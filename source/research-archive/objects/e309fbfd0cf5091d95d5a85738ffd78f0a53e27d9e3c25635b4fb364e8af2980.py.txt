"""Per-PASS ALUWR for the F05 family, grouped by shader width.

Read-only: evidence/STALL-TARGETS.json (the same file stall_by_family.py aggregates).
"""
import sys, re
from collections import defaultdict
CAMP = r"@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/fullnet-continuous-20261027"
sys.path.insert(0, CAMP)
import COMMON as K

rows = K.read_json(K.C / "evidence" / "STALL-TARGETS.json")["rows"]
name2fam = {}
for r in K.rows():
    name2fam.setdefault(r["name"], r["family"])
clean, art = [], []
for r in rows:
    pct = [r["aluwr"], r["sendwr"], r["barrier"]]
    (art if sum(1 for v in pct if abs(v - r["gpu"]) < 0.002) >= 2 else clean).append(r)
f05 = [r for r in clean if name2fam.get(r["name"]) == "F05"]


def width_of(name):
    m = re.match(r"block(\d+)_swin_body_3", name)
    n = int(m.group(1))
    if n <= 8 or n >= 62:
        return 64
    if n <= 14 or (56 <= n <= 61):
        return 128
    return 256


g = defaultdict(lambda: dict(n=0, aluwr=0.0, gpu=0.0, sendwr=0.0, barrier=0.0, names=[]))
for r in f05:
    e = g[width_of(r["name"])]
    e["n"] += 1
    for k in ("aluwr", "gpu", "sendwr", "barrier"):
        e[k] += r[k]
    e["names"].append(r["name"])

print("%-5s %3s %9s %9s %10s %8s %8s" % ("grp", "n", "gpu", "aluwr", "aluwr/PASS", "share%", "sendwr"))
tot = dict(n=0, gpu=0.0, aluwr=0.0)
for w in (64, 128, 256):
    e = g[w]
    tot["n"] += e["n"]; tot["gpu"] += e["gpu"]; tot["aluwr"] += e["aluwr"]
    print("C%-4d %3d %9.6f %9.6f %10.6f %7.2f%% %8.6f" %
          (w, e["n"], e["gpu"], e["aluwr"], e["aluwr"] / e["n"],
           100 * e["aluwr"] / e["gpu"], e["sendwr"]))
print("%-5s %3d %9.6f %9.6f %10.6f %7.2f%%" %
      ("ALL", tot["n"], tot["gpu"], tot["aluwr"], tot["aluwr"] / tot["n"],
       100 * tot["aluwr"] / tot["gpu"]))
print("\nC64 names :", " ".join(sorted(g[64]["names"], key=lambda s: int(re.findall(r"\d+", s)[0]))))
print("C128 names:", " ".join(sorted(g[128]["names"], key=lambda s: int(re.findall(r"\d+", s)[0]))))
print("C256 names:", " ".join(sorted(g[256]["names"], key=lambda s: int(re.findall(r"\d+", s)[0]))))
