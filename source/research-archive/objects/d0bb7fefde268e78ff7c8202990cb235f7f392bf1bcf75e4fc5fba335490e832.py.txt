"""Minimal plan reader for the R09/R10/R11 card (CPU only, no GPU).

Parses PORT_NETWORK_1 header + BUFFER lines and returns a dict of
resource index -> (name, bytes, path) using the index convention that the
toolchain in this workspace uses: BUFFER lines are numbered in file order
starting at 0 for the first BUFFER line (verified against the resource ids
and declared sizes that the PASS/BIND lines reference).
"""
import sys, json


def parse(path):
    lines = open(path, encoding='utf-8').read().splitlines()
    head = None
    bufs = []
    passes = []
    cur = None
    for ln in lines:
        if not ln.strip():
            continue
        w = ln.split()
        k = w[0]
        if k == 'PORT_NETWORK_1':
            head = w
        elif k == 'BUFFER':
            bufs.append({'idx': len(bufs), 'name': w[1].strip('"'),
                         'bytes': int(w[2]), 'path': w[3]})
        elif k == 'PASS':
            cur = {'order': int(w[1]), 'name': w[2].strip('"'),
                   'spv': w[3].strip('"'), 'wg': int(w[4]), 'shared': int(w[5]),
                   'nbind': int(w[6]), 'unknown': int(w[7]),
                   'grid': [int(w[8]), int(w[9]), int(w[10])],
                   'binds': [], 'push': None}
            passes.append(cur)
        elif k == 'BIND':
            cur['binds'].append({'binding': int(w[1]), 'resource': int(w[2]),
                                 'offset': int(w[3]), 'bytes': int(w[4])})
        elif k == 'PUSH':
            cur['push'] = [int(x) for x in w[1:]]
    return {'head': head, 'buffers': bufs, 'passes': passes}


def find(path, name):
    p = parse(path)
    buf_by_idx = {b['idx']: b for b in p['buffers']}
    out = []
    for ps in p['passes']:
        if ps['name'] == name:
            rec = dict(ps)
            rec['resources'] = [dict(buf_by_idx[b['resource']], binding=b['binding'],
                                     offset=b['offset'], bind_bytes=b['bytes'])
                                for b in ps['binds']]
            out.append(rec)
    return out


if __name__ == '__main__':
    p = parse(sys.argv[1])
    print(json.dumps({'head': p['head'], 'nbuf': len(p['buffers']),
                      'npass': len(p['passes'])}, indent=1))
    if len(sys.argv) > 2:
        for r in find(sys.argv[1], sys.argv[2]):
            print(json.dumps(r, indent=1, ensure_ascii=False))
