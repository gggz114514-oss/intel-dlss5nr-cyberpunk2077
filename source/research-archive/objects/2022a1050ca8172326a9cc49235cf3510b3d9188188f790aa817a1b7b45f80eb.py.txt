"""S03-tail: scan the tail-layer SPVs (F2, F20/block70, block39 projection+blend).

Reads nothing but the SPV binaries listed below; writes only into S03-tail/evidence.
Reuses the ledger's spvscan.py verbatim (copied into this directory so that no
__pycache__ is created outside the S03-tail write zone).
"""
from __future__ import annotations

import json
import sys
from pathlib import Path

HERE = Path(__file__).resolve().parent
sys.path.insert(0, str(HERE))
import spvscan  # noqa: E402

B = Path("@NR_WORKSPACE@")

TARGETS = {
    # INT8 side
    "int8_f20_postblend": B / "candidates/amd-map-20261015/round4-integrated/candidate/spv/io-f22-postblend.spv",
    "int8_f20_body_b70": B / "work/dsh-int8-fullnet-impl-20261024/candidates/c32-f1-f2-b70/spv/b70_i8_full.spv",
    "int8_f20_surfaces": B / "candidates/amd-map-20261015/round4-integrated/candidate/spv/io-f27-surfaces.spv",
    "int8_f2_block66": B / "work/codex-astra-fp16-int8-audit-20261024/int8-v15-offline-20261028/corrected-20261029/assets/f2.spv",
    "int8_b39_project_blend": B / "work/dsh-int8-fullnet-impl-20261024/candidates/r16-r18-r19-proj/spv/r16_b39_project_blend_i8mma.spv",
    # FP16 incumbent side
    "fp16_f20_s37_fold_body_r1a": B / "work/codex-astra-fp16-int8-audit-20261024/f20-coarse-owner-20261026/candidate.spv",
    "fp16_f20_tail_four_output": B / "work/codex-astra-fp16-int8-audit-20261024/c32-boundary-candidates-20261026/f20-four-output/tail.spv",
    "fp16_f2_block66": B / "work/codex-astra-fp16-int8-audit-20261024/c32-rest-20261026/root-f2/candidate.spv",
    # extra: the "swap+blend" projection kernel that shares the f37 slot name
    "fp16_projection_swap_f37": B / "candidates/amd-map-20261015/round4-integrated/candidate/spv/projection-swap-f37.spv",
}

out = {}
for key, path in TARGETS.items():
    try:
        out[key] = spvscan.scan(path)
    except Exception as exc:  # noqa: BLE001
        out[key] = {"error": f"{type(exc).__name__}: {exc}", "path": str(path)}
    out[key]["_key"] = key

dest = HERE.parent / "evidence" / "spv-tail-scan.json"
dest.parent.mkdir(parents=True, exist_ok=True)
dest.write_text(json.dumps(out, indent=1, ensure_ascii=False), encoding="utf-8")
print(f"wrote {dest} ({dest.stat().st_size} B) entries={len(out)}")
