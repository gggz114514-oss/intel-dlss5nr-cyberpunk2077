"""Map the audited kernels to families/ordinals and rank them for parallel cards."""
import json, sys
from collections import defaultdict
from pathlib import Path
sys.path.insert(0, str(Path(__file__).parent))
import COMMON as K

aud = K.read_json(K.C / 'evidence' / 'LATENCY-EXPOSURE.json')
idx = K.read_json(K.C / 'evidence' / 'NATIVE-INDEX.json')
by_ord = {}
def walk(n):
    if isinstance(n, dict):
        if 'ordinal' in n and 'baseline_kernel_hash' in n:
            by_ord.setdefault(str(n['baseline_kernel_hash'])[:16], []).append(int(n['ordinal']))
        for v in n.values():
            walk(v)
    elif isinstance(n, list):
        for v in n:
            walk(v)
walk(idx)
rows = {r['ordinal']: r for r in K.rows()}

out = []
for k in aud['kernels']:
    h = k['kernel'].split('_')[1] if '_' in k['kernel'] else ''
    ords = by_ord.get(h, [])
    fams = sorted({rows[o]['family'] for o in ords if o in rows})
    gate = [o for o in ords if rows[o]['family'] in ('F01', 'F02', 'F19', 'F07')]
    out.append(dict(kernel=k['kernel'], hash=h, ordinals=ords, families=fams,
                    total=k['total'], exposed=k['exposed'], partial=k['partial'],
                    hidden=k['hidden'], ops=k['ops']))
out.sort(key=lambda z: -z['exposed'])
print('%-22s %-6s %-22s %4s %4s %4s  %s' % ('hash', 'fam', 'ordinals', 'tot', '<=2', '3-10', 'ops'))
for z in out:
    if z['exposed'] == 0:
        continue
    print('%-22s %-6s %-22s %4d %4d %4d  %s' % (z['hash'], ','.join(z['families'])[:6],
                                                ','.join(map(str, z['ordinals']))[:22],
                                                z['total'], z['exposed'], z['partial'], z['ops']))
K.write_json(K.C / 'evidence' / 'LATENCY-TARGETS.json', out)
