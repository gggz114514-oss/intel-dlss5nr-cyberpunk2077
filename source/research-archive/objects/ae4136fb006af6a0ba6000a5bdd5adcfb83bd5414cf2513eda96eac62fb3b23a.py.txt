"""Adopt an ablation plan as the new chain head (removing members that degrade the chain).

The ablation was measured same-batch against the previous head; `improvement_ms` is
mean(previous head) - mean(ablation) with its CI. Only a FASTER_95 result is accepted.
"""
import json, shutil, sys
from pathlib import Path
sys.path.insert(0, str(Path(__file__).parent))
import COMMON as K
import ANALYZE

name = sys.argv[1]
note = sys.argv[2] if len(sys.argv) > 2 else ''
q = K.C / 'queue' / name
abl = K.read_json(q / 'ABLATION.json')
tim = K.read_json(q / 'TIMING.json')
c = tim['comparisons'][0]
# run_probe convention: saving = mean(base=head) - mean(probe=ablation)
improvement = -c['saving_ms']
ci = [-c['CI95_ms'][1], -c['CI95_ms'][0]]
if improvement <= 0 or ci[0] <= 0:
    print('REFUSED: the ablation is not faster (improvement %.6f, CI %s)' % (improvement, ci))
    sys.exit(1)

plan = Path(abl['plan']['path'])
prev = K.read_json(K.C / 'CURRENT.json')
acc = K.C / 'accepted' / ('%s-20261027' % name)
acc.mkdir(parents=True, exist_ok=True)
shutil.copyfile(plan, acc / 'plan.txt')
assert K.sha(acc / 'plan.txt') == K.sha(plan)
entry = dict(path=str(acc / 'plan.txt'), sha256=K.sha(acc / 'plan.txt'),
             label=name, saving_ms=improvement, CI95_ms=ci,
             kind='ablation (removes members that degrade the composed chain)',
             reverted=abl['reverted'], note=note)
cur = prev
cur['chain'].append(entry)
cur['increments_ms'].append(improvement)
cur['best_plan'] = dict(path=entry['path'], sha256=entry['sha256'])
cur['cumulative_ms_in_chain'] = sum(cur['increments_ms'])
K.write_json(K.C / 'CURRENT.json', cur)
K.write_json(acc / 'ADOPT.json', entry)
print(json.dumps(dict(adopted=name, improvement_ms=improvement, CI95_ms=ci,
                      reverted=abl['reverted'], chain_len=len(cur['chain']),
                      sum=cur['cumulative_ms_in_chain']), indent=2))
