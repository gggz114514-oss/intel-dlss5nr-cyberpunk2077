"""Second view: QKV weight-load / DPAS skeleton of the C64 candidate ISA."""
import json
import re
import sys
from pathlib import Path

sys.path.insert(0, str(Path(__file__).parent))
from isa_scan import parse, ASM, send_rows

ins = parse(ASM)


def find(kind_sub, pred=None):
    return [i for i in ins if kind_sub in i['tail']]


wl = find('load.ugm.d32x8.a32.bss[a0.2]')
print('weight loads:', len(wl))
for i in wl:
    tok = re.findall(r'\$(\d+)', i['tail'])
    print('%s  %-52s  tok=%s' % (hex(i['off']), i['head'][:52], tok))

dp = [i for i in ins if i['mnemonic'].startswith('dpas')]
print('\ndpas:', len(dp))
for i in dp:
    print('%s  %s  %s' % (hex(i['off']), i['head'][:96], i['tail'][:40]))
