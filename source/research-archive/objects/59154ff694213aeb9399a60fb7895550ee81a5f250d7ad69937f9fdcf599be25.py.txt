"""Emit SOURCE-DIFF.patch = parent source -> candidate source (CPU only)."""
import difflib
import hashlib
from pathlib import Path

CARD = Path(__file__).resolve().parent.parent
BUILD = CARD / '_build'
P = CARD / 'parent-rebuild.comp'
C = CARD / 'candidate.comp'
REL_P = ('fullnet-mechanism-transfer-20261026/f05-kv-consumer-layout/family/c128/'
         'candidate.comp')
REL_C = 'fullnet-continuous-20261027/cards/n05e-f05-atile/candidate.comp'


def main():
    a = P.read_text(encoding='utf-8').splitlines(keepends=True)
    b = C.read_text(encoding='utf-8').splitlines(keepends=True)
    diff = list(difflib.unified_diff(a, b, fromfile='a/' + REL_P, tofile='b/' + REL_C, n=3))
    out = CARD / 'SOURCE-DIFF.patch'
    out.write_text(''.join(diff), encoding='utf-8')
    added = sum(1 for l in diff if l.startswith('+') and not l.startswith('+++'))
    removed = sum(1 for l in diff if l.startswith('-') and not l.startswith('---'))
    hunks = sum(1 for l in diff if l.startswith('@@'))
    print('hunks %d  +%d  -%d  patch bytes %d' % (hunks, added, removed, out.stat().st_size))
    print('parent bytes %d  candidate bytes %d' % ((BUILD / 'parent-raw.spv').stat().st_size,
                                                   (BUILD / 'cand-raw.spv').stat().st_size))
    for name in ('parent-raw.spv', 'cand-raw.spv'):
        p = BUILD / name
        print(name, p.stat().st_size, hashlib.sha256(p.read_bytes()).hexdigest())


if __name__ == '__main__':
    main()
