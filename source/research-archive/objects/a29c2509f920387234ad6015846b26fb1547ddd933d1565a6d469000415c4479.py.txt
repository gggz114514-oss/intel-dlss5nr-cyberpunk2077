"""N02B F20 baseline native read (CPU only): what the current identity looks like.

Reads the existing baseline capture of ord 230 (pipeline 38, kernel 76902ef344609c8a)
from the F10 handoff capture set -- read-only, no GPU -- and records the exact
observables this card predicts will change.  Writes build/native-base.json.
"""
import collections
import hashlib
import json
import pathlib
import sys

CARD = pathlib.Path(__file__).resolve().parent
CAMP = CARD.parent.parent
sys.path.insert(0, str(CAMP))
import ISA                                       # noqa: E402

W = pathlib.Path(r'@NR_WORKSPACE@')
CAP = (W / 'work/codex-astra-fp16-int8-audit-20261024/fullnet-mechanism-transfer-20261026'
       / 'f10-weight-return-separation-20261027/captures/native-f20-pixel-owner/ISA')
ISA_F20 = CAP / 'CS_asm76902ef344609c8a_simd32_0000000000000027_38_orig.asm'
ISA_F21 = CAP / 'CS_asm575e62536177b76f_simd32_0000000000000028_39_orig.asm'
F20_NATIVE_REVIEW = (W / 'work/codex-astra-fp16-int8-audit-20261024/f20-coarse-owner-20261026'
                     / 'NATIVE-REVIEW.json')


def sha(p):
    return hashlib.sha256(pathlib.Path(p).read_bytes()).hexdigest()


def main():
    isa = ISA.parse(ISA_F20)
    sends = [i for i in isa if i['op'].startswith('send')]
    msg = collections.Counter(i['comment'] for i in sends)
    ops = collections.Counter(i['op'] for i in isa)
    dpas = [i for i in isa if i['op'].startswith('dpas')]
    out = {
        'card': 'n02b-f20-store', 'ordinal': 230, 'pipeline': 38,
        'kernel_hash': '76902ef344609c8a',
        'isa': {'path': str(ISA_F20), 'sha256': sha(ISA_F20)},
        'baseline_capture_plan': 'a024c3c5aa1d1432ab3f0591bf03a897880d86be2c99d6de1e1492a2a7b2befe',
        'sibling_isa_f21': {'path': str(ISA_F21), 'sha256': sha(ISA_F21)},
        'instructions': len(isa),
        'sends_total': len(sends),
        'sends_by_op': {k: v for k, v in sorted(collections.Counter(
            i['op'] for i in sends).items())},
        'dpas': len(dpas),
        'dpas_first_addr': hex(dpas[0]['addr']) if dpas else None,
        'dpas_last_addr': hex(dpas[-1]['addr']) if dpas else None,
        'sync': {k: v for k, v in sorted(ops.items()) if k.startswith('sync')},
        'barrier_like': 0,
        'top_ops': {k: v for k, v in ops.most_common(12)},
        'messages': dict(sorted(((m, n) for m, n in msg.items()),
                                key=lambda kv: (-kv[1], str(kv[0])))),
        'stores': {m: n for m, n in msg.items() if m and 'store.ugm' in m},
        'wide16B_store_count': sum(n for m, n in msg.items()
                                   if m and ('store.ugm.d32x4' in m or 'store.ugm.d32x64t' in m)),
    }

    # the four scalar store instructions of the residual_out record
    hits = [(k, i) for k, i in enumerate(isa)
            if i['comment'] and 'store.ugm.d32.a32.bss' in i['comment']]
    out['record_store_site'] = {
        'instruction_count': len(hits),
        'entries': [{'index': k, 'addr': hex(i['addr']), 'raw': i['raw'].strip()}
                    for k, i in hits],
    }
    if hits:
        k0 = hits[0][0]
        out['record_store_site']['prefix_window'] = [
            {'index': k, 'addr': hex(isa[k]['addr']), 'raw': isa[k]['raw'].strip()}
            for k in range(max(0, k0 - 8), min(len(isa), hits[-1][0] + 3))]

    # scratch / workgroup memory / basic blocks as reported by the F20 card review
    rev = json.loads(F20_NATIVE_REVIEW.read_text(encoding='utf-8-sig'))
    out['kernel_attributes_baseline'] = rev.get('candidate_F20')
    out['kernel_attributes_parent_of_baseline'] = rev.get('parent_F20')

    # isolation: which pipeline is which, and that F21 is a different shader
    out['isolation'] = {
        'pipe38': 'F20 ord 230 s37_fold_body_r1a_v1 (this card)',
        'pipe39': 'F21 ord 231 ioport_block70_2 (n02 card, separate PASS)',
        'note': 'the two pipelines are different SPIR-V modules with different kernel hashes',
    }

    (CARD / 'build').mkdir(exist_ok=True)
    (CARD / 'build/native-base.json').write_text(json.dumps(out, indent=1), encoding='utf-8')
    print(json.dumps({k: out[k] for k in ('instructions', 'sends_total', 'sends_by_op',
                                          'dpas', 'sync', 'stores', 'wide16B_store_count',
                                          'kernel_attributes_baseline')}, indent=1)[:3000])


if __name__ == '__main__':
    main()
