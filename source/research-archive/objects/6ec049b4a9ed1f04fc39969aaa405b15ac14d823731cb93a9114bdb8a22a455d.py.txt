"""N17 F20 body (ord 230) build: intra-subgroup Q fragment round trip removed.

Mechanism (one class): the two H(Q) fragments produced by stage 4 (QKV) are
published to shArena and immediately re-read by stage 6 (score) *inside the same
subgroup* (rows 8*sg on both sides).  The module already keeps an identical
MAT_H -> MAT_A handoff in registers for the stage8 -> stage9 context transport
(S37-CONTEXT-A), so the Q round trip is pure transport: it costs 2 shared
fragment stores + 2 shared fragment loads.  The candidate keeps Q in registers,
makes the Q-plane publication debug-only, and assembles score_q0/score_q1 from
the retained half components (identical lane/component map, so the half values
are bit-identical: no new rounding point).

Steps (CPU only, no GPU, writes only inside this card directory):
  1. parent byte reproduction : baseline source (f20-coarse-owner candidate.comp)
     + recorded recipe (glslang 16.5.0 + NoContraction pattern transfer from the
     f20-pixel-owner decoration parent) -> must be byte identical to 215aca67...
  2. candidate source          : the same parent source with the minimal edit.
  3. candidate compile + the same NoContraction transfer -> candidate.spv; validate.
  4. plan.txt on the CURRENT chain head (n09c-f09-b-share), replacing only the
     ord-230 PASS line; assert replaced == 1.
  5. cpu-check of both plans.
  6. read-only ISA accounting for the parent (build/isa.json via _isa.py).

Run:  @NR_USER_HOME@/AppData/Local/Programs/Python/Python312/python.exe build.py
"""

# ------------------------------------------------------------------ the edit
# Design rule: the patch adds *no* new source-level locals (no compiler-visible
# lifetime change beyond the Q fragment itself), so the FP expression multiset and
# the NoContraction site signatures stay bit-identical.  The stage-6 A fragments
# are declared early and filled by stage 4 -- the same "keep the accumulator half
# in an A fragment" idiom the module already uses for the stage8 -> stage9 context
# handoff (lines 553-558) and for the stage-8 P handoff (lines 450/502).
OLD_DECL = (
    b'    for (uint step = 0u; step < 3u; ++step) {\r\n')
NEW_DECL = (
    b'    // N17-F20: Q (region 0) is produced in step 2 of this loop and consumed by\r\n'
    b'    // stage 6 in the SAME subgroup (both own rows 8*sg of the Q plane), so the\r\n'
    b'    // shArena publication + reload is transport only.  score_q0/score_q1 are\r\n'
    b'    // declared here and filled below from the H(Q) components; MAT_H -> MAT_A is\r\n'
    b'    // component-wise on this target (see the stage8 -> stage9 context handoff).\r\n'
    b'    MAT_A score_q0, score_q1;\r\n'
    b'    for (uint step = 0u; step < 3u; ++step) {\r\n')

OLD_PUB = (
    b'        coopMatStore(half0, shArena, region*2048u+m0*32u,32u,LROW);\r\n'
    b'        coopMatStore(half1, shArena, region*2048u+m0*32u+16u,32u,LROW);\r\n')
NEW_PUB = (
    b'        // N17-F20: the Q pair never leaves the subgroup; it goes straight into the\r\n'
    b'        // stage-6 A fragments.  The shared Q plane is debug-only from here on (the\r\n'
    b'        // debug dump below still reads NR_Q unchanged when debug != 0).\r\n'
    b'        if(region==0u) {\r\n'
    b'            score_q0[0]=half0[0];score_q0[1]=half0[1];score_q0[2]=half0[2];score_q0[3]=half0[3];\r\n'
    b'            score_q1[0]=half1[0];score_q1[1]=half1[1];score_q1[2]=half1[2];score_q1[3]=half1[3];\r\n'
    b'        }\r\n'
    b'        if(debug || region!=0u) {\r\n'
    b'            coopMatStore(half0, shArena, region*2048u+m0*32u,32u,LROW);\r\n'
    b'            coopMatStore(half1, shArena, region*2048u+m0*32u+16u,32u,LROW);\r\n'
    b'        }\r\n')

OLD_LOAD = (
    b'    MAT_A score_q0,score_q1;\r\n'
    b'    coopMatLoad(score_q0,shArena,sg*8u*32u,32u,LROW);\r\n'
    b'    coopMatLoad(score_q1,shArena,sg*8u*32u+16u,32u,LROW);\r\n')
NEW_LOAD = (
    b'    // N17-F20: score_q0/score_q1 were filled in registers by stage 4 (this SG owns\r\n'
    b'    // rows 8*sg of Q on both sides), so there is no shared fragment load here.\r\n')


def sha(p):
    return hashlib.sha256(pathlib.Path(p).read_bytes()).hexdigest()


def run(argv):
    r = subprocess.run([str(x) for x in argv], capture_output=True, text=True,
                       encoding='utf-8', errors='replace')
    return r.returncode, (r.stdout or '').strip(), (r.stderr or '').strip()


def compile_glsl(src, out, incs):
    argv = [GLSLANG, '-V', '--target-env', 'vulkan1.3']
    for d in incs:
        argv.append('-I' + str(d))
    argv += [str(src), '-o', str(out)]
    return argv, run(argv)


def build_spv(src, raw, final, incs):
    """compile + NoContraction pattern transfer from the recorded decoration parent."""
    argv, (rc, so, se) = compile_glsl(src, raw, incs)
    assert rc == 0, 'glslang failed\n%s\n%s' % (so, se)
    pm = ST.Module(DECOR_PARENT)
    cm = ST.Module(raw)
    keep, info = ST.transfer_decoration_pattern_aligned(pm, cm, set())
    ST.set_no_contraction(cm, keep, final)
    return dict(argv=[str(x) for x in argv], glslang_rc=rc, glslang_stderr=se[-400:],
                raw_sha256=sha(raw), raw_bytes=pathlib.Path(raw).stat().st_size,
                nc_count=len(keep), nc_parent=len(ST.no_contraction_ids(pm)),
                nc_info={k: info[k] for k in ('alignment', 'functions_mapped',
                                              'child_functions_without_parent_counterpart',
                                              'parent_decorations_total')},
                spv_sha256=sha(final), spv_bytes=pathlib.Path(final).stat().st_size)


def unified_diff(old_bytes, new_bytes, a_name, b_name):
    """LF-terminated unified diff of the two sources (CRLF source lines are stripped)."""
    import difflib
    a = old_bytes.decode('utf-8-sig').replace('\r\n', '\n').split('\n')
    b = new_bytes.decode('utf-8-sig').replace('\r\n', '\n').split('\n')
    lines = [l.rstrip('\n') for l in difflib.unified_diff(a, b, fromfile=a_name,
                                                          tofile=b_name, n=4)]
    return '\n'.join(lines) + '\n'


def main():
    out = {'card': 'n17-f20-body', 'family': 'F20', 'ordinal': 230,
           'gpu_used': False, 'deployed': False}
    (CARD / 'build').mkdir(exist_ok=True)

    # ---------------- 0. identity of the inputs ----------------
    src_bytes = PARENT_SRC.read_bytes()
    assert sha(PARENT_SRC) == PARENT_SRC_SHA, 'parent source sha changed'
    assert sha(PARENT_SPV) == PARENT_SPV_SHA, 'baseline parent SPV sha changed'
    out['parent'] = {
        'source': K.ident(PARENT_SRC), 'source_sha_expected': PARENT_SRC_SHA,
        'spv': K.ident(PARENT_SPV), 'spv_sha_expected': PARENT_SPV_SHA,
        'decoration_parent': K.ident(DECOR_PARENT),
        'recipe': {'include_dirs': [str(PINNED_INC), str(RECIPE_INC)],
                   'pinned_include_dir': str(PINNED_INC),
                   'nc': 'transfer_decoration_pattern_aligned(decoration_parent -> child), keep_ids=empty',
                   'source': 'evidence/RECIPES.json entry 215aca67... (nc_recipe mode=parent)'},
    }

    # ---------------- 1. parent byte reproduction ----------------
    incs = [PINNED_INC] if PINNED_INC.exists() else [RECIPE_INC]
    incs = incs + [RECIPE_INC]
    out['parent_rebuild'] = build_spv(PARENT_SRC, CARD / 'build/parent-raw.spv',
                                      CARD / 'parent-rebuild.spv', incs)
    out['parent_rebuild']['byte_match'] = (out['parent_rebuild']['spv_sha256'] == PARENT_SPV_SHA)
    assert out['parent_rebuild']['byte_match'], 'parent did not reproduce byte for byte'

    # ---------------- 2. candidate source (minimal patch) ----------------
    assert src_bytes.count(OLD_DECL) == 1, 'declaration anchor not unique'
    assert src_bytes.count(OLD_PUB) == 1, 'publication anchor not unique'
    assert src_bytes.count(OLD_LOAD) == 1, 'load anchor not unique'
    cand_bytes = (src_bytes.replace(OLD_DECL, NEW_DECL)
                  .replace(OLD_PUB, NEW_PUB).replace(OLD_LOAD, NEW_LOAD))
    (CARD / 'parent.comp').write_bytes(src_bytes)
    (CARD / 'candidate.comp').write_bytes(cand_bytes)
    patch = unified_diff(src_bytes, cand_bytes,
                         'a/f20-coarse-owner-20261026/candidate.comp',
                         f'b/cards/n17-f20-body/candidate.comp')
    (CARD / 'SOURCE-DIFF.patch').write_text(patch, encoding='utf-8')
    out['candidate_source'] = {
        'path': str(CARD / 'candidate.comp'), 'sha256': sha(CARD / 'candidate.comp'),
        'bytes': len(cand_bytes),
        'hunks': 3,
        'diff_lines': {'removed': sum(1 for l in patch.splitlines()
                                      if l.startswith('-') and not l.startswith('---')),
                       'added': sum(1 for l in patch.splitlines()
                                    if l.startswith('+') and not l.startswith('+++'))},
        'only_site': ('ord 230, stage 4 (QKV) Q-plane publication + stage 6 (score) Q '
                      'fragment load, src lines 341 / 386-387 / 402-404'),
    }

    # ---------------- 3. candidate compile + validate ----------------
    out['candidate'] = build_spv(CARD / 'candidate.comp', CARD / 'build/candidate-raw.spv',
                                 CARD / 'candidate.spv', incs)
    out['candidate']['spv'] = str(CARD / 'candidate.spv')
    out['candidate']['nc_equal_to_parent'] = out['candidate']['nc_count'] == out['parent_rebuild']['nc_count']

    rc, so, se = run([VALIDATE, CARD / 'candidate.spv'])
    out['spv_validate_cpu'] = {'candidate': {'rc': rc, 'stdout': so, 'stderr': se}}
    rc2, so2, se2 = run([VALIDATE, CARD / 'parent-rebuild.spv'])
    out['spv_validate_cpu']['parent_rebuild'] = {'rc': rc2, 'stdout': so2, 'stderr': se2}
    assert rc == 0, 'candidate failed spv_validate_cpu: %s %s' % (so, se)

    # ---------------- 4. plan on the current chain head ----------------
    base_plan, base_sha, chain = K.current_best()
    base_text = pathlib.Path(base_plan).read_text(encoding='utf-8-sig')
    n = K.make_plan(CARD / 'candidate.spv', CARD / 'plan.txt', parent_plan=base_plan,
                    spv_shas={PARENT_SPV_SHA})
    assert n == 1, 'replaced %d passes, expected 1' % n
    base_lines = base_text.splitlines()
    cand_lines = (CARD / 'plan.txt').read_text(encoding='utf-8-sig').splitlines()
    assert len(base_lines) == len(cand_lines), 'plan line count changed'
    diff_lines = [i for i, (a, b) in enumerate(zip(base_lines, cand_lines)) if a != b]
    assert len(diff_lines) == 1, 'plan differs on %d lines: %s' % (len(diff_lines), diff_lines)
    changed = diff_lines[0]
    assert base_lines[changed] == K.pass_lines_of(base_plan)[230]['line'], 'changed line is not ord 230'
    out['plan'] = {
        'path': str(CARD / 'plan.txt'), 'sha256': sha(CARD / 'plan.txt'),
        'parent_plan': {'path': str(base_plan), 'sha256': base_sha},
        'chain_head_label': chain[-1]['label'] if chain else None,
        'replaced_passes': n, 'expected_passes': 1,
        'diff_lines': diff_lines, 'line_count': len(cand_lines),
        'changed_line_ordinal': 230,
        'old_line': base_lines[changed], 'new_line': cand_lines[changed],
        'pass_label_151_occurrences': sum(1 for l in cand_lines
                                          if l.startswith('PASS 151 ')),
    }
    assert out['plan']['pass_label_151_occurrences'] == 2, 'PASS 151 must appear twice (ord 230+231)'

    # ---------------- 5. CPU plan check ----------------
    rc, so, se = run([RUNNER, '--plan', CARD / 'plan.txt', '--cpu-check'])
    out['cpu_check'] = {'argv': [str(RUNNER), '--plan', str(CARD / 'plan.txt'), '--cpu-check'],
                       'rc': rc, 'stdout': so, 'stderr': se}
    rc2, so2, se2 = run([RUNNER, '--plan', base_plan, '--cpu-check'])
    out['cpu_check']['parent_stdout'] = so2
    out['cpu_check']['identical_to_parent'] = (so == so2)
    assert rc == 0 and out['cpu_check']['identical_to_parent'], 'cpu-check differs from chain head'

    # NoContraction accounting for the contract (float FMA site parity).
    out['float_NC_checked'] = out['parent_rebuild']['nc_count']
    out['float_NC_note'] = ('the same 45 NoContraction decorations are transferred from the recorded '
                            'decoration parent onto the same FP instructions in the candidate; the '
                            'transfer verifies each mapped pair by ID-free operand signature, and the '
                            'float/GLSL.std.450 expression multiset is identical (see _spv_check.py)')

    # ---------------- 6. ord 230 plan row: BIND / PUSH audit ----------------
    recs = K.pass_lines_of(CARD / 'plan.txt')
    lines = (CARD / 'plan.txt').read_text(encoding='utf-8-sig').splitlines()
    i0 = lines.index(recs[230]['line'])
    binds, push = [], None
    for l in lines[i0 + 1:]:
        if l.startswith('PASS '):
            break
        if l.startswith('BIND '):
            binds.append([int(x) for x in l.split()[1:]])
        elif l.startswith('PUSH '):
            push = [int(x) for x in l.split()[1:]]
    out['ord230_bind'] = [{'set': b[0], 'buffer_id': b[1], 'offset': b[2], 'size_bytes': b[3]}
                          for b in binds]
    out['ord230_push_dwords'] = push
    out['ord230_push_pc'] = {('p%d' % (k // 4)): push[k:k + 4] for k in range(0, len(push), 4)}
    out['ord230_push_dwords_as_i32'] = [x - (1 << 32) if x >= (1 << 31) else x for x in push]
    out['ord230_pass_header'] = recs[230]['line']

    K.write_json(CARD / 'BUILD.json', out)
    print(json.dumps({k: out[k] for k in ('parent_rebuild', 'candidate', 'spv_validate_cpu',
                                          'plan', 'cpu_check', 'float_NC_checked')}, indent=1)[:6000])


if __name__ == '__main__':
    main()
