$ErrorActionPreference='Stop'
$game='@NR_GAME@/bin/x64'
$package='@NR_WORKSPACE@/work/github-experimental-game-install-20261024/package'
try {
 & (Join-Path $PSScriptRoot 'Launch-Game.ps1')
} finally {
 if (!(Get-Process Cyberpunk2077 -ErrorAction SilentlyContinue)) {
  Copy-Item -LiteralPath "$game/plugins/CyberpunkNRBridge.log" -Destination "$PSScriptRoot/after-CyberpunkNRBridge.log"
  Copy-Item -LiteralPath "$package/config/component_status.json" -Destination "$PSScriptRoot/after-component_status.json"
  Copy-Item -LiteralPath "$package/config/panel_state.json" -Destination "$PSScriptRoot/after-panel_state.json"
  Copy-Item -LiteralPath "$PSScriptRoot/before/game/nrvk_bridge.dll" -Destination "$game/nrvk_bridge.dll" -Force
  Copy-Item -LiteralPath "$PSScriptRoot/before/game/plugins/nr_bridge.ini" -Destination "$game/plugins/nr_bridge.ini" -Force
  Copy-Item -LiteralPath "$PSScriptRoot/before/config/nr_panel.ini" -Destination "$package/config/nr_panel.ini" -Force
  'Old Small boot pair restored; optimized Full preserved.' | Set-Content "$PSScriptRoot/RESTORED.txt"
 } else { throw 'Game still alive: do not restore loaded files' }
}
