import json, copy
from pathlib import Path
D=Path(__file__).resolve().parent
old=json.loads((D/'LUNA-PILOT-JOBS.json').read_text(encoding='utf-8-sig'))
jobs=[]
for source,label in [('baseline','recovery-baseline'),('f02-whole','f02-maxbuckets'),('f20-whole','f20-maxbuckets')]:
 j=copy.deepcopy(next(j for j in old['jobs'] if j['id']==source))
 def rename(v):
  if isinstance(v,str): return v.replace('/'+source,'/'+label).replace('\\'+source,'\\'+label) if v!=source else label
  if isinstance(v,list): return [rename(x) for x in v]
  if isinstance(v,dict): return {k:rename(x) for k,x in v.items()}
  return v
 j=rename(j)
 if source!='baseline':
  pos=j['argv'].index('--');j['argv'][pos:pos]=['--num_thread_buckets','0','--allow_sregs','0','--dump_final_isa']
 jobs.append(j)
old['jobs']=jobs
old['status']='DIAGNOSIS_ONLY_COUNTER_CONTENTION_CONTROL'
old['rules']=['Run recovery-baseline first; require gold, clean exit, complete receipt and normal frame time before following jobs.','Any device lost, GPU timeout, frame mismatch or missing receipt stops all remaining jobs. No automatic continuation after such errors.','No thread_group_scope. Original shader/plan unchanged. num_thread_buckets=0 selects model maximum per local API.','Capture profile allocation size, frequency, final ISA and target PASS times. No optimization.']
(D/'LUNA-BUCKET-CONTROL-JOBS.json').write_text(json.dumps(old,indent=2),encoding='utf-8')
print([j['id'] for j in jobs])
