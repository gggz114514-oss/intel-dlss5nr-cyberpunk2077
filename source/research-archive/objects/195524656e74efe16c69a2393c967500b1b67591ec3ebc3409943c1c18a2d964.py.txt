"""n24-s5-gm-fold build driver (CPU only: glslang + spv_validate_cpu + clock_runner --cpu-check).

Steps
  1. parent-byte reproduction for ord 193 (producer) and ord 194 (consumer)
     using the pinned include dirs recorded in evidence/RECIPES.json
  2. compile candidate-193.comp / candidate-194.comp, validate both
  3. build plan.txt from the chain-head plan with exactly two SPV path
     replacements (ord 193 -> candidate-193.spv, ord 194 -> candidate-194.spv)
  4. cpu_check(plan) and write BUILD.json

No GPU, no deployment, writes only inside this card directory.
"""
import json
import subprocess
import sys
from pathlib import Path

CARD = Path(__file__).resolve().parent
C = CARD.parents[1]
sys.path.insert(0, str(C))
import COMMON as K

PINNED = C / 'cards/idt-identity-recipe/evidence/headers'
SRC193 = Path('@NR_WORKSPACE@/candidates/amd-netorg-t3b-20261018/'
              'c128-canonical-only-v1/source/c3x_body_c128_r09.comp')
SRC194 = Path('@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/'
              'fullnet-mechanism-transfer-20261026/contract-scale-fragments/family/c128/candidate.comp')
INC193 = [SRC193.parent, PINNED / '12569d767adf']
INC194 = [Path('@NR_WORKSPACE@/work/dsh-fp16-same-family-20261023/build/rebuild/K22'),
          PINNED / 'ffde3c3d2379', SRC194.parent]

PARENT = {193: dict(src=SRC193, spv_sha='12569d767adf9eb02398f87e39ccb24593f932cac2b0395702df769b897ed341',
                    incs=INC193, cand='candidate-193.comp', out='candidate-193.spv'),
          194: dict(src=SRC194, spv_sha='ffde3c3d2379b4651274ceafd14251e9f1916c22c885f245802020ffb491cc64',
                    incs=INC194, cand='candidate-194.comp', out='candidate-194.spv')}


def build_dir(ordinal):
    d = CARD / 'build' / str(ordinal)
    d.mkdir(parents=True, exist_ok=True)
    return d


def main():
    rep = {'card': 'n24-s5-gm-fold', 'pass_ordinals': [193, 194], 'steps': {}}
    plan_base, plan_base_sha, _ = K.current_best()
    rep['parent_plan'] = {'path': str(plan_base), 'sha256': plan_base_sha}

    # 1 -------------------------------------------------- parent reproduction
    for o, spec in PARENT.items():
        d = build_dir(o)
        raw = d / 'parent-raw.spv'
        K.compile_glsl(spec['src'], raw, spec['incs'])
        sha = K.sha(raw)
        ok = sha == spec['spv_sha']
        if ok:
            (CARD / ('parent-rebuild-%d.spv' % o)).write_bytes(raw.read_bytes())
        rep['steps']['parent_%d' % o] = {
            'source': str(spec['src']), 'source_sha256': K.sha(spec['src']),
            'include_dirs': [str(x) for x in spec['incs']],
            'raw_spv_sha256': sha, 'target_sha256': spec['spv_sha'],
            'raw_match': ok, 'recipe': 'raw' if ok else 'MISMATCH'}
        if not ok:
            print('PARENT MISMATCH', o, sha, spec['spv_sha'])
            return 1

    # 2 -------------------------------------------------- candidates
    for o, spec in PARENT.items():
        d = build_dir(o)
        cand = CARD / spec['cand']
        out = CARD / spec['out']
        K.compile_glsl(cand, out, spec['incs'])
        K.validate_spv(out)
        rep['steps']['candidate_%d' % o] = {
            'source': str(cand), 'source_sha256': K.sha(cand),
            'spv': str(out), 'spv_sha256': K.sha(out),
            'spv_bytes': out.stat().st_size,
            'spv_validated': True}

    # 3 -------------------------------------------------- plan (two replacements)
    base_text = plan_base.read_text(encoding='utf-8-sig')
    r193, r194 = K.row_by_ordinal(193), K.row_by_ordinal(194)
    p193 = Path(r193['spv']).as_posix()
    p194 = Path(r194['spv']).as_posix()
    c193 = (CARD / 'candidate-193.spv').as_posix()
    c194 = (CARD / 'candidate-194.spv').as_posix()
    n193 = base_text.count(p193)
    n194 = base_text.count(p194)
    text = base_text.replace(p193, c193).replace(p194, c194)
    plan = CARD / 'plan.txt'
    plan.write_text(text, encoding='utf-8')
    replaced = n193 + n194
    rep['steps']['plan'] = {
        'parent_plan': str(plan_base), 'parent_plan_sha256': plan_base_sha,
        'plan': str(plan), 'plan_sha256': K.sha(plan),
        'replacements': {'ord_193_path': p193, 'count_193': n193,
                         'ord_194_path': p194, 'count_194': n194},
        'replaced_total': replaced, 'replaced_expected': 2,
        'passes_before': len([l for l in base_text.splitlines() if l.startswith('PASS ')]),
        'passes_after': len([l for l in text.splitlines() if l.startswith('PASS ')]),
        'buffers': len([l for l in text.splitlines() if l.startswith('BUFFER ')]),
    }
    assert replaced == 2, 'expected exactly 2 PASS replacements, got %d' % replaced
    assert rep['steps']['plan']['passes_before'] == rep['steps']['plan']['passes_after'] == 232

    # 4 -------------------------------------------------- cpu check
    out = K.cpu_check(plan)
    rep['steps']['cpu_check'] = {'argv': [str(K.RUNNER), '--plan', str(plan), '--cpu-check'],
                                 'stdout': out}
    K.write_json(CARD / 'BUILD.json', rep)
    print(json.dumps(rep, indent=2)[:4000])
    return 0


if __name__ == '__main__':
    sys.exit(main())
