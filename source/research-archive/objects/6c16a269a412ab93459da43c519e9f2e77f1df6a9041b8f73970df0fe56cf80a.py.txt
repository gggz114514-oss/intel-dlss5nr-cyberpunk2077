# SPIR-V 1.6 cooperative matrix opcodes used by glslang 16.5:
for k, v in ((4456, 'OpTypeCooperativeMatrixKHR'), (4457, 'OpCooperativeMatrixLengthKHR'),
             (4458, 'OpCooperativeMatrixMulAddKHR'), (4459, 'OpCooperativeMatrixConvertKHR'),
             (4421, 'OpCooperativeMatrixLoadKHR'), (4422, 'OpCooperativeMatrixStoreKHR'),
             (4423, 'OpCooperativeMatrixLengthKHR'), (4424, 'OpCooperativeMatrixMulAddKHR')):
    NAMES.setdefault(k, v)


def name(op):
    return NAMES.get(op, 'Op%d' % op)


def skel(mod, region):
    return [(mod.instructions[i][0], mod.instructions[i][3]) for i in region]


def main():
    row = K.row_by_ordinal(7)
    pm = ST.Module(Path(row['spv']))
    cm = ST.Module(Path('@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/'
                        'fullnet-continuous-20261027/cards/n04a-f05-c64-feed/build/candidate-raw.spv'))
    pre = ST.function_regions(pm)
    cre = ST.function_regions(cm)
    print('parent functions:', len(pre), 'candidate functions:', len(cre))
    tot = {}
    for fi, (pr, cr) in enumerate(zip(pre, cre)):
        a = skel(pm, pr)
        b = skel(cm, cr)
        sm = difflib.SequenceMatcher(None, a, b, autojunk=False)
        hunks = []
        for tag, i1, i2, j1, j2 in sm.get_opcodes():
            if tag == 'equal':
                continue
            removed = [name(pm.instructions[pr[k]][0]) for k in range(i1, i2)]
            added = [name(cm.instructions[cr[k]][0]) for k in range(j1, j2)]
            hunks.append(dict(tag=tag, parent_pos=i1, candidate_pos=j1,
                              removed=removed, added=added))
            for x in removed:
                tot[x] = tot.get(x, 0) - 1
            for x in added:
                tot[x] = tot.get(x, 0) + 1
        if hunks:
            print('\n=== function %d : parent %d instrs, candidate %d instrs, %d hunks ==='
                  % (fi, len(a), len(b), len(hunks)))
            for h in hunks[:14]:
                print('  %-6s parent@%d cand@%d  -%s  +%s' % (
                    h['tag'], h['parent_pos'], h['candidate_pos'],
                    ','.join(h['removed']) or '-', ','.join(h['added']) or '-'))
    print('\ntotal opcode delta:')
    for k, v in sorted(tot.items(), key=lambda kv: -abs(kv[1])):
        if v:
            print('  %+4d  %s' % (v, k))


if __name__ == '__main__':
    main()
