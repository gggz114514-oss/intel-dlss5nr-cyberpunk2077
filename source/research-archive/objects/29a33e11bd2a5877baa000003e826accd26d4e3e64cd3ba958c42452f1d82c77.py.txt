#!/usr/bin/env python3
"""Probe 3: dump instruction windows in program order (with word offsets)."""
import sys

sys.path.insert(0, r"@NR_WORKSPACE@/work/dsh-xmx-handoff-validation-20261024/candidates/b3-c32-seed/tools")
from b3_spv import Module, OPNAME  # noqa: E402

P = r"@NR_WORKSPACE@/work/dsh-xmx-handoff-validation-20261024/build/b3-c32-seed/parent_repro.spv"
M = Module(P)


def show(lo, hi, label=""):
    print("---- insts[%d:%d] %s ----" % (lo, hi, label))
    for k in range(lo, min(hi, len(M.insts))):
        ins = M.insts[k]
        print("%5d w%-6d %-28s %s" % (k, ins.pos, ins.name, " ".join(str(x) for x in ins.ops)))


if __name__ == "__main__":
    what = sys.argv[1] if len(sys.argv) > 1 else "fconvert"
    if what == "fconvert":
        print("== OpFConvert sites ==")
        for ins in M.insts:
            if ins.op == 115:
                print("  inst#%-5d word %-6d ops=%s" % (ins.idx, ins.pos, list(ins.ops)))
    elif what == "fmul":
        print("== OpFMul whose result type is a cooperative matrix ==")
        for ins in M.insts:
            if ins.op == 133 and ins.rtype in (33, 35, 42, 2573):
                print("  inst#%-5d word %-6d type=%-5s ops=%s" % (ins.idx, ins.pos, ins.rtype, list(ins.ops)))
    elif what == "window":
        lo, hi = int(sys.argv[2]), int(sys.argv[3])
        show(lo, hi)
    elif what == "funcs":
        start = None
        for ins in M.insts:
            if ins.op == 54:
                print("inst#%-5d word %-6d OpFunction type=%d id=%d ctrl=%d" %
                      (ins.idx, ins.pos, ins.ops[0], ins.ops[1], ins.ops[2]))
    elif what == "calls":
        print("== OpFunctionCall ==")
        n = 0
        for ins in M.insts:
            if ins.op == 57:
                n += 1
                if n <= 60:
                    print("  inst#%-5d word %-6d type=%-5s id=%-5s fn=%-5s args=%s" %
                          (ins.idx, ins.pos, ins.ops[0], ins.ops[1], ins.ops[2], list(ins.ops[3:])))
        print("  total", n)
