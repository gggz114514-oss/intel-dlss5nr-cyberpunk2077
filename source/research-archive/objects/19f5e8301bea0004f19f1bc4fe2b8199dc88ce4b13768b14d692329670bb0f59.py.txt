from pathlib import Path
import hashlib,json
R=Path(__file__).resolve().parent;P=R/'f20-score-owner'
def ident(p):return dict(path=str(p),sha256=hashlib.sha256(p.read_bytes()).hexdigest())
names=['f20-q-consumer-scale','f20-ffn-fouracc-rebase','f20-k-consumer-tile','f20-packed-weights','f20-packed-bitdecode']
rows=[]
for name in names:
 n=R/name;perf=json.loads((n/'ORDINARY-REVIEW.json').read_text());nat=json.loads((n/'NATIVE-REVIEW.json').read_text());res=json.loads((n/'RESOURCE-REVIEW.json').read_text())
 st={x['name']:x['value'] for x in res['pipeline']['executables'][0]['statistics']};assert st['Scratch Memory Size']==0
 row=dict(name=name,plan=ident(n/'plan.txt'),spv=ident(n/'candidate.spv'),gate='fixed+fb2 gold; ordinary feedback final gold',savings_ms=perf['mean_savings_ms'],ci95=perf['ci95'],verdict=perf['verdict'],pairs=perf['savings_ms'],scratch=0,shared=st['Workgroup Memory Size'],native_instructions=nat['candidate']['instruction_count'],native_counts=nat['candidate']['counts'])
 rows.append(row)
 state=dict(status='MEASURED',adoption='NOT_ADOPTED' if perf['verdict']!='FASTER' else 'VERIFIED_GAIN_CANDIDATE_PENDING_BASELINE_DECISION',result=row,baseline=ident(P/'plan.txt'))
 (n/'STATE.json').write_text(json.dumps(state,indent=2),encoding='utf-8')
 (n/'RESUME.txt').write_text('Completed exact fixed/fb2, native, one 3-pair ordinary batch and driver resources. All samples retained. Do not repeat same identity to chase significance. See STATE and parent EXACT-FOUR-RESULTS.json.\n',encoding='utf-8')
out=dict(baseline=ident(P/'plan.txt'),results=rows,scope='Four directions; packed-bitdecode is a repair of the fourth implementation, not another direction.',ordinary_jobs=30,ordinary_samples=900,gpu_jobs_total=50,confidence='95% t interval over 3 process-pair means, positive savings means faster. No outlier removal.',method=ident(R/'EXACT-FOUR-METHOD.md'),limits=['Static instruction counts include debug and alternative paths. Not elapsed time.','Known synthetic fixed/feedback fixture; no general all-input correctness proof.','Native absence of scratch plus driver scratch0; physical GRF allocation/occupancy still unknown.'])
(R/'EXACT-FOUR-RESULTS.json').write_text(json.dumps(out,indent=2),encoding='utf-8')
lines=['# Exact F20 four-direction results','','Baseline remains separately preserved: f20-score-owner. Positive savings means faster. Each row is an independent arm, not additive.','','| Trial | Savings ms | 95% CI | Decision |','|---|---:|---|---|']
for x in rows:lines.append(f"| {x['name']} | {x['savings_ms']:+.6f} | [{x['ci95'][0]:+.6f}, {x['ci95'][1]:+.6f}] | {x['verdict']} |")
lines+=['','All arms passed fixed/fb2 and ordinary feedback exact-frame checks. Driver F20 scratch0 and shared25088B for every arm, no increased scratch in other pipelines. All900 ordinary samples retained; no profiling times used.','', 'Packed initial version used a floating decode path. Native hf<-f/f<-hf MOV counts exposed that unnecessary conversion; bitdecode repairs four calls using unpackFloat2x16, preserving pack and mapping. Both identities/results are retained.','','Mapping, rounding, lifecycle and native cost discussion: EXACT-FOUR-METHOD.md. Per-arm PROOF/CONTRACT/NATIVE/ORDINARY/RESOURCE records and Luna execution directories contain the reproducible evidence.']
(R/'EXACT-FOUR-RESULTS.md').write_text('\n'.join(lines)+'\n',encoding='utf-8')
files=[R/'build_exact_four.py',R/'build_packed_bitdecode.py',R/'review_exact_four.py',R/'EXACT-FOUR-METHOD.md',R/'EXACT-FOUR-RESULTS.json',R/'EXACT-FOUR-RESULTS.md']
for name in names:files.extend(p for p in (R/name).iterdir() if p.is_file() and p.suffix in ['.json','.comp','.spv','.txt','.bin','.py'])
(R/'EXACT-FOUR-MANIFEST.json').write_text(json.dumps([ident(p) for p in files],indent=2),encoding='utf-8')
print(json.dumps([{k:x[k] for k in ['name','savings_ms','ci95','verdict']} for x in rows],indent=2))
