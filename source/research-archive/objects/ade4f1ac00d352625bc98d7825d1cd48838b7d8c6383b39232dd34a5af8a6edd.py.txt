def spec_from(path):
    s = importlib.util.spec_from_file_location("ro_spv_tools", SPVTOOLS)
    m = importlib.util.module_from_spec(s)
    s.loader.exec_module(m)
    return m


def sha(p):
    return hashlib.sha256(Path(p).read_bytes()).hexdigest()


def ident(p):
    p = Path(p)
    return {"path": p.as_posix(), "bytes": p.stat().st_size, "sha256": sha(p)}

def copy_source(case):
    """Copy the frozen source tree (comp + its includes) verbatim into the card."""
    dst = CARD / "source/parent" / case["card"]
    dst.mkdir(parents=True, exist_ok=True)
    srcdir = R / case["src_dir"]
    copies = []
    for name in [case["src"]] + case["headers"]:
        src = R / case["header_src"][name] if case.get("header_src") and name in case["header_src"] \
            else srcdir / name
        out = dst / name
        shutil.copyfile(src, out)
        copies.append({"name": name, "from": src.as_posix(), **ident(out)})
    return copies


def compile_raw(case, src_dir, out):
    argv = [str(GLSLANG), "-V", "--target-env", "vulkan1.3"]
    if case.get("spirv_val"):
        argv.append("--spirv-val")
    argv += ["-I" + str(src_dir)]
    if case.get("extra_inc"):
        argv += ["-I" + str(R / case["extra_inc"])]
    argv += case["defines"]
    argv += ["-o", str(out), str(src_dir / case["src"])]
    t0 = time.time()
    r = subprocess.run(argv, cwd=str(R), capture_output=True)
    log = (BRUN / ("%s-raw-compile.log" % case["card"]))
    log.write_bytes(r.stdout + r.stderr)
    if r.returncode != 0:
        raise SystemExit("compile failed %s: %s" % (case["card"], (r.stdout + r.stderr)[:2000]))
    return {"argv": argv, "cwd": str(R), "rc": r.returncode,
            "log": ident(log), "elapsed_s": round(time.time() - t0, 2)}


def nc_transfer(case, raw, out, sp):
    """NoContraction transfer exactly as the deployed build did it."""
    if not case.get("nc_parent"):
        shutil.copyfile(raw, out)
        return {"applied": False, "reason": "module is its own parent (no earlier F3 SPV)"}
    parent = sp.Module(R / case["nc_parent"])
    child = sp.Module(raw)
    keep, info = sp.transfer_decoration_pattern_aligned(
        parent, child, sp.no_contraction_ids(parent), verify=True)
    sp.set_no_contraction(child, set(keep) | set(sp.no_contraction_ids(child)), out)
    return {"applied": True, "parent_spv": ident(R / case["nc_parent"]),
            "parent_nc": len(sp.no_contraction_ids(parent)),
            "final_nc": len(sp.no_contraction_ids(sp.Module(out))),
            "mapping": info}


def validate(path):
    r = subprocess.run([str(VALIDATOR), str(path)], capture_output=True)
    txt = (r.stdout + r.stderr).decode(errors="replace").strip()
    return {"rc": r.returncode, "out": txt,
            "result_zero": "result=0" in txt,
            "words": int(txt.split("words=")[1].split()[0]) if "words=" in txt else None}


def main():
    BRUN.mkdir(parents=True, exist_ok=True)
    (CARD / "spv").mkdir(parents=True, exist_ok=True)
    sp = spec_from(SPVTOOLS)
    rec = {"status": "CPU_ONLY", "generated": time.strftime("%Y-%m-%dT%H:%M:%S%z"),
           "toolchain": {"compiler": ident(GLSLANG), "validator": ident(VALIDATOR),
                         "nc_transfer": ident(SPVTOOLS)},
           "cases": []}
    ok_all = True
    for case in CASES:
        src_copy = copy_source(case)
        assert src_copy[0]["sha256"] == case["src_sha"], (case["card"], "source sha mismatch")
        raw = BRUN / ("%s.parent.raw.spv" % case["card"])
        cp = compile_raw(case, CARD / "source/parent" / case["card"], raw)
        raw_id = ident(raw)
        final = CARD / "spv" / ("parent_repro-%s.spv" % case["pass"])
        nc = nc_transfer(case, raw, final, sp)
        fin_id = ident(final)
        dep = R / case["deployed"]
        deployed = ident(dep)
        val = validate(final)
        raw_match_doc = (case["raw_doc_sha"] is None) or (raw_id["sha256"] == case["raw_doc_sha"])
        byte_exact = fin_id["sha256"] == case["deployed_sha"] == deployed["sha256"]
        ok = byte_exact and val["result_zero"] and raw_match_doc
        ok_all = ok_all and ok
        rec["cases"].append({
            "card": case["card"], "pass": case["pass"], "g3_order": case["order"],
            "sources": src_copy, "compile": cp,
            "raw_spv": raw_id,
            "raw_matches_documented": raw_match_doc,
            "raw_documented_sha256": case["raw_doc_sha"],
            "raw_documented_bytes": case["raw_doc_bytes"],
            "nc_transfer": nc,
            "reproduced_spv": fin_id,
            "deployed_spv": deployed,
            "deployed_sha256_expected": case["deployed_sha"],
            "deployed_bytes_expected": case["deployed_bytes"],
            "byte_identical": byte_exact,
            "validator": val,
            "verdict": "PASS" if ok else "FAIL",
        })
    rec["all_byte_identical"] = all(c["byte_identical"] for c in rec["cases"])
    rec["all_validator_result_zero"] = all(c["validator"]["result_zero"] for c in rec["cases"])
    rec["verdict"] = "PASS" if ok_all else "FAIL"
    out = BRUN / "parent-replay.json"
    out.write_text(json.dumps(rec, indent=1, ensure_ascii=False) + "\n", encoding="utf-8")
    print(json.dumps({"verdict": rec["verdict"], "cases": [
        {"card": c["card"], "pass": c["pass"],
         "raw": c["raw_spv"]["sha256"][:16], "raw_bytes": c["raw_spv"]["bytes"],
         "raw_matches_documented": c["raw_matches_documented"],
         "final": c["reproduced_spv"]["sha256"][:16],
         "final_bytes": c["reproduced_spv"]["bytes"],
         "byte_identical": c["byte_identical"],
         "validator": c["validator"]["out"]} for c in rec["cases"]]}, indent=1))
    return 0 if ok_all else 1


if __name__ == "__main__":
    sys.exit(main())
