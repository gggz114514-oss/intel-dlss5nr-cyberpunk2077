#!/usr/bin/env python3
"""Collect every card into RESULTS.json / MANIFEST.json and emit the STATUS table.

Reads only what the cards wrote; never invents a measurement.  A card contributes:
  * its CANDIDATE.json (identity + qualification),
  * the sha256 of every file it produced,
  * the jobs it defined (listed, not run).
"""
import hashlib
import json
from pathlib import Path

W = Path(r"@NR_WORKSPACE@/work/dsh-xmx-handoff-validation-20261024")
ROOT = Path(r"@NR_WORKSPACE@")


def sha(p):
    h = hashlib.sha256()
    with open(p, "rb") as f:
        for chunk in iter(lambda: f.read(1 << 20), b""):
            h.update(chunk)
    return h.hexdigest()


def main():
    cards, manifest = [], []
    for cdir in sorted((W / "candidates").iterdir()):
        if not cdir.is_dir():
            continue
        cand_json = cdir / "CANDIDATE.json"
        entry = {"card": cdir.name,
                 "candidate_json": str(cand_json) if cand_json.exists() else None,
                 "report": str(cdir / "REPORT.md") if (cdir / "REPORT.md").exists() else None,
                 "files": [], "qualification": None, "jobs": None}
        if cand_json.exists():
            try:
                doc = json.loads(cand_json.read_text(encoding="utf-8"))
                entry["qualification"] = (doc.get("qualification")
                                          or doc.get("qualifications")
                                          or doc.get("verdict", {}).get("qualification"))
                entry["summary"] = {k: doc.get(k) for k in
                                    ("what", "one_line", "title", "declared_shared_bytes",
                                     "dynamic_removed_cost", "verdict") if k in doc}
            except Exception as exc:                       # noqa: BLE001
                entry["candidate_json_error"] = str(exc)
        jobs = W / "run" / cdir.name / "jobs.json"
        if jobs.exists():
            try:
                jd = json.loads(jobs.read_text(encoding="utf-8"))
                entry["jobs"] = {"path": str(jobs), "count": len(jd.get("jobs", [])),
                                 "sha256": sha(jobs)}
                entry["job_list"] = jd.get("jobs", [])
            except Exception as exc:                       # noqa: BLE001
                entry["jobs_error"] = str(exc)
        for f in sorted(cdir.rglob("*")):
            if f.is_file():
                entry["files"].append({"path": str(f), "bytes": f.stat().st_size,
                                       "sha256": sha(f)})
        cards.append(entry)

    for sub in ("tools", "tickets"):
        for f in sorted((W / sub).glob("*")):
            if f.is_file():
                manifest.append({"path": str(f), "bytes": f.stat().st_size, "sha256": sha(f)})
    for f in sorted(W.glob("*.md")):
        manifest.append({"path": str(f), "bytes": f.stat().st_size, "sha256": sha(f)})

    results = {"work": str(W), "generated_by": "tools/collect.py",
               "qualification_vocabulary": ["CPU_STRUCT", "OFFLINE_COMPILE", "SPV_IDENTITY",
                                            "PLAN_IDENTITY", "NATIVE", "GATE", "ORDINARY",
                                            "QUALITY"],
               "gpu": "not run: no GPU slot allocated for this work package; every job is "
                      "defined in run/<card>/jobs.json and not executed",
               "cards": cards}
    (W / "RESULTS.json").write_text(json.dumps(results, indent=1) + "\n",
                                    encoding="utf-8", newline="\n")
    (W / "MANIFEST.json").write_text(json.dumps({"work": str(W), "top_level": manifest,
                                                 "cards": [{"card": c["card"], "files": c["files"]}
                                                           for c in cards]},
                                                indent=1) + "\n",
                                     encoding="utf-8", newline="\n")
    print(f"{'card':24} {'files':>5} {'jobs':>5}  qualification")
    for c in cards:
        q = c.get("qualification")
        q = ",".join(q) if isinstance(q, list) else (q or "-")
        print(f"{c['card']:24} {len(c['files']):5d} {(c['jobs'] or {}).get('count', 0):5d}  {q}")
    print("total files", sum(len(c['files']) for c in cards))


if __name__ == "__main__":
    main()
