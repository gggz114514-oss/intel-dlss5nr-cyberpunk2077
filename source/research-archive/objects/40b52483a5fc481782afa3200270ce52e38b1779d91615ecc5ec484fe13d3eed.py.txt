import json, os
ROOT = r"@NR_WORKSPACE@\work\codex-astra-fp16-int8-audit-20261024\fullnet-continuous-20261027"
fg = json.load(open(os.path.join(ROOT, "evidence", "FAMILY-GROUPS.json"), encoding="utf-8"))
rows = {}
for sha, g in fg["groups"].items():
    for o in g["ordinals"]:
        rows[o] = g
want = [0, 5, 62, 63, 91, 92, 133, 166, 167, 192, 193, 212, 213, 226, 230, 231]
seen = set()
for o in want:
    g = rows[o]
    src = g.get("source")
    if src in seen:
        print(o, g["family"], g["names"][0], "SAME SOURCE", src)
        continue
    seen.add(src)
    try:
        n = sum(1 for _ in open(src, encoding="utf-8", errors="replace"))
        b = os.path.getsize(src)
    except Exception as e:
        n, b = "ERR", str(e)
    print(o, g["family"], g["names"][0], "lines=%s bytes=%s" % (n, b))
    print("   ", src)
