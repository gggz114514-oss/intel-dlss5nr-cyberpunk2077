"""Measure the small network properly: 6 balanced pairs of (small 118-pass plan) vs (our head).

The old number for small was 13.721 ms from a single sample, explicitly not a performance result.
This runs the same plan through the same runner as everything else, warm 1, reps 30, pool on,
feedback on, six balanced orderings, all samples retained. small is a different model so only its
timing is compared; our head arm carries its own gold proofs.
"""
import json, sys
from pathlib import Path
sys.path.insert(0, str(Path(__file__).parent))
import COMMON as K
import STAGE

SMALL = Path('@NR_WORKSPACE@/work/vk102-small-310-9-16-20261019/'
             'package/manifest/port_network_runtime.txt')
cur = K.read_json(K.C / 'CURRENT.json')
head = Path(cur['best_plan']['path'])
arms = {'S': (SMALL, K.sha(SMALL)), 'C': (head, cur['best_plan']['sha256'])}
out = K.C / 'queue' / 'small-measure'
out.mkdir(parents=True, exist_ok=True)

PERM = [('S', 'C'), ('C', 'S'), ('S', 'C'), ('C', 'S'), ('S', 'C'), ('C', 'S')]
jobs = []
for pi, order in enumerate(PERM):
    for aid in order:
        p, s = arms[aid]
        jobs.append(K.ordinary_job('pair-%d-%s' % (pi, aid), p, s,
                                   out / 'run' / ('pair-%d-%s' % (pi, aid)), aid, pi))
spec = dict(stage='ordinary', owner='LUNA-CONTINUOUS-SMALL',
            authorization='Main authorized measuring the small network plan and the current head in '
                          'one balanced batch. small is a different model, timing only.',
            output_root=str(out), identities=[str(v[0]) for v in arms.values()] + [str(K.RUNNER)],
            jobs=jobs, rules='six balanced pairs; all samples retained')
K.write_json(out / 'spec-ordinary.json', spec)
manifest, ps1 = STAGE.prepare(spec)
STAGE.run(manifest, ps1)
s = STAGE.summarize(spec['output_root'])
K.write_json(out / 'STAGE-SUMMARY.json', s)
print('ordinary', s['status'])
if s['status'] != 'COMPLETED':
    print([(j['id'], j['status'], j['frame']) for j in s['jobs']])
    sys.exit(1)

# per-run means from the receipts
means = {}
for run in sorted((out / 'run').glob('pair-*')):
    rc = K.read_json(run / 'port_network_receipt.json')
    arm = run.name.split('-')[-1]
    means.setdefault(arm, []).append((int(run.name.split('-')[1]), rc['gpu_mean_ms'],
                                      rc.get('gpu_passes'), rc.get('nan'), rc.get('inf')))
T = 2.570581835636305
for a in means:
    v = sorted(means[a])
    avg = sum(x[1] for x in v) / len(v)
    print('%s: %d runs, passes %s, mean %.3f ms, per-run %s' % (
        a, len(v), {x[2] for x in v}, avg, ['%.2f' % x[1] for x in v]))
s_ = sorted(means['S'])
c_ = sorted(means['C'])
n = min(len(s_), len(c_))
d = [c_[i][1] - s_[i][1] for i in range(n)]
m = sum(d) / n
sd = (sum((x - m) ** 2 for x in d) / (n - 1)) ** 0.5
half = T * sd / (n ** 0.5)
print('\nhead - small = %+.4f ms  CI95[%+.4f, %+.4f]  (positive means the head is slower)' % (m, m - half, m + half))
K.write_json(K.C / 'evidence' / 'SMALL-NETWORK-TIMING.json',
             dict(arms={k: sorted(v) for k, v in means.items()},
                  head_minus_small_ms=m, CI95_ms=[m - half, m + half], groups=n))
