"""Build the campaign ledger (CAMPAIGN.json + CAMPAIGN.md) from measured artifacts.

Everything is derived: accepted chain from CURRENT.json, per-card verdicts from
cards/<id>/RESULTS.json + TIMING.json, family coverage from evidence/COVERAGE-232.json.
"""
import json, sys
from pathlib import Path
sys.path.insert(0, str(Path(__file__).parent))
import COMMON as K


def card_rows():
    out = []
    for d in sorted((K.C / 'cards').iterdir()):
        if not d.is_dir():
            continue
        row = dict(card=d.name)
        r = d / 'RESULTS.json'
        t = d / 'TIMING.json'
        req = d / 'GPU-REQUEST.json'
        if req.exists():
            try:
                q = K.read_json(req)
                row['shader_group'] = (q.get('shader_group') or {}).get('pass_ordinals')
                row['family_declared'] = q.get('card')
            except Exception:
                pass
        if (d / 'candidate.spv').exists():
            row['candidate_spv_sha256'] = K.sha(d / 'candidate.spv')
        for plan_name in ('plan.txt', 'plan-v2.txt'):
            if (d / plan_name).exists():
                row['plan_' + plan_name.replace('.', '_')] = K.sha(d / plan_name)
        if t.exists():
            try:
                tim = K.read_json(t)
                c = tim['comparisons'][0]
                row['timing'] = dict(saving_ms=c['saving_ms'], CI95_ms=c['CI95_ms'],
                                     status=c['status'], groups=c['groups'],
                                     base_mean=c.get('group_means') and
                                     sum(g[tim['arms'][0]] for g in c['group_means']) / len(c['group_means']),
                                     cand_mean=c.get('group_means') and
                                     sum(g[tim['arms'][1]] for g in c['group_means']) / len(c['group_means']))
            except Exception as e:
                row['timing_error'] = str(e)[:200]
        if r.exists():
            try:
                res = K.read_json(r)
                row['status'] = res.get('status')
                row['triage'] = res.get('triage')
                row['decision'] = res.get('decision')
                n = res.get('layer2_native')
                if isinstance(n, dict) and 'base' in n:
                    row['native'] = dict(instr=[n['base']['instructions'], n['cand']['instructions']],
                                         send=[n['base']['sends'], n['cand']['sends']],
                                         dpas=[n['base']['dpas'], n['cand']['dpas']],
                                         max_r=[n['base']['max_r'], n['cand']['max_r']],
                                         msg_delta=n.get('message_delta'))
            except Exception as e:
                row['results_error'] = str(e)[:200]
        if 'status' not in row:
            row['status'] = 'INFRA_OR_IN_PROGRESS'
        out.append(row)
    return out


def main():
    cur = K.read_json(K.C / 'CURRENT.json') if (K.C / 'CURRENT.json').exists() else None
    cov = None
    cp = K.C / 'evidence' / 'COVERAGE-232.json'
    if cp.exists():
        try:
            cov = K.read_json(cp)
        except Exception:
            cov = None
    led = dict(baseline=dict(handoff=str(K.BASELINE), handoff_sha256=K.BASELINE_SHA),
               chain=cur['chain'] if cur else None,
               chain_head=cur['best_plan'] if cur else None,
               cumulative_ms_in_chain=(cur or {}).get('cumulative_ms_in_chain'),
               protocol=dict(gates_fixed=K.FIXED_GOLD, gates_fb2=K.FB2_GATE_GOLD,
                             ordinary=K.ORDINARY_GOLD,
                             lock=str(K.LOCK),
                             ordinary_design='6 alternating process pairs, 30 samples/arm, all retained'),
               cards=card_rows())
    if cov:
        fam = cov.get('family_summary') or cov.get('families') or {}
        led['coverage_source'] = str(cp)
        led['coverage_family_summary'] = fam if isinstance(fam, dict) else None
    K.write_json(K.C / 'CAMPAIGN.json', led)

    lines = ['# 全后端连续改造 — 战役账本', '',
             '基线（handoff，未动）: `%s`  sha256 `%s`' % (K.BASELINE, K.BASELINE_SHA), '']
    if cur:
        lines += ['## 接受链', '', '| # | 计划 | sha256 | 同批增量 ms |', '|---|---|---|---|']
        for i, e in enumerate(cur['chain']):
            lines.append('| %d | %s | `%s` | %s |' % (i, e.get('label', ''), e['sha256'][:16],
                                                      ('%.6f' % e['saving_ms']) if e.get('saving_ms') else '—'))
        lines += ['', '链头累计 +%.6f ms（相对 handoff，不相加历史毫秒）' % cur.get('cumulative_ms_in_chain', 0), '']
    lines += ['## 各卡', '', '| 卡 | 状态 | 分诊 | native instr | send | 整体 saving ms | CI95 |', '|---|---|---|---|---|---|---|']
    for c in led['cards']:
        n = c.get('native') or {}
        t = c.get('timing') or {}
        lines.append('| %s | %s | %s | %s | %s | %s | %s |' % (
            c['card'], c.get('status'), (c.get('triage') or '')[:28],
            (n.get('instr') and '%s→%s' % tuple(n['instr'])) or '',
            (n.get('send') and '%s→%s' % tuple(n['send'])) or '',
            ('%.6f' % t['saving_ms']) if t else '',
            ('[%.4f, %.4f]' % tuple(t['CI95_ms'])) if t else ''))
    (K.C / 'CAMPAIGN.md').write_text('\n'.join(lines) + '\n', encoding='utf-8')
    print('cards', len(led['cards']), '->', K.C / 'CAMPAIGN.json')
    print('\n'.join(lines[-14:]).encode('utf-8','replace').decode('ascii','replace'))


if __name__ == '__main__':
    main()
