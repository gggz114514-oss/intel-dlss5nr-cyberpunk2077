"""Is the small network's own plan runnable through our runner? Check every BUFFER and PASS path."""
import re, sys
from pathlib import Path
sys.path.insert(0, str(Path(__file__).parent))
import COMMON as K

plan = Path('@NR_WORKSPACE@/work/vk102-small-310-9-16-20261019/'
            'package/manifest/port_network_runtime.txt')
raw = plan.read_text(encoding='utf-8-sig', errors='replace')
lines = raw.splitlines()
print('header:', lines[0][:120])
missing_buf, missing_spv = [], []
nbuf = npass = 0
for line in lines:
    if line.startswith('BUFFER '):
        nbuf += 1
        parts = line.split('"')
        if len(parts) >= 4:
            p = Path(re.sub(r'\\{2,}', r'\\', parts[3].strip()))
            if not p.exists():
                missing_buf.append(str(p))
    elif line.startswith('PASS '):
        npass += 1
        parts = line.split('"')
        if len(parts) >= 4:
            p = Path(re.sub(r'\\{2,}', r'\\', parts[3].strip()))
            if not p.exists():
                missing_spv.append(str(p))
print('BUFFER %d (missing %d), PASS %d (missing %d)' % (nbuf, len(missing_buf), npass, len(missing_spv)))
for m in missing_buf[:5]:
    print('  missing buffer:', m)
for m in missing_spv[:5]:
    print('  missing spv   :', m)
print('layout/policy lines:', sum(1 for l in lines if l.startswith('POLICY ')))
