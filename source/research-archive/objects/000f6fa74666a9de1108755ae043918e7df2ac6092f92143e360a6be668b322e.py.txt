import sys
from collections import Counter
sys.path.insert(0, r"@NR_WORKSPACE@/work/dsh-xmx-handoff-validation-20261024/candidates/b1-c32-xe-reuse/tools")
from spvlite import Mod
card = r"@NR_WORKSPACE@/work/dsh-xmx-handoff-validation-20261024/candidates/b1-c32-xe-reuse"
p = Mod(card + "/spv/parent_repro.spv")
c = Mod(card + "/spv/b1_body1_xe.spv")
def wc(m):
    out = Counter()
    for it in m.insts:
        out[it.name] += 1 + len(it.a)
    return out
wp, wc_ = wc(p), wc(c)
print("total words: parent", sum(wp.values()), "candidate", sum(wc_.values()),
      "delta", sum(wc_.values()) - sum(wp.values()))
print("header words: parent", p.w[:5].__len__()+0, " -> both modules have 5 header words + trailing", p.trailing, c.trailing)
for k in sorted(set(wp) | set(wc_)):
    if wp[k] != wc_[k]:
        print("  %-34s %6d -> %6d  (%+d words)" % (k, wp[k], wc_[k], wc_[k]-wp[k]))
print("instruction count parent", len(p.insts), "candidate", len(c.insts))
