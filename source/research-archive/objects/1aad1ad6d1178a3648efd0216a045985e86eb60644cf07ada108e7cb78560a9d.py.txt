"""n35-f18-latency build: parent byte reproduction, candidate compile, SPV validate.

CPU only.  Reads the parent source / header (read-only) and writes only inside this
card directory.
"""
import json, sys, shutil, subprocess
from pathlib import Path

CAMP = Path(__file__).resolve().parent.parent.parent
sys.path.insert(0, str(CAMP))
import COMMON as K

HERE = Path(__file__).resolve().parent
SRC = Path(r'@NR_WORKSPACE@/candidates/amd-netorg-t3d-20261018/c256-canonical-only-v1/source/c3x_pre_c256.comp')
PIN = CAMP / 'cards/idt-identity-recipe/evidence/headers/c27fa8c9e2e1'
PARENT_SPV = Path(r'@NR_WORKSPACE@/candidates/amd-netorg-t3d-20261018/c256-canonical-only-v1/c3x_pre_c256.spv')
CAND_COMP = HERE / 'candidate.comp'


def main():
    out = dict(card='n35-f18-latency', ordinal=166, name='c3x_pre_c256', family='F18')
    row = K.row_by_ordinal(166)
    base, base_sha, _ = K.current_best()
    out['parent_plan'] = dict(path=str(base), sha256=base_sha)
    out['baseline_plan'] = dict(path=str(K.BASELINE), sha256=K.BASELINE_SHA)
    out['parent'] = dict(spv=str(PARENT_SPV), sha256=K.sha(PARENT_SPV),
                         source=str(SRC), source_sha256=K.sha(SRC))
    out['parent']['row_spv_sha256'] = row['spv_sha256']
    out['parent']['header'] = dict(path=str(SRC.parent / 'nb05_swin_c256_common.glsl'),
                                   sha256=K.sha(SRC.parent / 'nb05_swin_c256_common.glsl'))
    out['parent']['pinned_header'] = dict(path=str(PIN / 'nb05_swin_c256_common.glsl'),
                                          sha256=K.sha(PIN / 'nb05_swin_c256_common.glsl'))
    out['parent']['header_pin_matches'] = out['parent']['header']['sha256'] == \
        out['parent']['pinned_header']['sha256']

    # ---- 1. parent byte reproduction (raw recipe from evidence/RECIPES.json)
    incs = K.include_dirs_for(166, SRC)
    out['include_dirs'] = [str(p) for p in incs]
    raw = HERE / 'parent-rebuild.spv'
    K.compile_glsl(SRC, raw, incs)
    out['parent_rebuild'] = dict(path=str(raw), sha256=K.sha(raw), size=raw.stat().st_size,
                                 raw_match=K.sha(raw) == row['spv_sha256'],
                                 target_sha256=row['spv_sha256'],
                                 target_size=PARENT_SPV.stat().st_size)
    if out['parent_rebuild']['raw_match']:
        out['parent_rebuild']['recipe'] = 'raw'
    else:
        out['parent_rebuild']['recipe'] = 'MISMATCH'

    # ---- 2. candidate compile
    cand = HERE / 'candidate.spv'
    K.compile_glsl(CAND_COMP, cand, [HERE] + incs)
    out['candidate'] = dict(comp=str(CAND_COMP), comp_sha256=K.sha(CAND_COMP),
                            spv=str(cand), spv_sha256=K.sha(cand), size=cand.stat().st_size)
    # ---- 3. validate
    K.validate_spv(cand)
    out['candidate']['spv_validate_cpu'] = 'PASS'

    # ---- 4. NoContraction accounting (spirv opcode census of both modules)
    sys.path.insert(0, str(K.SPV_TOOLS))
    import spv_tools_local as ST
    out['no_contraction'] = dict(
        parent=int(len(ST.no_contraction_ids(ST.Module(PARENT_SPV)))),
        candidate=int(len(ST.no_contraction_ids(ST.Module(cand)))),
    )

    NAMED = {128: 'OpIAdd', 129: 'OpFAdd', 130: 'OpISub', 131: 'OpFSub', 132: 'OpIMul',
             133: 'OpFMul', 134: 'OpUDiv', 135: 'OpSDiv', 136: 'OpFDiv', 137: 'OpUMod',
             138: 'OpSRem', 139: 'OpSMod', 169: 'OpSelect', 109: 'OpConvertFToU',
             110: 'OpConvertFToS', 111: 'OpConvertSToF', 112: 'OpConvertUToF',
             115: 'OpFConvert', 124: 'OpBitcast', 126: 'OpSNegate', 71: 'OpDecorate',
             54: 'OpFunctionCall', 61: 'OpStore', 62: 'OpLoad', 248: 'OpFunctionEnd',
             81: 'OpImageTexelPointer'}

    def census(p):
        from collections import Counter
        m = ST.Module(Path(p))
        c = Counter(t[0] for t in ST.code_instructions(m))
        return dict(total=sum(c.values()),
                    named={NAMED[k]: v for k, v in sorted(c.items()) if k in NAMED},
                    other=sum(v for k, v in c.items() if k not in NAMED))
    out['spirv_opcodes'] = dict(parent=census(PARENT_SPV), candidate=census(cand))

    K.write_json(HERE / 'BUILD.json', out)
    print(json.dumps({k: v for k, v in out.items() if k not in ('spirv_opcodes',)}, indent=1))
    print('parent opcodes', out['spirv_opcodes']['parent'])
    print('cand   opcodes', out['spirv_opcodes']['candidate'])


if __name__ == '__main__':
    main()
