def toks(line):
    try: return shlex.split(line)
    except ValueError: return line.split()

class Plan:
    def __init__(self, path):
        self.path = path; self.passes = {}; self.order = []
        for ln in open(path, encoding="utf-8", errors="replace"):
            s = ln.strip()
            if not s: continue
            t = toks(s); k = t[0]
            if k == "PASS":
                d = {"group": int(t[1]), "name": t[2], "spv": t[3], "args": tuple(t[4:])}
                nm = t[2] if t[2] not in self.passes else t[2] + "#dup%d" % len(self.passes)
                self.passes[nm] = d; self.order.append(d)

def sha(p):
    if not os.path.exists(p): return "MISSING"
    h = hashlib.sha256()
    with open(p, "rb") as f:
        for b in iter(lambda: f.read(1 << 20), b""):
            h.update(b)
    return h.hexdigest()

cache = {}
