"""Add the residency-pinned diag plan (diag2) so the three dumps are actually readable.

Evidence for why: the 2026-10-29 arm ran candidate-34189-diag.txt where
  POLICY 343 0 ... / POLICY 395 0 ... / POLICY 396 0 ...
(resident=0). The runner receipt shows those three buffers share their memory_offset
with unrelated tensors (vit_block31_A @111476736 is also swin_5_Y/block9_out/..., Y
@128540672 is also swin_6_Y/7/8..., block30_out @137887744 is also split16_b23_Y/...),
so the end-of-run dumps are other tenants' bytes.
The proven b31 gate plans in candidates/dsh-int8-same-family-20261023/{b0-repair,vit-qx-fuse}
dump the same buffer with POLICY 395 *1* (resident=1).
"""
from pathlib import Path
import hashlib
import json
import sys

R = Path("@NR_WORKSPACE@")
S = R / "work/dsh-fullnet-static-20261029/S08-vit"
sys.path.insert(0, str(R / "work/dsh-fullnet-static-20261029/tools"))
import planlib  # noqa: E402

PIN = ("block30_out", "vit_block31_A", "vit_block31_Y")


def sha(p):
    return hashlib.sha256(Path(p).read_bytes()).hexdigest()


def main():
    src = S / "candidate-34189-diag.txt"
    p = planlib.Plan(src)
    p.assert_valid()
    log = [p.set_policy(name, resident=1) for name in PIN]
    out = S / "candidate-34189-diag2.txt"
    text = p.to_text()
    lines = [l.rstrip("\r") for l in text.split("\n")]
    lines = [l for l in lines if l != ""]
    out.write_bytes(("\r\n".join(lines) + "\r\n").encode("utf-8"))
    q = planlib.Plan(out)
    q.assert_valid()
    base = planlib.Plan(src)
    a_lines = [l.rstrip("\r") for l in src.read_text(encoding="utf-8").split("\n") if l.strip()]
    b_lines = [l.rstrip("\r") for l in out.read_text(encoding="utf-8").split("\n") if l.strip()]
    changed = [(i + 1, x, y) for i, (x, y) in enumerate(zip(a_lines, b_lines)) if x != y]
    rep = dict(
        source=dict(path=str(src), sha256=sha(src), header=base.header),
        out=dict(path=str(out), sha256=sha(out), header=q.header,
                 dispatches=len(q.passes), buffers=len(q.buffers), outputs=len(q.outputs)),
        policy_edits=log, changed_line_count=len(changed),
        changed_lines=[dict(line=i, old=o, new=n) for i, o, n in changed],
        why="resident=0 -> the pool reuses the allocation and the end-of-run dump is another tensor",
        precedent="candidates/dsh-int8-same-family-20261023/b0-repair/plan-b31-*-gate.txt uses "
                  "POLICY 395 1 118 121 0 and its dump_vit_block31_A.bin IS the A tensor",
    )
    (S / "PLANS-DIAG2.json").write_text(json.dumps(rep, indent=1) + "\n", encoding="utf-8")
    print(json.dumps(dict(out=rep["out"], changed=rep["changed_lines"]), indent=1))
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
