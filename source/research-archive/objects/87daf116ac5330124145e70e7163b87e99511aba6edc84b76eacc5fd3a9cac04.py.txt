"""probe the OpCooperativeMatrixLoad/StoreKHR operand encoding (CPU only, read-only)"""
import sys
from pathlib import Path

HERE = Path(__file__).resolve().parent
sys.path.insert(0, str(HERE))
import spvlib  # noqa: E402

P = Path("@NR_WORKSPACE@/candidates/amd-map-20261015"
         "/round4-integrated/candidate/spv/output-c64.spv")
m = spvlib.Mod(P)
print("constants:", {k: v for k, v in sorted(m.consts.items()) if v in (0, 1, 16, 64, 128, 256)})
for it in m.insts:
    if it.name in ("OpCooperativeMatrixLoadKHR", "OpCooperativeMatrixStoreKHR"):
        print(it.woff, it.name, "operands(a)=", it.a,
              "| consts:", [m.consts.get(x, "-") for x in it.a[2:]],
              "| cmat:", m.cm_type(it.a[0]))
