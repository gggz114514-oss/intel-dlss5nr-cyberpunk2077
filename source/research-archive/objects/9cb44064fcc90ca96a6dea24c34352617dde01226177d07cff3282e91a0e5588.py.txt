"""Generate SOURCE-DIFF.patch (parent source -> candidate source) and verify round trip.

parent = cards/n12-vit-qkv/candidate.comp (the F14 package's existing member; this
card's diff sits on top of it), candidate = this card's candidate.comp.

The patch is applied back to the parent text by a minimal unified-diff applier and
must reproduce the candidate byte for byte.
"""
import difflib
import hashlib
import json
import sys
from pathlib import Path

CARD = Path(__file__).resolve().parent.parent
CAMPAIGN = CARD.parent.parent
sys.path.insert(0, str(CAMPAIGN))
import COMMON as K  # noqa: E402

EVID = CARD / 'evidence'
PARENT = CAMPAIGN / 'cards' / 'n12-vit-qkv' / 'candidate.comp'
CAND = CARD / 'candidate.comp'
OUT = CARD / 'SOURCE-DIFF.patch'


def apply_unified(patch_text, base_text):
    """Apply a single-file unified diff (no /dev/null) to base_text."""
    out = []
    base = base_text.splitlines(keepends=True)
    pos = 0
    lines = patch_text.splitlines(keepends=True)
    i = 0
    while i < len(lines):
        ln = lines[i]
        if ln.startswith('@@'):
            import re
            m = re.match(r'@@ -(\d+)(?:,(\d+))? \+(\d+)(?:,(\d+))? @@', ln)
            assert m, ln
            a = int(m.group(1))
            out.extend(base[pos:a - 1])
            pos = a - 1
            i += 1
            continue
        if ln.startswith(('---', '+++')):
            i += 1
            continue
        if ln.startswith(' '):
            out.append(base[pos])
            pos += 1
        elif ln.startswith('-'):
            pos += 1
        elif ln.startswith('+'):
            out.append(ln[1:])
        i += 1
    out.extend(base[pos:])
    return ''.join(out)


def main():
    parent_text = PARENT.read_text(encoding='utf-8')
    cand_text = CAND.read_text(encoding='utf-8')
    a = parent_text.splitlines()
    b = cand_text.splitlines()
    diff = list(difflib.unified_diff(
        a, b,
        fromfile='a/cards/n12-vit-qkv/candidate.comp (F14 package member n12: norm tree)',
        tofile='b/cards/n12b-f14-scale/candidate.comp (F14 package: + scale coefficient hoist)',
        n=6, lineterm=''))
    patch_text = '\n'.join(diff) + '\n'
    OUT.write_text(patch_text, encoding='utf-8')

    round_trip = apply_unified(patch_text, parent_text)
    added = [l for l in diff if l.startswith('+') and not l.startswith('+++')]
    removed = [l for l in diff if l.startswith('-') and not l.startswith('---')]
    hunks = sum(1 for l in diff if l.startswith('@@'))

    # semantic scope: the added/removed code lines (comments excluded)
    code_added = [l[1:].strip() for l in added if l[1:].strip() and not l[1:].strip().startswith('//')]
    code_removed = [l[1:].strip() for l in removed if l[1:].strip() and not l[1:].strip().startswith('//')]

    out = dict(
        schema='n12b-f14-scale/source-diff/v1',
        card='n12b-f14-scale',
        parent=dict(path=str(PARENT), sha256=K.sha(PARENT),
                    note='n12-vit-qkv candidate (F14 package member already built and measured)'),
        candidate=dict(path=str(CAND), sha256=K.sha(CAND), bytes=CAND.stat().st_size),
        patch=dict(path=str(OUT), sha256=K.sha(OUT), bytes=OUT.stat().st_size,
                   hunks=hunks, added_lines=len(added), removed_lines=len(removed)),
        round_trip_ok=round_trip == cand_text,
        semantic_added=code_added,
        semantic_removed=code_removed,
        scope=dict(
            statement='the only code change is that the Q coefficient read moves above the '
                      'two-iteration publication loop; the in-loop declaration becomes a use '
                      'of the hoisted value',
            norm_tree_untouched=True,
            k_order_untouched=True,
            write_domain_untouched=True,
            qkv_branch_untouched=True,
            scale_expression_untouched=True,
        ),
    )
    K.write_json(EVID / 'SOURCE-DIFF.json', out)
    print(json.dumps(out, indent=2)[:2500])
    assert out['round_trip_ok'], 'patch round trip failed'


if __name__ == '__main__':
    main()
