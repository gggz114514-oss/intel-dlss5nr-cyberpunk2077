"""Generate cards/n29-f01-alu-chain/candidate.comp from the card parent
(n18-f01-entry/candidate.comp) by three literal edits, then emit the diff.

Every edit replaces a range-fixup that the caller's clamp already makes an
identity; none of them touches a float expression, a rounding point, a K order
or a reduction tree.  The script asserts each anchor occurs exactly once, so a
drifted parent cannot silently produce a different candidate.
"""
import difflib
import sys
from pathlib import Path

CARD = Path(__file__).resolve().parents[1]
PARENT = CARD.parent / 'n18-f01-entry' / 'candidate.comp'
OUT = CARD / 'candidate.comp'

EDITS = [
    # ---------------------------------------------------------------- (1)
    # nrf_hist_linear: the four corner clamps -> ij and min(ij+1, hi)
    (
        """vec3 nrf_hist_linear(vec2 pos) {
    uint W=uint(nrf_ctl(9)),H=uint(nrf_ctl(10));
    vec2 q=pos-0.5,fr=fract(q);
    ivec2 ij=ivec2(floor(q));
    ivec2 hi=ivec2(int(W)-1,int(H)-1);
    vec3 a=nrf_rgba_clamped(pc.p6.y,W,clamp(ij,ivec2(0),hi)).rgb;
    vec3 b=nrf_rgba_clamped(pc.p6.y,W,clamp(ij+ivec2(1,0),ivec2(0),hi)).rgb;
    vec3 c=nrf_rgba_clamped(pc.p6.y,W,clamp(ij+ivec2(0,1),ivec2(0),hi)).rgb;
    vec3 e=nrf_rgba_clamped(pc.p6.y,W,clamp(ij+ivec2(1),ivec2(0),hi)).rgb;
    return mix(mix(a,b,fr.x),mix(c,e,fr.x),fr.y);
}""",
        """vec3 nrf_hist_linear(vec2 pos) {
    uint W=uint(nrf_ctl(9)),H=uint(nrf_ctl(10));
    vec2 q=pos-0.5,fr=fract(q);
    ivec2 ij=ivec2(floor(q));
    ivec2 hi=ivec2(int(W)-1,int(H)-1);
    // The five call sites below pass p0/pm/p2, every component of which the
    // caller clamps into [0.5, size-0.5]; pos-0.5 is then exact in fp32 (all
    // values below 2^24 are multiples of 2^-24), so ij=floor(pos-0.5) already
    // lies in [0, size-1] and both bounds of the "ij" corner are dead.  The
    // "+1" corners only need their upper bound: ij+1 >= 1 > 0.
    const ivec2 i1=min(ij+ivec2(1),hi);
    vec3 a=nrf_rgba_clamped(pc.p6.y,W,ij).rgb;
    vec3 b=nrf_rgba_clamped(pc.p6.y,W,ivec2(i1.x,ij.y)).rgb;
    vec3 c=nrf_rgba_clamped(pc.p6.y,W,ivec2(ij.x,i1.y)).rgb;
    vec3 e=nrf_rgba_clamped(pc.p6.y,W,i1).rgb;
    return mix(mix(a,b,fr.x),mix(c,e,fr.x),fr.y);
}""",
    ),
    # ---------------------------------------------------------------- (2)
    # nrf_rgba: the only caller passes the already-reflected pixel (sx,sy),
    # which is inside [0,W-1]x[0,H-1], so the mirror there is the identity.
    (
        """vec4 nrf_rgba(uint base,uint W,uint H,ivec2 xy) {
    uint a=base+uint(nrf_mirror(xy.y,int(H)))*W*4u+uint(nrf_mirror(xy.x,int(W)))*4u;
    return vec4(f[a],f[a+1u],f[a+2u],f[a+3u]);
}
""",
        "",
    ),
    (
        """// the j<0 fixup is dead, and j<size selects j.  Reading the already clamped
// pixel directly is therefore the same address, the same bits and the same
// order for every input; only the modulo/reflection arithmetic disappears.
vec4 nrf_rgba_clamped(uint base,uint W,ivec2 xy) {""",
        """// the j<0 fixup is dead, and j<size selects j.  Reading the already clamped
// pixel directly is therefore the same address, the same bits and the same
// order for every input; only the modulo/reflection arithmetic disappears.
//
// nrf_rgba (the mirrored variant) is gone with this card: its single caller is
// the source-pixel read, whose xy is sx/sy from nrf_prepare_A's own
// reflection.  For the tile-grid launch (grid = ceil(dim/8); 168x96 = 1344x768
// here, and the plan's 16515072-byte image buffer pins W*H at 1344*768) the
// reflection already lands inside [0,W-1]x[0,H-1], so both nrf_mirror calls
// there were the identity as well.  nrf_mirror itself stays: nrf_depth_at and
// nrf_motion_uv read coordinates that really can leave the image.
vec4 nrf_rgba_clamped(uint base,uint W,ivec2 xy) {""",
    ),
    # ---------------------------------------------------------------- (3)
    # nrf_depth_at: the centre read is xy itself (in range); only the four
    # neighbour reads can leave the image and keep the mirror.
    (
        """float nrf_depth_at(ivec2 xy) {
    uint W=pc.p5.x,H=pc.p5.y;
    return f[pc.p6.w+uint(nrf_mirror(xy.y,int(H)))*W+uint(nrf_mirror(xy.x,int(W)))];
}""",
        """float nrf_depth_at(ivec2 xy) {
    uint W=pc.p5.x,H=pc.p5.y;
    return f[pc.p6.w+uint(nrf_mirror(xy.y,int(H)))*W+uint(nrf_mirror(xy.x,int(W)))];
}
// Centre probe of the 3x3 depth search: xy is the already-reflected pixel and
// is in range, so the mirror is the identity there (its four neighbours are
// xy+(+-1,+-1) and do leave the image, so they keep nrf_depth_at).
float nrf_depth_at_in(ivec2 xy) {
    uint W=pc.p5.x,H=pc.p5.y;
    return f[pc.p6.w+uint(xy.y)*W+uint(xy.x)];
}""",
    ),
    # ---------------------------------------------------------------- (2b, 3b)
    # the two call sites that now use the in-range forms
    (
        """        const vec3 sampled=nrf_rgba(pc.p4.z,W,H,xy).rgb;""",
        """        const vec3 sampled=nrf_rgba_clamped(pc.p4.z,W,xy).rgb;""",
    ),
    (
        """                float best=nrf_depth_at(xy);""",
        """                float best=nrf_depth_at_in(xy);""",
    ),
]


def main():
    src = PARENT.read_text(encoding='utf-8')
    text = src
    for i, (old, new) in enumerate(EDITS):
        n = text.count(old)
        assert n == 1, 'edit %d: anchor occurs %d times' % (i + 1, n)
        text = text.replace(old, new)
    OUT.write_text(text, encoding='utf-8')
    diff = ''.join(difflib.unified_diff(
        src.splitlines(keepends=True), text.splitlines(keepends=True),
        fromfile='a/cards/n29-f01-alu-chain/parent-source.comp (n18 candidate.comp)',
        tofile='b/cards/n29-f01-alu-chain/candidate.comp', n=4))
    (CARD / 'SOURCE-DIFF.patch').write_text(diff, encoding='utf-8')
    print(diff)
    print('candidate bytes: %d -> %d' % (len(src), len(text)))


if __name__ == '__main__':
    main()
