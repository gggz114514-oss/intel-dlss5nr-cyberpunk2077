Required by the S00 card:

  T1  no-op patch on combo-34189 -> byte-identical output, SHA 2b5481d5..., CRLF kept
  T2  single C64 scale-pack patch -> output passes planlib.validate, nothing else moves
  T3  freeze identity -> matches HANDOFF-IDENTITIES.json for combo-34189 and QUALITY-24

Supplementary negative controls show the conflict machinery fails closed:

  N1  wrong base_sha256
  N2  one patch carrying every conflict class (unknown name, shared target, duplicate
      name, BIND overrun, missing file, SPV name drift, missing SPV SHA, non-basename
      OUTPUT, duplicate/out-of-range PUSH)

Run with the tool-env python::

    <Q>/tool-env/Scripts/python.exe -X utf8 run_tests.py

Everything is written under this card directory (out/, logs/, TESTS.json).
"""

def now() -> str:
    return datetime.now(timezone.utc).astimezone().isoformat(timespec="seconds")


def posix(p: Path | str) -> str:
    return Path(p).as_posix()


PYTHON = posix(sys.executable)


def display(argv: List[str]) -> str:
    return " ".join(f'"{a}"' if " " in a else a for a in argv)


def run(argv: List[str], timeout: int = 1800) -> Dict[str, Any]:
    start = time.time()
    proc = subprocess.run(argv, capture_output=True, text=True, encoding="utf-8", errors="replace", timeout=timeout)
    return {
        "argv": argv,
        "command": display(argv),
        "exit_code": proc.returncode,
        "duration_seconds": round(time.time() - start, 3),
        "stdout": proc.stdout,
        "stderr": proc.stderr,
    }


def assertions() -> List[Dict[str, Any]]:
    return []


def add(items: List[Dict[str, Any]], name: str, ok: bool, detail: Any = None) -> None:
    items.append({"name": name, "ok": bool(ok), "detail": detail})


def sha(path: Path) -> str:
    return planlib.sha256_file(path)


def test_t1(t: Dict[str, Any]) -> Dict[str, Any]:
    a = assertions()
    out = OUT / "test1-noop-34189.txt"
    report = OUT / "test1-report.json"
    for p in (out, report):
        p.unlink(missing_ok=True)
    r = run([PYTHON, "-X", "utf8", posix(HERE / "patch_merge.py"),
             "--base", posix(BASE_C64), "--patch", posix(HERE / "patches" / "test1-noop.json"),
             "--out", posix(out), "--report", posix(report)])
    t["command"] = r["command"]
    t["exit_code"] = r["exit_code"]
    t["duration_seconds"] = r["duration_seconds"]
    if r["stderr"].strip():
        t["stderr"] = r["stderr"][:4000]
    rep = json.loads(report.read_text(encoding="utf-8")) if report.is_file() else {}
    base_bytes = BASE_C64.read_bytes()
    add(a, "exit_code == 0", r["exit_code"] == 0, r["exit_code"])
    add(a, "output file exists", out.is_file(), posix(out))
    out_bytes = out.read_bytes() if out.is_file() else b""
    add(a, "output is byte-identical to base", out_bytes == base_bytes,
        {"out_bytes": len(out_bytes), "base_bytes": len(base_bytes)})
    out_sha = hashlib.sha256(out_bytes).hexdigest() if out_bytes else None
    add(a, "output SHA256 == 2b5481d5...", out_sha == EXPECTED["combo_34189_sha"], out_sha)
    add(a, "CRLF line endings preserved",
        out_bytes.count(b"\r\n") == base_bytes.count(b"\r\n") == out_bytes.count(b"\n"),
        {"crlf": out_bytes.count(b"\r\n"), "lf": out_bytes.count(b"\n")})
    add(a, "report ok=true", rep.get("ok") is True, rep.get("ok"))
    add(a, "report changed_lines == 0", rep.get("out", {}).get("changed_lines") == 0,
        rep.get("out", {}).get("changed_lines"))
    add(a, "report byte_identical_to_base", rep.get("out", {}).get("byte_identical_to_base") is True, None)
    add(a, "report base_sha256 matches expected", rep.get("base", {}).get("sha256") == EXPECTED["combo_34189_sha"], None)
    add(a, "report planlib_validate == ok", rep.get("verified", {}).get("planlib_validate") == "ok", None)
    t["assertions"] = a
    t["pass"] = all(x["ok"] for x in a)
    t["artifacts"] = {"out": posix(out), "report": posix(report), "out_sha256": out_sha}
    return t


def test_t2(t: Dict[str, Any]) -> Dict[str, Any]:
    a = assertions()
    out = OUT / "test2-b5-scale-34189.txt"
    report = OUT / "test2-report.json"
    for p in (out, report):
        p.unlink(missing_ok=True)
    r = run([PYTHON, "-X", "utf8", posix(HERE / "patch_merge.py"),
             "--base", posix(BASE_C32), "--patch", posix(HERE / "patches" / "test2-c64-b5-scale.json"),
             "--out", posix(out), "--report", posix(report)])
    t["command"] = r["command"]
    t["exit_code"] = r["exit_code"]
    t["duration_seconds"] = r["duration_seconds"]
    if r["stderr"].strip():
        t["stderr"] = r["stderr"][:4000]
    rep = json.loads(report.read_text(encoding="utf-8")) if report.is_file() else {}
    add(a, "exit_code == 0", r["exit_code"] == 0, r["exit_code"])
    add(a, "output file exists", out.is_file(), posix(out))
    problems: List[str] = []
    if out.is_file():
        try:
            cand = planlib.Plan(out)
            problems = cand.validate()
            base = planlib.Plan(BASE_C32)
        except Exception as exc:  # noqa: BLE001 - recorded as a failed assertion
            problems = [f"parse failed: {exc}"]
            cand = base = None
    else:
        cand = base = None
    add(a, "output passes planlib.validate", problems == [], problems)
    if cand is not None and base is not None:
        b5 = cand.buffers_by_name["c64_b5_scales"]
        add(a, "c64_b5_scales bytes == 2368", b5.bytes == 2368, b5.bytes)
        add(a, "c64_b5_scales file == new C64 pack", b5.file == posix(C64 / "b5-scales.bin"), b5.file)
        add(a, "c64_b5_scales file SHA == bd57f21a...",
            sha(Path(b5.file)) == EXPECTED["b5_scale_sha"], sha(Path(b5.file)))
        bind_ok = True
        bind_detail = []
        for ordinal in (6, 7, 8):
            p = cand.passes[ordinal]
            binds = [x for x in p.binds if x.buffer == b5.index]
            ok = len(binds) == 1 and binds[0].bytes == 2368 and binds[0].offset == 0
            bind_ok &= ok
            bind_detail.append({"ordinal": ordinal, "dispatch": p.name, "ok": ok,
                                "binds": [[x.binding, x.offset, x.bytes] for x in binds]})
        add(a, "BIND 1 for c64_b5_scales followed the resize at dispatches 6/7/8", bind_ok, bind_detail)
        others = ["c64_b6_scales", "c64_b7_scales", "c64_b8_scales",
                  "c64_b62_scales", "c64_b63_scales", "c64_b64_scales", "c64_b65_scales"]
        untouched = []
        for name in others:
            before, after = base.buffers_by_name[name], cand.buffers_by_name[name]
            untouched.append({"buffer": name, "bytes_before": before.bytes, "bytes_after": after.bytes,
                              "file_same": before.file == after.file,
                              "ok": before.bytes == after.bytes == 1856 and before.file == after.file})
        add(a, "the other seven C64 scale packs untouched (1856 B, same file)", all(x["ok"] for x in untouched), untouched)
        weights = [n for n in cand.buffers_by_name if "c64_b" in n and n.endswith("_weights")]
        w_ok = all((cand.buffers_by_name[n].file, cand.buffers_by_name[n].bytes) ==
                   (base.buffers_by_name[n].file, base.buffers_by_name[n].bytes) for n in weights)
        add(a, f"all {len(weights)} C64 weight packs untouched", w_ok, None)
        changed = [(x.name, x.bytes, x.file) for x in cand.buffers
                   if (x.bytes, x.file) != (base.buffers_by_name[x.name].bytes, base.buffers_by_name[x.name].file)]
        add(a, "exactly one BUFFER changed", len(changed) == 1, changed)
        push_same = all(cand.passes[k].push == base.passes[k].push for k in range(len(base.passes)))
        add(a, "all 288 PUSH lines unchanged", push_same and len(cand.passes) == len(base.passes), len(cand.passes))
        add(a, "OUTPUT identity unchanged", cand.outputs[:1] == base.outputs[:1], cand.outputs)
        cap_before = sorted((b.name, b.file) for b in base.buffers if b.file != "-" and "/dataset/" in b.file)
        cap_after = sorted((b.name, b.file) for b in cand.buffers if b.file != "-" and "/dataset/" in b.file)
        add(a, "capture input bindings unchanged", cap_before == cap_after, cap_before)
        add(a, "report changed_lines == 4 (1 BUFFER + 3 BIND)", rep.get("out", {}).get("changed_lines") == 4,
            rep.get("out", {}).get("changed_lines"))
        add(a, "report only_planned_lines_changed", rep.get("verified", {}).get("only_planned_lines_changed") is True, None)
        add(a, "report new file SHA recorded", rep.get("new_file_sha256", {}).get(posix(C64 / "b5-scales.bin")) == EXPECTED["b5_scale_sha"],
            rep.get("new_file_sha256"))
    t["assertions"] = a
    t["pass"] = all(x["ok"] for x in a)
    t["artifacts"] = {"out": posix(out), "report": posix(report),
                      "out_sha256": sha(out) if out.is_file() else None}
    return t


def projection(freeze: Dict[str, Any]) -> Dict[str, Any]:
    return {
        "plans": [(p["frame"], p["sha256"], p["resource_digest"], p["spv_digest"]) for p in freeze["plans"]],
        "quality": {k: v["sha256"] for k, v in freeze["quality"].items()},
        "files": {k: v["sha256"] for k, v in freeze["files"].items()},
        "spvs": {k: v["sha256"] for k, v in freeze["spvs"].items()},
        "buffer_layout": freeze["buffer_layout"],
        "capture_input_buffers": freeze["summary"]["capture_input_buffers"],
    }

             expect_min_conflicts: int = 1) -> Dict[str, Any]:

    out = OUT / out_name
    report = OUT / f"{out_name}.report.json"
    for p in (out, report):
        p.unlink(missing_ok=True)
    r = run([PYTHON, "-X", "utf8", posix(HERE / "patch_merge.py"),
             "--base", posix(BASE_C64), "--patch", posix(HERE / "patches" / patch_name),
             "--out", posix(out), "--report", posix(report)])
    t["command"] = r["command"]
    t["exit_code"] = r["exit_code"]
    t["duration_seconds"] = r["duration_seconds"]
    rep = json.loads(report.read_text(encoding="utf-8")) if report.is_file() else {}
    codes = [c["code"] for c in rep.get("conflicts", [])]
    add(a, "exit_code == 3 (conflict reported, fail closed)", r["exit_code"] == 3, r["exit_code"])

    t["pass"] = all(x["ok"] for x in a)

        changed = [(x.name, x.file) for x in cand.buffers
                   if (x.bytes, x.file) != (base.buffers_by_name[x.name].bytes, base.buffers_by_name[x.name].file)]
        add(a, "exactly 2 BUFFERs changed (block-5 weights/scales)", len(changed) == 2, changed)
        changed_spv = [k for k in range(len(base.passes)) if base.passes[k].spv != cand.passes[k].spv]
        add(a, "exactly 2 SPVs changed (ordinals 6 and 8)", changed_spv == [6, 8], changed_spv)
        add(a, "report changed_lines == 7 (2 BUFFER + 3 BIND + 2 SPV)",
            rep.get("out", {}).get("changed_lines") == 7, rep.get("out", {}).get("changed_lines"))
        # ---- T5b: dry-run PUSH + OUTPUT on the new plan -------------------
        patch_b = {
            "schema": "dsh-plan-patch/1",
            "comment": "T5b dry-run machinery only: PUSH word and OUTPUT name; nothing is written.",
            "base_sha256": cand.sha256(),
            "set_push": [{"ordinal": 0, "name": cand.passes[0].name, "word": 22,
                          "value": 12345, "expect_old": int(cand.passes[0].push[22])}],
            "output": {"index": 0, "name": "dryrun-network.bin", "expect_old": "replayed-network.bin"},
        }
        pb = OUT / "T5b-patch.json"
        pb.write_text(json.dumps(patch_b, indent=2) + "\n", encoding="utf-8")
        dry_out = OUT / "T5b-dryrun.txt"
        dry_out.unlink(missing_ok=True)
        rb = run([PYTHON, "-X", "utf8", posix(HERE / "patch_merge.py"),
                  "--base", posix(out), "--patch", posix(pb), "--out", posix(dry_out), "--dry-run"])
        rep_b = json.loads(rb["stdout"]) if rb["stdout"].strip().startswith("{") else {}
        kinds = [c.get("kind") for c in rep_b.get("changes", [])]
        add(a, "dry-run exit_code == 0", rb["exit_code"] == 0, rb["exit_code"])
        add(a, "dry-run reports PUSH and OUTPUT changes",
            "PUSH" in kinds and "OUTPUT" in kinds, kinds)
        add(a, "dry-run writes nothing", rep_b.get("out", {}).get("written") is False and not dry_out.exists(),
            rep_b.get("out", {}).get("written"))
        add(a, "dry-run output SHA differs from the base plan",
            rep_b.get("out", {}).get("sha256") not in (None, cand.sha256()),
            rep_b.get("out", {}).get("sha256"))
        add(a, "dry-run planlib_validate == ok", rep_b.get("verified", {}).get("planlib_validate") == "ok", None)
    t["assertions"] = a
    t["pass"] = all(x["ok"] for x in a)
    t["artifacts"] = {"out": posix(out), "report": posix(report),
                      "out_sha256": sha(out) if out.is_file() else None,
                      "dryrun_patch": posix(OUT / "T5b-patch.json")}
    return t


def test_n3(t: Dict[str, Any]) -> Dict[str, Any]:
    """Supplementary: freeze --verify must detect a changed hash, a missing file and a changed plan."""
    import copy

    a = assertions()
    frozen = json.loads((HERE / "CANDIDATE-FREEZE.json").read_text(encoding="utf-8"))
    delivered_sha_before = sha(HERE / "CANDIDATE-FREEZE.json")
    tampered = copy.deepcopy(frozen)
    victim_file = sorted(tampered["files"])[0]
    tampered["files"][victim_file]["sha256"] = "0" * 64
    victim_layout = sorted(tampered["buffer_layout"])[0]
    missing_path = posix(OUT / "missing-resource.bin")
    tampered["buffer_layout"][victim_layout]["file"] = missing_path
    tampered["plans"][0]["sha256"] = "1" * 64
    tampered_path = OUT / "N3-tampered-freeze.json"
    tampered_path.write_text(json.dumps(tampered, indent=2, ensure_ascii=False) + "\n", encoding="utf-8")
    r = run([PYTHON, "-X", "utf8", posix(HERE / "freeze_candidates.py"), "--verify", posix(tampered_path)])
    t["command"] = r["command"]
    t["exit_code"] = r["exit_code"]
    t["duration_seconds"] = r["duration_seconds"]
    rep = json.loads(r["stdout"]) if r["stdout"].strip().startswith("{") else {}
    problems = rep.get("problems", {})
    add(a, "exit_code == 2 (drift detected)", r["exit_code"] == 2, r["exit_code"])
    add(a, "ok == false", rep.get("ok") is False, rep.get("ok"))
    add(a, "changed file hash detected", any(x.get("path") == victim_file for x in problems.get("files_changed", [])),
        problems.get("files_changed", [])[:2])
    add(a, "missing resource file detected", missing_path in problems.get("files_missing", []),
        problems.get("files_missing", [])[:2])
    add(a, "changed plan detected", any(x.get("frame") == frozen["plans"][0]["frame"]
                                        for x in problems.get("plans_changed", [])),
        problems.get("plans_changed", [])[:2])
    add(a, "delivered freeze untouched",
        sha(HERE / "CANDIDATE-FREEZE.json") == delivered_sha_before, delivered_sha_before)
    t["assertions"] = a
    t["pass"] = all(x["ok"] for x in a)
    t["artifacts"] = {"tampered_freeze": posix(tampered_path), "problems": {k: len(v) for k, v in problems.items()}}
    return t
