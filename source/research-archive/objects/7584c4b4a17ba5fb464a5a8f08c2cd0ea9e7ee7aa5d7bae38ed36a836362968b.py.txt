"""Static SPIR-V enumeration of cooperative-matrix multiply-add sites.

CPU only: reads the module bytes, walks the instruction stream, resolves the
component types of A/B/C through the id -> result-type map and reports every
matrix multiply with its word offset, result id, function and type triple.

The definition of a "site" is the one used by
work/codex-astra-fp16-int8-audit-20261024/FULLNET-RESIDUAL-MMA-SITES.json:
one OpCooperativeMatrixMulAddKHR / NV instruction, offset = index of the
instruction's first word in the module's 32-bit word array.  Running this on
the same modules must reproduce those offsets and ids; that reproduction is
the calibration check for every number reported for the new modules.

usage: python spv_mma.py <module.spv> [<module.spv> ...] [--json out.json]
"""
import hashlib
import json
import sys
from pathlib import Path

OP_NAME = {
    21: "OpTypeInt",
    22: "OpTypeFloat",
    23: "OpTypeVector",
    24: "OpTypeMatrix",
    43: "OpConstant",
    46: "OpFma",
    54: "OpFunction",
    56: "OpFunctionEnd",
    71: "OpDecorate",
    4456: "OpTypeCooperativeMatrixKHR",
    4457: "OpCooperativeMatrixLoadKHR",
    4458: "OpCooperativeMatrixStoreKHR",
    4459: "OpCooperativeMatrixMulAddKHR",
    4460: "OpCooperativeMatrixLengthKHR",
    5356: "OpTypeCooperativeMatrixNV",
    5357: "OpCooperativeMatrixMulAddNV",
    5358: "OpCooperativeMatrixLoadNV",
    5359: "OpCooperativeMatrixStoreNV",
}
MULADD = {4459: "KHR", 5357: "NV"}
CMTYPE = {4456: "KHR", 5356: "NV"}
F16 = {"kind": "float", "width": 16}


def words_of(path):
    raw = Path(path).read_bytes()
    n = len(raw) // 4
    return list(int.from_bytes(raw[i * 4:i * 4 + 4], "little") for i in range(n))


def sha256(path):
    h = hashlib.sha256()
    with open(path, "rb") as f:
        for b in iter(lambda: f.read(1 << 20), b""):
            h.update(b)
    return h.hexdigest()


def scan(path):
    w = words_of(path)
    if not w or w[0] != 0x07230203:
        return {"path": str(path), "error": "not a SPIR-V module"}
    i = 5
    insts = []          # (offset, opcode, words)
    while i < len(w):
        wc = w[i] >> 16
        op = w[i] & 0xFFFF
        if wc == 0:
            break
        insts.append((i, op, w[i:i + wc]))
        i += wc
    # Type declarations put their own result id in word 1 and must be excluded
    # from the generic "word 1 is the result type" rule below.
    TYPE_DECL = set(range(19, 40)) | {4456, 5356}
    type_ids = {ws[1] for _, op, ws in insts if op in TYPE_DECL and len(ws) > 1}
    defs = {}
    for off, op, ws in insts:
        if op in TYPE_DECL and len(ws) > 1:
            defs[ws[1]] = (op, ws)
        elif op not in TYPE_DECL and len(ws) >= 3 and ws[1] in type_ids:
            defs[ws[2]] = (op, ws)

    def const_val(idv):
        d = defs.get(idv)
        if d and d[0] == 43 and len(d[1]) > 3:
            return d[1][3]
        return idv

    def type_of(idv, depth=0):
        d = defs.get(idv)
        if not d or depth > 4:
            return {"kind": "unknown", "id": idv}
        op, ws = d
        if op in (21, 22):
            return {"kind": "int" if op == 21 else "float",
                    "width": ws[2], "type_id": ws[1], "op": OP_NAME[op]}
        if op in CMTYPE:
            comp = ws[2]
            c = type_of(comp, depth + 1)
            out = {"kind": "coopmat_" + ("float" if c.get("kind") == "float" else "int"),
                   "component": c, "type_id": ws[1], "op": OP_NAME[op]}
            if op == 4456:
                out.update(rows=const_val(ws[4]), cols=const_val(ws[5]),
                           use=ws[6], scope=ws[3])
            else:
                out.update(rows=const_val(ws[4]), cols=const_val(ws[5]), scope=ws[3])
            return out
        if op == 23:
            return {"kind": "vector", "component": type_of(ws[2], depth + 1),
                    "type_id": ws[1]}
        # any other result-producing instruction: its declared result type is
        # word 1, and the value's component type is that type's component type
        if len(ws) > 1 and ws[1] != idv and ws[1] in defs:
            return type_of(ws[1], depth + 1)
        return {"kind": "other", "op": OP_NAME.get(op, f"op{op}"), "type_id": ws[1] if len(ws) > 1 else None}

    # function ranges
    funcs = []
    cur = None
    for off, op, ws in insts:
        if op == 54:
            cur = {"id": ws[2], "first": off, "last": None}
            funcs.append(cur)
        elif op == 56 and cur is not None:
            cur["last"] = off
            cur = None

    def func_at(off):
        for f in funcs:
            if f["last"] is not None and f["first"] <= off <= f["last"]:
                return f["id"]
        return None

    sites = []
    counts = {}
    for off, op, ws in insts:
        counts[OP_NAME.get(op, f"op{op}")] = counts.get(OP_NAME.get(op, f"op{op}"), 0) + 1
        if op not in MULADD:
            continue
        rtype, rid, a, b, c = ws[1], ws[2], ws[3], ws[4], ws[5]
        ta, tb, tc = type_of(a), type_of(b), type_of(c)
        sites.append({
            "word_offset": off,
            "result_id": rid,
            "function_id": func_at(off),
            "result_type_id": rtype,
            "A_id": a, "B_id": b, "C_id": c,
            "A": ta, "B": tb, "C": tc,
            "op": OP_NAME[op],
        })

    def combo(s):
        def key(t):
            if t.get("kind") == "coopmat_int":
                c = t["component"]
                return f"i{c['width']}"
            if t.get("kind") == "coopmat_float":
                return f"f{t['component']['width']}"
            return t.get("kind", "?")
        return [key(s["A"]), key(s["B"]), key(s["C"])]

    fp16 = []
    int8 = []
    other = []
    for s in sites:
        k = combo(s)
        s["combo"] = k
        if k[0] == "f16" and k[1] == "f16":
            fp16.append(s)
        elif k[0] in ("i8", "u8") and k[1] in ("i8", "u8") and k[2] == "i32":
            int8.append(s)
        else:
            other.append(s)

    return {
        "path": Path(path).as_posix(),
        "sha256": sha256(path),
        "bytes": Path(path).stat().st_size,
        "words": len(w),
        "functions": [{"id": f["id"], "first_word": f["first"], "last_word": f["last"]}
                      for f in funcs],
        "matrix_type_count": sum(1 for _, op, _ in insts if op in CMTYPE),
        "muladd_count": len(sites),
        "fp16_mma_count": len(fp16),
        "int8_mma_count": len(int8),
        "other_mma_count": len(other),
        "fp16_mma_sites": [{"word_offset": s["word_offset"], "result_id": s["result_id"],
                            "function_id": s["function_id"], "A_id": s["A_id"],
                            "B_id": s["B_id"], "C_id": s["C_id"]} for s in fp16],
        "int8_mma_sites": [{"word_offset": s["word_offset"], "result_id": s["result_id"],
                            "function_id": s["function_id"], "combo": s["combo"],
                            "A_rows": s["A"].get("rows"), "A_cols": s["A"].get("cols"),
                            "B_rows": s["B"].get("rows"), "B_cols": s["B"].get("cols"),
                            "C_rows": s["C"].get("rows"), "C_cols": s["C"].get("cols")}
                           for s in int8],
        "other_mma_sites": [{"word_offset": s["word_offset"], "combo": s["combo"]}
                            for s in other],
        "op_histogram": dict(sorted(counts.items())),
        "coopmat_load": counts.get("OpCooperativeMatrixLoadKHR", 0) + counts.get("OpCooperativeMatrixLoadNV", 0),
        "coopmat_store": counts.get("OpCooperativeMatrixStoreKHR", 0) + counts.get("OpCooperativeMatrixStoreNV", 0),
        "op_fma": counts.get("OpFma", 0),
    }


if __name__ == "__main__":
    args = [a for a in sys.argv[1:] if a != "--json"]
    out = None
    if "--json" in sys.argv:
        out = sys.argv[sys.argv.index("--json") + 1]
        args = [a for a in args if a != out]
    recs = [scan(p) for p in args]
    txt = json.dumps(recs if len(recs) > 1 else recs[0], indent=1, ensure_ascii=False)
    if out:
        Path(out).write_text(txt + "\n", encoding="utf-8")
    print(txt)
