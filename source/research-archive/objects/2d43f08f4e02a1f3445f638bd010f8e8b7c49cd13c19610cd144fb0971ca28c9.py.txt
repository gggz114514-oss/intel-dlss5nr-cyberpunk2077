#!/usr/bin/env python3
"""The 129-position map: original G3 identity -> current cumulative PASS -> status.

Identity of an original position is (G3 plan sha256, 0-based ordinal in G3, pass name).
The plan's `PASS <n>` label is not unique and ordinals moved when R13/R14 inserted a PASS,
so the map is keyed by name and records both ordinals.

usage: build_position_map.py <g3 census.json> <current census.json> <current label> --out dir
"""
import hashlib
import json
import sys
from pathlib import Path


def load(path, needle):
    d = json.loads(Path(path).read_text(encoding="utf-8"))
    return [p for p in d["plans"] if needle in p["plan"]][0]


def main(argv):
    out = None
    if "--out" in argv:
        i = argv.index("--out")
        out = Path(argv[i + 1])
        argv = argv[:i] + argv[i + 2:]
    g3 = load(argv[0], "g3")
    cur = load(argv[1], argv[2])
    cur_by_name = {r["pass_name"]: r for r in cur["rows"]}
    g3_fp16 = [r for r in g3["rows"] if r["fp16_mma"] > 0]
    rows = []
    for i, r in enumerate(g3_fp16):
        c = cur_by_name[r["pass_name"]]
        rows.append({
            "position": i,
            "g3_ordinal": r["ordinal"],
            "pass_name": r["pass_name"],
            "g3_spv": r["spv"], "g3_spv_sha256": r["spv_sha256"],
            "g3_fp16_mma": r["fp16_mma"],
            "current_ordinal": c["ordinal"],
            "current_spv": c["spv"], "current_spv_sha256": c["spv_sha256"],
            "current_fp16_mma": c["fp16_mma"], "current_int8_mma": c["int8_mma"],
            "closed": c["fp16_mma"] == 0,
            "status": "静态零残余" if c["fp16_mma"] == 0 else (
                "段级替换" if c["int8_mma"] > 0 and c["fp16_mma"] < r["fp16_mma"] else "未关闭"),
        })
    closed = [r for r in rows if r["closed"]]
    doc = {
        "key": "(g3 plan sha256, g3 0-based ordinal, pass name) — the plan's PASS number is "
               "not unique, and ordinals moved when R13/R14 inserted a PASS",
        "g3": {"plan": g3["plan"], "sha256": g3["plan_sha256"], "passes": g3["passes"],
               "fp16_sites": g3["fp16_static_mma_total"],
               "positions": g3["passes_with_fp16"]},
        "current": {"plan": cur["plan"], "sha256": cur["plan_sha256"], "passes": cur["passes"],
                    "buffers": cur["buffers"],
                    "fp16_sites": cur["fp16_static_mma_total"],
                    "positions": cur["passes_with_fp16"]},
        "closed_positions": len(closed),
        "open_positions": len(rows) - len(closed),
        "fp16_sites_removed": g3["fp16_static_mma_total"] - cur["fp16_static_mma_total"],
        "positions": rows,
    }
    if out:
        out.mkdir(parents=True, exist_ok=True)
        (out / "POSITION-MAP-129.json").write_text(
            json.dumps(doc, indent=1, ensure_ascii=False) + "\n", encoding="utf-8", newline="\n")
        L = ["# 129 位置映射（G3 原身份 → 当前累计）", "",
             f"G3 `{Path(g3['plan']).name}` `{g3['plan_sha256'][:16]}…`："
             f"{g3['passes_with_fp16']} 个位置 / {g3['fp16_static_mma_total']} 条 FP16 MMA",
             f"当前 `{Path(cur['plan']).name}` `{cur['plan_sha256'][:16]}…`："
             f"**关闭 {len(closed)} / {len(rows)}**，残余 {cur['passes_with_fp16']} 位置 / "
             f"{cur['fp16_static_mma_total']} 条", "",
             "| # | G3 ord | pass | G3 FP16 | 现 ord | 现模块 | 现 FP16 | 现 INT8 | 状态 |",
             "|---:|---:|---|---:|---:|---|---:|---:|---|"]
        for r in rows:
            L.append(f"| {r['position']} | {r['g3_ordinal']} | `{r['pass_name']}` | "
                     f"{r['g3_fp16_mma']} | {r['current_ordinal']} | `{r['current_spv']}` | "
                     f"{r['current_fp16_mma']} | {r['current_int8_mma']} | {r['status']} |")
        (out / "POSITION-MAP-129.md").write_text("\n".join(L) + "\n", encoding="utf-8",
                                                 newline="\n")
    print(json.dumps({"closed": len(closed), "open": len(rows) - len(closed),
                      "fp16_removed": doc["fp16_sites_removed"],
                      "current_positions": cur["passes_with_fp16"]}, indent=1))
    return 0


if __name__ == "__main__":
    sys.exit(main(sys.argv[1:]))
