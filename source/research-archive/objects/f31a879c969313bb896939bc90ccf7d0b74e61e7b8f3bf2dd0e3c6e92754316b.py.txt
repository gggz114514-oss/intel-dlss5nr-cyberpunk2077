"""n18b-f01-gather: reproduce the *parent* (current chain head, ordinal 0) SPV bytes.

Parent = cards/n18-f01-entry/candidate.spv (sha ecc9dd39...), i.e. the adopted
n18 candidate, which is what the chain-head plan (accepted/n18-f01-entry) binds
on PASS 0.  No candidate is built by this card (see RESUME): only the identity
of the bytes this analysis is anchored on is re-established here, with the same
recipe that n18 used (evidence/RECIPES.json[31c2fd9a].nc_recipe).

Usage:  python tools/build_parent.py
Writes: parent-rebuild.spv, parent-raw.spv, tools/parent-rebuild.json
"""
import sys, json
from pathlib import Path

CARD = Path(__file__).resolve().parents[1]
CAMPAIGN = CARD.parents[1]
sys.path.insert(0, str(CAMPAIGN))
import COMMON as K
import spv_tools_local as ST

ORD = 0
RECIPES = K.read_json(CAMPAIGN / 'evidence' / 'RECIPES.json')
RECIPE = RECIPES[K.row_by_ordinal(ORD)['spv_sha256']]
PARENT_SPV_FOR_NC = Path(RECIPE['nc_recipe']['parent'])   # f01.spv body: NC pattern source
SRC = CARD.parent / 'n18-f01-entry' / 'candidate.comp'    # chain-head source (ord 0)
EXPECTED = 'ecc9dd3921d8054a01ce2626739df006852b5690d1bd0f75a370c85fff280787'
INCLUDES = [Path(p) for p in RECIPE['compile']['include_dirs']]
PINNED = Path(RECIPE['compile']['pinned_include_dir'])


def includes_for(comp):
    dirs = [comp.parent, PINNED]
    for d in INCLUDES:
        if d not in dirs:
            dirs.append(d)
    for d in K.include_dirs_for(ORD, comp):
        if Path(d) not in dirs:
            dirs.append(Path(d))
    return dirs


def main():
    incs = includes_for(SRC)
    raw = CARD / 'parent-raw.spv'
    out = CARD / 'parent-rebuild.spv'
    K.compile_glsl(SRC, raw, incs)
    pm = ST.Module(PARENT_SPV_FOR_NC)
    cm = ST.Module(raw)
    keep, info = ST.transfer_decoration_pattern_aligned(pm, cm, set())
    ST.set_no_contraction(cm, keep, out)
    res = dict(
        card='n18b-f01-gather', role='parent(chain head, ord 0)',
        source=K.ident(SRC),
        nc_pattern_parent=str(PARENT_SPV_FOR_NC),
        includes=[str(p) for p in incs],
        raw=K.ident(raw), decorated=K.ident(out),
        nc_keep=len(keep), parent_nc=len(ST.no_contraction_ids(pm)),
        functions_mapped=info.get('functions_mapped'),
        child_own_decorations_outside_pattern=info.get('child_own_decorations_outside_pattern'),
        expected=EXPECTED,
        raw_match=K.sha(raw) == EXPECTED,
        decorated_match=K.sha(out) == EXPECTED,
    )
    K.write_json(CARD / 'tools' / 'parent-rebuild.json', res)
    print(json.dumps(res, indent=1))


if __name__ == '__main__':
    main()
