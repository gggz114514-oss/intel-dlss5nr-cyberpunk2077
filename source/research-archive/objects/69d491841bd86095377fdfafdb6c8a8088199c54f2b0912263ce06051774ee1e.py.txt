Each family entry carries:
  ordinal / name / kernel hash / baseline SPV + sha256
  ISA file + sha256 + the capture dir
  the cited site: line number L#, address, and the verbatim instruction text

  the verdict (worth a card or not), the deletable count, the widening option, risks

def sha(p):
    h = hashlib.sha256()
    with open(p, "rb") as f:
        for b in iter(lambda: f.read(1 << 20), b""):
            h.update(b)
    return h.hexdigest()


def resolve(o):
    for c in (H2F.get(str(o)) or []):
        if os.path.exists(c):
            return c


def line_of(path, addr):
    n = 0
    for n, raw in enumerate(open(path, encoding="utf-8", errors="replace"), 1):
        m = LINE_RE.match(raw)
        if m and m.group(1).upper() == addr.upper():
            return n, m.group(2).strip()
    return None, None


def fam_agg(fam):
    tot = {"narrow_kernel_total": 0, "messages": Counter(), "narrow": Counter(),
           "half_idioms": Counter(), "instr": [], "ordinals": [],
           "narrow_pk": [], "idioms_pk": []}
    for f in sorted(glob.glob(os.path.join(CARD, "_scan", "*.json"))):
        s = json.load(open(f))
        if s["family"] != fam:
            continue
        tot["ordinals"].append(s["ordinal"])
        tot["instr"].append(s["n_instr"])
        tot["narrow_pk"].append(sum(s["narrow_messages"].values()))
        tot["idioms_pk"].append(sum(s["stride_ops"].values()))
        tot["messages"].update(s["messages"])
        tot["narrow"].update(s["narrow_messages"])
        tot["half_idioms"].update(s["stride_ops"])
    return tot


def idiom_split(isa_path):
    """Exact split of the half-width register idioms (classify_idioms.py)."""
    try:
        import classify_idioms as CI
        c, _ = CI.classify(isa_path)
        return dict(c)
    except Exception as e:               # never fail the census on the optional split
        return {"error": str(e)}


def fam_addr(fam):
    tot = Counter()
    for f in sorted(glob.glob(os.path.join(CARD, "_addr", "*.json"))):
        d = json.load(open(f))
        if d.get("family") != fam:
            continue
        for k, v in d.items():
            if isinstance(v, int):
                tot[k] += v
    return {"sends": tot["sends"], "with_immediate": tot["with_immediate"],
            "same_addr_reg_as_prev": tot["same_addr_reg_as_prev"],
            "narrow": tot["narrow"], "narrow_with_immediate": tot["narrow_with_immediate"],
            "narrow_same_addr_reg": tot["narrow_same_addr_reg"]}


def main():
    verdicts = json.load(open(os.path.join(CARD, "_verdicts.json"), encoding="utf-8"))
    out = {"schema": "n15-narrow-read-census/v1",
           "generated": "2026-10-27",
           "card": "n15-narrow-read",
           "scope": "per-family census of narrow reads / per-half-byte assembly in the CURRENT native ISA; no GPU, no new capture, no candidate",
           "baseline": {"plan": IDX["baseline"]["plan"], "plan_sha256": IDX["baseline"]["plan_sha256"]},
           "method": {
               "isa_resolution": "evidence/NATIVE-INDEX.json baseline_kernel_hash per ordinal -> .asm file on disk with the same kernel hash (CURRENT identity)",
               "counts": "message mix and half-width register idioms counted from the disassembler text; line numbers are 1-based file lines, addresses are the ISA [hex] column",
               "not_claimed": ["bytes per message", "lane stride", "DRAM sector count"]},
           "families": []}

    for fam in sorted(REP):
        o = REP[fam]
        isa = resolve(o)
        sites, itxt = {}, None
        addr = SITE[fam][0]
        if addr:
            ln, itxt = line_of(isa, addr)
            sites["cited"] = {"addr": addr, "line": ln, "text": itxt, "proves": SITE[fam][1]}
        wkey = WIDE.get(fam)
        sc = json.load(open(os.path.join(CARD, "_scan", "%03d.json" % o)))
        if wkey and wkey in sc["message_sites"]:
            wa = sc["message_sites"][wkey][0]
            ln, txt = line_of(isa, wa)
            sites["wide_reference"] = {"message": wkey.replace("ugm.", ".ugm."), "addr": wa,
                                       "line": ln, "text": txt,
                                       "count_kernel": sc["messages"].get(wkey, 0)}
        agg = fam_agg(fam)
        v = verdicts[fam]
        out["families"].append({
            "family": fam,
            "pass_count": len(agg["ordinals"]),
            "ordinals": agg["ordinals"],
            "representative_ordinal": o,
            "name": sc["name"],
            "kernel_hash": sc["kernel_hash"],
            "baseline_spv": {"path": sc["baseline_spv"], "sha256": sc["baseline_spv_sha256"]},
            "isa": {"path": isa, "sha256": sha(isa),
                    "capture_dir": os.path.dirname(os.path.dirname(isa))},
            "site": sites,
            "profile_representative_kernel": {
                "instructions": sc["n_instr"],
                "messages": sc["messages"],
                "narrow_messages": sc["narrow_messages"],
                "narrow_total": sum(sc["narrow_messages"].values()),
                "half_idioms": sc["stride_ops"],
                "half_idiom_total": sum(sc["stride_ops"].values()),
                "half_idiom_split": idiom_split(isa),
            },
            "profile_family": {
                "narrow_total_all_pass": sum(agg["narrow"].values()),
                "narrow_by_kind": dict(agg["narrow"]),
                "half_idioms_all_pass": sum(agg["half_idioms"].values()),
                "instructions_min": min(agg["instr"]), "instructions_max": max(agg["instr"]),
                "narrow_per_kernel_min": min(agg["narrow_pk"]), "narrow_per_kernel_max": max(agg["narrow_pk"]),
                "idioms_per_kernel_min": min(agg["idioms_pk"]), "idioms_per_kernel_max": max(agg["idioms_pk"]),
                "kernels_with_narrow": sum(1 for x in agg["narrow_pk"] if x > 0),
            },
            "address_layer": fam_addr(fam),
            "verdict": v["verdict"],
            "worth_card": v["worth_card"],
            "deletable": v["deletable"],
            "widening_option": v["widening"],
            "risk": v["risk"],
            "prior_evidence": v["prior"],
            "notes": v.get("notes", ""),
        })
    with open(os.path.join(CARD, "NARROW-READ-CENSUS.json"), "w", encoding="utf-8") as fh:
        json.dump(out, fh, indent=2, ensure_ascii=False)
    print("wrote NARROW-READ-CENSUS.json with %d families" % len(out["families"]))
    print("worth_card:", sum(1 for f in out["families"] if f["worth_card"]))


if __name__ == "__main__":
    main()
