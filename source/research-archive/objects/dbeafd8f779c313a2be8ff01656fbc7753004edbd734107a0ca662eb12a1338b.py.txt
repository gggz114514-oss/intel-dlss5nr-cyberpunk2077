"""Idempotent post-pass over evidence/NATIVE-INDEX.json:

Reject `driver_stats` that cannot belong to the pipeline's current ISA.  The driver dump is
taken from a `pipeline-statistics.json` found in the same card, which may be the *candidate*
arm of that card; the driver's Instruction Count tracks the ISA's static instruction lines to
within 3 in every case checked, so a larger delta means the dump describes other code.

Same rule as build_index.py, applied without rescanning 4.4 GB of ISA.
"""
import json
from pathlib import Path

CARD = Path(__file__).parent
P = CARD / 'evidence' / 'NATIVE-INDEX.json'
TOL = 16

j = json.loads(P.read_text(encoding='utf-8'))
rejected = kept = 0
for c in j['captures']:
    for e in c['records']:
        st, scan = e.get('driver_stats'), (e.get('isa_scan') or {})
        if not st or not scan.get('instruction_lines'):
            continue
        di = st.get('Instruction Count')
        if di is None:
            continue
        if abs(di - scan['instruction_lines']) > TOL:
            e['driver_stats_rejected'] = dict(source=st.get('source'), instruction_count=di,
                                              static_instruction_lines=scan['instruction_lines'])
            e['driver_stats'] = None
            rejected += 1
        else:
            kept += 1

j['summary']['driver_stats_records'] = kept
j['summary']['driver_stats_rejected'] = rejected
j['postprocess'] = ('cards/nat-index/post_process.py: driver stats rejected when '
                    '|InstructionCount - static instruction lines| > %d' % TOL)
P.write_text(json.dumps(j, indent=1, ensure_ascii=False), encoding='utf-8')
print('driver stats kept %d / rejected %d' % (kept, rejected))
print('wrote', P, '%.1f MB' % (P.stat().st_size / 1e6))
