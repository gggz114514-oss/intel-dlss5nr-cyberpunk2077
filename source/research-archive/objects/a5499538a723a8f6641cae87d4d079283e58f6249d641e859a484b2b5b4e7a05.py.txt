import difflib
from pathlib import Path
parent = Path(r'@NR_WORKSPACE@/candidates/amd-netorg-full-scope-workbuddy-20261017/c2-layout/layout0-unify-l2-v4/source/projection-swap-family-24/split_projection_swap.comp')
