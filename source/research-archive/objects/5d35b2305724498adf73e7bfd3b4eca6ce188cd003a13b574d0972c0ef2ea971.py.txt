#!/usr/bin/env python3
"""Apply the 2026-10-24 audit corrections inside the package.

1. C2: record the scale-pack baseline correction, point CANDIDATE.json/jobs at the
   G3-rebuilt plan, and retract the parent-side "saturation" observation.
2. A2: make CANDIDATE.json point at the ruled `plan-a2-ups3-229-fullpush.txt`
   instead of the superseded word17-only plan, and mark it not-numerically-verified.
"""
import hashlib
import json
from pathlib import Path

W = Path(r"@NR_WORKSPACE@/work/dsh-xmx-handoff-validation-20261024")
ROOT = Path(r"@NR_WORKSPACE@")

G3_PACK = ROOT / "candidates/int8-coverage-v2-20261020/C/c32-scale-from-x-v2/pack/c32_b1_scales.f32.bin"
DIAG_PACK = ROOT / "candidates/int8-coverage-v2-20261020/C/pack/c32_b1_scales.f32.bin"


def sha(p):
    return hashlib.sha256(Path(p).read_bytes()).hexdigest()


def patch_c2():
    card = W / "candidates/c2-c32-regk32"
    plan = card / "plan-c2-regk32-g3.txt"
    doc = json.loads((card / "CANDIDATE.json").read_text(encoding="utf-8"))
    doc["baseline_correction"] = {
        "date": "2026-10-24",
        "what": "the first C2 plan was derived from the diagnostic plan "
                "plan-c32-i8-diag-b1-eact.txt, which binds the OLD scale pack; it is not the "
                "current G3 baseline",
        "scale_packs": {
            "current_g3": {"path": str(G3_PACK), "sha256": sha(G3_PACK),
                           "sS_times_invS": 1.00000002,
                           "sE_times_invE": 1.00000004,
                           "verdict": "corrected: invS == 1/sS"},
            "old_diagnostic": {"path": str(DIAG_PACK), "sha256": sha(DIAG_PACK),
                               "sS_times_invS": 127.000002,
                               "sE_times_invE": 127.000009,
                               "verdict": "diagnostic-only; must not be used as the verification "
                                          "baseline"},
        },
        "retraction": "the earlier 'parent saturates almost everything to +-127' observation was "
                      "measured on the old diagnostic pack. On the current G3 pack the product is "
                      "1.00000002, so the observation does NOT apply to the G3 parent and no "
                      "parent-line change is needed.",
        "rebuilt_plan": {"path": str(plan), "sha256": sha(plan), "passes": 287,
                         "changed_passes": 7,
                         "change": "spv c32_body1_i8.spv -> c2_body1_regk32.spv and shared "
                                   "18432 -> 10240 for the seven body1 passes; everything else "
                                   "byte-identical to G3"},
        "old_plan_status": "plan-c2-regk32-323.txt stays on disk as a diagnostic-baseline record "
                           "and MUST NOT be used for the G3 numeric arm",
    }
    (card / "CANDIDATE.json").write_text(json.dumps(doc, indent=1) + "\n",
                                         encoding="utf-8", newline="\n")
    note = card / "BASELINE-CORRECTION.md"
    note.write_text(
        "# C2 基线更正（2026-10-24，主审计复核要求）\n\n"
        "C2 初版计划来自诊断计划 `C/c32-i8-debug-eact-v1/plans/plan-c32-i8-diag-b1-eact.txt`，\n"
        "它绑的是**旧诊断 scale pack**；当前 G3 绑的是**修正版 pack**。两者同名不同目录，数值差 127 倍。\n\n"
        "| 基线 | 文件 | sha256 | sS×invS | sE×invE |\n"
        "|---|---|---|---:|---:|\n"
        f"| **当前 G3（正确）** | `C/c32-scale-from-x-v2/pack/c32_b1_scales.f32.bin` | `{sha(G3_PACK)[:16]}…` | **1.00000002** | **1.00000004** |\n"
        f"| 旧诊断（不可用） | `C/pack/c32_b1_scales.f32.bin` | `{sha(DIAG_PACK)[:16]}…` | 127.000002 | 127.000009 |\n\n"
        "**撤回**：C2 报告里\"父代 |E| ≥ sE 即被夹到 ±127\"的观察只对旧诊断 pack 成立；\n"
        "当前 G3 pack 的乘积是 1.00000002（即 invS = 1/sS），**不成立**，也不需要改父线。\n\n"
        "**因此**：C2 的验证身份已重建为 `plan-c2-regk32-g3.txt`（287 PASS，只改 7 个 body1 PASS 的\n"
        "`spv` 与 `shared`，其余与 G3 逐字节相同；`plan_lint ok=true`、`validate_pool_upload PASS`）。\n"
        "旧 `plan-c2-regk32-323.txt` 留在盘上只作诊断基线记录，**不得用于 G3 数值臂**。\n",
        encoding="utf-8", newline="\n")
    jobs = W / "run/c2-c32-regk32/jobs.json"
    jd = json.loads(jobs.read_text(encoding="utf-8"))
    for job in jd.get("jobs", []):
        for key in ("plan",):
            if isinstance(job.get(key), str):
                job[key] = str(plan)
        args = job.get("args")
        if isinstance(args, list):
            job["args"] = [str(plan) if isinstance(a, str) and a.endswith(".txt") and "plan" in a.lower()
                           else a for a in args]
        job["expected_passes"] = 287
    jd["plan"] = str(plan)
    jd.setdefault("notes", []).append(
        "2026-10-24 baseline correction: arms now run on plan-c2-regk32-g3.txt (current G3 pack "
        "sha ecdcea95..., sS*invS=1.00000002); the 323-pass diagnostic plan is retired for gates.")
    jobs.write_text(json.dumps(jd, indent=1, ensure_ascii=False) + "\n", encoding="utf-8", newline="\n")
    print("C2 patched:", sha(plan)[:16], "jobs", len(jd["jobs"]))


def patch_a2():
    card = W / "candidates/a2-ups3-body-contract"
    jp = card / "plan-a2-ups3-229-fullpush.txt"
    wp = card / "plan-a2-ups3-229-word17only.txt"
    doc = json.loads((card / "CANDIDATE.json").read_text(encoding="utf-8"))
    repl = {
        "candidate_plan": str(jp),
        "candidate_plan_sha256": sha(jp),
        "passes": 229,
        "identity_note": "DSH ruling: the candidate arm is the FULL 32-word contract PUSH copy. "
                         "plan-a2-ups3-229.txt (word17-only) is superseded and kept as an "
                         "alternative; plan-a2-ups3-strict.txt is unusable.",
        "alternative_plan": {"path": str(wp), "sha256": sha(wp),
                             "status": "superseded alternative, not the candidate arm"},
        "numeric_verification": "NOT RUN - no GPU slot; no gate and no ordinary were executed "
                                "for this card",
    }
    # update any field that names the old plan
    def walk(o):
        if isinstance(o, dict):
            for k, v in list(o.items()):
                if isinstance(v, str) and "plan-a2-ups3-229.txt" in v:
                    o[k] = str(jp)
                elif isinstance(v, dict):
                    walk(v)
                elif isinstance(v, list):
                    for i, x in enumerate(v):
                        if isinstance(x, str) and "plan-a2-ups3-229.txt" in x:
                            v[i] = str(jp)
                        elif isinstance(x, (dict, list)):
                            walk(x)
        elif isinstance(o, list):
            for x in o:
                walk(x)
    walk(doc)
    doc["ruling_applied"] = repl
    (card / "CANDIDATE.json").write_text(json.dumps(doc, indent=1, ensure_ascii=False) + "\n",
                                         encoding="utf-8", newline="\n")
    print("A2 patched ->", sha(jp)[:16])


if __name__ == "__main__":
    patch_c2()
    patch_a2()
