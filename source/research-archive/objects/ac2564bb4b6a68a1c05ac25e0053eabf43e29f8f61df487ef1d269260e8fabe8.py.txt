"""Census of the *current baseline* C128 native (kv-consumer-layout family, ordinal 19).

Confirms on the baseline identity (not the older rolling dump) that the A supply is
8 x 32 B row sends + 4 same-width 32 B packing copies per 8-row fragment, and that
the six-accumulator QKV loop carries 48 weight returns per phase.
"""
import json, re, collections
from pathlib import Path

CARD = Path(r'@NR_WORKSPACE@\work\codex-astra-fp16-int8-audit-20261024\fullnet-continuous-20261027\cards\n04b-f05-c128-feed')
D = Path(r'@NR_WORKSPACE@\work\codex-astra-fp16-int8-audit-20261024\fullnet-mechanism-transfer-20261026')
ISA = D / 'f05-kv-consumer-layout' / 'family' / 'captures' / 'native-f20-pixel-owner' / 'ISA'
target = ISA / 'CS_asmf98d5ab4e239dd54_simd32_000000000000000a_9_orig.asm'
assert target.exists(), target
lines = [l for l in target.read_text(encoding='utf-8', errors='replace').splitlines() if l.strip()]

M = lambda s: re.search(s, s)
counts = collections.Counter()
a_rows = 0
a_rows_before_qkv_end = 0
weight_sends = 0
dpas = 0
pack_movs = 0
addr = collections.Counter()
dpas_pcs = []
send_pcs = []
for i, l in enumerate(lines):
    body = re.sub(r'^\s*/\* \[([0-9A-Fa-f]+)\]\s*\*/\s*', r'\1 ', l)
    pc = body.split(' ', 1)[0]
    if 'send.ugm' in l:
        counts['send'] += 1
        send_pcs.append(pc)
        if 'd32x8t' in l:
            a_rows += 1
        elif 'rd:8' in l:
            weight_sends += 1
    if 'dpas' in l:
        dpas += 1
        dpas_pcs.append(pc)
    if re.search(r'\bmov \(8\|M0\)\s+r\d+\.8<1>:ud\s+r\d+\.0<1;1,0>:ud', l):
        pack_movs += 1
    if 'mov' in l:
        counts['mov'] += 1
    m = re.search(r'\((1|8|16|32)\|M0\)\s+(add3?|shl|mul|or)\b', l)
    if m:
        addr[(m.group(1), m.group(2))] += 1
    if 'sync.' in l:
        counts['sync'] += 1

out = dict(native=str(target), sha256=None, total_instructions=len(lines),
           sends=counts['send'], movs=counts['mov'], dpas=dpas, syncs=counts['sync'],
           a_row_sends_d32x8t=a_rows, weight_sends_rd8=weight_sends,
           a_pack_movs_upper_half=pack_movs,
           addr_ops=dict(sorted((f'{k[0]}|{k[1]}', v) for k, v in addr.items())),
           addr_ops_total=sum(addr.values()),
           first_dpas=dpas_pcs[0] if dpas_pcs else None,
           last_dpas=dpas_pcs[-1] if dpas_pcs else None,
           first_send=send_pcs[0] if send_pcs else None)
print(json.dumps(out, indent=1))
(CARD / 'build' / 'baseline-native-census.json').write_text(json.dumps(out, indent=1), encoding='utf-8')
