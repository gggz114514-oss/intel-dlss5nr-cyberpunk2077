"""S12: run spvprobe over every unique SPV in the current candidate plan.

Outputs evidence/spv-probe-all.json (all 60 unique SPVs) and a per-target summary.
"""
import json, sys, os
from pathlib import Path

sys.path.insert(0, r"@NR_WORKSPACE@/work/dsh-fullnet-static-20261029/tools")
sys.path.insert(0, r"@NR_WORKSPACE@/work/dsh-fullnet-static-20261029/S12-proj/_work")
import planlib
import spvprobe

PLAN = r"@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/int8-quality-first-20261029/real-int8-24/c64-family-static/combo-34189.txt"
OUT = r"@NR_WORKSPACE@/work/dsh-fullnet-static-20261029/S12-proj/evidence/spv-probe-all.json"

p = planlib.Plan(PLAN)
uniq = {}
for ps in p.passes:
    uniq.setdefault(ps.spv, []).append(ps.ordinal)
print("unique spv:", len(uniq))
res = {}
for spv, ords in uniq.items():
    try:
        r = spvprobe.scan(Path(spv))
    except Exception as e:  # noqa: BLE001
        r = {"error": f"{type(e).__name__}: {e}"}
    r["ordinals"] = sorted(ords)
    r["basename"] = os.path.basename(spv)
    res[spv.replace("\\", "/")] = r
json.dump(res, open(OUT, "w", encoding="utf-8"), indent=1)

nf = [(k, v) for k, v in res.items() if v.get("mma_fp16")]
print("SPVs with any FP16 coopmat:", len(nf))
for k, v in nf:
    print("  ", v["basename"], v["mma_fp16"], v["sha256"][:16])
ni8 = sum(1 for v in res.values() if v.get("mma_int8"))
print("SPVs with INT8 coopmat:", ni8)
print("\n%-34s %6s %6s %6s %6s %6s %6s %6s %6s" % (
    "basename", "int8", "fp16", "fdiv", "div127", "mul127", "FAbs", "FMax", "127c"))
for k, v in sorted(res.items(), key=lambda kv: kv[1].get("basename", "")):
    if not v.get("mma_int8"):
        continue
    e = v.get("ext", {})
    print("%-34s %6d %6d %6d %6d %6d %6d %6d %6s" % (
        v["basename"][:34], v["mma_int8"], v["mma_fp16"], v["opfdiv"], v["div_by_const127"],
        v["mul_by_inv127"], e.get("FAbs", 0), e.get("FMax", 0), v["has_float_127"]))
