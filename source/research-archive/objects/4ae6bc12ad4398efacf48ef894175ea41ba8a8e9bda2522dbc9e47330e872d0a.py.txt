#!/usr/bin/env python3
"""Independent FP16/INT8 matrix-multiply counter for SPIR-V modules.

Authoritative operand grammar (calibrated against the audit's
`FULLNET-RESIDUAL-MMA-SITES.json`: at the listed word offsets the opcode is 4459 and
the listed result id is operand 1):

  4456 OpTypeCooperativeMatrixKHR   [result, component, scope, rows, cols, use]
  4457 OpCooperativeMatrixLoadKHR   [type, result, pointer, layout, stride, memop]
  4458 OpCooperativeMatrixStoreKHR  [pointer, object, layout, stride, memop]   (no result)
  4459 OpCooperativeMatrixMulAddKHR [type, result, A, B, C]

A MulAdd's result type is the accumulator type: `f32` means FP16 operands (FP16 matrix
multiply), `i32` means INT8 operands.  That is exactly the ABC classification the audit
records, so this tool is an independent second opinion, not a copy of it.

Usage:
  verify_mma_dtype.py <spv> [<spv> ...]
  verify_mma_dtype.py --json <out.json> <spv> ...
"""
import hashlib
import json
import sys
from collections import Counter
from pathlib import Path

sys.path.insert(0, r"@NR_WORKSPACE@/build/zcode-fp16-compute-backend-20261021/scripts")
import spv_tools_local as S  # noqa: E402


def rid_of(op, ops):
    """Result id with the cooperative-matrix opcodes corrected (see module docstring)."""
    if op == 4458:
        return None
    if op == 4456:
        return ops[0]
    if op in (4457, 4459, 4460, 5358):
        return ops[1]
    return S.result_id_of(op, ops)


def analyse(path):
    m = S.Module(Path(path))
    by = {}
    for idx, (op, ops, _, _) in enumerate(m.instructions):
        r = rid_of(op, ops)
        if r is not None:
            by.setdefault(r, idx)

    def kind(tid, depth=0):
        if depth > 4:
            return "?"
        i = by.get(tid)
        if i is None:
            return "?"
        op, ops = m.instructions[i][0], m.instructions[i][1]
        if op == 4456:                      # cooperative matrix type
            c = by.get(ops[1])
            if c is None:
                return "?"
            cop, cops = m.instructions[c][0], m.instructions[c][1]
            if cop == 22:                   # OpTypeFloat
                return "f%d" % cops[1]
            if cop == 21:                   # OpTypeInt
                return "i%d" % cops[1]
            return "?"
        if op == 32:                        # OpTypePointer -> pointee
            return kind(ops[2], depth + 1)
        return "?"

    kinds = Counter()
    sites = []
    for idx, (op, ops, word_off, _) in enumerate(m.instructions):
        if op == 4459:
            k = kind(ops[0])
            kinds[k] += 1
            sites.append({"word_offset": word_off, "result_id": ops[1], "accumulator": k})
    fp16 = kinds.get("f32", 0)
    int8 = kinds.get("i32", 0)
    return {
        "path": str(path),
        "sha256": hashlib.sha256(Path(path).read_bytes()).hexdigest(),
        "bytes": Path(path).stat().st_size,
        "instructions": len(m.instructions),
        "muladd_total": sum(kinds.values()),
        "muladd_by_accumulator": dict(kinds),
        "fp16_matrix_multiplies": fp16,
        "int8_matrix_multiplies": int8,
        "verdict": ("FP16_FREE" if fp16 == 0 and int8 > 0 else
                    ("NO_MMA" if not kinds else "STILL_FP16")),
        "sites": sites,
    }


def main(argv):
    out = None
    if argv and argv[0] == "--json":
        out = Path(argv[1])
        argv = argv[2:]
    if not argv:
        raise SystemExit(__doc__)
    res = [analyse(p) for p in argv]
    for r in res:
        print(f"{Path(r['path']).name:42} muladd={r['muladd_total']:3d} "
              f"fp16={r['fp16_matrix_multiplies']:3d} int8={r['int8_matrix_multiplies']:3d} "
              f"{r['verdict']}")
    if out:
        out.write_text(json.dumps({"tool": "tools/verify_mma_dtype.py", "modules": res},
                                  indent=1) + "\n", encoding="utf-8", newline="\n")
        print("wrote", out)
    return 0 if all(r["fp16_matrix_multiplies"] == 0 for r in res) else 1


if __name__ == "__main__":
    sys.exit(main(sys.argv[1:]))
