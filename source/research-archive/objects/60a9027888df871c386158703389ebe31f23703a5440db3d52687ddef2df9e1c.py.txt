def name(op):
    return NAMED.get(op, 'Op%d' % op)


def analyse(path):
    m = ST.Module(Path(path))
    hist = {}
    for op, ops, _, _ in m.instructions:
        hist[name(op)] = hist.get(name(op), 0) + 1
    # storage classes: OpVariable storage-class operand (SPIR-V: OpVariable result type, result id, storage class)
    storage = {}
    for op, ops, _, _ in m.instructions:
        if op == 59 and len(ops) >= 3:
            sc = ops[2]
            storage[sc] = storage.get(sc, 0) + 1
    sc_name = {0: 'UniformConstant', 1: 'Input', 2: 'Uniform', 3: 'Output',
               4: 'Workgroup', 5: 'CrossWorkgroup', 6: 'Private', 7: 'Function',
               8: 'Generic', 9: 'PushConstant', 10: 'AtomicCounter', 11: 'Image',
               12: 'StorageBuffer'}
    storage = {sc_name.get(k2, str(k2)): v for k2, v in sorted(storage.items())}
    dec_hist = {}
    nc_ids = set()
    for op, ops, _, _ in m.instructions:
        if op == 71 and len(ops) >= 2:
            dec_hist[ops[1]] = dec_hist.get(ops[1], 0) + 1
            if ops[1] == 42:                          # Decoration NoContraction
                nc_ids.add(ops[0])
    id2 = m.id_to_instruction()
    nc_sites = {}
    for i in sorted(nc_ids):
        op, ops = id2.get(i, (None, None))
        if op is None:
            nc_sites['<undef:%d>' % i] = nc_sites.get('<undef:%d>' % i, 0) + 1
            continue
        nc_sites[name(op)] = nc_sites.get(name(op), 0) + 1
    nc_float = sum(v for k2, v in nc_sites.items() if k2.startswith('OpF') or k2 in ('OpExtInst',))
    nc_int = sum(v for k2, v in nc_sites.items() if k2 in set(OPNAME.values()) | CONTROL_FLOW_NAMES)
    return dict(path=str(path), sha256=K.sha(path), words=len(m.instructions),
                bound=m.bound, op_histogram=dict(sorted(hist.items())),
                storage_variables=storage,
                workgroup_variables=storage.get('Workgroup', 0),
                control_barriers=hist.get('OpControlBarrier', 0),
                memory_barriers=hist.get('OpMemoryBarrier', 0),
                coop_load=hist.get('OpCooperativeMatrixLoadKHR', 0),
                coop_store=hist.get('OpCooperativeMatrixStoreKHR', 0),
                coop_muladd=hist.get('OpCooperativeMatrixMulAddKHR', 0),
                no_contraction_total=len(nc_ids),
                no_contraction_sites=dict(sorted(nc_sites.items())),
                no_contraction_float=nc_float,
                decoration_histogram={str(k2): v for k2, v in sorted(dec_hist.items())},
                no_contraction_ids=sorted(nc_ids))


CONTROL_FLOW_NAMES = {OPNAME[k2] for k2 in CONTROL_FLOW}


def workgroup_bytes(path):
    """Byte size of every Workgroup-storage array in the module."""
    m = ST.Module(Path(path))
    id2 = m.id_to_instruction()
    # constant values
    consts = {}
    for op, ops, _, _ in m.instructions:
        if op == 43 and len(ops) >= 3:      # OpConstant result-type, result-id, value
            consts[ops[1]] = ops[2]
        if op == 44 and len(ops) >= 3:
            consts[ops[1]] = [consts.get(x) for x in ops[2:]]
    # pointee -> element type
    ptr = {}
    for op, ops, _, _ in m.instructions:
        if op == 32 and len(ops) >= 3:      # OpTypePointer storage, type
            ptr[ops[0]] = (ops[1], ops[2])
    arr = {}
    for op, ops, _, _ in m.instructions:
        if op == 28 and len(ops) >= 3:      # OpTypeArray element, length-id
            arr[ops[0]] = (ops[1], ops[2])
    total = 0
    detail = []
    for op, ops, _, _ in m.instructions:
        if op == 59 and len(ops) >= 3 and ops[2] == 4:   # Workgroup
            sc, pointee = ptr.get(ops[0], (None, None))
            if pointee in arr:
                elem, length_id = arr[pointee]
                n = consts.get(length_id)
                # element size: half = 2, float = 4
                esz = 2
                # find OpTypeFloat width
                for op2, ops2, _, _ in m.instructions:
                    if op2 == 22 and ops2[0] == elem:
                        esz = ops2[1] // 8
                detail.append(dict(var=ops[1], elements=n, element_bytes=esz,
                                   bytes=(n or 0) * esz))
                total += (n or 0) * esz
    return total, detail


def main():
    out = {}
    # 1 ---- parent byte reproduction from the frozen source
    assert K.sha(PARENT_SRC) == PARENT_SRC_SHA, 'parent source drifted'
    inc = SRC_INC
    K.compile_glsl(PARENT_SRC, BUILD / 'parent-raw.spv', [inc])
    parent_raw_sha = K.sha(BUILD / 'parent-raw.spv')
    assert parent_raw_sha == GROUP_SHA, ('parent not reproduced', parent_raw_sha)
    shutil.copyfile(BUILD / 'parent-raw.spv', CARD / 'parent-rebuild.spv')
    out['parent'] = dict(spv=K.ident(PARENT_SPV), source=K.ident(PARENT_SRC),
                         rebuilt=K.ident(CARD / 'parent-rebuild.spv'),
                         recipe='raw', defines=[], include_dirs=[str(inc)],
                         raw_match=True,
                         pinned_include_dir=str(PINNED_INC),
                         pinned_include_sha=K.sha(PINNED_INC / 'split16_common.glsl') if (PINNED_INC / 'split16_common.glsl').exists() else None,
                         source_include_sha=K.sha(SRC_INC / 'split16_common.glsl'))
    # 2 ---- candidate
    K.compile_glsl(CARD / 'candidate.comp', CARD / 'candidate.spv', [inc])
    K.validate_spv(CARD / 'candidate.spv')
    K.validate_spv(CARD / 'parent-rebuild.spv')
    out['candidate'] = dict(source=K.ident(CARD / 'candidate.comp'),
                            spv=K.ident(CARD / 'candidate.spv'),
                            validate='result=0')
    # 3 ---- plan
    n = K.make_plan(CARD / 'candidate.spv', CARD / 'plan.txt',
                    spv_shas={GROUP_SHA})
    out['plan'] = dict(path=str(CARD / 'plan.txt'), sha256=K.sha(CARD / 'plan.txt'),
                       replaced_passes=n,
                       parent_plan=K.ident(K.BASELINE))
    # 4 ---- CPU checks
    base_check = K.cpu_check(K.BASELINE)
    cand_check = K.cpu_check(CARD / 'plan.txt')
    out['cpu_check'] = dict(baseline=base_check.strip(), candidate=cand_check.strip(),
                            identical=base_check.strip() == cand_check.strip())
    # 5 ---- SPIR-V analysis
    P = analyse(CARD / 'parent-rebuild.spv')
    Q = analyse(CARD / 'candidate.spv')
    pw, pdetail = workgroup_bytes(CARD / 'parent-rebuild.spv')
    qw, qdetail = workgroup_bytes(CARD / 'candidate.spv')
    out['spirv'] = dict(
        parent=dict(words=P['words'], workgroup_variables=P['workgroup_variables'],
                    workgroup_bytes=pw, workgroup_detail=pdetail,
                    control_barriers=P['control_barriers'],
                    coop_load=P['coop_load'], coop_store=P['coop_store'],
                    coop_muladd=P['coop_muladd'],
                    no_contraction_total=P['no_contraction_total'],
                    no_contraction_sites=P['no_contraction_sites'],
                    storage_variables=P['storage_variables'],
                    op_histogram=P['op_histogram']),
        candidate=dict(words=Q['words'], workgroup_variables=Q['workgroup_variables'],
                       workgroup_bytes=qw, workgroup_detail=qdetail,
                       control_barriers=Q['control_barriers'],
                       coop_load=Q['coop_load'], coop_store=Q['coop_store'],
                       coop_muladd=Q['coop_muladd'],
                       no_contraction_total=Q['no_contraction_total'],
                       no_contraction_sites=Q['no_contraction_sites'],
                       storage_variables=Q['storage_variables'],
                       op_histogram=Q['op_histogram']),
        delta=dict(words=Q['words'] - P['words'],
                   workgroup_variables=Q['workgroup_variables'] - P['workgroup_variables'],
                   workgroup_bytes=qw - pw,
                   control_barriers=Q['control_barriers'] - P['control_barriers'],
                   coop_load=Q['coop_load'] - P['coop_load'],
                   coop_store=Q['coop_store'] - P['coop_store'],
                   coop_muladd=Q['coop_muladd'] - P['coop_muladd'],
                   no_contraction_total=Q['no_contraction_total'] - P['no_contraction_total']))
    # float NoContraction correspondence (contract: one-to-one with the parent)
    pf = {k2: v for k2, v in P['no_contraction_sites'].items() if k2.startswith('OpF') or k2 == 'OpExtInst'}
    qf = {k2: v for k2, v in Q['no_contraction_sites'].items() if k2.startswith('OpF') or k2 == 'OpExtInst'}
    out['nc'] = dict(float_parent=sum(pf.values()), float_candidate=sum(qf.values()),
                     float_sites_parent=pf, float_sites_candidate=qf,
                     float_sites_one_to_one=pf == qf)
    out['float_NC_checked'] = sum(pf.values())
    out['float_NC_one_to_one'] = pf == qf
    # 6 ---- group rows
    rows = [r for r in K.rows() if r['spv_sha256'] == GROUP_SHA]
    out['group'] = dict(spv_sha256=GROUP_SHA, passes=len(rows),
                        ordinals=[r['ordinal'] for r in rows],
                        names=sorted({r['name'] for r in rows}),
                        grid=rows[0]['grid'], header_fields=rows[0]['header_fields'])
    K.write_json(CARD / 'BUILD.json', out)
    print(json.dumps({k2: (v if not isinstance(v, dict) else '<dict>') for k2, v in out.items()},
                     indent=2, ensure_ascii=False))
    print('parent rebuild sha ', out['parent']['rebuilt']['sha256'])
    print('candidate spv sha  ', out['candidate']['spv']['sha256'])
    print('plan sha           ', out['plan']['sha256'], 'replaced', n)
    print('cpu-check          ', cand_check.strip())
    print('workgroup bytes    ', pw, '->', qw, ' barriers', P['control_barriers'], '->', Q['control_barriers'])
    print('words              ', P['words'], '->', Q['words'])
    print('float_NC           ', out['nc']['float_parent'], '->', out['nc']['float_candidate'],
          'one_to_one', out['nc']['float_sites_one_to_one'])
    print('NC sites parent    ', P['no_contraction_sites'])
    print('NC sites candidate ', Q['no_contraction_sites'])


if __name__ == '__main__':
    main()
