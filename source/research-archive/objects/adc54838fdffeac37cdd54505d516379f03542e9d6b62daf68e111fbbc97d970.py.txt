#!/usr/bin/env python3
"""verify the local opcode map against the two modules in play (CPU only)"""
import sys
from collections import Counter
from pathlib import Path

HERE = Path(__file__).resolve().parent
sys.path.insert(0, str(HERE))
import spvlib  # noqa: E402

CARD = HERE.parent
MODS = [CARD / "spv/parent_repro-c64_output.spv", CARD / "spv/r07_c64_output_wo_i8.spv"]
OPNUM = {v: k for k, v in spvlib.OP.items()}
for path in MODS:
    m = spvlib.Mod(path)
    c = Counter(it.op for it in m.insts)
    print("==", path.name, "words", len(m.w), "insts", len(m.insts))
    unknown = sorted(k for k in c if k not in spvlib.OP)
    print("   unknown opcodes:", [(k, c[k]) for k in unknown])
    for k in unknown:
        ex = [it for it in m.insts if it.op == k][0]
        print("     op %d x%d  example operands=%s  resulttype=%s" % (
            k, c[k], ex.a, m.types.get(ex.a[0]) if ex.a else None))
    # sanity: prove the shift/mask opcodes by their operand shapes
    for name in ("OpShiftRightLogical", "OpBitwiseAnd", "OpIEqual", "OpUDiv",
                 "OpUMod", "OpULessThan", "OpConvertSToF", "OpCompositeConstruct"):
        op = OPNUM.get(name)
        print("   %-22s opcode=%-4s count=%d" % (name, op, c.get(op, 0)))
