"""n06-f06-output: exhaustive owner/address/K-order proof (CPU only).

Models the parent and candidate C128 F06 kernel at the (token, channel) element level
and proves: one owner per output element, identical store address, identical seed tile,
identical per-output K accumulation sequence, identical half rounding points.

No GPU, no Vulkan: pure integer/sets over the launch geometry.
"""
import json
from pathlib import Path

CARD = Path(r'@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/fullnet-continuous-20261027/cards/n06-f06-output')
TOK = 64          # tokens per workgroup (w*64 + m0)
CH = 128          # channels per workgroup (C128)
KOFF = [16 * i for i in range(8)]


def img_half_index(t, c, wx=3, wy=2, tiles=(21, 12), shift=(0, 0)):
    """Verbatim c128_img_half_index from the pinned header (neutral wx/wy)."""
    q, s = t >> 4, t & 15
    tx = 2 * wx + shift[0] + (q & 1)
    ty = 2 * wy + shift[1] + (q >> 1)
    oob = tx < 0 or ty < 0 or tx >= tiles[0] or ty >= tiles[1]
    stx = min(max(tx, 0), tiles[0] - 1)
    sty = min(max(ty, 0), tiles[1] - 1)
    return (((sty * tiles[0] + stx) * 8 + c // 16) * 256 + s * 16 + c % 16), oob


def parent_tiles():
    """(sg, j, nblk, m0) tiles of the parent: 8 SG x 4 j x 2 n-blocks."""
    out = []
    for sg in range(8):
        n0 = (sg % 4) * 32
        for j in range(4):
            m0 = (sg // 4 + 2 * j) * 8
            for h in (0, 1):
                out.append(dict(sg=sg, j=j, jp=None, blk=None, m0=m0,
                                nblk=n0 + 16 * h, n0=n0, h=h,
                                acc='acc%d' % h))
    return out


def cand_tiles():
    """(sg, jp, block, nblk) tiles of the candidate: 8 SG x 2 jp x 2 blocks x 2 n-blocks."""
    out = []
    for sg in range(8):
        n0 = (sg % 4) * 32
        for jp in range(2):
            mbase = (sg // 4) * 32 + 16 * jp
            for bi, m0 in enumerate((mbase, mbase + 8)):
                for h in (0, 1):
                    tag = {0: 0, 1: 2}[bi] + h
                    out.append(dict(sg=sg, j=None, jp=jp, blk=bi, m0=m0,
                                    nblk=n0 + 16 * h, n0=n0, h=h,
                                    acc='acc%d%d' % (bi, h), tag=tag))
    return out


def k_seq(m0, nblk, w=5, wo=777):
    """Per-output K accumulation sequence: (a_off, b_off) for k0 = 0..112 step 16."""
    return [((w * TOK + m0) * CH + k0, wo + (nblk // 16) * 2048 + (k0 // 16) * 256)
            for k0 in KOFF]


def owner_map(tiles):
    """(token, channel) -> single owner tile; record missing/duplicate."""
    owners, dup = {}, []
    for tl in tiles:
        for r in range(8):
            for c in range(16):
                t = tl['m0'] + r
                ch = tl['nblk'] + c
                key = (t, ch)
                if key in owners:
                    dup.append((key, owners[key], tl))
                owners[key] = tl
    return owners, dup


def main():
    pt, ct = parent_tiles(), cand_tiles()
    assert len(pt) == 64 and len(ct) == 64
    po, pdup = owner_map(pt)
    co, cdup = owner_map(ct)

    missing = [(t, c) for t in range(TOK) for c in range(CH) if (t, c) not in po or (t, c) not in co]
    addr_diff, k_diff, seed_diff, scale_diff = [], [], [], []
    for t in range(TOK):
        for c in range(CH):
            a, b = po[(t, c)], co[(t, c)]
            pa, _ = img_half_index(a['m0'], a['nblk'])
            ca, _ = img_half_index(b['m0'], b['nblk'])
            pa += (t - a['m0']) * 16 + (c - a['nblk'])
            ca += (t - b['m0']) * 16 + (c - b['nblk'])
            if pa != ca:
                addr_diff.append((t, c, pa, ca))
            if k_seq(a['m0'], a['nblk']) != k_seq(b['m0'], b['nblk']):
                k_diff.append((t, c))
            ps = ((a['m0']) * CH, a['nblk'])
            cs = ((b['m0']) * CH, b['nblk'])
            if ps != cs:
                seed_diff.append((t, c, ps, cs))
            # scale index: pack[ars + nblk + (lane & 15)] -> depends only on nblk
            if a['nblk'] != b['nblk']:
                scale_diff.append((t, c))

    # ---- per-output-tile K order (equivalently a tile-level check over both kernels)
    tile_pairs = {}
    for a in pt:
        tile_pairs.setdefault((a['m0'], a['nblk']), set()).add('P')
    for b in ct:
        tile_pairs.setdefault((b['m0'], b['nblk']), set()).add('C')
    tile_mismatch = {k: sorted(v) for k, v in tile_pairs.items() if v != {'P', 'C'}}

    # ---- weight reuse accounting (per subgroup)
    def b_loads(tiles):
        # number of distinct (m0-group, nblk, k0) loads actually issued
        # parent: one load per (j, nblk, ki); candidate: one per (jp, nblk, ki)
        return len({(t['sg'], t.get('j'), t.get('jp'), t['nblk'], k)
                    for t in tiles for k in KOFF
                    for _ in [0]})

    parent_loads = len(pt) * len(KOFF)          # each tile issues its own 8 loads
    cand_loads = len(ct) * len(KOFF)
    # a tile owns 2 accumulators in both kernels, so raw tile counts already
    # reflect the loads: parent 64 tiles -> 512 tile-loads/SG? no: per SG the
    # tile list above spans all 8 SG; normalise per SG below.
    per_sg = {}
    for name, tiles in (('parent', pt), ('candidate', ct)):
        b_l = {}
        for t in tiles:
            for k in KOFF:
                key = (t.get('j') if name == 'parent' else t.get('jp'), t['nblk'], k)
                b_l[(t['sg'],) + key] = 1
        per_sg[name] = dict(b_tile_loads=len(b_l),
                            tiles=len(tiles),
                            a_tile_loads=len({(t['sg'], t['m0'], k)
                                              for t in tiles for k in KOFF}))

    # ---- store-tile addresses: disjoint per WG, and the two blocks of one pair
    #      complete one contiguous 256-half image block
    store_ranges = {}
    for name, tiles in (('parent', pt), ('candidate', ct)):
        rng = {}
        for t in tiles:
            base, oob = img_half_index(t['m0'], t['nblk'])
            assert not oob
            rng[(t['m0'], t['nblk'])] = (base, base + 8 * 16 - 1)
        store_ranges[name] = rng
    def overlap(rng):
        items = sorted(rng.items(), key=lambda kv: kv[1][0])
        bad = []
        for i in range(len(items) - 1):
            if items[i][1][1] >= items[i + 1][1][0]:
                bad.append((items[i][0], items[i + 1][0]))
        return bad
    pair_blocks = []
    for m0 in (0, 16, 32, 48):
        for n0 in (0, 32, 64, 96):
            rs = []
            for blk in (m0, m0 + 8):
                for nblk in (n0, n0 + 16):
                    lo, oob = img_half_index(blk, nblk)
                    assert not oob
                    rs.append((lo, lo + 127))
            rs.sort()
            covered = set()
            for lo, hi in rs:
                covered.update(range(lo, hi + 1))
            pair_blocks.append(dict(m0=m0, n0=n0, ranges=rs,
                                    union_lo=min(r[0] for r in rs),
                                    union_hi=max(r[1] for r in rs),
                                    union_contiguous=(len(covered) ==
                                                      max(r[1] for r in rs) - min(r[0] for r in rs) + 1),
                                    union_halves=len(covered)))

    lifetime = dict(
        weight_pair=dict(
            parent='per SG the n-block weight tile (nblk,k0) is loaded once per j '
                   '(4x) and consumed by 1 DPAS each time',
            candidate='same tile is loaded once per jp (2x) and consumed by two '
                      'back-to-back DPAS (acc for M block m0, then m0+8)',
            per_sg_loads=dict(parent=64, candidate=32),
            uses_per_load=dict(parent=1, candidate=2),
            war_distance_dpas=dict(parent=1, candidate=2),
            rule='a weight return window may only be overwritten after the second DPAS '
                 'of the pair; the source orders b0 (2 DPAS) before b1 (2 DPAS) so each '
                 'weight tile is retired before the next one is needed'),
        a_tiles=dict(parent='1 load -> 2 DPAS', candidate='1 load -> 2 DPAS',
                     note='unchanged per-tile reuse; only the base address is shared by '
                          'the pair (+1024 halves)'),
        accumulators=dict(parent='2 live MAT_C (8 GRF/lane)',
                          candidate='4 live MAT_C (16 GRF/lane)',
                          delta_grf_per_lane=8),
        seeds=dict(parent='2 transient MAT_H per j',
                   candidate='4 transient MAT_H per jp, converted to MAT_C immediately'),
        store_disjoint=dict(
            tile_ranges_disjoint=dict(parent=len(overlap(store_ranges['parent'])),
                                      candidate=len(overlap(store_ranges['candidate']))),
            same_tile_set=set(store_ranges['parent']) == set(store_ranges['candidate'])),
        store_pair_contiguous=pair_blocks,
        ordering='the kernel has no intra-WG reduction and no shared memory: every output '
                 'tile is written by exactly one subgroup, so there is no RAW/WAR/WAW '
                 'between store tiles and no barrier is required (parent had none)',
    )

    proof = dict(
        card='n06-f06-output',
        geometry=dict(tokens_per_wg=TOK, channels_per_wg=CH, subgroups=8,
                      lanes=32, wg_per_pass=21 * 12),
        mapping=dict(
            tile_owners_parent=len(pt), tile_owners_candidate=len(ct),
            output_elements=TOK * CH,
            owners_per_output_parent=1, owners_per_output_candidate=1,
            duplicate_parent=len(pdup), duplicate_candidate=len(cdup),
            missing=len(missing),
            store_address_mismatches=len(addr_diff),
            k_sequence_mismatches=len(k_diff),
            seed_tile_mismatches=len(seed_diff),
            scale_index_mismatches=len(scale_diff),
            tile_set_mismatch=tile_mismatch,
            equal=(not missing and not addr_diff and not k_diff and not seed_diff
                   and not scale_diff and not tile_mismatch and not cdup)),
        k_order=dict(
            rule='each output accumulator accumulates seed then exactly 8 coopMatMulAdd '
                 'steps with k0 = 0,16,...,112 in that order; the candidate reorders '
                 'only *which* output accumulator is issued, never the k index order',
            steps=len(KOFF), a_offsets=[k_seq(0, 0)[i][0] for i in range(8)],
            b_offsets=[k_seq(0, 0)[i][1] for i in range(8)],
            mismatch=len(k_diff)),
        rounding=dict(
            seed='coopMatLoad(MAT_H) from gY, then seed = seed * scaleM in half '
                 '(float16_t multiply, identical operand order)',
            accumulate='MAT_C(float) += half A * half B via coopMatMulAdd, same 8 steps',
            narrow='MAT_H(acc) once per output tile at the end, identical narrow point',
            exp_recip_clamp='untouched (not present in this kernel)'),
        reuse=per_sg,
        lifetime=lifetime,
        tail=dict(
            note='no tail: 64 tokens and 128 channels per WG are exact multiples of the '
                 '8x16 tile; grid 21x12 covers the full 42x24 image tile grid, oob handled '
                 'by the unchanged c128_img_half_index clamp'),
        pair_layout=dict(
            a_slab='a pair = rows m0..m0+15 of gT; a0 base = (w*64+m0)*128, '
                   'a1 base = a0 + 1024 halves (+2048 B); m0 is 16-aligned so the pair is one '
                   'contiguous 16x128 half (4096 B) slab',
            b_pair='b0 base = wo + nblk*2048 + kblk*256, b1 base = b0 + 2048 halves '
                   '(+4096 B): same k-block of the next n-block, one base index + constant',
            store_pair='the two M blocks are s=0..7 and s=8..15 of the same (tx,ty) quad, '
                       'so the pair completes one 256-half image store block'),
    )
    assert proof['mapping']['equal'], proof['mapping']
    (CARD / 'PROOF.json').write_text(json.dumps(proof, indent=2), encoding='utf-8')
    print(json.dumps({k: v for k, v in proof.items() if k != 'pair_layout'}, indent=1)[:4000])


if __name__ == '__main__':
    main()
