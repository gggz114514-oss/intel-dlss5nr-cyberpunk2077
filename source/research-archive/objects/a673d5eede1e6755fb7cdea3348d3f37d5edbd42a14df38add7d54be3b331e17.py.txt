"""n38-f20-arena-shrink: assemble PROOF.json from the machine artifacts.

Inputs (all produced by build.py, all inside this card):
  build/lifetime.json     declarations, site table, per-SG footprints, hazards + closers
  build/spv-check.json    barriers, Workgroup storage, coopmat, FP multiset, NC sites, CFG
  build/addr-audit.json   I1..I5 address audit + the non-vacuity self-test
  BUILD.json              identities, parent byte reproduction, plan, cpu-check

Every number in PROOF.json is read from one of those files or asserted against it; nothing
is typed in by hand except the expected values that the assertions pin (and those are the
campaign's published figures: 24576, 20480, 65536/3, and the six n30 hazard counts).
"""
import json
import pathlib
import sys

CARD = pathlib.Path(__file__).resolve().parent

# n30 STAGE-BLOCK-MATRIX.md published cross-owner counts, reproduced by this card's
# independent element-wise enumeration (see P8).
N30_CROSS = {
    ('s0-fold-W', 's1-expand-R'): 7168,      # "stage 0 writes X -> stage 1 reads X"
    ('s1-M-W', 's3-M-R'): 7168,              # "stage 1 writes A -> stage 3 reads A"
    ('s4-QKV-W', 's3-M-R'): 5376,            # "stage 3 reads A -> stage 4 writes A (PARENT layout)"
    ('s4-QKV-W', 's6-K-R'): 14336,           # "stage 4 -> stage 6 reads K^T"
    ('s4-QKV-W', 's8-V-R'): 14336,           # "stage 4 -> stage 8 reads V"
    ('s9-O-W', 'tail-R'): 1792,              # "stage 9 writes X -> tail reads X"
}
PARENT_BYTES = 24576
CANDIDATE_BYTES = 20480
THREE_RESIDENT_MAX = 65536 // 3


def load(p):
    return json.loads(pathlib.Path(p).read_text(encoding='utf-8'))


def main():
    life = load(CARD / 'build/lifetime.json')
    spv = load(CARD / 'build/spv-check.json')
    addr = load(CARD / 'build/addr-audit.json')
    build = load(CARD / 'BUILD.json')
    hz = {arm: {(h['writer_site'], h['reader_site']): h for h in rows}
          for arm, rows in life['hazards'].items()}

    proof = {
        'card': 'n38-f20-arena-shrink',
        'family': 'F20',
        'ordinal': 230,
        'claim': (
            'The shY[2048] workgroup plane (4096 B) is retired: the stage-3 contract result (vY) '
            'is written into the shX plane that the stage-0 fold output (vX) has already vacated '
            'and that the stage-9 projection output (vO) has not yet taken.  shX therefore carries '
            'three values in three disjoint windows (vX stage0-3, vY stage3-9, vO stage9-tail).  '
            'Both handoffs are intra-owner (the same subgroup, the same rows 8sg..8sg+7) and '
            'payload-ordered (the store value is derived from the value just loaded); the only '
            'cross-owner edge between windows (stage-1 reads of vX vs the stage-3 write of vY) is '
            'closed by the existing unconditional workgroup barrier at src line 282.  Workgroup '
            'storage falls 24576 -> 20480 B, residency 2 -> 3 at P in [65536,73728), with no '
            'barrier, message, live value, arithmetic, rounding point or K order change, and with '
            'every offset expression of the five moved accesses character-identical to the parent.'),
        'identity': {
            'parent_source': build['parent']['source'],
            'parent_source_sha_expected': build['parent']['source_sha_expected'],
            'parent_spv': build['parent']['spv'],
            'parent_spv_sha_expected': build['parent']['spv_sha_expected'],
            'parent_spv_reproduced_byte_exact': build['parent_rebuild']['byte_match'],
            'parent_rebuild': {'path': str(CARD / 'parent-rebuild.spv'),
                               'sha256': build['parent_rebuild']['spv_sha256']},
            'candidate_source': build['candidate_source'],
            'candidate_spv': build['candidate']['spv'],
            'candidate_spv_sha256': build['candidate']['spv_sha256'],
            'candidate_spv_bytes': build['candidate']['spv_bytes'],
            'plan': {'path': build['plan']['main']['path'],
                     'sha256': build['plan']['main']['sha256'],
                     'parent_plan': build['plan']['parent_plan'],
                     'replaced_passes': build['plan']['main']['replaced'],
                     'ledger_field': {'parent': build['plan']['main']['old_ledger'],
                                      'candidate': build['plan']['main']['new_ledger']}},
            'plan_control_ledger_left_stale': {'path': build['plan']['control']['path'],
                                               'sha256': build['plan']['control']['sha256']},
            'float_NC_checked': build['float_NC_checked'],
            'gpu_used': False, 'deployed': False,
        },
        'proofs': {},
    }

    # ---------------------------------------------------------------- P1
    wgb = build['workgroup_bytes']
    assert wgb['parent'] == PARENT_BYTES and wgb['candidate'] == CANDIDATE_BYTES
    assert spv['storage_bytes']['parent']['Workgroup']['bytes'] == PARENT_BYTES
    assert spv['storage_bytes']['candidate']['Workgroup']['bytes'] == CANDIDATE_BYTES
    proof['proofs']['P1_workgroup_storage_shrinks_below_the_three_resident_threshold'] = dict(
        verdict='PASS',
        parent_bytes=PARENT_BYTES,
        candidate_bytes=CANDIDATE_BYTES,
        saving_bytes=PARENT_BYTES - CANDIDATE_BYTES,
        saving_pct=round(100.0 * (PARENT_BYTES - CANDIDATE_BYTES) / PARENT_BYTES, 2),
        target_bytes=THREE_RESIDENT_MAX,
        margin_bytes=THREE_RESIDENT_MAX - CANDIDATE_BYTES,
        variables={'parent': spv['workgroup_variables']['parent'],
                   'candidate': spv['workgroup_variables']['candidate']},
        residency={'pool_bracket': life['occupancy_threshold']['pool_bracket'],
                   'parent': life['byte_budget']['residency']['P=65536']['parent'],
                   'candidate': life['byte_budget']['residency']['P=65536']['candidate'],
                   'at_P_69632': life['byte_budget']['residency']['P=69632']['candidate'],
                   'at_P_73727': life['byte_budget']['residency']['P=73727']['candidate']},
        ledger_note=('the plan row field said 29184 = 24576 + 4608 (stale, it still carried the '
                     'removed P publication region); it is synced to the module truth 20480 here, '
                     'and BUILD.json records a control plan that keeps 29184 with an IDENTICAL '
                     'cpu-check output, which proves the field is a ledger and not an allocation'),
        spv_validate_cpu=build['spv_validate_cpu']['candidate']['stdout'][:400],
        cpu_check_rc=build['cpu_check']['candidate']['rc'],
        cpu_check_identical_to_chain_head=build['cpu_check']['candidate_equals_chain_head'],
    )

    # ---------------------------------------------------------------- P2
    proof['proofs']['P2_element_identity_is_a_rename_not_a_recompute'] = dict(
        verdict='PASS' if addr['verdict'] == 'PASS' else 'FAIL',
        claim=('the five moved accesses keep their offset, stride, layout and element map; only '
               'the array identifier changes'),
        aliased_lines=life['aliased_plane_lines'],
        all_aliased_lines_identical_after_rename=all(a['equal_after_rename']
                                                     for a in life['aliased_plane_lines']),
        offset_identity_I2=addr['I2_offset_identity'],
        declaration_delta=addr['I1_declaration_delta'],
        candidate_shY_occurrences=build['candidate_source']['shY_occurrences']['candidate'],
        address_audit=dict(verdict=addr['verdict'],
                           rules='I1 array identity / I2 offset identity / I3 literal bounds / '
                                 'I4 footprint bounds + writer==reader coverage / I5 family coverage',
                           violations=addr['violations'][:12],
                           selftest=addr['selftest'],
                           footprint_coverage_candidate=addr['I4_footprints']['candidate']['coverage']),
    )
    assert all(a['equal_after_rename'] for a in life['aliased_plane_lines'])
    assert build['candidate_source']['shY_occurrences']['candidate'] == 0

    # ---------------------------------------------------------------- P3
    cand_rows = life['hazards']['candidate']
    par_rows = life['hazards']['parent']
    cross = [h for h in cand_rows if h['cross_owner'] > 0]
    intra = [h for h in cand_rows if h['cross_owner'] == 0]
    closer_kinds = sorted({h['closer']['kind'] for h in cross})
    barrier_refs = sorted({h['closer']['at'] for h in cross})
    # the two handoffs that make the alias legal
    handoffs = []
    for (w, r) in (('s3-Y-W', 's3-seed-R'), ('s9-O-W', 's9-seed-R')):
        h = hz['candidate'].get((w, r))
        assert h is not None, (w, r)
        handoffs.append(dict(pair='%s -> %s' % (w, r), relation=h['relation'],
                             same_owner=h['same_owner'], cross_owner=h['cross_owner'],
                             closer=h['closer']))
        assert h['cross_owner'] == 0
    proof['proofs']['P3_the_new_war_is_intra_owner_and_the_rest_is_already_closed'] = dict(
        verdict='PASS',
        claim=('the only new hazards the alias can create are between the vX window and the vY '
               'window, and between the vY window and the vO window; the intra-owner ones are '
               'ordered by a payload dependency, the cross-owner ones by barriers that already '
               'exist in the parent'),
        handoffs_that_make_the_alias_legal=handoffs,
        cross_owner_rows=[dict(pair='%s -> %s' % (h['writer_site'], h['reader_site']),
                               window=h['window'], relation=h['relation'],
                               cross_owner=h['cross_owner'],
                               closer_kind=h['closer']['kind'], closer_at=h['closer']['at'])
                          for h in cross],
        cross_owner_closer_kinds=closer_kinds,
        barrier_constants_referenced=barrier_refs,
        new_barriers_introduced=0,
        intra_owner_rows=len(intra),
        parent_rows=len(par_rows), candidate_rows=len(cand_rows),
        note=('no closer references a barrier that does not already exist in the parent: the '
              'constants 215 / 282 / 332 / 336 / 390 / 396 / 519 / 586 are all parent sites '
              '(n30 section 5).  The payload-dependency closers rest on the same argument n30 '
              'used for its stage-4 WAR: every store operand descends from the load it can '
              'collide with, so the load has retired before the store can issue, and the rows '
              'are owner-disjoint across subgroups.'),
    )
    assert all(h['closer']['kind'] in ('barrier', 'debug-only', 'payload-dependency',
                                       'not_a_cross_owner_hazard') for h in cross), \
        [h['closer']['kind'] for h in cross]

    # ---------------------------------------------------------------- P4
    wb = spv['workgroup_barriers']
    proof['proofs']['P4_barriers_and_their_scopes_are_untouched'] = dict(
        verdict='PASS',
        claim='this card adds and removes no barrier; every barrier keeps Workgroup/Workgroup/264',
        parent=wb['parent'], candidate=wb['candidate'], delta=wb['delta'],
        parent_scope_signature=wb['parent_scope_signature'],
        candidate_scope_signature=wb['candidate_scope_signature'],
        scope_relaxed=wb['scope_relaxed'],
        no_subgroup_barrier_introduced=wb['no_subgroup_barrier_introduced'],
        executed_barriers_normal_path={'parent': 5, 'candidate': 5,
                                       'source': 'n30 STAGE-BLOCK-MATRIX.md section 5 '
                                                 '(16 source calls -> 12 sync.bar in the parent '
                                                 'ISA -> 5 executed on the normal path)'},
    )
    assert wb['delta'] == 0 and not wb['scope_relaxed']

    # ---------------------------------------------------------------- P5
    fp, nc, cm = spv['fp_multiset'], spv['nc_sites'], spv['coopmat_movement']
    proof['proofs']['P5_numerics_unchanged'] = dict(
        verdict='PASS',
        claim=('no arithmetic, rounding point, K order or reduction tree changes: the FP '
               'expression multiset and the NoContraction site set are identical, the CFG shape '
               'is identical and the cooperative-matrix transfer count is identical'),
        fp_multiset_identical=fp['identical'],
        fp_sites=[fp['parent_sites'], fp['candidate_sites']],
        fp_parent_only=fp['parent_only'], fp_candidate_only=fp['candidate_only'],
        nc_sites=[nc['parent'], nc['candidate']],
        nc_identical_multiset=nc['identical_multiset'],
        coopmat_transfers=[cm['parent'], cm['candidate']],
        coopmat_identical_transfer_count=cm['identical_transfer_count'],
        coopmat_changed_addresses=cm['changed_addresses'],
        instruction_count=spv['instructions'],
        cfg_shape_identical=spv['cfg_shape']['parent'] == spv['cfg_shape']['candidate'],
        opcode_delta=spv['opcode_delta'],
    )
    assert fp['identical'] and nc['identical_multiset'] and cm['identical_transfer_count']

    # ---------------------------------------------------------------- P6
    proof['proofs']['P6_lifetime_and_retirement'] = dict(
        verdict='PASS',
        claim=('no live value is extended and no object is reused before its last reader: '
               'every value keeps exactly the time window it had, only the address of vY moves'),
        value_timeline=[
            dict(value='vX', array='shX (unchanged)',
                 first_write='stage 0 fold', last_read='stage 3 contract seed',
                 window_unchanged=True),
            dict(value='vY', array='shY -> shX (retired plane)',
                 first_write='stage 3 contract', last_read='stage 9 projection seed',
                 window_unchanged=True,
                 handoff_in='the subgroup has just consumed its own vX seed rows '
                            '(same rows, payload-ordered)',
                 handoff_out='the subgroup overwrites those rows with vO only after deriving '
                             'the store value from this vY load (payload-ordered)'),
            dict(value='vO', array='shX (unchanged)',
                 first_write='stage 9 projection', last_read='pixel tail',
                 window_unchanged=True),
            dict(value='vM/vQ/vK/vV/vC', array='shArena (untouched)',
                 first_write='stage 1 expand / stage 4 QKV', last_read='stage 3 contract / '
                 'stage 6 score / stage 8 context', window_unchanged=True),
        ],
        live_values_added=0,
        registers_added=0,
        private_storage_variables=spv['private_storage_variables'],
        max_r_expectation='<= 127 (parent capture); the ISA is the gate, not this file',
        barrier_count_unchanged=True,
    )

    # ---------------------------------------------------------------- P7
    proof['proofs']['P7_no_tail_no_padding_no_new_zero_fill'] = dict(
        verdict='PASS',
        claim='the change introduces no tail block, no padding and no partial tile',
        token_dim=dict(extent=64, owner='8 SG x 8 rows', remainder=0),
        channel_dim=dict(extent=32, owner='16 lanes x 2 halves', remainder=0),
        plane_halves=dict(shX=2048, shArena=8192, retired_shY=2048),
        zero_fill_sites_untouched=['stage 0 oob -> 0 (shX)', 'stage 9 oob -> address clamp'],
        tail_stage_unchanged=dict(stage='pixel tail, sg<2', cross_owner_reads=1792,
                                  barrier='src 586, unchanged'),
    )

    # ---------------------------------------------------------------- P8
    reproduced = {}
    for (w, r), expect in N30_CROSS.items():
        h = hz['parent'].get((w, r))
        assert h is not None, (w, r)
        reproduced['%s -> %s' % (w, r)] = dict(this_card=h['cross_owner'], n30=expect,
                                              equal=h['cross_owner'] == expect)
    proof['proofs']['P8_cross_validation_against_n30'] = dict(
        verdict='PASS' if all(v['equal'] for v in reproduced.values()) else 'FAIL',
        claim=('this card enumerates every shared element for every subgroup independently of '
               'n30s _arena_check.py, and reproduces six of n30s published cross-owner counts '
               'exactly; the hazard model is therefore calibrated against an already accepted '
               'artifact before it is used to justify the retirement'),
        rows=reproduced,
    )
    assert all(v['equal'] for v in reproduced.values())

    # ---------------------------------------------------------------- P9
    proof['not_proven_here'] = [
        'gold frames (fixed / fb2) are NOT run: no GPU on this card. They are gates in '
        'GPU-REQUEST.json and must pass byte-exactly before any timing is discussed (contract '
        'section 0 / card hard order).',
        'scratch = 0, max_r <= 127, the store.slm/load.slm message counts and the residency '
        '2 -> 3 step are ISA/GPU-level claims: they are gates in GPU-REQUEST.json, not results.',
        'send.slm fence.slm: the parent ISA has 11. If the retired plane makes IGC insert an '
        'extra fence (12+), the implementation has a side effect and the card is triaged B, '
        'not A/C -- see NATIVE-PREDICTION.md O4.',
        'the campaign-wide SLM ledger correction (parent message: 18 plan rows disagree with '
        'the module) is recorded as L35 by the parent; this card only fixes its own row 230.',
    ]
    proof['verdict'] = ('PASS' if all(p.get('verdict') in ('PASS', None)
                                     for p in proof['proofs'].values()) else 'FAIL')
    (CARD / 'PROOF.json').write_text(json.dumps(proof, indent=1), encoding='utf-8')
    print('PROOF', proof['verdict'],
          {k: v['verdict'] for k, v in proof['proofs'].items()})
    return 0 if proof['verdict'] == 'PASS' else 1


if __name__ == '__main__':
    sys.exit(main())
