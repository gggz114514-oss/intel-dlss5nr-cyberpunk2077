"""n31-f01-dpas-chain: dependency graph of the F01 kernel AFTER 0x4A90 (task A).

Identity: chain head = cards/n29-f01-alu-chain/candidate.spv (spv sha c12c0f9d...),
captured as queue/n29-f01-alu-chain/native/captures/native-native-cand/ISA/
CS_asma0888c5a3c6703a9_simd32_0000000000000001_0_orig.asm.  Read-only, CPU only.

Register model (Xe GRF = 8 dwords = 32 B), derived from the dump itself:
    r12.4<1>:d        one dword at r12.4
    r11:4             an operand region of 4 GRF = 32 dwords from r11.0
    r19:f             no region given: <1> -> 1 element per lane (span = exec)
    acc0.0<1>:hf      accumulator temporaries (sel/add write them)
    a0.0, null, fN.M  address reg / discard / flag

Definitions used in DPAS-CHAIN-AUDIT.md (all reproducible from this file):
  region  every instruction with addr >= 0x4A90
  live    region minus the 37 `(W&f1.0) jmpi` debug spans (debug==0, plan p0.z=0)
  edge    use -> reaching definition of the same dword (program order)
  sink    every live `send.*` that writes memory (store.*, signal barrier, end of thread)
  chain   union of live backward slices from the sinks
  side    live \ chain
  depth   longest dependent path in the chain; weighted depth uses LAT
  N(p)    per-lane independent chains: greedy def-use path cover, then the number of
          chains whose [min_addr,max_addr] extent covers program point p; max N(p)
          is the ILP number that must not regress.
"""
import json
import re
import sys
from collections import Counter, defaultdict
from pathlib import Path

CAMP = Path('@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/fullnet-continuous-20261027')
sys.path.insert(0, str(CAMP))
import ISA  # noqa: E402

ISA_PATH = CAMP / 'queue/n29-f01-alu-chain/native/captures/native-native-cand/ISA/CS_asma0888c5a3c6703a9_simd32_0000000000000001_0_orig.asm'
ENTRY_END = 0x4A90
OUT = CAMP / 'cards/n31-f01-dpas-chain/tools/post-chain.json'

DPAS_DST = int(__import__('os').environ.get('DPAS_DST', 4))
DPAS_AB = int(__import__('os').environ.get('DPAS_AB', 8))
TYPE_W = {'ud': 1, 'd': 1, 'w': 1, 'uw': 1, 'f': 1, 'v': 1, 'hf': 0.5,
          'q': 2, 'uq': 2, 'df': 2, 'b': 0.25, 'ub': 0.25, 'bf': 0.25}
LAT = {'mul': 5, 'mad': 5, 'macl': 5, 'mach': 5, 'fma': 5, 'dpas.8x8': 8,
       'math.inv': 12, 'math.rsqt': 12, 'math.sqrt': 12, 'math.log': 12,
       'math.exp': 12, 'rndd': 6, 'rndu': 6, 'rnde': 6, 'rndz': 6, 'frc': 6,
       'add': 4, 'add3': 4, 'sub': 4, 'shl': 4, 'shr': 4, 'asr': 4, 'and': 4,
       'or': 4, 'xor': 4, 'not': 4, 'bfn.0xFE': 4, 'bfn.0xF8': 4, 'cmp': 4,
       'sel': 4, 'min': 4, 'max': 4, 'mov': 1, 'movf': 1, 'sync.nop': 0,
       'send.ugm': 20, 'send.slm': 20, 'send.gtwy': 20, 'illegal': 0,
       'jmpi': 0, 'goto': 0, 'join': 1, 'sync.allrd': 4, 'sync.allwr': 4,
       'sync.bar': 4, 'mov.hf': 1}
WEIGHT = {'math.inv': 30, 'math.rsqt': 30, 'math.log': 30, 'math.exp': 30,
          'dpas.8x8': 6, 'send.slm': 12, 'send.ugm': 12, 'sync.bar': 12,
          'sync.allwr': 8, 'sync.allrd': 6, 'fma': 2, 'mad': 2, 'mul': 2,
          'add': 1, 'add3': 1, 'sel': 1, 'cmp': 1, 'mov': 0.3, 'and': 1,
          'or': 1, 'shl': 1, 'shr': 1, 'bfn.0xFE': 1, 'bfn.0xF8': 1,
          'sync.nop': 0.1, 'illegal': 1}


def exec_size(flags, default=32):
    m = re.search(r'\((\d+)\|M\d+\)', flags or '')
    return int(m.group(1)) if m else default


def region_spec(op):
    m = re.search(r'<(\d+);(\d+),(\d+)>', op)
    return (int(m.group(1)), int(m.group(2)), int(m.group(3))) if m else None


def elem_type(op):
    m = re.search(r':([a-z0-9]+)\s*$', op)
    return m.group(1) if m else None


def grf_suffix(op):
    """r11:4 -> 4  (region in GRF, only when there is no <..> spec)."""
    m = re.search(r'^[ra]?\d+(?:\.\d+)?:(\d+)$', op.strip())
    return int(m.group(1)) if m else None


def reg_base(op):
    o = op.lstrip('-').replace('(abs)', '').strip()
    m = re.match(r'^r(\d+)\.(\d+)', o)
    if m:
        return ('r', int(m.group(1)), int(m.group(2)))
    m = re.match(r'^r(\d+)(?![\d.])', o)
    if m:
        return ('r', int(m.group(1)), 0)
    m = re.match(r'^acc(\d+)\.(\d+)', o)
    if m:
        return ('acc', int(m.group(1)), int(m.group(2)))
    return None


def span_of(op, flags, opcode):
    """dwords written/read by this operand (per lane region, 1 lane = 1 dword)."""
    ex = exec_size(flags)
    g = grf_suffix(op)
    if g is not None:
        return max(1, g * 8)
    rs = region_spec(op)
    if rs is not None and rs[0] == 0:
        return 1
    t = elem_type(op)
    w = TYPE_W.get(t, 1) if t else 1
    if opcode.startswith('dpas'):
        return DPAS_DST
    return max(1, int(round(w * ex)))


def is_send(ins):
    return ins['op'].startswith('send')


def classify(ins):
    """returns (writes, reads) as list of (file, regno, dword, count)."""
    op = ins['op']
    ops = ins['ops']
    com = ins['comment'] or ''
    w, r = [], []
    if is_send(ins):
        store = 'store.' in com or 'fence' in com
        if store:
            if len(ops) > 2:
                b = reg_base(ops[2])
                if b:
                    r.append((b[0], b[1], b[2], span_of(ops[2], ins['flags'], op)))
        else:
            if ops:
                b = reg_base(ops[0])
                if b:
                    g = grf_suffix(ops[0])
                    n = max(1, g * 8) if g is not None else span_of(ops[0], ins['flags'], op)
                    w.append((b[0], b[1], b[2], n))
        for o in ops:
            if re.match(r'^a0\.\d+', o):
                r.append(('a0', 0, 0, 2))
        return w, r
    if op.startswith('dpas'):
        if ops:
            b = reg_base(ops[0])
            if b:
                w.append((b[0], b[1], b[2], 4))
        for o in ops[2:4]:
            b = reg_base(o)
            if b:
                r.append((b[0], b[1], b[2], DPAS_AB))
        return w, r
    if not ops:
        return w, r
    # generic ALU: ops[0] = dst unless it is null
    if ops[0] != 'null':
        b = reg_base(ops[0])
        if b:
            w.append((b[0], b[1], b[2], span_of(ops[0], ins['flags'], op)))
        else:
            m = re.match(r'^f(\d)\.(\d)<', ops[0])
            if m:
                w.append(('f', int(m.group(1)), int(m.group(2)), 1))
    for o in ops[1:]:
        if o.startswith('null'):
            continue
        b = reg_base(o)
        if b:
            if b[0] == 'a0':
                r.append(('a0', 0, 0, 2))
            else:
                r.append((b[0], b[1], b[2], span_of(o, ins['flags'], op)))
    for o in ops:
        if re.match(r'^a0\.\d+', o):
            if not any(x[0] == 'a0' for x in r):
                r.append(('a0', 0, 0, 2))
    return w, r


def load():
    isa = ISA.parse(ISA_PATH)
    mask = json.loads((CAMP / 'cards/n31-f01-dpas-chain/tools/exec-mask.json').read_text(encoding='utf-8'))
    spans = [(int(a, 16), int(b, 16)) for a, b in mask['spans']]
    for k, ins in enumerate(isa):
        ins['idx'] = k
        ins['w'], ins['r'] = classify(ins)
        ins['dead'] = False
        if ins['addr'] >= ENTRY_END:
            for a, b in spans:
                if a < ins['addr'] < b:
                    ins['dead'] = True
                    break
    by_dw = defaultdict(list)
    for ins in isa:
        for f, no, dw, n in ins['w']:
            for d in range(dw, dw + n):
                by_dw[(f, no, d)].append(ins['idx'])
    for v in by_dw.values():
        v.sort()
    return isa, by_dw


def reaching(by_dw, key, idx):
    lst = by_dw.get(key)
    if not lst:
        return None
    hit = None
    for w in lst:
        if w < idx:
            hit = w
        else:
            break
    return hit


def roots(isa, by_dw):
    n = 0
    tot = 0
    for ins in isa:
        if ins['addr'] < ENTRY_END or ins['dead']:
            continue
        for f, no, dw, cnt in ins['r']:
            for d in range(dw, dw + cnt):
                tot += 1
                if reaching(by_dw, (f, no, d), ins['idx']) is None:
                    n += 1
    return n, tot


def analyse():
    isa, by_dw = load()
    post = [i for i in isa if i['addr'] >= ENTRY_END]
    live = [i for i in post if not i['dead']]
    live_idx = {i['idx'] for i in live}
    rn, rt = roots(isa, by_dw)

    def par(k):
        out = set()
        for f, no, dw, cnt in isa[k]['r']:
            for d in range(dw, dw + cnt):
                p = reaching(by_dw, (f, no, d), k)
                if p is not None and p in live_idx:
                    out.add(p)
        return out

    store_ops = [i['idx'] for i in live if is_send(i) and
                 ('store.' in (i['comment'] or '') or 'signal barrier' in (i['comment'] or '')
                  or 'end of thread' in (i['comment'] or ''))]
    chain = set()
    stack = list(store_ops)
    while stack:
        k = stack.pop()
        if k in chain:
            continue
        chain.add(k)
        stack.extend(par(k))
    side = live_idx - chain

    memo, wmemo = {}, {}
    for ins in isa:
        k = ins['idx']
        if k not in live_idx:
            continue
        ps = par(k)
        memo[k] = 1 + max((memo.get(p, 0) for p in ps), default=0)
        wmemo[k] = LAT.get(ins['op'], 4) + max((wmemo.get(p, 0) for p in ps), default=0)

    # greedy path cover -> chains
    order = sorted(chain, key=lambda k: isa[k]['addr'])
    owner, chains = {}, []
    for k in order:
        cand = None
        for p in sorted(par(k), reverse=True):
            if p in owner:
                cand = owner[p]
                break
        if cand is None:
            cand = len(chains)
            chains.append(dict(id=cand, lo=isa[k]['addr'], hi=isa[k]['addr'], n=0, ops=Counter()))
        owner[k] = cand
        c = chains[cand]
        c['n'] += 1
        c['lo'] = min(c['lo'], isa[k]['addr'])
        c['hi'] = max(c['hi'], isa[k]['addr'])
        c['ops'][isa[k]['op']] += 1
    ev = []
    for c in chains:
        ev.append((c['lo'], 1))
        ev.append((c['hi'] + 1, -1))
    ev.sort()
    cur = best = 0
    best_at = 0
    for a, d in ev:
        cur += d
        if cur > best:
            best, best_at = cur, a

    def tab(s):
        return Counter(isa[k]['op'] for k in s)

    def msgs(s):
        return Counter((isa[k]['comment'] or '').split(';')[-1].strip()
                       for k in s if is_send(isa[k]) and isa[k]['comment'])

    price = defaultdict(lambda: [0, 0.0, 0])
    for k in chain:
        op = isa[k]['op']
        w = WEIGHT.get(op, 1.0)
        price[op][0] += 1
        price[op][1] += w
        price[op][2] = max(price[op][2], wmemo[k])

    out = dict(isa=str(ISA_PATH), entry_end=hex(ENTRY_END),
               region=dict(n=len(post), lo=hex(post[0]['addr']), hi=hex(post[-1]['addr'])),
               live=dict(n=len(live), dead=len(post) - len(live)),
               roots=dict(unresolved_read_dwords=rn, total_read_dwords=rt,
                          pct=round(100.0 * rn / max(1, rt), 2)),
               sinks=len(store_ops),
               chain=dict(n=len(chain), side=len(side),
                          pct=round(100.0 * len(chain) / max(1, len(live)), 2)),
               depth=dict(chain=max(memo[k] for k in chain),
                          chain_weighted=max(wmemo[k] for k in chain)),
               chains=dict(count=len(chains), max_concurrent=best, at=hex(best_at),
                           top=[dict(id=c['id'], n=c['n'], lo=hex(c['lo']), hi=hex(c['hi']),
                                     ops=dict(c['ops'].most_common(5)))
                                for c in sorted(chains, key=lambda c: -c['n'])[:12]]),
               op_live=dict(tab(live_idx).most_common()),
               op_chain=dict(tab(chain).most_common()),
               op_side=dict(tab(side).most_common()),
               msg_live=dict(msgs(live_idx).most_common()),
               msg_chain=dict(msgs(chain).most_common()),
               price={k: dict(count=v[0], weight=round(v[1], 1), max_wdepth=v[2])
                      for k, v in sorted(price.items(), key=lambda t: -t[1][1])},
               wdeepest=[[wmemo[k], hex(isa[k]['addr']), isa[k]['op']]
                         for k in sorted(chain, key=lambda k: -wmemo[k])[:15]],
               deepest=[[memo[k], hex(isa[k]['addr']), isa[k]['op']]
                        for k in sorted(chain, key=lambda k: -memo[k])[:15]])
    OUT.write_text(json.dumps(out, indent=1), encoding='utf-8')

    print('region  : %d  %s..%s' % (len(post), out['region']['lo'], out['region']['hi']))
    print('live    : %d  (debug-skipped %d)' % (len(live), len(post) - len(live)))
    print('roots   : %d / %d read dwords unresolved (%.2f%%)'
          % (rn, rt, 100.0 * rn / max(1, rt)))
    print('sinks   : %d' % len(store_ops))
    print('CHAIN   : %d   SIDE: %d   (%.1f%% on chain)'
          % (len(chain), len(side), 100.0 * len(chain) / len(live)))
    print('depth   : %d   weighted %d' % (out['depth']['chain'], out['depth']['chain_weighted']))
    print('chains  : %d   max concurrent per lane %d at %s'
          % (len(chains), best, hex(best_at)))
    print('\nop           live  chain   side')
    for op in sorted(set(tab(live_idx)), key=lambda o: -tab(live_idx)[o]):
        print('%-12s %5d %6d %6d' % (op, tab(live_idx)[op], tab(chain)[op], tab(side)[op]))
    print('\nL24 weighted price on chain (top 12):')
    for k, v in list(out['price'].items())[:12]:
        print('  %-12s count %5d weight %8.1f max_wdepth %4d' % (k, v['count'], v['weight'], v['max_wdepth']))
    print('\nweighted-deepest:')
    for w, a, o in out['wdeepest'][:8]:
        print('  %4d %s %s' % (w, a, o))
    return out


def cmd_reduce():
    """census of the 4 softmax-denominator reduction blocks."""
    isa, by_dw = load()
    live = [i for i in isa if i['addr'] >= ENTRY_END and not i['dead']]
    inv = [i for i in live if i['op'] == 'math.inv']
    print('live math.inv %d @ %s' % (len(inv), ' '.join(hex(i['addr']) for i in inv)))
    print('live math.rsqt %d' % sum(1 for i in live if i['op'] == 'math.rsqt'))
    ends = [i['addr'] for i in inv]
    blocks = [(ends[0], ends[1]), (ends[2], ends[3]), (ends[4], ends[5]), (ends[6], ends[7])]
    starts = [0x11300, blocks[0][1], blocks[1][1], blocks[2][1]]
    tot = Counter()
    print('\n%-9s %-9s %5s %5s %5s %5s %5s %5s %5s' %
          ('from', 'to', 'n', 'sel', 'add', 'mov', 'mul', 'shl', 'inv'))
    for (a, b), lo in zip(blocks, starts):
        body = [i for i in live if lo < i['addr'] <= b]
        c = Counter(i['op'] for i in body)
        tot.update(c)
        print('%-9s %-9s %5d %5d %5d %5d %5d %5d %5d' %
              (hex(lo), hex(b), len(body), c['sel'], c['add'], c['mov'], c['mul'], c['shl'], c['math.inv']))
    print('total:', dict(tot.most_common(12)))


if __name__ == '__main__':
    cmd = sys.argv[1] if len(sys.argv) > 1 else 'all'
    if cmd in ('all', 'chain'):
        analyse()
    if cmd in ('all', 'reduce'):
        print()
        cmd_reduce()
