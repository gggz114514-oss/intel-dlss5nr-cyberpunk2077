def census(path):
    raw = open(path, "rb").read()
    magic = struct.unpack("<I", raw[:4])[0]
    assert magic == 0x07230203, hex(magic)
    n = (len(raw) - 20) // 4
    words = struct.unpack("<%dI" % n, raw[20:20 + 4 * n])
    counts = {}
    i = 0
    while i < n:
        w = words[i]
        op = w & 0xFFFF
        wc = w >> 16
        if wc == 0:
            break
        counts[op] = counts.get(op, 0) + 1
        i += wc
    return {
        "sha256": hashlib.sha256(raw).hexdigest(),
        "bytes": len(raw),
        "OpAtomicUMax": counts.get(239, 0),
        "OpAtomicSMax": counts.get(238, 0),
        "OpAtomicIAdd": counts.get(234, 0),
        "OpConvertFToS": counts.get(110, 0),
        "total_instructions": sum(counts.values()),
    }


def main():
    out = {}
    for name, p in SPVS.items():
        if not os.path.exists(p):
            out[name] = {"MISSING": p}
            continue
        c = census(p)
        out[name] = c
        print("%-28s sha %s  UMax=%-2d SMax=%-2d FToS=%-3d bytes=%d" % (
            name, c["sha256"][:8], c["OpAtomicUMax"], c["OpAtomicSMax"],
            c["OpConvertFToS"], c["bytes"]))
    dst = os.path.join(os.path.dirname(os.path.dirname(os.path.abspath(__file__))),
                       "tools", "spv-opcode-census.json")
    with open(dst, "w", encoding="utf-8") as f:
        json.dump(out, f, indent=1)
    print("wrote", dst)


if __name__ == "__main__":
    main()
