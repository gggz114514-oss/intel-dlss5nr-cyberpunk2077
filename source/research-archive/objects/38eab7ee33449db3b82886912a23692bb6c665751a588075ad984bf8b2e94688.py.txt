from pathlib import Path
import json,re,collections
R=Path(__file__).resolve().parent;N=R/'f20-owner-bridge'
review=json.loads((N/'NATIVE-REVIEW.json').read_text())
def instructions(path):
 out=[]
 for line in Path(path).read_text().splitlines():
  m=re.match(r'/\* \[([0-9A-Fa-f]+)\].*?\*/\s*(.*)',line)
  if m:out.append((int(m[1],16),m[2]))
 return out
def movclass(line):
 # Conversion instructions are mov too. Do not equate total mov with copies.
 tail=line.split(')',1)[-1]
 types=re.findall(r':(hf|f|ud|d|uw|w|uq|q|ub|b)\b',tail)
 if len(types)>=2:
  if types[:2]==['hf','f']:return 'f32_to_half'
  if types[:2]==['f','hf']:return 'half_to_f32'
  if types[0]!=types[1]:return 'other_type_conversion'
  return 'same_type_copy_or_immediate'
 return 'other_mov'
o={}
for arm in ['parent','candidate']:
 ins=instructions(review[arm]['isa']);counts=collections.Counter(movclass(l) for pc,l in ins if re.search(r'\bmov\s+\(',l))
 o[arm]=dict(isa=review[arm]['isa'],static_mov_categories=dict(counts),total_mov=sum(counts.values()))
ins=instructions(review['candidate']['isa'])
bank_records=[]
for lo in [4,12,20,28]:
 dpas=[(pc,l) for pc,l in ins if 0x20c8<=pc<=0x6518 and re.search(rf'\bdpas\.8x8 .*?r{lo}:f\s+r{lo}:f',l)]
 assert len(dpas)==8,(lo,len(dpas))
 # Restrict to the interval AFTER the first update and BEFORE last update.
 # Covers direct GRF destinations; send returns additionally checked via rd size.
 overwrites=[]
 for pc,l in ins:
  if not dpas[0][0]<pc<dpas[-1][0] or 'dpas.8x8' in l:continue
  dst=re.search(r'\(\d+\|M\d+\)\s+r(\d+)',l)
  if dst:
   base=int(dst[1]);extent=1
   if 'send.' in l:
    rd=re.search(r'rd:(\d+)',l);extent=int(rd[1]) if rd else 1
   if base<=lo+7 and base+extent-1>=lo:overwrites.append(dict(pc=hex(pc),line=l))
 assert not overwrites,(lo,overwrites)
 narrow=[dict(pc=hex(pc),line=l) for pc,l in ins if 0x6530<=pc<=0x6b70 and re.search(r'\bmov\s+',l) and re.search(r':hf\s+r('+ '|'.join(str(x) for x in range(lo,lo+8))+r')\.',l) and ':f' in l]
 assert len(narrow)==8,(lo,narrow)
 bank_records.append(dict(bank=f'r{lo}..r{lo+7}',updates=[hex(pc) for pc,l in dpas],non_DPAS_destination_overlap_between_updates=overwrites,narrowing_consumers=narrow))
o['contract_banks']=bank_records
o['scope']='Direct destinations and send return extents checked. Not a general byte-precise liveness analyzer; register region extents of arbitrary arithmetic not inferred. No claim that static mov delta is dynamic copy overhead.'
o['finding']='Four stable contract banks survive all eight hidden tiles. No bank ping-pong observed. Distinct expansion banks r47/r39 are narrowed, activated and reused as A. Source increases overlap, not repeated accumulator cloning.'
(N/'ACC-LIFETIME-AUDIT.json').write_text(json.dumps(o,indent=2));print(json.dumps({k:o[k] for k in ['parent','candidate','finding']},indent=2))
