from pathlib import Path
import numpy as np,json
from PIL import Image,ImageDraw
R=Path(__file__).parent.parent;D=Path(__file__).parent;P=R/'corrected-20261029';rows=[]
for i in range(8):
 a=np.fromfile(R/'run/feedback-fp16'/f'frame_{i}.bin',dtype='<f4').reshape(2,720,1280,4)[0,:,:,:3].astype(np.float64);b=np.fromfile(D/'body-vscale-corrected/feedback'/f'frame_{i}.bin',dtype='<f4').reshape(2,720,1280,4)[0,:,:,:3].astype(np.float64);e=b-a;rows.append({'frame':i,'PSNR_RGB_peak1_dB':float(-10*np.log10(np.mean(e*e))),'RMSE_RGB':float(np.sqrt(np.mean(e*e))),'MAE_RGB':float(abs(e).mean()),'max_error_RGB':float(abs(e).max()),'nonfinite':int((~np.isfinite(b)).sum())})
(P/'FEEDBACK-QUALITY.json').write_text(json.dumps(rows,indent=2));print(rows)
a=np.fromfile(R/'run/fixed-fp16/frame_0.bin',dtype='<f4').reshape(2,720,1280,4)[0,:,:,:3];old=np.fromfile(R/'run/fixed-int8/frame_0.bin',dtype='<f4').reshape(2,720,1280,4)[0,:,:,:3];b=np.fromfile(D/'body-vscale-corrected/run/frame_0.bin',dtype='<f4').reshape(2,720,1280,4)[0,:,:,:3]
canvas=Image.new('RGB',(1280,776),(24,24,24));draw=ImageDraw.Draw(canvas)
for i,(title,x) in enumerate([('FP16 reference',a),('Original INT8 v15: 7.93 dB',old),('Corrected INT8: 26.12 dB',b),('Absolute error x4',abs(b-a)*4)]):
 im=Image.fromarray(np.rint(np.clip(x,0,1)*255).astype(np.uint8)).resize((640,360));cx=(i%2)*640;cy=(i//2)*388;draw.text((cx+8,cy+7),title,fill='white');canvas.paste(im,(cx,cy+28))
canvas.save(P/'COMPARISON.png')
