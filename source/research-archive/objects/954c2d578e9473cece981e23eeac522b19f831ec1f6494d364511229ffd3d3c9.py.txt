"""Private capture package and reversible game mount; never starts the game."""
from pathlib import Path
import argparse,configparser,hashlib,json,shutil,subprocess
Q=Path(__file__).resolve().parent
B=Q.parents[2]; A=Q.parent
P=Q/'game-capture'; source=A/'game-latest-20261028/full'
game=Path('@NR_GAME@/bin/x64')
panel=B/'work/github-experimental-game-install-20261024/package'
paths={'nrvk_bridge.dll':game/'nrvk_bridge.dll','nr_bridge.ini':game/'plugins/nr_bridge.ini',
       'nr_panel.ini':panel/'config/nr_panel.ini','panel-config.json':panel/'panel/panel-config.json'}
def sha(p):return hashlib.sha256(p.read_bytes()).hexdigest()
def dump(p,v):p.write_text(json.dumps(v,ensure_ascii=False,indent=2),encoding='utf-8')
def copy_installed(src,dst):
    subprocess.run(['powershell','-NoProfile','-ExecutionPolicy','Bypass','-File',str(Q/'copy_file.ps1'),'-Source',str(src),'-Destination',str(dst)],check=True)
def stopped():
    r=subprocess.run(['tasklist','/FI','IMAGENAME eq Cyberpunk2077.exe','/FO','CSV','/NH'],capture_output=True)
    if b'cyberpunk2077.exe' in r.stdout.lower():raise RuntimeError('Exit Cyberpunk normally first')
def preserved():
    return {str(p):sha(p) for p in [game/'plugins/CyberpunkNRBridge.asi',
        source/'bin/nrvk_bridge.dll',A/'game-latest-20261028/small/bin/nrvk_bridge.dll',
        A/'fullnet-continuous-20261027/backend-merged/plan.txt']}
def prepare():
    P.mkdir(exist_ok=False)
    for name in ['assets','manifest']:shutil.copytree(source/name,P/name)
    (P/'bin').mkdir()
    for name in ['nrvk_bridge.dll','port_network_bridge_backend.dll']:
        shutil.copy2(Q/'capture/build/Release'/name,P/'bin'/name)
    (P/'manifest/capture-plan-sha256.txt').write_text(sha(Q/'capture/capture-source-plan.txt')+'\n',encoding='ascii')
    m=json.loads((P/'manifest/port_network_manifest.json').read_text(encoding='utf-8'))
    checked=0
    for section in ['network_shaders','bridge_shaders','cold']:
        for item in m[section].values():
            f=P/'manifest'/item['path']
            if f.stat().st_size!=item['bytes'] or sha(f)!=item['sha256']:raise RuntimeError(str(f))
            checked+=1
    dump(P/'PACKAGE.json',{'status':'PREPARED_NOT_MOUNTED','assets_checked':checked,
        'capture_dll_sha256':sha(P/'bin/nrvk_bridge.dll'),'preserved':preserved(),
        'runtime_sha256':sha(P/'manifest/port_network_runtime.txt'),
        'manifest_sha256':sha(P/'manifest/port_network_manifest.json')})
    print('PACKAGE_PREPARED',checked)
def edit_ini(path,section,key,value):
    c=configparser.ConfigParser();c.read(path,encoding='utf-8-sig');c[section][key]=value
    staged=P/('staged-'+path.name)
    with staged.open('w',encoding='utf-8',newline='\n') as f:c.write(f)
    copy_installed(staged,path)
def install():
    stopped()
    package=json.loads((P/'PACKAGE.json').read_text(encoding='utf-8'))
    if preserved()!=package['preserved']:raise RuntimeError('accepted identities changed since preparation')
    if (P/'manifest/capture.request').exists():raise RuntimeError('capture must not be armed at startup')
    backup=P/'before'
    if (P/'BEFORE.json').exists():
        before=json.loads((P/'BEFORE.json').read_text(encoding='utf-8'))
        for name,path in paths.items():
            if sha(backup/name)!=before[name]['sha256']:raise RuntimeError('saved backup changed')
            allowed=[before[name]['sha256']]
            if name=='nrvk_bridge.dll':allowed.append(package['capture_dll_sha256'])
            if sha(path) not in allowed:raise RuntimeError('unexpected partial-install state')
    else:
        backup.mkdir(exist_ok=False);before={}
        for name,path in paths.items():
            shutil.copy2(path,backup/name);before[name]={'path':str(path),'sha256':sha(path)}
        dump(P/'BEFORE.json',before)
    copy_installed(P/'bin/nrvk_bridge.dll',paths['nrvk_bridge.dll'])
    edit_ini(paths['nr_bridge.ini'],'bridge','manifest_dir',(P/'manifest').as_posix())
    edit_ini(paths['nr_panel.ini'],'NR','backend','102')
    edit_ini(paths['nr_panel.ini'],'Backends','102',(P/'manifest').as_posix())
    settings=json.loads(paths['panel-config.json'].read_text(encoding='utf-8-sig'))
    settings['vk102_manifest']=(P/'manifest').as_posix()
    dump(paths['panel-config.json'],settings)
    if sha(paths['nrvk_bridge.dll'])!=package['capture_dll_sha256']:raise RuntimeError('mounted DLL mismatch')
    if preserved()!=package['preserved']:raise RuntimeError('accepted artifacts were modified')
    dump(P/'INSTALL.json',{'status':'MOUNTED_GAME_NOT_STARTED','before':before,
        'after':{name:sha(path) for name,path in paths.items()},'preserved':preserved(),
        'selected':'102 / Full capture','capture_armed':False})
    print('CAPTURE_MOUNTED; game not started; capture not armed')
def restore():
    stopped()
    receipt=json.loads((P/'INSTALL.json').read_text(encoding='utf-8'))
    current=sha(paths['nrvk_bridge.dll'])
    if current not in [receipt['after']['nrvk_bridge.dll'],receipt['before']['nrvk_bridge.dll']['sha256']]:
        raise RuntimeError('game DLL changed by another operation; refusing to overwrite')
    for name,path in paths.items():
        saved=P/'before'/name
        if sha(saved)!=receipt['before'][name]['sha256']:raise RuntimeError('backup identity mismatch')
        copy_installed(saved,path)
        if sha(path)!=receipt['before'][name]['sha256']:raise RuntimeError('restore identity mismatch')
    dump(P/'RESTORED.json',{'status':'ORIGINAL_FILES_BYTE_RESTORED','files':{name:sha(path) for name,path in paths.items()},'preserved':preserved()})
    print('ORIGINAL_FILES_BYTE_RESTORED')
if __name__=='__main__':
    ap=argparse.ArgumentParser();ap.add_argument('mode',choices=['prepare','install','restore']);args=ap.parse_args()
    {'prepare':prepare,'install':install,'restore':restore}[args.mode]()
