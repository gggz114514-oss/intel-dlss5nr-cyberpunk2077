def scan(n, tag):

    m = ST.Module(Path(row['spv']))
    from collections import Counter
    c = Counter(op for op, _, _, _ in m.instructions)
    print('==', n, tag, row['spv_sha256'][:12], 'instr', len(m.instructions))
    print('   ', {OPNAME.get(k, k): v for k, v in sorted(c.items()) if k in OPNAME})
    print('    funcs', len(ST.functions(m)) if hasattr(ST, 'functions') else '?')
    # names of functions
    for op, ops, w, _c in m.instructions:
        if op == 54:
            print('     OpFunction result id', ops[1] if len(ops) > 1 else ops)
    # named
    for op, ops, w, _c in m.instructions:
        if op == 5:  # OpName
            pass
