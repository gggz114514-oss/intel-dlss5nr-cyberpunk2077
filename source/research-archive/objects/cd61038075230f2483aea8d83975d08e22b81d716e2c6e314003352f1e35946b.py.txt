"""Fair test of the X-equalised arm: re-tune the four scalar steps on top of the
power-of-two diagonal, so the comparison against the scalar-only arm is honest."""
import json
import sys
from pathlib import Path

import numpy as np

B = Path("@NR_WORKSPACE@")
W = B / "work/dsh-fullnet-static-20261029"
sys.path.insert(0, str(W / "tools"))
sys.path.insert(0, str(W / "S07-split16/tools"))
from calibrate import Chain  # noqa: E402
from equalise import requant, diag_from  # noqa: E402
import model as M  # noqa: E402

OUT = W / "S07-split16"


def main():
    ch = Chain()
    cal = json.loads((OUT / "CALIBRATION.json").read_text())
    s0 = dict(cal["new"]["scales"])
    Xa = np.zeros(512, np.float32)
    for fr in ch.frames:
        Xa = np.maximum(Xa, np.abs(fr["X"][:1056]).max(0))
    D1 = diag_from(Xa)
    q1, s1 = requant(ch.pk.W1, D1)
    w = {"W1": q1, "sW1": s1}

    def score(s):
        ys, fs = [], []
        for fr in ch.frames:
            F, A, Y, *_ = ch.forward(fr, s, dx=D1, w=w)
            ys.append(M.rel_l2(Y, fr["Y"]))
            fs.append(M.rel_l2(F, fr["F"]))
        return float(np.mean(np.square(ys))), float(np.mean(ys)), float(np.mean(fs)), ys

    cur = dict(s0)
    best = score(cur)
    print("start", best[:3], flush=True)
    for sweep in range(2):
        for dim in ["sX", "sR", "sA", "sF"]:
            opts = []
            for f in [0.5, 0.7, 0.85, 1.0, 1.2, 1.45]:
                trial = dict(cur)
                trial[dim] = cur[dim] * f
                m, my, mf, ys = score(trial)
                opts.append((m, trial, my, mf))
            m, cur, my, mf = min(opts, key=lambda z: z[0])
            print(f"xeq sweep{sweep} {dim} -> {cur[dim]:.6g} Y={my:.6e} F={mf:.6e}", flush=True)
    m, my, mf, ys = score(cur)
    out = {"arm": "X_equalised_retuned", "D1": D1.tolist(), "scales": cur,
           "mse_Y": m, "mean_Y_rel_l2": my, "mean_F_rel_l2": mf, "per_frame_Y": ys,
           "scalar_only_new": cal["new"]["mean_Y_rel_l2"], "scalar_only_new_F": cal["new"]["mean_F_rel_l2"],
           "requant_rel_l2": float(np.linalg.norm(q1 * s1[:, None] - ch.pk.W1) / np.linalg.norm(ch.pk.W1))}
    (OUT / "XEQ-RETUNE.json").write_text(json.dumps(out, indent=1))
    print(json.dumps(out, indent=1))


if __name__ == "__main__":
    main()
