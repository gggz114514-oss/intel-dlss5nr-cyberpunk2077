import sys
from pathlib import Path
import numpy as np

sys.path.insert(0, str(Path(__file__).parent))
import c32blk as M
from validate import bk, PUSH, ptr

B = Path(r"@NR_WORKSPACE@")
W = B / "work/dsh-fullnet-static-20261029"
src = W / "C00-color-baseline/runs/diag65-69"
tea = B / "work/codex-astra-fp16-int8-audit-20261024/int8-quality-first-20261029/real-int8-24/dump-fp16"

# teacher input -> model -> compare teacher block67 out
x, valid = M.load_block(tea / "dump_block66_out.bin", -1, -1)
blk = M.Block(bk("half", 67), bk("i8", 67), bk("sc", 67), bk("aux", 67), ptr(PUSH))
y, _ = blk.forward(x, valid)
idx, vv = M.token_index(-1, -1)
tch = np.fromfile(tea / "dump_block67_out.bin", dtype="<f2")[idx][vv].astype(np.float64)
pred = y[vv].astype(np.float64)
print("model(teacher in) vs teacher67:", M.rel_l2(pred, tch), M.corr(pred, tch))

# int8 input -> model; histogram comparison with gpu
x2, valid2 = M.load_block(src / "c00-block66_out.bin", -1, -1)
y2, _ = blk.forward(x2, valid2)
gpu = np.fromfile(src / "c00-block67_out.bin", dtype="<f2")[idx][vv].astype(np.float64)
p2 = y2[vv].astype(np.float64)
print("model(int8 in) vs gpu:", M.rel_l2(p2, gpu), M.corr(p2, gpu))
qs = [0, 1, 5, 25, 50, 75, 95, 99, 100]
print("quantiles gpu ", np.round(np.percentile(gpu, qs), 2).tolist())
print("quantiles pred", np.round(np.percentile(p2, qs), 2).tolist())
print("rms gpu/pred", np.sqrt((gpu ** 2).mean()), np.sqrt((p2 ** 2).mean()))
print("mean abs gpu/pred", np.abs(gpu).mean(), np.abs(p2).mean())
# best global scale
a = float((p2 * gpu).sum() / (p2 * p2).sum())
print("best global scale", a, "relL2 after scale", M.rel_l2(p2 * a, gpu))
