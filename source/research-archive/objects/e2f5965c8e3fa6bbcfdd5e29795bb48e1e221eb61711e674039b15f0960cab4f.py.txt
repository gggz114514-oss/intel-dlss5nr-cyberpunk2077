"""Print ISA instructions in an address range (CPU only, read-only ISA).

Usage: python dump_range.py 0x0200 0x0400
"""
import sys
from pathlib import Path

CARD = Path(__file__).resolve().parents[1]
sys.path.insert(0, str(CARD))
from tools.isa_chain import ISA_PATH, load  # noqa: E402


def main():
    lo = int(sys.argv[1], 16)
    hi = int(sys.argv[2], 16)
    isa, _ = load()
    for ins in isa:
        if lo <= ins['addr'] < hi:
            print('%04X %s' % (ins['addr'], ins['raw'].rstrip()))


if __name__ == '__main__':
    main()
