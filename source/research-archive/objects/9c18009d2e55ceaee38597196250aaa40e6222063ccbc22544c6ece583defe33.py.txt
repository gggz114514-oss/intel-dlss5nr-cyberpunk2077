"""Parent vs candidate SPIR-V diff for the n24 fold (CPU only).

Reports, for each of the two PASSes:
  * instruction multiset diff (op, operands) -- everything that changed
  * opcode multiset diff (op only)
  * NoContraction (OpDecorate ... 44) count and MEMBER-agnostic float op counts
  * the constants whose value changed

Purpose: prove the candidate differs from its parent only in the gM address
arithmetic / row stride of the two folded stores (producer) or two folded loads
(consumer), and that every floating-point operation is untouched.
"""
import collections
import json
import sys
from pathlib import Path

CARD = Path(__file__).resolve().parent
sys.path.insert(0, str(CARD))
import spv_access as S      # reuse the exact instruction parser

NO_CONTRACTION = 44
FLOATS = {127: 'OpFNegate', 129: 'OpFAdd', 131: 'OpFSub', 133: 'OpFMul',
          136: 'OpFDiv', 138: 'OpFRem', 140: 'OpFMod', 142: 'OpVectorTimesScalar',
          144: 'OpMatrixTimesScalar', 46: 'OpFmaKHR?', 180: 'OpFConvert',
          168: 'OpFConvert16', 126: 'OpBitcast'}
INT_CTRL = {128, 130, 132, 134, 135, 137, 138, 139}


def load(path):
    ops = S.parse(path)
    return ops


def nc_count(ops):
    n = 0
    for op, a in ops:
        if op == 71 and len(a) >= 3 and a[1] == NO_CONTRACTION:
            n += 1
    return n


def inst_key(op, a):
    return (op, tuple(a))


def diff(parent_path, cand_path):
    po, co = load(parent_path), load(cand_path)
    pk = collections.Counter(inst_key(op, a) for op, a in po)
    ck = collections.Counter(inst_key(op, a) for op, a in co)
    removed = pk - ck
    added = ck - pk
    pop = collections.Counter(op for op, a in po)
    cop = collections.Counter(op for op, a in co)
    op_delta = {str(k): [pop.get(k, 0), cop.get(k, 0)]
                for k in set(pop) | set(cop) if pop.get(k, 0) != cop.get(k, 0)}
    return {
        'parent': str(parent_path), 'candidate': str(cand_path),
        'parent_instructions': len(po), 'candidate_instructions': len(co),
        'parent_nc': nc_count(po), 'candidate_nc': nc_count(co),
        'removed_instructions': [{'op': k[0], 'operands': list(k[1]), 'count': v}
                                 for k, v in removed.items()],
        'added_instructions': [{'op': k[0], 'operands': list(k[1]), 'count': v}
                               for k, v in added.items()],
        'opcode_count_delta': op_delta,
        'removed_total': sum(removed.values()), 'added_total': sum(added.values()),
    }


def main():
    card = CARD
    pairs = [('ord_193', card / 'parent-rebuild-193.spv', card / 'candidate-193.spv'),
             ('ord_194', card / 'parent-rebuild-194.spv', card / 'candidate-194.spv')]
    out = {'card': 'n24-s5-gm-fold', 'passes': {}}
    for name, p, c in pairs:
        assert p.exists() and c.exists(), (p, c)
        out['passes'][name] = diff(p, c)
    txt = json.dumps(out, indent=2)
    (card / 'spv-diff.json').write_text(txt, encoding='utf-8')
    for name, d in out['passes'].items():
        print('===', name, 'instructions', d['parent_instructions'], '->', d['candidate_instructions'],
              'NC', d['parent_nc'], '->', d['candidate_nc'],
              'removed', d['removed_total'], 'added', d['added_total'])
        print('   opcode deltas:', d['opcode_count_delta'])
        for r in d['removed_instructions']:
            print('   -', r['count'], 'x op', r['op'], r['operands'])
        for r in d['added_instructions']:
            print('   +', r['count'], 'x op', r['op'], r['operands'])
    return 0


if __name__ == '__main__':
    sys.exit(main())
