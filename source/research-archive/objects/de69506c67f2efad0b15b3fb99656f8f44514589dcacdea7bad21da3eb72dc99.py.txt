"""Issue a bounded request to the private capture backend, after deployment verification."""
import argparse
import json
from pathlib import Path
ap=argparse.ArgumentParser()
ap.add_argument('--manifest',type=Path,required=True)
ap.add_argument('--output',type=Path,required=True)
ap.add_argument('--scene',required=True)
ap.add_argument('--split',choices=['calibration','holdout','smoke'],required=True)
ap.add_argument('--frames',type=int,default=8)
args=ap.parse_args()
if not 1<=args.frames<=32:raise ValueError('frames must be 1..32')
if not (args.manifest/'capture-plan-sha256.txt').is_file():
    raise ValueError('private capture manifest identity is missing')
if (args.manifest/'capture.request').exists():raise ValueError('a request is already pending')
args.output.mkdir(parents=True,exist_ok=False)
(args.output/'sequence.json').write_text(json.dumps({
    'schema':1,'scene_label':args.scene,'split':args.split,'requested_frames':args.frames,
    'status':'REQUESTED_NOT_ACCEPTED','source_plan_sha256':(args.manifest/'capture-plan-sha256.txt').read_text().strip(),
    'note':'Raw inputs/history required. Calibration and holdout must be independent scenes/clips.'
},indent=2),encoding='utf-8')
# Publish by rename so the rendering thread cannot read a half-written request.
tmp=args.manifest/'capture.request.tmp'
with tmp.open('x',encoding='utf-8',newline='\n') as f:
    f.write(args.output.resolve().as_posix()+'\n'+str(args.frames)+'\n')
tmp.rename(args.manifest/'capture.request')
print('CAPTURE_REQUESTED',args.output)
