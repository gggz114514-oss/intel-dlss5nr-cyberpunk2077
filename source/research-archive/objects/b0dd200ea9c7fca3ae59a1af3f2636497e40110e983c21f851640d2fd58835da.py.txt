"""n36-f05-arena-shrink: assemble BUILD.json from the machine artifacts.  CPU only."""
import hashlib
import json
import pathlib
import shutil

CARD = pathlib.Path(__file__).resolve().parent
TIER1_SPV = CARD / 'candidate-c256.spv'
CANON = CARD / 'candidate.spv'


def sha(p):
    return hashlib.sha256(pathlib.Path(p).read_bytes()).hexdigest()


def read(name):
    return json.loads((CARD / 'build' / name).read_text(encoding='utf-8'))


def main():
    # canonical candidate.spv = the tier-1 (C256) candidate, byte-identical copy, so the
    # campaign helpers (COMMON.rebase_candidate_plan / make_plan) work unchanged.
    shutil.copyfile(TIER1_SPV, CANON)
    assert sha(CANON) == sha(TIER1_SPV)

    repro = read('repro.json')
    build = read('build.json')
    plan = read('plan.json')
    life = read('lifetime.json')
    rows = read('rows.json')
    verify = read('verify.json')

    out = dict(
        card='n36-f05-arena-shrink',
        family='F05',
        status='BUILT_AWAITING_GPU',
        date='2026-10-28',
        change=('shared workgroup memory 24576 B -> 16384 B: the V projection is relocated '
                'after the last Q/K read and stored into the retired Q plane, and its token '
                'ownership moves from {8t,8t+32} to {16t,16t+16). Grid, token tile, DPAS '
                'count, SLM message count and every FP expression are unchanged.'),
        delivery=dict(
            tier1=dict(label='PRIMARY (C256 only, 16 PASS, zero extra UGM messages)',
                       plan=plan['tiers']['tier1-c256']['plan'],
                       per_group=plan['tiers']['tier1-c256']['per_group'],
                       cpu_check=plan['tiers']['tier1-c256']['cpu_check']),
            tier2=dict(label='SECONDARY (36 PASS, C64/C128 pay +50% UGM; built, not dispatched)',
                       plan=plan['tiers']['tier2-36']['plan'],
                       per_group=plan['tiers']['tier2-36']['per_group'],
                       cpu_check=plan['tiers']['tier2-36']['cpu_check'])),
        baseline_plan=dict(path=plan['baseline']['path'],
                           sha256=plan['baseline']['sha256'],
                           sha256_ok=plan['baseline']['verified'],
                           label='chain head: accepted/n29-f01-alu-chain-20261027/plan.txt'),
        canonical_candidate=dict(
            path=str(CANON), sha256=sha(CANON),
            note='byte-identical copy of candidate-c256.spv, i.e. the tier-1 candidate; '
                 'candidate-c64.spv / candidate-c128.spv are the other two groups'),
        groups={},
        plan=dict(tier1=plan['tiers']['tier1-c256']['plan'],
                  tier2=plan['tiers']['tier2-36']['plan']),
        cpu_check=dict(tier1=plan['tiers']['tier1-c256']['cpu_check'],
                       tier2=plan['tiers']['tier2-36']['cpu_check'],
                       baseline=plan['cpu_check_baseline'],
                       shared_field_unsynced_control=dict(
                           tier1=plan['tiers']['tier1-c256']['cpu_check_shared_unsynced'],
                           tier2=plan['tiers']['tier2-36']['cpu_check_shared_unsynced'],
                           reading='both tiers still pass --cpu-check with the ledger field '
                                   'left at 24576, so the field is a ledger value and not an '
                                   'allocation input (same conclusion as n30 1.1). It is '
                                   'synced anyway because the card requires the plan to '
                                   'agree with the module.')),
        gpu='none in this card (no Vulkan task, no lock taken)')

    for g, rec in repro.items():
        b = build[g]
        v = verify[g]
        out['groups'][g] = dict(
            passes=rec['passes'],
            parent=dict(spv=rec['spv'], source=rec['source'],
                        byte_reproduction=dict(
                            argv=rec['argv'], raw_sha256=rec['raw_sha256'],
                            raw_match=rec['raw_match'], recipe='raw',
                            note='plain glslang 16.5.0 -V --target-env vulkan1.3, no '
                                 'NoContraction transfer needed'),
                        rebuild_spv=rec.get('parent_rebuild')),
            include_headers=rec['headers'],
            candidate=dict(source=b['source'], spv=b['spv'],
                           compile_argv=b['compile_argv'],
                           validate=b['validate']),
            arithmetic_alignment=dict(
                no_contraction=v['no_contraction'],
                float_NC_checked=v['no_contraction']['candidate'],
                fp_multiset=b['fp_multiset'],
                int_control_flow=v['int_control_flow'],
                int_control_flow_total=v['int_control_flow_total'],
                private_storage_variables=v['private_storage_variables'],
                note='the FP expression multiset is identical site for site and the '
                     'NoContraction multiset (function, opcode, operand signature) is '
                     'identical, so no rounding point or contraction policy changed. The '
                     'integer control-flow delta is the moved block\'s address arithmetic '
                     'and is reported separately, as CONTRACT.md 2 requires.'),
            instruction_delta=dict(module_words=v['module_words'],
                                   instructions=v['instructions'],
                                   opcode_delta=b['opcode_delta'],
                                   dpas=b['dpas']),
            workgroup_storage=b['storage_bytes'],
            barriers=b['barriers'],
            coopmat=b['coopmat'],
            dynamics=life['dynamic_messages'][g],
            plan_rows=[dict(plan_line=r['plan_line'], label=r['label'], name=r['name'],
                            grid=r['grid'], workgroups=r['workgroups'], shared=r['shared'])
                       for r in rows['rows'] if r['group'] == g])

    out['workgroups'] = dict(per_group={g: rows['per_group'][g]['workgroups']
                                       for g in rows['per_group']},
                             total_36=rows['total_workgroups'],
                             messages_delta_per_frame={
                                 g: rows['per_group'][g]['ugm_messages_delta_per_frame']
                                 for g in rows['per_group']},
                             total_messages_delta_per_frame_36=rows[
                                 'total_ugm_messages_delta_per_frame'])
    (CARD / 'BUILD.json').write_text(json.dumps(out, indent=2), encoding='utf-8')
    print('canonical candidate.spv', sha(CANON)[:16])
    for g in out['groups']:
        print('%-5s parent %s  cand %s  wg %d->%d  NC %d'
              % (g, out['groups'][g]['parent']['spv']['sha256'][:16],
                 out['groups'][g]['candidate']['spv']['sha256'][:16],
                 out['groups'][g]['workgroup_storage']['parent']['Workgroup']['bytes'],
                 out['groups'][g]['workgroup_storage']['candidate']['Workgroup']['bytes'],
                 out['groups'][g]['arithmetic_alignment']['float_NC_checked']))


if __name__ == '__main__':
    main()
