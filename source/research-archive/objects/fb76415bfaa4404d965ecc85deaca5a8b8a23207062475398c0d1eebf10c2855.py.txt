"""Compare the F13 expand native against the F13 contract native (both CURRENT).

Purpose: the contract kernel already hoists its live guards outside the K loop, so it is
the natural counter-example for the "in-loop join causes the accumulator copies" claim.
"""
import json, sys, collections, re
from pathlib import Path
sys.path.insert(0, r'@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/fullnet-continuous-20261027/cards/n11-vit-ffn/_tools')
from isa_scan import parse

CASES = {
    'expand': dict(spv='54029e6beb917f447e71add99a8b61c196d3e6428cc1df0113db82a00cb9c2a2',
                   isa=r'@NR_WORKSPACE@/run/dsh-fp16-same-family-20261023/nat-k11-s32-native-000/gtpin/ISA/CS_asm48ac0230660a2c52_simd32_0000000000000001_0_orig.asm',
                   guard='inside the K loop (if(mA<live){...} if(mB<live){...})',
                   accs=4, unrolled=True, ordinals=[93, 98, 103, 108, 113, 118, 123, 128]),
    'contract': dict(spv='bc629c9b8e24651dde19de3cba45520b5c5fa899bb991ceb69dac559675ac23f',
                     isa=r'@NR_WORKSPACE@/run/codex-fp16-continuation-20261021/C/ct01-f17-native-001/gtpin/ISA/CS_asm479f1bb119e3b522_simd32_0000000000000002_1_orig.asm',
                     guard='outside the K loop (if(mA<live&&mB<live){...} else {guarded loop})',
                     accs=4, unrolled=False, ordinals=[94, 99, 104, 109, 114, 119, 124, 129]),
    'projection': dict(spv='ae5cb966cf4ec2f3134fd45769c32f83067d24cc81bbb3f7a1588280b5933546',
                       isa=r'@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/fullnet-mechanism-transfer-20261026/f08-register-chain-20261027/captures/native-f20-pixel-owner/ISA/CS_asm61c5144f4358fd6f_simd32_000000000000001b_26_orig.asm',
                       guard='inside the K loop (if(mA<live){...} if(mB<live){...})',
                       accs=2, unrolled=True, ordinals=[97, 102, 107, 112, 117, 122, 127, 132]),
}

out = {}
for name, c in CASES.items():
    ins = parse(c['isa'])
    cls = collections.Counter()
    for i in ins:
        t = i['text']
        if i['op'].startswith('send'):
            cls[i['op'] + ('.wide16' if '(16|M0)' in t else '.narrow1')] += 1
        elif i['op'] == 'mov':
            m = re.search(r'\((\d+)\|M0\)', t)
            cls['mov(%s)' % (m.group(1) if m else '?')] += 1
        else:
            cls[i['op']] += 1
    dpas = cls['dpas.8x8']
    out[name] = dict(
        spv_sha256=c['spv'], isa=c['isa'], isa_sha256=__import__('hashlib').sha256(Path(c['isa']).read_bytes()).hexdigest(),
        isa_bytes=Path(c['isa']).stat().st_size,
        guard_location=c['guard'], accumulators=c['accs'], fully_unrolled=c['unrolled'],
        ordinals=c['ordinals'],
        instructions=len(ins), dpas=dpas,
        mov32=cls['mov(32)'], mov8=cls['mov(8)'], mov1=cls['mov(1)'],
        jmpi=cls['jmpi'], sync=cls['sync.nop'],
        sends=cls['send.ugm.wide16'] + cls['send.ugm.narrow1'],
        wide=cls['send.ugm.wide16'], narrow=cls['send.ugm.narrow1'],
        instr_per_dpas=round(len(ins) / dpas, 2),
        mov32_per_dpas=round(cls['mov(32)'] / dpas, 2),
    )
print(json.dumps(out, indent=1))
Path(r'@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/fullnet-continuous-20261027/cards/n11-vit-ffn/VIT-FFN-NATIVE-COMPARISON.json').write_text(
    json.dumps(out, indent=1), encoding='utf-8')
