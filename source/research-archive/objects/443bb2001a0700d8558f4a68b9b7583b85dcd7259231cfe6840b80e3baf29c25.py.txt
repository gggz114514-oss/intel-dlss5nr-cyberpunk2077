"""Final acceptance sweep for the card (CPU only): plan diff, cpu-check, shas, JSON validity."""
import json
import hashlib
import pathlib
import subprocess
import sys

CARD = pathlib.Path(__file__).resolve().parent
CAMP = pathlib.Path(r'@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/fullnet-continuous-20261027')
RUNNER = pathlib.Path(r'@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/'
                      r'phase-diagnosis-20261025/direct-clock-build/Release/clock_runner.exe')


def sha(p):
    return hashlib.sha256(pathlib.Path(p).read_bytes()).hexdigest()


build = json.loads((CARD / 'BUILD.json').read_text(encoding='utf-8'))
head = pathlib.Path(build['plan']['parent_plan']['path'])
plan = CARD / 'plan.txt'

a = head.read_text(encoding='utf-8-sig').splitlines()
b = plan.read_text(encoding='utf-8-sig').splitlines()
diff = [i for i, (x, y) in enumerate(zip(a, b)) if x != y]
print('chain head      :', head)
print('head sha        :', build['plan']['parent_plan']['sha256'], '->', sha(head))
print('plan sha        :', sha(plan))
print('line counts     :', len(a), len(b), 'row0:', b[0])
print('differing lines :', diff, '(0-based); PASS-row index of ord 230 =', diff == [2987])
print('ord230 new line :', b[2987][:120], '...')
print('ord230 spv ok   :', str(CARD / 'candidate.spv').replace('\\', '/') in b[2987])
print('head ord230 line:', a[2987][:120], '...')

r = subprocess.run([str(RUNNER), '--plan', str(plan), '--cpu-check'], capture_output=True, text=True)
r2 = subprocess.run([str(RUNNER), '--plan', str(head), '--cpu-check'], capture_output=True, text=True)
print('cpu-check cand  :', r.returncode, (r.stdout or '').strip())
print('cpu-check base  :', r2.returncode, (r2.stdout or '').strip())
print('cpu-check equal :', (r.stdout or '').strip() == (r2.stdout or '').strip())

print('\n-- deliverable shas --')
need = ['CARD.txt', 'ANALYSIS.md', 'STAGE-BLOCK-MATRIX.md', 'PROOF.json', 'SOURCE-DIFF.patch',
        'COST-LEDGER.json', 'NATIVE-PREDICTION.md', 'BUILD.json', 'GPU-REQUEST.json', 'RESUME.txt',
        'candidate.comp', 'candidate.spv', 'parent.comp', 'parent-rebuild.spv', 'plan.txt',
        'build/arena-check.json', 'build/spv-check.json', 'build/isa.json',
        'build.py', '_spv_check.py', '_isa.py', '_arena_check.py', '_proof.py']
missing = []
for n in need:
    p = CARD / n
    if not p.exists():
        missing.append(n)
        continue
    print('  %-26s %8d  %s' % (n, p.stat().st_size, sha(p)))
print('missing:', missing or 'none')

print('\n-- JSON validity --')
for n in ('PROOF.json', 'COST-LEDGER.json', 'GPU-REQUEST.json', 'BUILD.json',
          'build/arena-check.json', 'build/spv-check.json', 'build/isa.json'):
    d = json.loads((CARD / n).read_text(encoding='utf-8'))
    print('  %-26s parses (%d top keys)' % (n, len(d)))

print('\n-- cross-checks --')
proof = json.loads((CARD / 'PROOF.json').read_text(encoding='utf-8'))
ar = json.loads((CARD / 'build/arena-check.json').read_text(encoding='utf-8'))
sp = json.loads((CARD / 'build/spv-check.json').read_text(encoding='utf-8'))
print('  proof slots            :', proof['proofs']['P2_element_identity_is_bijective']['slots_checked'],
      'mismatches', proof['proofs']['P2_element_identity_is_bijective']['mismatches'])
print('  WAR cross parent->cand :', ar['hazards']['stage3_read -> stage4_write (PARENT layout)']['cross_owner_matches'],
      '->', ar['hazards']['stage3_read -> stage4_write (CANDIDATE layout)']['cross_owner_matches'])
print('  source barriers        :', ar['barriers']['parent']['static'], '->', ar['barriers']['candidate']['static'],
      '| normal path', ar['barriers']['parent']['normal_path'], '->', ar['barriers']['candidate']['normal_path'])
print('  spirv barriers         :', sp['workgroup_barriers']['parent'], '->', sp['workgroup_barriers']['candidate'],
      '| scope_relaxed', sp['workgroup_barriers']['scope_relaxed'])
print('  wg storage             :', sp['storage_bytes']['parent']['Workgroup']['bytes'], '->',
      sp['storage_bytes']['candidate']['Workgroup']['bytes'], '| identical', sp['storage_bytes']['identical'])
print('  build.json cpu_check   :', build['cpu_check']['stdout'])
print('  plan in build.json     :', build['plan']['sha256'] == sha(plan),
      '| spv in build.json', build['candidate']['spv_sha256'] == sha(CARD / 'candidate.spv'))
print('  gpu_used/deployed      :', build['gpu_used'], build['deployed'])
