def read(p):
 return json.loads(Path(p).read_text(encoding='utf-8-sig'))
def save(p,v):
 Path(p).write_text(json.dumps(v,indent=2,ensure_ascii=False),encoding='utf8')
def tokenize(l):
 return [a or b for a,b in re.findall(r'"([^"]*)"|(\S+)',l)]
def parse(path):
 bs=[];ps=[];tail=[]
 for l in Path(path).read_text(encoding='utf-8-sig').splitlines():
  t=tokenize(l)
  if not t:continue
  if t[0]=='PORT_NETWORK_1':header=t
  elif t[0]=='BUFFER':bs.append(dict(name=t[1],size=int(t[2]),path=t[3]))
  elif t[0]=='PASS':ps.append(dict(row=int(t[1]),name=t[2],spv=t[3],sg=int(t[4]),shared=int(t[5]),grid=list(map(int,t[8:11])),binds=[],push=[],ordinal=len(ps)))
  elif t[0]=='BIND':ps[-1]['binds'].append(list(map(int,t[1:])))
  elif t[0]=='PUSH':ps[-1]['push']=list(map(int,t[1:]))
  else:tail.append(t)
 return dict(header=header,buffers=bs,passes=ps,tail=tail)

def literal(name):
 return next(ast.literal_eval(n.value) for n in tree.body if isinstance(n,ast.Assign) and any(isinstance(t,ast.Name) and t.id==name for t in n.targets))
TEMPLATE=literal('TEMPLATE');BLOCK_MAP=literal('BLOCK_MAP');EXACT_RENAME=literal('EXACT_RENAME')
def rename(name):
 if name in EXACT_RENAME:return EXACT_RENAME[name]

  m=re.fullmatch(pattern,name)
  if m:return fmt.format(BLOCK_MAP[int(m[1])],*m.groups()[1:])
 raise ValueError(name)
