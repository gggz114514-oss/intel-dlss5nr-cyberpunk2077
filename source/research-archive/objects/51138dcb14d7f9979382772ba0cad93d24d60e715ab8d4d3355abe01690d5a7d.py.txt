import sys, json, re, collections
from pathlib import Path

CARD = Path(__file__).resolve().parent
C = CARD.parent.parent
sys.path.insert(0, str(C))
import COMMON as K
ST = K.ST

HEAD, HEAD_SHA, _ = K.current_best()
cp_head = K.cpu_check(HEAD)
print('parent cpu-check:', cp_head.strip())

# ---- store classification: which cooperative-matrix stores survive?
HPP = Path('@NR_WORKSPACE@/src/project/tools/glslang-16.5.0/include/glslang/SPIRV/spirv.hpp11')
NAMES = {}
for m in re.finditer(r'\b(Op[A-Za-z0-9_]+)\s*=\s*(\d+)', HPP.read_text(encoding='utf-8', errors='replace')):
    NAMES.setdefault(int(m.group(2)), m.group(1))
RESULT_AT_ZERO = tuple(range(19, 40)) + (7, 11, 248, 4456, 5358)
NO_RESULT = (0, 2, 3, 4, 5, 6, 8, 10, 14, 15, 16, 17, 62, 63, 64, 71, 72, 73, 74, 75, 77,
             224, 225, 246, 247, 249, 250, 251, 252, 253, 254, 255, 4458, 5360)


def rid(op, ops):
    if not ops or op in NO_RESULT:
        return None
    return ops[0] if op in RESULT_AT_ZERO else (ops[1] if len(ops) > 1 else None)


def analyse(path):
    m = ST.Module(Path(path))
    defs, consts = {}, {}
    for i, (op, ops, _, _) in enumerate(m.instructions):
        r = rid(op, ops)
        if r is not None:
            defs.setdefault(r, i)
        if op == 43:
            consts[ops[1]] = ops[2]
    dbg = [i for i, (op, ops, _, _) in enumerate(m.instructions)
           if op == 43 and list(ops)[-1] == 81920]
    dbg_id = m.instructions[dbg[0]][1][1] if dbg else None
    rows = []
    for i, (op, ops, _, _) in enumerate(m.instructions):
        if op != 4458:                       # OpCooperativeMatrixStoreKHR
            continue
        ptr = ops[0]
        # walk the pointer chain: AccessChain indices -> any use of the dbg constant
        chain, hits, frontier = [], False, [ptr]
        seen = set()
        while frontier:
            v = frontier.pop()
            if v in seen:
                continue
            seen.add(v)
            if v == dbg_id:
                hits = True
            j = defs.get(v)
            if j is None:
                continue
            dop, dops, _, _ = m.instructions[j]
            chain.append(NAMES.get(dop, dop))
            if dop == 43:
                continue
            frontier.extend([x for x in dops if x in defs])
        rows.append(dict(site=i, uses_dbg_pointer=hits, chain_ops=sorted(set(chain))))
    return rows, consts


prow, _ = analyse(K.row_by_ordinal(194)['spv'])
crow, _ = analyse(CARD / 'candidate.spv')
print('parent stores:', len(prow), 'with dbg pointer:', sum(r['uses_dbg_pointer'] for r in prow))
print('cand   stores:', len(crow), 'with dbg pointer:', sum(r['uses_dbg_pointer'] for r in crow))

K.write_json(CARD / 'build' / 'store-classification.json', dict(
    parent_cpu_check=cp_head.strip(),
    candidate_cpu_check=(CARD / 'build' / 'cpu-check.txt').read_text(encoding='utf-8').strip(),
    parent=dict(stores=len(prow), dbg_pointer=sum(r['uses_dbg_pointer'] for r in prow), rows=prow),
    candidate=dict(stores=len(crow), dbg_pointer=sum(r['uses_dbg_pointer'] for r in crow), rows=crow),
))
