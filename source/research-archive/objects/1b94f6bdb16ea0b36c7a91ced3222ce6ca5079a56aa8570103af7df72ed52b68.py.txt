import json,struct,statistics as s,collections

D=Path(__file__).resolve().parent

positions=rd(D/'remaining-family/POSITIONS.json');wall={r['ordinal']:r for r in rd(D/'REMAINING-FULLNET-LEDGER.json')['rows']};cache={}
for rep in range(3):
 for a in ['control','whole','null','phase1','phase2','phase3']:
  p=D/'run'/f'remaining-family-r{rep}-{a}';r=rd(p/'port_network_receipt.json');cache[rep,a]={v['index']:v for v in r['per_pass_gpu']}

for pos in positions:
 phases=['phase'+str(i+1) for i in range(len(pos['phase_names']))];reps=[]
 for rep in range(3):
  arms={}
  for a in ['control','whole','null','phase1','phase2','phase3']:
   raw=(D/'run'/f'remaining-family-r{rep}-{a}'/pos['file']).read_bytes();assert len(raw)==pos['bytes'];ticks=[];by=[[] for _ in range(pos['sg'])];inactive=a=='control' or a=='phase3' and len(phases)==2
   for i,(x,y,z,w) in enumerate(struct.iter_unpack('<IIII',raw)):
    nz=bool(x or y or z or w);sel=not inactive and (i//pos['sg'])%16==0
    assert nz==sel,(pos['ordinal'],rep,a,i,nz,sel)
    if nz:
     t=((z+(w<<32))-(x+(y<<32)))%2**64;ticks.append(t);by[i%pos['sg']].append(t)
   arms[a]={'ticks':s.mean(ticks) if ticks else 0,'by_sg':[s.mean(v) if v else 0 for v in by],'records':len(ticks),'max':max(ticks) if ticks else 0,'ms':s.median(cache[rep,a][pos['ordinal']]['samples_ms'])}
  closure=sum(arms[a]['ticks'] for a in phases)/arms['whole']['ticks'];perturb={a:arms[a]['ms']/arms['control']['ms']-1 for a in ['whole','null']+phases};empty={a:arms['null']['ticks']/arms[a]['ticks'] for a in phases}
  reps.append({'rep':rep,'arms':arms,'closure':closure,'perturbation':perturb,'empty_fraction':empty})
 spread={a:max(r['arms'][a]['ticks'] for r in reps)/min(r['arms'][a]['ticks'] for r in reps)-1 for a in ['whole']+phases}

 active=list(range(pos['sg']))

 rows.append({**pos,'pass_interval_ms':wall[pos['ordinal']]['mean_pass_interval_ms'],'repeats':reps,'spread':spread,'failures':fails,'timing_screen_pass':not fails,'native_status':'PENDING','summary':summary})

print('TOTAL',len(rows),'pass',sum(r['timing_screen_pass'] for r in rows))
