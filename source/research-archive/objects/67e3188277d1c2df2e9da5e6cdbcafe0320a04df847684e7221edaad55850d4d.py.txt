"""R09 SPV identity: parent f3c64_body_ds vs the INT8-WO candidate.

CPU only, word level.  Five sections plus the NoContraction role analysis:

  P0 identity        files, sha256, bytes, and a probe of the family's
                     NoContraction-transfer step (run on the candidate as well,
                     to show whether it is numerically inert here)
  P1 interface       bindings, shared arrays, push-constant size, entry point,
                     execution mode, cooperative-matrix type table, functions
  P2 operand streams cooperative-matrix load/store streams compared one by one
                     on (slot, address expression, stride, layout, component,
                     rows, cols, use): the residual seed reads and the whole
                     store stream must be kept; the FP16 WO operand loads must be
                     removed and every added load must belong to a new stream
  P3 matrix sites    four FP16 MulAdd sites gone, four INT8 MulAdd sites added,
                     the two INT8 DS sites untouched, with A/B/C provenance
  P4 opcode deltas   every per-opcode count delta with its source-level reason
                     and, where it is mechanically determined, its derivation
  P5 decorations     NoContraction / binding / stride deltas, and the role
                     comparison of every NoContraction the parent carried

usage: python tools/spv_identity.py
"""

def norm(s):
    """Normalise an SSA-id-bearing symbol.

    `emit_qx` is a real function, so its store strides/addresses read push-constant
    values through OpLoad; those loads get fresh SSA ids in any recompile.  The
    *expression* is what must be compared, so id numbers are folded.  No other
    stream in this module uses a loaded stride (the four body stores use the
    literal 64/16), so the fold cannot hide a real change.
    """
    return re.sub(r"load#\d+", "load#*", str(s))


def sha(p):
    import hashlib
    return hashlib.sha256(Path(p).read_bytes()).hexdigest()


def slot_of(m, buf):
    return SLOT.get(buf, SLOT_EXTRA.get(buf, buf))


def stream(m):
    loads, stores = [], []
    for it in m.insts:
        if it.name == "OpCooperativeMatrixLoadKHR":
            bv = m.base_var(it.a[2])
            buf = m.block_name(bv) if bv is not None else None
            if buf is None:
                buf = m.var_name(bv) if bv is not None else "?"
            loads.append({"slot": slot_of(m, buf), "buffer": buf,
                          "address": m.resolve(it.a[2])[0],
                          "stride": m.fval(it.a[4]) if m.fval(it.a[4]) is not None
                          else m.resolve(it.a[4])[0],
                          "layout": "ColumnMajor" if m.fval(it.a[3]) == 1 else "RowMajor",
                          "cmat": m.cm_type(it.a[0])})
        elif it.name == "OpCooperativeMatrixStoreKHR":
            obj = it.a[1]
            bv = m.base_var(it.a[0])
            buf = m.block_name(bv) if bv is not None else None
            if buf is None:
                buf = m.var_name(bv) if bv is not None else "?"
            stores.append({"slot": slot_of(m, buf), "buffer": buf,
                           "address": m.resolve(it.a[0])[0],
                           "stride": m.fval(it.a[3]) if m.fval(it.a[3]) is not None
                           else m.resolve(it.a[3])[0],
                           "layout": "ColumnMajor" if m.fval(it.a[2]) == 1 else "RowMajor",
                           "cmat": m.cm_type(m.byid[obj].a[0]) if obj in m.byid else None})
    return loads, stores


def tup(r):
    c = r["cmat"] or {}
    return (r["slot"], norm(r["address"]), norm(r["stride"]), r["layout"],
            c.get("component"), c.get("rows"), c.get("cols"), c.get("use"))


def mma(m):
    out = []
    for it in m.insts:
        if it.name != "OpCooperativeMatrixMulAddKHR":
            continue
        rc = m.cm_type(it.a[0])
        out.append({"word_offset": it.woff, "result_id": it.a[1],
                    "function": m.funcs.get(it.func, "?"),
                    "accumulator": rc and rc.get("component"),
                    "A_cmat": m.cm_type(m.byid[it.a[2]].a[0]) if it.a[2] in m.byid else None,
                    "B_cmat": m.cm_type(m.byid[it.a[3]].a[0]) if it.a[3] in m.byid else None,
                    "C_id": it.a[4]})
    return out


def pc_bytes(m):
    for vid, sc, tid in m.vars:
        if STORAGE.get(sc) != "PushConstant":
            continue
        t = m.types.get(tid)
        if t and t[0] == "ptr":
            return m.size(t[2])
    return None


def iface(m):
    bindings, shared = {}, []
    for vid, sc, tid in m.vars:
        d = m.decor.get(vid, {})
        b = d.get(33, [[None]])[0][0]
        if b is not None:
            bindings[str(b)] = {"name": m.names.get(vid), "block": m.block_name(vid),
                                "storage": STORAGE.get(sc), "bytes": m.size(tid)}
        if STORAGE.get(sc) == "Workgroup":
            shared.append({"name": m.var_name(vid), "bytes": m.size(tid)})
    ep = [it for it in m.insts if it.name == "OpEntryPoint"]
    ex = [[m.fval(x) if x in m.consts else x for x in it.a]
          for it in m.insts if it.name == "OpExecutionMode"]
    return {
        "bindings": bindings, "shared": shared,
        "lds_bytes": sum(s["bytes"] for s in shared),
        "entry_point": {"model": ep[0].a[0], "id": ep[0].a[1],
                        "name": bytes(b"".join(struct.pack("<I", x)
                                               for x in ep[0].a[2:])).split(b"\0")[0].decode()}
        if ep else None,
        "execution_modes": ex,
        "cm_types": sorted({(m.ctype(v[0]), m.fval(v[1]), m.fval(v[2]), m.fval(v[3]),
                             m.fval(v[4])) for v in m.cm_types.values()}, key=str),
        "push_constant_bytes": pc_bytes(m),
        "functions": [m.funcs[k] for k in m.funcs],
    }

def nc_roles(m):
    """every NoContraction-decorated instruction with its opcode and word offset"""
    ids = set()
    for vid, d in m.decor.items():
        if 42 in d:
            ids.add(vid)
    out = []
    for it in m.insts:
        if it.op in spvlib.NO_RESULT:
            continue
        rid = it.a[0] if it.op in spvlib.RESULT_FIRST else (it.a[1] if len(it.a) > 1 else None)
        if rid in ids:
            out.append({"word_offset": it.woff, "opcode": it.name,
                        "function": m.funcs.get(it.func, "?"),
                        "float_arith": it.name in ("OpFMul", "OpFAdd", "OpFSub", "OpFDiv",
                                                   "OpFNegate", "OpExtInst"),
                        "result_id": rid})
    return out


def main():
    p, c = spvlib.Mod(PARENT), spvlib.Mod(CAND)
    rec = {"parent": {"path": PARENT.as_posix(), "bytes": PARENT.stat().st_size,
                      "sha256": sha(PARENT), "words": len(p.w)},
           "candidate": {"path": CAND.as_posix(), "bytes": CAND.stat().st_size,
                         "sha256": sha(CAND), "words": len(c.w)}}
    # ---------------------------------------------------------------- P0
    tprobe = {}
    if CAND_T.exists():
        tprobe = {"transfer_spv": {"path": CAND_T.as_posix(),
                                   "bytes": CAND_T.stat().st_size,
                                   "sha256": sha(CAND_T)},
                  "donor": {"path": DONOR.as_posix(), "sha256": sha(DONOR)},
                  "byte_identical_to_candidate": sha(CAND_T) == sha(CAND)}
    rec["P0_identity"] = {
        "parent_nocontraction_count": len([r for r in nc_roles(p)]),
        "candidate_nocontraction_count": len([r for r in nc_roles(c)]),
        "candidate_is_plain_compile": True,
        "nocontraction_transfer_probe": tprobe,
    }
    # ---------------------------------------------------------------- P1
    ip, ic = iface(p), iface(c)
    tadd = [t for t in ic["cm_types"] if t not in ip["cm_types"]]
    trem = [t for t in ip["cm_types"] if t not in ic["cm_types"]]
    rec["P1_interface"] = {
        "bindings_added": {k: ic["bindings"][k] for k in ic["bindings"]
                           if k not in ip["bindings"]},
        "bindings_removed": sorted(set(ip["bindings"]) - set(ic["bindings"])),
        "bindings_kept": {k: ip["bindings"][k] for k in ip["bindings"] if k in ic["bindings"]},
        "bindings_kept_identical": all(ip["bindings"][k]["block"] == ic["bindings"][k]["block"]
                                       and ip["bindings"][k]["storage"] == ic["bindings"][k]["storage"]
                                       for k in ip["bindings"] if k in ic["bindings"]),
        "shared_before": ip["shared"], "shared_after": ic["shared"],
        "lds_before": ip["lds_bytes"], "lds_after": ic["lds_bytes"],
        "push_constant_bytes": [ip["push_constant_bytes"], ic["push_constant_bytes"]],
        "entry_point": [ip["entry_point"], ic["entry_point"]],
        "execution_modes": [ip["execution_modes"], ic["execution_modes"]],
        "functions": [ip["functions"], ic["functions"]],
        "cm_types_added": tadd, "cm_types_removed": trem,
    }
    # ---------------------------------------------------------------- P2
    lp, sp = stream(p)
    lc, sc = stream(c)
    keep_p = Counter(tup(x) for x in lp if LOAD_KEEP(x))
    keep_c = Counter(tup(x) for x in lc if LOAD_KEEP(x))
    st_p = Counter(tup(x) for x in sp if STORE_KEEP(x))
    st_c = Counter(tup(x) for x in sc if STORE_KEEP(x))
    rec["P2_operand_streams"] = {
        "residual_and_scale_loads_before": [list(k) for k in keep_p],
        "residual_and_scale_loads_after": [list(k) for k in keep_c],
        "residual_loads_identical": keep_p == keep_c,
        "store_stream_identical": st_p == st_c,
        "store_stream": [list(k) for k in st_p],
        "loads_removed": {str(k): v for k, v in (Counter(tup(x) for x in lp) -
                                                 Counter(tup(x) for x in lc)).items()},
        "loads_added": {str(k): v for k, v in (Counter(tup(x) for x in lc) -
                                               Counter(tup(x) for x in lp)).items()},
        "stores_removed": {str(k): v for k, v in (Counter(tup(x) for x in sp) -
                                                  Counter(tup(x) for x in sc)).items()},
        "stores_added": {str(k): v for k, v in (Counter(tup(x) for x in sc) -
                                                Counter(tup(x) for x in sp)).items()},
        "parent_load_count": len(lp), "candidate_load_count": len(lc),
        "parent_store_count": len(sp), "candidate_store_count": len(sc),
        "normalisation": "SSA ids inside address/stride symbols are folded to 'load#*'; "
                         "the only streams using a loaded stride are the two emit_qx "
                         "stores of a real function, and the four body stores use the "
                         "literal strides 64/16, so the fold cannot hide a real change",
        "kept_load_stream": sorted(set(k[0] for k in keep_p)),
    }
    # ---------------------------------------------------------------- P3
    mp, mc = mma(p), mma(c)
    dsp = Counter((s["accumulator"], (s["A_cmat"] or {}).get("component"),
                   (s["A_cmat"] or {}).get("rows"), (s["A_cmat"] or {}).get("cols"),
                   (s["B_cmat"] or {}).get("component"), (s["B_cmat"] or {}).get("rows"),
                   (s["B_cmat"] or {}).get("cols"))
                  for s in mp if s["function"] == "main")
    dsc = Counter((s["accumulator"], (s["A_cmat"] or {}).get("component"),
                   (s["A_cmat"] or {}).get("rows"), (s["A_cmat"] or {}).get("cols"),
                   (s["B_cmat"] or {}).get("component"), (s["B_cmat"] or {}).get("rows"),
                   (s["B_cmat"] or {}).get("cols"))
                  for s in mc if s["function"] == "main")
    rec["P3_matrix_sites"] = {
        "parent_count": len(mp),
        "parent_fp16": sum(1 for s in mp if s["accumulator"] == "f32"),
        "parent_int8": sum(1 for s in mp if s["accumulator"] == "i32"),
        "candidate_count": len(mc),
        "candidate_fp16": sum(1 for s in mc if s["accumulator"] == "f32"),
        "candidate_int8": sum(1 for s in mc if s["accumulator"] == "i32"),
        "parent_sites": mp, "candidate_sites": mc,
        "ds_sites": {"parent": [list(k) + [v] for k, v in dsp.items()],
                     "candidate": [list(k) + [v] for k, v in dsc.items()],
                     "OPERANDS_UNCHANGED": dsp == dsc},
        "ds_sites_preserved": [s for s in mc if s["function"] == "main"],
        "no_fp16_matrix_multiply_left":
            sum(1 for s in mc if s["accumulator"] == "f32") == 0,
    }
    # ---------------------------------------------------------------- P4
    hp, hc = Counter(), Counter()
    for it in p.insts:
        hp[it.name] += 1
    for it in c.insts:
        hc[it.name] += 1
    deltas, unexplained = {}, []
    for k in sorted(set(hp) | set(hc)):
        d = hc[k] - hp[k]
        if not d:
            continue
        d_ = {"before": hp[k], "after": hc[k], "delta": d, "attribution": ATTRIB.get(k)}
        deltas[k] = d_
        if d_["attribution"] is None:
            unexplained.append(k)
    mz = {k: hc[k] - hp[k] for k in MUST_ZERO}
    mk = {k: [hp[k], hc[k]] for k in MUST_KEEP}
    rec["P4_opcodes"] = {"deltas": deltas, "unexplained": unexplained,
                         "must_be_zero": mz,
                         "must_be_zero_ok": all(v == 0 for v in mz.values()),
                         "must_be_kept": mk,
                         "must_be_kept_ok": all(v[0] == v[1] for v in mk.values()),
                         "parent_histogram": dict(sorted(hp.items())),
                         "candidate_histogram": dict(sorted(hc.items()))}
    # ---------------------------------------------------------------- P5
    def dec(m):
        out = Counter()
        for vid, d in m.decor.items():
            for k, ops in d.items():
                out[str(k)] += len(ops)
        return out
    dp, dc = dec(p), dec(c)
    DNAME = {"2": "Block", "6": "ArrayStride", "11": "BuiltIn", "24": "NonWritable",
             "33": "Binding", "34": "DescriptorSet", "42": "NoContraction"}
    rp, rc_ = nc_roles(p), nc_roles(c)
    role_p = Counter((r["function"], r["opcode"]) for r in rp if r["float_arith"])
    role_c = Counter((r["function"], r["opcode"]) for r in rc_ if r["float_arith"])
    rec["P5_decorations"] = {
        "deltas": {DNAME.get(k, "decoration_" + k): {"before": dp.get(k, 0),
                                                     "after": dc.get(k, 0)}
                   for k in sorted(set(dp) | set(dc)) if dc.get(k, 0) != dp.get(k, 0)},
        "nocontraction_before": dp.get("42", 0), "nocontraction_after": dc.get("42", 0),
        "nocontraction_roles_parent": rp,
        "nocontraction_roles_candidate": rc_,
        "parent_float_arith_nc": sum(1 for r in rp if r["float_arith"]),
        "candidate_float_arith_nc": sum(1 for r in rc_ if r["float_arith"]),
        "parent_non_arith_nc": [r for r in rp if not r["float_arith"]],
        "candidate_non_arith_nc": [r for r in rc_ if not r["float_arith"]],
        "parent_role_multiset": {"%s / %s" % k: v for k, v in role_p.items()},
        "candidate_role_multiset": {"%s / %s" % k: v for k, v in role_c.items()},
        "PARENT_FLOAT_NC_ALL_PRESERVED": all(role_c[k] >= v for k, v in role_p.items()),
        "candidate_nc_extra": {"%s / %s" % k: role_c[k] - role_p.get(k, 0)
                               for k in role_c if role_c[k] > role_p.get(k, 0)},
        "note": "NoContraction is emitted by glslang for every `precise` expression. "
                "The parent's deployed SPV additionally carries four stray "
                "NoContraction decorations on non-float instructions, inherited by the "
                "family's NoContraction-transfer step (aligning a changed function "
                "against the old FP16 body donor); the candidate is a plain compile and "
                "carries none. The transfer probe (P0) shows that re-running the step on "
                "the candidate adds four *different* stray decorations and leaves every "
                "float-arithmetic one untouched, i.e. the step is numerically inert here.",
    }
    ok = (not unexplained
