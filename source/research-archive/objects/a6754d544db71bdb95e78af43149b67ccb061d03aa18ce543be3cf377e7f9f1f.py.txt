from pathlib import Path

p = Path(__file__).with_name('COMMON.py')
t = p.read_text(encoding='utf-8')
old = """        if spv_idx is not None:
            new_toks[spv_idx] = cand"""
new = """        if spv_idx is not None:
            q = '"' if head_toks[spv_idx].startswith('"') else ''
            new_toks[spv_idx] = q + cand + q"""
assert t.count(old) == 1, t.count(old)
p.write_text(t.replace(old, new, 1), encoding='utf-8')
print('quote preservation patched')
