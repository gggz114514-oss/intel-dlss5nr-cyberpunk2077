"""Append an adopted step to the chain from an explicit plan path (for cards with several tiers)."""
import json, shutil, sys
from pathlib import Path
sys.path.insert(0, str(Path(__file__).parent))
import COMMON as K

label = sys.argv[1]
plan = Path(sys.argv[2]).resolve()
saving = float(sys.argv[3])
cio, cih = float(sys.argv[4]), float(sys.argv[5])
note = sys.argv[6] if len(sys.argv) > 6 else ''

cur = K.read_json(K.C / 'CURRENT.json')
acc = K.C / 'accepted' / ('%s-20261027' % label)
acc.mkdir(parents=True, exist_ok=True)
shutil.copyfile(plan, acc / 'plan.txt')
assert K.sha(acc / 'plan.txt') == K.sha(plan)
entry = dict(path=str(acc / 'plan.txt'), sha256=K.sha(acc / 'plan.txt'), label=label,
             saving_ms=saving, CI95_ms=[cio, cih], kind='adopt', note=note)
cur['chain'].append(entry)
cur['increments_ms'].append(saving)
cur['best_plan'] = dict(path=entry['path'], sha256=entry['sha256'])
cur['cumulative_ms_in_chain'] = sum(cur['increments_ms'])
K.write_json(K.C / 'CURRENT.json', cur)
K.write_json(acc / 'ADOPT.json', entry)
print(json.dumps(dict(adopted=label, plan=entry['path'], sha=entry['sha256'][:12],
                      chain_len=len(cur['chain']), sum_ms=round(cur['cumulative_ms_in_chain'], 6)), indent=2))
