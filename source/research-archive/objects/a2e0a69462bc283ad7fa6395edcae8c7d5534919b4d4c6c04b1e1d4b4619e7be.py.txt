"""Patch: support per-ordinal candidate SPVs in rebase_by_ordinals.

A card may need a different candidate SPV for each ordinal (e.g. a producer/consumer pair).
GPU-REQUEST then declares rebase.mode == 'per_ordinal' plus spv_per_ordinal[<ordinal>].
The single-candidate path stays the default and is unchanged.
"""
from pathlib import Path

p = Path(__file__).with_name('COMMON.py')
t = p.read_text(encoding='utf-8')

old = """    cand = (d / 'candidate.spv').as_posix()
    carried = []"""
new = """    per_ord = None
    reb = req.get('rebase') or {}
    if reb.get('mode') == 'per_ordinal':
        raw_map = req.get('spv_per_ordinal') or {}
        per_ord = {}
        for k, v in raw_map.items():
            spv = v.get('candidate_spv') if isinstance(v, dict) else v
            assert spv, ('per_ordinal entry without candidate_spv', k)
            per_ord[int(k)] = Path(spv).as_posix()
        missing = [o for o in ords if o not in per_ord]
        assert not missing, ('per_ordinal missing ordinals', missing)
    cand = (d / 'candidate.spv').as_posix()
    carried = []"""
assert t.count(old) == 1
t = t.replace(old, new, 1)

old2 = """        if spv_idx is not None:
            q = '"' if head_toks[spv_idx].startswith('"') else ''
            new_toks[spv_idx] = q + cand + q"""
new2 = """        this_cand = per_ord[o] if per_ord else cand
        if spv_idx is not None:
            q = '"' if head_toks[spv_idx].startswith('"') else ''
            new_toks[spv_idx] = q + this_cand + q"""
assert t.count(old2) == 1
t = t.replace(old2, new2, 1)
p.write_text(t, encoding='utf-8')
print('per-ordinal rebase patched')
