import json, sys
from collections import Counter
from pathlib import Path
CAMP = Path('@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/fullnet-continuous-20261027')
sys.path.insert(0, str(CAMP)); import ISA
ISA_PATH = CAMP / 'queue/n29-f01-alu-chain/native/captures/native-native-cand/ISA/CS_asma0888c5a3c6703a9_simd32_0000000000000001_0_orig.asm'
ENTRY_END = 0x4A90
mask = json.loads((CAMP/'cards/n31-f01-dpas-chain/tools/exec-mask.json').read_text(encoding='utf-8'))
spans = [(int(a,16), int(b,16)) for a,b in mask['spans']]
def dead(a): return any(x < a < y for x,y in spans)
isa = ISA.parse(ISA_PATH)
live = [i for i in isa if i['addr'] >= ENTRY_END and not dead(i['addr'])]
def base(o):
    o = o.split('<')[0]
    return o
selfmove = [i for i in live if i['op']=='mov' and len(i['ops'])>=2 and base(i['ops'][0])==base(i['ops'][1])]
print('self-mov (identical register field):', len(selfmove))
# mov chains: dst equals some other mov's src in the immediately preceding 2 instructions
chain2 = 0
for a,b in zip(live, live[1:]):
    if a['op']=='mov' and b['op']=='mov' and len(a['ops'])>=2 and len(b['ops'])>=2:
        if base(a['ops'][0]) == base(b['ops'][1]): chain2 += 1
print('adjacent mov->mov (dst1 == src2):', chain2)
# mov with same width type both sides & same reg
w = Counter()
for i in live:
    if i['op']!='mov' or len(i['ops'])<2: continue
    d = i['ops'][0].split(':')[-1] if ':' in i['ops'][0] else '?'
    s = i['ops'][1].split(':')[-1] if ':' in i['ops'][1] else '?'
    w[(d,s)] += 1
print('mov type pairs (top):', w.most_common(10))
