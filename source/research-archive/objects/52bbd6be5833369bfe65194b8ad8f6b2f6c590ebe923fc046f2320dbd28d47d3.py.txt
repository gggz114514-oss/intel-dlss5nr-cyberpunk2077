"""n10b-f10-pipeline build: F10 software pipeline over the n10 double window.

Writes only into this card directory.  No GPU, no deployment, no shared edits.

Pipeline
  1. parent byte reproduction: the chain-head F10 source (n10-f10-next/candidate.comp)
     must recompile byte for byte to the chain-head F10 SPV (228dee8a...).
  2. candidate source: the kc loop becomes a nine-step producer/consumer ring; the
     publish barrier moves to the loop head, staging slice i precedes consuming
     slice i-1, so the next sixteen global reads are issued before the twenty-four
     DPAS and the barrier that waits for them sits after the DPAS.
  3. structural proof: NoContraction sites pair 1:1 with the parent, every float
     instruction keeps its opcode and operand signature, the mapping and the
     hazard schedule are re-derived from scratch.
  4. candidate.spv must pass spv_validate_cpu.
  5. plan.txt: exactly the sixteen F10 PASS lines of the chain head are swapped.
  6. plan must pass clock_runner --cpu-check.
"""

def sha(p):
    return hashlib.sha256(Path(p).read_bytes()).hexdigest()


def ident(p):
    return dict(path=str(Path(p)), sha256=sha(p))


def run(argv, allow_fail=False):
    r = subprocess.run(list(map(str, argv)), capture_output=True, text=True,
                       encoding='utf-8', errors='replace')
    if r.returncode != 0 and not allow_fail:
        raise RuntimeError('FAILED %s\n%s\n%s' % (argv, r.stdout, r.stderr))
    return r.returncode, (r.stdout or '') + (r.stderr or '')

def sub(old, new, count=1):
    global cand
    n = cand.count(old)
    assert n == count, 'expected %d occurrence(s), found %d for %r' % (count, n, old[:90])
    cand = cand.replace(old, new)
    edits.append(dict(kind='replace', count=count, before=old, after=new))


def sub_region(start_marker, end_marker, new, must_contain=(), must_not_contain=()):
    """Replace the inclusive region [start_marker .. end_marker] exactly once."""
    global cand
    assert cand.count(start_marker) == 1, 'start marker count %d' % cand.count(start_marker)
    assert cand.count(end_marker) == 1, 'end marker count %d' % cand.count(end_marker)
    i0 = cand.index(start_marker)
    i1 = cand.index(end_marker) + len(end_marker)
    old = cand[i0:i1]
    for m in must_contain:
        assert m in old, 'region does not contain %r' % m
    for m in must_not_contain:
        assert m not in old, 'region unexpectedly contains %r' % m
    cand = cand[:i0] + new + cand[i1:]
    edits.append(dict(kind='region', count=1, before=old, after=new))

# 2b. the kc loop -> the nine-step ring
NEW_LOOP = '''        // N10B-F10 software pipeline.  The two staging windows n10 built are used as a
        // producer/consumer ring instead of one window per slice:
        //   iteration i : publish barrier (i>0) -> stage slice i (i<8) -> consume slice i-1 (i>0)
        // so the sixteen global reads of the next slice are issued before the twenty-four
        // DPAS of the current one, and the barrier that has to wait for those reads comes
        // after the DPAS.  The window of slice i is [4096,8192) for even i and [0,4096) for
        // odd i, so the two live windows of an iteration are the two halves of smem[0,8192)
        // and the last staged slice (i=7, odd) is in [0,4096): the QKV handoff stores in
        // [4096,10240) are still ordered only by the existing post-store barrier.  The
        // number of barriers executed per workgroup is eight, as in n10.
        // The guards are uniform (i is the loop counter, the same for every invocation).
        uint abase=QKV_OFF;   // staging window of the slice this iteration produces
        uint kc=0u;           // channel base of the slice this iteration consumes
        for(uint i=0u;i<9u;i++) {
            // publish slice i-1 (staged in the previous iteration) and retire its readers
            if(i>0u) barrier();
            if(i<8u) {
#define STAGE(A,V,O) smem[abase+qkv_a_offset(lid+(O))]=(V)?gy[(A)]:float16_t(0.0)
                STAGE(a00,v00,0u);      STAGE(a00+64u,v00,256u);
                STAGE(a00+128u,v00,512u);STAGE(a00+192u,v00,768u);
                STAGE(a10,v10,1024u);   STAGE(a10+64u,v10,1280u);
                STAGE(a10+128u,v10,1536u);STAGE(a10+192u,v10,1792u);
                STAGE(a01,v01,2048u);   STAGE(a01+64u,v01,2304u);
                STAGE(a01+128u,v01,2560u);STAGE(a01+192u,v01,2816u);
                STAGE(a11,v11,3072u);   STAGE(a11+64u,v11,3328u);
                STAGE(a11+128u,v11,3584u);STAGE(a11+192u,v11,3840u);
#undef STAGE
                // the next slice starts 64 channels further into the same tiles
                a00+=1024u;a10+=1024u;a01+=1024u;a11+=1024u;
            }
            if(i>0u) {
                // the slice published by the barrier above lives in the other window
                uint acons=QKV_OFF-abase;
                for(uint k=0u;k<64u;k+=16u) {
                    MA a0,a1;
                    coopMatLoad(a0,smem,acons+((mA/8u)*4u+k/16u)*128u,8u,COL);
                    coopMatLoad(a1,smem,acons+((mB/8u)*4u+k/16u)*128u,8u,COL);
                    MB b0,b1,b2;
                    coopMatLoad(b0,weights,wq_tile(h*96u+n0+0u,kc+k),16u,COL);
                    coopMatLoad(b1,weights,wq_tile(h*96u+n0+16u,kc+k),16u,COL);
                    coopMatLoad(b2,weights,wq_tile(h*96u+n0+32u,kc+k),16u,COL);
                    acc00=coopMatMulAdd(a0,b0,acc00);
                    acc10=coopMatMulAdd(a1,b0,acc10);
                    acc01=coopMatMulAdd(a0,b1,acc01);
                    acc11=coopMatMulAdd(a1,b1,acc11);
                    acc02=coopMatMulAdd(a0,b2,acc02);
                    acc12=coopMatMulAdd(a1,b2,acc12);
                }
                kc+=64u;
            }
            // ring step: the window staged by this iteration is the one consumed next
            abase=QKV_OFF-abase;
        }'''

def classify(m, ids):
    out = {}
    for op, a, _, _ in m.instructions:
        if ST.result_id_of(op, a) in ids:

def op_seq(m, opcodes):
    out = []
    for fi, reg in enumerate(ST.function_regions(m)):
        for i in reg:
            op = m.instructions[i][0]
            if op in opcodes:
                out.append((fi, op, ST.result_id_of(op, m.instructions[i][1])))
    return out

def nc_multiset(m, ids):
    seen, out = {}, []
    for op, a, _, _ in m.instructions:
        rid = ST.result_id_of(op, a)
        if rid in ids:
            key = (op, str(m.operand_signature(rid, 3)))
            seen[key] = seen.get(key, 0) + 1
            out.append((key, seen[key]))
    return sorted(out)

# ------------------------------------------------------ 5. loop / barrier shape
OP_LABEL, OP_LOOP_MERGE, OP_BARRIER, OP_STORE, OP_LOAD = 248, 246, 224, 62, 61


def loops_of(m):
    ins = m.instructions
    out = []

    def label_index(lbl):
        for j, (o2, a2, _, _) in enumerate(ins):
            if o2 == OP_LABEL and a2 and a2[0] == lbl:
                return j
        return None

    for i, (op, a, _, _) in enumerate(ins):
        if op != OP_LOOP_MERGE:
            continue
        head_i = None
        for j in range(i - 1, -1, -1):
            if ins[j][0] == OP_LABEL:
                head_i = j
                break
        mg = label_index(a[0])
        out.append(dict(head_i=head_i, merge_i=mg))
    return out


def loop_report(m):
    ins = m.instructions
    loops = loops_of(m)
    recs = []
    for lp in loops:
        h, mg = lp['head_i'], lp['merge_i']
        inner = [x for x in loops if x is not lp and x['head_i'] > h and x['merge_i'] <= mg]
        own = [j for j in range(h, mg)
               if not any(x['head_i'] <= j < x['merge_i'] for x in inner)]
        body = mg - h
        bar = [j for j in range(h, mg) if ins[j][0] == OP_BARRIER]
        rec = dict(head_word='0x%X' % (ins[h][2] * 4), body_instructions=body,
                   barriers_in_loop=len(bar),
                   barrier_words=['0x%X' % (ins[j][2] * 4) for j in bar])
        if inner:
            rec['nested_loops'] = [dict(head_word='0x%X' % (ins[x['head_i']][2] * 4),
                                        body_instructions=x['merge_i'] - x['head_i'])
                                   for x in inner]
            rec['own_body_instructions'] = len(own)
            rec['own_histogram'] = {str(k): v for k, v in sorted(opcount_idx(ins, own).items())}
            rec['own_barrier_offsets'] = [j - h for j in own if ins[j][0] == OP_BARRIER]
            rec['own_store_offsets'] = [j - h for j in own if ins[j][0] == OP_STORE]
            rec['own_first_load_offset'] = next((j - h for j in own if ins[j][0] == OP_LOAD), None)
            rec['barrier_offsets'] = [j - h for j in bar]
            rec['first_muladd_offset'] = next(
                (j - h for j in range(h, mg) if ins[j][0] == MULADD), None)
        recs.append(rec)
    recs.sort(key=lambda d: -d['body_instructions'])
    return dict(total_barriers=sum(1 for op, _, _, _ in ins if op == OP_BARRIER),
                loops=recs[:4])


def opcount_idx(ins, idxs):
    c = {}
    for j in idxs:
        c[ins[j][0]] = c.get(ins[j][0], 0) + 1
    return c

# ------------------------------------------------- 8. mapping / hazard schedule
def a_off(e):
    r, k = e // 64, e % 64
    return ((r // 8) * 4 + k // 16) * 128 + (k % 16) * 8 + r % 8


cells = {}
for r in range(64):
    for k in range(64):
        addr = a_off(r * 64 + k)
        assert addr not in cells
        cells[addr] = (r, k)
assert set(cells) == set(range(4096)), 'staged slice is not a bijection'
fragment_ok = 0
for sg in range(8):
    for abase in (0, 4096):
        for m in ((sg // 2) * 16, (sg // 2) * 16 + 8):
            for k0 in range(0, 64, 16):
                base = abase + ((m // 8) * 4 + k0 // 16) * 128
                for r in range(8):
                    for c in range(16):
                        assert cells[base - abase + c * 8 + r] == (m + r, k0 + c)
                        fragment_ok += 1


def tiled(token, c, C=512):
    return ((token // 16) * (C // 16) + c // 16) * 256 + (token % 16) * 16 + c % 16


hoist_checks = 0
for token in range(1024):
    base_c0 = [tiled(token, c0) for c0 in range(64)]
    for i in range(8):
        for c0 in range(64):
            assert base_c0[c0] + 1024 * i == tiled(token, 64 * i + c0)
            hoist_checks += 1


def schedule(kind):
    """Program-order event list per thread, with the barrier sequence number of each
    event (the number of barriers that have completed before it in that thread)."""
    events = []
    bseq = 0
    if kind == 'parent':                       # n10: stage i, barrier, consume i
        stage_win, cons_win = [], []
        for i in range(8):
            w = 4096 if i % 2 == 0 else 0
            stage_win.append(w)
            stage_cons = 0
            events.append(dict(iter=i, ev='stage', slice=i, window=w, bseq=bseq))
            bseq += 1
            events.append(dict(iter=i, ev='barrier', slice=None, window=None, bseq=bseq))
            events.append(dict(iter=i, ev='consume', slice=i, window=w, bseq=bseq))
    else:                                      # candidate: nine-step ring
        stage_win = []
        for i in range(9):
            w = 4096 if i % 2 == 0 else 0
            if i > 0:
                bseq += 1
                events.append(dict(iter=i, ev='barrier', slice=None, window=None, bseq=bseq))
            if i < 8:
                stage_win.append(w)
                events.append(dict(iter=i, ev='stage', slice=i, window=w, bseq=bseq))
            if i > 0:
                acons = 4096 - w
                events.append(dict(iter=i, ev='consume', slice=i - 1, window=acons, bseq=bseq))
    return events


hazard = {}
for kind in ('parent', 'candidate'):
    ev = schedule(kind)
    stage_w = {e['slice']: e['window'] for e in ev if e['ev'] == 'stage'}
    cons_w = {e['slice']: e['window'] for e in ev if e['ev'] == 'consume'}
    assert set(stage_w) == set(range(8)) and set(cons_w) == set(range(8))
    assert all(stage_w[s] == cons_w[s] for s in range(8)), (stage_w, cons_w)
    first = {}
    for e in ev:
        first.setdefault((e['ev'], e['slice']), e)
    checks = []
    for s in range(8):
        w = stage_w[s]
        a, b = first[('stage', s)], first[('consume', s)]
        checks.append(dict(kind='RAW', item='staging write -> A fragment read', slice=s,
                           window=w, stage_bseq=a['bseq'], consume_bseq=b['bseq'],
                           barriers_between=b['bseq'] - a['bseq']))
        if s + 2 <= 7:
            nxt = first[('stage', s + 2)]
            checks.append(dict(kind='WAR', item='A fragment read -> next staging write',
                               slice=s, window=w, consume_bseq=b['bseq'],
                               next_stage_bseq=nxt['bseq'],
                               barriers_between=nxt['bseq'] - b['bseq']))
            checks.append(dict(kind='WAW', item='staging write -> next staging write',
                               slice=s, window=w, stage_bseq=a['bseq'],
                               next_stage_bseq=nxt['bseq'],
                               barriers_between=nxt['bseq'] - a['bseq']))
    handoff = None
    if kind == 'candidate':
        last_stage = first[('stage', 7)]
        handoff = dict(
            last_staged_slice=7, last_staged_window=stage_w[7],
            handoff_region=[4096, 10240],
            overlap=(stage_w[7] == 4096),
            store_qkv_bseq=8,
            last_read_of_handoff_window=first[('consume', 6)]['bseq'],
            barriers_between=8 - first[('consume', 6)]['bseq'])
        assert not handoff['overlap'], 'last staged slice shares the handoff window'
        assert handoff['barriers_between'] >= 1
    bad = [c for c in checks if c['barriers_between'] < 1]
    assert not bad, bad
    hazard[kind] = dict(events=ev, checks=checks, handoff=handoff,
                        barriers_executed=sum(1 for e in ev if e['ev'] == 'barrier'))

# ------------------------------------------- 8b. ring index algebra (S1)
# There are exactly two statements in the source that touch abase or kc:
#   abase=QKV_OFF;  ...  abase=QKV_OFF-abase;   and   uint kc=0u; ... kc+=64u;
# So the table below is a transliteration of the source, not a second model.
QKV_OFF = 4096
ring_table = []
abase, kc = QKV_OFF, 0
for i in range(9):
    row = dict(iteration=i, abase_at_entry=abase)
    if i < 8:
        row['staged_slice'] = i
        row['stage_window'] = abase
    if i > 0:
        acons = QKV_OFF - abase
        row['consumed_slice'] = i - 1
        row['consume_window'] = acons
        row['kc'] = kc
        assert kc == 64 * (i - 1), (kc, i)
        kc += 64
    abase = QKV_OFF - abase
    ring_table.append(row)
stage_win_of = {r['staged_slice']: r['stage_window'] for r in ring_table if 'staged_slice' in r}
cons_win_of = {r['consumed_slice']: r['consume_window'] for r in ring_table if 'consumed_slice' in r}
assert stage_win_of == cons_win_of, (stage_win_of, cons_win_of)
assert sorted(stage_win_of) == list(range(8))
assert [r['consumed_slice'] for r in ring_table if 'consumed_slice' in r] == list(range(8))

# the accepted asset: the three weight loads and the six accumulator statements must be
# byte-identical to the parent and in the same order (only their surroundings changed).
def b_asset_lines(text):
    out = []
    for l in text.splitlines():
        s = l.strip()
        if (s.startswith('coopMatLoad(b0,weights') or s.startswith('coopMatLoad(b1,weights')
                or s.startswith('coopMatLoad(b2,weights')):
            out.append(s)
        elif (s.startswith(('acc00=', 'acc10=', 'acc01=', 'acc11=', 'acc02=', 'acc12='))
              and '=coopMatMulAdd' in s):
            out.append(s)
    return out
