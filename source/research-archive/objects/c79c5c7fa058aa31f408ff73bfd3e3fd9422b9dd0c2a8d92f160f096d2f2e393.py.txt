"""N05-a build: candidate compile + identity checks. CPU only, no GPU."""
import sys, json, shutil, difflib
from pathlib import Path

C = Path("@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/fullnet-continuous-20261027")
sys.path.insert(0, str(C))
import COMMON as K
import spv_tools_local as ST

CARD = C / "cards" / "n05a-f05-c256-mreuse"
CARD.mkdir(parents=True, exist_ok=True)
BUILD = CARD / "_build"
BUILD.mkdir(exist_ok=True)

R = K.row_by_ordinal(37)
PARENT_SPV = Path(R["spv"])
PARENT_SRC = Path(R["source"])
CAND_SRC = CARD / "candidate.comp"
CAND_SPV = CARD / "candidate.spv"

print("== 1. identify ==")
for label, p, want in (("parent spv", PARENT_SPV, R["spv_sha256"]),
                       ("parent src", PARENT_SRC, R["source_sha256"])):
    got = K.sha(p)
    print("%-12s %s %s %s" % (label, got, "OK" if got == want else "MISMATCH(want %s)" % want, p))

print("== 2. parent byte reproduction ==")
repro = K.reproduce_current(37, BUILD / "repro")
print(json.dumps(repro, indent=2))
shutil.copyfile(BUILD / "repro" / "rebuild.spv", CARD / "parent-rebuild.spv")

print("== 3. compile candidate ==")
incs = K.include_dirs_for(37, CAND_SRC)
print("include dirs:")
for d in incs:
    print("   ", d)
K.compile_glsl(CAND_SRC, CAND_SPV, incs)
print("candidate spv:", CAND_SPV, K.sha(CAND_SPV), CAND_SPV.stat().st_size, "bytes")

print("== 4. spv_validate_cpu ==")
K.validate_spv(CAND_SPV)
print("validate: OK (exit 0)")

print("== 5. NoContraction / decoration alignment ==")
pm = ST.Module(PARENT_SPV)
cm = ST.Module(CAND_SPV)
pnc = ST.no_contraction_ids(pm)
cnc = ST.no_contraction_ids(cm)
print("parent NC ids:", len(pnc), "candidate NC ids:", len(cnc))
keep, info = ST.transfer_decoration_pattern_aligned(pm, cm, set())
print("transfer info:", json.dumps({k: info[k] for k in (
    'functions_mapped', 'child_functions_without_parent_counterpart',
    'parent_decorations_total', 'child_own_decorations_outside_pattern')}, indent=2))
print("keep (parent NC ids matched 1:1 into child):", len(keep))
print("parent fma ordinals:", len(pm.ext_inst_fma_ordinals()))
print("cand   fma ordinals:", len(cm.ext_inst_fma_ordinals()))

json.dump(dict(repro=repro, nc_parent=len(pnc), nc_candidate=len(cnc),
               keep=len(keep), info={k: info[k] for k in (
                   'functions_mapped', 'child_functions_without_parent_counterpart',
                   'parent_decorations_total', 'child_own_decorations_outside_pattern')},
               incdirs=[str(d) for d in incs]),
          open(BUILD / "nc.json", "w"), indent=2)

print("== 6. source diff ==")
ptxt = PARENT_SRC.read_text(encoding="utf-8-sig").splitlines(keepends=True)
ctxt = CAND_SRC.read_text(encoding="utf-8-sig").splitlines(keepends=True)
diff = list(difflib.unified_diff(ptxt, ctxt,
                                 fromfile="a/" + PARENT_SRC.name,
                                 tofile="b/candidate.comp", n=3))
(CARD / "SOURCE-DIFF.patch").write_text("".join(diff), encoding="utf-8")
print("diff lines:", len(diff))
