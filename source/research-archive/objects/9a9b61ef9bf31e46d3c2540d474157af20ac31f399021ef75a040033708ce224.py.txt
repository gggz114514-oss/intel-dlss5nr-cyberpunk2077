import json, os
import numpy as np

W = r"@NR_WORKSPACE@/work/dsh-fullnet-static-20261029"
Q = r"@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/int8-quality-first-20261029"
D = os.path.join(Q, "real-int8-24")
REF = os.path.join(Q, "dataset/day-street-static-formal-01/0-34189/network-output.bin")
ref = np.fromfile(REF, dtype="<f4").reshape(2, 720, 1280, 4)

def rd(p):
    return np.fromfile(p, dtype="<f4").reshape(2, 720, 1280, 4)

pairs = [("XE candidate combo-34189", os.path.join(D, "c64-family-static/run-34189/replayed-network.bin"))]
out = {}
for name, p in pairs:
    x = rd(p)
    a = x[0, :, :, :3].astype(np.float64); b = ref[0, :, :, :3].astype(np.float64)
    lum = b.mean(axis=2)
    bright = lum >= 0.7
    r = {"bright_pixels": int(bright.sum())}
    for ci, ch in enumerate("RGB"):
        rb = b[:, :, ci][bright]; ra = a[:, :, ci][bright]
        r[ch] = {"ref_mean": float(rb.mean()), "arm_mean": float(ra.mean()),
                 "diff": float((ra - rb).mean()),
                 "ref_clip_frac_at_1": float((rb >= 1 - 1e-6).mean()),
                 "arm_clip_frac_at_1": float((ra >= 1 - 1e-6).mean()),
                 "ref_p50": float(np.percentile(rb, 50)), "ref_p90": float(np.percentile(rb, 90)),
                 "arm_p90": float(np.percentile(ra, 90))}
        # on pixels where the reference channel is exactly clipped to 1.0
        m = rb >= 1 - 1e-6
        r[ch]["at_ref_clip_n"] = int(m.sum())
        r[ch]["at_ref_clip_arm_mean"] = float(ra[m].mean()) if m.sum() else None
    # bin by reference channel value
    bins = [(0.7, 0.8), (0.8, 0.9), (0.9, 0.99), (0.99, 1.000001)]
    rbins = []
    for lo, hi in bins:
        rr = (b >= lo) & (b < hi)
        if rr.sum() == 0:
            continue
        rbins.append({"ref_value_band": [lo, hi], "n": int(rr.sum()), "arm_minus_ref_mean": float((a[rr] - b[rr]).mean()),
                      "ref_mean": float(b[rr].mean()), "arm_mean": float(a[rr].mean())})
    r["band_arm_minus_ref"] = rbins
    out[name] = r
    print("\n===", name)
    print("bright px:", r["bright_pixels"])
    for ch in "RGB":
        v = r[ch]
        print("  %s ref_mean %.4f arm_mean %.4f diff %+.4f | clip ref %.4f arm %.4f | at ref=1.0 (n=%d) arm=%.4f" % (
            ch, v["ref_mean"], v["arm_mean"], v["diff"], v["ref_clip_frac_at_1"], v["arm_clip_frac_at_1"],
            v["at_ref_clip_n"], v["at_ref_clip_arm_mean"] if v["at_ref_clip_arm_mean"] is not None else float("nan")))
    print("  per-value-band arm-ref (all channels):")
    for bk in rbins:
        print("    band %s n=%d ref %.4f arm %.4f diff %+.5f" % (bk["ref_value_band"], bk["n"], bk["ref_mean"], bk["arm_mean"], bk["arm_minus_ref_mean"]))

json.dump(out, open(os.path.join(W, "C00-first-divergence/_work/rgb-highlight-34189.json"), "w", encoding="utf-8"), ensure_ascii=False, indent=1)
print("\nwrote rgb-highlight-34189.json")
