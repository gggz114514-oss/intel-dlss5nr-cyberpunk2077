def run(path, op_name):
    """reuse the F02 pipeline on an arbitrary listing"""
    import _t01_identity as T
    import _isa
    old = _isa.CAP, _isa.FILES
    _isa.CAP = Path(path).parent
    _isa.FILES = [Path(path).name]
    try:
        ins, labels = _isa.parse(Path(path).name)
        seen = reachable(ins, labels)
        if sum(seen) < max(8, len(ins) // 10):
            # this listing's CFG does not start at index 0 (different prologue):
            # value identity (reaching definitions) does not need the live mask
            seen = [True] * len(ins)
            print("   [note] reachability failed on this listing, using the full listing")
        for i, it in enumerate(ins):
            it["live"] = seen[i]
            it["w"], it["r"] = T.classify(it)
        by_dw = defaultdict(list)
        for it in ins:
            for f, no, dw, n in it["w"]:
                for d in range(dw, dw + n):
                    by_dw[(f, no, d)].append(it["idx"])
        for v in by_dw.values():
            v.sort()
        sites = [i for i in range(len(ins)) if ins[i]["live"] and ins[i]["op"] == op_name]
        groups = defaultdict(list)
        for k in sites:
            src = ins[k]["ops"][0] if ins[k]["ops"] else None
            groups[src].append(k)
        out = []
        for src, ks in groups.items():
            ks.sort()
            prods = []
            for k in ks:
                key = T.reg_base(ins[k]["ops"][0])
                prods.append(T.reaching(by_dw, key, k))
            minw = None
            for i in range(len(ks)):
                for j in range(i + 1, len(ks)):
                    key = T.reg_base(ins[ks[i]]["ops"][0])
                    w = [x for x in by_dw.get(key, []) if ks[i] < x < ks[j]]
                    if minw is None or len(w) < minw:
                        minw = len(w)
            out.append(dict(src=src, sites=len(ks), distinct_producers=len(set(prods)),
                            min_writes_between=minw,
                            producers=prods, sites_idx=ks))
        return out
    finally:
        _isa.CAP, _isa.FILES = old
