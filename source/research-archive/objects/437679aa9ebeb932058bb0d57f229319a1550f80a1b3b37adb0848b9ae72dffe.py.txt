"""n31: generate candidate.comp from the chain-head source (n29's candidate.comp).

Single mechanism (see DPAS-CHAIN-AUDIT.md §5/§6): delete the operand-swap identity
selects of the four softmax-denominator reductions and of nr_pool_natural.

    (lower ? x : peer) + (lower ? peer : x)   ->   x + peer

IEEE-754 addition is commutative, so the two forms are bit-identical for every
input; the `lower` predicate exists only to order the operands.  No add is
reordered, no rounding point moves (CONTRACT §3).

Every replacement is counted and asserted, so the edit cannot silently miss a
site.  Writes candidate.comp next to the card.
"""
import hashlib
import sys
from pathlib import Path

CARD = Path(__file__).resolve().parents[1]
CAMP = CARD.parents[1]
SRC = CAMP / 'cards/n29-f01-alu-chain/candidate.comp'      # the chain head source
DST = CARD / 'candidate.comp'

# (old, new, expected count) -- order matters: longer patterns first
EDITS = [
    # --- denominator reduce: 4 accumulate steps per block, 4 blocks
    ('bool lower=(lane&8u)==0u;precise f16vec2 pair=(lower?x:peer)+(lower?peer:x);d=d+pair;',
     'precise f16vec2 pair=x+peer;d=d+pair;', 16),
    # --- denominator butterfly: xor 2 / 4 / 1, once per block
    ('bool lower=(lane&2u)==0u;d=(lower?d:peer)+(lower?peer:d);', 'd=d+peer;', 4),
    ('bool lower=(lane&4u)==0u;d=(lower?d:peer)+(lower?peer:d);', 'd=d+peer;', 4),
    ('bool lower=(lane&1u)==0u;d=(lower?d:peer)+(lower?peer:d);', 'd=d+peer;', 4),
    # --- nr_pool_natural: the two 16-lane mean pairs
    ('precise f16vec2 sx0=(lower?a:ap)+(lower?ap:a);', 'precise f16vec2 sx0=a+ap;', 1),
    ('precise f16vec2 sx1=(lower?b:bp)+(lower?bp:b);', 'precise f16vec2 sx1=b+bp;', 1),
    # --- the now-unused predicate in nr_pool_natural
    ('bool lower=(lane&16u)==0u;\n', '', 1),
]

# every `lower` that must be gone afterwards
FORBIDDEN = ['(lower?', 'bool lower=']


def main():
    text = SRC.read_text(encoding='utf-8')
    orig_sha = hashlib.sha256(text.encode('utf-8')).hexdigest()
    for old, new, want in EDITS:
        got = text.count(old)
        assert got == want, 'pattern %r: found %d, expected %d' % (old[:48], got, want)
        text = text.replace(old, new)
    for bad in FORBIDDEN:
        assert bad not in text, 'leftover %r' % bad
    DST.write_text(text, encoding='utf-8')
    new_sha = hashlib.sha256(text.encode('utf-8')).hexdigest()
    print('parent src %s  %s' % (orig_sha[:16], SRC))
    print('candidate  %s  %s' % (new_sha[:16], DST))
    print('lines %d -> %d ; sites edited %d'
          % (len(SRC.read_text(encoding="utf-8").splitlines()),
             len(text.splitlines()), sum(e[2] for e in EDITS)))


if __name__ == '__main__':
    main()
