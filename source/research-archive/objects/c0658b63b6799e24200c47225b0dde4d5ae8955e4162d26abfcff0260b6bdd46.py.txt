"""Element census of the F02 body ISA: opcode counts, send message forms, and
the exact shape census of every `mov`.  CPU only, read-only.
"""
import sys, collections
sys.path.insert(0, '@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/fullnet-continuous-20261027/cards/n01b-f02-load-side')
import isaparse as P
import re

path = (sys.argv[1] if len(sys.argv) > 1 else
        '@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/'
        'fullnet-continuous-20261027/queue/n01-c32-chain/native/captures/native-native-base/'
        'ISA/CS_asm7a38a17947f21ae6_simd32_0000000000000003_2_orig.asm')

isa = P.parse(path)
print('instructions', len(isa))
print()
print('--- opcode census')
for k, v in collections.Counter(i['op'] for i in isa).most_common():
    print('%6d  %s' % (v, k))
print()
print('--- send message forms')
for k, v in collections.Counter((i['op'], P.rd_of(i), P.msg_of(i)) for i in isa
                                if i['op'].startswith('send')).most_common():
    print('%6d  %s' % (v, k))
print()
print('--- mov region-shape census (dst width, src region)')
c = collections.Counter()
for i in isa:
    if i['op'] != 'mov' or len(i['ops']) != 2:
        continue
    d, s = i['ops']
    dm = re.search(r'<([^>]*)>', d)
    sm = re.search(r'<([^>]*)>', s)
    c[(i['flags'], dm.group(1) if dm else '-', sm.group(1) if sm else '-')] += 1
for k, v in c.most_common():
    print('%6d  flags=%-6s dst<%-10s src<%-10s' % (v, k[0], k[1], k[2]))
print()
print('--- dpas census')
for k, v in collections.Counter((i['op'], i['flags']) for i in isa
                                if i['op'].startswith('dpas')).most_common():
    print('%6d  %s' % (v, k))
