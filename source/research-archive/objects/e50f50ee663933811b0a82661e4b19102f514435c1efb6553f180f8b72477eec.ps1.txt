 Write-Output ('INTERVAL_JOB '+$job.id+' '+$entry.status+' exit='+$exit+' frame='+$frameSha+' ts='+$(if($rec){$rec.timestamp_count_per_frame}else{'missing'})+' selected_samples='+$entry.selected_samples_count)
