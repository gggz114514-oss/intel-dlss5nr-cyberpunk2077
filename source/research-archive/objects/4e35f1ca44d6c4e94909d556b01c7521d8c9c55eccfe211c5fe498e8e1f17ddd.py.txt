import sys
from pathlib import Path
import numpy as np

sys.path.insert(0, str(Path(__file__).parent))
import c32blk as M
from validate import bk, PUSH, ptr

B = Path(r"@NR_WORKSPACE@")
W = B / "work/dsh-fullnet-static-20261029"
src = W / "C00-color-baseline/runs/diag65-69"
tea = B / "work/codex-astra-fp16-int8-audit-20261024/int8-quality-first-20261029/real-int8-24/dump-fp16"

x, valid = M.load_block(src / "c00-block66_out.bin", -1, -1)
blk = M.Block(bk("half", 67), bk("i8", 67), bk("sc", 67), bk("aux", 67), ptr(PUSH))
y, tr = blk.forward(x, valid, chunk_windows=M.NWIN, trace=True)
idx, vv = M.token_index(-1, -1)
gpu = np.fromfile(src / "c00-block67_out.bin", dtype="<f2")[idx][vv].astype(np.float64)
seed = tr["seed"][vv].astype(np.float64)
attn = tr["attn"][vv].astype(np.float64)
print("rms seed/attn", np.sqrt((seed ** 2).mean()), np.sqrt((attn ** 2).mean()))
A = np.stack([seed.ravel(), attn.ravel()], 1)
coef, *_ = np.linalg.lstsq(A, gpu.ravel(), rcond=None)
print("fit gpu = a*seed + b*attn:", coef)
print("relL2 fit", M.rel_l2((A @ coef).reshape(gpu.shape), gpu))
res = gpu - A @ coef
print("residual rms", np.sqrt((res ** 2).mean()))
# try scalars on attention
for b in (0.5, 0.6, 0.7, 0.75, 0.8, 0.9, 1.0, 1.1):
    print(f" b={b:.2f} relL2", round(M.rel_l2(seed + b * attn, gpu), 4))
print("teacher in, attention fit:")
x2, v2 = M.load_block(tea / "dump_block66_out.bin", -1, -1)
y2, tr2 = blk.forward(x2, v2, chunk_windows=M.NWIN, trace=True)
tch = np.fromfile(tea / "dump_block67_out.bin", dtype="<f2")[idx][vv].astype(np.float64)
s2 = tr2["seed"][vv].astype(np.float64)
a2 = tr2["attn"][vv].astype(np.float64)
print(" model(teacher) relL2:", M.rel_l2((s2 + a2), tch))
for b in (0.7, 0.8, 0.9, 1.0, 1.1, 1.2):
    print(f" b={b:.2f} relL2", round(M.rel_l2(s2 + b * a2, tch), 4))
