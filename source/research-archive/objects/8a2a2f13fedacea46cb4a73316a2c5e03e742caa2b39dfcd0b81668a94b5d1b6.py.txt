"""Generate a CARD.txt whose shader identities come from CURRENT-232-ROWS, never hand-written.

Usage: python make_card.py <card-id> <family> <width> "<title>" [<body file>]

Resolves the family/width group from the ledger, prints the ordinal list, the per-row SPV
paths and their sha256, and writes CARD.txt with those values baked in. This exists because
a hand-written ordinal list was wrong once (n25: the real F04 C64 rows were 6/9/12/15/217/220/223,
not the 13/16/196..208 I had guessed).
"""
import sys
from collections import defaultdict
from pathlib import Path
sys.path.insert(0, str(Path(__file__).parent))
import COMMON as K

card_id, family, width, title = sys.argv[1], sys.argv[2], sys.argv[3], sys.argv[4]
body = Path(sys.argv[5]).read_text(encoding='utf-8') if len(sys.argv) > 5 else ''

rows = K.rows()
group = [r for r in rows if r['family'] == family and (width.lower() in str(r.get('width', '')).lower()
                                                        or width.lower() in str(r['name']).lower()
                                                        or width.lower() in str(r['source']).lower())]
if not group:
    # fall back: same family + same spv (the group is defined by the shader, not the label)
    bysha = defaultdict(list)
    for r in rows:
        bysha[r['spv_sha256']].append(r)
    for sha, rs in bysha.items():
        if rs[0]['family'] == family:
            group = rs
            break
ords = sorted(r['ordinal'] for r in group)
spvs = sorted({r['spv'] for r in group})
srcs = sorted({r['source'] for r in group})
sha = sorted({r['spv_sha256'] for r in group})

lines = ['卡 id: %s' % card_id, '族: %s 宽度: %s' % (family, width), '标题: %s' % title, '',
         '**身份（由 make_card.py 从 CURRENT-232-ROWS 现取，勿手写）**:',
         '  该 shader 覆盖 ordinal = %s' % ords,
         '  spv_sha256 = %s' % sha,
         '  spv(s)     = %s' % spvs,
         '  source(s)  = %s' % srcs, '',
         '> 手写 ordinal 出过一次错（n25 把 F05/F06 的行当成了 F04 C64）。凡本卡要替换的行，'
         '必须以 GPU-REQUEST.pass_ordinals 为准，且 rebase 后由 replaced==N 断言 + cpu-check + 金帧三重兜底。',
         '', body]
d = K.C / 'cards' / card_id
d.mkdir(parents=True, exist_ok=True)
(d / 'CARD.txt').write_text('\n'.join(lines) + '\n', encoding='utf-8')
print('wrote', d / 'CARD.txt')
print('ordinals', ords)
print('spv count', len(spvs), 'sha count', len(sha))
