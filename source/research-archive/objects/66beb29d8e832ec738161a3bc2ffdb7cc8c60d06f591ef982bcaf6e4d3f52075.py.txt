#!/usr/bin/env python3
"""dump a compact view of SITE-REGISTRY.json (CPU only)"""
import json
import sys
from pathlib import Path

W = Path("@NR_WORKSPACE@/work/dsh-xmx-handoff-validation-20261024")
d = json.load(open(W / "candidates/r07-output-proj/evidence/SITE-REGISTRY.json",
                    encoding="utf-8"))
what = sys.argv[1] if len(sys.argv) > 1 else "sites"
if what == "sites":
    for k, v in d["modules"].items():
        print("=== %s  family=%s  lds=%dB  %s" % (k, v["family"], v["lds_bytes"],
                                                  [x["name"] + ":" + str(x["bytes"]) for x in v["lds"]]))
        for s in v["fp16_sites"]:
            b = s["B_origin"]
            print("  woff=%-6d rid=%-5d %-4s B=%s[%s] stride=%s %s" % (
                s["word_offset"], s["result_id"], s["class"], b.get("slot"),
                b.get("address"), b.get("stride"), b.get("layout")))
            print("      C=%s" % json.dumps(s["C_chain"])[:400])
        print("  fp16 stores:", v["half_store_sites"])
        print("  int8 stores:", [(c["word_offset"],
                                   c["slot"], c["address"], c["stride"], c["layout"])
                                  for c in v["int8_store_sites"]])
        print("  loads:", [(c["word_offset"], c["slot"], c["cmat"] and c["cmat"]["component"],
                             c["address"], c["stride"], c["layout"]) for c in v["cm_loads"]])
elif what == "passes":
    for p in d["passes"]:
        print("%-3d %-24s %-22s grid=%s mma=%d" % (p["plan_order"], p["pass_name"],
                                                   p["module_file"], p["grid"],
                                                   p["fp16_static_mma_count"]))
        for s in p["sites"]:
            print("      woff=%-6d rid=%-5d %s  (%s)" % (s["word_offset"], s["result_id"],
                                                         s["class"], s["class_reason"]))
