"""Inspect a window of a function region in two SPIR-V modules. CPU only."""
import sys
from pathlib import Path
CARD = Path(__file__).parent
sys.path.insert(0, str(CARD.parent.parent))
import COMMON as K
ST = K.ST

a = CARD / sys.argv[1]
b = CARD / sys.argv[2]
lo = int(sys.argv[3])
hi = int(sys.argv[4])
pm, cm = ST.Module(a), ST.Module(b)
pr, cr = ST.function_regions(pm), ST.function_regions(cm)
for j in range(lo, hi):
    if j < len(pr[0]):
        op, ops, wc, wr = pm.instructions[pr[0][j]]
        print('P %5d op=%-5d wc=%-3d %s' % (j, op, wc, ops[:7]))
print()
for j in range(lo, hi):
    if j < len(cr[0]):
        op, ops, wc, wr = cm.instructions[cr[0][j]]
        print('C %5d op=%-5d wc=%-3d %s' % (j, op, wc, ops[:7]))
