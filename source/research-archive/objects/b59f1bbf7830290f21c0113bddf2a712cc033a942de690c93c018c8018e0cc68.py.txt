"""Patch health.py: normalise path separators when comparing head rows to the handoff rows."""
from pathlib import Path

p = Path(__file__).with_name('health.py')
t = p.read_text(encoding='utf-8')
old = "    if r['spv'] != h['spv']:"
new = "    if str(r['spv']).replace('/', chr(92)) != str(h['spv']).replace('/', chr(92)):"
if old in t and new not in t:
    t = t.replace(old, new)
    p.write_text(t, encoding='utf-8')
    print('patched')
else:
    print('already patched or anchor missing')
