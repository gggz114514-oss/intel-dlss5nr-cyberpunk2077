"""n13b-f13-contract: SPIR-V census / parent-vs-candidate structural comparison.

Reports opcode multisets, NoContraction id sets, storage-class byte totals and the
FP-expression census (Fma / FAdd / FMul / FNegate / FDiv / conversions / cooperative
matrix ops), so the numerics contract can be checked item by item.
CPU only.
"""
import json
import sys
from collections import Counter
from pathlib import Path

CAMP = Path(__file__).resolve().parents[3]
sys.path.insert(0, str(CAMP))
import COMMON as K            # noqa: E402
import spv_tools_local as ST  # noqa: E402

NAMES = {
    12: 'OpExtInst', 14: 'OpTypeVoid', 59: 'OpVariable', 61: 'OpLoad', 62: 'OpStore',
    65: 'OpAccessChain', 79: 'OpUndef', 80: 'OpCompositeConstruct', 81: 'OpCompositeExtract',
    93: 'OpSConvert', 109: 'OpConvertFToU', 112: 'OpConvertUToF',
    126: 'OpFNegate', 128: 'OpIAdd', 129: 'OpFAdd', 130: 'OpISub', 131: 'OpFSub',
    132: 'OpIMul', 133: 'OpFMul', 134: 'OpUDiv', 135: 'OpSDiv', 136: 'OpFDiv',
    137: 'OpUMod', 138: 'OpSRem', 139: 'OpSMod', 141: 'OpFRem', 142: 'OpFMod',
    154: 'OpLogicalAnd', 155: 'OpLogicalNot', 164: 'OpBitwiseOr', 165: 'OpBitwiseAnd',
    169: 'OpSelect', 171: 'OpIEqual', 173: 'OpUGreaterThan', 174: 'OpSGreaterThan',
    177: 'OpULessThan', 178: 'OpSLessThan', 195: 'OpShiftRightLogical',
    199: 'OpShiftLeftLogical', 245: 'OpPhi', 246: 'OpLoopMerge', 247: 'OpSelectionMerge',
    249: 'OpBranch', 250: 'OpBranchConditional', 251: 'OpSwitch', 252: 'OpKill',
    253: 'OpReturn', 255: 'OpFunctionEnd', 54: 'OpFunction', 56: 'OpFunctionCall',
    248: 'OpLabel',
    # KHR cooperative matrix
    4455: 'OpCooperativeMatrixLengthKHR', 4456: 'OpCooperativeMatrixLoadKHR',
    4457: 'OpCooperativeMatrixStoreKHR', 4458: 'OpCooperativeMatrixMulAddKHR',
    5358: 'OpCooperativeMatrixConvertKHR',
    # fp16 conversions
    4418: 'OpFConvert', 4419: 'OpFConvert(?)', 4420: 'OpQuantizeToF16',
}
FP = {126: 'FNegate', 129: 'FAdd', 131: 'FSub', 133: 'FMul', 136: 'FDiv', 141: 'FRem', 142: 'FMod'}
CONV = {109: 'FToU', 112: 'UToF', 4418: 'FConvert', 4420: 'QuantizeToF16'}


def census(path):
    m = ST.Module(Path(path))
    ops = Counter()
    fma_sites = []       # OpExtInst Fma (ext instruction 50)
    for op, o, idx, count in m.instructions:
        ops[op] += 1
        if op == 12 and len(o) >= 5 and o[3] == 50:
            fma_sites.append((o[1], tuple(o[4:7])))
    nc = set(ST.no_contraction_ids(m)) if hasattr(ST, 'no_contraction_ids') else set()
    return dict(path=str(path), bytes=Path(path).stat().st_size,
                instructions=len(m.instructions), bound=m.bound,
                opcodes={str(k): v for k, v in sorted(ops.items())},
                fp={v: ops[k] for k, v in FP.items() if ops[k]},
                conversions={v: ops[k] for k, v in CONV.items() if ops[k]},
                coop_matrix={v: ops[k] for k, v in NAMES.items()
                             if v.startswith('OpCooperative') and ops[k]},
                fma_sites=len(fma_sites),
                no_contraction=sorted(nc), nc_count=len(nc),
                storage=m.storage_bytes())


def main():
    card = Path(__file__).resolve().parents[1]
    par = card / 'parent-rebuild.spv'
    cand = card / 'candidate.spv'
    a = census(par)
    b = census(cand)
    ka, kb = set(a['opcodes']), set(b['opcodes'])
    delta = {}
    for k in sorted(ka | kb, key=int):
        va, vb = a['opcodes'].get(k, 0), b['opcodes'].get(k, 0)
        if va != vb:
            delta[NAMES.get(int(k), 'op' + k)] = [va, vb]
    out = dict(parent=a, candidate=b, opcode_delta=delta,
               nc_equal=(a['no_contraction'] == b['no_contraction']),
               fp_equal=(a['fp'] == b['fp']),
               conv_equal=(a['conversions'] == b['conversions']),
               coop_equal=(a['coop_matrix'] == b['coop_matrix']),
               fma_equal=(a['fma_sites'] == b['fma_sites']),
               storage_equal=(a['storage'] == b['storage']))
    card.joinpath('build').mkdir(exist_ok=True)
    card.joinpath('build', 'spv-census.json').write_text(json.dumps(out, indent=2), encoding='utf-8')
    print('parent ', a['instructions'], 'instr,', a['bytes'], 'B, nc', a['nc_count'])
    print('cand   ', b['instructions'], 'instr,', b['bytes'], 'B, nc', b['nc_count'])
    print('fp      ', a['fp'], '->', b['fp'], 'equal', out['fp_equal'])
    print('conv    ', a['conversions'], '->', b['conversions'], 'equal', out['conv_equal'])
    print('coop    ', a['coop_matrix'], '->', b['coop_matrix'], 'equal', out['coop_equal'])
    print('fma     ', a['fma_sites'], '->', b['fma_sites'], 'equal', out['fma_equal'])
    print('storage ', a['storage'], '->', b['storage'], 'equal', out['storage_equal'])
    print('nc      ', out['nc_equal'])
    print('opcode delta', json.dumps(delta, indent=2))


if __name__ == '__main__':
    main()
