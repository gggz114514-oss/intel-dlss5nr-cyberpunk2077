import re, collections, json
from pathlib import Path
F = Path(r"@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/fullnet-mechanism-transfer-20261026/f05-k-consumer-tile/c256/captures/native-f20-pixel-owner/ISA/CS_asmfd78c91bb8744f0f_simd32_000000000000000e_13_orig.asm")
lines = F.read_text(encoding="utf-8", errors="replace").splitlines()

msgs = collections.Counter()
send_by_msg = []
for i, l in enumerate(lines):
    if "//" not in l:
        continue
    cmt = l.split("//", 1)[1].strip()
    m = re.search(r"(wr:\d+\+\d+, rd:\d+); (\w+)", cmt)
    if not m:
        continue
    kind = m.group(2)
    msgs[kind] += 1
    if "send" in l:
        send_by_msg.append((i, kind, l))

print("== instruction classes in the ISA (by comment) ==")
for k, v in msgs.most_common(40):
    print("%5d %s" % (v, k))
print()
print("total send lines:", sum(1 for l in lines if re.search(r"\bsend\b", l)))
print("total dpas lines:", sum(1 for l in lines if re.search(r"\bdpas\b", l)))
print()
print("== send breakdown ==")
c = collections.Counter(k for _, k, _ in send_by_msg)
for k, v in c.most_common():
    print("%5d %s" % (v, k))
