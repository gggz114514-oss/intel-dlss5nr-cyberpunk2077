"""Detailed send-message accounting + targeted diff lines for a card's two captures."""
import json, re, sys
from collections import Counter
from pathlib import Path
sys.path.insert(0, str(Path(__file__).parent))
import COMMON as K
import ISA


def msg_counts(path):
    isa = ISA.parse(path)
    c = Counter()
    for i in isa:
        if i['op'].startswith('send') and i['comment']:
            m = re.search(r';\s*([a-z0-9_.]+)', i['comment'])
            c[(i['op'], m.group(1) if m else i['comment'])] += 1
        elif i['op'].startswith('send'):
            c[(i['op'], i['comment'] or '?')] += 1
    return c


def main(card):
    a = K.C / 'queue' / card / 'native/captures/native-native-base'
    b = K.C / 'queue' / card / 'native/captures/native-native-cand'
    fa = {p.name: p for p in a.rglob('*.asm')}
    fb = {p.name: p for p in b.rglob('*.asm')}
    oa = [n for n in fa if n not in fb]
    ob = [n for n in fb if n not in fa]
    out = dict(card=card, base_file=oa, cand_file=ob)
    if len(oa) == 1 and len(ob) == 1:
        ca, cb = msg_counts(fa[oa[0]]), msg_counts(fb[ob[0]])
        out['messages_base'] = {'%s %s' % k: v for k, v in sorted(ca.items())}
        out['messages_cand'] = {'%s %s' % k: v for k, v in sorted(cb.items())}
        keys = {('%s %s' % k) for k in set(ca) | set(cb)}
        out['message_delta'] = {k: [ca.get(tuple(k.split(' ', 1)), 0), cb.get(tuple(k.split(' ', 1)), 0)]
                                for k in sorted(keys)
                                if ca.get(tuple(k.split(' ', 1)), 0) != cb.get(tuple(k.split(' ', 1)), 0)}
        d = ISA.diff(fa[oa[0]], fb[ob[0]])
        out['diff_counts'] = {tag: len(v) for tag, v in d['changes'].items()}
        out['inserted'] = [x['raw'] for v in d['changes'].get('insert', []) for x in v['b']][:20]
        out['deleted'] = [x['raw'] for v in d['changes'].get('delete', []) for x in v['a']][:20]
        out['replaced'] = [[x['raw'] for x in v['a']] + [y['raw'] for y in v['b']]
                           for v in d['changes'].get('replace', [])][:14]
    K.write_json(K.C / 'cards' / card / 'ISA-ACCOUNT.json', out)
    print(json.dumps({k: v for k, v in out.items() if k not in ('inserted', 'deleted', 'replaced')},
                     indent=2)[:3000])
    print('--- inserted ---')
    for x in out.get('inserted', []):
        print('  ', x)
    print('--- deleted ---')
    for x in out.get('deleted', []):
        print('  ', x)
    print('--- replaced (first 8) ---')
    for x in out.get('replaced', [])[:8]:
        print('   A:', x[0][:120] if x else None)
        print('   B:', x[1][:120] if len(x) > 1 else None)


if __name__ == '__main__':
    main(sys.argv[1])
