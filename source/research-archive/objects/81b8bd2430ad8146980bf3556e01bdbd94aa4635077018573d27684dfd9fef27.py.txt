from pathlib import Path
import json,hashlib,statistics,math,sys
D=Path(__file__).resolve().parent
rd=lambda p:json.loads(p.read_text(encoding='utf-8-sig'))
gold='2b63d4ba1967eebcda61698d538963f59bccaf6d690ef8168f1695a46c5d658d'
def check(j):
 p=Path(j['runner_output']);r=rd(p/'port_network_receipt.json')
 assert hashlib.sha256(Path(j['plan']['path']).read_bytes()).hexdigest()==j['plan']['sha256']
 assert hashlib.sha256((p/'frame_out.bin').read_bytes()).hexdigest()==gold
 assert r['normal_close'] and r['warm']==1 and r['reps']==30 and r['feedback'] and r['gpu_passes']==232 and r['nan']==r['inf']==0
 return p,r
def ci(vals):
 m=statistics.mean(vals)
 if len(vals)==3:
  e=4.302652729911275*statistics.stdev(vals)/math.sqrt(3)
  return dict(values=vals,mean=m,ci95=[m-e,m+e])
 return dict(values=vals,mean=m,ci95=None)
mode=sys.argv[1]
if mode=='timestamps':
 vals={}
 for j in rd(D/'LUNA-TIMESTAMPS.json')['jobs']:
  p,r=check(j);assert r['per_pass_timing'] and len(r['per_pass_gpu'])==232
  rowmeans=[]
  for i,x in enumerate(r['per_pass_gpu']):
   assert x['index']==i and len(x['samples_ms'])==30
   rowmeans.append(statistics.mean(x['samples_ms']))
  vals[(j['pair'],j['candidate'])]=dict(network=statistics.mean(r['gpu_samples_ms']),F20=rowmeans[230],F01=rowmeans[0],F2=rowmeans[226],body7=sum(rowmeans[i] for i in [1,2,3,4,227,228,229]),other_intervals=sum(rowmeans)-rowmeans[230],interval_sum=sum(rowmeans))
 out=dict(process_means={str(k):v for k,v in vals.items()},parent_minus_candidate={k:ci([vals[(i,'parent')][k]-vals[(i,'candidate')][k] for i in range(3)]) for k in next(iter(vals.values()))},all180_samples_retained=True)
 (D/'TIMESTAMP-REVIEW.json').write_text(json.dumps(out,indent=2));print(json.dumps(out,indent=2))
else:
 jobs=rd(D/'LUNA-COUNTERS-PILOT.json')['jobs']
 if mode=='counters':jobs+=rd(D/'LUNA-COUNTERS-REST.json')['jobs']
 vals={};stats={};definitions={};frames={}
 for j in jobs:
  p,r=check(j);c=rd(p/'hardware-counters.json')
  assert c['hardware_passes']==1 and c['feedback'] and len(c['frames'])==30
  names=[x['name'] for x in c['metrics']];assert set(names)==set(j['expected_metrics'])
  definitions[j['metric_group']]=c['metrics']
  selected={}
  for ordinal in [0,1,226,230]:
   samples=[]
   for f in c['frames']:
    assert len(f['passes'])==232
    x=f['passes'][ordinal];assert x['index']==ordinal
    assert len(x['values'])==len(names) and all(math.isfinite(v) for v in x['values'])
    samples.append(x['values'])
   selected[str(ordinal)]={name:statistics.mean(x[k] for x in samples) for k,name in enumerate(names)}
  key=(j['metric_group'],j['pair'],j['candidate']);vals[key]=selected
  frames[j['id']]=len(c['frames'])
  ps=rd(p/'pipeline-statistics.json');p38=next(x for x in ps['pipelines'] if x['index']==38)
  stats[j['id']]=p38
 out=dict(process_means={str(k):v for k,v in vals.items()},pipeline38=stats,definitions=definitions,metric_frame_counts=frames,all_frames_gold=True,all_hardware_passes_one=True)
 if mode=='counters':
  comparison={}
  for group in ['traffic','occupancy','stalls']:
   for name in vals[(group,0,'parent')]['230']:
    pa=[vals[(group,i,'parent')]['230'][name] for i in range(3)];ca=[vals[(group,i,'candidate')]['230'][name] for i in range(3)]
    comparison[group+':'+name]=dict(parent=pa,candidate=ca,candidate_minus_parent=ci([c-p for p,c in zip(pa,ca)]))
  out['F20_comparison']=comparison
 (D/('COUNTER-REVIEW.json' if mode=='counters' else 'PILOT-REVIEW.json')).write_text(json.dumps(out,indent=2))
 print(json.dumps(out.get('F20_comparison',dict(process_means=out['process_means'],all_frames_gold=True)),indent=2))
