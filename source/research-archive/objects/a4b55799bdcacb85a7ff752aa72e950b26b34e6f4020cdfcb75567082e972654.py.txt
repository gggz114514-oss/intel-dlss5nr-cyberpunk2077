"""Build card n28's candidate for F20 / ord 230.

Candidate: the stage-11 pixel-owner tail is re-owned to the subgroup that produced the rows
it reads (rows 8*sg..8*sg+7, one output channel per lane) and the stage9->tail workgroup
barrier is removed.  Nothing else changes.

Steps: compile -> NoContraction transfer (45 IDs, same recorded decoration parent) ->
SPIR-V validation -> structural census (barriers 16 -> 15, shared vars identical, FP op
multiset of the tail) -> plan on the chain head (one line) -> cpu_check -> BUILD.json.

CPU only: no Vulkan device is touched.
"""

def census(spv):
    b = Path(spv).read_bytes()
    w = list(struct.unpack('<%dI' % (len(b) // 4), b))
    i, hist, blocks, funcs = 5, collections.Counter(), 0, 0
    barriers, consts = [], {}
    while i < len(w):
        wc = w[i] >> 16
        op = w[i] & 0xFFFF
        hist[op] += 1
        if op == 43:                                     # OpConstant
            consts[w[i + 2]] = w[i + 3]
        elif op == 224:
            barriers.append(tuple(w[i + 1:i + wc]))
        elif op == 248:
            blocks += 1
        elif op == 54:
            funcs += 1
        i += wc
    # the scope / memory-semantics operands are ids of OpConstant instructions here, so the
    # comparison that matters is on the resolved values (execution scope, memory scope,
    # semantics), not on the ids.
    resolved = sorted({tuple(consts.get(x, 'id?') for x in bar) for bar in barriers})
    return dict(words=len(w), bytes=len(b), funcs=funcs, blocks=blocks,
                barriers=len(barriers),
                barrier_scopes=[[int(x) for x in t] for t in resolved],
                hist={OPS.get(k, str(k)): v for k, v in hist.items() if k in OPS})


def wg_vars(spv):
    """Workgroup variables with their element counts (shared memory footprint)."""
    b = Path(spv).read_bytes()
    w = list(struct.unpack('<%dI' % (len(b) // 4), b))
    i, insts = 5, []
    while i < len(w):
        wc = w[i] >> 16
        insts.append((w[i] & 0xFFFF, list(w[i + 1:i + wc])))
        i += wc
    consts = {o[1]: o[2] for op, o in insts if op == 43}
    arrays = {o[0]: (o[1], consts.get(o[2])) for op, o in insts if op == 28}
    out = []
    for op, o in insts:
        if op == 59 and o[2] == 4:
            base, t = o[1], o[0]
            ptr = None
            for op2, o2 in insts:
                if op2 == 32 and o2[0] == t:       # OpTypePointer result==var type
                    ptr = o2[2]
            elem, n = arrays.get(ptr, (ptr, '?'))
            out.append(dict(id=base, array_len=n, elem_type=elem))
    return out

# ---------------------------------------------------------------- 2. NC transfer
#
# Same policy as the campaign recipe (transfer the recorded decoration parent's
# NoContraction pattern), with one difference that the candidate forces: the parent's
# `boundary_project()` function no longer exists (the tail fold is inlined per channel), so
# the parent function that carries 16 of the 45 decorations has no counterpart.  The
# campaign helper asserts on that case, so the alignment is done here with the same
# difflib + operand-signature procedure and the vanished function is *recorded* instead of
# asserted.  The child's own `precise`-derived decorations inside the new tail are kept on
# top, so every FP operation of the live tail is NoContraction, exactly as all 16 FP
# operations of the parent's tail function were.
import difflib

pm = ST.Module(DEC_PARENT)
rm = ST.Module(raw)


def tolerant_transfer(parent, child):
    pregions = ST.function_regions(parent)
    cregions = ST.function_regions(child)
    p_skel = {i: [(parent.instructions[x][0], parent.instructions[x][3]) for x in r]
              for i, r in enumerate(pregions)}
    c_skel = {i: [(child.instructions[x][0], child.instructions[x][3]) for x in r]
              for i, r in enumerate(cregions)}
    keep, used, mapping, vanished = set(), set(), [], []
    for pi, preg in enumerate(pregions):
        pattern = ST.decoration_pattern(parent, preg)
        if not pattern:
            continue
        want, best = p_skel[pi], None
        for ci, have in c_skel.items():
            if ci in used or not have:
                continue
            blocks = difflib.SequenceMatcher(None, want, have, autojunk=False).get_matching_blocks()
            pos = {}
            for b in blocks:
                for k in range(b.size):
                    pos[b.a + k] = b.b + k
            if not all(p in pos for p in pattern):
                continue
            matched = sum(b.size for b in blocks)
            if best is None or matched > best[0]:
                best = (matched, ci, pos)
        if best is None:
            vanished.append(dict(parent_function_index=pi, instructions=len(want),
                                 decorations=len(pattern)))
            continue
        matched, target, pos = best
        used.add(target)
        creg = cregions[target]
        for p in sorted(pattern):
            pidx, cidx = preg[p], creg[pos[p]]
            pop, pops = parent.instructions[pidx][0], parent.instructions[pidx][1]
            cop, cops = child.instructions[cidx][0], child.instructions[cidx][1]
            assert pop == cop, (pidx, cidx, pop, cop)
            assert parent.instructions[pidx][3] == child.instructions[cidx][3]
            prid = ST.result_id_of(pop, pops)
            crid = ST.result_id_of(cop, cops)
            assert prid is not None and crid is not None
            assert parent.operand_signature(prid, 3) == child.operand_signature(crid, 3), \
                ('mapped instruction differs', pi, p)
            keep.add(crid)
        mapping.append(dict(parent_function_index=pi, child_function_index=target,
                            parent_instructions=len(want),
                            child_instructions=len(c_skel[target]),
                            matched_instructions=matched, pattern_size=len(pattern)))
    own = set(ST.no_contraction_ids(child))
    extra = own - keep
    # The campaign recipe discards glslang's own `precise` decorations wherever the recorded
    # pattern does not cover them (that is how the parent's 45 are produced).  The candidate
    # must keep exactly the decorations of the *new tail code*, because the parent's tail
    # function was 16/16 NoContraction: every FP multiply and add of that fold must stay
    # non-contractible.  The tail is the last code of main, so its operations are the last
    # FMul / FAdd of function 0.
    main_region = cregions[0]
    own_in_main = []
    for x in main_region:
        op, ops = child.instructions[x][0], child.instructions[x][1]
        rid = ST.result_id_of(op, ops)
        if op in (133, 129) and rid in extra:
            own_in_main.append((x, op, rid))
    own_in_main.sort()
    tail_ops = own_in_main[-2:]
    assert sorted(op for _, op, _ in tail_ops) == [129, 133], tail_ops
    keep |= {rid for _, _, rid in tail_ops}
    extra_ops = collections.Counter(op for _, op, _ in tail_ops)
    record_extra = dict(extra_ops)
    info = dict(alignment='longest-matching-blocks with operand-signature verification',
                functions_mapped=len(mapping), mapping=mapping,
                parent_functions_without_child_counterpart=vanished,
                new_tail_decorations=[dict(instruction_index=x, opcode=op)
                                      for x, op, _ in tail_ops],
                discarded_child_own_decorations=[dict(instruction_index=x, opcode=op)
                                                 for x, op, rid in own_in_main[:-2]],
                new_tail_decorations_ops=record_extra,
                parent_decorations_total=len(ST.no_contraction_ids(parent)),
                child_decorations_total=len(keep))
    return sorted(keep), info
