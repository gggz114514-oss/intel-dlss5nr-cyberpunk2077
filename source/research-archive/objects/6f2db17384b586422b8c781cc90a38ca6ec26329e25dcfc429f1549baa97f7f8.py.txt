from pathlib import Path
import shlex,numpy as np,json
D=Path(__file__).parent;P=D.parent/'corrected-20261029/plan.txt';out=[]
for line in P.read_text().splitlines():
 if not line.startswith('BUFFER '):continue
 t=shlex.split(line)
 if ('scale' not in t[1] and '_sc' not in t[1]) or t[3]=='-':continue
 v=np.fromfile(t[3],dtype='<f4')
 if len(v)<4:continue
 out.append({'name':t[1],'path':t[3],'first4':v[:4].tolist(),'pair_products':[float(v[0]*v[1]),float(v[2]*v[3])]})
(D/'SCALE-PAIR-CENSUS.json').write_text(json.dumps(out,indent=2),encoding='utf-8')
for x in out:
 if any(abs(z-127)<.01 for z in x['pair_products']): print(x)
print('Checked',len(out),'buffers; pair products are screening only, require consumer ABI.')
