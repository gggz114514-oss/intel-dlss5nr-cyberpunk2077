"""Where does the frame time actually sit? Rank the 232 PASSes by the recorded per-PASS timing.

Source: evidence of the census (INTEL-ROWS.json, field current_baseline_timing_ms) which
timed each PASS in isolation (fixed frame0 and fb2 frame1, warm0 short feedback). Those
numbers are NOT additive (section 3 of the book) and are not ordinary timing, but as a
ranking they say which dispatch positions are worth attacking.
"""
import json, sys
from collections import defaultdict
from pathlib import Path
sys.path.insert(0, str(Path(__file__).parent))
import COMMON as K

src = K.D / 'remaining-amd-production-audit-20261027' / 'INTEL-ROWS.json'
rows = K.read_json(src)
cov = K.read_json(K.C / 'evidence' / 'COVERAGE-NOW.json')
introduced = {t['ordinal']: t for t in cov['table']}

recs = []
for r in rows:
    t = r.get('current_baseline_timing_ms') or {}
    f0 = (t.get('fixed-frame0') or {}).get('median')
    f1 = (t.get('fb2-frame1') or {}).get('median')
    recs.append(dict(ordinal=r['ordinal'], name=r['name'], family=r['family'],
                     fixed_ms=f0, fb2_ms=f1,
                     step=introduced.get(r['ordinal'], {}).get('step_label'),
                     changed=introduced.get(r['ordinal'], {}).get('introduced_by_step') != 0))
tot0 = sum(x['fixed_ms'] or 0 for x in recs)
tot1 = sum(x['fb2_ms'] or 0 for x in recs)
by_fam = defaultdict(float)
for x in recs:
    by_fam[x['family']] += x['fixed_ms'] or 0
ranked = sorted(recs, key=lambda x: -(x['fixed_ms'] or 0))
out = dict(source=str(src), note='isolated per-PASS GpuTime medians, NOT additive and not ordinary',
           n=len(recs), sum_fixed_ms=tot0, sum_fb2_ms=tot1,
           families=dict(sorted(by_fam.items(), key=lambda kv: -kv[1])),
           top30=ranked[:30], rows=recs)
K.write_json(K.C / 'evidence' / 'PASS-TIME-RANKING.json', out)

print('sum of isolated per-PASS medians: %.3f ms (frame is ~18 ms; not additive)' % tot0)
print('\nfamilies:')
for f, v in sorted(by_fam.items(), key=lambda kv: -kv[1]):
    n = sum(1 for x in recs if x['family'] == f)
    print('  %-4s %8.3f ms  %2d PASS  share %.1f%%' % (f, v, n, 100 * v / tot0))
print('\ntop 15 PASS:')
for x in ranked[:15]:
    print('  ord %3d %-30s %-4s %8.3f ms  %s' % (x['ordinal'], x['name'], x['family'],
                                                 x['fixed_ms'] or 0, x['step']))
