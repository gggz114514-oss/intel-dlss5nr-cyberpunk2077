"""Emit MISSING-NATIVE.md (and the numbers quoted in RESUME.txt) from NATIVE-INDEX.json."""
import json
from pathlib import Path
from collections import defaultdict

CARD = Path(__file__).parent
j = json.loads((CARD / 'evidence' / 'NATIVE-INDEX.json').read_text(encoding='utf-8'))
S = j['summary']
cov = j['coverage']
caps = j['captures']
base = j['baseline']

baseline_dirs = {c['isa_dir'] for c in caps if (c['plan'] or {}).get('equals_baseline')}
per_pipe = defaultdict(list)
for c in caps:
    for e in c['records']:
        if e['identity'] != 'CURRENT':
            continue
        for p in e['baseline_pipelines']:
            per_pipe[p].append((c, e))

rows = []
for p in sorted(int(x) for x in base['pipelines']):
    bp = base['pipelines'][str(p)]
    ev = per_pipe.get(p, [])
    # prefer: baseline-plan capture *with* driver statistics, then any with statistics
    order = sorted(ev, key=lambda ce: (0 if (ce[0]['isa_dir'] in baseline_dirs and
                                             (ce[1].get('driver_stats'))) else
                                       1 if ce[1].get('driver_stats') else
                                       2 if ce[0]['isa_dir'] in baseline_dirs else 3))
    c, e = (order[0] if order else (None, {}))
    ds = (e or {}).get('driver_stats') or {}
    scan = (e or {}).get('isa_scan') or {}
    na = lambda v: 'n/a' if v is None else v
    rows.append(dict(
        p=p, fam=','.join(bp['families']), npass=len(bp['passes']),
        first=bp['first_pass'], spv_sha=bp['spv_sha256'][:12], khash=bp['kernel_hash'],
        insn=na(ds.get('Instruction Count', scan.get('instruction_lines'))),
        blocks=na(ds.get('Basic Block Count', scan.get('basic_blocks'))),
        scratch=na(ds.get('Scratch Memory Size')),
        wg=na(ds.get('Workgroup Memory Size', bp['shared_bytes'])),
        lo=na(ds.get('Loop Count')),
        ncaps=len({x[0]['isa_dir'] for x in ev}),
    ))

NON_GTPIN = 695

byfam = defaultdict(list)
for c in cov:
    byfam[c['family']].append(c['current_count'])
fam_lines = ['| 族 | PASS | 当前身份 | 历史仅 | 缺失 | 每 PASS 当前证据份数 min/median/max |',
             '|---|---|---|---|---|---|']
for f in sorted(S['per_family']):
    v = S['per_family'][f]
    cc = sorted(byfam[f])
    fam_lines.append('| %s | %d | %d | %d | %d | %d / %d / %d |'
                     % (f, sum(v.values()), v.get('CURRENT', 0), v.get('HISTORICAL-ONLY', 0),
                        v.get('MISSING', 0), cc[0], cc[len(cc) // 2], cc[-1]))
fam_table = '\n'.join(fam_lines)

thin = sorted(cov, key=lambda c: (c['current_count'], c['ordinal']))
minc = thin[0]['current_count']
thin_list = [x for x in thin if x['current_count'] == minc]

cur_hashes = len({r['khash'] for r in rows})
md = []
A = md.append
A('# 当前基线 native ISA 缺失清单 / nat-index')
A('')
A('生成于 %s（索引扫描时点 %s）。工作区仍在被并行卡写入（`queue/` 下其他卡在跑），'
  '捕获会话数会继续增长；本文与 `evidence/NATIVE-INDEX.json` 是同一时点快照。'
  % (__import__('time').strftime('%Y-%m-%dT%H:%M:%S'), j['generated']))
A('')
A('- 基线 plan: `%s`' % base['plan'])
A('- SHA256: `%s`（%s）' % (base['plan_sha256'],
                            '与合同 sha 一致' if base['plan_sha_matches'] else '与合同 sha 不一致'))
A('')
A('## 0. 结论')
A('')
A('- 扫到 **%d** 个 GTPin native ISA 文件、**%d** 个捕获会话；另有 %d 个 `*.text.*.asm` 是 IGA 静态反汇编，'
  '不是 native 捕获，已排除。' % (S['isa_files'], S['capture_sessions'], NON_GTPIN))
A('- **232 / 232 个 PASS 都有当前身份 ISA**（40 / 40 个 pipeline 都有）。')
A('- **缺失列表为空：当前身份下没有任何 PASS 需要补捕获。**')
A('- **现在最少需要的新 native 捕获次数 = 0。**')
A('- 未来每张候选卡：**每个臂 1 次** `--dump_isa`（一次跑完整 232 个派发 -> 40 个 ISA）。'
  '基线臂通常不需要再采，除非基线 plan/SPV 变了。')
A('')
A('## 1. 缺失列表（按 shader 组）')
A('')
A('**空。** 下面是每族的当前身份覆盖，也就是以后每卡 native 复核要花多少证据的底账：')
A('')
A(fam_table)
A('')
A('“每 PASS 当前证据份数”= 有多少个不同捕获会话能为该 PASS 提供当前身份 ISA（按会话去重）。'
  'min 列是最薄的 PASS：')
A('')
for x in thin_list[:8]:
    A('- PASS %d `%s`（%s, pipeline %d）当前证据 %d 份'
      % (x['ordinal'], x['name'], x['family'], x['pipeline'], x['current_count']))
A('')
A('## 2. 当前身份的可复现实证')
A('')
A('- 有 **%d** 个捕获会话，其 40 个 pipeline 的 kernel hash 全部等于基线指纹'
  '（`captures_full_current`）。它们来自不同卡/不同时间，但逐 pipeline 编译结果相同：'
  % S['captures_full_current'])
for d in S['fingerprint_sources']:
    A('  - `%s`' % d['isa_dir'])
A('- 这说明：只要 plan 与 SPV 不变，一次 `--dump_isa` 就能复现同一套 ISA，不存在'
  '“每次捕获都不一样所以不能复用”的问题。')
A('')
A('## 3. 每 pipeline 当前身份指纹表（native 复核的对照物）')
A('')
A('| pipe | 族 | PASS 数 | 首个 PASS | SPV sha(12) | kernel hash | Instruction | BasicBlock | Scratch | WG mem | Loop | 当前 ISA 会话数 |')
A('|---|---|---|---|---|---|---|---|---|---|---|---|')
for r in rows:
    A('| %d | %s | %d | %s | `%s` | `%s` | %s | %s | %s | %s | %s | %d |'
      % (r['p'], r['fam'], r['npass'], r['first'], r['spv_sha'], r['khash'],
         r['insn'], r['blocks'], r['scratch'], r['wg'], r['lo'], r['ncaps']))
A('')
A('Instruction/BasicBlock/Scratch/WG/Loop 优先取同卡 `pipeline-statistics.json`（驱动 '
  '`VkPipelineExecutableStatistics`）。没有同卡资源运行时回落为 ISA 文本静态计数'
  '（`isa_scan.instruction_lines` / `basic_blocks`），两者口径不同；每条记录的 '
  '`driver_stats.source` 标明来源，`driver_stats = null` 表示该记录只有静态计数。')
A('')
A('## 4. 历史身份 ISA（不得当成当前身份）')
A('')
A('- `CURRENT` 记录 %d 条 / `HISTORICAL` 记录 %d 条。'
  % (S['isa_identity_counts'].get('CURRENT', 0), S['isa_identity_counts'].get('HISTORICAL', 0)))
A('- 全区共 %d 个不同 kernel hash，当前身份只占 %d 个。'
  % (S['distinct_native_hashes'], cur_hashes))
A('- 历史 ISA 只能作具名历史对照（旧候选、旧 plan 的 native 形状）；任何“当前身份”断言必须引用第 3 节的 kernel hash。')
A('- 反例检查：在“结构完整 + plan SPV 与基线同 sha”的捕获里逐 pipeline 重比 kernel hash，'
  '**不一致记录 = %d 条**。' % S['spv_equal_binary_different_records'])
A('- 文件名完整性：ISA 尾部常量里内嵌同一 kernel hash，已抽查 %d 条，'
  '文件名与内嵌 hash 不一致 **%d** 条（说明文件名没有被改名/搬运污染）。'
  % (S['filename_hash_integrity']['checked'], S['filename_hash_integrity']['mismatch']))
A('- 驱动统计口径守卫：同卡 `pipeline-statistics.json` 可能是**候选臂**的，因此逐条校验'
  '`|Instruction Count - 静态指令行| <= 16`；保留 %d 条、剔除 %d 条'
  '（剔除的写在记录的 `driver_stats_rejected` 里，不用它冒充当前身份统计）。'
  % (S.get('driver_stats_records', 0), S.get('driver_stats_rejected', 0)))
A('')
A('## 5. 身份判定方法（可复核）')
A('')
A('1. **pipeline -> PASS 映射**：runner 用 `key = spv + npush + subgroup + bindings[]` 去重、'
  '按首次出现顺序编号（`phase-diagnosis-20261025/direct-clock/clock_runner.cpp:90` 的 plan 语法、'
  '`:354` 的 key）。用当前基线 plan 重算该顺序，与驱动 `pass_pipeline_indices` 逐元素一致'
  '（crosscheck = %s）。' % j['pipeline_map']['crosscheck']['derived_matches_driver'])
A('2. **指纹**：基线 plan 的捕获里，每个 pipeline 的 ISA 内嵌 64-bit kernel hash，构成 40 条指纹。')
A('3. **身份**：任一 ISA 的 kernel hash 命中某条指纹 => `CURRENT`（即基线在该 pipeline 上编译出的同一个二进制）；否则 `HISTORICAL`。')
A('4. **部分捕获的 index 不可信**：只有 1-6 个 ISA 的会话里，文件名中的 pipeline index 在该会话内从 0 重编号，'
  '不能当 plan pipeline 使用；这类记录只按 kernel hash 判身份，不做 PASS 归属。')
A('5. **plan 侧证据**逐条记录：来源 plan 路径+sha（manifest 记录优先，否则现算并标注 sha_source）、'
  '该 plan 内该 pipeline 的 SPV 路径+sha。')
A('')
A('## 6. 声明了但没产出 ISA 的运行（`--dump_isa` exit0 != 有 ISA 文件）')
A('')
noisa_p = CARD / 'evidence' / 'NO-ISA-RUNS.json'
if noisa_p.exists():
    N = json.loads(noisa_p.read_text(encoding='utf-8'))
    bad = [r for r in N['runs'] if r.get('status') == 'NO-ISA']
    ran = [r for r in bad if r.get('msg_file_bytes')]
    A('- manifest 里声明过的捕获目录 **%d** 个：有 ISA 的 %d 个，**没有 ISA 的 %d 个**。'
      % (N['declared_capture_dirs'], N['dirs_with_isa'], N['dirs_without_isa']))
    A('- 其中 **%d** 个既没有捕获目录也没有 ISA 目录（声明了但没跑/已清理）；'
      '**%d** 个目录在、`ISA/` 也在但一个 ISA 文件都没有；'
      '**%d** 个目录在但没有 `ISA/`。' %
      (sum(1 for r in bad if not r['capture_dir_exists']),
       sum(1 for r in bad if r['capture_dir_exists'] and r['isa_dir_exists']),
       sum(1 for r in bad if r['capture_dir_exists'] and not r['isa_dir_exists'])))
    A('- **%d** 个运行留有非空 GTPin `--msg_file` 日志却没有任何 ISA 文件——这就是'
      '“exit0 但没有 ISA 产物”，一律记缺失，不当成功：' % len(ran))
    for r in ran:
        A('  - `%s`（日志 %d B，plan `%s`）'
          % (r['capture_dir'], r['msg_file_bytes'],
             (r['plan_path'][0] if r['plan_path'] else 'n/a')))
    A('- 明细：`evidence/NO-ISA-RUNS.json`')
else:
    A('- `evidence/NO-ISA-RUNS.json` 未生成')
A('')
A('## 7. 复现')
A('')
A('```')
A('@NR_USER_HOME@/AppData/Local/Programs/Python/Python312/python.exe %s' % (CARD / 'build_index.py').as_posix())
A('@NR_USER_HOME@/AppData/Local/Programs/Python/Python312/python.exe %s' % (CARD / 'no_isa_runs.py').as_posix())
A('@NR_USER_HOME@/AppData/Local/Programs/Python/Python312/python.exe %s' % (CARD / 'report.py').as_posix())
A('```')
A('')
A('纯磁盘只读扫描：不运行 gtpin / clock_runner / port_zprof，不取 GPU 锁，不写卡目录以外任何文件。')

out = CARD / 'MISSING-NATIVE.md'
out.write_text('\n'.join(md), encoding='utf-8')
print('wrote', out, out.stat().st_size, 'bytes')
print('\n'.join(md[:46]))
