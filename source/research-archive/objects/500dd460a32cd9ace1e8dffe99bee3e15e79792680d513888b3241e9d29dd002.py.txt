import json, os, collections

W = r"@NR_WORKSPACE@/work/dsh-fullnet-static-20261029"
OUT = os.path.join(W, "C00-first-divergence/_work")
o = json.load(open(os.path.join(W, "ledger/FULLNET-ROWS-V2.json"), encoding="utf-8"))
rows = o["rows"]
print("candidate:", json.dumps(o["candidate"], ensure_ascii=False)[:800])
print("parents:", json.dumps(o["parents"], ensure_ascii=False)[:600])
print("lineage:", json.dumps(o["lineage"], ensure_ascii=False)[:800])
print("frame_invariance:", json.dumps(o["frame_invariance"], ensure_ascii=False)[:600])

fam = collections.Counter(r["family"] for r in rows)
print("\nfamily counts:", dict(fam))
print("\n--- stage table (unique stage per family) ---")
seen = []
for r in rows:
    key = (r["family"], r["stage"])
    if key not in seen:
        seen.append(key)
for f, s in seen:
    print("%-10s %s" % (f, s))

# resource adjacency structure sample
print("\n--- sample adjacency/bindings for key ordinals ---")
for r in rows:
    if r["name"] in ("f1c32_image_pre", "block1_swin_body_1", "c64_b5_qx_producer"):
        print(r["ordinal"], r["name"], json.dumps(r.get("resource_adjacency"), ensure_ascii=False)[:700])
        print("   bindings:", json.dumps(r.get("bindings"), ensure_ascii=False)[:500])
        print("   residual_feedback:", json.dumps(r.get("residual_feedback"), ensure_ascii=False)[:300])

with open(os.path.join(OUT, "rows-table.tsv"), "w", encoding="utf-8") as f:
    f.write("ordinal\tname\tpass_group\tfamily\tstage\tchanged\tstatus\n")
    for r in rows:
        f.write("%d\t%s\t%s\t%s\t%s\t%s\t%s\n" % (
            r["ordinal"], r["name"], r["pass_group"], r["family"], r["stage"],
            json.dumps(r.get("changed_by_current_candidate"), ensure_ascii=False)[:200], r.get("status")))
print("\nwrote rows-table.tsv")
