Used by the B1 card to (a) census every shared-memory access site of a module,
(b) resolve the index expressions symbolically, (c) diff two modules word by word
with an id-independent normal form.

No spirv-dis exists in this tree, so the module is decoded from the raw words.
"""

def read_words(path):
    raw = Path(path).read_bytes()
    n = len(raw) // 4
    return list(struct.unpack("<%dI" % n, raw[:n * 4])), raw


class Inst(object):
    __slots__ = ("idx", "woff", "op", "name", "a", "func")

    def __init__(self, idx, woff, op, a):
        self.idx, self.woff, self.op, self.a = idx, woff, op, a
        self.name = OP.get(op, "Op%d" % op)
        self.func = None

    @property
    def res(self):
        if self.name in RESULT_IS_FIRST:
            return self.a[0] if self.a else None
        if self.name in RESULT_IS_SECOND:
            return self.a[1] if len(self.a) > 1 else None
        return None

    @property
    def rtype(self):
        if self.name in RESULT_IS_FIRST:
            return None
        if self.name in RESULT_IS_SECOND:
            return self.a[0]
        return None

    def __repr__(self):
        return "<%s @%d/%d %s>" % (self.name, self.idx, self.woff, self.a[:6])


class Mod(object):
    def __init__(self, path):
        self.path = str(path)
        self.w, self.raw = read_words(path)
        assert self.w[0] == 0x07230203, "not SPIR-V: %s" % path
        self.magic, self.version, self.generator, self.bound, self.schema = self.w[:5]
        self.insts = []
        i, n = 5, len(self.w)
        while i < n:
            head = self.w[i]
            wc, op = head >> 16, head & 0xFFFF
            if wc == 0:
                break
            self.insts.append(Inst(len(self.insts), i, op, self.w[i + 1:i + wc]))
            i += wc
        self.trailing = n - i
        self._index()

    # ---------- indexing ----------
    def _index(self):
        self.byid = {}
        self.n_stores = {}
        for it in self.insts:
            if it.name == "OpStore":
                self.n_stores[it.a[0]] = self.n_stores.get(it.a[0], 0) + 1
        self.names = {}
        self.types = {}          # id -> (kind, operands)
        self.consts = {}         # id -> int value
        self.vars = []           # (id, storage, type id, inst)
        self.decor = {}          # id -> {dec: [operands]}
        self.member_decor = {}   # (type id, member) -> {dec: [operands]}
        self.entry = []
        self.exec_mode = []
        self.funcs = {}          # func id -> name
        for it in self.insts:
            if it.res is not None:
                self.byid[it.res] = it
        for it in self.insts:
            nm = it.name
            if nm == "OpName":
                s = b"".join(struct.pack("<I", x) for x in it.a[1:]).split(b"\0")[0]
                self.names[it.a[0]] = s.decode("latin1")
            elif nm == "OpTypeInt":
                self.types[it.a[0]] = ("int", it.a[1], it.a[2])
            elif nm == "OpTypeFloat":
                self.types[it.a[0]] = ("float", it.a[1])
            elif nm == "OpTypeVector":
                self.types[it.a[0]] = ("vec", it.a[1], it.a[2])
            elif nm == "OpTypeArray":
                self.types[it.a[0]] = ("array", it.a[1], it.a[2])
            elif nm == "OpTypeRuntimeArray":
                self.types[it.a[0]] = ("rarray", it.a[1])
            elif nm == "OpTypeStruct":
                self.types[it.a[0]] = ("struct", list(it.a[1:]))
            elif nm == "OpTypePointer":
                self.types[it.a[0]] = ("ptr", it.a[1], it.a[2])
            elif nm in ("OpConstant", "OpConstantTrue", "OpConstantFalse"):
                self.consts[it.a[1]] = it.a[2] if len(it.a) > 2 else (
                    1 if nm == "OpConstantTrue" else 0)
            elif nm == "OpConstantNull":
                self.consts[it.a[1]] = 0
            elif nm == "OpVariable":
                self.vars.append((it.a[1], it.a[2], it.a[0], it))
            elif nm == "OpDecorate":
                self.decor.setdefault(it.a[0], {}).setdefault(it.a[1], []).append(it.a[2:])
            elif nm == "OpMemberDecorate":
                self.member_decor.setdefault((it.a[0], it.a[1]), {}).setdefault(
                    it.a[2], []).append(it.a[3:])
            elif nm == "OpEntryPoint":
                self.entry.append(it.a)
            elif nm == "OpExecutionMode":
                self.exec_mode.append(it.a)
            elif nm == "OpFunction":
                self.funcs[it.a[1]] = self.names.get(it.a[1], "f%d" % it.a[1])
        # function containment
        cur = None
        for it in self.insts:
            if it.name == "OpFunction":
                cur = it.a[1]
            it.func = cur
            if it.name == "OpFunctionEnd":
                cur = None

    # ---------- sizes ----------
    def size(self, tid, depth=0):
        if depth > 8 or tid not in self.types:
            return 0
        t = self.types[tid]
        if t[0] == "int":
            return t[1] // 8
        if t[0] == "float":
            return t[1] // 8
        if t[0] == "vec":
            return self.size(t[1], depth + 1) * t[2]
        if t[0] == "array":
            cnt = self.consts.get(t[2], 0)
            stride = None
            dec = self.decor.get(tid, {})
            if 4 in dec:
                stride = dec[4][0][0]
            cover = self.size(t[1], depth + 1)
            return max((stride or cover) * cnt, cover * cnt)
        if t[0] == "rarray":
            dec = self.decor.get(tid, {})
            return dec[4][0][0] if 4 in dec else self.size(t[1], depth + 1)
        if t[0] == "struct":
            total = 0
            for m, mt in enumerate(t[1]):
                md = self.member_decor.get((tid, m), {})
                off = md[6][0][0] if 6 in md else total
                total = max(total, off + self.size(mt, depth + 1))
            return total
        if t[0] == "ptr":
            return self.size(t[2], depth + 1)
        return 0

    def var_name(self, vid):
        return self.names.get(vid, "#%d" % vid)

    def wg_vars(self):
        out = []
        for vid, sc, tid, it in self.vars:
            if STORAGE.get(sc) == "Workgroup":
                d = self.decor.get(vid, {})
                out.append({"id": vid, "name": self.var_name(vid),
                            "bytes": self.size(tid),
                            "elem_bytes": self.size(self.types[tid][1]) if
                            self.types.get(tid, ("?",))[0] == "array" else self.size(tid),
                            "count": self.consts.get(self.types[tid][2]) if
                            self.types.get(tid, ("?",))[0] == "array" else None,
                            "note": "woff=%d" % it.woff})
        return out

    # ---------- id resolution ----------
    def last_store_before(self, vid, idx):
        for it in reversed(self.insts[:idx]):
            if it.name == "OpStore" and it.a[0] == vid:
                return it.a[1]
        return None

    def resolve(self, id_, idx, depth=0, seen=None):
        """symbolic (str, const-or-None) for an id at instruction position idx"""
        if depth > 24:
            return ("<deep>", None)
        if id_ in self.consts:
            return (str(self.consts[id_]), self.consts[id_])
        it = self.byid.get(id_)
        if it is None:
            nm = self.names.get(id_, "")
            return (nm or "?", None)
        nm = it.name
        if nm in ("OpCopyObject", "OpBitcast"):
            return self.resolve(it.a[2], idx, depth + 1)
        if nm in ("OpAccessChain", "OpInBoundsAccessChain"):
            base = it.a[2]
            subs = [self.resolve(x, idx, depth + 1)[0] for x in it.a[3:]]
            nmv = self.names.get(base)
            if nmv:
                return ("%s%s" % (nmv, "".join("[%s]" % s for s in subs)), None)
            return (self.resolve(base, idx, depth + 1)[0] +
                    "".join("[%s]" % s for s in subs), None)
        if nm == "OpLoad":
            ptr = it.a[2]
            if self.n_stores.get(ptr, 0) > 1:
                # loop-carried / reassigned local: do NOT fold a stale value.
                # The id itself is NOT part of the answer: ids are renumbered by
                # glslang whenever a variable disappears, so printing one would fake
                # a diff between two otherwise identical modules.
                return ("%s<var>" % (self.names.get(ptr) or "?"), None)
            src = self.last_store_before(ptr, it.idx)
            if src is not None:
                return self.resolve(src, it.idx, depth + 1)
            if ptr in self.byid:
                s, v = self.resolve(ptr, idx, depth + 1)
                return (s, v)
            return (self.names.get(ptr) or "?", None)
        if nm in ("OpIAdd", "OpISub", "OpIMul", "OpUDiv", "OpSDiv", "OpShiftLeftLogical",
                  "OpShiftRightLogical", "OpBitwiseOr", "OpBitwiseAnd", "OpBitwiseXor",
                  "OpUMod"):
            ls, lv = self.resolve(it.a[2], idx, depth + 1)
            rs, rv = self.resolve(it.a[3], idx, depth + 1)
            sym = {"OpIAdd": "+", "OpISub": "-", "OpIMul": "*", "OpUDiv": "/",
                   "OpSDiv": "/", "OpShiftLeftLogical": "<<", "OpShiftRightLogical": ">>",
                   "OpBitwiseOr": "|", "OpBitwiseAnd": "&", "OpBitwiseXor": "^",
                   "OpUMod": "%"}[nm]
            val = None
            if lv is not None and rv is not None:
                try:
                    val = {"+": lambda a, b: a + b, "-": lambda a, b: a - b,
                           "*": lambda a, b: a * b, "/": lambda a, b: a // b,
                           "<<": lambda a, b: a << b, ">>": lambda a, b: a >> b,
                           "|": lambda a, b: a | b, "&": lambda a, b: a & b,
                           "^": lambda a, b: a ^ b, "%": lambda a, b: a % b}[sym](lv, rv)
                except Exception:
                    val = None
            if val is not None:
                return (str(val), val)
            return ("(%s%s%s)" % (ls, sym, rs), None)
        if nm in ("OpCompositeExtract",):
            base = it.a[2]
            lit = it.a[3:]
            return (self.resolve(base, idx, depth + 1)[0] + "".join("[%d]" % x for x in lit),
                    None)
        if nm == "OpVariable":
            return (self.names.get(it.a[1]) or "?", None)
        if nm in ("OpGroupNonUniformBroadcastFirst", "OpGroupNonUniformBallot"):
            return ("%s(%s)" % (nm.replace("OpGroupNonUniform", "sg"), 
                                self.resolve(it.a[2], idx, depth + 1)[0]), None)
        if nm in ("OpGroupNonUniformShuffle", "OpGroupNonUniformShuffleXor"):
            return ("%s(%s,%s)" % (nm.replace("OpGroupNonUniform", "sg"),
                                   self.resolve(it.a[2], idx, depth + 1)[0],
                                   self.resolve(it.a[3], idx, depth + 1)[0]), None)
        return (self.names.get(id_, "?"), None)

    # ---------- shared-memory access census ----------
    def base_var(self, ptr_id):
        """walk AccessChain/InBoundsAccessChain back to the root variable id"""
        seen = 0
        while seen < 8:
            it = self.byid.get(ptr_id)
            if it is None:
                return None
            if it.name in ("OpAccessChain", "OpInBoundsAccessChain"):
                ptr_id = it.a[2]
                seen += 1
                continue
            if it.name == "OpVariable":
                return it.a[1]
            return None
        return None

    def accesses(self, var_id):
        """every read/write site of a shared/workgroup variable, in program order"""
        out = []
        for it in self.insts:
            nm = it.name
            if nm == "OpStore":
                ptr = it.a[0]
                if self.base_var(ptr) == var_id:
                    chain = self.byid.get(ptr)
                    idx = [self.resolve(x, it.idx) for x in chain.a[3:]] if chain else []
                    out.append({"kind": "store", "idx": it.idx, "woff": it.woff,
                                "chain_type": chain.a[0] if chain else None,
                                "index": [s for s, _ in idx],
                                "index_const": [v for _, v in idx],
                                "sfid": ptr})
            elif nm in ("OpCooperativeMatrixLoadKHR",):
                ptr = it.a[2]
                if self.base_var(ptr) == var_id:
                    chain = self.byid.get(ptr)
                    idx = [self.resolve(x, it.idx) for x in chain.a[3:]] if chain else []
                    out.append({"kind": "cm_load", "idx": it.idx, "woff": it.woff,
                                "res": it.a[1], "rowtype": it.a[0],
                                "chain_type": chain.a[0] if chain else None,
                                "index": [s for s, _ in idx],
                                "index_const": [v for _, v in idx],
                                "sfid": ptr, "sfid_word": chain.a[3] if chain else None})
            elif nm in ("OpCooperativeMatrixStoreKHR",):
                ptr = it.a[0]
                if self.base_var(ptr) == var_id:
                    chain = self.byid.get(ptr)
                    idx = [self.resolve(x, it.idx) for x in chain.a[3:]] if chain else []
                    out.append({"kind": "cm_store", "idx": it.idx, "woff": it.woff,
                                "chain_type": chain.a[0] if chain else None,
                                "index": [s for s, _ in idx],
                                "index_const": [v for _, v in idx],
                                "sfid": ptr})
            elif nm == "OpLoad":
                ptr = it.a[2]
                if self.base_var(ptr) == var_id:
                    out.append({"kind": "load", "idx": it.idx, "woff": it.woff,
                                "res": it.a[1]})
            elif nm in ("OpCopyMemory",):
                for ptr in it.a[:2]:
                    if self.base_var(ptr) == var_id:
                        out.append({"kind": "copy", "idx": it.idx, "woff": it.woff})
            elif nm == "OpAtomicLoad" or nm.startswith("OpAtomic"):
                pass
        return out

    def call_sites(self, func_name):
        """every OpFunctionCall to func_name, with its arguments resolved"""
        fid = None
        for f, n in self.funcs.items():
            if n.split("(")[0] == func_name or n == func_name:
                fid = f
        if fid is None:
            return []
        out = []
        for it in self.insts:
            if it.name != "OpFunctionCall" or it.a[2] != fid:
                continue
            args = []
            for p in it.a[3:]:
                args.append(self.resolve(p, it.idx)[0])
            out.append({"inst_index": it.idx, "word_offset": it.woff,
                        "callee": self.funcs.get(fid), "args": args,
                        "res": it.a[1]})
        return out

    def barriers(self):
        out = []
        for it in self.insts:
            if it.name in ("OpControlBarrier", "OpMemoryBarrier"):
                tgt = self.resolve(it.a[0], it.idx)[0]
                if it.name == "OpControlBarrier":
                    out.append({"kind": it.name, "idx": it.idx, "woff": it.woff,
                                "exec_scope": tgt,
                                "mem_scope": self.resolve(it.a[1], it.idx)[0],
                                "semantics": self.resolve(it.a[2], it.idx)[0]})
                else:
                    out.append({"kind": it.name, "idx": it.idx, "woff": it.woff,
                                "mem_scope": tgt,
                                "semantics": self.resolve(it.a[1], it.idx)[0]})
        return out

    def cm_ops(self):
        out = []
        for it in self.insts:
            if it.name in ("OpCooperativeMatrixLoadKHR", "OpCooperativeMatrixStoreKHR",
                           "OpCooperativeMatrixMulAddKHR"):
                out.append({"kind": it.name, "idx": it.idx, "woff": it.woff,
                            "res": it.a[1] if it.name != "OpCooperativeMatrixStoreKHR"
                            else None})
        return out

    def histogram(self):
        h = {}
        for it in self.insts:
            h[it.name] = h.get(it.name, 0) + 1
        return dict(sorted(h.items()))

    def summary(self):
        return {"spv": self.path, "bytes": len(self.raw), "words": len(self.w),
                "bound": self.bound, "version": hex(self.version),
                "generator": hex(self.generator),
                "entry_points": [{"model": e[0], "name": self.names.get(e[1], "?")}
                                 for e in self.entry],
                "workgroup_vars": self.wg_vars(),
                "lds_bytes": sum(v["bytes"] for v in self.wg_vars()),
                "histogram": self.histogram(),
                "barriers": self.barriers()}
