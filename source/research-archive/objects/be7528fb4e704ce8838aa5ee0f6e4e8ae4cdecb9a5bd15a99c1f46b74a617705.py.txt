"""Predict the runner pipeline vector for the candidate plan (clock_runner.cpp:354 rule).

key = spv_path + ":" + npush + ":" + subgroup + ":" + <binding numbers in order>
pipeline index = first-appearance order of that key over the 232 PASS lines.
CPU only (plan parsing, no Vulkan).
"""
import json, sys
from pathlib import Path

C = Path(r'@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/fullnet-continuous-20261027')
sys.path.insert(0, str(C))
import COMMON as K


def parse(path):
    toks = Path(path).read_text(encoding='utf-8-sig').split('\n')
    passes = []
    i = 0
    while i < len(toks):
        line = toks[i].strip()
        if line.startswith('PASS '):
            parts = line.split()
            row = int(parts[1])
            name = parts[2].strip('"')
            spv = parts[3].strip('"')
            subgroup, shared, nbind, npush = map(int, parts[4:8])
            gx, gy, gz = map(int, parts[8:11])
            binds, push = [], []
            for j in range(nbind):
                i += 1
                b = toks[i].split()
                binds.append(int(b[1]))
            i += 1
            p = toks[i].split()
            assert p[0] == 'PUSH', p[:3]
            push = [int(x) for x in p[1:1 + npush]]
            passes.append(dict(row=row, name=name, spv=spv, subgroup=subgroup, shared=shared,
                               nbind=nbind, npush=npush, binds=binds, push=push,
                               gx=gx, gy=gy, gz=gz))
        i += 1
    return passes


def vector(passes):
    cache, vec = {}, []
    for p in passes:
        key = p['spv'] + ':' + str(p['npush']) + ':' + str(p['subgroup'])
        for b in p['binds']:
            key += ':' + str(b)
        if key not in cache:
            cache[key] = len(cache)
        vec.append(cache[key])
    return vec, len(cache)


if __name__ == '__main__':
    base, base_sha, _ = K.current_best()
    cand_plan = C / 'cards/n27-f07/plan.txt'
    pb, pc = parse(base), parse(cand_plan)
    assert len(pb) == len(pc) == 232
    vb, nb = vector(pb)
    vc, nc = vector(pc)
    diff = [(i, vb[i], vc[i], pb[i]['name']) for i in range(232) if vb[i] != vc[i]]
    changed = [i for i in range(232) if pb[i]['spv'] != pc[i]['spv']]
    fields = [(i, {k: (pb[i][k], pc[i][k]) for k in pb[i] if pb[i][k] != pc[i][k]})
              for i in range(232) if pb[i] != pc[i]]
    out = dict(baseline=dict(path=str(base), sha256=base_sha),
               candidate_plan=dict(path=str(cand_plan), sha256=K.sha(cand_plan)),
               passes=232, pipelines_before=nb, pipelines_after=nc,
               pass_lines_changed=changed, fields_changed=fields,
               pipeline_vector_equal=(vb == vc),
               pipeline_index_changes=diff[:8], pipeline_index_changes_count=len(diff),
               note='pipeline key is the SPV *path* (clock_runner.cpp:354). The old C256 path is '
                    'used by ordinal 59 only, and the candidate path is first seen at the same '
                    'ordinal, so the first-appearance index is unchanged: 40 pipelines and '
                    'ordinal 59 = pipeline 15 in both plans. Native pipeline mapping is stable.')
    K.write_json(C / 'cards/n27-f07/PIPELINE-PREDICTION.json', out)
    print(json.dumps({k: v for k, v in out.items() if k != 'fields_changed'}, indent=1))
    print('fields_changed:', fields)
    print('old pipeline of ord 59 =', vb[59], ' new =', vc[59])
