import re, collections
from pathlib import Path
F = Path(r"@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/fullnet-mechanism-transfer-20261026/f05-k-consumer-tile/c256/captures/native-f20-pixel-owner/ISA/CS_asmfd78c91bb8744f0f_simd32_000000000000000e_13_orig.asm")
lines = F.read_text(encoding="utf-8", errors="replace").splitlines()

def dsc(l):
    if "//" not in l:
        return None
    m = re.search(r"rd:\d+; ([a-z0-9_.]+)", l.split("//", 1)[1])
    return m.group(1) if m else None

fence = [i for i, l in enumerate(lines) if "fence.slm" in l]
dpas = [i for i, l in enumerate(lines) if re.search(r"\bdpas\b", l)]
print("fence lines:", fence, " dpas lines:", len(dpas))

for f_i in fence:
    before = [i for i in range(len(lines)) if i < f_i]
    def cnt(sel, a, b):
        return sum(1 for i in sel if a <= i < b)
    for lab, a, b in (("QKV (pre-fence)", 0, f_i), ("attention (post-fence)", f_i, len(lines))):
        g = collections.Counter(dsc(lines[i]) for i in range(a, b)
                                if re.search(r"\bsend\b", lines[i]))
        nd = cnt(dpas, a, b)
        print()
        print("== %s: lines %d..%d  dpas=%d ==" % (lab, a, b, nd))
        for k, v in g.most_common():
            print("   %5d %s" % (v, k))
