"""NoContraction-transfer probe for the R09 candidate (CPU only).

The deployed parent of this module family is *not* a plain compile: its build
ran a NoContraction decoration transfer from the old FP16 body module
(fused_c64_body_ds.spv) and that step left four stray NoContraction decorations
on non-float instructions.  The R09 candidate is a plain compile.  This probe
re-runs the identical transfer step on the candidate so the report can state,
with a file, whether the step changes any float-arithmetic decoration here.

Output: spv/r09_f3c64_body_ds_wo_i8.transfer.spv  (probe only, NOT deployed)
"""
import hashlib
import importlib.util
import json
import sys
from pathlib import Path

sys.dont_write_bytecode = True
HERE = Path(__file__).resolve().parent
CARD = HERE.parent
R = Path("@NR_WORKSPACE@")
SPVTOOLS = R / "build/amd-netorg-supply-layout-v5-workbuddy-20261016/spv_tools.py"
DONOR = R / "candidates/amd-netorg-acceleration-workbuddy-20261016/f3-c64-fused-half-patch-v1/fused_c64_body_ds.spv"
CAND = CARD / "spv/r09_f3c64_body_ds_wo_i8.spv"
OUT = CARD / "spv/r09_f3c64_body_ds_wo_i8.transfer.spv"


def sha(p):
    return hashlib.sha256(Path(p).read_bytes()).hexdigest()


def main():
    s = importlib.util.spec_from_file_location("sp", SPVTOOLS)
    sp = importlib.util.module_from_spec(s)
    s.loader.exec_module(sp)
    pm, cm = sp.Module(DONOR), sp.Module(CAND)
    keep, info = sp.transfer_decoration_pattern_aligned(
        pm, cm, sp.no_contraction_ids(pm), verify=True)
    sp.set_no_contraction(cm, set(keep) | set(sp.no_contraction_ids(cm)), OUT)
    rec = {"donor": {"path": DONOR.as_posix(), "sha256": sha(DONOR)},
           "candidate": {"path": CAND.as_posix(), "sha256": sha(CAND)},
           "probe_output": {"path": OUT.as_posix(), "bytes": OUT.stat().st_size,
                            "sha256": sha(OUT)},
           "mapping": info,
           "byte_identical_to_candidate": sha(OUT) == sha(CAND),
           "nc_before": len(sp.no_contraction_ids(cm)),
           "nc_after": len(sp.no_contraction_ids(sp.Module(OUT))),
           "note": "probe only; the deployed candidate is the plain compile"}
    (CARD / "evidence/NC-TRANSFER-PROBE.json").write_text(
        json.dumps(rec, indent=1, ensure_ascii=False) + "\n", encoding="utf-8")
    print(json.dumps({k: v for k, v in rec.items() if k != "mapping"}, indent=1))
    return 0


if __name__ == "__main__":
    sys.exit(main())
