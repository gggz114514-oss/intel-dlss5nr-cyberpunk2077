"""S01 final verification: pack identity claims, plan validity, SHA ledger."""
import hashlib
import json
import sys
from pathlib import Path

import numpy as np

sys.path.insert(0, r"@NR_WORKSPACE@/work/dsh-fullnet-static-20261029/tools")
from planlib import Plan, sha256_file  # noqa: E402

B = Path(r"@NR_WORKSPACE@")
D = B / "work/codex-astra-fp16-int8-audit-20261024/int8-quality-first-20261029/real-int8-24"
HERE = Path(__file__).resolve().parent
PACK = HERE / "PACK"

old8 = np.fromfile(D / "c32-equalized-b4/b4-weights.bin", dtype=np.int8)
new8 = np.fromfile(PACK / "c32_block4_xe_weights.i8.bin", dtype=np.int8)
oldsc = np.fromfile(D / "c32-equalized-b4/b4-scales.bin", dtype="<f4")
newsc = np.fromfile(PACK / "c32_block4_xe_scales.f32.bin", dtype="<f4")
print("weights sizes", old8.size, new8.size)
print("expand  0..16383 identical:", bool(np.array_equal(old8[:16384], new8[:16384])))
print("qkv/wo 32768..49151 identical:", bool(np.array_equal(old8[32768:], new8[32768:])))
print("contract 16384..32767 changed bytes:", int((old8[16384:32768] != new8[16384:32768]).sum()), "/", 16384)
print("fsc entries", oldsc.size, "->", newsc.size, "bytes", newsc.nbytes)
print("fsc[4..132] identical:", bool(np.array_equal(oldsc[4:132], newsc[4:132])))
print("fsc[164..324] identical:", bool(np.array_equal(oldsc[164:324], newsc[164:324])))
print("fsc[0] identical:", bool(oldsc[0] == newsc[0]))
we = new8[:16384].reshape(16, 32, 32)[:, :, :8]
print("expand row absmax unique:", sorted({int(v) for v in np.abs(we).max(axis=1).max(axis=1)}) or "n/a")
rowmax = np.abs(we.astype(np.int16)).max(axis=1).max(axis=1)   # per tile
print("per-tile expand absmax min/max:", int(rowmax.min()), int(rowmax.max()))
ct = new8[16384:32768].reshape(16, 32, 32)[:, :, :8].astype(np.int16)
cm = np.abs(ct).max(axis=2).max(axis=1)
print("per-tile contract absmax min/max:", int(cm.min()), int(cm.max()))
assert np.array_equal(old8[:16384], new8[:16384]) and np.array_equal(old8[32768:], new8[32768:])
assert np.array_equal(oldsc[4:132], newsc[4:132]) and np.array_equal(oldsc[164:324], newsc[164:324])

ledger = {}
for name in sorted(f.name for f in HERE.glob("candidate-*.txt")) + ["oracle-local-b4-debug.txt"]:
    p = HERE / name
    q = Plan(p)
    problems = q.validate()
    assert not problems, (name, problems)
    ledger[name] = dict(sha256=sha256_file(p), bytes=p.stat().st_size,
                        header=list(q.header), dispatches=len(q.passes),
                        block4_spv_sha256=sha256_file(q.dispatch(4).spv),
                        block4_scales_bytes=q.buffers[872].bytes,
                        block4_scales_sha256=sha256_file(q.buffers[872].file),
                        outputs=[list(o[:3]) for o in q.outputs])
    print(f"{name:28s} ok sha={ledger[name]['sha256'][:16]} sc={ledger[name]['block4_scales_bytes']}")

for f in sorted(PACK.glob("*.bin")):
    ledger[f"PACK/{f.name}"] = dict(sha256=sha256_file(f), bytes=f.stat().st_size)
for rel in ("SOURCE/c32_b4_xe.spv", "SOURCE/parent2.spv", "SOURCE/c32_body1_i8.comp",
            "SOURCE/c2_i8_regk32.glsl", "SOURCE/c32_i8_common.glsl",
            "SOURCE/register_handoff.glsl", "SOURCE/build_spv.py"):
    f = HERE / rel
    if f.is_file():
        ledger[rel] = dict(sha256=sha256_file(f), bytes=f.stat().st_size)

for doc in ("CONTRACT.json", "CALIBRATION.json", "QUALITY.json", "JOBS.json", "COST.txt",
            "STATUS.txt", "PLANS.json", "CPU/c32lib.py", "CPU/calibration_e_b4.json",
            "CPU/calibrate_e_b4.py", "CPU/check_align.py", "CPU/align_sweep.json",
            "CPU/pack_meta.json", "CPU/oracle_crossmatch.json", "CPU/real_input_audit.json",
            "CPU/real_calibration.json", "CPU/feed_regimes.json", "CPU/exit_distribution.json",
            "CPU/exit_error_shape.json", "CPU/exit_bias.json", "CPU/ablation_preregistration.json",
            "assemble_contract.py", "build_plans.py", "emit_docs.py", "update_docs.py",
            "update_docs2.py", "CPU/audit_real_x.py", "CPU/recalibrate_real_x.py",
            "CPU/build_ablation_packs.py", "CPU/parse_oracle.py", "CPU/oracle_columns.py"):
    f = HERE / doc
    if f.is_file():
        ledger[doc] = dict(sha256=sha256_file(f), bytes=f.stat().st_size)

(PACK / "SHA256.json").write_text(json.dumps(ledger, indent=2))
print("\nledger entries", len(ledger))
print(json.dumps({k: v["sha256"][:16] for k, v in ledger.items()}, indent=1))
