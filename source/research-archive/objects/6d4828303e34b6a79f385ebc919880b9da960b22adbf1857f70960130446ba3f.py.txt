import struct
p = r"@NR_WORKSPACE@/candidates/int8-coverage-v2-20261020/B/spv/c64_d1.spv"
data=open(p,'rb').read(); n=len(data)//4
w=struct.unpack("<%dI"%n, data[:n*4]); i=5
seq=[]
while i<n:
    wc=w[i]>>16; op=w[i]&0xFFFF
    if wc==0: break
    seq.append((op,list(w[i+1:i+wc])))
    i+=wc
idx=[k for k,(op,_) in enumerate(seq) if op==4457]
print("muladd idx", idx[:12], "total", len(seq))
for k in range(max(0,idx[0]-26), idx[0]+4):
    op,ops=seq[k]
    print(k, op, ops)
