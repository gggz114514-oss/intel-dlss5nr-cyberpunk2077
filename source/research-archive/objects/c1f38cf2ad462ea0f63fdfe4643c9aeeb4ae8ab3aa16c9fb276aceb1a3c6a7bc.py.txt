"""How much of the campaign's optimised shader set would transfer to the vk102-small network?

The small network (310.9.16) is a different plan: 26 rows, 118 PASS. Our 12 adoptions are compiled
SPIR-V modules keyed by shader identity, so they transfer wherever the small plan dispatches the
same baseline shader. This compares the two shader sets.
"""
import hashlib, json, sys
from collections import Counter
from pathlib import Path
sys.path.insert(0, str(Path(__file__).parent))
import COMMON as K

SMALL = Path('@NR_WORKSPACE@/work/vk102-small-310-9-16-20261019')
plan = SMALL / 'package' / 'manifest' / 'port_network_runtime.txt'
lines = plan.read_text(encoding='utf-8-sig', errors='replace').splitlines()
passes = [l for l in lines if l.startswith('PASS ')]
print('small plan: %d PASS lines, %d lines total' % (len(passes), len(lines)))


def toks(line):
    return line.split('"')


small_spv = {}
for l in passes:
    parts = l.split('"')
    if len(parts) >= 4:
        p = Path(parts[1].replace(chr(92)*2, chr(92)))
        small_spv.setdefault(p.as_posix(), 0)
        small_spv[p.as_posix()] += 1
print('small distinct spv paths: %d' % len(small_spv))
resolved = {}
for p in small_spv:
    f = Path(p)
    if f.exists():
        resolved[p] = hashlib.sha256(f.read_bytes()).hexdigest()
print('small spv files resolved on disk: %d' % len(resolved))

base_rows = K.rows()
base_sha = {}
for r in base_rows:
    base_sha.setdefault(r['spv_sha256'], []).append(r['family'])
opt = K.read_json(K.C / 'backend-merged' / 'MANIFEST.json')
opt_map = {}
for row in opt['rows']:
    opt_map.setdefault(row['ordinal'], row)

small_shas = Counter(resolved.values())
print('\nsmall shader shas (top):')
for sha, n in small_shas.most_common(12):
    fams = base_sha.get(sha, [])
    print('  %s  %2d PASS  %s' % (sha[:12], n, ('baseline shader known: ' + ','.join(sorted(set(fams)))) if fams else 'not in the 232-row baseline set'))
hit = sum(n for sha, n in small_shas.items() if sha in base_sha)
print('\nsmall PASS using a baseline shader we have: %d / %d' % (hit, len(passes)))
# of those baseline shaders, how many do we have an optimised version for?
adopted_shas = {r['spv_sha256'] for r in opt['rows']}
print('distinct baseline shaders we optimised: %d' % len(adopted_shas))
covered = sum(n for sha, n in small_shas.items() if sha in adopted_shas)
print('small PASS whose baseline shader is one we optimised: %d' % covered)
K.write_json(K.C / 'evidence' / 'SMALL-TRANSFER.json',
             dict(small_passes=len(passes), small_distinct=len(small_spv),
                  resolved=len(resolved), baseline_hit=hit, optimised_hit=covered,
                  shas={k: v for k, v in small_shas.items()}))
