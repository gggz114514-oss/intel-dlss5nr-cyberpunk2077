import json, hashlib, subprocess, sys
from pathlib import Path

C = Path('@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/fullnet-continuous-20261027')
W = Path('@NR_WORKSPACE@')
GLSLANG = W / 'src/project/tools/glslang-16.5.0/bin/glslang.exe'
CARD = C / 'cards/n03b-f04-c256'
BUILD = CARD / 'build'
BUILD.mkdir(parents=True, exist_ok=True)

src = W / 'work/codex-astra-fp16-int8-audit-20261024/fullnet-mechanism-transfer-20261026/backend-uniform-owner/d034-family/c256/candidate.comp'
target = W / 'work/codex-astra-fp16-int8-audit-20261024/fullnet-mechanism-transfer-20261026/backend-uniform-owner/d034-family/c256/candidate.spv'
pin = C / 'cards/idt-identity-recipe/evidence/headers/dc5f1634de7e'
k32 = W / 'work/dsh-fp16-same-family-20261023/build/rebuild/K32'

def sha(p):
    return hashlib.sha256(Path(p).read_bytes()).hexdigest()

print('pin header sha', sha(pin / 'nb05_swin_c256_common.glsl'))
print('k32 header sha', sha(k32 / 'nb05_swin_c256_common.glsl'))

for tag, incs in [('pin-only', [pin]), ('pin+k32', [pin, k32]), ('k32+pin', [k32, pin])]:
    out = BUILD / ('parent-%s.spv' % tag)
    argv = [str(GLSLANG), '-V', '--target-env', 'vulkan1.3']
    for d in incs:
        argv += ['-I' + str(d)]
    argv += [str(src), '-o', str(out)]
    r = subprocess.run(argv, capture_output=True, text=True, errors='replace')
    print(tag, 'rc', r.returncode, 'sha', sha(out) if out.exists() else None,
          'match', out.exists() and sha(out) == sha(target))
    if r.returncode != 0:
        print(r.stdout[-800:], r.stderr[-800:])
print('target sha', sha(target), 'bytes', target.stat().st_size)
