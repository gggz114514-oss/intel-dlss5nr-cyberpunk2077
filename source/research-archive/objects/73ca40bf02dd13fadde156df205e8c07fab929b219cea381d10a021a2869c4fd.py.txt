import json,struct,statistics
from pathlib import Path
D=Path(__file__).resolve().parent
out={'status':'ALL_FRAMES_RETAINED_NO_TRIMMING','old_observation':'r2 p-normalize WG(24,24) all 8 SG about 300880 ticks; retained; cause unknown','repeats':[]}
for rep in range(3):
 arms={}
 for name in ['control','p-normalize','clock-whole']:
  root=D/'run'/f'f20-tail-r{rep}-{name}';r=json.loads((root/'port_network_receipt.json').read_text(encoding='utf-8-sig'));files=list(root.glob('clock_frame_*.bin'));assert len(files)==10
  frames=[];allvalues=[]
  for p in sorted(files):
   data=p.read_bytes();v=struct.unpack('<%dI'%(len(data)//4),data);ticks=[]
   for y in range(8,97,16):
    for x in range(8,169,16):
     for sg in range(8):
      k=((y*169+x)*8+sg)*4;a,b,c,d=v[k:k+4]
      ticks.append(((c|(d<<32))-(a|(b<<32)))&((1<<64)-1))
   frames.append({'frame':p.name,'mean_ticks':statistics.mean(ticks),'median_ticks':statistics.median(ticks),'max_ticks':max(ticks)})
   allvalues+=ticks
  arms[name]={'target_median_ms':statistics.median(next(x['samples_ms'] for x in r['per_pass_gpu'] if x['index']==230)),'all_records_mean_ticks':statistics.mean(allvalues),'all_records_median_ticks':statistics.median(allvalues),'max_ticks':max(allvalues),'frames':frames}
 out['repeats'].append({'rep':rep,'arms':arms,'pnorm_over_whole':arms['p-normalize']['all_records_mean_ticks']/arms['clock-whole']['all_records_mean_ticks'],'perturbation':{n:arms[n]['target_median_ms']/arms['control']['target_median_ms']-1 for n in ['p-normalize','clock-whole']}})
(D/'MAIN-F20-TAIL-REVIEW.json').write_text(json.dumps(out,indent=2),encoding='utf-8')
for r in out['repeats']:print({'rep':r['rep'],'pnorm_over_whole':r['pnorm_over_whole'],'perturbation':r['perturbation'],'means':{n:a['all_records_mean_ticks'] for n,a in r['arms'].items()},'pnorm_max':r['arms']['p-normalize']['max_ticks']})
