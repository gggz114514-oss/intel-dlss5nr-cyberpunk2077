def edge_probe_numbers(probe: dict, ordinal: int) -> dict:
    for t in probe["targets"]:
        if t["ordinal"] == ordinal:
            return {"opfdiv_by_127": t["opfdiv_by_127"], "opfdiv_total": t["opfdiv_total"],
                    "fabs_ext_inst": t["fabs_ext_inst"], "has_127": t["has_127"],
                    "has_1_over_127": t["has_1_over_127"], "mma_int8": t["mma_int8"],
                    "any_scale_buffer_written": t["any_scale_buffer_written"]}
    return {}


def worst_class(row) -> str:
    cls = {e["scale_A_source"] for e in row["quant_edges"]}
    if not cls:
        return "NO_MATRIX"
    for k in ("DYNAMIC_RUNTIME", "UNKNOWN", "FP16", "STATIC_REFINED", "STATIC_COARSE"):
        if k in cls:
            return k
    return "UNKNOWN"


def status_for(row) -> tuple[str, str]:
    cls = {e["scale_A_source"] for e in row["quant_edges"]}
    if not row["quant_edges"]:
        return "CLOSED", "no INT8 matrix edge in this dispatch (blend/IO/quantiser-free kernel)"
    if "DYNAMIC_RUNTIME" in cls:
        return "OPEN", ("activation-side scale still measured in kernel (see "
                        "DYNAMIC-SCALE-REMAINDERS-V2.json)")
    if "UNKNOWN" in cls:
        return "OPEN", "activation-side scale provenance not proven (UNKNOWN)"
    return "CLOSED", "all matrix edges carry a static scale from a bound pack"

def main() -> None:
    # ---- integrity of the inputs
    assert sha_file(V1_ROWS) == V1_ROWS_SHA, "FULLNET-ROWS.json changed since v1 was documented"
    assert sha_file(V1_REMAINDERS) == V1_REMAINDERS_SHA, "v1 remainders changed"
    assert sha_file(V1_SUMMARY) == V1_SUMMARY_SHA, "v1 summary changed"
    v1 = json.load(open(V1_ROWS, encoding="utf-8"))
    v1_rem = json.load(open(V1_REMAINDERS, encoding="utf-8"))
    v1_sum = json.load(open(V1_SUMMARY, encoding="utf-8"))
    probe = json.load(open(PROBE, encoding="utf-8"))
    s12 = json.load(open(S12_ROWS, encoding="utf-8"))
    s03 = json.load(open(S03_RISKS, encoding="utf-8"))

    rows = v1["rows"]
    by_ord = {r["ordinal"]: r for r in rows}
    assert len(rows) == 288

    before = Counter(e["scale_A_source"] for r in rows for e in r["quant_edges"])
    assert dict(before) == EXPECTED_V1_CLASSES, before
    v1_unknown = {(r["ordinal"], i) for r in rows for i, e in enumerate(r["quant_edges"])
                  if e["scale_A_source"] == "UNKNOWN"}
    assert len(v1_unknown) == 36, len(v1_unknown)

    # the closure must be exactly the 36 UNKNOWN positions, by ordinal
    closure = {}
    for c in s12["ledger_unknown_closure"]:
        closure.setdefault(c["ordinal"], []).append(c)
    s12_ords = sorted({c["ordinal"] for c in s12["ledger_unknown_closure"]})
    assert set(s12_ords) == {o for o, _ in v1_unknown}, (
        sorted(set(s12_ords) - {o for o, _ in v1_unknown}),
        sorted({o for o, _ in v1_unknown} - set(s12_ords)))
    # every patched ordinal carries at least one probe record
    for o in P:
        assert edge_probe_numbers(probe, o), f"missing probe for ordinal {o}"
        assert probe_sha_ok(probe, o, by_ord[o]["spv"]["sha256"]), f"probe sha mismatch at {o}"

    # ---- apply
    applied = []
    for o, spec in P.items():
        r = by_ord[o]
        before_classes = [e["scale_A_source"] for e in r["quant_edges"]]
        for idx, (cls, conf, ev) in spec["edges"].items():
            e = r["quant_edges"][idx]
            assert e["scale_A_source"] == "UNKNOWN", (o, idx, e["scale_A_source"])
            old_ev = e["evidence"]
            e["scale_A_source"] = cls
            e["confidence"] = conf
            e["evidence"] = ev

def probe_sha_ok(probe: dict, ordinal: int, sha: str) -> bool:
    for t in probe["targets"]:
        if t["ordinal"] == ordinal:
            return t["sha256"] == sha
    return False
