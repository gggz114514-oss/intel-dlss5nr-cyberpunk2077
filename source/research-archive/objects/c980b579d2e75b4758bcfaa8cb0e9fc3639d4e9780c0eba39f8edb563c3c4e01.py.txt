"""n26-f08-order: calibrate SPIR-V opcode numbers against this workspace's
glslang 16.5.0 by compiling one tiny shader per operation and reporting which
opcode number showed up.  Removes any hand-typed opcode table from the
evidence chain.

CPU only, writes only into cards/n26-f08-order/_probe/.
"""
import json
import subprocess
import sys
from pathlib import Path

HERE = Path(__file__).resolve().parent
OUT = HERE / '_probe'
GLSLANG = r'@NR_WORKSPACE@\src\project\tools\glslang-16.5.0\bin\glslang.exe'

HEAD = '''#version 450
#extension GL_KHR_shader_subgroup_basic : require
layout(local_size_x=256) in;
layout(push_constant) uniform P { int a; } pc;
layout(set=0,binding=0) buffer O { uint o[]; };
'''

CASES = {
    'imul': '    o[0] = gl_WorkGroupID.x * 16u + 7u;\n',
    'udiv': '    o[0] = gl_WorkGroupID.x / 16u + 7u;\n',
    'umod': '    o[0] = (gl_WorkGroupID.x % 2u) + 7u;\n',
    'shl': '    o[0] = (gl_WorkGroupID.x << 4u) + 7u;\n',
    'and': '    o[0] = (gl_WorkGroupID.x & 3u) + 7u;\n',
    'iadd': '    o[0] = gl_WorkGroupID.x + gl_WorkGroupID.y + 7u;\n',
    'atomic': '    atomicAdd(o[0], 1u);\n',
    'barrier': '    barrier();\n    o[0] = 1u;\n',
    'subgroupbarrier': '    subgroupBarrier();\n    o[0] = 1u;\n',
    'sgid': '    o[0] = gl_SubgroupID + gl_SubgroupInvocationID;\n',
}


def words(p):
    import struct
    d = Path(p).read_bytes()
    return list(struct.unpack('<%dI' % (len(d) // 4), d))


def histogram(w):
    i, h = 5, {}
    while i < len(w):
        op, n = w[i] & 0xFFFF, w[i] >> 16
        h[op] = h.get(op, 0) + 1
        i += n
    return h


def main():
    OUT.mkdir(exist_ok=True)
    result = {}
    for name, body in CASES.items():
        src = OUT / ('%s.comp' % name)
        spv = OUT / ('%s.spv' % name)
        src.write_text(HEAD + 'void main(){\n' + body + '}\n', encoding='utf-8')
        r = subprocess.run([GLSLANG, '-V', '--target-env', 'vulkan1.3',
                            '-o', str(spv), str(src)],
                           capture_output=True, text=True)
        if not spv.exists():
            result[name] = {'error': r.stdout + r.stderr}
            continue
        h = histogram(words(spv))
        result[name] = {'opcodes': {str(k): v for k, v in sorted(h.items())}}
    (HERE / 'OPCODE-CALIBRATION.json').write_text(
        json.dumps(result, indent=1), encoding='utf-8')
    # print a compact view: which opcode appears in exactly the probe module
    base = histogram(words(OUT / 'iadd.spv'))
    for name, info in result.items():
        if 'opcodes' not in info:
            print(name, 'ERROR', info['error'][:200])
            continue
        h = {int(k): v for k, v in info['opcodes'].items()}
        print('%-16s' % name, sorted(h.items()))


if __name__ == '__main__':
    sys.exit(main())
