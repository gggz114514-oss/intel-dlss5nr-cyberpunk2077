import json, os, hashlib
import numpy as np

W = r"@NR_WORKSPACE@/work/dsh-fullnet-static-20261029"
Q = r"@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/int8-quality-first-20261029"
D = os.path.join(Q, "real-int8-24")
OUT = os.path.join(W, "C00-first-divergence/_work")

def load(p):
    return np.fromfile(p, dtype="<f2").astype(np.float32)

def sha(p):
    h = hashlib.sha256()
    with open(p, "rb") as f:
        for b in iter(lambda: f.read(1 << 20), b""):
            h.update(b)
    return h.hexdigest()

# boundary -> (channel count, int8 arm file, teacher dump file, chain label)
CASES = [
    ("block1_out (C32 b1 out)", 32, "local-resident-v2/local-b1/replayed-network.bin", "dump-fp16/dump_block1_out.bin", "corrected-INT8 chain dump arm"),
    ("block2_out (C32 b2 out)", 32, "local-resident-v2/local-b2/replayed-network.bin", "dump-fp16/dump_block2_out.bin", "corrected-INT8 chain dump arm"),
    ("block3_out (C32 b3 out)", 32, "local-resident-v2/local-b3/replayed-network.bin", "dump-fp16/dump_block3_out.bin", "corrected-INT8 chain dump arm"),
    ("skip4 (C32 b4 FFN out)", 32, "local-resident-v2/local-b4/replayed-network.bin", "dump-fp16/dump_skip4.bin", "corrected-INT8 chain dump arm"),
    ("block5_out (C64 b5 out)", 64, "family-isolation/run-block5/replayed-network.bin", "dump-fp16/dump_block5_out.bin", "corrected-INT8 chain dump arm"),
    ("block5_out (C64 b5 out) [cand-C32]", 64, "c64-family-static/local-b5/replayed-network.bin", "dump-fp16/dump_block5_out.bin", "candidate chain, old C32"),
    ("block6_out (C64 b6 out)", 64, "family-isolation/run-block6/replayed-network.bin", "dump-fp16/dump_block6_out.bin", "corrected-INT8 chain dump arm"),
    ("block6_out (C64 b6 out) [cand-C32]", 64, "c64-family-static/local-b6/replayed-network.bin", "dump-fp16/dump_block6_out.bin", "candidate chain, old C32"),
    ("block6_out teacher-Y (FP16 Y in)", 64, "c64-stage/teacher-Y/replayed-network.bin", "dump-fp16/dump_block6_out.bin", "SAME-INPUT isolation: Y from FP16 teacher"),
    ("block6_out teacher-YT (FP16 Y,T in)", 64, "c64-stage/teacher-YT/replayed-network.bin", "dump-fp16/dump_block6_out.bin", "SAME-INPUT isolation: Y,T from FP16 teacher"),
    ("block7_out (C64 b7 out)", 64, "family-isolation/run-block7/replayed-network.bin", "dump-fp16/dump_block7_out.bin", "corrected-INT8 chain dump arm"),
    ("block7_out (C64 b7 out) [cand-C32]", 64, "c64-family-static/local-b7/replayed-network.bin", "dump-fp16/dump_block7_out.bin", "candidate chain, old C32"),
    ("block15_out (C256 b15 out)", 256, "family-isolation/run-block15/replayed-network.bin", "dump-fp16/dump_block15_out.bin", "corrected-INT8 chain dump arm"),
    ("block31_out (ViT b31 out)", 256, "family-isolation/run-block31/replayed-network.bin", "dump-fp16/dump_block31_out.bin", "corrected-INT8 chain dump arm"),
    ("block38_out (ViT b38 out)", 256, "family-isolation/run-block38/replayed-network.bin", "dump-fp16/dump_block38_out.bin", "corrected-INT8 chain dump arm"),
]

BUCKETS = [(0.0, 50.0), (50.0, 90.0), (90.0, 99.0), (99.0, 99.9), (99.9, 100.01)]

results = []
for label, C, rel_i8, rel_fp, chain in CASES:
    pi, pf = os.path.join(D, rel_i8), os.path.join(D, rel_fp)
    if not (os.path.exists(pi) and os.path.exists(pf)):
        print("MISSING", label, os.path.exists(pi), os.path.exists(pf)); continue
    a = load(pi); b = load(pf)
    assert a.size == b.size, (label, a.size, b.size)
    assert a.size % (C * 16) == 0, (label, a.size, C)
    d = a - b
    n = a.size
    # per-(channel) stats under verified C32/C64 layout (nblk, C/16, 16-token, 16-chan)
    A = a.reshape(-1, C // 16, 16, 16)
    B = b.reshape(-1, C // 16, 16, 16)
    Dd = d.reshape(-1, C // 16, 16, 16)
    bias = Dd.mean(axis=(0, 2)).reshape(-1)            # C
    fp_abs = np.abs(B).mean(axis=(0, 2)).reshape(-1)   # mean |fp| per channel
    rel = np.abs(Dd).mean(axis=(0, 2)).reshape(-1) / np.maximum(fp_abs, 1e-12)
    rec = {
        "boundary": label, "chain": chain, "C": int(C), "n": int(n),
        "int8_file": rel_i8, "int8_sha256": sha(pi), "int8_bytes": os.path.getsize(pi),
        "fp16_file": rel_fp, "fp16_sha256": sha(pf), "fp16_bytes": os.path.getsize(pf),
        "global": {
            "mean_signed_diff": float(d.mean()),
            "mean_abs_diff": float(np.abs(d).mean()),
            "rmse": float(np.sqrt((d.astype(np.float64) ** 2).mean())),
            "rel_l2": float(np.linalg.norm(d) / np.linalg.norm(b)),
            "fp16_absmax": float(np.abs(b).max()),
            "int8_absmax": float(np.abs(a).max()),
            "fp16_absmax_ratio_int8_over_fp16": float(np.abs(a).max() / np.abs(b).max()),
            "fp16_std": float(b.std()),
            "int8_std": float(a.std()),
            "exact_equal_fraction": float((a == b).mean()),
            "frac_int8_beyond_fp16_absmax": float((np.abs(a) > np.abs(b).max()).mean()),
        },
        "channel": {
            "layout": "reshape(-1,C/16,16,16); channel=(g,k)=g*16+k; reduce axes (0,2)",
            "bias_signed": [round(float(x), 6) for x in bias],
            "mean_abs_fp16": [round(float(x), 5) for x in fp_abs],
            "rel_abs_err": [round(float(x), 5) for x in rel],
            "bias_max": float(bias.max()), "bias_min": float(bias.min()),
            "bias_argmax": int(np.argmax(bias)), "bias_argmin": int(np.argmin(bias)),
            "bias_mean": float(bias.mean()), "bias_l1": float(np.abs(bias).mean()),
            "n_channels_bias_pos": int((bias > 0).sum()), "n_channels_bias_neg": int((bias < 0).sum()),
            "rel_err_max": float(rel.max()), "rel_err_argmax": int(np.argmax(rel)),
            "bias_over_rel_corr": float(np.corrcoef(bias, rel)[0, 1]) if C > 2 else None,
        },
        "magnitude_buckets": [],
    }
    ab = np.abs(b).reshape(-1)
    qs = np.percentile(ab, [50, 90, 99, 99.9])
    for (lo, hi) in BUCKETS:
        if lo == 0.0:
            m = ab <= qs[0]
        elif hi >= 100:
            m = ab > qs[3]
        elif lo == 50.0:
            m = (ab > qs[0]) & (ab <= qs[1])
        elif lo == 90.0:
            m = (ab > qs[1]) & (ab <= qs[2])
        else:
            m = (ab > qs[2]) & (ab <= qs[3])
        if m.sum() == 0:
            continue
        rec["magnitude_buckets"].append({
            "fp16_abs_range": [float(lo), float(hi)] if hi < 100 else [float(lo), None],
            "pixel_fraction": float(m.mean()),
            "mean_signed_diff": float(d.reshape(-1)[m].mean()),
            "mean_abs_diff": float(np.abs(d.reshape(-1)[m]).mean()),
            "fp16_mean_signed": float(b.reshape(-1)[m].mean()),
            "int8_over_fp16_signed_ratio": float(a.reshape(-1)[m].mean() / b.reshape(-1)[m].mean()) if abs(b.reshape(-1)[m].mean()) > 1e-9 else None,
        })
    results.append(rec)
    g = rec["global"]; c = rec["channel"]
    print("\n%s  [%s]" % (label, chain))
    print("  n=%d C=%d relL2=%.4f meanD=%+.5f mean|D|=%.5f rmse=%.5f  fp_absmax=%.3f i8_absmax=%.3f (i8/fp=%.4f) std fp=%.4f i8=%.4f" % (
        n, C, g["rel_l2"], g["mean_signed_diff"], g["mean_abs_diff"], g["rmse"],
        g["fp16_absmax"], g["int8_absmax"], g["fp16_absmax_ratio_int8_over_fp16"], g["fp16_std"], g["int8_std"]))
    print("  channels: bias range [%+.4f .. %+.4f] mean %+.4f L1 %.4f pos/neg %d/%d  rel_err max %.4f@c%d" % (
        c["bias_min"], c["bias_max"], c["bias_mean"], c["bias_l1"], c["n_channels_bias_pos"], c["n_channels_bias_neg"],
        c["rel_err_max"], c["rel_err_argmax"]))
    print("  top-|bias| channels:", sorted(range(C), key=lambda i: -abs(bias[i]))[:8], [round(float(bias[i]), 4) for i in sorted(range(C), key=lambda i: -abs(bias[i]))[:8]])
    print("  magnitude buckets (by fp16 |x| percentile):")
    for bk in rec["magnitude_buckets"]:
        print("     %-14s frac=%.5f meanD=%+.5f mean|D|=%.5f fp_mean=%+.4f signed_ratio=%s" % (
            bk["fp16_abs_range"], bk["pixel_fraction"], bk["mean_signed_diff"], bk["mean_abs_diff"], bk["fp16_mean_signed"],
            ("%.4f" % bk["int8_over_fp16_signed_ratio"]) if bk["int8_over_fp16_signed_ratio"] is not None else "n/a"))

with open(os.path.join(W, "C00-first-divergence/_work/boundary-channel-stats.json"), "w", encoding="utf-8") as f:
    json.dump(results, f, ensure_ascii=False, indent=1)
print("\nwrote boundary-channel-stats.json  (%d cases)" % len(results))
