"""Corrected per-channel analysis in the TRUE channel coordinate system.

block*_out is the C64 canonical image layout (c64_img_half_index):
    idx = (tile*4 + c/16)*256 + slot*16 + (c%16)      [tile][g][slot][k]
A reshape(-1,64) therefore gives  col = (slot%4)*16 + k, i.e. the column index
mixes slot bits with channel bits -> an apparent period-16 pattern that is a
reshape artifact, not a channel property.  Here both tensors are permuted to
token-major [tile][slot][g*16+k] first.
"""
import json
import numpy as np
from pathlib import Path

B = Path("@NR_WORKSPACE@")
W = B / "work/dsh-fullnet-static-20261029"
CHAIN = W / "C00-color-baseline/runs/diag-chain62-66"
TEACH = (B / "work/codex-astra-fp16-int8-audit-20261024/int8-quality-first-20261029"
         / "real-int8-24/dump-fp16")


def load(p):
    return np.fromfile(p, dtype="<f2").astype(np.float64)


def to_token(a):
    return np.ascontiguousarray(a.reshape(-1, 4, 16, 16).transpose(0, 2, 1, 3)).reshape(-1, 64)


def main():
    out = {"layout_note": "canonical [tile][g=4][slot=16][k=16] -> token [tile][slot][g*16+k]"}
    for blk in (62, 63, 64, 65):
        g = to_token(load(CHAIN / f"chain-block{blk}_out.bin"))
        t = to_token(load(TEACH / f"dump_block{blk}_out.bin"))
        gr = np.sqrt((g * g).mean(0)); tr = np.sqrt((t * t).mean(0))
        ratio = gr / np.maximum(tr, 1e-30)
        slope = (g * t).sum(0) / np.maximum((t * t).sum(0), 1e-30)
        out[f"b{blk}"] = {
            "rms_gpu": float(np.sqrt((g * g).mean())), "rms_teach": float(np.sqrt((t * t).mean())),
            "ratio_global": float(np.sqrt((g * g).mean() / (t * t).mean())),
            "relL2": float(np.linalg.norm(g - t) / np.linalg.norm(t)),
            "corr": float(np.corrcoef(g.ravel(), t.ravel())[0, 1]),
            "slope_global": float((g * t).sum() / (t * t).sum()),
            "ch_ratio_64": [round(float(x), 4) for x in ratio],
            "ch_ratio_min": float(ratio.min()), "ch_ratio_max": float(ratio.max()),
            "ch_ratio_mean": float(ratio.mean()), "ch_ratio_std": float(ratio.std()),
            "ch_slope_64": [round(float(x), 4) for x in slope],
            "ch_slope_min": float(slope.min()), "ch_slope_max": float(slope.max()),
            "ch_slope_std": float(slope.std()),
            "ratio_16period_dev": float(np.abs(np.tile(ratio[:16], 4) - ratio).max()),
            "ratio_32period_dev": float(np.abs(np.tile(ratio[:32], 2) - ratio).max()),
        }
    print(json.dumps(out, indent=1))


if __name__ == "__main__":
    main()
