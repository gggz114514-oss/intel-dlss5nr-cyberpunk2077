# -*- coding: utf-8 -*-
"""Verify plane semantics on all nine files (ref + two arms x three frames)."""
import json
import sys
from pathlib import Path

import numpy as np

OUT = Path(__file__).resolve().parent.parent
B = Path(r"@NR_WORKSPACE@")
Q = B / "work/codex-astra-fp16-int8-audit-20261024/int8-quality-first-20261029"
W = B / "work/dsh-fullnet-static-20261029"
FRAMES = [(34189, "day-street-static-formal-01"),
          (41444, "day-street-motion-formal-01"),
          (43654, "day-street-motion-formal-02")]
FILES = {
    "ref": lambda f, s: Q / f"dataset/{s}/0-{f}/network-output.bin",
    "xe": lambda f, s: W / f"S02-f01/runs24/run-{f}/replayed-network.bin",
    "c64": lambda f, s: Q / f"real-int8-24/c64-family-static/run-{f}/replayed-network.bin",
}
out = []
for f, s in FRAMES:
    for arm, fn in FILES.items():
        a = np.fromfile(fn(f, s), dtype="<f4").reshape(2, 720, 1280, 4).astype(np.float64)
        p0, p1 = a[0, :, :, :3], a[1, :, :, :3]
        d = np.abs(p0 - np.clip(p1, 0.0, 1.0))
        rec = {"frame": f, "arm": arm,
               "plane0_equals_clip_plane1_max_abs": float(d.max()),
               "plane0_equals_clip_plane1_n_gt_1e-6": int((d > 1e-6).sum()),
               "plane0_alpha_all_one": bool((a[0, :, :, 3] == 1.0).all()),
               "plane1_rgb_min": float(p1.min()), "plane1_rgb_max": float(p1.max()),
               "plane1_alpha_min": float(a[1, :, :, 3].min()),
               "plane1_alpha_max": float(a[1, :, :, 3].max()),
               "plane1_alpha_mean": float(a[1, :, :, 3].mean()),
               "plane1_exceed_1": int((p1 > 1.0).sum()),
               "plane1_below_0": int((p1 < 0.0).sum())}
        out.append(rec)
        print(f"{f} {arm:3s}: |p0-clip(p1)|max={rec['plane0_equals_clip_plane1_max_abs']:.2e} "
              f"n>1e-6={rec['plane0_equals_clip_plane1_n_gt_1e-6']} "
              f"p0a=1:{rec['plane0_alpha_all_one']} p1 range=[{rec['plane1_rgb_min']:+.4f},"
              f"{rec['plane1_rgb_max']:+.4f}] a1 mean={rec['plane1_alpha_mean']:.4f} "
              f"exceed={rec['plane1_exceed_1']} below0={rec['plane1_below_0']}")
(OUT / "evidence/layout.json").write_text(json.dumps(out, indent=1), encoding="utf-8")
