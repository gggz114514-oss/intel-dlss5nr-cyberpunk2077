"""Build the three F05 candidate SPVs, validate them, and diff against the parents.

CPU only.  Writes build/<g>/candidate.spv and build/candidate-build.json.
"""
import json, sys, hashlib
from pathlib import Path

HERE = Path(__file__).parent
CAMP = HERE.parent.parent
sys.path.insert(0, str(CAMP))
import COMMON as K

BUILD = HERE / "build"
INC = {
    "c64": CAMP.parent.parent / "zcode-fp16-fullnet-handoff-20261024/candidates/f01-f02-f05-f21/f05-c64-join-t3-v1/source",
    "c128": CAMP.parent.parent / "zcode-fp16-fullnet-handoff-20261024/candidates/f01-f02-f05-f21/f05-c128-join-t3-v1/source",
    "c256": CAMP.parent.parent / "zcode-fp16-fullnet-handoff-20261024/candidates/f01-f02-f05-f21/f05-c256-join-t3-v1/source",
}
PARENT_SPV = {
    "c64": K.D / "f05-kv-consumer-layout/family/c64/candidate.spv",
    "c128": K.D / "f05-kv-consumer-layout/family/c128/candidate.spv",
    "c256": CAMP / "cards/n05a-f05-c256-mreuse/candidate.spv",
}
PARENT_SHA = {
    "c64": "0d83192b5368cc6074e00bbefed67ec71f8f540a9088d4d96752887390459641",
    "c128": "20f4074fc70c32c0e74f0ad000053492e5f5c420259da6044e7a00cfc3ee3f4f",
    "c256": "7a9576d15a18b4698b2e930c82b9658f2345e40479351e6ad0a65b034abc0af1",
}


def sha(p):
    return hashlib.sha256(Path(p).read_bytes()).hexdigest()


out = {}
for g in ("c64", "c128", "c256"):
    d = BUILD / g
    d.mkdir(exist_ok=True)
    src = HERE / ("candidate-%s.comp" % g)
    spv = d / "candidate.spv" if False else HERE / ("candidate-%s.spv" % g)
    K.compile_glsl(src, spv, [src.parent, INC[g]])
    K.validate_spv(spv)                       # raises on failure
    rec = dict(group=g, source=str(src), source_sha=sha(src),
               candidate_spv=str(spv), candidate_spv_sha=sha(spv), spv_bytes=Path(spv).stat().st_size,
               parent_spv=str(PARENT_SPV[g]), parent_spv_sha=sha(PARENT_SPV[g]),
               parent_expected_sha=PARENT_SHA[g], parent_byte_match=sha(PARENT_SPV[g]) == PARENT_SHA[g],
               spv_validated=True, include_dir=str(INC[g]))
    out[g] = rec
    print("%-5s %s -> %s  (%d B)  validate OK" % (g, PARENT_SHA[g][:16], rec["candidate_spv_sha"][:16],
                                                  rec["spv_bytes"]))

# SPIR-V level statistics + which functions changed
sys.path.insert(0, str(K.SPV_TOOLS))
import spv_tools_local as ST  # noqa: E402


def funcs(p):
    m = ST.Module(Path(p))
    return m


for g in ("c64", "c128", "c256"):
    a = Path(PARENT_SPV[g]); b = Path(HERE / ("candidate-%s.spv" % g))
    same = sha(a) == sha(b)
    print("%-5s parent %d B / candidate %d B  identical=%s" %
          (g, a.stat().st_size, b.stat().st_size, same))

(BUILD / "candidate-build.json").write_text(json.dumps(out, indent=1), encoding="utf-8")
print("\nwrote", BUILD / "candidate-build.json")
