import json, os, collections

W = r"@NR_WORKSPACE@/work/dsh-fullnet-static-20261029"
o = json.load(open(os.path.join(W, "ledger/FULLNET-ROWS-V2.json"), encoding="utf-8"))
rows = o["rows"]

# inspect binding roles / access values
roles = collections.Counter()
acc = collections.Counter()
for r in rows:
    for b in r.get("bindings") or []:
        roles[b.get("role")] += 1
        acc[b.get("spv_access")] += 1
print("roles:", dict(roles))
print("access:", dict(acc))
print("sample binding:", json.dumps(rows[5]["bindings"][0], ensure_ascii=False))
print("row keys sample ordinal5:", {k: rows[5][k] for k in ("ordinal", "name", "family", "stage", "status")})
