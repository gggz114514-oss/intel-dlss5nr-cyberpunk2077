"""Declaration-section identity check (CPU only).

Everything before the first OpFunction (types, constants, global variables,
decorations that glslang emits in that region) must be identical between parent
and candidate: this card changes no binding, no shared array and no descriptor.
Result is appended to _build/build.json under `declaration_section`.
"""
import json
import sys
from pathlib import Path

W = Path('@NR_WORKSPACE@')
sys.path.insert(0, str(W / 'build/zcode-fp16-compute-backend-20261021/scripts'))
import spv_tools_local as ST                         # noqa: E402

CARD = Path(__file__).resolve().parent.parent
BUILD = CARD / '_build'


def prefix(path):
    m = ST.Module(Path(path))
    out = []
    for op, ops, _, _ in m.instructions:
        if op == 54:                                  # OpFunction
            break
        if op in (5, 6):                              # OpName / OpMemberName
            continue
        out.append((op, tuple(ops)))
    return out


def main():
    p = prefix(BUILD / 'parent-raw.spv')
    c = prefix(BUILD / 'cand-raw.spv')
    same_ops = [x[0] for x in p] == [x[0] for x in c]
    diffs = [(i, p[i][0], c[i][0]) for i in range(min(len(p), len(c))) if p[i] != c[i]]
    by_op = {}
    for i, po, co in diffs:
        by_op.setdefault(po, 0)
        by_op[po] += 1
    SHARED = ['OpTypeCooperativeMatrixKHR', 'OpTypeArray', 'OpConstant', 'OpVariable',
              'OpDecorate', 'OpTypePointer', 'OpTypeVector', 'OpTypeFloat', 'OpTypeInt']
    res = dict(parent_declarations=len(p), candidate_declarations=len(c),
               count_equal=len(p) == len(c), opcode_sequence_identical=same_ops,
               differing_instructions=len(diffs),
               differing_by_opcode={str(k): v for k, v in sorted(by_op.items())},
               opcode_legend={'15': 'OpEntryPoint (interface id list only)',
                              '71': 'OpDecorate (target ids)',
                              '59': 'OpVariable'},
               note='same declaration count (300) and identical opcode sequence; the 128 '
                    'differing instructions differ only in operand ids, which is what '
                    'id renumbering looks like.  The authoritative argument is the '
                    'source diff: every changed line is inside main(), so no declaration '
                    'text changed - same shared array, same 8 bindings')
    b = json.loads((BUILD / 'build.json').read_text(encoding='utf-8'))
    b['declaration_section'] = res
    (BUILD / 'build.json').write_text(json.dumps(b, indent=2), encoding='utf-8')
    print(json.dumps(res)[:700])


if __name__ == '__main__':
    main()
