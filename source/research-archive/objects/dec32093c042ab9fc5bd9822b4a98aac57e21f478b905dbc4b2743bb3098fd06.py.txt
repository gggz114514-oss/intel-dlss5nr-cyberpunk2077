def P(p):
    return str(p).replace("\\", "/")


def dispatch_facts(plan_path, ordinal=4):
    p = Plan(plan_path)
    d = p.dispatch(ordinal)
    buf = {b.binding: p.buffers[b.buffer] for b in d.binds}
    push = [int(w) for w in d.push]
    sgn = lambda u: u - 2**32 if u >= 2**31 else u
    binds = []
    for b in d.binds:
        bb = p.buffers[b.buffer]
        binds.append(dict(binding=b.binding, buffer_index=b.buffer, name=bb.name,
                          buffer_bytes=bb.bytes, offset=b.offset, bytes=b.bytes,
                          file=P(bb.file) if bb.file != "-" else "-",
                          file_sha256=(sha256_file(bb.file) if bb.file not in ("-",) and Path(bb.file).is_file() else None)))
    return dict(
        plan=P(plan_path), plan_sha256=p.sha256(), ordinal=ordinal, name=d.name,
        group=d.group, spv=P(d.spv),
        spv_sha256=(sha256_file(d.spv) if Path(d.spv).is_file() else None),
        grid_workgroups=dict(x=int(d.tail_tokens[4]), y=int(d.tail_tokens[5]), z=int(d.tail_tokens[6])),
        local_size=128, push16=push[:16],
        push_semantics=dict(p0=["tiles_x", "shift_x(signed)", "debug_flag", "window_x_if_p3y"],
                            p1=["unused", "unused", "unused_fp16_pack", "0"],
                            p2=["rs_fp16_offset", "ars_fp16_offset", "fb_scalar_index", "qk_bias_fb_offset"],
                            p3=["window_y_if_p3y", "use_window_coords", "tiles_y", "shift_y(signed)"]),
        tiles_x=push[0], tiles_y=push[14], shift_x=sgn(push[1]), shift_y=sgn(push[15]),
        binds=binds)
