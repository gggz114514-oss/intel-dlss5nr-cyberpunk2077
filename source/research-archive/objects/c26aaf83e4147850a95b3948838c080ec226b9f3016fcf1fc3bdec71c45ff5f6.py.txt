"""Structural SPIR-V audit for a card: what actually differs between parent and candidate.

Usage: python verify_spv_diff.py <card-id> [--parent <spv>]

Reports instruction counts, opcode-stream equality, differing positions with their
opcodes/operands, NoContraction sets, and entry-point/execution-mode (grid-ish) data.
CPU only.
"""
import json, re, sys
from pathlib import Path
sys.path.insert(0, str(Path(__file__).parent))
import COMMON as K

NAME = {12: 'OpExtInst', 61: 'OpLoad', 62: 'OpStore', 65: 'OpAccessChain', 126: 'OpImageRead',
        128: 'OpIAdd', 130: 'OpIMul', 132: 'OpUDiv', 133: 'OpSDiv', 134: 'OpUMod', 135: 'OpSRem',
        137: 'OpShiftRightLogical', 138: 'OpShiftLeftLogical', 139: 'OpBitwiseOr', 140: 'OpBitwiseXor',
        141: 'OpBitwiseAnd', 155: 'OpFAdd', 157: 'OpFMul', 159: 'OpFSub', 199: 'OpSelect'}


def opname(op):
    return NAME.get(op, 'op%d' % op)


def card_spv(card):
    d = K.C / 'cards' / card
    b = K.read_json(d / 'BUILD.json')
    cand = d / 'candidate.spv'
    base_spv = {str(r['spv']).replace('/', '\\') for r in K.rows()}
    from audit_card import idents, norm
    parent = None
    for i in idents(b):
        if norm(i['path']) in base_spv:
            parent = Path(i['path'])
            break
    return d, parent, cand


def main():
    card = sys.argv[1]
    d, parent, cand = card_spv(card)
    if '--parent' in sys.argv:
        parent = Path(sys.argv[sys.argv.index('--parent') + 1])
    assert parent and Path(parent).exists(), 'parent not found'
    pm = K.ST.Module(Path(parent))
    cm = K.ST.Module(cand)
    out = dict(card=card, parent=dict(path=str(parent), sha256=K.sha(parent)),
               candidate=dict(path=str(cand), sha256=K.sha(cand)),
               parent_instructions=len(pm.instructions), candidate_instructions=len(cm.instructions),
               parent_words=len(pm.words), candidate_words=len(cm.words))
    same_len = len(pm.instructions) == len(cm.instructions)
    out['same_instruction_count'] = same_len
    diff = []
    if same_len:
        for i, (a, b) in enumerate(zip(pm.instructions, cm.instructions)):
            if a[0] != b[0] or a[1] != b[1]:
                diff.append(dict(index=i, parent=dict(op=opname(a[0]), operands=a[1][:8]),
                                 candidate=dict(op=opname(b[0]), operands=b[1][:8])))
    out['differing_instructions'] = len(diff)
    out['differences'] = diff[:40]
    out['opcode_histogram_delta'] = {}
    ha, hb = {}, {}
    for a in pm.instructions:
        ha[a[0]] = ha.get(a[0], 0) + 1
    for b in cm.instructions:
        hb[b[0]] = hb.get(b[0], 0) + 1
    for k in set(ha) | set(hb):
        if ha.get(k) != hb.get(k):
            out['opcode_histogram_delta'][opname(k)] = [ha.get(k, 0), hb.get(k, 0)]
    out['no_contraction'] = dict(parent=len(K.ST.no_contraction_ids(pm)),
                                 candidate=len(K.ST.no_contraction_ids(cm)))
    # float sites comparison on the largest function
    try:
        pr = K.ST.function_regions(pm)
        cr = K.ST.function_regions(cm)
        out['functions'] = [len(pr), len(cr)]
    except Exception:
        pass
    K.write_json(d / 'SPV-DIFF-main.json', out)
    print(json.dumps({k: v for k, v in out.items() if k != 'differences'}, indent=2, default=str)[:4000])


if __name__ == '__main__':
    main()
