from pathlib import Path
import json,shlex
import numpy as np
Q=Path(__file__).resolve().parent;D=Q/'real-int8-24';O=D/'c32-minmax';O.mkdir(exist_ok=False)
data=json.loads((Q/'ACTIVE-ROWS.json').read_text());buf=data['buffers'];changes={};report=[]
teachers=[D/'dump-fp16',D/'teacher-diagnostics/teacher-41444',D/'teacher-diagnostics/teacher-43654']
for block in [1,2,3,4,67,68,69]:
 r=next(r for r in data['rows'] if r['name']==f'block{block}_swin_body_1');bind={int(k):v for k,v in r['bindings'].items()};push=list(map(int,r['push']))
 wp=np.fromfile(buf[bind[0][0]]['file'],dtype='<f2').astype(np.float32);old=Path(buf[bind[6][0]]['file']);sc=np.fromfile(old,dtype='<f4').copy();before=sc[:4].copy()
 W=np.array([[wp[push[4]+(f//16)*512+((f%16)//8)*128+(k//16)*256+(f%8)*16+k%16] for k in range(32)] for f in range(128)],dtype=np.float32)
 xmax=0.;emax=0.;frames=[]
 for directory in teachers:
  path=directory/('dump_'+buf[bind[2][0]]['name']+'.bin')
  x=np.fromfile(path,dtype='<f2').reshape(-1,2,16,16).transpose(0,2,1,3).reshape(-1,32).astype(np.float32)
  xmax=max(xmax,float(abs(x).max()));localmax=0.
  for start in range(0,len(x),8192):
   e=(x[start:start+8192]@W.T).astype(np.float16).astype(np.float32);t=np.clip(e,-4,4)
   u=(-.055908203125*abs(t)+.447265625).astype(np.float16).astype(np.float32);v=(t*u+.89453125).astype(np.float16).astype(np.float32)
   act=(e*v).astype(np.float16).astype(np.float32);localmax=max(localmax,float(abs(act).max()))
  emax=max(emax,localmax);frames.append({'source':str(path),'xmax':float(abs(x).max()),'cpu_emax':localmax})
 sx=np.float32(xmax/127);se=np.float32(emax/127)
 sc[4:132]*=sx/sc[0];sc[132:164]*=se/sc[2];sc[:4]=[sx,1/sx,se,1/se]
 assert np.array_equal(sc[164:],np.fromfile(old,dtype='<f4')[164:])
 new=O/f'b{block}-scales.bin';sc.tofile(new);changes[old.as_posix()]=new.as_posix()
 report.append({'block':block,'before':before.tolist(),'after':sc[:4].tolist(),'frames':frames})
jobs=[]
for j in json.loads((D/'JOBS.json').read_text()):
 if j['dump']:continue

 for old,new in changes.items():assert old in s;s=s.replace(old,new)
 p=O/f'candidate-{j["frame"]}.txt';p.write_text(s);jobs.append({**j,'plan':str(p),'out':str(O/f'run-{j["frame"]}'),'expected_output_sha256':None})
for j in json.loads((D/'local-resident-v2/JOBS.json').read_text()):
 if not j['sequence'].startswith('local-'):continue

 for old,new in changes.items():s=s.replace(old,new)
 p=O/f'{j["sequence"]}.txt';p.write_text(s);jobs.append({**j,'plan':str(p),'out':str(O/j['sequence'])})
(O/'JOBS.json').write_text(json.dumps(jobs,indent=2));(O/'CALIBRATION.json').write_text(json.dumps({'scope':'three representative daytime frames; all 24 are calibration evaluation, not holdout','method':'max absolute teacher X and CPU half FFN activation, per block static sX/sE; folded dequant taps updated coherently; weights unchanged','limitation':'CPU activation range estimate is not bit-exact GPU activation oracle','rows':report},indent=2));print('Prepared',len(jobs),'jobs')
