"""n39-f06-c256-slack: recompute the parent's ISA facts from the native capture that already
exists for this exact SPIR-V (881dff8c), so COST-LEDGER / NATIVE-PREDICTION cite numbers this
card derived itself instead of quoting another card.

Source: queue/n06c-f06-c256-sht/native/captures/native-native-base/ISA/
        CS_asm6b7176820f28c8ca_simd32_000000000000000f_14_orig.asm
(that card's base arm dispatched the same module, so the static listing is the parent's.)

Also counts the same facts for the accepted F05 C256 base (its A operand already comes from a
global tile) as the in-chain precedent: the message class the candidate is expected to produce.

Writes build/isa-parent.json.
"""
import collections
import hashlib
import json
import pathlib
import re
import sys

CARD = pathlib.Path(__file__).resolve().parent
C = pathlib.Path(r'@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024'
                 r'/fullnet-continuous-20261027')
FILES = {
    'parent_f06_c256': C / 'queue/n06c-f06-c256-sht/native/captures/native-native-base/ISA/'
                          'CS_asm6b7176820f28c8ca_simd32_000000000000000f_14_orig.asm',
    'precedent_f05_c256': C / 'queue/n05a-f05-c256-mreuse/native/captures/native-native-base/ISA/'
                              'CS_asmfd78c91bb8744f0f_simd32_000000000000000e_13_orig.asm',
}
INSN = re.compile(r'^/\* \[([0-9A-F]{4,})\]\s*\*/(.*)$')
MSG = re.compile(r'//\s*wr:([0-9]+\+[0-9]+),\s*rd:([0-9]+);\s*(\S+)')
REG = re.compile(r'\br(\d+)(?:\.\d+)?\b')


def facts(path):
    lines = pathlib.Path(path).read_text(encoding='utf-8', errors='replace').splitlines()
    msgs = collections.Counter()
    insns = 0
    dpas = 0
    syncs = 0
    max_r = -1
    first_dpas = None
    for line in lines:
        m = INSN.match(line)
        if not m:
            continue
        insns += 1
        body = m.group(2)
        if 'dpas' in body:
            dpas += 1
            if first_dpas is None:
                first_dpas = insns
        if re.search(r'\bsync\.', body):
            syncs += 1
        mm = MSG.search(line)
        if mm:
            msgs[mm.group(3)] += 1
        for r in REG.findall(body.split('//')[0]):
            max_r = max(max_r, int(r))
    return dict(path=str(path), sha256=hashlib.sha256(pathlib.Path(path).read_bytes()).hexdigest(),
                instructions=insns, dpas=dpas, syncs=syncs, max_r=max_r,
                first_dpas_instruction=first_dpas, sends=sum(msgs.values()),
                messages={k: v for k, v in msgs.most_common()})


def main():
    out = {}
    for key, p in FILES.items():
        out[key] = facts(p)
        f = out[key]
        print('%-20s insns=%4d dpas=%3d syncs=%4d max_r=%3d first_dpas=%4d sends=%4d'
              % (key, f['instructions'], f['dpas'], f['syncs'], f['max_r'],
                 f['first_dpas_instruction'] or -1, f['sends']))
        for k, v in list(f['messages'].items())[:8]:
            print('    %4d  %s' % (v, k))
    # the classes this card moves
    p = out['parent_f06_c256']['messages']
    out['delta_prediction_basis'] = dict(
        parent=dict(slm_a_loads=p.get('load.slm.d32x8t.a32', 0),
                    ugm_seed_loads=p.get('load.ugm.d32x8t.a32.bss[a0.2]', 0),
                    ugm_weight_loads=p.get('load.ugm.d32x8.a32.bss[a0.0]', 0),
                    staging_global_loads=p.get('load.ugm.d16u32.a32.bss[a0.0]', 0),
                    staging_slm_stores=p.get('store.slm.d16u32.a32', 0),
                    slm_fence=p.get('fence.slm.none.group', 0),
                    total_sends=out['parent_f06_c256']['sends']),
        prediction='the 1024 load.slm.d32x8t become 1024 load.ugm.d32x8t (same message count and '
                   'shape as the F05 C256 precedent below), the 16 store.slm + 18 staging loads '
                   '+ 1 fence disappear',
        precedent_f05_c256=dict(ugm_a_loads=out['precedent_f05_c256']['messages'].get(
            'load.ugm.d32x8t.a32.bss[a0.0]', 0),
            slm_a_loads=out['precedent_f05_c256']['messages'].get('load.slm.d32x8t.a32', 0),
            note='the accepted F05 C256 kernel loads its DPAS A operand straight from a global '
                 'tile and has zero d32x8t SLM loads'))
    (CARD / 'build' / 'isa-parent.json').write_text(json.dumps(out, indent=2), encoding='utf-8')
    print('A-class criterion: if load.slm.d32x8t stays 1024 the mechanism did not land')
    return 0


if __name__ == '__main__':
    sys.exit(main())
