"""Aggregate the measured stall milliseconds per family, skipping counter artifacts.

An artifact is a pass where ALUWR, SENDWR and BARRIER all report ~the same value as GpuTime
(i.e. 100% each), which is impossible; block69 showed this earlier.
"""
import json, sys
from collections import defaultdict
from pathlib import Path
sys.path.insert(0, str(Path(__file__).parent))
import COMMON as K

rows = K.read_json(K.C / 'evidence' / 'STALL-TARGETS.json')['rows']
by_ord = {r['ordinal']: r for r in K.rows()}
name2fam = {}
for r in K.rows():
    name2fam.setdefault(r['name'], r['family'])
art, clean = [], []
for r in rows:
    pct = [r['aluwr'], r['sendwr'], r['barrier']]
    if sum(1 for v in pct if abs(v - r['gpu']) < 0.002) >= 2:
        art.append(r)
    else:
        clean.append(r)
print('artifacts (>=2 of 3 stalls equal to gpu time): %d passes, %.3f ms'
      % (len(art), sum(r['gpu'] for r in art)))
for r in art:
    print('   %-30s gpu %.3f' % (r['name'], r['gpu']))
fam = defaultdict(lambda: dict(n=0, gpu=0.0, aluwr=0.0, sendwr=0.0, barrier=0.0))
for r in clean:
    f = name2fam.get(r['name'], '?')
    e = fam[f]
    e['n'] += 1
    e['gpu'] += r['gpu']
    e['aluwr'] += r['aluwr']
    e['sendwr'] += r['sendwr']
    e['barrier'] += r['barrier']
print('\nfamilies by ALUWR ms (artifacts excluded):')
print('  %-6s %3s %8s %8s %8s %8s' % ('fam', 'n', 'gpu', 'aluwr', 'sendwr', 'barrier'))
for f, e in sorted(fam.items(), key=lambda kv: -kv[1]['aluwr']):
    print('  %-6s %3d %8.3f %8.3f %8.3f %8.3f' % (f, e['n'], e['gpu'], e['aluwr'], e['sendwr'], e['barrier']))
tg = sum(e['gpu'] for e in fam.values())
print('\nclean total: %.3f ms, aluwr %.3f, sendwr %.3f, barrier %.3f'
      % (tg, sum(e['aluwr'] for e in fam.values()), sum(e['sendwr'] for e in fam.values()),
         sum(e['barrier'] for e in fam.values())))
K.write_json(K.C / 'evidence' / 'STALL-BY-FAMILY.json',
             dict(artifacts=[r['name'] for r in art], families=fam, clean_total_ms=tg))
