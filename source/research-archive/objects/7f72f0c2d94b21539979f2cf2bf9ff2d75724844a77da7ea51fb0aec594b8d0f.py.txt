import struct, collections
p = r"@NR_WORKSPACE@/candidates/int8-coverage-v2-20261020/B/spv/c64_d1.spv"
data=open(p,'rb').read(); n=len(data)//4
w=struct.unpack("<%dI"%n, data[:n*4]); i=5
names={}; exts=[]; hist=collections.Counter()
while i<n:
    wc=w[i]>>16; op=w[i]&0xFFFF
    if wc==0: break
    ops=list(w[i+1:i+wc])
    hist[op]+=1
    if op==5:
        raw=struct.pack("<%dI"%len(ops[1:]), *ops[1:]).split(b"\x00")[0].decode("utf-8","replace")
        names[ops[0]]=raw
    if op==10:
        raw=struct.pack("<%dI"%len(ops), *ops).split(b"\x00")[0].decode("utf-8","replace")
        exts.append(raw)
    i+=wc
print("extensions", exts)
print("hist 4440-4470", {k:v for k,v in sorted(hist.items()) if 4440<=k<=4470})
for k in (319,48,46,23,50,452,30,361,368,342,353,88,22,375,381):
    print(k, repr(names.get(k)))
