"""Global dependency-depth census of the F20 ISA: for every instruction compute the longest
dependent producer path that ends at it, then report the deepest chains and where they live.

This is the evidence for task B of card n28: it locates where XVE_STALL_ALUWR can come from,
instead of assuming it is the fragment-load address chains.

Read-only.
"""
import sys, json, collections
from pathlib import Path

sys.stdout.reconfigure(encoding='utf-8')
sys.path.insert(0, str(Path(__file__).parent))
from addr_cones import ISA, parse, build_writers, slice_depth, cone, regs

ins = parse(ISA)
writers = build_writers(ins)
bars = [i for i, x in enumerate(ins) if x['op'] == 'sync.bar']


def region_of(i):
    n = 0
    for b in bars:
        if i > b:
            n += 1
    return n


SKIP = ('sync.bar', 'sync.nop', 'sync.allrd', 'sync.allwr', 'join', 'jmpi', 'goto',
        'nop', 'illegal', 'wait')

depths = []
for i, x in enumerate(ins):
    d = slice_depth(ins, writers, i)
    depths.append(d)

order = sorted(range(len(ins)), key=lambda i: -depths[i])
print('=== deepest dependent chains (instruction -> path length) ===')
seen_ops = collections.Counter()
shown = 0
for i in order:
    if shown >= 25:
        break
    x = ins[i]
    if x['op'] in SKIP:
        continue
    shown += 1
    print('  depth=%3d ln=%4d [%s] region=%2d %-12s %s' %
          (depths[i], x['ln'], hex(x['addr']), region_of(i), x['op'],
           (x['msg'] or ' '.join(x['srcs']))[:56]))
    seen_ops[x['op']] += 1

print('\n=== depth histogram by region (max depth per region) ===')
byreg = collections.defaultdict(list)
for i, d in enumerate(depths):
    byreg[region_of(i)].append(d)
for r in sorted(byreg):
    v = byreg[r]
    print('  region %2d  n=%4d  max=%3d  p99=%3d  mean=%.1f' %
          (r, len(v), max(v), sorted(v)[int(len(v) * 0.99) - 1], sum(v) / len(v)))

# opcode histogram of the deepest 200 instructions: tells which pipes the critical path uses
top = [i for i in order[:200]]
print('\nopcodes in the 200 deepest instructions:',
      dict(collections.Counter(ins[i]['op'] for i in top).most_common(12)))

Path('cards/n28-f20-barrier-alu/tools/depth-census.json').write_text(json.dumps(
    [dict(ln=ins[i]['ln'], addr=hex(ins[i]['addr']), region=region_of(i),
          op=ins[i]['op'], depth=depths[i]) for i in order[:400]], indent=2), encoding='utf-8')
