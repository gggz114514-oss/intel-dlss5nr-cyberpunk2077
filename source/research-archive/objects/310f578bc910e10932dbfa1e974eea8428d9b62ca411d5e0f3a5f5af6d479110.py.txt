# -*- coding: utf-8 -*-
"""Final ROI candidate measurement + motion-frame context check."""
import sys
from pathlib import Path

import numpy as np

sys.path.insert(0, str(Path(__file__).resolve().parent))
import pngio  # noqa: E402

Q = Path(r"@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/int8-quality-first-20261029")
OUT = Path(__file__).resolve().parent.parent
PNG = OUT / "PNG"
W_ = np.array([0.2126, 0.7152, 0.0722])
FR = {34189: "day-street-static-formal-01", 41444: "day-street-motion-formal-01",
      43654: "day-street-motion-formal-02"}

BOX = {
    "P1-sky-a": (638, 148, 704, 212),
    "P1-sky-b": (620, 150, 700, 215),
    "P1-sky-c": (645, 150, 700, 210),
    "P2-building": (706, 150, 762, 232),
    "P2-building-b": (700, 160, 760, 230),
    "P3-ncpd": (450, 252, 640, 308),
    "P3-ncpd-b": (455, 255, 620, 305),
    "P4-bluewall": (640, 368, 720, 430),
    "P4-bluewall-b": (650, 370, 760, 425),
    "P5-blueground": (100, 600, 300, 672),
    "P6-neutral-a": (1075, 185, 1140, 245),
    "P6-neutral-b": (1090, 192, 1130, 230),
    "P6-neutral-c": (1000, 215, 1060, 275),
}
for frame, seq in FR.items():
    ref = np.fromfile(Q / f"dataset/{seq}/0-{frame}/network-output.bin", dtype="<f4")
    ref = ref.reshape(2, 720, 1280, 4)[0, :, :, :3].astype(np.float64)
    img8 = np.rint(np.clip(ref, 0, 1) * 255).astype(np.uint8)
    pngio.write_rgb(PNG / f"fp16-{frame}-from-bin.png", img8)
    print("=" * 60)
    print("frame", frame, seq)
    for name, (x0, y0, x1, y1) in BOX.items():
        c = ref[y0:y1, x0:x1].reshape(-1, 3)
        m = c.mean(0)
        L = c @ W_
        print(f"  {name:16s} ({x0:4d},{y0:3d},{x1:4d},{y1:3d}) mean=({m[0]:.3f},{m[1]:.3f},{m[2]:.3f}) "
              f"L={L.mean():.3f} R-G={m[0]-m[1]:+.3f} B-G={m[2]-m[1]:+.3f} "
              f"fracL>0.7={float((L>0.7).mean()):.3f} fracL<0.25={float((L<0.25).mean()):.3f} "
              f"clip_any={float((c>=1.0).any(1).mean()):.4f}")
