"""Write a compact SPIR-V listing (index, opcode, operands) for eyeballing/proofs."""
import sys
from pathlib import Path
CARD = Path(__file__).resolve().parent.parent
sys.path.insert(0, str(CARD / 'scratch'))
import spvstat, struct

def const_map(m):
    """id -> ('const', value) for OpConstant / OpConstantNull etc."""
    out = {}
    for op, o in m.instr:
        if op == 43 and len(o) >= 3:
            out[o[1]] = struct.unpack('<I', struct.pack('<I', o[2]))[0]
    return out

def dump(path, out):
    m = spvstat.Mod(path)
    cm = const_map(m)
    lines = []
    for i, (op, o) in enumerate(m.instr):
        ops = []
        for x in o:
            if x in cm:
                ops.append('%d(=%d)' % (x, cm[x]))
            else:
                ops.append(str(x))
        lines.append('%4d %-34s %s' % (i, m.opname(op), ' '.join(ops)))
    Path(out).write_text('\n'.join(lines) + '\n', encoding='utf-8')
    return m

if __name__ == '__main__':
    dump(sys.argv[1], sys.argv[2])
    print('wrote', sys.argv[2])
