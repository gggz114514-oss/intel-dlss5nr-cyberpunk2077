import sys, json
sys.path.insert(0, r"@NR_WORKSPACE@/work/dsh-xmx-handoff-validation-20261024/candidates/b3-c32-seed/tools")
from b3_spv import Module
CARD=r"@NR_WORKSPACE@/work/dsh-xmx-handoff-validation-20261024/candidates/b3-c32-seed"
out={}
for tag,p in (("parent",CARD+"/spv/parent_repro.spv"),("child",CARD+"/spv/b3_c32_body1_i8_seed.spv")):
    M=Module(p); names={}
    for ins in M.insts:
        if ins.op==5: names[ins.ops[0]]=b"".join(x.to_bytes(4,"little") for x in ins.ops[1:]).split(b"\0")[0].decode()
    nc=[i for i in M.insts if i.op==71 and len(i.ops)>=2 and i.ops[1]==42]
    dec={}
    for i in M.insts:
        if i.op==71: dec[i.ops[1]]=dec.get(i.ops[1],0)+1
    out[tag]={"nocontraction_count":len(nc),"decorations":dec,
              "deq_add_fn":[k for k,v in names.items() if v.startswith("nr_i8_deq_add")][0]}
    fn=out[tag]["deq_add_fn"]
    body=[i for i in M.insts if i.idx>=0]
    # print the body of nr_i8_deq_add
    start=None
    for ins in M.insts:
        if ins.op==54 and ins.ops[1]==fn: start=ins.idx
        elif start is not None and ins.op==56 and ins.idx>start: end=ins.idx; break
    out[tag]["deq_add_body"]=[ "%s %s"%(M.insts[k].name," ".join(str(x) for x in M.insts[k].ops)) for k in range(start,end+1)]
print(json.dumps({k:{kk:vv for kk,vv in v.items() if kk!="deq_add_body"} for k,v in out.items()}, indent=1))
print("\nparent nr_i8_deq_add body (%d insts):"%len(out["parent"]["deq_add_body"]))
for l in out["parent"]["deq_add_body"]: print("  ",l)
