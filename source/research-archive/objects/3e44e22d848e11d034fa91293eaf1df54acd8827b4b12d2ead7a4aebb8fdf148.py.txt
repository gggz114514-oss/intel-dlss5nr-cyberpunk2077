"""S06: locate the largest block15 Y errors of the new arm (peak-risk audit)."""
import json
import sys
from pathlib import Path

import numpy as np

sys.path.insert(0, str(Path(__file__).parent))
import c256b15 as L
from importlib import util

# reuse the emitter's helpers by exec'ing the module source up to the arms
src = (L.W / "tools/08_emit.py").read_text(encoding="utf-8-sig").split("arms = {")[0]
ns = {"__name__": "emit", "__file__": str(L.W / "tools/08_emit.py")}
exec(compile(src, "08_emit_head", "exec"), ns)
run = ns["run"]
PACKS = ns["PACKS"]
raw_old = ns["raw_old"]
idx = ns["idx"]; valid = ns["valid"]; NW = ns["NW"]

name, td = "34189", L.D / "dump-fp16"
Xc = L.load_x_canonical(td / "dump_block14_out.bin")
Y = np.fromfile(td / "dump_swin_15_Y.bin", dtype="<f2").reshape(NW, 256).astype(np.float32)
Xw = np.zeros((NW, 256), dtype=np.float32)
Xw[valid] = Xc[idx[valid]]
ro = run(raw_old, Xw, False)
rn = run(PACKS["A_mse"]["raw"], Xw, True)
Yv = Y[valid]; yov = ro["y"][valid]; ynv = rn["y"][valid]
de = np.abs(ynv.astype(np.float64) - Yv.astype(np.float64))
do = np.abs(yov.astype(np.float64) - Yv.astype(np.float64))
order = np.argsort(de.ravel())[::-1][:12]
rows = []
for flat in order:
    t, c = divmod(int(flat), 256)
    tok = np.nonzero(valid)[0][t]
    rows.append(dict(token=int(tok), window=int(tok // 64), t_in_win=int(tok % 64),
                     channel=int(c),
                     Y=float(Yv[t, c]), y_new=float(ynv[t, c]), y_old=float(yov[t, c]),
                     err_new=float(de[t, c]), err_old=float(do[t, c]),
                     qx_new=int(rn["qx"][tok, c]), qm_new=int(rn["qm"][tok, c])))
out = dict(
    top_errors_new=rows,
    quantiles=dict(
        p9999_new=float(np.quantile(de, 0.9999)), p9999_old=float(np.quantile(do, 0.9999)),
        p99999_new=float(np.quantile(de, 0.99999)), p99999_old=float(np.quantile(do, 0.99999)),
        p999_new=float(np.quantile(de, 0.999)), p999_old=float(np.quantile(do, 0.999)),
        max_new=float(de.max()), max_old=float(do.max()),
        n_elements=int(de.size), n_gt_1_new=int((de > 1.0).sum()), n_gt_1_old=int((do > 1.0).sum()),
        n_gt_05_new=int((de > 0.5).sum()), n_gt_05_old=int((do > 0.5).sum())),
    note="Y absmax teacher = %.4f" % float(np.abs(Yv).max()))
(L.W / "evidence/09-peak.json").write_text(json.dumps(out, indent=1), encoding="utf-8")
print(json.dumps(out, indent=1)[:4000])
