"""Emit SOURCE-DIFF.patch (unified) and SPIR-V size statistics for the card."""
import sys, json, difflib
from pathlib import Path

C = Path(r'@NR_WORKSPACE@\work\codex-astra-fp16-int8-audit-20261024\fullnet-continuous-20261027')
CARD = C / 'cards' / 'n04b-f05-c128-feed'
sys.path.insert(0, str(C))
import COMMON as K

parent = Path(K.row_by_ordinal(19)['source'])
cand = CARD / 'candidate.comp'
a = parent.read_text(encoding='utf-8-sig').splitlines()
b = cand.read_text(encoding='utf-8-sig').splitlines()
diff = list(difflib.unified_diff(a, b, fromfile='parent/' + parent.name,
                                 tofile='candidate.comp', lineterm='', n=4))
(CARD / 'SOURCE-DIFF.patch').write_text('\n'.join(diff) + '\n', encoding='utf-8')

added = sum(1 for l in diff if l.startswith('+') and not l.startswith('+++'))
removed = sum(1 for l in diff if l.startswith('-') and not l.startswith('---'))


def count_instr(path):
    m = K.ST.Module(Path(path))
    n = 0
    for op, ops, idx, cnt in m.instructions:
        if op not in (0, 1, 2, 3, 4, 5, 6, 7, 8, 11, 14, 15, 16, 17, 71, 72, 73, 74, 75, 77, 248, 331, 332, 4422, 4423, 4424, 4425, 4456, 4457, 4458, 5632, 5633, 5634, 5635, 5636, 5637):
            n += 1
    return n


stats = dict(
    diff_added_lines=added, diff_removed_lines=removed,
    parent_source_lines=len(a), candidate_source_lines=len(b),
    parent_spv_bytes=Path(K.row_by_ordinal(19)['spv']).stat().st_size,
    candidate_spv_bytes=(CARD / 'candidate.spv').stat().st_size,
    parent_code_instructions=count_instr(K.row_by_ordinal(19)['spv']),
    candidate_code_instructions=count_instr(CARD / 'candidate.spv'),
    parent_code_insns=len(K.ST.code_instructions(K.ST.Module(Path(K.row_by_ordinal(19)['spv'])))),
    candidate_code_insns=len(K.ST.code_instructions(K.ST.Module(CARD / 'candidate.spv'))),
)
print(json.dumps(stats, indent=1))
(CARD / 'build' / 'spv-stats.json').write_text(json.dumps(stats, indent=1), encoding='utf-8')
print('\n'.join(diff))
