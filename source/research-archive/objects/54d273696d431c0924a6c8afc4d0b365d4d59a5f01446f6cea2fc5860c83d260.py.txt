"""Stage 2: coordinate search of the 16-channel lift diagonal D at the tuned X/E scales."""
import json
import numpy as np
import f01_eval as E
import f01_model as M

xs, ys = E.make_sample(64)
E.set_caches(xs, ys)
base = E.base_cfg()
cfg = dict(base); cfg["sx"] = 0.0135; cfg["se"] = 0.0052


def ev(d, c=None):
    c = dict(cfg); c["d"] = [float(v) for v in d]
    return E.run_cfg(xs, c)["pooled"]["relL2"]


d = np.array(base["d"], np.float32)
best = ev(d)
print("start", best, list(np.round(d, 5)))
hist = []
for it in range(3):
    improved = False
    for k in range(16):
        cur = d[k]
        for fac in [0.6, 0.75, 0.9, 1.0, 1.12, 1.35, 1.7]:
            d[k] = np.float32(cur * fac)
            r = ev(d)
            hist.append({"it": it, "k": k, "fac": fac, "d": float(d[k]), "relL2": r})
            if r < best - 1e-9:
                best = r; cur = d[k]; improved = True
        d[k] = cur
    # global scale
    for fac in [0.85, 0.95, 1.0, 1.06, 1.18]:
        d2 = (d * np.float32(fac)).astype(np.float32)
        r = ev(d2)
        hist.append({"it": it, "k": -1, "fac": fac, "relL2": r})
        if r < best - 1e-9:
            best = r; d = d2; improved = True
    print("iter", it, "best", best, list(np.round(d, 6)), flush=True)
    if not improved:
        break

print("FINAL", best, json.dumps([float(v) for v in d]))
open("search-d.json", "w", encoding="utf-8").write(json.dumps(
    {"best_relL2": best, "d": [float(v) for v in d], "sx": cfg["sx"], "se": cfg["se"],
     "sf": cfg["sf"], "history": hist, "sample_stride": 64}, indent=2))
