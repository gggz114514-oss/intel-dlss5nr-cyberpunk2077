"""Measured native base-vs-candidate accounting for ord 226 (pipeline 37).

Reads the two GTPin captures the main agent already produced under
queue/n19-f19-root/native/captures/ (no GPU work here, read-only) and writes
cards/n19-f19-root/ISA-ACCOUNT.json.
"""
import json
import re
import sys
from collections import Counter
from pathlib import Path

CAMP = Path('@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/fullnet-continuous-20261027')
CARD = CAMP / 'cards/n19-f19-root'
sys.path.insert(0, str(CAMP))
import COMMON as K  # noqa: E402
import ISA          # noqa: E402

Q = CAMP / 'queue/n19-f19-root/native/captures'
BASE = Q / 'native-native-base/ISA/CS_asmf3942341c6275bc3_simd32_0000000000000026_37_orig.asm'
CAND = Q / 'native-native-cand/ISA/CS_asmf3dc59a88b9855df_simd32_0000000000000026_37_orig.asm'
INVARIANT = {'dpas.8x8': 129, 'sync.bar': 3, 'math.rsqt': 16, 'math.inv': 8, 'mad': 176}


def prof(p):
    isa = ISA.parse(p)
    ops = Counter(i['op'] for i in isa)
    msgs = Counter()
    for i in isa:
        if i['op'].startswith('send') and i['comment']:
            m = re.search(r'rd:\d+;\s*(.*)$', i['comment'])
            msgs[(m.group(1) if m else i['comment']).strip()] += 1
    regs = set()
    for i in isa:
        for m in re.finditer(r'\br(\d+)\.', i['raw']):
            regs.add(int(m.group(1)))
    scratch = sum(1 for i in isa if 'scratch' in i['raw'])
    return dict(instrs=len(isa), ops=dict(ops), msgs=dict(msgs),
                max_grf=max(regs) if regs else None, distinct_grf=len(regs),
                scratch=scratch, path=str(p))


def main():
    b, c = prof(BASE), prof(CAND)
    opdelta = []
    for k in sorted(set(b['ops']) | set(c['ops'])):
        d = c['ops'].get(k, 0) - b['ops'].get(k, 0)
        if d:
            opdelta.append(dict(op=k, baseline=b['ops'].get(k, 0), candidate=c['ops'].get(k, 0), delta=d))
    msgdelta = []
    for k in sorted(set(b['msgs']) | set(c['msgs'])):
        d = c['msgs'].get(k, 0) - b['msgs'].get(k, 0)
        if d:
            msgdelta.append(dict(message=k, baseline=b['msgs'].get(k, 0), candidate=c['msgs'].get(k, 0), delta=d))
    inv = {k: dict(baseline=b['ops'].get(k, 0), candidate=c['ops'].get(k, 0),
                   expected=v, ok=c['ops'].get(k, 0) == v) for k, v in INVARIANT.items()}
    res = dict(
        note='measured from the captures the main agent produced under '
             'queue/n19-f19-root/native/captures (plan sha 6a665698… = this card plan)',
        base_capture=b['path'], cand_capture=c['path'],
        instrs=dict(baseline=b['instrs'], candidate=c['instrs'],
                    delta=c['instrs'] - b['instrs'],
                    pct=round(100.0 * (c['instrs'] - b['instrs']) / b['instrs'], 2)),
        gathers_d16u32=dict(baseline=sum(v for k, v in b['msgs'].items() if 'd16u32' in k and 'ugm' in k),
                            candidate=sum(v for k, v in c['msgs'].items() if 'd16u32' in k and 'ugm' in k)),
        send_ugm=dict(baseline=b['ops'].get('send.ugm', 0), candidate=c['ops'].get('send.ugm', 0)),
        send_slm=dict(baseline=b['ops'].get('send.slm', 0), candidate=c['ops'].get('send.slm', 0)),
        invariants=inv, invariants_all_ok=all(v['ok'] for v in inv.values()),
        scratch=dict(baseline=b['scratch'], candidate=c['scratch']),
        registers=dict(baseline=dict(max_grf=b['max_grf'], distinct=b['distinct_grf']),
                       candidate=dict(max_grf=c['max_grf'], distinct=c['distinct_grf'])),
        opcode_delta=opdelta, message_delta=msgdelta,
        side_effects=[x for x in opdelta if x['op'] in ('sync.nop', 'sel', 'and', 'sync.allwr',
                                                        'sync.allrd') and x['delta'] > 0],
    )
    K.write_json(CARD / 'ISA-ACCOUNT.json', res)
    print(json.dumps({k: v for k, v in res.items()
                      if k not in ('opcode_delta', 'message_delta')}, indent=2))
    print('--- opcode delta ---')
    for x in opdelta:
        print('  %+4d  %-12s %d -> %d' % (x['delta'], x['op'], x['baseline'], x['candidate']))
    print('--- message delta ---')
    for x in msgdelta:
        print('  %+4d  %s  %d -> %d' % (x['delta'], x['message'], x['baseline'], x['candidate']))


if __name__ == '__main__':
    main()
