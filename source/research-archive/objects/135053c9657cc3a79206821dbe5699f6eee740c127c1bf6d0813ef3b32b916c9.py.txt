from pathlib import Path
import numpy as np,json
D=Path(__file__).parent; P=D/'c32-scale-recalibration'; out={}
for name,dt in [('first_pool.bin','<f2'),('first_image.bin','<f2'),('frame_0.bin','<f4')]:
 a=np.fromfile(D/'run/fp16'/name,dtype=dt).astype(np.float64);b=np.fromfile(P/'run'/name,dtype=dt).astype(np.float64)
 if name=='frame_0.bin':a=a.reshape(2,720,1280,4)[0,:,:,:3].ravel();b=b.reshape(2,720,1280,4)[0,:,:,:3].ravel()
 e=b-a;mse=np.mean(e*e);out[name]={'fp16_absmax':float(abs(a).max()),'corrected_int8_absmax':float(abs(b).max()),'rel_l2':float(np.linalg.norm(e)/np.linalg.norm(a)),'rmse':float(np.sqrt(mse)),'max_abs_error':float(abs(e).max()),'correlation':float(np.corrcoef(a,b)[0,1]),'nonfinite':int((~np.isfinite(b)).sum())}
 if name=='frame_0.bin':out[name]['psnr_peak1_db']=float(-10*np.log10(mse))
(P/'COMPARISON.json').write_text(json.dumps(out,indent=2));print(json.dumps(out,indent=2))
