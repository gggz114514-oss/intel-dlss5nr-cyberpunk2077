#!/usr/bin/env python3
"""Probe the parent SPV: header, capabilities, coopmat types, opcode census."""
import sys

sys.path.insert(0, r"@NR_WORKSPACE@/work/dsh-xmx-handoff-validation-20261024/candidates/b3-c32-seed/tools")
from b3_spv import Module, OPNAME  # noqa: E402

P = r"@NR_WORKSPACE@/work/dsh-xmx-handoff-validation-20261024/build/b3-c32-seed/parent_repro.spv"
M = Module(P)
print("bytes=%d words=%d bound=%d version=0x%08x generator=0x%08x" %
      (M.nbytes, len(M.words), M.bound, M.version, M.generator))
print("extensions:", [i.ops[0].to_bytes(4, "little").split(b"\0")[0].decode()
                      for i in M.insts if i.op == 10])
print("ext inst imports:", [(i.ops[0], i.ops[1].to_bytes(4, "little").split(b"\0")[0].decode())
                            for i in M.insts if i.op == 11])

CAP = {0: "Matrix", 1: "Shader", 9: "ImageBasic", 11: "SampledBuffer",
       32: "ImageBuffer", 37: "ShaderNonUniform", 38: "RuntimeDescriptorArray",
       39: "StorageBuffer16BitAccess", 4431: "StorageImageReadWithoutFormat",
       4432: "StorageImageWriteWithoutFormat", 4427: "StorageBuffer8BitAccess",
       4423: "GroupNonUniform", 4424: "GroupNonUniformVote",
       4425: "GroupNonUniformArithmetic", 4426: "GroupNonUniformBallot",
       4428: "GroupNonUniformShuffle", 4429: "GroupNonUniformShuffleRelative",
       4430: "GroupNonUniformClustered", 4433: "GroupNonUniformQuad",
       5341: "ShaderLayer", 5342: "ShaderViewportIndex",
       6022: "CooperativeMatrixKHR", 6023: "Float16", 6024: "Int8",
       6019: "Float16ImageAMD", 5286: "Int8?", 6086: "Float16?",
       }
print("capabilities:", [(i.ops[0], CAP.get(i.ops[0], "?")) for i in M.insts if i.op == 17])
print("memory model:", [(i.ops[0], i.ops[1]) for i in M.insts if i.op == 14])
print("entry points:", [(i.ops[1].to_bytes(4, "little").split(b"\0")[0].decode(), i.ops[0], i.ops[2])
                         for i in M.insts if i.op == 15])
print("execution modes:", [(i.ops[0], i.ops[1], list(i.ops[2:])) for i in M.insts if i.op == 16])

# coopmat types
print("\n-- OpTypeCooperativeMatrixKHR (4456) --")
for ins in M.insts:
    if ins.op == 4456:
        print("  id=%d ops=%s" % (ins.rid, list(ins.ops)))
print("\n-- OpTypeFloat / OpTypeInt --")
for ins in M.insts:
    if ins.op in (21, 22):
        print("  %s id=%d ops=%s" % (OPNAME[ins.op], ins.rid, list(ins.ops)))

# census
h = {}
for ins in M.insts:
    k = (ins.op, ins.n)
    h[k] = h.get(k, 0) + 1
print("\n-- opcode/wordcount census --")
for (op, n), c in sorted(h.items()):
    print("  %-34s wc=%-3d x%d" % (OPNAME.get(op, "op%d" % op), n, c))
