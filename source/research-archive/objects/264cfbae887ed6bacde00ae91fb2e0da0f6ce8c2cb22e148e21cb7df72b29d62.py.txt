def sha(p):
    return hashlib.sha256(Path(p).read_bytes()).hexdigest()


def glslang(src, out):
    argv = GLSL + [str(src), '-o', str(out)]
    r = subprocess.run(argv, capture_output=True, text=True)
    if r.returncode != 0:
        raise RuntimeError('glslang failed: %s\n%s\n%s' % (argv, r.stdout, r.stderr))
    return dict(argv=argv, stderr=r.stderr.strip())

# ------------------------------------------------------------------ plan parse
def plan_passes(plan_path=None):
    """Record order in plan.txt is: PASS, BIND*, PUSH  (the PUSH follows its own pass)."""
    plan_path = Path(plan_path or K.BASELINE)
    out = {}
    pending = None
    for i, l in enumerate(plan_path.read_text(encoding='utf-8-sig').splitlines()):
        if l.startswith('PUSH '):
            if pending is not None and out[pending]['push'] is None:
                out[pending]['push'] = [int(x) for x in l.split()[1:]]
            continue
        if not l.startswith('PASS '):
            continue
        m = re.match(r'PASS\s+(\d+)\s+(?:"([^"]+)"|(\S+))\s+(?:"([^"]+)"|(\S+))\s+(.*)$', l)
        name = m.group(2) or m.group(3)
        nums = m.group(6).split()
        out[name] = dict(line=i + 1, pass_index=int(m.group(1)),
                         spv=(m.group(4) or m.group(5)), nums=nums,
                         local=int(nums[0]), shared=int(nums[1]), grid=[int(x) for x in nums[-3:]],
                         push=None)
        pending = name
    return out

def ds_algebra():
    """Exhaustive: parent per-element store address == candidate 8x16 block store
    address, and the DS output domain is covered exactly once."""
    P = plan_passes()['f3c256_body_ds']
    push = P['push']
    OW, OH, N, Kk = push[21], push[22], push[23], push[24]
    whalf, obase = push[25], push[26]
    valid_W, valid_H = push[28], push[29]
    gx, gy, gz = P['grid']
    assert (OW, OH, N, Kk, obase, valid_W, valid_H) == (44, 24, 512, 256, 0, 42, 24), (OW, OH, N, Kk, obase, valid_W, valid_H)

    def io_at0(px, py, c, W, C):
        t = ((py // 4) * (W // 4) + px // 4) * 16 + 4 * (py % 4) + px % 4
        return ((t // 16) * (C // 16) + c // 16) * 256 + (t % 16) * 16 + c % 16, t

    rows = []
    bad_addr = 0
    contig_bad = 0
    tbase_mod_bad = 0
    covered = Counter()
    masked = []
    unmasked = 0
    for wx in range(gx):
        for wy in range(gy):
            px0 = 4 * wx
            for sy in (0, 1):
                py0 = 4 * wy - 2 + 2 * sy
                half_ok = (px0 >= 0) and (px0 + 3 < OW) and (py0 >= 0) and (py0 + 1 < OH)
                if not half_ok:
                    masked.append([wx, wy, sy, px0, py0])
                    continue
                unmasked += 1
                cb = None
                # sg encodes (sy, cb): sy = sg & 1, cb = sg >> 1
                for cb in range(4):
                    base, tb = io_at0(px0, py0, (cb * 8) * 16, OW, N)
                    base += obase
                    if tb % 16 not in (0, 8):
                        tbase_mod_bad += 1
                    # parent path: per element (m,c) of the 8x16 fragment
                    for m in range(8):
                        px = px0 + (m & 3)
                        py = py0 + (m >> 2)
                        for cc in range(16):
                            for rnd in range(2):
                                for j in range(4):
                                    group = cb * 8 + rnd * 4 + j
                                    par, t_par = io_at0(px, py, group * 16 + cc, OW, N)
                                    cand, _ = io_at0(px0, py0, group * 16, OW, N)
                                    cand_off = cand + m * 16 + cc
                                    if (cand_off + obase) != (par + obase):
                                        bad_addr += 1
                                    # contiguity: the fragment row m must be t_base + m
                                    if t_par != tb + m:
                                        contig_bad += 1
                                    covered[(t_par, group * 16 + cc)] += 1
    T = (OH // 4) * (OW // 4) * 16
    missing = [t for t in range(T) for c in range(N) if (t, c) not in covered]
    dup = [k for k, v in covered.items() if v != 1]
    oob = [k for k in covered if k[0] >= T or k[1] >= N]
    return dict(
        tokens=T, channels=N, expected_elements=T * N,
        elements_written=len(covered),
        address_identity_mismatches=bad_addr,
        row_contiguity_mismatches=contig_bad,
        tbase_mod16_violations=tbase_mod_bad,
        duplicate_writes=len(dup), missing_writes=len(missing), out_of_range=len(oob),
        half_ok_true_sites=unmasked, half_ok_false_sites=len(masked),
        masked_sites=masked,
        mask_rule='half_ok = px0>=0 && px0+3<OW && py0>=0 && py0+1<OH (the parent nrf_write_group guard, unchanged)',
        tbase_rule='tbase = ((py0/4)*(OW/4)+px0/4)*16 + 4*(py0%4) + m ; tbase%16 in {0,8} because py0%4 in {0,2}',
        identity='nrf_ds_at(px0+m&3, py0+(m>>2), g*16+cc) == nrf_ds_at(px0,py0,g*16) + m*16 + cc for all m,cc',
        push_words=dict(OW=OW, OH=OH, N=N, K=Kk, whalf=whalf, obase=obase, valid_W=valid_W, valid_H=valid_H),
        grid=[gx, gy, gz],
    )


def arena_ledger():
    """Byte-exact live ranges of the workgroup arena of F07 C256 (and the F18 set)."""
    # F07 C256, ord 59: nrf_pair[2048] + nrf_mean[4096] + shT[16384] halves = 45056 B
    sg_span = 256          # halves per subgroup slot in nrf_pair
    frag = 128             # halves per staged 8x16 fragment
    K = 256
    pair = dict(name='nrf_pair', halves=8 * 256, bytes=8 * 256 * 2,
                phaseA=dict(desc='body per-j staging of the two narrowed output fragments',
                            writes='[sg*256 + acc*128, +128) for sg<8, acc<2 -> [0,2048)',
                            write_range=[0, 2048],
                            readers='the same subgroup\'s pooling block: base+(r0+k)*16+ch, k in {0,1,4,5}',
                            read_range=[0, 2047],
                            lifetime='per j iteration (8 times), inside nr_body_stage',
                            barrier='subgroupBarrier before and after'))
    pair['phaseB_parent_only'] = dict(
        desc='DS output staging: the completed fp32->fp16 accumulator',
        writes='[sg*256+128, sg*256+256) -> one 128-half slot per subgroup',
        write_range=[128, 2047],
        readers='nrf_write_group: nrf_pair[sg*256+128 + m*16 + c] (identity copy, no transpose)',
        read_range=[128, 2047],
        lifetime='main(), 8 times (2 rounds x 4 channel groups), removed by this card',
        barrier='publish subgroupBarrier + retire subgroupBarrier per fragment')
    mean = dict(name='nrf_mean', halves=16 * K, bytes=16 * K * 2,
                desc='pooled tile [16 slots][K], slot = 4*ly+lx',
                writes='[(4*ly+lx)*K + kk, +1) per lane, kk = (sg%8)*32 + acc*16 + ch -> [0,4096)',
                write_range=[0, 4096],
                readers='DS A operand: nrf_mean[rowbase*K + k, stride K] over all 8 rows x K',
                read_range=[0, 4096],
                lifetime='written progressively over j=0..7, read once after the WG barrier',
                barrier='barrier() after nr_body_stage (cross-subgroup RAW)')
    sht = dict(name='shT', halves=16384, bytes=16384 * 2,
               desc='body A operand [64 rows][256 K]',
               writes='[0,16384) once (256 lanes x 64 halves, stride 256)',
               write_range=[0, 16384],
               readers='coopMatLoad(a, shT, m0*256 + k0, stride 256) with m0 = 8j -> rows [8j,8j+8)',
               read_range=[0, 16384],
               lifetime='whole j loop; rows [0,8j) die after iteration j-1',
               barrier='barrier() after the fill')
    total = pair['bytes'] + mean['bytes'] + sht['bytes']
    # interleaving evidence for the card's first try (mutually exclusive co-location)
    coloc = []
    # 1) nrf_mean -> shT: first nrf_mean write happens inside j=0 (slots 8,9)
    j0_slots = [8, 9]
    j0_write = [j0_slots[0] * K, (j0_slots[-1] + 1) * K]
    live_shT_at_j0_write = [2048 * 1, 16384]      # rows 8..63 are still needed by j>=1
    coloc.append(dict(
        host='shT', guest='nrf_mean', verdict='IMPOSSIBLE without new sync',
        first_guest_write=j0_write,
        host_live_at_that_moment=live_shT_at_j0_write,
        overlap=[max(j0_write[0], live_shT_at_j0_write[0]), min(j0_write[1], live_shT_at_j0_write[1])],
        why='the body writes pooled slots during the same j loop that still has to read shT '
            'rows 8j..63; subgroups are not synchronized inside the loop, so row death is '
            'per-subgroup, not per-workgroup.  A WG-wide retirement proof needs one barrier '
            'per j (8 extra WG barriers per dispatch) - new synchronization, not this card.'))
    # 2) nrf_pair -> shT
    coloc.append(dict(
        host='shT', guest='nrf_pair', verdict='IMPOSSIBLE',
        first_guest_write=[0, 256], host_live_at_that_moment=[0, 16384],
        overlap=[0, 256],
        why='j=0 stages its two fragments at [sg*256, sg*256+256) = [0,2048), which is '
            'exactly shT rows 0..7 that j=0 itself is reading.'))
    # 3) nrf_mean -> nrf_pair
    coloc.append(dict(
        host='nrf_pair', guest='nrf_mean', verdict='IMPOSSIBLE (capacity + lifetime)',
        first_guest_write=[0, 512], host_live_at_that_moment=[0, 2048],
        overlap=[0, 512],
        why='nrf_mean is 4096 halves > nrf_pair 2048 halves, and both are live across the '
            'same j loop (pooling staging in, pooled tile out).'))
    return dict(declared_shared_bytes=total, plan_shared_bytes=45056,
                arenas=[pair, mean, sht], colocation_attempts=coloc,
                phases_note='the only true mutually exclusive pair in this shader is '
                            'nrf_pair phase A (body) vs phase B (DS output); they already share '
                            'one arena, and phase B is retired by this card instead of co-located')


def f18_arena_table():
    """The F18 nine shaders: what arena exists and whether it has a second phase."""
    passes = plan_passes()
    rows = []
    for n in (166, 167, 168, 192, 193, 194, 212, 213, 214):
        r = K.row_by_ordinal(n)
        m = ST.Module(Path(r['spv']))
        sb = m.storage_bytes().get('Workgroup') or dict(count=0, bytes=0, unknown=0)
        p = passes.get(r['name'], {})
        rows.append(dict(ordinal=n, family=r['family'], name=r['name'],
                         plan_shared=p.get('shared'), spv_workgroup_bytes=sb['bytes'],
                         grid=p.get('grid'), phases=_f18_phase(r['name'])))
    return rows


def _f18_phase(name):
    if name.startswith('c3x_pre'):
        return ('3 phases in one arena: (1) gather input tile -> (2) projection A read + '
                'narrowed acc store -> (3) blend scalar read.  Already fully aliased: '
                'declared = max(phase) = gather, and the acc store region [0,2048) lies '
                'inside the still-live A region, so the WAR barrier before the acc store '
                'is required and must NOT be deleted.')
    if name.startswith('c3x_body'):
        return ('2 phases in one arena: expand writes [0,4096) (C256) / [0,8192) (C128) / '
                '[0,8192) (C64), mid reads all of it.  Single RAW phase pair, no second '
                'arena to co-locate, no retired phase.')
    if name.startswith('block'):
        return ('1 phase: the whole A tile is copied in and read by every subgroup '
                '(C256 keeps 32768 B; C128/C64 adopted siblings read A straight from gM '
                'and declare 0).  No second phase inside the shader.')
    if name == 't3_b39_project_blend':
        return 'no workgroup arena (plan shared 0, SPV has no Workgroup variable)'
    return 'unknown'


def plan_diff(plan, base=None):
    """Changed lines of the candidate plan vs the baseline plan."""
    import difflib
    b = Path(base or K.BASELINE).read_text(encoding='utf-8-sig').splitlines()
    c = Path(plan).read_text(encoding='utf-8').splitlines()
    changed = [l for l in difflib.unified_diff(b, c, 'base', 'cand', n=0)
               if l.startswith(('+', '-')) and not l.startswith(('+++', '---'))]
    return dict(base_lines=len(b), candidate_lines=len(c), changed_lines=len(changed),
                changed_pass_lines=len([l for l in changed if l.startswith(('+PASS', '-PASS'))]),
                changed_non_pass_lines=[l[:120] for l in changed
                                        if not l.startswith(('+PASS', '-PASS'))],
                only_pass_lines_changed=all(l.startswith(('+PASS', '-PASS')) for l in changed))


def declared_bindings(m):
    """[(set, binding)] from OpDecorate Binding, sorted - the interface declaration."""
    out = []
    for op, ops, _, _ in m.instructions:
        if op == 71 and ops[1] == 33:
            out.append(ops[2])
    return sorted(out)


def wg_store_audit(path):
    """Which storage class does every store in the module land in?

    Walks each OpStore / OpCooperativeMatrixStoreKHR pointer back through
    OpAccessChain / OpInBoundsAccessChain to its root OpVariable and reports the
    storage class plus the interface binding.  This is the machine check behind
    the retirement claim: after this card no store of the DS section can reach a
    Workgroup pointer.
    """
    m = ST.Module(Path(path))
    SC = {0: 'UniformConstant', 1: 'Input', 2: 'Uniform', 3: 'Output', 4: 'Workgroup',
          5: 'CrossWorkgroup', 6: 'Private', 7: 'Function', 8: 'Generic', 9: 'PushConstant',
          10: 'AtomicCounter', 11: 'Image', 12: 'StorageBuffer'}
    dec = {}
    for op, ops, _, _ in m.instructions:
        if op == 71 and ops[1] == 33:                      # OpDecorate ... Binding
            dec.setdefault(ops[0], {})['binding'] = ops[2]

    def root(cid, depth=0):
        if depth > 6:
            return None
        op, ops = m.defining_instruction(cid)
        if op is None:
            return None
        if op == 59:                                       # OpVariable: [type, result, sc]
            return ops[1]
        if op in (65, 66):                                 # AccessChain: [type, result, base, ...]
            return root(ops[2], depth + 1)
        if op == 61:                                       # OpLoad: [type, result, ptr]
            return root(ops[2], depth + 1)
        if op == 55:                                       # OpFunctionParameter
            return cid
        return None

    def sc_of(var_id):
        op, ops = m.defining_instruction(var_id)
        if op == 55:
            return 'FunctionParameter', None
        if op != 59:
            return None
        return SC.get(ops[2], 'sc%d' % ops[2]), dec.get(ops[1])

    counts = Counter()
    for op, ops, _, _ in m.instructions:
        if op == 62:
            ptr = ops[0]
        elif op == 4458:
            ptr = ops[0]
        else:
            continue
        v = root(ptr)
        sc = sc_of(v) if v is not None else None
        counts[(('OpStore' if op == 62 else 'CoopMatStore'), sc[0] if sc else 'unresolved',
                sc[1].get('binding') if sc and sc[1] else None)] += 1
    return [dict(op=k[0], storage_class=k[1], binding=k[2], count=v)
            for k, v in sorted(counts.items(), key=lambda kv: (kv[0][0], str(kv[0][1]), str(kv[0][2])))]

def barrier_ledger():
    return [
        dict(site='nr_body_stage: after the shT fill', kind='workgroup barrier()',
             covers='shT RAW (fill -> every subgroup A read) and the debug copy',
             action='KEPT (identical in the candidate)'),
        dict(site='nr_body_stage: inside if(debug) after the debug shT copy', kind='workgroup barrier()',
             covers='debug copy visibility', action='KEPT'),
        dict(site='nr_body_stage: before the pooling read of nrf_pair', kind='subgroupBarrier()',
             covers='nrf_pair phase-A RAW: the two narrowed fragments staged by coopMatStore are '
                    'read by the same subgroup\'s lanes (lane/16 selects the fragment, lane%16 the '
                    'channel), so the hazard is intra-subgroup and a subgroup barrier suffices',
             action='KEPT'),
        dict(site='nr_body_stage: after the pooling read of nrf_pair', kind='subgroupBarrier()',
             covers='nrf_pair phase-A WAR: the next j iteration restages the same 256-half slot',
             action='KEPT'),
        dict(site='main: after nr_body_stage()', kind='workgroup barrier()',
             covers='nrf_mean RAW across subgroups (each subgroup writes its own 32-channel slice of '
                    'a slot; the DS A operand reads all 256 channels of a slot)',
             action='KEPT'),
        dict(site='main: DS section, per fragment (4 per round x 2 rounds)', kind='2 x subgroupBarrier()',
             covers='nrf_pair phase-B RAW (staging store -> scalar read) and WAR (read -> next '
                    'fragment\'s staging store of the same 128-half slot)',
             action='DELETED: phase B no longer writes or reads nrf_pair, so no hazard remains. '
                    'The reuser was the subgroup itself (slot sg*256+128 of its own phase-A '
                    'region), so this reuse never had a cross-owner exposure'),
    ]

        first_writer_after_retirement='none - the DS section writes only nrf_o (global) and reads '
                                      'only nrf_mean and nrf_w',
        proof='machine check (PROOF.store_storage_audit): OpCooperativeMatrixStoreKHR whose pointer '
              'resolves to an OpVariable(Workgroup) is 6 in the parent and 2 in the candidate.  '
              'The two survivors are the per-j pooling staging of the two narrowed fragments in '

        declared_shared_before=45056, declared_shared_after=45056,

def _tail_block():
    return dict(
        tail_domain='the DS output tile grid is 44x24 pixels = 11x6 4x4 tiles = 1056 tokens; '
                    'grid.y = 7 so the last row of workgroups (wy=6) writes py0=22 (sy=0) and '
                    'py0=24 (sy=1, rejected).  Every valid pixel is written exactly once (see '
                    'ds_output_address_identity.*), and the 2 masked (wx,wy,sy) row-halves per wx '
                    'are the parent\'s own half_ok rejects.',
        padding='tokens [1056, ...) of the io_at t space are never written in either arm; '
                'px = 42..43 columns are written in both arms (OW=44 > valid_W=42) exactly as the '
                'parent writes them.',
        identity='the candidate keeps the parent\'s half_ok predicate verbatim, so the write '
                 'domain is set-equal, not merely cardinality-equal.',
    )
