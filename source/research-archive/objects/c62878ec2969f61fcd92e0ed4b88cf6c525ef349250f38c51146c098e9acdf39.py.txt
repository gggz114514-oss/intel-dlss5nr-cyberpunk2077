"""N02: emit SOURCE-DIFF.patch (site parent source/header -> card candidate)."""
import difflib, pathlib

CARD = pathlib.Path(r'@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/fullnet-continuous-20261027/cards/n02-f20-f21-tail')
SITE = pathlib.Path(r'@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/c32-boundary-candidates-20261026/f20-four-output')

pairs = [
    (SITE / 'tail.comp', CARD / 'parent.comp', CARD / 'candidate.comp',
     'c32-boundary-candidates-20261026/f20-four-output/tail.comp',
     'cards/n02-f20-f21-tail/candidate.comp'),
    (SITE / 'ioport_common.glsl', CARD / 'headers/ioport_common.glsl', None,
     'c32-boundary-candidates-20261026/f20-four-output/ioport_common.glsl',
     'cards/n02-f20-f21-tail/headers/ioport_common.glsl'),
]

out = []
for src, parent_copy, cand, rel_a, rel_b in pairs:
    a = pathlib.Path(src).read_text(encoding='utf-8')
    if cand is None:
        b = pathlib.Path(parent_copy).read_text(encoding='utf-8')
    else:
        b = pathlib.Path(cand).read_text(encoding='utf-8')
    diff = difflib.unified_diff(a.splitlines(keepends=True), b.splitlines(keepends=True),
                                fromfile='a/' + rel_a, tofile='b/' + rel_b, n=3)
    out.extend(diff)

(CARD / 'SOURCE-DIFF.patch').write_text(''.join(out), encoding='utf-8')
print(''.join(out))
