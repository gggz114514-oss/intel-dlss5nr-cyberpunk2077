"""n18b-f01-gather: CPU-only identity checks for the parent (chain head) and its plan.

No candidate exists for this card (see RESUME); these two checks only re-establish
that the bytes and the plan this analysis is anchored on are intact:
  - spv_validate_cpu on parent-rebuild.spv (should equal the chain-head PASS 0 SPV),
  - clock_runner --cpu-check on the accepted chain-head plan (no Vulkan device).

Writes tools/identity-checks.json.
"""
import json, sys
from pathlib import Path

CARD = Path(__file__).resolve().parents[1]
CAMPAIGN = CARD.parents[1]
sys.path.insert(0, str(CAMPAIGN))
import COMMON as K

PLAN = CAMPAIGN / 'accepted' / 'n18-f01-entry-20261027' / 'plan.txt'
SPV = CARD / 'parent-rebuild.spv'


def main():
    out = dict(card='n18b-f01-gather')
    K.validate_spv(SPV)
    out['spv_validate'] = dict(path=str(SPV), sha256=K.sha(SPV), result='PASS',
                              tool=str(K.VALIDATE))
    txt = K.cpu_check(PLAN)
    low = txt.lower()
    out['cpu_check'] = dict(plan=str(PLAN), sha256=K.sha(PLAN), tool=str(K.RUNNER),
                            output=' '.join(txt.split())[:400],
                            vulkan_loaded=('vulkan' in low and 'no vulkan loaded' not in low))
    out['chain_head_expected_plan_sha'] = '244e919260fa416e8b915b029435b05c5db481f511e2f261777d4bbee82fae8e'
    out['plan_sha_matches_chain_head'] = K.sha(PLAN) == out['chain_head_expected_plan_sha']
    (CARD / 'tools' / 'identity-checks.json').write_text(json.dumps(out, indent=2), encoding='utf-8')
    print(json.dumps(out, indent=1))


if __name__ == '__main__':
    main()
