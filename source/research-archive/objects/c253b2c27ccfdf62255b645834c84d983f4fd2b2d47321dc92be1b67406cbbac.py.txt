"""Read-only ISA scan for n20-f18-ups-contract (F18 ord 194, C128 UPS contract).

No GPU: reads an ISA capture that already exists on disk (identity CURRENT, SPV sha
ffde3c3d...), plus the SPIR-V of the same shader.
"""
import re, sys, json, collections
from pathlib import Path

W = Path('@NR_WORKSPACE@')
C = W / 'work/codex-astra-fp16-int8-audit-20261024/fullnet-continuous-20261027'
ISA_DEFAULT = (W / 'work/codex-astra-fp16-int8-audit-20261024/fullnet-mechanism-transfer-20261026'
               / 'f05-a-return-phase1-20261027/captures/boundary/ISA'
               / 'CS_asm4e6ba3ddc1931535_simd32_0000000000000022_33_orig.asm')

MSG = re.compile(r'//\s*wr:(\d+)\+(\d+),\s*rd:(\d+);\s*(.+?)\s*$')
MASK = re.compile(r'\((\d+)\|M0\)')
PRED = re.compile(r'^\(([^)]*)\)$')


def parse(path):
    inst = []
    for raw in Path(path).read_text(encoding='utf-8', errors='replace').splitlines():
        if '/*' not in raw or '*/' not in raw:
            continue
        addr = re.search(r'/\*\s*\[([0-9A-Fa-f]+)\]', raw)
        body = raw.split('*/', 1)[1]
        toks = body.split()
        pred = ''
        while toks and PRED.match(toks[0]):
            pred += toks.pop(0)
        if not toks:
            continue
        op = toks[0]
        m = MASK.search(body)
        inst.append(dict(addr=addr.group(1) if addr else None, pred=pred, op=op,
                         mask=int(m.group(1)) if m else None, text=raw.strip(),
                         tail=' '.join(toks[1:])))
    return inst


def width_bytes(name):
    m = re.search(r'\.d(\d+)x(\d+)(t?)', name)
    if m:
        n = int(m.group(1)) * int(m.group(2))
        return n, n * 4
    m = re.search(r'\.d(\d+)u(\d+)', name)
    if m:
        n = int(m.group(1))
        return n, n * int(m.group(2)) // 8
    m = re.search(r'\.d(\d+)t', name)
    if m:
        n = int(m.group(1))
        return n, n * 4
    return None, None


def main():
    p = Path(sys.argv[1]) if len(sys.argv) > 1 else ISA_DEFAULT
    inst = parse(p)
    sends = [i for i in inst if i['op'].startswith('send')]
    dpas = [i for i in inst if i['op'].startswith('dpas')]
    sync = [i for i in inst if i['op'].startswith('sync')]
    kinds = collections.Counter()
    for s in sends:
        g = MSG.search(s['text'])
        kinds[g.group(4) if g else 'UNPARSED'] += 1
    print('file      ', p)
    print('instructions', len(inst), ' send', len(sends), ' dpas', len(dpas), ' sync', len(sync))
    print('sync kinds  ', dict(collections.Counter(i['op'] for i in sync)))
    print('masks      ', dict(sorted(collections.Counter(i['mask'] for i in inst).items(), key=lambda x: (x[0] is None, x[0]))))
    print('opcodes    ', dict(collections.Counter(i['op'] for i in inst).most_common(30)))
    print()
    print('%-54s %5s %9s %11s' % ('message', 'n', 'dwords', 'bytes'))
    tot_d = tot_b = 0
    for k, v in sorted(kinds.items(), key=lambda x: -x[1]):
        d, b = width_bytes(k)
        tot_d += (d or 0) * v
        tot_b += (b or 0) * v
        print('%-54s %5d %9s %11s' % (k, v, (d or 0) * v, (b or 0) * v))
    print('%-54s %5s %9d %11d' % ('TOTAL', len(sends), tot_d, tot_b))
    print()
    print('address range: first', inst[0]['addr'], 'last', inst[-1]['addr'])
    # send positions in program order
    idx = {id(s): n for n, s in enumerate(inst)}
    print('send indices (first 40):', [idx[id(s)] for s in sends[:40]])
    print('dpas indices:', [idx[id(d)] for d in dpas])
    print('dpas ops   ', collections.Counter(d['op'] for d in dpas))
    print('dpas first text:', dpas[0]['text'] if dpas else None)
    print('send text samples:')
    seen = set()
    for s in sends:
        k = MSG.search(s['text'])
        key = (s['op'], k.group(4) if k else None)
        if key in seen:
            continue
        seen.add(key)
        print('   ', s['text'][:200])


if __name__ == '__main__':
    main()
