"""Clean the campaign's intermediate artifacts (frames, rebuild scratch, one-off probes).

Keeps every deliverable and every piece of verifiable evidence:
  - cards/<id>/             full card deliverables (ANALYSIS/PROOF/BUILD/plan/candidate.spv/...)
  - accepted/<step>/        the adopted plans
  - evidence/*.json         recipes, coverage, native index, lessons, audits, slim report
  - queue/<id>/*/           receipts, manifests, stage summaries, rebased plans, changed-ISA only
Deletes only:
  - frame_*.bin (28 MB each) after recording their sha256 in FRAME-HASHES.json
  - cards/*/repro/          the auditor's rebuild scratch
  - evidence/identity*/ evidence/parent-byte/  the rebuild scratch of the identity stage
  - one-off probe scripts and stale plan-only queue dirs
"""
import hashlib, json, shutil, sys
from pathlib import Path
sys.path.insert(0, str(Path(__file__).parent))
import COMMON as K

report = dict(frames=[], frame_bytes=0, dirs=[], dir_bytes=0, files=[], file_bytes=0)


def sha(p):
    h = hashlib.sha256()
    with open(p, 'rb') as f:
        for chunk in iter(lambda: f.read(1 << 20), b''):
            h.update(chunk)
    return h.hexdigest()


def dirsize(p):
    return sum(f.stat().st_size for f in p.rglob('*') if f.is_file())


# 1) every frame binary anywhere under queue/, hashed then removed
for f in list((K.C / 'queue').rglob('frame_*.bin')):
    h = sha(f)
    rep = f.parent / 'FRAME-HASHES.json'
    d = K.read_json(rep) if rep.exists() else {}
    d[f.name] = dict(sha256=h, bytes=f.stat().st_size)
    K.write_json(rep, d)
    report['frames'].append(str(f.relative_to(K.C)))
    report['frame_bytes'] += f.stat().st_size
    f.unlink()

# 2) audit rebuild scratch
for rel in ('cards/n01-c32-chain/repro', 'cards/n05a-f05-c256-mreuse/_build/repro',
            'cards/n11-vit-ffn/repro', 'cards/n11b-vit-projection-hoist/repro',
            'cards/n11-f13-expand-last8/repro', 'cards/n09b-f09-sx-stage/repro',
            'evidence/identity', 'evidence/identity2', 'evidence/parent-byte'):
    p = K.C / rel
    if p.exists():
        report['dirs'].append(dict(path=rel, files=sum(1 for _ in p.rglob('*') if _.is_file())))
        report['dir_bytes'] += dirsize(p)
        shutil.rmtree(p)

# 3) one-off probes / superseded generators
for rel in ('probe.py', 'probe_n08b.py', 'report_two.py', 'make_smoke.py',
            'verify_identity.py', 'verify_identity2.py', 'reproduce_with_parent.py',
            'find_recipes.py', 'make_cards.py', 'make_cards2.py', 'make_cards3.py',
            'make_cards4.py', 'make_cards5.py', 'make_n11_last8.py',
            'queue/ISA-DIFF-TMP.json', 'queue/pkg-vit-hoist', 'queue/pkg-f04-c128',
            'queue/smoke-baseline'):
    p = K.C / rel
    if not p.exists():
        continue
    if p.is_dir():
        report['dirs'].append(dict(path=rel, files=sum(1 for _ in p.rglob('*') if _.is_file())))
        report['dir_bytes'] += dirsize(p)
        shutil.rmtree(p)
    else:
        report['files'].append(rel)
        report['file_bytes'] += p.stat().st_size
        p.unlink()

# 4) python byte-code caches
for p in list(K.C.rglob('__pycache__')):
    if p.is_dir():
        report['dirs'].append(dict(path=str(p.relative_to(K.C)), files=len(list(p.glob('*')))))
        report['dir_bytes'] += dirsize(p)
        shutil.rmtree(p)

report['reclaimed_bytes'] = report['frame_bytes'] + report['dir_bytes'] + report['file_bytes']
K.write_json(K.C / 'evidence' / 'CLEANUP-REPORT.json', report)
print('frames removed %d (%.1f MB)' % (len(report['frames']), report['frame_bytes'] / 1e6))
print('dirs removed %d (%.1f MB)' % (len(report['dirs']), report['dir_bytes'] / 1e6))
print('files removed %d (%.1f MB)' % (len(report['files']), report['file_bytes'] / 1e6))
print('total reclaimed %.1f MB' % (report['reclaimed_bytes'] / 1e6))
