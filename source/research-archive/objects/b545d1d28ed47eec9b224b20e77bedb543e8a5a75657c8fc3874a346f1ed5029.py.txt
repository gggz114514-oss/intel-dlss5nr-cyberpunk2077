from pathlib import Path
import json, re
import numpy as np
from PIL import Image,ImageDraw
Q=Path(__file__).resolve().parent;D=Q/'real-int8-24'
def metrics(a,b):
    a=a.astype(np.float64);b=b.astype(np.float64);e=b-a
    if not np.isfinite(b).all():return {'nonfinite':int((~np.isfinite(b)).sum())}
    mse=float(np.mean(e*e));den=float(np.sum(a*a))
    return dict(rmse=float(np.sqrt(mse)),mae=float(abs(e).mean()),max_abs=float(abs(e).max()),rel_l2=float(np.sqrt(np.sum(e*e)/max(den,1e-30))),psnr_peak1_db=float(-10*np.log10(max(mse,1e-30))),nonfinite=0)
rows=[]
for j in json.loads((D/'JOBS.json').read_text()):
    if j['dump']:continue
    p=Path(j['out'])/'replayed-network.bin'
    if not p.exists():continue
    a=np.fromfile(j['reference'],dtype='<f4').reshape(2,720,1280,4);b=np.fromfile(p,dtype='<f4').reshape(a.shape)
    rows.append(dict(sequence=j['sequence'],frame=j['frame'],application_rgb=metrics(a[0,:,:,:3],b[0,:,:,:3]),history_rgb=metrics(a[1,:,:,:3],b[1,:,:,:3])))
    if j['frame'] in [34189,41444,43654]:
        canvas=Image.new('RGB',(2560,1496),(25,25,25));draw=ImageDraw.Draw(canvas)
        for i,(label,v) in enumerate([('FP16 captured output',a[0,:,:,:3]),('Corrected INT8 output',b[0,:,:,:3]),('Absolute RGB error x4',abs(a[0,:,:,:3]-b[0,:,:,:3])*4),('Absolute RGB error x16',abs(a[0,:,:,:3]-b[0,:,:,:3])*16)]):
            x=(i%2)*1280;y=(i//2)*748;draw.text((x+8,y+8),label,fill='white');canvas.paste(Image.fromarray(np.rint(np.clip(v,0,1)*255).astype(np.uint8)),(x,y+28))
        canvas.save(D/f'comparison-{j["frame"]}.png')
(D/'FRAME-METRICS.json').write_text(json.dumps(rows,indent=2))
print('Frames',len(rows))
if rows:print('Application PSNR range',min(r['application_rgb'].get('psnr_peak1_db',-999) for r in rows),max(r['application_rgb'].get('psnr_peak1_db',-999) for r in rows))
bound=[]
for p in (D/'dump-fp16').glob('dump_*.bin'):
    name=p.name
    # Match semantic boundaries, not raw workspace/weight buffers or repacked layouts.
    if not (re.fullmatch(r'dump_block\d+_out.bin',name) or name.startswith('dump_ioport_') or re.fullmatch(r'dump_skip\d+.bin',name)):continue
    other=D/'dump-int8'/name
    if not other.exists() or p.stat().st_size!=other.stat().st_size:continue
    a=np.fromfile(p,dtype='<f2');b=np.fromfile(other,dtype='<f2')
    bound.append(dict(name=name,bytes=p.stat().st_size,fp16_absmax=float(np.max(abs(a.astype(float)))),**metrics(a,b)))
(D/'BOUNDARY-METRICS.json').write_text(json.dumps(bound,indent=2))
for r in sorted(bound,key=lambda r:r.get('rel_l2',1e9),reverse=True)[:20]:print(r['name'],r.get('rel_l2'),r.get('nonfinite'))
