"""n31: GPU-REQUEST.json + plan.txt on the current chain head, then cpu-check.

Uses COMMON.rebase_by_ordinals (the sanctioned path for a follow-up card whose
parent SPV is itself an adopted candidate): it copies the chain-head plan and
replaces only the SPV path of the card's PASS ordinals, carrying every other
field (grid, push constants, bindings) verbatim.

out_plan is pinned inside the card: the helper's default target is
C/queue/<card>/, which is outside this card's write boundary.
"""
import json
import sys
from pathlib import Path

CARD = Path(__file__).resolve().parents[1]
CAMP = CARD.parents[1]
sys.path.insert(0, str(CAMP))
import COMMON as K  # noqa: E402

BASE_SPV_SHA = 'c12c0f9d6e0420066f79fd057f4a78da048d2f6c75c5e64fc8ad75c3764b39db'   # n29 chain head

REQ = dict(
    card='n31-f01-dpas-chain',
    shader_group=dict(spv_sha256_baseline=[BASE_SPV_SHA], pass_ordinals=[0]),
    plan=dict(path=str(CARD / 'plan.txt'), sha256=None),
    arms=[dict(id='base', plan='baseline', role='baseline'),
          dict(id='cand', plan='candidate', role='candidate')],
    gates=['fixed', 'fb2'],
    native=True,
    resources=True,
    ordinary=dict(design='two-arm-6-alternating', reps=30, warm=1, feedback=True),
    hypothesis=('if native ordinal 0 shows sel 376 -> ~240 (the operand-swap sels of the four '
                'softmax-denominator blocks and of nr_pool_natural are gone) and the '
                'XVE_STALL_ALUWR milliseconds fall, the identity-select deletion landed'),
    native_observables=[
        'sel (whole kernel) 376 -> 240 +- 8; the 4-sel groups of the four denominator blocks gone',
        'the 29 lower predicates ((lane&8/4/2/1/16)) gone from and/cmp',
        'math.inv 8 unchanged, math.rsqt 16 unchanged (math is frozen)',
        'dpas.8x8 128 unchanged, send.slm 27 unchanged, max_r not above parent, scratch 0',
        'region instructions 4996 -> 4828 +- 20; live 3927 -> 3791 +- 20',
        'integer control-flow opcode set {128,130,132,134,135,137,138,139} unchanged',
    ],
    refuted_if=['sel still 376 => class A: the implementation did not land',
                'dpas / send.slm / math.inv counts move => class E: wrong site',
                'fixed or fb2 gold frame mismatch => class E'],
)


def main():
    K.write_json(CARD / 'GPU-REQUEST.json', REQ)
    plan = CARD / 'plan.txt'
    out, psha, n, meta = K.rebase_by_ordinals('n31-f01-dpas-chain', out_plan=plan)
    assert Path(out) == plan, out
    print('rebase: %d ordinal(s) replaced; baseline %s' % (n, meta['baseline_sha'][:12]))

    base, base_sha, _ = K.current_best()
    a = Path(base).read_text(encoding='utf-8-sig').splitlines()
    b = plan.read_text(encoding='utf-8-sig').splitlines()
    assert len(a) == len(b), (len(a), len(b))
    diff = [(i + 1, x, y) for i, (x, y) in enumerate(zip(a, b)) if x != y]
    print('base plan %s  sha %s' % (base, base_sha))
    print('card plan %s  sha %s' % (plan, psha))
    print('lines %d ; differing rows %d' % (len(b), len(diff)))
    for ln, x, y in diff:
        print('  line %d\n    - %s\n    + %s' % (ln, x[:150], y[:150]))
    assert len(diff) == 1, 'expected exactly one replaced PASS row'
    assert 'n31-f01-dpas-chain/candidate.spv' in diff[0][2]
    assert diff[0][1].startswith('PASS ')

    req = dict(REQ)
    req['plan']['sha256'] = psha
    req['plan']['base_plan'] = dict(path=str(base), sha256=base_sha)
    req['plan']['diff_rows'] = [diff[0][0]]
    req['plan']['passes'] = 232
    K.write_json(CARD / 'GPU-REQUEST.json', req)

    res = K.cpu_check(plan)
    print('cpu_check:', res.strip())
    K.write_json(CARD / 'tools/cpu-check.json',
                 dict(plan=str(plan), sha256=psha, stdout=res, returncode=0))


if __name__ == '__main__':
    main()
