"""Build cards/n29-f01-alu-chain/plan.txt from the current chain head.

Only the PASS 0 row may change (its SPV path); every other byte is carried.
Assertions: exactly one PASS row differs, 232 PASS rows, 502 buffers, the
PASS 0 header fields (local size / SLM / grid) are untouched.
"""
import re
import sys
from pathlib import Path

CARD = Path(__file__).resolve().parents[1]
sys.path.insert(0, str(CARD.parents[1]))
import COMMON as K                      # noqa: E402

OUT = CARD / 'plan.txt'
PARENT_SPV = (CARD.parent / 'n18-f01-entry' / 'candidate.spv').as_posix()
CAND_SPV = (CARD / 'candidate.spv').as_posix()


def main():
    base, base_sha, _ = K.current_best()
    raw = Path(base).read_text(encoding='utf-8-sig')
    lines = raw.splitlines()
    hits = [i for i, l in enumerate(lines) if PARENT_SPV in l]
    assert len(hits) == 1, ('expected exactly one row carrying the parent SPV', hits)
    i = hits[0]
    assert lines[i].startswith('PASS 0 '), lines[i]
    new = lines[i].replace(PARENT_SPV, CAND_SPV)
    lines[i] = new
    OUT.write_text('\n'.join(lines) + ('\n' if raw.endswith('\n') else ''), encoding='utf-8')
    a = K.pass_lines_of(base)
    b = K.pass_lines_of(OUT)
    assert len(a) == len(b) == 232, (len(a), len(b))
    diff = [k for k in range(232) if a[k]['line'] != b[k]['line']]
    assert diff == [0], diff
    pa, pb = a[0]['line'].split(), b[0]['line'].split()
    assert len(pa) == len(pb)
    strip = lambda toks: [t for t in toks if not t.strip('"').endswith('.spv')]  # noqa: E731
    assert strip(pa) == strip(pb), (strip(pa), strip(pb))
    print('base plan      :', base, base_sha)
    print('base PASS 0    :', a[0]['line'])
    print('card PASS 0    :', b[0]['line'])
    print('rows           : %d passes, buffers=%d' % (
        len(b), sum(1 for l in lines if l.startswith('BUFFER '))))
    print('changed rows   :', diff)
    print('plan sha256    :', K.sha(OUT))
    print('header fields  :', ' '.join(pa[2:]))
    chk = K.cpu_check(OUT)
    print('cpu_check      :', chk.strip())
    K.write_json(CARD / 'new-plan.json',
                 dict(base=str(base), base_sha256=base_sha, plan=K.ident(OUT),
                      changed_rows=diff, passes=len(b),
                      buffers=sum(1 for l in lines if l.startswith('BUFFER ')),
                      pass0=a[0]['line'], pass0_new=b[0]['line'], cpu_check=chk.strip()))


if __name__ == '__main__':
    main()
