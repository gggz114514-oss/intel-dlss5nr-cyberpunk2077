from pathlib import Path
import json,hashlib
import numpy as np
from PIL import Image,ImageDraw
Q=Path(__file__).resolve().parent;D=Q/'real-int8-24';jobs=[j for j in json.loads((D/'JOBS.json').read_text()) if not j['dump']]
arms={'original':lambda f:D/f'run-{f}','c32_static':lambda f:D/f'c32-minmax/run-{f}','lift_c32':lambda f:D/f'lift-equalized/lift-c32-{f}','fine_c32':lambda f:D/f'c32-equalized/run-{f}'}
rows=[]
for j in jobs:
 a=np.fromfile(j['reference'],dtype='<f4').reshape(2,720,1280,4)
 for arm,folder in arms.items():
  p=folder(j['frame'])/'replayed-network.bin'
  if not p.exists():continue
  b=np.fromfile(p,dtype='<f4').reshape(a.shape);assert np.isfinite(b).all()
  e=b[0,:,:,:3].astype(float)-a[0,:,:,:3].astype(float);mse=float(np.mean(e*e))
  rows.append({'arm':arm,'sequence':j['sequence'],'frame':j['frame'],'psnr_peak1':float(-10*np.log10(mse)),'mse':mse,'mae':float(abs(e).mean()),'max_abs':float(abs(e).max()),'nonfinite':0,'output_sha256':hashlib.sha256(p.read_bytes()).hexdigest()})
summary={}
for arm in arms:
 rs=[r for r in rows if r['arm']==arm]
 if rs:summary[arm]={'frames':len(rs),'psnr_mean':float(np.mean([r['psnr_peak1'] for r in rs])),'psnr_min':min(r['psnr_peak1'] for r in rs),'psnr_max':max(r['psnr_peak1'] for r in rs),'pooled_psnr':float(-10*np.log10(np.mean([r['mse'] for r in rs]))),'pooled_rmse':float(np.sqrt(np.mean([r['mse'] for r in rs]))),'mean_mae':float(np.mean([r['mae'] for r in rs]))}
(D/'QUALITY-SUMMARY.json').write_text(json.dumps({'scope':'24 daytime frames with shared captured FP16 history; three representative frames calibrate; not holdout or autonomous feedback','summary':summary,'frames':rows},indent=2));print(json.dumps(summary,indent=2))
for frame in [34189,41444,43654]:
 j=next(j for j in jobs if j['frame']==frame);a=np.fromfile(j['reference'],dtype='<f4').reshape(2,720,1280,4)[0,:,:,:3];old=np.fromfile(D/f'run-{frame}/replayed-network.bin',dtype='<f4').reshape(2,720,1280,4)[0,:,:,:3];new=np.fromfile(D/f'c32-equalized/run-{frame}/replayed-network.bin',dtype='<f4').reshape(2,720,1280,4)[0,:,:,:3]
 canvas=Image.new('RGB',(2560,1496),(25,25,25));draw=ImageDraw.Draw(canvas)
 for i,(label,v) in enumerate([('FP16 reference',a),('Original corrected INT8',old),('Offline equalized INT8 candidate',new),('Candidate absolute RGB error x4',abs(new-a)*4)]):
  x=(i%2)*1280;y=(i//2)*748;draw.text((x+8,y+8),label,fill='white');canvas.paste(Image.fromarray(np.rint(np.clip(v,0,1)*255).astype(np.uint8)),(x,y+28))
 canvas.save(D/f'fine-comparison-{frame}.png')
