"""Structural / NoContraction alignment proof for n05c (CPU only).

Uses the campaign's own pattern alignment (the same routine the identity cards
use) to ask: do the parent's 73 NoContraction sites map one-to-one onto the
candidate, and does the candidate carry decorations outside that pattern?

  * parent_n / candidate_n            : number of NoContraction decorated ids
  * parent_ids_matched_into_candidate : how many parent sites the aligned
                                        pattern transfer finds in the candidate
  * candidate_own_decorations_outside_pattern : decorations the candidate would
                                        have to add itself (must be 0)
"""
import json
import sys
from pathlib import Path

CARD = Path(__file__).resolve().parent.parent
BUILD = CARD / '_build'
W = Path('@NR_WORKSPACE@')
SPV_TOOLS = W / 'build/zcode-fp16-compute-backend-20261021/scripts'
sys.path.insert(0, str(SPV_TOOLS))
import spv_tools_local as ST                         # noqa: E402


def main():
    pm = ST.Module(BUILD / 'parent-raw.spv')
    cm = ST.Module(BUILD / 'cand-raw.spv')
    pid = ST.no_contraction_ids(pm)
    cid = ST.no_contraction_ids(cm)
    keep, info = ST.transfer_decoration_pattern_aligned(pm, cm, set())
    res = dict(
        parent_n=len(pid), candidate_n=len(cid),
        parent_ids_matched_into_candidate=len(keep),
        count_equal=len(pid) == len(cid),
        info={k: info[k] for k in info if isinstance(info[k], (int, float, str, bool, list))},
        float_NC_checked=len(cid),
        dynamic_per_sg=dict(
            parent=dict(weight_loads=96, a_row_loads=128, dpas=96),
            candidate=dict(weight_loads=48, a_row_loads=256, dpas=96),
            note='8 ki x 6 weight + 8 ki x 1 A per row_phase x 2 row_phases (parent); '
                 'QK pass 8 ki x 4 weight + 2 A and V pass 8 ki x 2 weight + 2 A '
                 '(candidate).  Static SPIR-V sites: see build.json.',
        ),
    )
    (BUILD / 'nc.json').write_text(json.dumps(res, indent=2, default=str), encoding='utf-8')
    print(json.dumps({k: res[k] for k in ('parent_n', 'candidate_n',
                                          'parent_ids_matched_into_candidate',
                                          'count_equal')}))
    print(json.dumps(res['info'], indent=1, default=str)[:1400])


if __name__ == '__main__':
    main()
