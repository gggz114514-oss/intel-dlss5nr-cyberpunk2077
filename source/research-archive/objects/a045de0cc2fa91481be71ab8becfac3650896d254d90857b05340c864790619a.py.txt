from pathlib import Path
import re,json,hashlib
D=Path(__file__).resolve().parent;A=D.parent
read=lambda p:json.loads(p.read_text(encoding='utf-8-sig'))
paths=read(A/'f20-wide-publish-20261026/publish-linear/NATIVE-REVIEW.json')['rows'][0]
rows={}
for arm,key in [('parent','parent'),('linear','candidate')]:
 p=Path(paths[key]['path']);ls=[x for x in p.read_text().splitlines() if re.match(r'/\* \[[0-9A-F]+\]',x)]
 dp=max(i for i,l in enumerate(ls) if 'dpas.' in l);bar=next(i for i in range(dp,len(ls)) if 'sync.bar' in ls[i]);first=next(i for i in range(dp,bar) if 'store.slm.' in ls[i]);fence=next(i for i in range(dp,bar) if 'fence.slm' in ls[i]);end=next(i for i in range(bar,len(ls)) if 'store.ugm' in ls[i])
 region=ls[dp+1:bar+1];tail=ls[bar+1:end+1]
 store=[l for l in region if 'store.slm.' in l]
 counts=lambda x:{k:sum(k in l for l in x) for k in ['mov (16','mov (32','mov (8','shl','bfn.','add','send.slm','sync.nop','sync.bar','mul (32']}
 row=dict(native_path=str(p),sha256=hashlib.sha256(p.read_bytes()).hexdigest(),last_dpas=ls[dp],first_store=ls[first],last_store=store[-1],fence_to_barrier=ls[fence:bar+1],publication_region=region,publication_counts=counts(region),post_dpas_to_barrier_instructions=len(region),before_first_store_instructions=first-dp-1,tail_counts=counts(tail),tail_instruction_count=len(tail),tail_wide_reads=[x for x in tail if 'load.slm.d32x4' in x],tail_weights=[x for x in tail if 'load.ugm.d32x16t' in x],payload_register_units=sum(sum(map(int,re.search(r'wr:(\d+)\+(\d+)',l).groups())) for l in store))
 rows[arm]=row
 (D/(arm+'-native-publication.txt')).write_text('\n'.join(ls[max(0,dp-30):end+1]),encoding='utf-8')
(D/'STATIC-REVIEW.json').write_text(json.dumps(rows,indent=2),encoding='utf-8')
for arm,r in rows.items():print(arm,{k:r[k] for k in ['post_dpas_to_barrier_instructions','before_first_store_instructions','publication_counts','tail_instruction_count','payload_register_units']})
