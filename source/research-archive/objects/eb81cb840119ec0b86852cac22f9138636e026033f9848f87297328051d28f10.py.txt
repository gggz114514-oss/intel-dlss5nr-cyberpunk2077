#!/usr/bin/env python3
"""qX edge ownership check for a cumulative plan.

For every C128 qX edge (producer body_4 -> consumer body_1) verify, from the plan itself:
  * the producer binds the DESTINATION block's qX buffer at BIND 8 and its scale table at BIND 9
  * the consumer binds the SAME qX resource at BIND 2 and the SAME scale table at BIND 1
  * the qX resource is written by exactly one PASS and read by exactly one PASS
  * the scale table is bound by producer and consumer with identical bytes
This is the check that would have caught a producer/consumer pair that disagree about which
buffer holds the edge.

usage: qx_edge_check.py <plan>
"""
import sys
from collections import defaultdict
from pathlib import Path

sys.path.insert(0, r"@NR_WORKSPACE@/work/dsh-int8-fullnet-impl-20261024/tools")
import planlib as L  # noqa: E402

EDGES = {12: 11, 13: 12, 14: 13, 57: 56, 58: 57, 59: 58, 60: 59, 61: 60}


def binds(p):
    return {b["binding"]: b for b in p["binds"]}


def main(argv):
    plan = L.Plan(argv[0])
    out, ok = [], True
    writers, readers = defaultdict(list), defaultdict(list)
    for n, prod in EDGES.items():
        pp = plan.by_name(f"block{prod}_swin_body_4")
        cp = plan.by_name(f"block{n}_swin_body_1")
        if len(pp) != 1 or len(cp) != 1:
            out.append({"edge": f"block{prod}->block{n}", "error": "pass not found"})
            ok = False
            continue
        pp, cp = pp[0], cp[0]
        pb, cb = binds(pp), binds(cp)
        qx_p = pb.get(8, {}).get("res")
        sc_p = pb.get(9, {}).get("res")
        qx_c = cb.get(2, {}).get("res")
        sc_c = cb.get(1, {}).get("res")
        same = (qx_p is not None and qx_p == qx_c and sc_p is not None and sc_p == sc_c)
        sizes = same and pb[8]["bytes"] == cb[2]["bytes"] and pb[9]["bytes"] == cb[1]["bytes"]
        rec = {"edge": f"block{prod}_swin_body_4 -> block{n}_swin_body_1",
               "producer_spv": Path(pp["spv"]).name,
               "consumer_spv": Path(cp["spv"]).name,
               "producer_qx_res": qx_p, "consumer_qx_res": qx_c,
               "producer_sc_res": sc_p, "consumer_sc_res": sc_c,
               "qx_buffer": plan.buffers[qx_p][0] if qx_p is not None else None,
               "sc_buffer": plan.buffers[sc_p][0] if sc_p is not None else None,
               "same_resources": bool(same), "same_bytes": bool(sizes)}
        if not (same and sizes):
            ok = False
        if qx_p is not None:
            writers[qx_p].append(pp["name"])
        if qx_c is not None:
            readers[qx_c].append(cp["name"])
        out.append(rec)
    single = {r: (writers[r], readers[r]) for r in set(writers) | set(readers)
              if len(writers[r]) > 1 or len(readers[r]) > 1}
    print(f"{'edge':46} {'producer':26} {'consumer':26} qx ok bytes")
    for r in out:
        print(f"{r['edge']:46} {r.get('producer_spv',''):26} {r.get('consumer_spv',''):26} "
              f"{r.get('same_resources')} {r.get('same_bytes')}")
    print("multi-writer/reader qX resources:", single or "none")
    print("ok", ok and not single)
    return 0 if (ok and not single) else 1


if __name__ == "__main__":
    sys.exit(main(sys.argv[1:]))
