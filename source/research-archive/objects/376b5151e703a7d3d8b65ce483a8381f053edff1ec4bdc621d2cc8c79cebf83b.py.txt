import struct, collections
p = r"@NR_WORKSPACE@/candidates/int8-coverage-v2-20261020/B/spv/c64_d1.spv"
data=open(p,'rb').read(); n=len(data)//4
w=struct.unpack("<%dI"%n, data[:n*4]); i=5
defs={}
instrs=[]
while i<n:
    wc=w[i]>>16; op=w[i]&0xFFFF
    if wc==0: break
    ops=list(w[i+1:i+wc])
    instrs.append((op,ops))
    if op not in (5,6,7,10,11,71,72,73,74,248,249,250,251,252,253,254,62,63,56,55) and len(ops)>=2:
        defs[ops[1]]=(op,ops)
    i+=wc
for target in (375, 88, 22, 48):
    print(target, defs.get(target))
# coopmat type decls
for op,ops in instrs:
    if op in (4456,5358):
        print("coopmat type", op, ops)
