"""Final self-check for n05e-f05-atile (CPU only, read-only on everything outside the card).

  1. every required deliverable exists and is non-empty;
  2. every JSON parses, and the identity chain (parent SPV, parent source, candidate
     comp, candidate SPV, plan) matches what BUILD.json / GPU-REQUEST.json claim;
  3. SOURCE-DIFF.patch applied to the parent source reproduces candidate.comp exactly;
  4. the read-only inputs this card depends on are unchanged (baseline plan, chain
     head plan, COMMON.py, parent SPV, parent source, CURRENT.json);
  5. the plan file references candidate.spv for exactly the 12 group PASS lines.
"""
import hashlib
import json
import sys
from pathlib import Path

CARD = Path(__file__).resolve().parent.parent
B = CARD / '_build'
C = CARD.parent.parent
sys.path.insert(0, str(C))
import COMMON as K                                   # noqa: E402

REQUIRED = ['CARD.txt', 'ANALYSIS.md', 'PROOF.json', 'SOURCE-DIFF.patch', 'candidate.comp',
            'BUILD.json', 'candidate.spv', 'plan.txt', 'COST-LEDGER.json',
            'NATIVE-PREDICTION.md', 'GPU-REQUEST.json', 'RESUME.txt', 'parent-rebuild.spv',
            'parent-rebuild.comp']
FROZEN = {
    'baseline plan': (K.BASELINE, K.BASELINE_SHA),
    'chain head plan': (C / 'accepted/n18-f01-entry-20261027/plan.txt',
                        '244e919260fa416e8b915b029435b05c5db481f511e2f261777d4bbee82fae8e'),
    'parent SPV': (Path('@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-'
                        '20261024/fullnet-mechanism-transfer-20261026/f05-kv-consumer-layout/'
                        'family/c128/candidate.spv'),
                   '20f4074fc70c32c0e74f0ad000053492e5f5c420259da6044e7a00cfc3ee3f4f'),
    'parent source': (Path('@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-'
                           '20261024/fullnet-mechanism-transfer-20261026/f05-kv-consumer-layout/'
                           'family/c128/candidate.comp'),
                      'd628a08e35bce7b148dc330b42b20004a404a6a9c1f03eb20ca3df90803699f2'),
    'COMMON.py': (C / 'COMMON.py', '0438138ccae004ab9fc0da8b9cd771f9f76b98327d5007b8c0b25ed9b1185596'),
}


def sha(p):
    return hashlib.sha256(Path(p).read_bytes()).hexdigest()


def main():
    ok = True

    def check(cond, label, detail=''):
        nonlocal ok
        print(('PASS ' if cond else 'FAIL ') + label + (('  ' + detail) if detail else ''))
        ok = ok and bool(cond)

    print('== 1. deliverables')
    for name in REQUIRED:
        p = CARD / name
        check(p.exists() and p.stat().st_size > 0, name,
              '%d B' % p.stat().st_size if p.exists() else 'MISSING')

    print('== 2. JSON + identity chain')
    js = {}
    for name in ('BUILD.json', 'PROOF.json', 'COST-LEDGER.json', 'GPU-REQUEST.json'):
        try:
            js[name] = json.loads((CARD / name).read_text(encoding='utf-8'))
            check(True, name + ' parses')
        except Exception as e:                        # noqa: BLE001
            check(False, name + ' parses', str(e))
    build = js['BUILD.json']
    req = js['GPU-REQUEST.json']
    check(build['candidate']['spv_sha256'] == sha(CARD / 'candidate.spv'),
          'BUILD.candidate.spv_sha256 == file')
    check(build['candidate']['comp_sha256'] == sha(CARD / 'candidate.comp'),
          'BUILD.candidate.comp_sha256 == file')
    check(build['parent_rebuild']['byte_identical'] is True, 'parent rebuild byte-identical')
    check(build['parent_rebuild']['produced_sha256'] ==
          build['parent']['spv']['sha256'], 'rebuilt SPV == baseline SPV sha')
    check(build['plan']['sha256'] == sha(CARD / 'plan.txt'), 'BUILD.plan.sha256 == plan.txt')
    check(req['plan']['sha256'] == sha(CARD / 'plan.txt'), 'GPU-REQUEST.plan.sha256 == plan.txt')
    check(req['shader_group']['spv_sha256_candidate'] == [sha(CARD / 'candidate.spv')],
          'GPU-REQUEST candidate sha')
    check(req['shader_group']['pass_ordinals'] ==
          [19, 22, 25, 28, 31, 34, 195, 198, 201, 204, 207, 210], '12 group ordinals')
    check(build['float_NC_checked'] == 73, 'float_NC_checked == 73')
    check(build['cpu_check']['stdout'].startswith('CPU_PLAN_OK'), 'cpu_check CPU_PLAN_OK')

    print('== 3. patch reproduces the candidate')
    parent = (CARD / 'parent-rebuild.comp').read_text(encoding='utf-8').splitlines(keepends=True)
    cand = (CARD / 'candidate.comp').read_text(encoding='utf-8').splitlines(keepends=True)
    patch = (CARD / 'SOURCE-DIFF.patch').read_text(encoding='utf-8').splitlines()
    # replay the unified diff (n=3 context) against the parent
    out, i, cur = [], 0, 0
    while i < len(patch):
        line = patch[i]
        if line.startswith('---') or line.startswith('+++'):
            i += 1
            continue
        if line.startswith('@@'):
            hdr = line.split('@@')[1].strip()
            a = hdr.split(' ')[0]
            start = int(a.split(',')[0][1:])
            target = start - 1
            while cur < target:
                out.append(parent[cur]); cur += 1
            i += 1
            while i < len(patch) and (patch[i][:1] in (' ', '+', '-')):
                l = patch[i]
                if l.startswith(' '):
                    assert parent[cur] == l[1:] + '\n' or parent[cur] == l[1:], \
                        ('context mismatch', l[:60])
                    out.append(parent[cur]); cur += 1
                elif l.startswith('-'):
                    assert parent[cur].rstrip('\n') == l[1:], ('delete mismatch', l[:60])
                    cur += 1
                else:
                    out.append(l[1:] + '\n')
                i += 1
            continue
        i += 1
    while cur < len(parent):
        out.append(parent[cur]); cur += 1
    check(''.join(out) == ''.join(cand), 'parent + SOURCE-DIFF.patch == candidate.comp')

    print('== 4. frozen read-only inputs')
    for label, (p, want) in FROZEN.items():
        got = sha(p)
        check(got == want, label, got[:16] + '…' if got != want else '')

    print('== 5. plan scope')
    text = (CARD / 'plan.txt').read_text(encoding='utf-8-sig')
    cand_path = (CARD / 'candidate.spv').as_posix()
    n_cand = text.count(cand_path)
    base_path = ('@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/'
                 'fullnet-mechanism-transfer-20261026/f05-kv-consumer-layout/family/c128/'
                 'candidate.spv')
    check(n_cand == 12, 'candidate path appears on 12 PASS lines', str(n_cand))
    check(text.count(base_path) == 0, 'no baseline F05 C128 path left')
    head = (C / 'accepted/n18-f01-entry-20261027/plan.txt').read_text(encoding='utf-8-sig')
    a, b = head.splitlines(), text.splitlines()
    pa = sum(1 for l in a if l.startswith('PASS '))
    pb = sum(1 for l in b if l.startswith('PASS '))
    check(pa == pb == 232, 'both plans have 232 PASS lines', '%d/%d' % (pa, pb))
    diff = [i for i, (x, y) in enumerate(zip(a, b)) if x != y]
    check(len(diff) == 12, 'exactly 12 differing lines', str(len(diff)))

    print()
    print('ALL CHECKS PASS' if ok else 'SOME CHECKS FAILED')
    return 0 if ok else 1


if __name__ == '__main__':
    sys.exit(main())
