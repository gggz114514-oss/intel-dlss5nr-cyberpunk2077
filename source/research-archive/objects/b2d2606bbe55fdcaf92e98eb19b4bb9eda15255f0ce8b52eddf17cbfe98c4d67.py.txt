"""Plan reader for the n23-f05-shape card (read-only).

Parses the campaign plan format (see clock_runner.cpp):
  PASS <row> "<name>" "<spv>" <subgroup> <shared> <nbind> <npush> <gx> <gy> <gz>
  BIND <binding> <resource> <offset> <bytes>
  PUSH <npush x uint32>
  BUFFER "<name>" <bytes> "<path>"        (resource index = order of appearance)
"""
from pathlib import Path
import re


def parse(plan_path):
    lines = Path(plan_path).read_text(encoding='utf-8-sig').splitlines()
    buffers, passes = [], []
    i, n = 0, len(lines)
    while i < n:
        l = lines[i]
        if l.startswith('BUFFER '):
            q = re.findall(r'"([^"]*)"', l)
            toks = l.split()
            buffers.append(dict(index=len(buffers), name=q[0], bytes=int(toks[2]), path=q[1], line=i))
        elif l.startswith('PASS '):
            q = re.findall(r'"([^"]*)"', l)
            toks = l.split()
            # some PASS rows quote name/spv, some do not (see CONTRACT 8.2: locate by row)
            name, spv = (q[0], q[1]) if len(q) == 2 else (toks[2], toks[3])
            p = dict(line=i, ordinal=len(passes), row=int(toks[1]), name=name, spv=spv,
                     subgroup=int(toks[4]), shared=int(toks[5]), nbind=int(toks[6]),
                     npush=int(toks[7]), gx=int(toks[8]), gy=int(toks[9]), gz=int(toks[10]),
                     binds=[], push=[])
            for k in range(p['nbind']):
                b = lines[i + 1 + k].split()
                assert b[0] == 'BIND', b
                p['binds'].append(dict(binding=int(b[1]), resource=int(b[2]),
                                       offset=int(b[3]), bytes=int(b[4])))
            if p['npush']:
                t = lines[i + 1 + p['nbind']].split()
                assert t[0] == 'PUSH', t
                vals = [int(x) for x in t[1:]]
                assert len(vals) == p['npush'], (len(vals), p['npush'])
                p['push'] = vals
            passes.append(p)
            i += 1 + p['nbind'] + (1 if p['npush'] else 0)
            continue
        i += 1
    return dict(buffers=buffers, passes=passes, lines=lines)


if __name__ == '__main__':
    import sys, json
    d = parse(sys.argv[1])
    print('buffers', len(d['buffers']), 'passes', len(d['passes']))
    for o in [19, 34]:
        p = d['passes'][o]
        print(json.dumps({k: p[k] for k in ('ordinal', 'row', 'name', 'gx', 'gy', 'gz', 'shared',
                                            'nbind', 'npush')}, indent=2))
