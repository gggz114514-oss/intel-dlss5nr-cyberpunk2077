"""Run a probe's ordinary timing with the probe arm's own frame hash as its expectation.

A probe deliberately changes numerics (it deletes work), so it cannot be gated against
the network gold. It is still timed against the same chain head in the same batch, and it
is never adopted. The probe arm's expected frame is its own deterministic output, taken
from the gate run that just produced it.
"""
import json, sys
from pathlib import Path
sys.path.insert(0, str(Path(__file__).parent))
import COMMON as K
import STAGE
import ANALYZE

name = sys.argv[1]
plan = Path(sys.argv[2])
base, base_sha, chain = K.current_best()
psha = K.sha(plan)
out = K.C / 'queue' / name
ref = None
for cand in out.rglob('FRAME-HASHES.json'):
    d = K.read_json(cand)
    if 'frame_out.bin' in d:
        ref = d['frame_out.bin']['sha256']
        break
assert ref, 'no probe frame hash found; run the probe gate first'
print('probe self-frame', ref)

run = out / 'run'
jobs = []
arms = [('base', base, base_sha, K.ORDINARY_GOLD), ('probe', plan, psha, ref)]
for pi in range(6):
    order = [arms[0], arms[1]] if pi % 2 == 0 else [arms[1], arms[0]]
    for aid, p, s, exp in order:
        j = K.ordinary_job('ord-%d-%s' % (pi, aid), p, s, run / ('ord2-%d-%s' % (pi, aid)), aid, pi)
        j['expected_frame_out'] = exp
        jobs.append(j)
spec = dict(stage='ordinary', owner='LUNA-CONTINUOUS-PROBE',
            authorization='Main (this session) authorized probe %s ordinary only; the probe arm is checked against its own frame, never adopted.' % name,
            output_root=str(out / 'ordinary2'), identities=[str(base), str(plan), str(K.RUNNER)],
            jobs=jobs, rules='probe timing only; not a candidate')
sp = out / 'spec-ordinary2.json'
K.write_json(sp, spec)
manifest, ps1 = STAGE.prepare(spec)
r = STAGE.run(manifest, ps1)
s = STAGE.summarize(spec['output_root'])
K.write_json(Path(spec['output_root']) / 'STAGE-SUMMARY.json', s)
print('ordinary', s['status'], [(j['id'], j['status'], j['frame']) for j in s['jobs']])
if s['status'] == 'COMPLETED':
    rep = ANALYZE.report(Path(spec['output_root']) / 'LUNA-ORDINARY.json',
                         [dict(arms=('base', 'probe'))], out / 'TIMING.json', extra=dict(probe=name))
    c = rep['comparisons'][0]
    print('PROBE %s: %s saving=%.6f CI95=[%.6f, %.6f] groups=%d' % (
        name, c['status'], c['saving_ms'], c['CI95_ms'][0], c['CI95_ms'][1], c['groups']))
    print('pairs', [(g['pair'], round(g['base'], 5), round(g['probe'], 5)) for g in c['group_means']])
