"""Shared ISA reader for n33-f02-latency (F02, kernel hash 7a38a17947f21ae6).

Identity (chain head accepted/n29-f01-alu-chain-20261027 == baseline for F02):
  F02 7 PASS ordinals 1,2,3,4,227,228,229 -> ONE spv
  c32-rest-20261026/body-pv-den-overlap/candidate.spv  sha 4c034a532ce60f5f...
  source candidate.comp                                  sha 75e3474dc274...
The same kernel hash appears in *every* campaign native batch (n01b at 2026-10-27
20:32 up to n31 at 2026-10-28 12:51); the two ISA files are byte-identical across
all of them (verified in _t01_identity.py), so this is one identity, one driver
batch of record: <CAP>.

File order == program order (GTPin --dump_isa asm5).  Labels `L<n>:` map to the
index of the next instruction.
"""
import re, hashlib
from pathlib import Path
from collections import Counter

C = Path(r"@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/fullnet-continuous-20261027")
CAP = C / "queue/n31-f01-dpas-chain/native/captures/native-native-base/ISA"
KERN = "7a38a17947f21ae6"
FILES = ["CS_asm7a38a17947f21ae6_simd32_0000000000000002_1_orig.asm",
         "CS_asm7a38a17947f21ae6_simd32_0000000000000003_2_orig.asm"]
ORDINALS = [1, 2, 3, 4, 227, 228, 229]
SPV = C.parent / "c32-rest-20261026/body-pv-den-overlap/candidate.spv"

LINE = re.compile(r"^/\* \[([0-9A-F]+)\]\s+\*/\s+(.*)$")
LBL = re.compile(r"^L(\d+):")
DEPS = re.compile(r"\{(?P<deps>[^}]*)\}\s*$")
COMMENT = re.compile(r"//\s*(?P<c>.*)$")


def path(name=None):
    return CAP / (name or FILES[0])


def parse(name=None):
    """-> list of dict(idx, addr, flags, op, ops, deps, comment, text)."""
    out, labels = [], {}
    for raw in path(name).read_text(encoding="utf-8", errors="replace").splitlines():
        s = raw.strip()
        m = LBL.match(s)
        if m:
            labels[int(m.group(1))] = len(out)
            continue
        m = LINE.match(raw)
        if not m:
            continue
        addr = int(m.group(1), 16)
        rest = m.group(2)
        deps = None
        dm = DEPS.search(rest)
        if dm:
            deps = dm.group("deps").strip()
            rest = rest[:dm.start()]
        comment = None
        cm = COMMENT.search(rest)
        if cm:
            comment = cm.group("c").strip()
            rest = rest[:cm.start()]
        toks = rest.split()
        if not toks:
            continue
        flags = ""
        if toks[0].startswith("(") and toks[0].endswith(")"):
            flags = toks[0]
            toks = toks[1:]
        if not toks:
            continue
        ops = [t for t in toks[1:] if not (t.startswith("(") and t.endswith(")"))]
        out.append(dict(idx=len(out), addr=addr, flags=flags, op=toks[0], ops=ops,
                        deps=deps, comment=comment,
                        text=" ".join(toks)))
    return out, labels


def sha(p):
    return hashlib.sha256(Path(p).read_bytes()).hexdigest()


if __name__ == "__main__":
    for f in FILES:
        ins, labels = parse(f)
        print("%s: %d instructions, %d labels, %d bytes, sha %s"
              % (f, len(ins), len(labels), path(f).stat().st_size, sha(path(f))[:16]))
        print("   ops:", dict(Counter(i["op"] for i in ins).most_common(12)))
