"""Static loop/barrier accounting for the parent and candidate SPIR-V.

Only SPIR-V structure is read: which natural loop contains which OpControlBarrier.
"""
from pathlib import Path
import sys, json
W = Path('@NR_WORKSPACE@')
sys.path.insert(0, str(W / 'build/zcode-fp16-compute-backend-20261021/scripts'))
import spv_tools_local as ST

S = Path(__file__).resolve().parent
OP_LABEL, OP_LOOP_MERGE, OP_BARRIER, OP_BRANCH = 248, 246, 224, 249

out = {}
for name in ('parent-rebuild.spv', 'candidate.spv'):
    m = ST.Module(S / name)
    ins = m.instructions
    loops = []

    def label_index(lbl):
        for j, (o2, a2, _, _) in enumerate(ins):
            if o2 == OP_LABEL and a2 and a2[0] == lbl:
                return j
        return None

    for i, (op, a, _, _) in enumerate(ins):
        if op == OP_LOOP_MERGE:
            merge = a[0]
            cont = a[1]
            # loop head = the OpLabel of the block that holds this OpLoopMerge
            head_i = None
            for j in range(i - 1, -1, -1):
                if ins[j][0] == OP_LABEL:
                    head_i = j
                    break
            mg = label_index(merge)
            span = [j for j in range(head_i, mg) if ins[j][0] == OP_BARRIER] if mg else []
            loops.append(dict(head_word='0x%X' % (ins[head_i][2] * 4),
                              body_instructions=mg - head_i if mg else None,
                              barriers_in_loop=len(span),
                              barrier_words=['0x%X' % (ins[j][2] * 4) for j in span]))
    loops.sort(key=lambda d: -(d['body_instructions'] or 0))
    out[name] = dict(total_barriers=sum(1 for op, _, _, _ in ins if op == OP_BARRIER),
                     loops=loops[:4])
print(json.dumps(out, indent=1))
