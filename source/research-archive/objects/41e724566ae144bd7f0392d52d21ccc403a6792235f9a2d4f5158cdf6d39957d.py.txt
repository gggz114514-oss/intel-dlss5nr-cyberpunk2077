"""Emit evidence/GROUPS.json + GROUPS.md with BOTH the baseline and the chain-head identity.

Baseline identity comes from CURRENT-232-ROWS; the head identity comes from the chain head's
own PASS lines. They differ for every adopted group (e.g. F05 C256 dispatches n05a's
7a9576d1… at the head, not the baseline 9d61c4e3…). Cards must take their parent from the
HEAD column, otherwise the parent-byte reproduction targets the wrong shader.
"""
import sys
from collections import defaultdict
from pathlib import Path
sys.path.insert(0, str(Path(__file__).parent))
import COMMON as K
from audit_card import pass_lines

rows = K.rows()
cur = K.read_json(K.C / 'CURRENT.json')
head_recs = pass_lines(Path(cur['best_plan']['path']))

base = defaultdict(lambda: defaultdict(list))
for r in rows:
    base[r['family']][r['spv_sha256']].append(r)

out = {'note': 'family -> baseline shader sha -> rows, plus the sha actually dispatched at the chain head',
       'head': cur['best_plan']['sha256'], 'families': {}}
lines = ['# 族 / shader 组 / ordinal 权威表', '',
         '> **基线身份**来自 CURRENT-232-ROWS；**链头身份**来自当前 best_plan 的 PASS 行。',
         '> 已采纳的组两者不同（例：F05 C256 在链头派发 n05a 的 `7a9576d1…`，不是基线的 `9d61c4e3…`）。',
         '> **卡片必须以"链头 sha"为父**，否则父字节复现会指向错 shader。', '',
         '| 族 | 基线 sha | PASS | ordinal | 链头 sha | 代表 PASS 名 |', '|---|---|---|---|---|---|']
for fam in sorted(base):
    out['families'][fam] = {}
    for sha, rs in sorted(base[fam].items(), key=lambda kv: -len(kv[1])):
        ords = sorted(r['ordinal'] for r in rs)
        head_spvs = {Path(head_recs[o]['spv']).as_posix() for o in ords}
        h0 = Path(head_recs[ords[0]]['spv'])
        head_sha12 = K.sha(h0)[:12] if h0.exists() else '?'
        changed = '*' if len(head_spvs) == 1 and h0.as_posix() != rs[0]['spv'].replace('\\', '/') else ''
        out['families'][fam][sha] = dict(count=len(rs), ordinals=ords, name=rs[0]['name'],
                                         baseline_spv=rs[0]['spv'],
                                         head_spv=sorted(head_spvs)[0] if len(head_spvs) == 1 else sorted(head_spvs),
                                         head_sha12=head_sha12, changed=bool(changed))
        lines.append('| %s | `%s` | %d | %s | `%s`%s | %s |' % (
            fam, sha[:12], len(rs), ', '.join(str(o) for o in ords), head_sha12, changed, rs[0]['name']))
K.write_json(K.C / 'evidence' / 'GROUPS.json', out)
(K.C / 'GROUPS.md').write_text('\n'.join(lines) + '\n', encoding='utf-8')
print('families', len(out['families']))
for fam in sorted(out['families']):
    for sha, e in out['families'][fam].items():
        if e['changed']:
            print('  %-4s %2d rows  base %s -> head %s' % (fam, e['count'], sha[:8], e['head_sha12']))
