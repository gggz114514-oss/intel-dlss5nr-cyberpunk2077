out = {'status':'CPU_VERIFIED_BOUNDARIES_NOT_TIMING_RESULTS',
 'semantics':'Before-start to before-end native arrival window. Labels include operand preparation, publication and waits where named. No conversion to exclusive network milliseconds.',
 'calibration_required':True, 'families':{}}
for family,(identity,regions) in definitions.items():
 m=next(m for m in modules if m['native_hash']==identity)
 p=Path(m['native']['path']); data=p.read_bytes()
 assert hashlib.sha256(data).hexdigest()==m['native']['sha256']
 ins={int(match[1],16):line for line in data.decode().splitlines() if (match:=re.search(r'/\* \[([0-9A-Fa-f]+)\]',line))}
 rows=[]
 for a,b,name in regions:
  assert a in ins and b in ins,(family,hex(a),hex(b))
  rows.append({'start_pc':a,'end_pc':b,'start_hex':hex(a),'end_hex':hex(b),'meaning':name,'start_instruction':ins[a],'end_instruction':ins[b]})
 assert all(a['end_pc']==b['start_pc'] for a,b in zip(rows,rows[1:]))
 out['families'][family]={'native':m['native'],'native_hash':identity,'orders':m['orders'],'regions':rows}
(D/'NATIVE-PHASE-BOUNDARIES.json').write_text(json.dumps(out,indent=2,ensure_ascii=False),encoding='utf-8')
