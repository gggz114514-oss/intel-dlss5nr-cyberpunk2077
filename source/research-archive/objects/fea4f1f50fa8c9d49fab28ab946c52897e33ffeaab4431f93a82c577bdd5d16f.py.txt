import datetime, json, time
from pathlib import Path

cfg = Path(r"@NR_WORKSPACE@\work\github-experimental-game-install-20261024\package\config")
out = Path(__file__).with_name("candidate-stationary-30s.jsonl")
if out.exists():
    raise SystemExit(f"output already exists: {out}")
start = time.perf_counter()
with out.open("x", encoding="utf-8", newline="\n") as f:
    for i in range(61):
        target = start + i * 0.5
        now = time.perf_counter()
        if now < target:
            time.sleep(target - now)
        elapsed = time.perf_counter() - start
        record = {
            "timestamp_utc": datetime.datetime.now(datetime.timezone.utc).isoformat(),
            "elapsed_sec": round(elapsed, 3),
            "settling": elapsed < 5.0,
        }
        try:
            component = json.loads((cfg / "component_status.json").read_text(encoding="utf-8"))
            panel = json.loads((cfg / "panel_state.json").read_text(encoding="utf-8"))
            stage = panel.get("stage") or {}
            handoff = panel.get("handoff") or {}
            record["component"] = component
            record["panel"] = panel
            record["stage_ms"] = {k: stage.get(k) for k in ("prep", "host", "composite", "tail", "total", "last")}
            record["handoff"] = {k: handoff.get(k) for k in ("bypasses", "cpu_waits", "motion_waits")}
            record["valid_context"] = (
                component.get("build_id") == "optimized-full-gpu-handoff-v1"
                and component.get("init_state") == 2
                and component.get("gpu_timeout") == 0
                and component.get("fallback") == 0
                and panel.get("backend") == "small"
                and panel.get("mounted") == 1
                and (handoff.get("bypasses") or 0) > 0
            )
        except Exception as exc:
            record["read_error"] = repr(exc)
            record["valid_context"] = False
        f.write(json.dumps(record, separators=(",", ":")) + "\n")
        f.flush()
print(json.dumps({"status": "COMPLETE", "records": 61, "settling_records": 10, "output": str(out)}))
