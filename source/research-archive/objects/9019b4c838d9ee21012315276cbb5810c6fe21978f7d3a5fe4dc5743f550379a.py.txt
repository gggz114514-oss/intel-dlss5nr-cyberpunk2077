def sha(p):
    h = hashlib.sha256()
    with open(p, 'rb') as f:
        for b in iter(lambda: f.read(1 << 20), b''):
            h.update(b)
    return h.hexdigest()


def rel(p):
    try:
        return os.path.relpath(p, WS).replace('\\', '/')
    except ValueError:
        return p.replace('\\', '/')


def md_link(p, label=None):
    return '[%s](<%s>)' % (label or os.path.basename(p), rel(p))


# ---------------------------------------------------------------- plan parse
def parse_plan():
    lines = open(PLAN, encoding='utf-8-sig').read().splitlines()
    passes = []
    for i, line in enumerate(lines):
        if not line.startswith('PASS '):
            continue
        body = line[5:]
        m = re.match(r'^(\S+)\s+(?:"([^"]*)"|(\S+))\s+(?:"([^"]*)"|(\S+))\s*(.*)$', body)
        assert m, 'unparsable PASS line %d' % (i + 1)
        passes.append(dict(
            line=i + 1, plan_index=m.group(1), name=m.group(2) or m.group(3),
            spv=m.group(4) or m.group(5), quoted=bool(m.group(2)),
            fields=(m.group(6) or '').split()))
    # structural check: BIND count declared in fields[2]; one PUSH after the binds
    for n, p in enumerate(passes):
        end = passes[n + 1]['line'] - 1 if n + 1 < len(passes) else len(lines)
        binds = [l for l in lines[p['line']:end] if l.startswith('BIND ')]
        push = [l for l in lines[p['line']:end] if l.startswith('PUSH ')]
        p['binds_declared'] = int(p['fields'][2])
        p['binds_actual'] = len(binds)
        p['push_actual'] = len(push)
        p['push_values'] = push[0][5:].split() if push else []
        assert len(p['fields']) == 7, p
        assert p['binds_declared'] == p['binds_actual'] == int(p['fields'][2])
        assert p['push_actual'] == 1
    return passes

def main():
    passes = parse_plan()
    rows = json.load(open(ROWS_P, encoding='utf-8'))
    assert len(passes) == 232 and len(rows) == 232

    # numbering facts
    idx_counts = collections.Counter(p['plan_index'] for p in passes)
    dup_idx = {k: v for k, v in idx_counts.items() if v > 1}
    idx_to_names = collections.defaultdict(list)
    for p in passes:
        idx_to_names[p['plan_index']].append(p['name'])

    # measured hashes
    pair = {}
    for r in rows:
        for kind in ('spv', 'source'):
            p = r[kind]
            k = (kind, p.lower())
            if k not in pair:
                pair[k] = (sha(p) if os.path.exists(p) else None)

    # identity evidence
    idn = json.load(open(ID232_P, encoding='utf-8'))
    id_by_sha = {}
    for s in idn['shaders']:
        key = None
        for r in rows:
            if r['name'] == s['name'] and r['family'] == s['family']:
                key = r['spv_sha256']
                break
        id_by_sha[key] = s
    recipe = json.load(open(RECIPE_P, encoding='utf-8'))
    recipe_by_sha = recipe['shaders']

    # census staleness (documented, never used as identity)
    census = json.load(open(CENSUS_P, encoding='utf-8'))
    crow = {x['ordinal']: x for x in census['rows']}
    census_spv_diff = [o for o in range(232)
                       if crow[o]['spv'].replace('\\', '/').lower() != rows[o]['spv'].replace('\\', '/').lower()]
    census_sha_diff = [o for o in range(232)
                       if (crow[o].get('sha256') or '').lower() != rows[o]['spv_sha256'].lower()]

    out_rows = []
    checks = dict(
        rows=len(rows), unique_spv=len({r['spv'].lower() for r in rows}),
        unique_source=len({r['source'].lower() for r in rows}),
        plan_pass_lines=len(passes),
        pass_label_mismatch=[], name_mismatch=[], spv_mismatch=[], grid_mismatch=[],
        spv_sha_mismatch=[], source_sha_mismatch=[], spv_missing=[], source_missing=[],
        binds_mismatch=[], push_mismatch=[],
    )

    for n, (p, r) in enumerate(zip(passes, rows)):
        o = r['ordinal']
        spv_m = pair[('spv', r['spv'].lower())]
        src_m = pair[('source', r['source'].lower())]
        if not os.path.exists(r['spv']):
            checks['spv_missing'].append(o)
        if not os.path.exists(r['source']):
            checks['source_missing'].append(o)
        if spv_m != r['spv_sha256']:
            checks['spv_sha_mismatch'].append(o)
        if src_m != r['source_sha256']:
            checks['source_sha_mismatch'].append(o)
        if p['plan_index'] != r['pass_label']:
            checks['pass_label_mismatch'].append(o)
        if p['name'] != r['name']:
            checks['name_mismatch'].append(o)
        if p['spv'].replace('\\', '/').lower() != r['spv'].replace('\\', '/').lower():
            checks['spv_mismatch'].append(o)
        if p['fields'][4:7] != list(r['grid']):
            checks['grid_mismatch'].append(o)
        if p['binds_declared'] != p['binds_actual']:
            checks['binds_mismatch'].append(o)
        if p['push_actual'] != 1:
            checks['push_mismatch'].append(o)

        # width from the measured SPV path / name, never from ordinal arithmetic
        low = r['spv'].lower()
        if r['family'] in ('F04', 'F05', 'F06'):
            if '\\c64\\' in low or low.endswith('\\c64\\candidate.spv'):
                width = 'C64'
            elif '\\c128\\' in low or low.endswith('\\c128\\candidate.spv'):
                width = 'C128'
            elif '\\c256\\' in low:
                width = 'C256'
            else:
                width = 'UNKNOWN'
        elif r['family'] == 'F07':
            width = 'C64' if 'c64' in r['name'] else ('C128' if 'c128' in r['name'] else 'C256')
        elif r['family'] in ('F08', 'F09', 'F10', 'F11', 'F12'):
            width = 'C512'
        elif r['family'] in ('F13', 'F14', 'F15', 'F16'):
            width = 'ViT1024'
        elif r['family'] == 'F18':
            width = 'UPS-C256' if 'c256' in low else ('UPS-C128' if 'c128' in low else 'UPS-C64')
        elif r['family'] in ('F01', 'F21'):
            width = '图像'
        elif r['family'] == 'F17':
            width = 'UPS'
        else:
            width = 'C32'

        fam = FAM[r['family']]
        card, secondary = NMAP[r['family']]
        if r['family'] == 'F05':
            card = 'N05' if width == 'C256' else 'N04'
            secondary = []
        status = fam['st']
        basis = fam['basis']
        ev_level, ev_scope, neg_keys = fam['ev'], fam['scope'], list(fam['neg'])
        # F13 splits by site: contract 8 派发被本轮测过（负），expand 8 派发本轮未动
        if r['family'] == 'F13' and '_ffn_expand' in r['name']:
            status, ev_level, ev_scope, neg_keys = '未动', '无', '无', []
            basis = ('本轮 vit-epilogue-gain-reuse 只替换 contract 8 派发，expand 8 派发字节未变；'
                     '当前字节为 dsh-fp16-same-family-20261023/s32-k11-v1 历史资产')
        # F18 body_2 rows carry the accepted coefficient-reuse shader
        if r['family'] == 'F18' and r['name'].endswith('swin_body_2'):
            status = '现役接受'
            basis = ('当前字节来自 coefficient-reuse-combined 的 contract-scale-fragments/family/%s'
                     '（b39fae00，combined FASTER_95 +0.072084ms），UPS contract 三宽 send -14/-30/-62；'
                     '同族 c3x pre/body 6 行仍属 ups-* 负结果族' % width.replace('UPS-', '').lower())

        sid = id_by_sha.get(r['spv_sha256'])
        if sid and sid.get('raw_match'):
            rebuild, rerr = 'BYTES_REPRODUCED_RAW', None
        elif sid and sid.get('decorated_match'):
            rebuild, rerr = 'BYTES_REPRODUCED_DECORATED', None
        else:
            rebuild, rerr = 'NOT_REPRODUCED', ((sid or {}).get('error') or 'unknown')[:160]

        rec = recipe_by_sha.get(r['spv_sha256']) or {}
        acc = ACCEPTED_ROUNDS[fam['acc']] if fam['acc'] else None

        out_rows.append(dict(
            ordinal=o,
            plan_index=p['plan_index'],
            plan_line=p['line'],
            plan_index_multiplicity=idx_counts[p['plan_index']],
            plan_index_siblings=idx_to_names[p['plan_index']] if idx_counts[p['plan_index']] > 1 else [],
            name=r['name'], family=r['family'], width=width, width_basis=fam['wb'],
            grid=list(r['grid']), pass_header_fields=list(r['header_fields']),
            pass_block=dict(binds=p['binds_actual'], push_values=p['push_values'],
                            push_value_count=len(p['push_values'])),
            spv=dict(path=r['spv'], sha256_declared=r['spv_sha256'], sha256_measured=spv_m,
                     match=spv_m == r['spv_sha256'], exists=os.path.exists(r['spv'])),
            source=dict(path=r['source'], sha256_declared=r['source_sha256'], sha256_measured=src_m,
                        match=src_m == r['source_sha256'], exists=os.path.exists(r['source'])),
            identity=dict(rebuild=rebuild, recipe_recorded=bool(rec.get('recipes')),
                          recipe_count=len(rec.get('recipes') or []),
                          rebuild_error=rerr),
            card=card, cards_secondary=secondary,
            status=status, status_basis=basis,
            evidence_level=ev_level, evidence_scope=ev_scope,
            accepted_round=(fam['acc'] if status == '现役接受' and fam['acc'] else None),
            negative_keys=neg_keys,
            history_card_dir=os.path.dirname(r['spv']),
            next_action=fam['next'],
        ))

    # ---- family rollup + self check
    fam_rows = collections.defaultdict(list)
    for x in out_rows:
        fam_rows[x['family']].append(x)
    fam_order = ['F%02d' % i for i in range(1, 22)]
    WORDER = {'图像': 0, 'C32': 1, 'C64': 2, 'C128': 3, 'C256': 4, 'C512': 5,
              'ViT1024': 6, 'UPS': 7, 'UPS-C64': 8, 'UPS-C128': 9, 'UPS-C256': 10}
    rollup = []
    for f in fam_order:
        rs = fam_rows[f]
        meth = collections.Counter(x['status'] for x in rs)
        ev = collections.Counter(x['evidence_level'] for x in rs)
        widths = sorted({x['width'] for x in rs}, key=lambda w: WORDER.get(w, 99))
        rollup.append(dict(family=f, pass_count=len(rs), widths=widths,
                           card=NMAP[f][0], status_counts=dict(meth),
                           evidence_counts=dict(ev),
                           accepted_via=(FAM[f]['acc'] if FAM[f]['acc'] else None),
                           negatives=FAM[f]['neg'],
                           forbidden_rerun=FORBID_RERUN.get(f, [])))
    total = sum(x['pass_count'] for x in rollup)
    coverage_txt = {}
    for line in open(COVERAGE_TXT, encoding='utf-8'):
        m = re.match(r'^(F\d\d): (\d+) PASS', line.strip())
        if m:
            coverage_txt[m.group(1)] = int(m.group(2))
    etalon_ok = all(coverage_txt.get(x['family']) == x['pass_count'] for x in rollup)
    coverage_total_ok = sum(coverage_txt.values()) == total == 232

    checks['family_sum'] = total
    checks['coverage_txt_per_family_match'] = etalon_ok
    checks['coverage_txt_total'] = sum(coverage_txt.values())
    checks['ordinals_contiguous_0_231'] = [x['ordinal'] for x in out_rows] == list(range(232))
    checks['plan_index_distinct'] = len(idx_counts)
    checks['plan_index_multiplicity_gt1'] = {k: v for k, v in sorted(dup_idx.items(), key=lambda kv: int(kv[0]))}
    checks['plan_index_max_multiplicity'] = max(idx_counts.values())
    checks['identity_rebuild_counts'] = dict(collections.Counter(x['identity']['rebuild'] for x in out_rows))

def _ranges(nums):
    nums = sorted(nums)
    out, i = [], 0
    while i < len(nums):
        j = i
        while j + 1 < len(nums) and nums[j + 1] == nums[j] + 1:
            j += 1
        out.append(str(nums[i]) if j == i else '%d-%d' % (nums[i], nums[j]))
        i = j + 1
    return ','.join(out)
