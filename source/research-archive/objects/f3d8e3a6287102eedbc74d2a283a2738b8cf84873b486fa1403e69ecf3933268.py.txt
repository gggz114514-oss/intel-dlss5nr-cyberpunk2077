"""Optional control arm for n08b-f08-order: PARENT content (2 tiles x 4 groups)
with the dispatch axes swapped, grid (2,66,1).

Only writes inside this card directory (control/).  No GPU, no deployment.
"""
import sys, json
from pathlib import Path

CAMPAIGN = Path(r'@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/fullnet-continuous-20261027')
sys.path.insert(0, str(CAMPAIGN))
import COMMON as K
CAMPAIGN_CARD = CAMPAIGN / 'cards' / 'n08b-f08-order'
sys.path.insert(0, str(CAMPAIGN_CARD))
import build as B          # reuse normalize/spirv_diff/plan helpers

CARD = CAMPAIGN_CARD / 'control'
CARD.mkdir(exist_ok=True)
PARENT_SPV = B.PARENT_SPV
PARENT_GRID = ('66', '2', '1')
CTL_GRID = ('2', '66', '1')
sha = K.sha


def build_plan(parent_plan, spv_rel):
    raw = Path(parent_plan).read_bytes()
    bom = raw.startswith(b'\xef\xbb\xbf')
    lines = raw.decode('utf-8-sig').split('\n')
    parent_posix = PARENT_SPV.as_posix()
    new_posix = Path(spv_rel).as_posix()
    changed = 0
    for i, line in enumerate(lines):
        cr = '\r' if line.endswith('\r') else ''
        body = line[:-1] if cr else line
        if not body.startswith('PASS ') or parent_posix not in body:
            continue
        parts = body.split(' ')
        assert tuple(parts[-3:]) == PARENT_GRID and parts[3] == parent_posix
        parts[-3:] = list(CTL_GRID)
        parts[3] = new_posix
        lines[i] = ' '.join(parts) + cr
        changed += 1
    assert changed == 16, changed
    out = CARD / 'plan.txt'
    out.write_bytes((('\ufeff' if bom else '') + '\n'.join(lines)).encode('utf-8'))
    return out, changed


def main():
    assert sha(PARENT_SPV) == B.PARENT_SPV_SHA
    chain_plan, chain_sha, _ = K.current_best()
    assert sha(chain_plan) == chain_sha

    ctl_spv = CARD / 'candidate-control.spv'
    K.compile_glsl(CAMPAIGN_CARD / 'control-candidate.comp', ctl_spv, B.INC_PRIMARY)
    K.validate_spv(ctl_spv)

    diff = B.spirv_diff(PARENT_SPV, ctl_spv, 'control - parent')
    pm = K.ST.Module(PARENT_SPV)
    cm = K.ST.Module(ctl_spv)
    analysis = B.analyze_vs_parent(pm, cm)

    plan, changed = build_plan(chain_plan, ctl_spv)
    cpu = K.cpu_check(plan)
    base_lines = Path(chain_plan).read_text(encoding='utf-8-sig').split('\n')
    ctl_lines = Path(plan).read_text(encoding='utf-8-sig').split('\n')
    plan_diff = [{'line': i + 1, 'parent': a, 'candidate': b}
                 for i, (a, b) in enumerate(zip(base_lines, ctl_lines)) if a != b]

    out = {
        'card': 'n08b-f08-order',
        'role': 'optional control arm (not the primary candidate)',
        'content': 'PARENT content: 2 token8 tiles x 4 weight groups per workgroup',
        'order': 'transposed: token block on y (66), group pair on x (2)',
        'source': K.ident(CAMPAIGN_CARD / 'control-candidate.comp'),
        'spv': K.ident(ctl_spv),
        'spv_validate_cpu': 'OK',
        'parent_spv': K.ident(PARENT_SPV),
        'plan': {'path': str(plan), 'sha256': sha(plan), 'built_on': str(chain_plan),
                 'passes_replaced': changed, 'parent_lines_changed': plan_diff},
        'cpu_check': cpu,
        'spirv_vs_parent': diff,
        'numerics': {k: analysis[k] for k in ('no_contraction_parent', 'no_contraction_candidate',
                                              'no_contraction_ids_equal', 'float_NC_checked',
                                              'float_sites', 'opcode_histogram_delta',
                                              'ext_inst_fma_signatures_equal')},
    }
    K.write_json(CARD / 'BUILD-CONTROL.json', out)
    print(json.dumps({'ctl_spv_sha': sha(ctl_spv), 'plan_sha': sha(plan),
                      'cpu': cpu.strip(), 'spv_validate_cpu': 'OK',
                      'diff_count': diff['differing_instructions'],
                      'diff': diff['differences'],
                      'float_NC_checked': analysis['float_NC_checked'],
                      'nc': (analysis['no_contraction_parent'], analysis['no_contraction_candidate']),
                      'opcode_delta': analysis['opcode_histogram_delta'],
                      'plan_lines_changed': len(plan_diff)}, indent=1))


if __name__ == '__main__':
    main()
