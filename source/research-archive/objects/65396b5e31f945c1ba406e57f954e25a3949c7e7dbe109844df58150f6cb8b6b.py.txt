def run(argv, allow_fail=False):
    r = subprocess.run(list(map(str, argv)), capture_output=True, text=True,
                       encoding='utf-8', errors='replace')
    if r.returncode != 0 and not allow_fail:
        raise RuntimeError('FAILED %s\n%s\n%s' % (argv, r.stdout, r.stderr))
    return r.returncode, (r.stdout or '') + (r.stderr or '')


def compile_glsl(comp, out):
    return run([GLSLANG, '-V', '--target-env', 'vulkan1.3', '-I' + str(SITE), comp, '-o', out])

# (b) NC placement: pair by position over the parent's NC opcode set.
def classify(m, ids):
    out = {}
    for op, a, _, _ in m.instructions:
        if ST.result_id_of(op, a) in ids:
            out[op] = out.get(op, 0) + 1
    return dict(sorted(out.items()))


def op_seq(m, opcodes):
    out = []
    for fi, reg in enumerate(ST.function_regions(m)):
        for i in reg:
            op = m.instructions[i][0]
            if op in opcodes:
                out.append((fi, op, ST.result_id_of(op, m.instructions[i][1])))
    return out

def nc_multiset(m, ids):
    seen, out = {}, []
    for op, a, _, _ in m.instructions:
        rid = ST.result_id_of(op, a)
        if rid in ids:
            key = (op, str(m.operand_signature(rid, 3)))
            seen[key] = seen.get(key, 0) + 1
            out.append((key, seen[key]))
    return sorted(out)

# ------------------------------------------- mapping / lifetime / barrier proof
# (a) staged layout is untouched: element (r,k) of a slice still lands at
#     qkv_a_offset(r*64+k); adding the window base is a pure translation.
def a_off(e):
    r, k = e // 64, e % 64
    return ((r // 8) * 4 + k // 16) * 128 + (k % 16) * 8 + r % 8

cells = {}
for r in range(64):
    for k in range(64):
        addr = a_off(r * 64 + k)
        assert addr not in cells
        cells[addr] = (r, k)
assert set(cells) == set(range(4096)), 'staged slice is not a bijection'
fragment_ok = 0
for sg in range(8):
    for abase in (0, 4096):
        for m in ((sg // 2) * 16, (sg // 2) * 16 + 8):
            for k0 in range(0, 64, 16):
                base = abase + ((m // 8) * 4 + k0 // 16) * 128
                for r in range(8):
                    for c in range(16):
                        assert cells[base - abase + c * 8 + r] == (m + r, k0 + c)
                        fragment_ok += 1

# (b) slice -> window map and the barrier that retires each window use
slices = []
window = 4096
for i in range(8):
    slices.append(dict(index=i, kc=64 * i, window=window))
    window = 4096 - window
uses = {}
for s in slices:
    uses.setdefault(s['window'], []).append(s['index'])
retire = {}
for w, idxs in uses.items():
    retire[w] = []
    for a, b in zip(idxs, idxs[1:]):
        # the publish barrier of slice a+1 sits between the two uses of w
        assert b - a >= 2, (w, a, b)
        retire[w].append(dict(from_slice=a, to_slice=b, barrier_of_slice=a + 1))
last_window = slices[-1]['window']
barrier_check = dict(
    slice_window=[dict(index=s['index'], kc=s['kc'], base=s['window']) for s in slices],
    window_uses={str(k): v for k, v in uses.items()},
    separation={str(k): v for k, v in retire.items()},
    last_slice_window=last_window,
    handoff_window=4096,
    handoff_store_barrier=None,
    note='odd slices stage into [0,4096); the last slice is odd, so the QKV handoff '
         'stores in [4096,10240) are ordered behind the slice-7 publish barrier and '
         'behind the existing post-store barrier only.')


# (c) address hoist: tiled(token, 64i+c0, 512) == tiled(token, c0, 512) + 1024*i
def tiled(token, c, C=512):
    return ((token // 16) * (C // 16) + c // 16) * 256 + (token % 16) * 16 + c % 16


hoist_checks = 0
for token in range(1024):
    base_c0 = [tiled(token, c0) for c0 in range(64)]
    for i in range(8):
        for c0 in range(64):
            ref = tiled(token, 64 * i + c0)
            assert base_c0[c0] + 1024 * i == ref, (token, i, c0, ref, base_c0[c0] + 1024 * i)
            hoist_checks += 1
# the same identity has to hold for the four quadrant token expressions, which only
# differ in `token`; the loop above is therefore exhaustive over the formula.
max_invalid_addr = 1024 * 7 + 192
input_halves = 1179648 // 2
assert max_invalid_addr < input_halves
