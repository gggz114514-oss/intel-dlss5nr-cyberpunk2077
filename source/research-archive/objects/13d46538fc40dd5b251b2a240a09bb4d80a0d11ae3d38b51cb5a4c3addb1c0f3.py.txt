"""SPIR-V census / opcode diff for n11b-vit-projection-hoist (CPU only, no GPU).

Parent: candidates/.../u03-f18-v2/shader.spv (ae5cb966..)
Candidate: cards/n11b-vit-projection-hoist/candidate.spv (55e04be5..)

Prints:
  * module word counts, NoContraction id count (both modules)
  * full opcode histogram diff (parent -> candidate)
  * control-flow instruction list (labels/branches/merges/phis/calls)
  * FP-opcode census that the contract's NoContraction rule cares about
"""
import sys, json, collections
from pathlib import Path
sys.path.insert(0, r'@NR_WORKSPACE@/build/zcode-fp16-compute-backend-20261021/scripts')
import spv_tools_local as ST

PARENT = Path(r'@NR_WORKSPACE@/candidates/codex-fp16-continuation-20261021/A/u03-f18-v2/shader.spv')
CARD = Path(r'@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/fullnet-continuous-20261027/cards/n11b-vit-projection-hoist')
CAND = CARD / 'candidate.spv'

# SPIR-V opcode names we care about (core + the ones this shader uses)
NAMES = {0: 'OpNop', 19: 'OpTypeVoid', 22: 'OpTypeFunction', 33: 'OpTypePointer', 43: 'OpConstant',
         59: 'OpVariable', 61: 'OpLoad', 62: 'OpStore', 65: 'OpAccessChain', 68: 'OpDecorate',
         71: 'OpDecorateId', 79: 'OpString', 128: 'OpIAdd', 130: 'OpIMul', 132: 'OpUDiv',
         134: 'OpUMod', 135: 'OpShiftRightLogical', 137: 'OpShiftLeftLogical',
         138: 'OpBitwiseOr', 139: 'OpBitwiseAnd', 141: 'OpNot', 143: 'OpBitCount',
         177: 'OpUConvert', 124: 'OpFNegate', 126: 'OpFDiv', 127: 'OpFRem', 128: 'OpIAdd',
         129: 'OpFAdd', 130: 'OpIMul', 131: 'OpFMul', 132: 'OpUDiv', 133: 'OpFDiv',
         134: 'OpUMod', 135: 'OpSRem', 136: 'OpSMod', 137: 'OpFRem', 138: 'OpFMod',
         139: 'OpVectorTimesScalar', 140: 'OpMatrixTimesScalar', 141: 'OpVectorTimesMatrix',
         142: 'OpMatrixTimesVector', 143: 'OpMatrixTimesMatrix', 144: 'OpOuterProduct',
         145: 'OpDot', 146: 'OpIAddCarry', 147: 'OpISubBorrow', 148: 'OpUMulExtended',
         149: 'OpSMulExtended', 128: 'OpIAdd', 245: 'OpPhi', 246: 'OpLoopMerge',
         247: 'OpSelectionMerge', 248: 'OpLabel', 249: 'OpBranch', 250: 'OpBranchConditional',
         251: 'OpSwitch', 252: 'OpKill', 253: 'OpReturn', 254: 'OpReturnValue',
         255: 'OpUnreachable', 57: 'OpFunction', 54: 'OpFunctionEnd', 56: 'OpFunctionCall',
         58: 'OpFunctionParameter', 60: 'OpExtInst', 81: 'OpName', 86: 'OpMemberName'}

FP = {124: 'OpFNegate', 126: 'OpFDiv', 127: 'OpFRem', 129: 'OpFAdd', 131: 'OpFMul'}
CONV = {109: 'OpBitcast', 110: 'OpSNegate', 111: 'OpFNegate', 112: 'OpIAdd',
        113: 'OpFAdd', 114: 'OpISub', 115: 'OpFSub', 116: 'OpIMul', 117: 'OpFMul',
        124: 'OpFNegate'}
CF = {245: 'OpPhi', 246: 'OpLoopMerge', 247: 'OpSelectionMerge', 248: 'OpLabel',
      249: 'OpBranch', 250: 'OpBranchConditional', 251: 'OpSwitch', 253: 'OpReturn',
      254: 'OpReturnValue', 255: 'OpUnreachable', 56: 'OpFunctionCall', 57: 'OpFunction',
      54: 'OpFunctionEnd'}


def hist(mod):
    h = collections.Counter()
    for ins in mod.instructions:
        h[ins[0]] += 1
    return h


def census(path):
    m = ST.Module(Path(path))
    h = hist(m)
    nc = ST.no_contraction_ids(m)
    exts = collections.Counter()
    for ins in m.instructions:
        # instruction = (opcode, [operands...], result_id, word_count)
        if ins[0] == 12:  # OpExtInst: operands = [result_type, result_id, set, literal...]
            exts[ins[1][3]] += 1
    return m, h, nc, exts


def main():
    pm, ph, pnc, pext = census(PARENT)
    cm, ch, cnc, cext = census(CAND)
    out = dict(parent=dict(path=str(PARENT), words=len(pm.words), bytes=PARENT.stat().st_size,
                           nc=sorted(pnc), nc_count=len(pnc)),
               candidate=dict(path=str(CAND), words=len(cm.words), bytes=CAND.stat().st_size,
                              nc=sorted(cnc), nc_count=len(cnc)))
    keys = sorted(set(ph) | set(ch))
    delta = []
    for k in keys:
        if ph.get(k, 0) != ch.get(k, 0):
            delta.append(dict(opcode=k, name=NAMES.get(k, 'op%d' % k),
                              parent=ph.get(k, 0), candidate=ch.get(k, 0),
                              delta=ch.get(k, 0) - ph.get(k, 0)))
    out['opcode_delta'] = delta
    out['opcode_identical_except_delta'] = True
    out['fp_opcodes'] = {NAMES.get(k, 'op%d' % k): dict(parent=ph.get(k, 0), candidate=ch.get(k, 0))
                         for k in sorted(set(FP) | set(CONV)) if ph.get(k, 0) or ch.get(k, 0)}
    out['ext_inst'] = dict(parent=dict(pext), candidate=dict(cext),
                           note='literal = GLSL.std.450 instruction number (50 = Fma)')
    out['control_flow'] = {NAMES.get(k, 'op%d' % k): dict(parent=ph.get(k, 0), candidate=ch.get(k, 0))
                           for k in sorted(CF) if ph.get(k, 0) or ch.get(k, 0)}
    print(json.dumps(out, indent=2))


if __name__ == '__main__':
    main()
