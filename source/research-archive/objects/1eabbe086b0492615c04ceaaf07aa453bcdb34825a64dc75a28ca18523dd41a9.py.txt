"""Unit test for the field-carrying rebase (what shape cards rely on).

Builds a doctored copy of an existing single-PASS card with a modified grid on its PASS
line, rebases it onto the current chain head, and asserts:
  - the target PASS line carries the doctored grid fields and the card's candidate SPV,
  - every other PASS line is byte-identical to the chain head.
No GPU.
"""
import shutil, sys
from pathlib import Path
sys.path.insert(0, str(Path(__file__).parent))
import COMMON as K
from audit_card import pass_lines

SRC = 'n02b-f20-store'
NAME = '_carry_test_card'
ORD = 230
tmp = K.C / 'cards' / NAME
if tmp.exists():
    shutil.rmtree(tmp)
tmp.mkdir(parents=True)
for f in ('candidate.spv', 'GPU-REQUEST.json', 'plan.txt'):
    shutil.copyfile(K.C / 'cards' / SRC / f, tmp / f)

lines = (tmp / 'plan.txt').read_text(encoding='utf-8-sig').splitlines()
recs = pass_lines(tmp / 'plan.txt')
line = recs[ORD]['line']
toks = line.split()
grid = toks[-3:]
doctored = [str(int(grid[0]) * 2) if grid[0].isdigit() else '128'] + grid[1:]
new_line = ' '.join(toks[:-3] + doctored)
idx = next(i for i, l in enumerate(lines) if l == line)
lines[idx] = new_line
(tmp / 'plan.txt').write_text('\n'.join(lines) + '\n', encoding='utf-8')
print('doctored grid', grid, '->', doctored)

try:
    out, sha, n, meta = K.rebase_by_ordinals(NAME, out_plan=K.C / 'queue' / '_carry_test' / 'rebased.txt')
    head = K.current_best()[0]
    a = pass_lines(out)
    b = pass_lines(head)
    changed = [i for i in range(232) if a[i]['line'] != b[i]['line']]
    print('rows differing from head:', changed)
    assert changed == [ORD], changed
    row = a[ORD]['line'].split()
    assert row[-3:] == doctored, (row[-3:], doctored)
    assert (tmp / 'candidate.spv').as_posix() in a[ORD]['line'].replace('\\', '/')
    print('PASS: only ord %d changed, grid %s carried, candidate SPV present' % (ORD, doctored))
    print('meta:', meta)
finally:
    shutil.rmtree(tmp, ignore_errors=True)
