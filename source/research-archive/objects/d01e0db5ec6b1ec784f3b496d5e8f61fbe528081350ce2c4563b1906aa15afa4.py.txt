"""Aggregate the S09 QK-static study into the tables used by QUALITY.md / STATUS.txt.

Reports, over the 24 (frame, block) combinations:
  * mean / median of the per-(frame,block) mean-over-32-heads relL2 for dynamic and static
  * the same for score, exp(P), context (pre-inverse numerator), T (final half)
  * static-vs-dynamic attribution (how much of the delta is the scale change itself)
  * saturation (clip fraction of Q/K codes) per frame
"""
import json
import os
import sys

import numpy as np

HERE = os.path.dirname(os.path.abspath(__file__))
rep = json.load(open(os.path.join(HERE, "s09-qk-static-study.json"), encoding="utf-8"))
FRAMES = list(rep["frames"])
BLOCKS = rep["blocks"]
metrics = ("score", "P", "ctx", "T")


def coll(fname, tag, metric):
    return [rep["per_frame"][fname]["block%d" % b][tag][metric]["mean"] for b in BLOCKS]


print("== per-frame, per-block mean-over-heads (median across the 8 blocks) ==")
print("%-10s %8s %8s %8s %8s | %8s %8s %8s %8s" % (
    "frame", "d_score", "d_P", "d_ctx", "d_T", "s_score", "s_P", "s_ctx", "s_T"))
for f in FRAMES:
    vals = []
    for tag in ("dyn", "static_qk"):
        for m in metrics:
            vals.append(float(np.median(coll(f, tag, m))))
    print("%-10s %8.5f %8.5f %8.5f %8.5f | %8.5f %8.5f %8.5f %8.5f" % (f, *vals))

print()
print("== 24-combo aggregate (mean over frames x blocks of the per-block head mean) ==")
print("%-10s %10s %10s %8s" % ("metric", "dynamic", "static_qk", "ratio"))
for m in metrics:
    d = float(np.mean([np.mean(coll(f, "dyn", m)) for f in FRAMES]))
    s = float(np.mean([np.mean(coll(f, "static_qk", m)) for f in FRAMES]))
    print("%-10s %10.5f %10.5f %8.3f" % (m, d, s, s / d))

print()
print("== static vs dynamic attribution (same V/P path, only the Q/K step differs) ==")
for m in ("ctx", "T"):
    vals = [rep["per_frame"][f]["block%d" % b]["static_vs_dyn"][m]["mean"] for f in FRAMES for b in BLOCKS]
    print("  static_vs_dyn %-4s mean=%.5f median=%.5f max=%.5f" % (m, np.mean(vals), np.median(vals), np.max(vals)))

print()
print("== saturation of the Q/K codes (roundEven(x/s) outside [-127,127]) ==")
print("%-10s %12s %12s %8s %8s" % ("frame", "clip_q_max", "clip_k_max", "nQheads", "nKheads"))
for f in FRAMES:
    q = max(rep["per_frame"][f]["block%d" % b]["static_qk"]["clip_q_max"] for b in BLOCKS)
    k = max(rep["per_frame"][f]["block%d" % b]["static_qk"]["clip_k_max"] for b in BLOCKS)
    nq = sum(rep["per_frame"][f]["block%d" % b]["static_qk"]["heads_with_q_clip"] for b in BLOCKS)
    nk = sum(rep["per_frame"][f]["block%d" % b]["static_qk"]["heads_with_k_clip"] for b in BLOCKS)
    print("%-10s %11.5f%% %11.5f%% %8d %8d" % (f, 100 * q, 100 * k, nq, nk))

print()
print("== resolution res = median(dynamic per-WG/chunk absmax) / static per-head absmax ==")
for f in FRAMES:
    for kk in ("Q", "K"):
        v = [rep["per_frame"][f]["block%d" % b]["resolution"][kk]["mean"] for b in BLOCKS]
        print("  %-10s res%s median=%.3f min=%.3f max=%.3f" % (f, kk, np.median(v), min(v), max(v)))

print()
print("== pack files ==")
for b in BLOCKS:
    e = rep["pack"]["block%d" % b]
    print("  block%-2d bytes=%d sha256=%s" % (b, e["bytes"], e["sha256"]))
