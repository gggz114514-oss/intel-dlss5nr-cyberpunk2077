"""Independent parent-byte verification for all 39 baseline shaders, using the idt recipes.

This is the auditor's own reproduction, not a card's claim. Writes
evidence/PARENT-BYTE-VERIFIED.json
"""
import json, sys, shutil
from pathlib import Path
sys.path.insert(0, str(Path(__file__).parent))
import COMMON as K

recipes = K.read_json(K.C / 'cards' / 'idt-identity-recipe' / 'RECIPES.json')
out_dir = K.C / 'evidence' / 'parent-byte'
out_dir.mkdir(parents=True, exist_ok=True)
rows = []
for sha, rec in recipes.items():
    d = out_dir / ('ord%03d' % rec['ordinals'][0])
    d.mkdir(parents=True, exist_ok=True)
    target = Path(rec['target'])
    res = dict(sha=sha, name=rec['name'], families=rec['families'], ordinals=rec['ordinals'],
               source=rec['source'], target=str(target), recipe=rec.get('compile'),
               decoration_parent=rec.get('decoration_parent'), nc_recipe=rec.get('nc_recipe'))
    try:
        comp = rec['compile']
        incs = comp.get('include_dirs') or []
        if comp.get('pinned_include_dir'):
            incs = [comp['pinned_include_dir']] + [x for x in incs if x != comp['pinned_include_dir']]
        raw = d / 'raw.spv'
        K.compile_glsl(Path(rec['source']), raw, [Path(x) for x in incs] + [Path(rec['source']).parent],
                       defines=comp.get('defines') or ())
        res['raw_ok'] = K.sha(raw) == sha
        final = raw
        if not res['raw_ok'] and rec.get('decoration_parent'):
            pm = K.ST.Module(Path(rec['decoration_parent']))
            rm = K.ST.Module(raw)
            keep, _ = K.ST.transfer_decoration_pattern_aligned(pm, rm, set())
            fixed = d / 'decorated.spv'
            K.ST.set_no_contraction(rm, keep, fixed)
            final = fixed
            res['decorated_ok'] = K.sha(fixed) == sha
        res['ok'] = K.sha(final) == sha
        if res['ok']:
            shutil.copyfile(final, d / 'rebuild.spv')
        res['produced_sha'] = K.sha(final)
    except Exception as e:
        res['ok'] = False
        res['error'] = str(e)[:600]
    rows.append(res)
    print('%-5s %-4s %-30s ok=%s' % (sha[:5], res['families'][0], res['name'], res['ok']), flush=True)

summary = dict(total=len(rows), verified=sum(1 for r in rows if r['ok']),
               failed=[r for r in rows if not r['ok']], rows=rows)
K.write_json(K.C / 'evidence' / 'PARENT-BYTE-VERIFIED.json', summary)
print('verified', summary['verified'], '/', summary['total'])
