"""Census of redundant expensive ops across the frame, with the proof requirement spelled out."""
import json, sys
from collections import Counter
from pathlib import Path
sys.path.insert(0, str(Path(__file__).parent))
import COMMON as K
import ISA

cap = K.C / 'queue' / 'n32-run' / 'run' / 'captures-base'
aud = K.read_json(K.C / 'evidence' / 'LATENCY-TARGETS.json')
hash2fam = {}
for z in aud:
    fams = z.get('families') or []
    if fams:
        hash2fam[z['hash']] = ','.join(fams)

rows = []
for p in sorted(cap.rglob('*.asm')):
    isa = ISA.parse(p)
    h = p.name.split('_')[1].replace('asm', '')
    for opname in ('math.inv', 'math.rsqt'):
        srcs = []
        for i in isa:
            if i['op'] != opname:
                continue
            ops = [o for o in i['ops'] if o]
            if len(ops) >= 2:
                srcs.append(ops[-1])
        c = Counter(srcs)
        if not c:
            continue
        copies = sum(c.values())
        distinct = len(c)
        if copies > distinct:
            rows.append(dict(kernel=p.name, hash=h, family=hash2fam.get(h, '?'), op=opname,
                             copies=copies, distinct_sources=distinct, redundant=copies - distinct,
                             sources=dict(c)))
rows.sort(key=lambda r: -r['redundant'])
print('%-18s %-5s %-9s %6s %6s %6s' % ('hash', 'fam', 'op', 'copies', 'srcs', 'redund'))
for r in rows:
    print('%-18s %-5s %-9s %6d %6d %6d' % (r['hash'], r['family'], r['op'], r['copies'],
                                           r['distinct_sources'], r['redundant']))
tot = sum(r['redundant'] for r in rows)
print('\ntotal redundant expensive ops in the frame: %d' % tot)
K.write_json(K.C / 'evidence' / 'REDUNDANT-LATENCY-OPS.json', dict(
    capture=str(cap), rows=rows, total_redundant=tot,
    proof_requirement='identical source-operand text is not sufficient: a card must show that the '
                      'register is not rewritten between the copies (or that the value is a push '
                      'constant / loop invariant), and that every consumer reads the same value.'))
