"""Read-only: print the plan row field tails for a few ordinals (plan format probe)."""
import sys

PLAN = ('@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/'
        'fullnet-continuous-20261027/cards/n30-f20-arena-rotate/plan.txt')

lines = open(PLAN, encoding='utf-8').read().splitlines()
idx = [i for i, l in enumerate(lines) if l.startswith('PASS ')]
print('rows:', len(idx), 'header:', lines[0])
for ordv in [int(a) for a in sys.argv[1:]] or [0, 62, 193, 230, 231]:
    parts = lines[idx[ordv]].split('"')
    print('\nord %-3d line %-5d tail: %s' % (ordv, idx[ordv] + 1, parts[-1].strip()))
