import json, sys
from pathlib import Path
sys.path.insert(0, r"@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/fullnet-continuous-20261027")
import COMMON as K

rows = K.rows()
d = json.load(open(sys.argv[1], encoding="utf-8-sig"))
vals = {p["index"]: (p["name"], p["values"][0]) for p in d["frames"][0]["passes"]}
TARGET = "881dff8c51454f79181a567d6b976ea21ec7c81174bf075ca7be723b09c4a3f0"
idxs = [i for i, r in enumerate(rows) if r["spv_sha256"] == TARGET]
print("C256 plan row indices:", idxs, "count", len(idxs))
tot = 0
for i in idxs:
    r = rows[i]
    n, v = vals[i]
    tot += v
    print("  %3d %-26s %-24s %8d ns  %6.2f us" % (i, r["name"], n, v, v / 1000.0))
frame = sum(v for _, v in vals.values())
print("  group total %.2f us = %.4f ms   frame %.3f ms  share %.2f%%" % (
    tot / 1000.0, tot / 1e6, frame / 1e6, 100.0 * tot / frame))
