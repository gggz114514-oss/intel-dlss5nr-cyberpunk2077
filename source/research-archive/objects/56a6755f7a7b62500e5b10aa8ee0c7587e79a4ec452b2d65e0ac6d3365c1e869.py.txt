"""Rebase this card's plan onto any parent plan, rewriting BOTH the SPV path and
the dispatch grid triple of the 16 F08 PASS lines.

COMMON.rebase_candidate_plan() only replaces the SPV path; for this card that
would leave `66 2 1` in place and the transposed mapping would compute only a
fraction of the tokens.  Always use this script (or replicate both edits).

usage:
  python rebase.py [parent-plan] [out-plan]
defaults: parent = CURRENT.json chain head, out = plan.txt
Only writes inside this card directory.
"""
import sys, json
from pathlib import Path

CAMPAIGN = Path(r'@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/fullnet-continuous-20261027')
sys.path.insert(0, str(CAMPAIGN))
import COMMON as K

CARD = CAMPAIGN / 'cards' / 'n08b-f08-order'
PARENT_SPV = Path(r'@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/fullnet-mechanism-transfer-20261026/f08-register-chain-20261027/candidate.spv')
PARENT_SPV_SHA = '32291c7ea61b5a1762ba492474806e37376a51e88a8c9c72e286752676709491'
PARENT_GRID = ('66', '2', '1')
CAND_GRID = ('4', '33', '1')


def rebase(parent_plan, out_plan, spv):
    assert K.sha(PARENT_SPV) == PARENT_SPV_SHA, K.sha(PARENT_SPV)
    raw = Path(parent_plan).read_bytes()
    bom = raw.startswith(b'\xef\xbb\xbf')
    lines = raw.decode('utf-8-sig').split('\n')
    parent_posix = PARENT_SPV.as_posix()
    new_posix = Path(spv).as_posix()
    changed = []
    for i, line in enumerate(lines):
        cr = '\r' if line.endswith('\r') else ''
        body = line[:-1] if cr else line
        if not body.startswith('PASS ') or parent_posix not in body:
            continue
        parts = body.split(' ')
        assert tuple(parts[-3:]) == PARENT_GRID, 'unexpected parent grid: %s' % parts[-3:]
        parts[-3:] = list(CAND_GRID)
        parts[3] = new_posix
        lines[i] = ' '.join(parts) + cr
        changed.append(i + 1)
    assert len(changed) == 16, 'replaced %d lines, expected 16' % len(changed)
    out = Path(out_plan)
    out.write_bytes((('\ufeff' if bom else '') + '\n'.join(lines)).encode('utf-8'))
    return out, changed


def main():
    argv = sys.argv[1:]
    chain_plan, chain_sha, _ = K.current_best()
    parent = Path(argv[0]) if len(argv) > 0 else chain_plan
    out = Path(argv[1]) if len(argv) > 1 else (CARD / 'plan.txt')
    spv = CARD / 'candidate.spv'
    plan, lines = rebase(parent, out, spv)
    cpu = K.cpu_check(plan)
    info = {'parent_plan': K.ident(parent), 'out_plan': K.ident(plan),
            'candidate_spv': K.ident(spv), 'grid': ' '.join(CAND_GRID),
            'lines_changed_1based': lines, 'cpu_check': cpu.strip()}
    print(json.dumps(info, indent=1))


if __name__ == '__main__':
    main()
