import sys
from pathlib import Path
import numpy as np

sys.path.insert(0, str(Path(__file__).parent))
import c32blk as M
from validate import bk, PUSH, ptr

B = Path(r"@NR_WORKSPACE@")
W = B / "work/dsh-fullnet-static-20261029"
src = W / "C00-color-baseline/runs/diag65-69"

x, valid = M.load_block(src / "c00-block66_out.bin", -1, -1)
blk = M.Block(bk("half", 67), bk("i8", 67), bk("sc", 67), bk("aux", 67), ptr(PUSH))
y, tr = blk.forward(x, valid, chunk_windows=M.NWIN, trace=True)
idx, vv = M.token_index(-1, -1)
gpu = np.fromfile(src / "c00-block67_out.bin", dtype="<f2")[idx][vv].astype(np.float64)
pred = y[vv].astype(np.float64)

print("per-channel affine fit gpu = a*pred+b")
res = []
for c in range(32):
    A = np.stack([pred[:, c], np.ones(pred.shape[0])], 1)
    coef, *_ = np.linalg.lstsq(A, gpu[:, c], rcond=None)
    fit = A @ coef
    res.append((coef[0], coef[1], M.rel_l2(fit, gpu[:, c]), M.rel_l2(pred[:, c], gpu[:, c])))
    if c < 6 or c in (19, 26, 30, 31):
        print(f"  ch{c:2d} a={coef[0]:.3f} b={coef[1]:8.3f} fit_relL2={res[-1][2]:.4f} raw={res[-1][3]:.4f}")
print("mean fitted a", np.mean([r[0] for r in res]), "mean b", np.mean([r[1] for r in res]))
print("mean fit relL2", np.mean([r[2] for r in res]), "mean raw", np.mean([r[3] for r in res]))
print("tap_o", np.round(blk.fsc[260:292], 5).tolist())
print("ars", np.round(blk.ars.astype(np.float64), 4).tolist())
print("rs", np.round(blk.rs.astype(np.float64), 4).tolist())
