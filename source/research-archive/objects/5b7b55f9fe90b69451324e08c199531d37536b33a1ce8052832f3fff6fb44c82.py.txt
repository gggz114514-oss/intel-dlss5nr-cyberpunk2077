import os, shlex, collections, json

W = r"@NR_WORKSPACE@/work/dsh-fullnet-static-20261029"
Q = r"@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/int8-quality-first-20261029"
D = os.path.join(Q, "real-int8-24")
PLAN = os.path.join(D, "c64-family-static/combo-34189.txt")

def toks(line):
    try: return shlex.split(line)
    except ValueError: return line.split()

bufs = []; passes = []; binds = collections.defaultdict(list)
for ln in open(PLAN, encoding="utf-8", errors="replace"):
    s = ln.strip()
    if not s: continue
    t = toks(s); k = t[0]
    if k == "BUFFER": bufs.append({"name": t[1], "bytes": int(t[2]), "file": t[3] if len(t) > 3 else "-"})
    elif k == "PASS": passes.append({"group": int(t[1]), "name": t[2], "spv": t[3], "args": t[4:]})
    elif k == "BIND": binds[int(t[1])].append({"binding": int(t[2]), "idx": int(t[3]), "offset": int(t[4]), "bytes": int(t[5]) if len(t) > 5 else None})

def show(ordinal):
    p = passes[ordinal]
    print("\nordinal %d  name=%s  group=%d" % (ordinal, p["name"], p["group"]))
    print("  push:", p["args"])
    for b in binds[ordinal]:
        nm = bufs[b["idx"]]["name"]
        print("   bind%-3d idx=%-4d %-26s off=%-10d bytes=%-9d file=%s" % (
            b["binding"], b["idx"], nm, b["offset"], b["bytes"] or 0, bufs[b["idx"]]["file"][-60:]))

# find key passes
idx = {p["name"]: i for i, p in enumerate(passes)}
for nm in ("f2c32_block66_fused", "ioport_block70_0", "ioport_block70_1", "ioport_block70_2", "f1c32_image_pre", "block65_swin_body_4"):
    if nm in idx:
        show(idx[nm])
    else:
        # multiple passes per name
        hits = [i for i, p in enumerate(passes) if p["name"] == nm]
        print("\n", nm, "hits:", hits)
        for i in hits:
            show(i)

print("\n=== buffers of interest (index -> name/bytes) ===")
for i, b in enumerate(bufs):
    if b["name"] in ("block66_out", "skip4", "block65_out", "skip0", "frame_out", "ioport_70_ioport_images", "ioport_70_ioport_blended", "ioport_70_ioport_body"):
        print(" idx=%-4d %-28s bytes=%-10d %s" % (i, b["name"], b["bytes"], b["file"][-70:]))
