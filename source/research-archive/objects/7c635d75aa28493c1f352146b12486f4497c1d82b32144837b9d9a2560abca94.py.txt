"""Census: redundant expensive ops — the same math.inv / math.rsqt source operand repeated.

A repeated reciprocal/sqrt with an identical source is dead work unless each copy sits in a
different guarded block (which is why the compiler did not CSE it). Deleting the copies is
the class that actually pays (n29: 2 divisions = +0.080 ms), unlike reordering (L26).
"""
import sys
from collections import Counter, defaultdict
from pathlib import Path
sys.path.insert(0, str(Path(__file__).parent))
import COMMON as K
import ISA

cap = K.C / 'queue' / 'n32-run' / 'run' / 'captures-base'
files = sorted(cap.rglob('*.asm'))
print('kernels:', len(files))
tot = Counter()
rows = []
for p in files:
    isa = ISA.parse(p)
    for opname in ('math.inv', 'math.rsqt'):
        srcs = []
        for i in isa:
            if i['op'] != opname:
                continue
            ops = [o for o in i['ops'] if o]
            if len(ops) >= 2:
                srcs.append(ops[-1])
        c = Counter(srcs)
        dup = {k: v for k, v in c.items() if v > 1}
        if dup:
            rows.append((p.name, opname, sum(dup.values()), len(dup), dict(list(dup.items())[:3])))
            tot[opname] += sum(dup.values()) - len(dup)
print('\nkernels with a repeated expensive op (identical source operand):')
for name, op, n, groups, sample in sorted(rows, key=lambda r: -r[2]):
    print('  %-46s %-9s copies %2d over %2d sources  e.g. %s' % (name[:46], op, n, groups, list(sample)[:2]))
print('\nredundant copies (beyond the first) per op:', dict(tot))
