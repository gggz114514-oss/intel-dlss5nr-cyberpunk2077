#!/usr/bin/env python3
"""Ad-hoc: identify unknown opcodes and instruction contexts in the R03 SPVs."""
import sys
from pathlib import Path

sys.path.insert(0, str(Path(__file__).resolve().parent))
import spv_read as S  # noqa: E402

CARD = Path("@NR_WORKSPACE@/work/dsh-xmx-handoff-validation-20261024/candidates/r03-ioport4")
parent = S.Module(CARD / "spv/parent_repro.spv")
half = S.Module(CARD / "spv/r03_ioport4_i8.spv")

for label, m in (("parent", parent), ("cand", half)):
    print("==", label)
    for op in (4, 176, 177):
        hits = [(i, a, wc) for i, (o, a, wc) in enumerate(m.ins) if o == op]
        print("  opcode", op, "count", len(hits))
        for i, a, wc in hits[:6]:
            print("    idx=%d wc=%d ops=%s" % (i, wc, a))
            for v in a[:4]:
                d = m.defining(v)
                if d:
                    print("        id %d <- %s %s" % (v, m.name(d[0]), d[1]))
    # names of the buffer variables
    print("  named vars:", {i: n for i, n in m.names.items() if n})
