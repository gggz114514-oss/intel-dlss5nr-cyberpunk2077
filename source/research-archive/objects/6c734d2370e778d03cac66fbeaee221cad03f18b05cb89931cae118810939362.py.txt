import json,struct,statistics as s
from pathlib import Path
D=Path(__file__).resolve().parent
def read(p):return json.loads(p.read_text(encoding='utf-8-sig'))
positions=read(D/'direct-c32-family/POSITIONS.json')
phases=['input-seed','ffn-only','stage-qkv','stage-norm-publish','attention-producer','stage-pv','stage-output']
rows=[]
for pos in positions:
 repeats=[]
 for rep in range(3):
  arms={}
  for name in ['control','whole',*phases]:
   folder=D/'run'/f'c32-family-r{rep}-{name}'
   receipt=read(folder/'port_network_receipt.json')
   timing=next(x for x in receipt['per_pass_gpu'] if x['index']==pos['ordinal'])
   data=(folder/pos['output']).read_bytes();assert len(data)==pos['bytes']
   vals=list(struct.iter_unpack('<IIII',data));ticks=[];missing=unexpected=0
   for y in range(pos['gy']):
    for x in range(pos['gx']):
     for sg in range(4):
      a,b,c,d=vals[(y*pos['gx']+x)*4+sg];nz=bool(a or b or c or d)
      selected=name!='control' and x%16==8 and y%16==8
      if selected and nz:ticks.append(((c+(d<<32))-(a+(b<<32)))%2**64)
      elif selected:missing+=1
      elif nz:unexpected+=1
   assert missing==unexpected==0,(pos,name,rep,missing,unexpected)
   arms[name]={'mean_ticks':s.mean(ticks) if ticks else 0,'median_ticks':s.median(ticks) if ticks else 0,'records':len(ticks),'median_ms':s.median(timing['samples_ms'])}
  repeats.append({'rep':rep,'arms':arms,'closure':sum(arms[n]['mean_ticks'] for n in phases)/arms['whole']['mean_ticks'],'max_abs_perturbation':max(abs(arms[n]['median_ms']/arms['control']['median_ms']-1) for n in ['whole',*phases])})
 row={'position':pos,'repeats':repeats,'control_ms':s.mean(r['arms']['control']['median_ms'] for r in repeats),'phases':{n:{'mean_ticks':s.mean(r['arms'][n]['mean_ticks'] for r in repeats),'whole_fraction':s.mean(r['arms'][n]['mean_ticks']/r['arms']['whole']['mean_ticks'] for r in repeats),'repeat_spread':max(r['arms'][n]['mean_ticks'] for r in repeats)/min(r['arms'][n]['mean_ticks'] for r in repeats)-1} for n in phases}}
 rows.append(row)
 print(pos['ordinal'],'ms',round(row['control_ms'],4),'closure',[round(r['closure'],4) for r in repeats],'perturb',round(max(r['max_abs_perturbation'] for r in repeats),4),'percent',{n:round(v['whole_fraction']*100,1) for n,v in row['phases'].items()})
den=sum(s.mean(r['arms']['whole']['mean_ticks'] for r in row['repeats'])*row['position']['gx']*row['position']['gy']*4 for row in rows)
weighted={n:sum(row['phases'][n]['mean_ticks']*row['position']['gx']*row['position']['gy']*4 for row in rows)/den for n in phases}
out={'unit':'subgroup clock ticks, NOT exclusive kernel milliseconds','sampling':'x%16=y%16=8; four subgroups; original grid','rows':rows,'weighted_phase_fractions':weighted,'family_control_ms':sum(row['control_ms'] for row in rows),'precision_limit':'Short QKV/output/norm windows require separate empty-probe budget qualification; closure alone does not remove timer overhead.'}
out['weighted_phase_fractions_status']='INVALID_FOR_INFERENCE: ordinal 4 phase closure fails in repeats 0 and 2; raw aggregate retained, no outliers removed'
out['closure_failed_positions']=[row['position']['ordinal'] for row in rows if any(abs(r['closure']-1)>.1 for r in row['repeats'])]
(D/'MAIN-C32-FAMILY-REVIEW.json').write_text(json.dumps(out,indent=2),encoding='utf-8')
print('family',out['family_control_ms'],'weighted',weighted)
