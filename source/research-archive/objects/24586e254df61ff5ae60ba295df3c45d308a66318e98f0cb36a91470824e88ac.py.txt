#!/usr/bin/env python3
"""Build the 129-row machine-readable coverage table (stage-1 deliverable).

(if any) with its operand-level audit result, and a status word from the fixed vocabulary:
  待实施 / 编译 / 接线 / 静态零残余 / 段级替换

#          after the card, status word)

def sha(p):
    return hashlib.sha256(Path(p).read_bytes()).hexdigest()


def main():
    led = json.loads((A / "FULLNET-INT8-IMPLEMENTATION-LEDGER.json").read_text(encoding="utf-8"))
    pos = led["positions"]
    rows = []
    for p in pos:
        name, order, card = p["pass_name"], p["plan_order"], p["card"]
        mod, covered, residual, status = CARDS.get(card, (None, [], None, "待实施"))
        own = name in covered or any(name.startswith(c) for c in covered if "*" in c)
        rows.append({
            "pass_name": name,
            "plan_order": order,
            "card": card,
            "g3_spv": p["spv"],
            "g3_spv_sha256": p["sha256"],
            "g3_fp16_static_mma": p["fp16_static_mma_count"],
            "candidate_module": str(mod) if (mod and own) else None,
            "candidate_sha256": sha(mod) if (mod and own) else None,
            "residual_fp16_after_card": residual if (mod and own) else p["fp16_static_mma_count"],
            "status": status if (mod and own) else "待实施",
        })
    counts = Counter(r["status"] for r in rows)
    doc = {
        "date": "2026-10-24",
        "baseline": {"plan": led.get("baseline_plan"), "sha256": led.get("baseline_sha256")},
        "key": "pass_name + plan_order (the plan's PASS number is NOT unique; see TOOL-DEFECTS D3)",
        "status_vocabulary": ["待实施", "编译", "接线", "静态零残余", "段级替换",
                              "数值通过(未取得)", "质量通过(未取得)", "性能通过(未取得)"],
        "counts": dict(counts),
        "positions": rows,
    }
    out = W / "FULLNET-129-ROWS.json"
    out.write_text(json.dumps(doc, indent=1, ensure_ascii=False) + "\n", encoding="utf-8", newline="\n")

    L = ["# 129 行覆盖表（0 基 plan_order + pass 名定位）", "",
         f"基线 G3 `{Path(led['baseline_plan']).name}` `{led['baseline_sha256'][:16]}…`；"
         f"共 {len(rows)} 行。**没有任何一行是数值通过或质量通过**。", "",
         "| 状态 | 行数 |", "|---|---:|"]
    for k, v in sorted(counts.items(), key=lambda kv: -kv[1]):
        L.append(f"| {k} | {v} |")
    L += ["", "| # | pass | order | 卡 | G3 FP16 | 候选模块 | 改后 FP16 | 状态 |",
          "|---:|---|---:|---|---:|---|---:|---|"]
    for i, r in enumerate(rows):
        mod = Path(r["candidate_module"]).name if r["candidate_module"] else "—"
        L.append(f"| {i} | `{r['pass_name']}` | {r['plan_order']} | {r['card']} | "
                 f"{r['g3_fp16_static_mma']} | {mod} | {r['residual_fp16_after_card']} | {r['status']} |")
    (W / "FULLNET-129-ROWS.md").write_text("\n".join(L) + "\n", encoding="utf-8", newline="\n")
    print(json.dumps(dict(counts), ensure_ascii=False), "rows", len(rows))


if __name__ == "__main__":
    main()
