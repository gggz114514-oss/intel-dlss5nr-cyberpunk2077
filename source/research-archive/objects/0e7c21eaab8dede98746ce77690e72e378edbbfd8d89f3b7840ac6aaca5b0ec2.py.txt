"""CFG / block census for the F13 expand ISA dump (read-only analysis)."""
import re, sys, json, collections
sys.path.insert(0, r'@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/fullnet-continuous-20261027/cards/n11-vit-ffn/_tools')
from isa_scan import parse

p = sys.argv[1]
ins = parse(p)
label_at = {}
for i in ins:
    if i['label'] and i['label'] not in label_at:
        label_at[i['label']] = i['off']

jumps = []
for i in ins:
    if i['op'].startswith('jmpi') or i['op'] == 'ret' or i['op'].startswith('goto'):
        tgt = None
        m = re.search(r'\bL(\d+)\b', i['text'])
        if m:
            tgt = int(m.group(1))
        jumps.append((i['off'], i['label'], i['op'], tgt, i['pred']))

# back edges
print('total instructions', len(ins))
print('labels', len(label_at))
print('jump sites', len(jumps))
back = [(o, l, t) for (o, l, op, t, pr) in jumps if t and label_at.get('L%d' % t, 1 << 60) <= o]
print('back edges:', len(back))
for o, l, t in back[:20]:
    print('  back edge at %05X (label %s) -> L%d @ %05X' % (o, l, t, label_at['L%d' % t]))
# block sizes: instruction counts between labels
sizes = collections.Counter()
for i in ins:
    sizes[i['label']] += 1
print('blocks (label: instrs) sorted by offset:')
for lab, off in sorted(label_at.items(), key=lambda kv: kv[1])[:40]:
    print('  %s @ %05X  n=%d' % (lab, off, sizes[lab]))
print('...')
# send / dpas / mov census
c = collections.Counter()
for i in ins:
    key = i['op']
    if key == 'mov' or key.startswith('mov'):
        m = re.search(r'\((\d+)\|M0\)', i['text'])
        key = 'mov(%s)' % (m.group(1) if m else '?')
    c[key] += 1
print(json.dumps(dict(c.most_common()), indent=1))
