import json, os, re, collections

ROOT = r"@NR_WORKSPACE@\work\codex-astra-fp16-int8-audit-20261024\fullnet-continuous-20261027"
OUT = os.path.join(ROOT, "cards", "n16-crossdispatch", "tools")
fg = json.load(open(os.path.join(ROOT, "evidence", "FAMILY-GROUPS.json"), encoding="utf-8"))

rows = {}
for sha, g in fg["groups"].items():
    for o in g["ordinals"]:
        rows[o] = g

# unique shader by source path
shaders = {}
for o in sorted(rows):
    g = rows[o]
    src = g["source"]
    if src in shaders:
        shaders[src]["ordinals"].append(o)
        continue
    txt = open(src, encoding="utf-8", errors="replace").read()
    decls = {}
    for m in re.finditer(r"layout\s*\(([^)]*)\)\s*(readonly|writeonly|)?\s*buffer\s+(\w+)\s*\{([^}]*)\}\s*(\w+)\s*;", txt):
        attrs, qual, bname, body, var = m.groups()
        b = re.search(r"binding\s*=\s*(\d+)", attrs)
        decls[var] = {"binding": int(b.group(1)) if b else None, "qual": qual or "",
                      "block": bname, "type": " ".join(body.split())[:80]}
    stores = sorted(set(re.findall(r"coopMatStore\s*\(\s*\w+\s*,\s*(\w+)", txt)))
    loads = sorted(set(re.findall(r"coopMatLoad\s*\(\s*\w+\s*,\s*(\w+)", txt)))
    # plain indexing writes/reads of declared buffer vars
    plain_w, plain_r = set(), set()
    for var in decls:
        for m in re.finditer(re.escape(var) + r"\s*\[([^\]]+)\]\s*(=|\+=|-=|\*=|/=)", txt):
            plain_w.add(var)
        if re.search(r"[=(]\s*" + re.escape(var) + r"\s*\[", txt):
            pass
    shaders[src] = {"family": g["family"], "names": g["names"], "ordinals": [o],
                    "source": src, "decls": decls, "cm_store": stores, "cm_load": loads,
                    "plain_write_vars": sorted(plain_w), "lines": txt.count("\n") + 1,
                    "text": txt}

# compact print per shader
for src, s in shaders.items():
    print("=" * 100)
    print(s["family"], s["names"][0], "ords", s["ordinals"][:6], "n=", len(s["ordinals"]), "lines", s["lines"])
    for var, d in sorted(s["decls"].items(), key=lambda kv: (kv[1]["binding"] is None, kv[1]["binding"])):
        print("   b%-2s %-10s %-10s %s" % (d["binding"], var, d["qual"], d["type"]))
    print("   cm_store:", s["cm_store"])
    print("   cm_load :", s["cm_load"])
    print("   plainW  :", s["plain_write_vars"])

json.dump({k: {kk: vv for kk, vv in v.items() if kk != "text"} for k, v in shaders.items()},
          open(os.path.join(OUT, "shader-io.json"), "w", encoding="utf-8"), indent=1)
