"""Build plan.txt on the CURRENT accepted chain head, replacing only the 8 F14 PASS lines.

CPU only: clock_runner --cpu-check.  No Vulkan device, no GPU lock, no deploy.

Assertions:
  * the chain-head plan sha equals CURRENT.json best_plan.sha256
  * exactly 8 PASS lines change (the F14 ordinals), all to this card's candidate.spv
  * every other plan line is byte-identical
  * candidate and chain-head cpu-check output are identical and rc == 0
"""
import json
import sys
from pathlib import Path

CARD = Path(__file__).resolve().parent
CAMPAIGN = CARD.parent.parent
sys.path.insert(0, str(CAMPAIGN))
import COMMON as K  # noqa: E402

EVID = CARD / 'evidence'
EVID.mkdir(exist_ok=True)

FAMILY = 'F14'
ORDINALS = [95, 100, 105, 110, 115, 120, 125, 130]
BASE_SPV_SHA = '72e2df7c7b1e542f375f5c7c2bdc572433a08c5f1a1a0d706026efa54b9a1e87'
CAND = CARD / 'candidate.spv'


def pass_lines(plan):
    out = []
    for i, l in enumerate(Path(plan).read_text(encoding='utf-8-sig').splitlines(), 1):
        if l.startswith('PASS '):
            out.append((i, l))
    return out


def main():
    best, best_sha, chain = K.current_best()
    assert K.sha(best) == best_sha, 'CURRENT.json best_plan sha != disk'
    base_lines = Path(best).read_text(encoding='utf-8-sig').splitlines(keepends=True)

    replaced = K.make_plan(CAND, CARD / 'plan.txt', parent_plan=best, spv_shas={BASE_SPV_SHA})
    assert replaced == 8, 'replaced %d, expected 8' % replaced

    cand_lines = (CARD / 'plan.txt').read_text(encoding='utf-8-sig').splitlines(keepends=True)
    assert len(cand_lines) == len(base_lines), 'line count changed'
    changed = [(i + 1, b, c) for i, (b, c) in enumerate(zip(base_lines, cand_lines)) if b != c]
    assert len(changed) == 8, 'changed %d lines, expected 8' % len(changed)

    cand_posix = CAND.as_posix()
    rows = [K.row_by_ordinal(o) for o in ORDINALS]
    expect_names = {r['name'] for r in rows}
    expect_labels = {r['pass_label'] for r in rows}
    details = []
    for ln, b, c in changed:
        bp = b.split()
        cp = c.split()
        assert bp[0] == 'PASS' and cp[0] == 'PASS'
        assert bp[1] == cp[1], 'PASS label changed'
        assert bp[2] == cp[2], 'PASS name changed'
        assert cp[3] == '"%s"' % cand_posix, 'candidate path not written'
        assert bp[4:] == cp[4:], 'PASS tail (grid) changed'
        details.append(dict(line=ln, pass_label=bp[1], name=bp[2],
                            old_spv=bp[3].strip('"'), new_spv=cp[3].strip('"')))
    assert {d['name'].strip('"') for d in details} == expect_names
    assert {d['pass_label'] for d in details} == expect_labels

    # every PASS line that is not one of the 8 keeps its parent SPV path
    changed_lines = {d['line'] for d in details}
    for i, (b, c) in enumerate(zip(base_lines, cand_lines), 1):
        if i in changed_lines or not b.startswith('PASS '):
            continue
        assert b == c, 'non-F14 PASS line %d changed' % i
    base_pass = pass_lines(best)
    cand_pass = pass_lines(CARD / 'plan.txt')
    assert len(base_pass) == len(cand_pass) == 232

    ch_base = K.cpu_check(best)
    ch_cand = K.cpu_check(CARD / 'plan.txt')
    assert 'CPU_PLAN_OK' in ch_cand
    out = dict(
        schema='n12b-f14-scale/plan/v1',
        card='n12b-f14-scale',
        family=FAMILY,
        chain_head=dict(path=str(best), sha256=best_sha, lines=len(base_lines),
                        chain=[c.get('label') for c in chain],
                        is_current_json_best=True),
        candidate_plan=dict(path=str(CARD / 'plan.txt'), sha256=K.sha(CARD / 'plan.txt'),
                            bytes=(CARD / 'plan.txt').stat().st_size,
                            lines=len(cand_lines)),
        replaced_passes=replaced,
        expected_passes=len(ORDINALS),
        changed_lines=[d['line'] for d in details],
        changed_pass_labels=[d['pass_label'] for d in details],
        changed_names=[d['name'].strip('"') for d in details],
        replaced=[dict(old=d['old_spv'], new=d['new_spv']) for d in details[:1]],
        pass_count=len(cand_pass),
        only_pass_lines_changed=True,
        pass_tail_unchanged=True,
        cpu_check=dict(baseline=dict(rc=0, stdout=ch_base.strip()),
                       candidate=dict(rc=0, stdout=ch_cand.strip()),
                       identical=ch_base.strip() == ch_cand.strip()),
        method='COMMON.make_plan(candidate, parent_plan=chain head, spv_shas={F14 baseline sha}); '
               'assert replaced == 8',
    )
    K.write_json(EVID / 'PLAN.json', out)
    print(json.dumps({k: out[k] for k in ('chain_head', 'candidate_plan', 'replaced_passes',
                                          'changed_lines', 'changed_names', 'cpu_check')},
                     indent=2))
    print('replaced', replaced, 'changed', len(changed), 'identical cpu-check',
          out['cpu_check']['identical'])


if __name__ == '__main__':
    main()
