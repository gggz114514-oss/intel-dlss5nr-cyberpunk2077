from pathlib import Path

R=Path(__file__).resolve().parent; N=R/'family-norm-owner'; P=R/'f20-score-owner'; C=N/'combo'
def read(p):return json.loads(p.read_text(encoding='utf-8-sig'))
def ident(p):return dict(path=str(p),sha256=hashlib.sha256(p.read_bytes()).hexdigest())
def save(p,x):p.write_text(json.dumps(x,indent=2),encoding='utf-8')
def native(base,index):
 ps=list((base/'captures').rglob(f'*_{index}_orig.asm'));assert len(ps)==1,(base,index,ps)
 p=ps[0];ls=[s for s in p.read_text().splitlines() if re.match(r'/\* \[[0-9A-F]+\]',s)]
 return dict(**ident(p),instructions=len(ls),counts={k:sum(k in s for s in ls) for k in ['math.rsqt','dpas.','send.slm','send.ugm','sync.bar','mov','sel']},scratch=[s for s in ls if re.search(r'\.ss\b',s)])
cov=read(N/'FAMILY-COVERAGE.json');reviews=[]
for f in cov['families']:
 d=N/f['family']
 for j in read(d/'LUNA-GATES.json')['jobs']:

  receipt=read(out/'port_network_receipt.json');assert receipt['normal_close'] and receipt['gpu_passes']==232 and receipt['nan']==receipt['inf']==0
  if j['mode']=='fb2':assert ident(out/'frame_0.bin')['sha256']==j['expected_frame0']
 for i in f['pipeline_indices']:
  a,b=native(P,i),native(d,i);assert not b['scratch'];assert a['counts']['dpas.']==b['counts']['dpas.']
  reviews.append(dict(family=f['family'],pipeline=i,parent=a,candidate=b))
save(N/'NATIVE-REVIEW.json',dict(rows=reviews,limits='Static counts, not runtime. Driver scratch still requires combo resource check.'))
for x in reviews:print(x['family'],x['pipeline'],x['parent']['instructions'],x['candidate']['instructions'],x['parent']['counts'],x['candidate']['counts'])
spvs=[f['spv'] for f in cov['families']]+[cov['F20_existing']]
for kind,template in [('GATES',N/'f2/LUNA-GATES.json'),('RESOURCES',P/'LUNA-RESOURCES.json')]:

 for j in t['jobs']:
  j['id']='combo-'+j['id'];j['plan']=ident(C/'plan.txt');j['runner_output']=str(C/'run'/j['id']);a=j['argv'];a[a.index('--plan')+1]=str(C/'plan.txt');a[a.index('--out')+1]=j['runner_output']
 save(C/f'LUNA-{kind}.json',t)
t=read(P/'LUNA-ORDINARY.json');t['output_root']=str(C);t['identities']=t['identities'][:1]+spvs;proto=t['jobs'][0];jobs=[]
for pair,order in enumerate([['parent','candidate'],['candidate','parent'],['parent','candidate']]):
 for arm in order:
  j=json.loads(json.dumps(proto));j.update(pair=pair,candidate=arm,id=f'ordinary-{pair}-{arm}');plan=(P if arm=='parent' else C)/'plan.txt';j['plan']=ident(plan);j['runner_output']=str(C/'run'/j['id']);a=j['argv'];a[a.index('--plan')+1]=str(plan);a[a.index('--out')+1]=j['runner_output'];jobs.append(j)
t['jobs']=jobs;t['rules']=['After main resource review only. Three alternating process pairs, all180 samples, no profiler, no retries.']
save(C/'LUNA-ORDINARY.json',t)
