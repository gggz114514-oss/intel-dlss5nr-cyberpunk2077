import difflib
from pathlib import Path

CAMP = Path('@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/fullnet-continuous-20261027')
CARD = CAMP / 'cards/n07n14-arena'
parent = Path('@NR_WORKSPACE@/candidates/amd-netorg-full-scope-workbuddy-20261017/c2-layout/layout0-unify-l2-v4/source/f3-c256-fused/fused_c256_body_ds.comp')
cand = CARD / 'candidate.comp'
a = parent.read_text(encoding='utf-8-sig').splitlines(keepends=True)
b = cand.read_text(encoding='utf-8-sig').splitlines(keepends=True)
d = difflib.unified_diff(a, b, fromfile='a/' + str(parent).replace('\\', '/'),
                         tofile='b/' + str(cand).replace('\\', '/'), n=8)
txt = ''.join(d)
(CARD / 'SOURCE-DIFF.patch').write_text(txt, encoding='utf-8')
print('lines', len(txt.splitlines()), 'changed +', sum(1 for l in txt.splitlines() if l.startswith('+') and not l.startswith('+++')),
      '-', sum(1 for l in txt.splitlines() if l.startswith('-') and not l.startswith('---')))
print(txt[:400])
