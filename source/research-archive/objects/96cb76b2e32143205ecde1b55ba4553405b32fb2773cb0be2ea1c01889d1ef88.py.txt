#!/usr/bin/env python3
"""Summarise a plan_mma_census.json: residual FP16 positions grouped by module.

usage: report_residual.py <census.json> <plan-substring> [--md out.md]
"""
import json
import sys
from collections import defaultdict
from pathlib import Path


def main(argv):
    md = None
    if "--md" in argv:
        i = argv.index("--md")
        md = Path(argv[i + 1])
        argv = argv[:i] + argv[i + 2:]
    census, needle = argv[0], argv[1]
    d = json.loads(Path(census).read_text(encoding="utf-8"))
    plan = [p for p in d["plans"] if needle in p["plan"]][0]
    rows = [r for r in plan["rows"] if r["fp16_mma"] > 0]
    agg = defaultdict(lambda: [0, 0, []])
    for r in rows:
        a = agg[r["spv"]]
        a[0] += 1
        a[1] += r["fp16_mma"]
        a[2].append(r["pass_name"])
    lines = [f"# residual FP16 — {Path(plan['plan']).name}",
             "",
             f"plan sha256 `{plan['plan_sha256']}`  header `{plan['header']}`",
             f"passes {plan['passes']}  buffers {plan['buffers']}  policies {plan['policies']}",
             f"**FP16 static MMA {plan['fp16_static_mma_total']} in {plan['passes_with_fp16']} passes**"
             f"  (INT8 static MMA {plan['int8_static_mma_total']})",
             "",
             "| FP16 | passes | module |", "|---:|---:|---|"]
    for k, (n, s, _) in sorted(agg.items(), key=lambda kv: -kv[1][1]):
        lines.append(f"| {s} | {n} | `{k}` |")
    lines += ["", "| # | pass | ordinal | module | FP16 | INT8 |", "|---:|---|---:|---|---:|---:|"]
    for i, r in enumerate(rows):
        lines.append(f"| {i} | `{r['pass_name']}` | {r['ordinal']} | `{r['spv']}` | "
                     f"{r['fp16_mma']} | {r['int8_mma']} |")
    text = "\n".join(lines) + "\n"
    print(text if not md else text[:1500])
    if md:
        md.write_text(text, encoding="utf-8", newline="\n")
        print("wrote", md)
    return 0


if __name__ == "__main__":
    sys.exit(main(sys.argv[1:]))
