#!/usr/bin/env python3
"""s37-fnorm localization step 2: dump norm-section float dataflow of both SPVs.

Prints, in program order, every float arithmetic / convert / shuffle /
extinst instruction with resolved result type. Parent's scalar norm section
and candidate's fragment norm section are then comparable node for node.
"""
import sys
from collections import Counter

sys.path.insert(0, r"@NR_WORKSPACE@/build/zcode-fp16-compute-backend-20261021/scripts")
import u09_build as U
S = U.S

CAND = (r"@NR_WORKSPACE@/work/zcode-fp16-fullnet-handoff-20261024"
        r"/candidates/s37-fnorm/spv")
ARITH = {"OpFMul", "OpFAdd", "OpFSub", "OpFDiv", "OpFNegate", "OpExtInst",
         "OpFConvert", "OpMatrixTimesScalar", "OpVectorTimesScalar",
         "OpGroupNonUniformShuffleXor", "OpGroupNonUniformShuffle",
         "OpGroupNonUniformBroadcast", "OpBitcast", "OpCompositeConstruct",
         "OpCompositeExtract", "OpCompositeInsert", "OpSelect"}
GLSL450 = {28: "Fma", 32: "InverseSqrt", 90: "NMax", 89: "NMin", 40: "FMax",
           37: "FMin", 91: "NClamp", 31: "Sqrt"}


def analyze(path, label):
    m = S.parse(path)
    types = {}
    consts = {}
    extnames = {}
    rows = []
    for inst in m:
        nm, ops = inst.name, inst.ops
        if nm == "OpTypeFloat":
            types[ops[0]] = "f%d" % ops[1]
        elif nm == "OpTypeVector":
            types[ops[0]] = "v%d%s" % (ops[2], types.get(ops[1], "?"))
        elif nm == "OpTypeMatrix":
            types[ops[0]] = "mat%d%s" % (ops[2], types.get(ops[1], "?"))
        elif nm == "OpTypeInt":
            types[ops[0]] = "i%d" % ops[1]
        elif nm == "OpTypeCooperativeMatrixKHR":
            types[ops[0]] = "coop%s" % types.get(ops[1], "?")
        elif nm == "OpConstant":
            consts[ops[1]] = ops[2]
        elif nm == "OpExtInstImport":
            extnames[ops[0]] = "GLSL450" if b"GLSL" in b"".join(w.to_bytes(4, "little") for w in ops[1:]) else "?"
        elif nm in ARITH:
            rtype = types.get(ops[0], "t%d" % ops[0])
            rid = ops[1]
            if nm == "OpExtInst":
                sub = GLSL450.get(ops[3], "ext%d" % ops[3])
                tag = "ExtInst.%s" % sub
                src = ops[4:]
            else:
                tag = nm
                src = ops[2:]
            rows.append((inst.idx, tag, rtype, rid, src))
    print("=== %s (%d float-flow insts) ===" % (label, len(rows)))
    for idx, tag, t, rid, src in rows:
        print("  [%4d] id=%-5d %-14s %-10s src=%s" % (idx, rid, tag, t, src))
    print()
    return m, types, consts, rows


pm, pt, pc, prows = analyze(CAND + "/parent_repro.spv", "PARENT")
cm, ct, cc, crows = analyze(CAND + "/s37_fnorm.spv", "CANDIDATE")
