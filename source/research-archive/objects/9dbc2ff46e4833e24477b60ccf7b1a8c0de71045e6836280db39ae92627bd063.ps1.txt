function SaveLock { $bytes=[Text.Encoding]::UTF8.GetBytes(($record|ConvertTo-Json -Depth 8));$stream.Position=0;$stream.SetLength(0);$stream.Write($bytes,0,$bytes.Length);$stream.Flush($true) }
