import json, os

W = r"@NR_WORKSPACE@/work/dsh-fullnet-static-20261029"
o = json.load(open(os.path.join(W, "S03-tail/TAIL-CONTRACT.json"), encoding="utf-8"))
for st in o["stages"]:
    print("=" * 110)
    print("STAGE", st["id"])
    print(" math_stages:")
    for ms in st["math_stages"]:
        print("   ", json.dumps(ms, ensure_ascii=False) if not isinstance(ms, str) else ms)
    for side in ("int8", "fp16_counterpart"):
        d = st[side]
        keep = {k: d[k] for k in ("ordinal", "name", "pass_group", "spv", "measured_dtype", "int8_mma", "fp16_mma", "int_dots", "has_127", "has_inv127") if k in d}
        print(" %s: %s" % (side, json.dumps(keep, ensure_ascii=False)))
        print("    geometry:", json.dumps(d.get("pass_geometry"), ensure_ascii=False)[:400])
        print("    ext_inst:", json.dumps(d.get("ext_inst"), ensure_ascii=False)[:300])
    print(" alignment:", json.dumps(st.get("alignment"), ensure_ascii=False, indent=1)[:1600])
