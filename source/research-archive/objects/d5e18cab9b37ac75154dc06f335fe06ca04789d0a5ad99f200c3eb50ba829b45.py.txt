"""qkv_pack_oracle.py - offline QKV INT8 pack builder + the arm's own numeric
reference (CPU only, no device, no pipeline, no lock).

Two independent jobs:

A. WEIGHT PACK.  For each of the seven R02 passes it reads that pass's own PUSH
   (never a constant) out of the deployed plan, gathers the FP16 Q/K/V weight
   planes from the bound half-weight buffer through the parent shader's literal
   address expression, quantizes per output channel exactly like the existing
   C32 INT8 family (s[n] = amax_k|W[n,k]|/127, RNE, clamp +-127), and writes the
   twelve 1024-byte I8A K32 tiles the candidate kernel loads at
   C32_I8_WQ/WK/WV.  A round-trip decode through the kernel's own tile
   addressing must return the quantized matrix element for element.

B. NUMERIC ORACLE.  On a real activation tile captured by the deployed parent
   SPV (the debug dump of the very plan that binds c32_body1_i8.spv), it runs
   both arms in numpy: the FP16 reference (half weights x half Y, float32
   accumulation, narrow to half) and the INT8 arm (int32 accumulation, one
   dequant multiply per output channel, narrow to half), and reports the error.
   The integer accumulation is cross-checked against int64 so the reference does
   not depend on the accumulation width.

The activation inverse scale invY is per tensor and needs that block's own
capture; only block1 is calibrated here, from the captured tile, and every other
block is marked PENDING_CALIBRATION rather than guessed.
"""
import hashlib
import json
import re
import sys
from pathlib import Path

import numpy as np

R = Path("@NR_WORKSPACE@")
CARD = R / "work/dsh-xmx-handoff-validation-20261024/candidates/r02-c32-body"
PACK = CARD / "pack"
EV = CARD / "evidence"
PLAN = R / "candidates/int8-coverage-v2-20261020/plan-c32pass8-production.txt"
PASSES = ("block1_swin_body_1", "block2_swin_body_1", "block3_swin_body_1",
          "block4_swin_body_1", "block67_swin_body_1", "block68_swin_body_1",
          "block69_swin_body_1")
SPV_SHA = "1476290831c4955ac01a7b8ea214f7cf472091c5cc047ff54df43340bd8d7de1"
DUMP = R / "run/int8-coverage-v2-20261020/C/c32-i8-diag-b1-eact-001/diag_b1_debug.bin"
DUMP_RECEIPT = R / "run/int8-coverage-v2-20261020/C/c32-i8-diag-b1-eact-001/port_network_receipt.json"
DUMP_LOCK = R / "run/codex-sol-int8-long-008-20261019/exports/c32-i8-diag-b1-eact-001.lock-receipt.json"
TOKEN = re.compile(r'"[^"]*"|\S+')
TILE = 1024          # bytes per 8-output-channel I8A K32 tile
PROJ = 4096          # bytes per projection (4 tiles)
WQ_BASE = 32768      # byte offset of the appended QKV region inside w8


def sha(p):
    return hashlib.sha256(Path(p).read_bytes()).hexdigest()


def ident(p):
    p = Path(p)
    return {"path": p.as_posix(), "bytes": p.stat().st_size, "sha256": sha(p)}


def parse_plan(path):
    lines = Path(path).read_text(encoding="utf-8").splitlines()
    h = lines[0].split()
    nb, npass, nout = int(h[2]), int(h[3]), int(h[4])
    bufs = []
    for i in range(nb):
        t = TOKEN.findall(lines[1 + i])
        bufs.append({"name": t[1].strip('"'), "bytes": int(t[2]),
                     "path": t[3].strip('"')})
    rest = lines[1 + nb:]
    passes, i = [], 0
    for _ in range(npass):
        t = TOKEN.findall(rest[i])
        nbind = int(t[6])
        binds = []
        for q in range(nbind):
            w = rest[i + 1 + q].split()
            binds.append({"binding": int(w[1]), "resource": int(w[2]),
                          "offset": int(w[3]), "bytes": int(w[4])})
        push = [int(x) if int(x) < 2 ** 31 else int(x) - 2 ** 32
                for x in rest[i + 1 + nbind].split()[1:]]
        passes.append({"name": t[2].strip('"'), "spv": t[3].strip('"'),
                       "row": int(t[1]), "shared": int(t[5]), "nbind": nbind,
                       "npush": int(t[7]), "grid": [int(t[8]), int(t[9]), int(t[10])],
                       "binds": binds, "push": push, "line": i + 1 + nb})
        i += 1 + nbind + 1
    return {"header": h, "buffers": bufs, "passes": passes}


def q_addr(n, k):
    """parent shader address of FP16 weight (n output channel, k input channel).

    coopMatLoad(w, wb, p1.z + base, 16, RowMajor) with
    base = (n0//16)*512 + (k_half)*256 + (n0%16)*16, so element (n,k) sits at
    plane + (n//16)*512 + (k//16)*256 + (n%16)*16 + (k%16).
    """
    return (n // 16) * 512 + (k // 16) * 256 + (n % 16) * 16 + (k % 16)


def gather(wplane, n=32, k=32):
    a = np.empty((n, k), np.int32)
    for i in range(n):
        for j in range(k):
            a[i, j] = q_addr(i, j)
    assert sorted(a.ravel().tolist()) == list(range(1024)), "address map not a bijection"
    return wplane[a]


def quant_row(w):
    """per output channel symmetric quantiser, identical to the C32 INT8 family"""
    amax = np.max(np.abs(w), axis=1).astype(np.float32)
    s = (amax / np.float32(127)).astype(np.float32)
    dead = np.where(s == 0)[0]
    q = np.zeros(w.shape, np.int8)
    live = s > 0
    if dead.size:
        assert not np.any(w[dead]), "zero scale but non-zero row"
    q[live] = np.clip(np.rint(w[live].astype(np.float32)
                              * (np.float32(1) / s[live, None])),
                      -127, 127).astype(np.int8)
    return q, s, [int(x) for x in dead]


def tiles_to_bytes(q):
    """pack (32,32) int8 into the kernel's 12-tile addressing -> 4096-4*1024 B"""
    buf = np.zeros((4, TILE), np.uint8)
    for g in range(4):
        for m in range(8):
            n = 8 * g + m
            for k in range(32):
                buf[g, k * 32 + m] = np.uint8(q[n, k])
    return buf


def bytes_to_q(buf):
    q = np.zeros((32, 32), np.int8)
    for g in range(4):
        for m in range(8):
            n = 8 * g + m
            for k in range(32):
                v = int(buf[g, k * 32 + m])
                q[n, k] = np.int8(v - 256 if v > 127 else v)
    return q


def relerr(q, s, wref):
    dq = q.astype(np.float32) * s[:, None].astype(np.float32)
    err = np.abs(dq - wref.astype(np.float32))
    return {"max_abs": float(err.max()),
            "rms": float(np.sqrt((err ** 2).mean())),
            "rel_l2": float(np.linalg.norm(err) / max(np.linalg.norm(wref), 1e-30)),
            "bound_ok": bool(err.max() <= s.max() / 2 + 1e-6)}


def main():
    PACK.mkdir(parents=True, exist_ok=True)
    EV.mkdir(parents=True, exist_ok=True)
    plan = parse_plan(PLAN)
    by_name = {p["name"]: p for p in plan["passes"]}
    assert sha(PLAN)
    out = {"label": "R02-QKV-INT8-PACK-AND-ORACLE",
           "plan": ident(PLAN),
           "spv_sha256_expected": SPV_SHA,
           "passes": {}}
    y_cache = {}
    for name in PASSES:
        p = by_name[name]
        assert p["spv"].endswith("c32_body1_i8.spv"), p["spv"]
        push = p["push"]
        p1z, p2x, p2y = push[6], push[8], push[9]
        wbuf = plan["buffers"][p["binds"][0]["resource"]]
        i8buf = plan["buffers"][p["binds"][5]["resource"]]
        scbuf = plan["buffers"][p["binds"][6]["resource"]]
        w = np.fromfile(wbuf["path"], dtype="<f2")
        assert w.size * 2 == wbuf["bytes"], (w.size, wbuf["bytes"])
        rec = {"pass_row": p["row"], "spv": p["spv"], "shared_bytes": p["shared"],
               "grid": p["grid"], "push": push,
               "p1_z_qkv_weights": p1z, "p2_x_contract_seed": p2x,
               "p2_y_output_seed": p2y,
               "weights_buffer": dict(wbuf, sha256=sha(wbuf["path"])),
               "int8_weights_buffer": dict(i8buf, sha256=sha(i8buf["path"])
                                           if Path(i8buf["path"]).is_file() else None),
               "int8_scales_buffer": dict(scbuf, sha256=sha(scbuf["path"])
                                          if Path(scbuf["path"]).is_file() else None)}
        planes = {}
        out_bytes = np.zeros(3 * PROJ, np.uint8)
        wscale = {}
        for pi, proj in enumerate(("Q", "K", "V")):
            plane = w[p1z + pi * 1024: p1z + pi * 1024 + 1024]
            assert plane.size == 1024, (name, proj, plane.size)
            mat = gather(plane)
            q, s, dead = quant_row(mat)
            buf = tiles_to_bytes(q)
            out_bytes[pi * PROJ:(pi + 1) * PROJ] = buf.reshape(-1)
            back = bytes_to_q(buf)
            assert np.array_equal(back, q), "tile round trip failed"
            wscale[proj] = {"scale": [float(x) for x in s], "dead_rows": dead,
                            "relerr": relerr(q, s, mat),
                            "amax": [float(x) for x in np.max(np.abs(mat), axis=1)]}
            planes[proj] = mat
        wp = PACK / ("c32_%s_qkvweights.i8.bin" % name.split("_")[0])
        wp.write_bytes(out_bytes.tobytes())
        rec["qkv_weights"] = dict(ident(wp), layout={
            "region": "w8[%d .. %d)" % (WQ_BASE, WQ_BASE + 3 * PROJ),
            "tile": "1024 B per 8 output channels, code(n=8g+m, k) at byte k*32+m",
            "order": "Q[%d..%d) K[%d..%d) V[%d..%d)" % (
                WQ_BASE, WQ_BASE + PROJ, WQ_BASE + PROJ, WQ_BASE + 2 * PROJ,
                WQ_BASE + 2 * PROJ, WQ_BASE + 3 * PROJ)},
            roundtrip_ok=True)
        rec["weight_scales"] = wscale
        rec["int8_w8_end"] = WQ_BASE + 3 * PROJ
        out["passes"][name] = rec
        y_cache[name] = planes

    # ---------------------------------------------------------------- oracle
    assert DUMP.is_file(), DUMP
    dbg = np.fromfile(DUMP, dtype="<f2")
    y = np.zeros((4, 16, 32), np.float32)      # [sg][token][channel]
    for sg in range(4):
        for t in range(16):
            base = 8192 + sg * 512 + t * 32
            y[sg, t, :] = dbg[base:base + 32].astype(np.float32)
    yhalf = np.zeros((4, 16, 32), np.float16)
    for sg in range(4):
        for t in range(16):
            base = 8192 + sg * 512 + t * 32
            yhalf[sg, t, :] = dbg[base:base + 32]
    planes = y_cache[PASSES[0]]
    oracle = {"dump": ident(DUMP), "dump_receipt": ident(DUMP_RECEIPT),
              "dump_lock_receipt": ident(DUMP_LOCK),
              "dump_lock_argv_plan": json.loads(
                  Path(DUMP_LOCK).read_text(encoding="utf-8"))["argv"][2],
              "activation_layout": "dbg[8192 + sg*512 + t*32 + c] = Y[c,t] (half)",
              "activation_alive": bool(np.any(y != 0)),
              "activation_absmax": float(np.abs(y).max()),
              "per_projection": {}}
    inv_y = float(np.float32(127) / np.float32(np.abs(y).max()))
    s_y = np.float32(1) / np.float32(inv_y)
    qy = np.clip(np.rint(yhalf.astype(np.float32) * inv_y), -127, 127)
    qy_i32 = qy.astype(np.int32)
    oracle["inv_y"] = inv_y
    oracle["saturation_fraction"] = float(np.mean(np.abs(qy) >= 127))
    for proj in ("Q", "K", "V"):
        mat = planes[proj]
        q, s, dead = quant_row(mat)
        qi = q.astype(np.int32)
        # FP16 reference arm: half x half -> float32 accumulate, narrow to half
        ref = np.empty((32, 4, 16), np.float32)
        acc32 = np.empty((32, 4, 16), np.int32)
        for sg in range(4):
            yt = yhalf[sg].astype(np.float32)          # (16 tokens, 32 channels)
            ref[:, sg, :] = mat.astype(np.float32) @ yt.T
            a32 = qi @ qy_i32[sg].T                    # (32 out, 16 tokens)
            a64 = qi.astype(np.int64) @ qy_i32[sg].T.astype(np.int64)
            assert np.array_equal(a32, a64), "int32 accumulation not exact"
            acc32[:, sg, :] = a32
        ref16 = ref.astype(np.float16)
        dq_factor = (s.astype(np.float32) * np.float32(s_y))[:, None, None]
        real = acc32.astype(np.float32) * dq_factor
        int8_16 = real.astype(np.float16)
        diff = int8_16.astype(np.float32) - ref16.astype(np.float32)
        rms_ref = float(np.sqrt((ref16.astype(np.float32) ** 2).mean()))
        rms_err = float(np.sqrt((diff ** 2).mean()))
        oracle["per_projection"][proj] = {
            "weight_scale_amax": float(np.abs(mat).max()),
            "dequant": {
                "dq_min": float(s.min() * s_y), "dq_max": float(s.max() * s_y)},
            "error_vs_fp16_reference": {
                "max_abs": float(np.abs(diff).max()),
                "rms": rms_err,
                "rel_l2": float(np.linalg.norm(diff) / max(np.linalg.norm(
                    ref16.astype(np.float32)), 1e-30)),
                "snr_db": float(20 * np.log10(rms_ref / max(rms_err, 1e-30))),
            },
            "reference_absmax": float(np.abs(ref16.astype(np.float32)).max()),
            "nonzero": bool(np.any(ref16) and np.any(int8_16)),
        }
    # block1 dequant table for the 96 channels + invY, in the candidate's slots.
    # The parent's own 164 entries are copied verbatim from the deployed scale
    # buffer: the candidate kernel still runs the parent expand/contract, so those
    # entries must stay byte-identical.
    p1 = by_name[PASSES[0]]
    parent_scales = np.fromfile(plan["buffers"][p1["binds"][6]["resource"]]["path"],
                                dtype="<f4")
    assert parent_scales.size == 164, parent_scales.size
    dq = np.zeros(261, np.float32)
    dq[:164] = parent_scales
    dq[260] = np.float32(inv_y)
    for pi, proj in enumerate(("Q", "K", "V")):
        _, s, _ = quant_row(planes[proj])
        dq[164 + 32 * pi: 164 + 32 * (pi + 1)] = s * np.float32(s_y)
    sp = PACK / "c32_block1_qkvscales.f32.bin"
    sp.write_bytes(dq.tobytes())
    oracle["parent_scale_entries_preserved"] = bool(np.array_equal(
        dq[:164], parent_scales))
    oracle["scale_file"] = dict(ident(sp), layout={
        "0..163": "parent entries copied verbatim from the deployed scale buffer",
        "164..195": "sY*sWq[32] dequant", "196..227": "sY*sWk[32]",
        "228..259": "sY*sWv[32]", "260": "invY (activation quantiser)"})
    oracle["note"] = (
        "Y is a real activation tile captured by the deployed parent SPV (the diag "
        "plan that binds it); the FP16 arm is a float32 accumulation reference, not "
        "a bit-exact hardware replay, so the reported error is the INT8 quantisation "
        "error of this arm and not a GPU gate result")
    out["oracle"] = oracle
    out["uncalibrated_blocks"] = [n for n in PASSES[1:]]
    (EV / "QKV-PACK-ORACLE.json").write_text(
        json.dumps(out, ensure_ascii=False, indent=1) + "\n", encoding="utf-8")
    print(json.dumps({
        "pack": {n: {"sha256": out["passes"][n]["qkv_weights"]["sha256"],
                     "bytes": out["passes"][n]["qkv_weights"]["bytes"],
                     "q_rel_l2": out["passes"][n]["weight_scales"]["Q"]["relerr"]["rel_l2"],
                     "k_rel_l2": out["passes"][n]["weight_scales"]["K"]["relerr"]["rel_l2"],
                     "v_rel_l2": out["passes"][n]["weight_scales"]["V"]["relerr"]["rel_l2"]}
                 for n in PASSES},
        "oracle": {p: out["oracle"]["per_projection"][p]["error_vs_fp16_reference"]
                   for p in ("Q", "K", "V")},
        "activation_absmax": out["oracle"]["activation_absmax"],
        "saturation_fraction": out["oracle"]["saturation_fraction"],
    }, indent=1))


if __name__ == "__main__":
    main()
