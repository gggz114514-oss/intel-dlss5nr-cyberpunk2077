"""drill into CURRENT records whose native hash disagrees with the majority."""
import json
from pathlib import Path
from collections import Counter, defaultdict

CARD = Path(__file__).parent
j = json.loads((CARD / 'evidence' / 'NATIVE-INDEX.json').read_text(encoding='utf-8'))

TARGETS = [int(x) for x in (len(__import__('sys').argv) > 1 and __import__('sys').argv[1:]) or [24, 0, 5]]

# majority hash per pass
maj = {}
for o in TARGETS:
    c = Counter()
    for cap in j['captures']:
        for e in cap['records']:
            if o in (e.get('pass_ordinals') or []) and e['identity'] == 'CURRENT':
                c[e['native_hash']] += 1
    maj[o] = c.most_common(1)[0][0]
    print('\n#### ordinal %d %s  majority=%s  %s' % (o, j['coverage'][o]['name'], maj[o], dict(c)))
    for cap in j['captures']:
        for e in cap['records']:
            if o in (e.get('pass_ordinals') or []) and e['identity'] == 'CURRENT' \
               and e['native_hash'] != maj[o]:
                print('  hash=%s  mode=%s  npass=%s' % (e['native_hash'], cap['mapping_mode'],
                                                        (cap['plan'] or {}).get('n_passes')))
                print('     capture:', cap['capture_root'])
                print('     spv    :', (e['spv'] or '')[:150], (e['spv_sha256_at_capture'] or '')[:12])
                print('     ords   :', e.get('pass_ordinals'), e.get('pass_names'))
                print('     mtime  :', e['mtime'], 'bytes', e['bytes'])
                print('     reason :', e['identity_reason'])
