def widths(path):
    data=open(path,'rb').read(); n=len(data)//4
    w=struct.unpack("<%dI"%n, data[:n*4]); i=5; fl=collections.Counter(); it=collections.Counter()
    while i<n:
        wc=w[i]>>16; op=w[i]&0xFFFF
        if wc==0: break
        ops=w[i+1:i+wc]
        if op==22: fl[ops[0]]=ops[0]
        i+=wc
    # second pass: collect OpTypeFloat widths
    i=5; out=collections.Counter()
    while i<n:
        wc=w[i]>>16; op=w[i]&0xFFFF
        if wc==0: break
        ops=w[i+1:i+wc]
        if op==22: out[ops[0]]+=1
        i+=wc
    return sorted(out)
