"""n23-f05-shape / B1: build candidate.comp from the parent source.

Four edits, all mechanical, on the measured NESTED law (probe: 8x16 control 128/128, 8x32
NESTED 256/256, runner-up 32/256):

  E1  three type aliases: f32 8x32 accumulator, f16 8x32 accumulator, 16x32 B fragment;
  E2  `nr_norm_qk_pair` (two matrices, component i of each) becomes `nr_norm_qk32` (one matrix,
      components i and i+4): under NESTED, merged component i holds exactly what the old q0[i] held
      and component i+4 exactly what q1[i] held, so the sealed tree, its term order, its rounding
      points and the scale products are the parent's, term for term;
  E3  the six QKV weight B loads become three 16x32 loads on the 32-channel-blocked pack, and the
      six fp32 8x16 accumulators become three fp32 8x32 accumulators (48 GRF both ways);
  E4  six narrowings -> three, six publish stores -> three, eight debug stores -> four; every
      written byte is unchanged (Q/V regions are [token][32 dims] with a 32-half row stride and the
      K region is contiguous across the two 16-dim halves at a stride of 8).

Nothing else is touched: the score stage, the denominator tree, the PV stage, the barrier and every
other fp32 expression stay byte-identical, so the NoContraction set must stay identical (73).
"""

E2_NEW = """// n23-f05-shape B1.  The two dim-halves now live inside one 8x32 accumulator.  The measured lane
// law (probe, NESTED: 8x32 components 0..3 are the 8x16 columns 0..15, components 4..7 are columns
// 16..31 with the same row law) makes merged component i hold exactly the element the old q0[i]
// held and component i+4 exactly the old q1[i], so this is the parent's sealed tree with the pair
// (i, i+4) inside one matrix instead of (i of q0, i of q1) across two.  Every add, multiply and
// half rounding point is the same expression in the same order.
void nr_norm_qk32(inout MAT_H32 q, inout MAT_H32 k, float16_t scale, uint lane) {
    { f16vec2 n=nr_norm_component(f16vec2(q[0],k[0]),f16vec2(q[4],k[4]),lane); q[0]=float16_t(q[0]*(n.x*scale)); q[4]=float16_t(q[4]*(n.x*scale)); k[0]=float16_t(k[0]*n.y); k[4]=float16_t(k[4]*n.y); }
    { f16vec2 n=nr_norm_component(f16vec2(q[1],k[1]),f16vec2(q[5],k[5]),lane); q[1]=float16_t(q[1]*(n.x*scale)); q[5]=float16_t(q[5]*(n.x*scale)); k[1]=float16_t(k[1]*n.y); k[5]=float16_t(k[5]*n.y); }
    { f16vec2 n=nr_norm_component(f16vec2(q[2],k[2]),f16vec2(q[6],k[6]),lane); q[2]=float16_t(q[2]*(n.x*scale)); q[6]=float16_t(q[6]*(n.x*scale)); k[2]=float16_t(k[2]*n.y); k[6]=float16_t(k[6]*n.y); }
    { f16vec2 n=nr_norm_component(f16vec2(q[3],k[3]),f16vec2(q[7],k[7]),lane); q[3]=float16_t(q[3]*(n.x*scale)); q[7]=float16_t(q[7]*(n.x*scale)); k[3]=float16_t(k[3]*n.y); k[7]=float16_t(k[7]*n.y); }
}
"""

def main():
    src = PARENT.read_text(encoding='utf-8')
    for name, old, new in (('E1', E1_OLD, E1_NEW), ('E2', E2_OLD, E2_NEW),
                           ('E3', E3_OLD, E3_NEW), ('E4', E4_OLD, E4_NEW)):
        assert src.count(old) == 1, (name, src.count(old))
        src = src.replace(old, new)
    out = CARD / 'candidate.comp'
    out.write_text(src, encoding='utf-8')
    print('wrote', out, len(src), 'chars; edited 4 regions of', PARENT)


if __name__ == '__main__':
    main()
