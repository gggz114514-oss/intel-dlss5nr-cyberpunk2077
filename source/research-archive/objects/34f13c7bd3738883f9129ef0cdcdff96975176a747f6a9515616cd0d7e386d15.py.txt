def load(p):
    return json.loads(Path(p).read_text(encoding='utf-8'))


def patch():
    a = (CARD / 'parent-rebuild.comp').read_text(encoding='utf-8').splitlines(keepends=True)
    b = (CARD / 'candidate.comp').read_text(encoding='utf-8').splitlines(keepends=True)
    d = list(difflib.unified_diff(a, b, fromfile='a/candidate.comp',
                                  tofile='b/candidate.comp', n=6))
    (CARD / 'SOURCE-DIFF.patch').write_text(''.join(d), encoding='utf-8')
    added = sum(1 for l in d if l.startswith('+') and not l.startswith('+++'))
    removed = sum(1 for l in d if l.startswith('-') and not l.startswith('---'))
    return dict(hunks=sum(1 for l in d if l.startswith('@@')),
                added_lines=added, removed_lines=removed,
                bytes=len(''.join(d).encode('utf-8')))
