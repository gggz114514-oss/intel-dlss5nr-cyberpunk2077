"""Independent verification of BUFFER-EDITS.json (main-agent ruling condition 2).

Applies the 12 recorded BUFFER rows to a copy of the chain-head plan, then checks:
  * every original_line still matches the head plan at the recorded line,
  * every new file exists, has the recorded sha256 and the recorded byte size,
  * the buffer count, buffer sizes and pool bytes are unchanged,
  * the edited plan still passes clock_runner --cpu-check (structural only: the shader half that
    reads the permuted layout is not built yet, so this plan is NOT a candidate).

Writes build/BUFFER-EDITS-VERIFY.json and build/plan-with-buffer-edits.txt.
"""
import json
import sys
from pathlib import Path

CARD = Path(__file__).parent
sys.path.insert(0, str(CARD.parent.parent))
import COMMON as K            # noqa: E402
import planio                 # noqa: E402


def sha(p):
    return K.sha(Path(p))


def main():
    bed = K.read_json(CARD / 'BUFFER-EDITS.json')
    perm = K.read_json(CARD / 'build' / 'PERMUTATION-PROOF.json')
    plan = Path(bed['applies_to_plan'])
    assert sha(plan) == bed['applies_to_plan_sha256'], 'head plan moved: re-run reblock.py'
    lines = plan.read_text(encoding='utf-8-sig').splitlines(True)
    head_buf = [l for l in lines if l.startswith('BUFFER ')]
    res = {'card': 'n23-f05-shape', 'plan': str(plan), 'plan_sha256': sha(plan),
           'checks': {}, 'edits': []}
    for e in bed['edits']:
        i = e['buffer_line']
        ok_line = lines[i].rstrip('\n') == e['original_line']
        nf = Path(e['new_file'])
        ok_file = nf.exists()
        ok_sha = ok_file and sha(nf) == e['new_sha256']
        ok_bytes = ok_file and nf.stat().st_size == e['new_bytes'] == e['source_bytes']
        if ok_line:
            lines[i] = e['new_line'] + '\n'
        res['edits'].append(dict(resource_index=e['resource_index'], buffer_line=i,
                                 name=e['name'], original_line_matched=ok_line,
                                 new_file_exists=ok_file, new_sha_matched=ok_sha,
                                 bytes_matched=ok_bytes))
    out = CARD / 'build' / 'plan-with-buffer-edits.txt'
    out.write_text(''.join(lines), encoding='utf-8')
    new_buf = [l for l in lines if l.startswith('BUFFER ')]
    py, psha, _ = K.current_best()
    old = [l for l in Path(py).read_text(encoding='utf-8-sig').splitlines(True)
           if l.startswith('BUFFER ')]
    sizes_old = [l.split()[2] for l in old]
    sizes_new = [l.split()[2] for l in new_buf]
    cc = K.cpu_check(out)
    (CARD / 'build' / 'cpu-check-with-buffer-edits.txt').write_text(cc, encoding='utf-8')
    res['checks'] = dict(
        edits=len(bed['edits']),
        all_original_lines_matched=all(x['original_line_matched'] for x in res['edits']),
        all_files_present_and_hashed=all(x['new_sha_matched'] for x in res['edits']),
        all_sizes_unchanged=all(x['bytes_matched'] for x in res['edits']),
        buffer_rows_before=len(head_buf), buffer_rows_after=len(new_buf),
        buffer_count_unchanged=len(head_buf) == len(new_buf),
        buffer_sizes_unchanged=sizes_old == sizes_new,
        cpu_check_ok=cc.startswith('CPU_PLAN_OK'),
        cpu_check_output=cc.strip(),
        structural_only=True,
        note='cpu-check validates plan/binding/pool/grid/lifetime structure only; the shader half '
             'that reads the 32-channel blocking is not built yet, so this plan is not a candidate',
        permutation_proof=perm['totals'],
    )
    K.write_json(CARD / 'build' / 'BUFFER-EDITS-VERIFY.json', res)
    print(json.dumps(res['checks'], indent=2))


if __name__ == '__main__':
    main()
