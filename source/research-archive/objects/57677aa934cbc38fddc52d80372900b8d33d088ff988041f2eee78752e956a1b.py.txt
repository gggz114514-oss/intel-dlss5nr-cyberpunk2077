def sha(p):
    h = hashlib.sha256()
    with open(p, "rb") as f:
        for b in iter(lambda: f.read(1 << 20), b""):
            h.update(b)
    return h.hexdigest()


for k, p in files.items():
    if not os.path.exists(p):
        print(f"{k}: MISSING {p}")
        continue
    a = np.fromfile(p, dtype=np.float16)
    fin = np.isfinite(a)
    print(f"{k}: {p.split('/')[-1]} bytes={os.path.getsize(p)} half={a.size} "
          f"min={np.nanmin(a):.6g} max={np.nanmax(a):.6g} zero_frac={(a==0).mean():.4f} "
          f"nonfinite={(~fin).sum()} sha={sha(p)[:16]}")
