def sha(p):
    return hashlib.sha256(Path(p).read_bytes()).hexdigest()


def run(argv):
    r = subprocess.run([str(a) for a in argv], capture_output=True, text=True,
                       encoding='utf-8', errors='replace')
    return r.returncode, (r.stdout or '').replace('\r\n', '\n'), (r.stderr or '').replace('\r\n', '\n')


def glslang(src, out, includes):
    argv = [GLSLANG, '-V', '--target-env', 'vulkan1.3']
    for d in includes:
        argv.append('-I' + str(d))
    argv += [str(src), '-o', str(out)]
    rc, so, se = run(argv)
    if rc != 0:
        raise SystemExit('glslang failed (%d)\n%s\n%s' % (rc, so, se))
    return [str(a) for a in argv], se


def validate(spv):
    rc, so, se = run([K.VALIDATE, str(spv)])
    return dict(rc=rc, stdout=so.strip(), stderr=se.strip())


def stage_headers():
    """Copy the two pinned headers into the card so the build is self-contained."""
    src_dir = PARENT_SRC.parent
    out = {}
    for name, pin in HEADER_PINS.items():
        s = src_dir / name
        got = sha(s)
        assert got == pin, 'header pin mismatch %s %s != %s' % (name, got, pin)
        d = CARD / name
        if not d.exists() or sha(d) != pin:
            shutil.copyfile(s, d)
        out[name] = dict(path=str(d), sha256=sha(d), pin=pin, pin_ok=sha(d) == pin,
                         origin=str(s))
    return out


def step_parent():
    assert sha(PARENT_SRC) == PARENT_SRC_SHA
    argv, se = glslang(PARENT_SRC, CARD / 'parent-rebuild.spv', [PARENT_SRC.parent])
    got = sha(CARD / 'parent-rebuild.spv')
    print('parent rebuild sha %s  match=%s' % (got, got == PARENT_SPV_SHA))
    return dict(argv=argv, stderr=se, sha256=got, byte_match=got == PARENT_SPV_SHA,
                target_sha256=PARENT_SPV_SHA)


def step_candidate():
    argv, se = glslang(CARD / 'candidate.comp', CARD / 'candidate.spv', [CARD])
    return dict(argv=argv, stderr=se, sha256=sha(CARD / 'candidate.spv'),
                bytes=(CARD / 'candidate.spv').stat().st_size)

def _regions(m):
    """Split module instructions into (header, [function bodies])."""
    funcs = []
    cur = None
    for op, ops, _, _ in m.instructions:
        if op == 54:                       # OpFunction
            cur = []
            funcs.append(cur)
            cur.append(op)
            continue
        if cur is not None:
            cur.append(op)
        if op == 56:                       # OpFunctionEnd
            cur = None
    return funcs


def spv_statics(path):
    m = K.ST.Module(Path(path))
    from collections import Counter
    c = Counter()
    for op, ops, _, _ in m.instructions:
        c[op] += 1
    funcs = _regions(m)
    body = Counter()
    for f in funcs:
        for op in f:
            body[op] += 1
    nc = sorted(K.ST.no_contraction_ids(m))
    fma = m.ext_inst_fma_ordinals()
    code = m.code if hasattr(m, 'code') else None
    return dict(words=len(m.words), bytes=Path(path).stat().st_size,
                instruction_count=len(m.instructions),
                function_count=len(funcs),
                body_instruction_count=sum(body.values()),
                opcodes_total={str(k): v for k, v in sorted(c.items())},
                opcodes_body={str(k): v for k, v in sorted(body.items())},
                opcodes_body_named={SPV_NAMES.get(k, 'op%d' % k): v for k, v in sorted(body.items())},
                opcodes_total_named={SPV_NAMES.get(k, 'op%d' % k): v for k, v in sorted(c.items())},
                no_contraction_count=len(nc), no_contraction_ids=nc,
                ext_inst_fma_sites=len(fma), ext_inst_fma_ordinals=fma,
                code=code)

    p, c = res['parent'], res['candidate']

    print('instr     %6d -> %6d' % (p['instruction_count'], c['instruction_count']))
    print('body      %6d -> %6d' % (p['body_instruction_count'], c['body_instruction_count']))
    print('NC        %6d -> %6d' % (p['no_contraction_count'], c['no_contraction_count']))
    print('Fma ext   %6d -> %6d' % (p['ext_inst_fma_sites'], c['ext_inst_fma_sites']))
    return res


def step_nc():
    """NoContraction: which operations carry it, and are the two arms one-to-one?"""
    out = {}
    for tag, path in (('parent', CARD / 'parent-rebuild.spv'),
                      ('candidate', CARD / 'candidate.spv')):
        m = K.ST.Module(Path(path))
        rows = []
        for cid in sorted(K.ST.no_contraction_ids(m)):
            op, ops = m.defining_instruction(cid)
            rows.append(dict(result_id=cid, opcode=op,
                             opcode_name=SPV_NAMES.get(op, 'op%d' % op),
                             signature=m.operand_signature(cid)))
        from collections import Counter
        fma = m.ext_inst_fma_ordinals()
        fma_sigs = [m.fma_signature(x) for x in fma] if fma else []
        out[tag] = dict(count=len(rows), sites=rows,
                        fma_ext_inst=len(fma), fma_signatures=fma_sigs,
                        signature_multiset=sorted(
                            '%s|%s' % (r['opcode_name'], r['signature']) for r in rows))
    out['one_to_one'] = (out['parent']['signature_multiset']
                         == out['candidate']['signature_multiset'])
    out['float_NC_checked'] = out['parent']['count']
    (EVID / 'NC-MAP.json').write_text(json.dumps(out, indent=2), encoding='utf-8')
    print(json.dumps({k: (v if k not in ('parent', 'candidate') else
                          {kk: vv for kk, vv in v.items() if kk != 'sites'})
                      for k, v in out.items()}, indent=2))
    return out
