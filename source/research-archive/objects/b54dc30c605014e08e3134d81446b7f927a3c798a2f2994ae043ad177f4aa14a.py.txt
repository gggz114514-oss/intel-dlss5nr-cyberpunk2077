"""Independent audit of the accepted F10 baseline result (CPU only).

Recomputes the 540-sample three-arm comparison from the raw receipts and checks
identity, gold frames, sample counts and the reported CI. Writes
evidence/AUDIT-F10-BASELINE.json
"""
import json, math, statistics, sys
from pathlib import Path
sys.path.insert(0, str(Path(__file__).parent))
import COMMON as K

S = K.D / 'f10-weight-return-separation-20261027'
task = K.read_json(S / 'LUNA-TRIPLET.json')
reported = K.read_json(S / 'RESULTS.json')
T975_5 = 2.570581835636305

jobs = {j['id']: j for j in task['jobs']}
triplets, problems = [], []
for n in range(6):
    arms = {}
    for arm in 'ABC':
        j = jobs['triplet-%d-%s' % (n, arm)]
        p = Path(j['runner_output'])
        rec = K.read_json(p / 'port_network_receipt.json')
        v = rec['gpu_samples_ms']
        exp = j.get('expected_fixed_frame_out') or j.get('expected_frame_out')
        fsha = K.sha(p / 'frame_out.bin')
        ok = (rec['normal_close'] and rec['nan'] == 0 and rec['inf'] == 0
              and rec['gpu_passes'] == 232 and len(v) == 30 and fsha == exp
              and K.sha(j['plan']['path']) == j['plan']['sha256'])
        if not ok:
            problems.append(dict(pair=n, arm=arm, gpu_passes=rec['gpu_passes'], n_samples=len(v),
                                 frame_gold=fsha == exp, normal_close=rec['normal_close']))
        arms[arm] = dict(mean_ms=statistics.mean(v), samples=v, plan_sha=K.sha(j['plan']['path']),
                         frame_sha=fsha, ok=ok)
    triplets.append(dict(index=n, arms=arms))

comp = {}
for x, y, name in (('B', 'C', 'B_minus_C_recovery'), ('A', 'C', 'A_minus_C_net'), ('A', 'B', 'A_minus_B_layout_cost')):
    d = [t['arms'][x]['mean_ms'] - t['arms'][y]['mean_ms'] for t in triplets]
    m = statistics.mean(d)
    h = T975_5 * statistics.stdev(d) / math.sqrt(6)
    comp[name] = dict(saving_ms=m, CI95_ms=[m - h, m + h],
                      status='FASTER_95' if m - h > 0 else 'SLOWER_95' if m + h < 0 else 'INDISTINCT',
                      diffs=d)

means = {a: statistics.mean(t['arms'][a]['mean_ms'] for t in triplets) for a in 'ABC'}
n_samples = sum(len(t['arms'][a]['samples']) for t in triplets for a in 'ABC')

# identity of the three arms
plans = {}
for a in 'ABC':
    p = jobs['triplet-0-' + a]['plan']['path']
    plans[a] = dict(path=p, sha256=K.sha(p))

# reported values for comparison
rep_comp = {}
try:
    r = reported
    def dig(o, *ks):
        for k in ks:
            if isinstance(o, dict) and k in o:
                o = o[k]
            else:
                return None
        return o
    rep_comp = dict(raw_keys=sorted(reported.keys())[:20])
except Exception as e:
    rep_comp = dict(error=str(e))

out = dict(
    method='Independent recomputation from LUNA-TRIPLET.json receipts: 6 balanced ABC permutations, 30 samples/arm, 540 samples, t df5 two-sided 95%, no trimming.',
    samples_total=n_samples, triplets=triplets, means_ms=means, comparisons=comp,
    arm_plans=plans, receipt_problems=problems,
    reported_file=str(S / 'RESULTS.json'), reported=rep_comp,
)
K.write_json(K.C / 'evidence' / 'AUDIT-F10-BASELINE.json', out)
print(json.dumps(dict(samples_total=n_samples, means_ms=means,
                      comparisons={k: dict(saving_ms=v['saving_ms'], CI95_ms=v['CI95_ms'], status=v['status'])
                                   for k, v in comp.items()},
                      receipt_problems=problems), indent=2))
