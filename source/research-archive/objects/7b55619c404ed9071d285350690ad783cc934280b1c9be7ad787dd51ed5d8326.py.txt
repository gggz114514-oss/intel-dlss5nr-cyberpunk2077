"""S09 diagnostic 2: find the live-row count in the ViT dumps."""
import os
import numpy as np

DUMPS = r"@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/int8-quality-first-20261029/real-int8-24/dump-fp16"
TOK, HEADS, HD = 288, 32, 32


def load(block, tag, n=None):
    a = np.fromfile(os.path.join(DUMPS, "dump_vit_block%d_%s.bin" % (block, tag)), dtype=np.float16)
    return a if n is None else a.reshape(n)


a = load(31, "QKV").reshape(TOK, 3072)
allz = np.all(a == 0, axis=1)
print("all-zero rows:", int(allz.sum()), "first:", int(np.argmax(allz)) if allz.any() else -1)
idx = np.where(allz)[0]
print("zero row indices:", idx[:8].tolist(), "...", idx[-8:].tolist() if idx.size else [])
for tag in ("T", "Y"):
    y = load(31, tag).reshape(TOK, 1024)
    z = np.all(y == 0, axis=1)
    iz = np.where(z)[0]
    print(tag, "all-zero rows:", int(z.sum()), "first:", iz[:6].tolist())
A = load(31, "A")
print("A size", A.size, A.size // 1024)
Aq = A.reshape(288, 4, 1024) if A.size == 288 * 4 * 1024 else None
if Aq is not None:
    nz = np.array([np.all(Aq[i] == 0) for i in range(288)])
    print("A blocks all-zero rows (per 288):", int(nz.sum()), np.where(nz)[0][:6].tolist())
