"""Detail pass for task B: for the 38 shared-source coopMatLoad sites (304 `load.slm.d32x8t`
messages) report the address dependency depth, its opcode cone and the literal instruction
text, so the deepest chain can be named rather than guessed.

Read-only.
"""
import re, sys, json, collections
from pathlib import Path

sys.stdout.reconfigure(encoding='utf-8')
sys.path.insert(0, str(Path(__file__).parent))
from addr_cones import ISA, parse, build_writers, slice_depth, cone, regs, single_reg

ins = parse(ISA)
writers = build_writers(ins)
loads = [i for i, x in enumerate(ins)
         if x['op'].startswith('send') and x['msg'] and 'load' in x['msg']]

SLM_FRAGMENT = 'load.slm.d32x8t'
slm = [i for i in loads if SLM_FRAGMENT in ins[i]['msg']]
print('load.slm.d32x8t messages:', len(slm), '(38 sites x 8 messages)')

rows = []
for i in slm:
    rows.append((slice_depth(ins, writers, i), len(cone(ins, writers, i)), i))
hist = collections.Counter(r[0] for r in rows)
print('depth histogram (slm d32x8t):', dict(sorted(hist.items())))

# the barrier region index of every instruction (to separate prologue from per-site ops)
bars = [i for i, x in enumerate(ins) if x['op'] == 'sync.bar']


def region_of(i):
    n = 0
    for b in bars:
        if i > b:
            n += 1
    return n


print('\n=== deepest slm d32x8t sites ===')
rows.sort(key=lambda r: (-r[0], r[1]))
shown = 0
for d, c, i in rows:
    if shown >= 4:
        break
    if shown and rows[shown - 1][0] != d:
        break
    shown += 1
    print('\n--- depth=%d cone=%d line=%d addr=%s region=%d' %
          (d, c, ins[i]['ln'], hex(ins[i]['addr']), region_of(i)))
    for j in sorted(cone(ins, writers, i)):
        x = ins[j]
        print('    %4d [%s] %-12s %s' % (x['ln'], hex(x['addr']), x['op'],
                                         ' '.join(x['srcs'])[:70]))

# cone size distribution
print('\ncone sizes:', dict(sorted(collections.Counter(r[1] for r in rows).items())))
print('distinct depths:', dict(sorted(hist.items())))

# per-region summary of the 304
per = collections.Counter(region_of(i) for i in slm)
print('messages per barrier region:', dict(sorted(per.items())))

# where do the deepest ones live
deep = [i for d, c, i in rows if d == rows[0][0]]
print('deepest site count:', len(deep), 'regions:', sorted({region_of(i) for i in deep}))
