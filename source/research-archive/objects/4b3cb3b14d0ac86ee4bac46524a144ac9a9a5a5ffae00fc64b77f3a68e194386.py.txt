"""n31: source outline of the F01 body (comments + stage headers), read-only.

Prints the comment lines of the chain-head source so the post-0x4A90 ISA
segments can be mapped onto source stages.  CPU only.
"""
from pathlib import Path
import re
import sys

CAMP = Path('@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/fullnet-continuous-20261027')
SRC = CAMP / 'cards/n29-f01-alu-chain/candidate.comp'


def main():
    text = SRC.read_text(encoding='utf-8', errors='replace')
    lines = text.splitlines()
    for i, l in enumerate(lines, 1):
        s = l.strip()
        if s.startswith('//') and len(s) > 4:
            print('%4d %s' % (i, s[:170]))
    print('--- totals ---')
    for pat in ['coopMatLoad(', 'coopMatStore(', 'coopMatMulAdd(', 'subgroupShuffle',
                'barrier', 'nr_act_h(', 'nrh_narrow(', 'nrh_to_A(', 'inversesqrt',
                'float16_t(1.0/', 'packed[', 'store']:
        print('%-20s %d' % (pat, text.count(pat)))


if __name__ == '__main__':
    main()
