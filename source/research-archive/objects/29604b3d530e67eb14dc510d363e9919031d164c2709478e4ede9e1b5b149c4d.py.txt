from pathlib import Path
import json,struct,hashlib
D=Path(__file__).resolve().parent
read=lambda p:json.loads(p.read_text(encoding='utf-8-sig'))
p=read(D/'PROOF.json');n=len(p['arithmetic_cases']);data=(D/'run/rounding/comparison.bin').read_bytes();ref=(D/'ordered-reference.bin').read_bytes();assert len(data)==n*256*4
bits=struct.unpack('<%dI'%(len(data)//4),data);rb=struct.unpack('<%dI'%(len(ref)//4),ref);decode=lambda x:struct.unpack('<f',struct.pack('<I',x))[0]
scalar_bad=[];matrix_bad=[];casecounts={}
for case in range(n):
 for e in range(128):
  expected=rb[case*128+e];scalar=bits[case*256+128+e];matrix=bits[case*256+e]
  if scalar!=expected:scalar_bad.append(dict(case=case,row=e//16,col=e%16,expected=expected,actual=scalar))
  if matrix!=scalar:
   casecounts[case]=casecounts.get(case,0)+1
   matrix_bad.append(dict(case=case,row=e//16,col=e%16,scalar_bits=hex(scalar),matrix_bits=hex(matrix),scalar=decode(scalar),matrix=decode(matrix)))
r=read(D/'run/rounding/port_network_receipt.json');assert r['normal_close'] and r['gpu_passes']==1 and r['nan']==r['inf']==0
assert not scalar_bad,scalar_bad[:3]
out=dict(status='MMA_NOT_EQUIVALENT_TO_ORDERED_SCALAR' if matrix_bad else 'NO_COUNTEREXAMPLE_IN_THIS_FINITE_PROBE',device=r['device'],driver=r['driver_version_raw'],cases=n,scalar_vs_CPU_bit_mismatches=len(scalar_bad),matrix_vs_scalar_bit_mismatches=len(matrix_bad),affected_cases=casecounts,examples=matrix_bad[:16],limits='Synthetic finite half-valued cancellation cases. This establishes or challenges semantic equivalence, not frame quality, performance, or frequency of differences in the actual network. No conclusion about every possible register-fused scalar implementation.')
(D/'ROUNDING-REVIEW.json').write_text(json.dumps(out,indent=2),encoding='utf-8')
(D/'ROUNDING-MISMATCHES.json').write_text(json.dumps(matrix_bad,indent=2),encoding='utf-8')
print(json.dumps(out,indent=2))
