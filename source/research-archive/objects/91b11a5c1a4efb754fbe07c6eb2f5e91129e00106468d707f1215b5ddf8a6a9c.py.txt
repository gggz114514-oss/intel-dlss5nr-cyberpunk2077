        print("MISSING", k, p)

    plans[k] = pl
    outs = [o for o in pl.outputs] if hasattr(pl, "outputs") else []
    hdr = pl.header
    print("%-22s PASS=%-4s buf=%-4s disp=%-4s out=%s sha=%s" % (
        k, hdr[0], hdr[1], hdr[2],
        [(o.buffer_index if hasattr(o, "buffer_index") else o) for o in outs],
        pl.sha256()[:16]))
