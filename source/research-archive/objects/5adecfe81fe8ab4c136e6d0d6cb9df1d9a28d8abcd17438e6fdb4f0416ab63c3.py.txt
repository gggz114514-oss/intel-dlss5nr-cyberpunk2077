"""Shared ISA reader for n32-f05-aluwr.

Current identity = the F05 kernels of the n29-f01-alu-chain batch capture
(native-native-base and native-native-cand are byte-identical for all three
F05 kernels, so this is one driver batch, one identity).
"""
import re, hashlib, json
from pathlib import Path

C = Path(r"@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/fullnet-continuous-20261027")
CAP = C / "queue/n29-f01-alu-chain/native/captures/native-native-base/ISA"

F05 = {
    "c64": dict(width=64, kernel="6a22246177be2653", ords=[7, 10, 13, 16, 215, 218, 221, 224],
                groups="F05 C64 8 PASS"),
    "c128": dict(width=128, kernel="f98d5ab4e239dd54", ords=[19, 22, 25, 28, 31, 34, 195, 198, 201, 204, 207, 210],
                 groups="F05 C128 12 PASS"),
    "c256": dict(width=256, kernel="abd166013c088927", ords=[37, 40, 43, 46, 49, 52, 55, 58, 169, 172, 175, 178, 181, 184, 187, 190],
                 groups="F05 C256 16 PASS"),
}

LINE = re.compile(r"^/\* \[([0-9A-F]+)\]\s+\*/\s+(.*)$")


def path(width):
    k = F05["c%d" % width]["kernel"]
    fs = list(CAP.glob("CS_asm%s*" % k))
    assert len(fs) == 1, fs
    return fs[0]


def parse(width):
    """-> list of dict(addr, text, pred, op, args) in file order."""
    out = []
    for raw in path(width).read_text(encoding="utf-8", errors="replace").splitlines():
        m = LINE.match(raw)
        if not m:
            continue
        addr = int(m.group(1), 16)
        text = m.group(2).strip()
        toks = text.split()
        pred = []
        i = 0
        while i < len(toks) and toks[i].startswith("(") and toks[i].endswith(")"):
            pred.append(toks[i])
            i += 1
        op = toks[i] if i < len(toks) else ""
        out.append(dict(addr=addr, text=text, pred=" ".join(pred), op=op,
                        args=" ".join(toks[i + 1:])))
    return out


def sha(p):
    return hashlib.sha256(Path(p).read_bytes()).hexdigest()


if __name__ == "__main__":
    for w in (64, 128, 256):
        ins = parse(w)
        print("c%d: %d instructions, %s bytes, %s" % (
            w, len(ins), path(w).stat().st_size, path(w).name))
