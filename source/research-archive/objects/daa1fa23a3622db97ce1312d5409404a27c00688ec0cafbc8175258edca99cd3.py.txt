"""Build the S02 F01 candidate plans with planlib (structure-checked, no string replace)."""
import json
import sys
import hashlib
sys.path.insert(0, r"@NR_WORKSPACE@/work/dsh-fullnet-static-20261029/tools")
import planlib

S02 = r"@NR_WORKSPACE@/work/dsh-fullnet-static-20261029/S02-f01"
PACK = S02 + "/PACK"
COMBO = (r"@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/"
         r"int8-quality-first-20261029/real-int8-24/c64-family-static")
FRAMES = [34189, 41444, 43654]
BASE_SHA = {34189: "2b5481d5ce6b8ec8c1f62e68d2057b39d5267825d655e844a2ee2c5500f1c690",
            41444: "1bb66ba9b4896f6b", 43654: "c30613cc9a7f3e9d"}


def sha(p):
    return hashlib.sha256(open(p, "rb").read()).hexdigest()


def main():
    out = {"arms": {}, "base": {}}
    for fr in FRAMES:
        base = COMBO + f"/combo-{fr}.txt"
        out["base"][fr] = {"path": base, "sha256": sha(base)}
        patch = {
            "base_sha256": out["base"][fr]["sha256"],
            "replace_buffers": {
                "c32_f1_i8_scales": {"file": PACK + "/f1-ffn-scales.bin", "bytes": 656},
            },
            "notes": "S02 F01 X/E static scales; no SPV/weight change",
        }
        dst = S02 + f"/candidate-{fr}.txt"
        log = planlib.apply_patch(base, patch, dst)
        out["arms"].setdefault("xe", {})[fr] = log
        # D variant
        patch2 = {
            "base_sha256": out["base"][fr]["sha256"],
            "replace_buffers": {
                "c32_f1_i8_scales": {"file": PACK + "/f1-ffn-scales.bin", "bytes": 656},
                "c32_f1_body_weights": {"file": PACK + "/f1-body-weights-dopt.bin", "bytes": 53248},
                "c32_f1_body_scales": {"file": PACK + "/f1-body-scales-dopt.bin", "bytes": 640},
            },
            "replace_spv": [{"ordinal": 0, "name": "f1c32_image_pre",
                             "spv": S02 + "/SOURCE/candidate-dopt.spv",
                             "expected_sha256": sha(S02 + "/SOURCE/candidate-dopt.spv")}],
            "notes": "S02 F01 X/E static scales + 16-channel lift D (new SPV)",
        }
        dst2 = S02 + f"/candidate-{fr}-dopt.txt"
        log2 = planlib.apply_patch(base, patch2, dst2)
        out["arms"].setdefault("xed", {})[fr] = log2
        # structural check: parse both and compare resource arrays
        a = planlib.Plan(base); b = planlib.Plan(dst); c = planlib.Plan(dst2)
        assert len(a.passes) == len(b.passes) == len(c.passes) == 288
        assert len(a.buffers) == len(b.buffers) == len(c.buffers) == 906
        assert (a.header == b.header == c.header)
        n_diff_b = sum(1 for x, y in zip(a.buffers, b.buffers)
                       if (x.file, x.bytes) != (y.file, y.bytes))
        n_diff_c = sum(1 for x, y in zip(a.buffers, c.buffers)
                       if (x.file, x.bytes) != (y.file, y.bytes))
        print(fr, "xe buffer diffs", n_diff_b, "xed buffer diffs", n_diff_c,
              "xe sha", log["out_sha256"][:16], "xed sha", log2["out_sha256"][:16])
        out["arms"]["xe"][fr]["n_buffer_diffs"] = n_diff_b
        out["arms"]["xed"][fr]["n_buffer_diffs"] = n_diff_c
        out["arms"]["xe"][fr]["pass0_spv"] = a.passes[0].spv
        out["arms"]["xed"][fr]["pass0_spv"] = c.passes[0].spv
    open(S02 + "/PLANS.json", "w", encoding="utf-8").write(json.dumps(out, indent=2))
    print("wrote plans")


if __name__ == "__main__":
    main()
