        candidate=dict(

        plan=P['plan'],
        plan_diff=dict(passes_replaced=P['passes_replaced'],

                       differing_lines=P['differing_lines'],
                       only_f10_lines_changed=P['only_f10_lines_changed'],
                       shared_field=P['shared_field'],
                       grid_fields_unchanged=P['grid_fields_unchanged']),
        spirv_static=dict(B['spirv_static'],
                          barriers=[B['barriers']['parent'], B['barriers']['candidate']],
                          barrier_signature=B['barriers']['candidate_signature'],
                          dpas=[B['dpas']['parent'], B['dpas']['candidate']],
