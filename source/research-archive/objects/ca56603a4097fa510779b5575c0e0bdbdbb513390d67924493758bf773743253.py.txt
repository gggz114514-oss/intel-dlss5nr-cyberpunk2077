# ---------------------------------------------------------------- native census
ins = parse(ISA)
cls = collections.Counter()
for i in ins:
    t = i['text']
    if i['op'].startswith('send'):
        cls[i['op'] + ('.wide16' if '(16|M0)' in t else '.narrow1')] += 1
    elif i['op'] == 'sync.nop':
        cls['sync.nop'] += 1
    elif i['op'] == 'mov':
        import re
        m = re.search(r'\((\d+)\|M0\)', t)
        cls['mov(%s)' % (m.group(1) if m else '?')] += 1
    else:
        cls[i['op']] += 1
steps = 64
step_win = (0xC90, 0x120F)
step_ins = [i for i in ins if step_win[0] <= i['off'] <= step_win[1]]
step_cls = collections.Counter()

    t = i['text']
    if i['op'].startswith('send'):
        step_cls[i['op'] + ('.wide16' if '(16|M0)' in t else '.narrow1')] += 1
    elif i['op'] == 'sync.nop':
        step_cls['sync.nop'] += 1
    elif i['op'] == 'mov':
        import re
        m = re.search(r'\((\d+)\|M0\)', t)
        step_cls['mov(%s)' % (m.group(1) if m else '?')] += 1
    elif i['op'] in ('add', 'add3', 'shl', 'mul', 'mad', 'sel', 'cmp', 'and', 'or', 'jmpi', 'bfn.0xEA'):
        step_cls['int/addr/br'] += 1
    else:
        step_cls[i['op']] += 1
