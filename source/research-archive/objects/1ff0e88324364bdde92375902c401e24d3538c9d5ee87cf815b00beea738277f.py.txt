"""Append a restore step: point the chain head back at an earlier accepted plan.

Used when a same-batch re-measurement contradicts the step that was just adopted (the
ablation case): the restore is itself measured against the current head in one balanced
batch, so the chain stays a sequence of measured steps.
"""
import json, shutil, sys
from pathlib import Path
sys.path.insert(0, str(Path(__file__).parent))
import COMMON as K

src_step = int(sys.argv[1])          # chain index to restore to
improvement = float(sys.argv[2])     # measured (current head - restored plan) in ms
ci_lo, ci_hi = float(sys.argv[3]), float(sys.argv[4])
note = sys.argv[5] if len(sys.argv) > 5 else ''

cur = K.read_json(K.C / 'CURRENT.json')
target = cur['chain'][src_step]
plan = Path(target['path'])
acc = K.C / 'accepted' / ('restore-%s-20261027' % target.get('label'))
acc.mkdir(parents=True, exist_ok=True)
shutil.copyfile(plan, acc / 'plan.txt')
assert K.sha(acc / 'plan.txt') == K.sha(plan)
entry = dict(path=str(acc / 'plan.txt'), sha256=K.sha(acc / 'plan.txt'),
             label='restore-' + str(target.get('label')), saving_ms=improvement,
             CI95_ms=[ci_lo, ci_hi], kind='restore (withdraws the preceding ablation)',
             note=note)
cur['chain'].append(entry)
cur['increments_ms'].append(improvement)
cur['best_plan'] = dict(path=entry['path'], sha256=entry['sha256'])
cur['cumulative_ms_in_chain'] = sum(cur['increments_ms'])
cur['increment_sum_note'] = 'with ablation/restore steps the sum is an accounting artifact; report only the direct A/B vs handoff'
K.write_json(K.C / 'CURRENT.json', cur)
K.write_json(acc / 'ADOPT.json', entry)
print(json.dumps(dict(restored=entry['label'], plan=entry['plan'], improvement_ms=improvement,
                      chain_len=len(cur['chain'])), indent=2))
