"""Refresh the 232-row coverage table against the current accepted chain head.

For each PASS line of the chain head: ordinal, name, family, SPV path+sha, and which
accepted step introduced that SPV (0 = handoff baseline). Writes
evidence/COVERAGE-NOW.json and COVERAGE-NOW.md.
"""
import json, sys
from collections import Counter, defaultdict
from pathlib import Path
sys.path.insert(0, str(Path(__file__).parent))
import COMMON as K

cur = K.read_json(K.C / 'CURRENT.json')
chain = cur['chain']
head = Path(cur['best_plan']['path'])
rows = K.rows()
lines = K.pass_lines_of(head)
assert len(lines) == 232, len(lines)

# which step owns each SPV path (candidate path recorded in that step's accepted plan)
owner = {}
for i, e in enumerate(chain):
    p = Path(e['path'])
    for r in K.pass_lines_of(p):
        owner.setdefault(str(r['spv']).replace('/', '\\'), i)

table = []
for r in rows:
    ln = lines[r['ordinal']]
    assert ln['name'] == r['name'], (r['ordinal'], ln['name'], r['name'])
    spv = ln['spv']
    introduced = owner.get(str(spv).replace('/', '\\'))
    if introduced is None:
        introduced = 0 if str(spv).replace('/', '\\') == str(r['spv']).replace('/', '\\') else -1
    table.append(dict(ordinal=r['ordinal'], name=r['name'], family=r['family'],
                      spv=spv, spv_sha256=(Path(spv).exists() and K.sha(Path(spv))) or None,
                      handleoff_spv=r['spv'], introduced_by_step=introduced,
                      step_label=(chain[introduced].get('label') if introduced is not None and introduced >= 0 else 'unknown')))

bystep = Counter(t['introduced_by_step'] for t in table)
byfam = defaultdict(lambda: [0, 0])
for t in table:
    byfam[t['family']][0] += 1
    if t['introduced_by_step'] != 0:
        byfam[t['family']][1] += 1

out = dict(head=dict(path=str(head), sha256=K.sha(head)), rows=len(table),
           changed_since_handoff=sum(1 for t in table if t['introduced_by_step'] != 0),
           rows_per_step={str(k): v for k, v in sorted(bystep.items())},
           families={f: dict(rows=v[0], changed=v[1]) for f, v in sorted(byfam.items())},
           table=table)
K.write_json(K.C / 'evidence' / 'COVERAGE-NOW.json', out)

md = ['# 232 位置覆盖表（按当前接受链头 %s）' % cur['best_plan']['sha256'][:12], '',
      '| 引入轮次 | PASS 数 |', '|---|---|']
for k, v in sorted(bystep.items()):
    md.append('| %s | %d |' % (chain[k].get('label') if k >= 0 else 'unknown', v))
md += ['', '| 族 | 行数 | 自 handoff 起变更 |', '|---|---|---|']
for f, v in sorted(byfam.items()):
    md.append('| %s | %d | %d |' % (f, v[0], v[1]))
(K.C / 'COVERAGE-NOW.md').write_text('\n'.join(md) + '\n', encoding='utf-8')
print('rows', out['rows'], 'changed', out['changed_since_handoff'])
print('per step', out['rows_per_step'])
print('families changed:', {f: v['changed'] for f, v in out['families'].items() if v['changed']})
unknown = [t['ordinal'] for t in table if t['introduced_by_step'] == -1]
print('unknown owner rows:', unknown[:10], 'count', len(unknown))
