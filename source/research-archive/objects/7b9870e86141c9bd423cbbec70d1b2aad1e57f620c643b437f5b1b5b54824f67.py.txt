"""Emit the native evidence JSON for card n04a-f05-c64-feed.

Everything is derived from the recorded ISA capture of the current C64 group
identity (SPV sha 0d83192b...) and the current baseline plan.  Read-only.
"""
import json
import re
import sys
from pathlib import Path

sys.path.insert(0, str(Path(__file__).parent))
import isa_scan as S
import isa_regs as R

CARD = Path(__file__).parent
WEIGHT = S.__dict__.get('WEIGHT', 'load.ugm.d32x8.a32.bss[a0.2]')
WEIGHT = 'load.ugm.d32x8.a32.bss[a0.2]'
W = 'load.ugm.d32x8.a32.bss[a0.2]'

ins = S.parse(S.ASM)


def fields(s):
    return [f for f in re.split(r'\s{2,}', s) if f]


def reg(f):
    m = re.match(r'^r(\d+)', (f or '').strip())
    return int(m.group(1)) if m else None


def block(lo, hi):
    sends, dpas = [], []
    for i in ins:
        if not (lo <= i['off'] < hi):
            continue
        f = fields(i['full'])
        toks = re.findall(r'\$(\d+)', i['tail'])
        if W in i['tail']:
            sends.append(dict(off='0x%04x' % i['off'], execmask=i['execmask'],
                              bank=reg(f[1]), payload=reg(f[2]),
                              tokens=toks, desc='load.ugm.d32x8.a32.bss[a0.2]',
                              rd='rd:8', wr='wr:1+0'))
        elif i['mnemonic'].startswith('dpas'):
            dpas.append(dict(off='0x%04x' % i['off'], execmask=i['execmask'],
                             acc=reg(f[1]), b=reg(f[3]), a=reg(f[4]),
                             tokens=toks, mnemonic=i['mnemonic']))
    return sends, dpas


def chains(sends, dpas):
    """Per bank: the ordered (send, dpas) pairs that reuse it."""
    out = {}
    for s in sends:
        b = s['bank']
        reader = [d for d in dpas if d['b'] == b and int(d['off'], 16) > int(s['off'], 16)]
        out.setdefault(b, []).append(dict(send=s['off'], payload=s['payload'],
                                          first_reader=reader[0]['off'] if reader else None,
                                          reader_acc=reader[0]['acc'] if reader else None))
    return out


def roles(sends):
    r = {}
    for s in sends:
        r.setdefault(s['bank'], set()).add('B_return')
        r.setdefault(s['payload'], set()).add('addr_payload')
    return {('r%d' % k): sorted(v) for k, v in sorted(r.items())}


def census():
    c = {}
    for i in ins:
        c[i['mnemonic']] = c.get(i['mnemonic'], 0) + 1
    return dict(sorted(c.items(), key=lambda kv: -kv[1]))


def main():
    phases = {}
    for name, lo, hi in (('row_phase0', 0x0C00, 0x2200), ('row_phase1', 0x4A00, 0x6200)):
        sends, dpas = block(lo, hi)
        phases[name] = dict(region=[hex(lo), hex(hi)], sends=sends, dpas=dpas,
                            bank_chains={('r%d' % k): v for k, v in chains(sends, dpas).items()},
                            register_roles=roles(sends),
                            dual_role=[k for k, v in roles(sends).items()
                                       if len(v) > 1])
    allexec = sorted({i['execmask'] for i in ins if i['execmask']})
    slm = {}
    for i in ins:
        if 'slm' in i['tail']:
            k = i['tail'].split(';')[-1].strip()
            slm[k] = slm.get(k, 0) + 1
    out = dict(
        card='n04a-f05-c64-feed',
        isa=dict(path=str(S.ASM), sha256=R.__dict__.get('sha256') if False else None,
                 instructions=len(ins), census=census(),
                 exec_masks=allexec, slm=slm),
        qkv_projection=phases,
        notes=[
            'The capture is the native compilation of the C64 group SPV sha 0d83192b...; '
            'the family plan that produced it has a byte-identical full C64 pass block '
            '(f05-kv-consumer-layout/PARENT-NATIVE-BINDING.json, COMBINED-NATIVE-BINDING.json).',
            'exec (16|M0) on every weight send and every DPAS: 16 active lanes.',
            'rd:8 is the descriptor allocation unit; it is NOT claimed to equal the '
            '512 B/SG logical B fragment.  Byte-level per-lane width is derived from the '
            'address stride (lane*32 B), not from rd:8.',
        ])
    (CARD / 'ISA-QKV-FEED.json').write_text(json.dumps(out, indent=2), encoding='utf-8')
    print('wrote ISA-QKV-FEED.json')
    for k, v in out['qkv_projection'].items():
        print(k, 'sends', len(v['sends']), 'dpas', len(v['dpas']),
              'banks', len(v['bank_chains']), 'dual-role', v['dual_role'])


if __name__ == '__main__':
    main()
