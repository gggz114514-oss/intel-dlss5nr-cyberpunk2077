"""Task 1: channel/layout alignment census between GPU chain dumps and FP16 teacher.

Reads:  C00-color-baseline/runs/diag-chain62-66/chain-block{62..66}_out.bin
        real-int8-24/dump-fp16/dump_block{62..66}_out.bin
Writes: nothing (stdout json).
"""
import json
import numpy as np
from pathlib import Path

B = Path("@NR_WORKSPACE@")
W = B / "work/dsh-fullnet-static-20261029"
CHAIN = W / "C00-color-baseline/runs/diag-chain62-66"
TEACH = (B / "work/codex-astra-fp16-int8-audit-20261024/int8-quality-first-20261029"
         / "real-int8-24/dump-fp16")

C64 = 64


def load(p):
    a = np.fromfile(p, dtype="<f2").astype(np.float32)
    return a


def rms(a):
    return float(np.sqrt(np.mean(a.astype(np.float64) ** 2)))


def per_channel_rms(a, C):
    t = a.reshape(-1, C)
    return np.sqrt(np.mean(t.astype(np.float64) ** 2, axis=0))


def main():
    out = {"blocks": {}}
    for blk in (62, 63, 64, 65, 66):
        g = load(CHAIN / f"chain-block{blk}_out.bin")
        t = load(TEACH / f"dump_block{blk}_out.bin")
        if g.size != t.size:
            out["blocks"][blk] = {"ERROR": f"size {g.size} vs {t.size}"}
            continue
        C = C64 if (g.size % C64 == 0) else 32
        gr = per_channel_rms(g, C)
        tr = per_channel_rms(t, C)
        ratio = gr / np.maximum(tr, 1e-30)
        # global regression slope
        slope = float(np.dot(g.astype(np.float64), t.astype(np.float64))
                      / max(np.dot(t.astype(np.float64), t.astype(np.float64)), 1e-30))
        relL2 = float(np.linalg.norm(g - t) / max(np.linalg.norm(t), 1e-30))
        corr = float(np.corrcoef(g, t)[0, 1])
        out["blocks"][blk] = {
            "n": int(g.size), "C": C,
            "rms_gpu": rms(g), "rms_teach": rms(t),
            "ratio_global": rms(g) / rms(t),
            "absmax_gpu": float(np.max(np.abs(g))), "absmax_teach": float(np.max(np.abs(t))),
            "regress_slope": slope, "relL2": relL2, "corr": corr,
            "ch_ratio_min": float(ratio.min()), "ch_ratio_max": float(ratio.max()),
            "ch_ratio_mean": float(ratio.mean()), "ch_ratio_std": float(ratio.std()),
            "ch_ratio_head16": [round(float(x), 4) for x in ratio[:16]],
            "ch_ratio_all64": [round(float(x), 4) for x in ratio[:64]],
            "zero_frac_gpu": float(np.mean(g == 0)), "zero_frac_teach": float(np.mean(t == 0)),
        }
    print(json.dumps(out, indent=1))


if __name__ == "__main__":
    main()
