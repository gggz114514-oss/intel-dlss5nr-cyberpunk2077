"""n36-f05-arena-shrink: enumerate the 36 plan rows, grids, workgroup counts and the
mapping between plan PASS index and census ordinal.  CPU only."""
import hashlib
import json
import pathlib

W = pathlib.Path('@NR_WORKSPACE@')
A = W / 'work/codex-astra-fp16-int8-audit-20261024'
C = A / 'fullnet-continuous-20261027'
CARD = C / 'cards/n36-f05-arena-shrink'
ROWS = pathlib.Path('@NR_USER_HOME@/Documents/Codex/2026-10-24/nr-fp16-int8-audit-20261024/'
                    'handoff/dsh-fullnet-20261027/CURRENT-232-ROWS.json')

# GROUPS.md (chain-head column) census ordinals per F05 shader group.  GROUPS.md is the
# authority the card names; the handoff CURRENT-232-ROWS.json is not present in this
# environment any more, so the list is transcribed from GROUPS.md rather than derived
# from that file.  Every ordinal below is also the list n05a/n05e used for the same SPV.
ORDINALS = {
    'c64': [7, 10, 13, 16, 215, 218, 221, 224],
    'c128': [19, 22, 25, 28, 31, 34, 195, 198, 201, 204, 207, 210],
    'c256': [37, 40, 43, 46, 49, 52, 55, 58, 169, 172, 175, 178, 181, 184, 187, 190],
}

GROUP_OF = {
    '0d83192b5368cc6074e00bbefed67ec71f8f540a9088d4d96752887390459641': 'c64',
    '20f4074fc70c32c0e74f0ad000053492e5f5c420259da6044e7a00cfc3ee3f4f': 'c128',
    '7a9576d15a18b4698b2e930c82b9658f2345e40479351e6ad0a65b034abc0af1': 'c256',
}
# per-subgroup UGM message delta from build/lifetime.json (A tile = 8 messages)
MESSAGE_DELTA = {'c64': 56, 'c128': 112, 'c256': 0}


def sha(p):
    return hashlib.sha256(pathlib.Path(p).read_bytes()).hexdigest()


def main():
    plan = (CARD / 'plan.txt').read_text(encoding='utf-8-sig').splitlines()
    recs = []
    for i, line in enumerate(plan):
        if not line.startswith('PASS '):
            continue
        t = line.split()
        spv = t[3].strip('"')
        if 'n36-f05-arena-shrink' not in spv:
            continue
        g = spv.split('candidate-')[1].split('.spv')[0]
        gx, gy, gz = int(t[8]), int(t[9]), int(t[10])
        recs.append(dict(plan_line=i, label=t[1], name=t[2], group=g,
                         grid=[gx, gy, gz], workgroups=gx * gy * gz,
                         shared=int(t[5]), spv=spv))
    per = {}
    for r in recs:
        d = per.setdefault(r['group'], dict(passes=0, workgroups=0, grid_hist={},
                                            plan_lines=[], census_ordinals=ORDINALS[r['group']]))
        d['passes'] += 1
        d['workgroups'] += r['workgroups']
        key = '%dx%dx%d' % tuple(r['grid'])
        d['grid_hist'][key] = d['grid_hist'].get(key, 0) + 1
        d['plan_lines'].append(r['plan_line'])
    for g, d in per.items():
        assert d['passes'] == len(ORDINALS[g]), (g, d['passes'], len(ORDINALS[g]))
        d['messages_per_sg_delta'] = MESSAGE_DELTA[g]
        d['ugm_messages_delta_per_frame'] = (d['workgroups'] * 8 * MESSAGE_DELTA[g])
    out = dict(rows=recs, per_group=per, total_passes=len(recs),
               total_workgroups=sum(r['workgroups'] for r in recs),
               total_ugm_messages_delta_per_frame=sum(
                   d['ugm_messages_delta_per_frame'] for d in per.values()),
               plan=dict(path=str(CARD / 'plan.txt'), sha256=sha(CARD / 'plan.txt')),
               note=('UGM message deltas are per subgroup; x8 subgroups x workgroups = '
                     'per frame.  A tile = 8 load.ugm.d32x8t messages, B tile = 1 '
                     'load.ugm.d32x8 (n05a census cross-check).'))
    (CARD / 'build/rows.json').write_text(json.dumps(out, indent=2), encoding='utf-8')
    print(json.dumps(per, indent=1))
    print('passes', out['total_passes'], 'workgroups', out['total_workgroups'],
          'plan', out['plan']['sha256'][:16])
    missing = []
    print('rows without a census ordinal:', missing)


if __name__ == '__main__':
    main()
