# Generate LUNA-RESOURCES.json and LUNA-ORDINARY.json for the C02 candidate.
# Ordinary parent arm = the fixed baseline plan f20-score-owner (d655b6b1...),
# NOT any earlier baseline. Candidate arm = c02-f20-merged-norm plan.
from pathlib import Path
import json, hashlib

R = Path(__file__).resolve().parent
B = R.parents[1]
BASE = B / 'codex-astra-fp16-int8-audit-20261024'
P = BASE / 'c32-stream-balanced-20261025/f20-score-owner'
N = R / 'c02-f20-merged-norm'
RUNNER = BASE / 'phase-diagnosis-20261025/direct-clock-build/Release/clock_runner.exe'
ZPROF = R.parents[2] / 'work/zcode-combo1-profile-20261025/build/port_network_driver_profile_zprof.exe'

def ident(p):
    return dict(path=str(p), sha256=hashlib.sha256(p.read_bytes()).hexdigest())

myplan = ident(N / 'plan.txt')
parentplan = ident(P / 'plan.txt')
myspv = ident(N / 'candidate.spv')

# --- resources (single uninstrumented driver-profile run for scratch/shared receipts) ---
rj = json.loads((P / 'LUNA-RESOURCES.json').read_text())
job = json.loads(json.dumps(rj['jobs'][0]))
job['id'] = 'resource-c02-f20-merged-norm'
job['candidate'] = 'c02-f20-merged-norm'
job['plan'] = myplan
job['runner_output'] = str(N / 'run/resource-c02-f20-merged-norm')
job['argv'][2] = str(N / 'plan.txt')
job['argv'][4] = str(N / 'run/resource-c02-f20-merged-norm')
res = dict(identities=[ident(RUNNER), ident(ZPROF), myspv], jobs=[job])
(N / 'LUNA-RESOURCES.json').write_text(json.dumps(res, indent=2), encoding='utf-8')

# --- ordinary: three alternating parent/candidate process pairs ---
pairs = [('0', 'parent', 'candidate'), ('1', 'candidate', 'parent'), ('2', 'parent', 'candidate')]
jobs = []
for pair, first, second in pairs:
    for arm in (first, second):
        plan = parentplan if arm == 'parent' else myplan
        out = N / f'run/ordinary-c02-{pair}-{arm}'
        jobs.append(dict(
            id=f'ordinary-c02-{pair}-{arm}', pair=int(pair),
            candidate=arm, plan=plan, runner_output=str(out),
            argv=[str(RUNNER), '--plan', str(plan['path'] if isinstance(plan, dict) else plan),
                  '--out', str(out), '--pool', '--warm', '1', '--reps', '30', '--feedback'],
            expected_frame_out='2b63d4ba1967eebcda61698d538963f59bccaf6d690ef8168f1695a46c5d658d'))
ordd = dict(identities=[ident(RUNNER), myspv,
                        dict(path=str(P / 'candidate.spv'), sha256=ident(P / 'candidate.spv')['sha256'])],
            jobs=jobs)
(N / 'LUNA-ORDINARY.json').write_text(json.dumps(ordd, indent=2), encoding='utf-8')
print('resources jobs:', [j['id'] for j in res['jobs']])
print('ordinary jobs:', [j['id'] for j in ordd['jobs']])
