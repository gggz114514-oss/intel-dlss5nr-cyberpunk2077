def m8_pack(qx_tok):
    out = np.zeros(C.WIN_X * C.WIN_Y * 8192, np.int8)
    for w in range(C.WIN_X * C.WIN_Y):
        blk = qx_tok[w * 64:(w + 1) * 64]
        for m in range(64):
            out[w * 8192 + ((m >> 3) * 128 + np.arange(128)) * 8 + (m & 7)] = blk[m]
    return out
