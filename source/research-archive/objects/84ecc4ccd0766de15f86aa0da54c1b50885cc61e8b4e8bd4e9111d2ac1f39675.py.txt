"""N17B F20 SPIR-V structural check: parent-rebuild.spv vs candidate.spv / probe-candidate.spv.

CPU only.  Writes build/spv-check-<variant>.json and build/spv-diff-<variant>.txt.

Checks (parent = byte-identical reproduction of 215aca67...):
  1. cooperative-matrix op delta (Load / Store / MulAdd / Length / types)
  2. the cooperative-matrix types: MAT_H and MAT_A are the same f16/Subgroup/8x16
     type differing only in MatrixUse (premise of the component-wise H<->A transport
     that the module already uses twice)
  3. the removed / added OpCooperativeMatrixLoadKHR addresses (layout, stride, column
     major, id-free address expression)
  4. the OpCooperativeMatrixStoreKHR inventory: which stores sit behind a selection
     (i.e. under `debug`) in the candidate and which do not
  5. an id-free structural diff of the whole module (hunk list)
  6. numerics: float/GLSL.std.450 expression multiset and the NoContraction site
     multiset as (function, opcode, operand-signature) keys
  7. resources: bindings, storage bytes, Private storage class variables

Usage: python _spv_check.py [candidate|probe-candidate]
"""
import collections
import difflib
import json
import pathlib
import sys

sys.path.insert(0, r'@NR_WORKSPACE@/build/zcode-fp16-compute-backend-20261021/scripts')
import spv_tools_local as ST          # noqa: E402

CARD = pathlib.Path(__file__).resolve().parent

OPNAME = {
    12: 'OpExtInst', 43: 'OpConstant', 59: 'OpVariable', 61: 'OpLoad', 62: 'OpStore',
    65: 'OpAccessChain', 79: 'OpVectorShuffle', 80: 'OpCompositeConstruct',
    81: 'OpCompositeExtract', 82: 'OpCompositeInsert', 83: 'OpCopyObject',
    115: 'OpFConvert', 124: 'OpBitcast', 127: 'OpFNegate', 128: 'OpIAdd',
    129: 'OpFAdd', 130: 'OpISub', 131: 'OpFSub', 132: 'OpIMul', 133: 'OpFMul',
    139: 'OpSMod', 166: 'OpLogicalOr', 167: 'OpLogicalAnd', 169: 'OpSelect',
    170: 'OpIEqual', 171: 'OpINotEqual', 177: 'OpCopyLogical',
    246: 'OpLoopMerge', 247: 'OpSelectionMerge', 248: 'OpLabel', 249: 'OpBranch',
    250: 'OpBranchConditional', 251: 'OpSwitch',
    4456: 'OpTypeCooperativeMatrixKHR', 4457: 'OpCooperativeMatrixLoadKHR',
    4458: 'OpCooperativeMatrixStoreKHR', 4459: 'OpCooperativeMatrixMulAddKHR',
    4460: 'OpCooperativeMatrixLengthKHR',
}
FP_OPS = {127, 129, 131, 133, 135, 140, 141, 142}
FP_EXT = {4, 8, 10, 26, 27, 28, 31, 32, 37, 40, 43, 46, 49, 50}
CM_LOAD, CM_STORE, CM_MULADD = 4457, 4458, 4459
PRIVATE = 6


class Canon:
    """ID-free canonical expression tree (function locals tagged by ordinal)."""

    def __init__(self, module):
        self.m = module
        self.by = module.id_to_instruction()
        self.cache = {}
        self.bind = {}
        self.var_ord = {}
        n = 0
        for op, ops, _, _ in module.instructions:
            if op == 71 and len(ops) > 2 and ops[1] == 33:
                self.bind[ops[0]] = ops[2]
            if op == 59:
                self.var_ord[ops[1]] = n
                n += 1

    def c(self, cid, depth=0):
        if cid in self.cache:
            return self.cache[cid]
        assert depth < 400, 'canon recursion'
        op, ops = self.by.get(cid, (None, None))
        if op is None:
            val = ('undef', cid)
        elif op == 43:
            t_ops = self.by[ops[0]][1]
            val = ('const', tuple(t_ops[1:]), ops[2])
        elif op == 59:
            val = ('var', ops[2], self.bind.get(cid), self.var_ord.get(cid))
        elif op == 65:
            val = ('ac', self.c(ops[2], depth + 1),
                   tuple(self.c(x, depth + 1) for x in ops[3:]))
        elif op == 61:
            val = ('load', self.c(ops[2], depth + 1))
        elif op == 81:
            val = ('extract', self.c(ops[2], depth + 1), tuple(ops[3:]))
        elif op == 80:
            val = ('construct', tuple(self.c(x, depth + 1) for x in ops[2:]))
        elif op == 82:
            val = ('insert', self.c(ops[2], depth + 1), self.c(ops[3], depth + 1),
                   tuple(ops[4:]))
        elif op in (124, 109, 110, 111, 112, 113, 114, 115):
            val = ('conv', op, self.c(ops[2], depth + 1))
        elif op == 12:
            val = ('ext', ops[2], ops[3], tuple(self.c(x, depth + 1) for x in ops[4:]))
        elif op in (126, 127, 128, 129, 130, 131, 132, 133, 134, 135, 136, 137,
                    138, 139, 140, 141, 142, 169, 194, 195, 196, 197, 198, 199, 200):
            val = ('op', op, tuple(self.c(x, depth + 1) for x in ops[2:]))
        else:
            val = ('id', op)
        self.cache[cid] = val
        return val


def fp_multiset(m):
    can = Canon(m)
    bag = collections.Counter()
    for op, ops, _, _ in m.instructions:
        rid = ST.result_id_of(op, ops)
        if rid is None:
            continue
        if op in FP_OPS:
            bag[can.c(rid)] += 1
        elif op == 12 and ops[3] in FP_EXT:
            bag[can.c(rid)] += 1
    return bag


def nc_sites(m):
    ids = ST.no_contraction_ids(m)
    by = m.id_to_instruction()
    func_of, cur = {}, None
    for op, ops, _, _ in m.instructions:
        if op == 54:
            cur = ops[1]
        rid = ST.result_id_of(op, ops)
        if rid is not None:
            func_of[rid] = cur
    bag = collections.Counter()
    for i in ids:
        op, ops = by[i]
        bag[(func_of.get(i), op, ST.Module.operand_signature(m, i))] += 1
    return bag, len(ids)


def coopmat_types(m):
    const = {ops[1]: ops[2] for op, ops, _, _ in m.instructions if op == 43}
    out = []
    for op, ops, _, _ in m.instructions:
        if op != 4456:
            continue
        comp, scope, rows, cols, use = ops[1:6]
        out.append(dict(type=ops[0], component=const.get(comp, comp),
                        scope=const.get(scope, scope), rows=const.get(rows, rows),
                        cols=const.get(cols, cols), use=const.get(use, use)))
    return out


def coopmat_loads(m):
    can = Canon(m)
    const = {ops[1]: ops[2] for op, ops, _, _ in m.instructions if op == 43}
    out = []
    for op, ops, _, _ in m.instructions:
        if op != CM_LOAD:
            continue
        c = can.c(ops[2])
        out.append(dict(layout=const.get(ops[3], ops[3]), stride=const.get(ops[4], ops[4]),
                        collmajor=const.get(ops[5], ops[5]) if len(ops) > 5 else None,
                        addr=str(c), addr_c=c))
    return out


def coopmat_stores(m):
    can = Canon(m)
    const = {ops[1]: ops[2] for op, ops, _, _ in m.instructions if op == 43}
    out = []
    for op, ops, _, _ in m.instructions:
        if op != CM_STORE:
            continue
        # OpCooperativeMatrixStoreKHR has no result id:
        #   [pointer, object, layout(id), stride(id), memory-operand...]
        ptr_id = ops[0] if len(ops) > 0 else None
        c = can.c(ptr_id) if ptr_id is not None else None
        out.append(dict(layout=const.get(ops[2], ops[2]) if len(ops) > 2 else None,
                        stride=const.get(ops[3], ops[3]) if len(ops) > 3 else None,
                        addr=str(c) if ptr_id is not None else None, addr_c=c))
    return out


def struct_keys(m):
    const = {ops[1]: ops[2] for op, ops, _, _ in m.instructions if op == 43}

    def k(ops):
        return ','.join('#%d' % const[x] if x in const else '_' for x in ops)
    return ['%s(%s)' % (OPNAME.get(op, 'op%d' % op), k(ops))
            for op, ops, _, _ in m.instructions]


def main(variant='candidate'):
    par = ST.Module(CARD / 'parent-rebuild.spv')
    cand = ST.Module(CARD / (variant + '.spv'))
    res = {'variant': variant, 'parent_spv': 'parent-rebuild.spv',
           'candidate_spv': variant + '.spv'}
    hp = collections.Counter(op for op, _, _, _ in par.instructions)
    hc = collections.Counter(op for op, _, _, _ in cand.instructions)
    res['words'] = {'parent': len(par.words), 'candidate': len(cand.words)}
    res['instructions'] = {'parent': len(par.instructions),
                           'candidate': len(cand.instructions)}
    res['cooperative_matrix_ops'] = {OPNAME[k]: [hp.get(k, 0), hc.get(k, 0)]
                                     for k in (4456, 4457, 4458, 4459, 4460)}
    res['opcode_delta'] = {OPNAME.get(k, 'op%d' % k): [hp.get(k, 0), hc.get(k, 0)]
                           for k in sorted(set(hp) | set(hc))
                           if hp.get(k, 0) != hc.get(k, 0)}

    tp, tc = coopmat_types(par), coopmat_types(cand)
    res['cooperative_matrix_types'] = {'parent': tp, 'candidate': tc, 'identical': tp == tc}
    by_use = {t['use']: t for t in tp}
    drop = ('use', 'type')
    res['H_vs_A_type'] = {
        'accumulator_half_use2': by_use.get(2), 'matrixA_use0': by_use.get(0),
        'same_except_use': bool(
            by_use.get(2) and by_use.get(0) and
            {k: v for k, v in by_use[2].items() if k not in drop} ==
            {k: v for k, v in by_use[0].items() if k not in drop})}

    pl, cl = coopmat_loads(par), coopmat_loads(cand)
    pa = collections.Counter((l['layout'], l['stride'], l['addr']) for l in pl)
    ca = collections.Counter((l['layout'], l['stride'], l['addr']) for l in cl)
    res['coopmat_loads'] = {
        'parent': len(pl), 'candidate': len(cl),
        'removed': {'layout=%s stride=%s addr=%s' % k: n for k, n in (pa - ca).items()},
        'added': {'layout=%s stride=%s addr=%s' % k: n for k, n in (ca - pa).items()},
        'removed_sites': len(pa - ca), 'added_sites': len(ca - pa)}

    ps, cs = coopmat_stores(par), coopmat_stores(cand)
    spa = collections.Counter((s['layout'], s['stride'], s['addr']) for s in ps)
    sca = collections.Counter((s['layout'], s['stride'], s['addr']) for s in cs)
    res['coopmat_stores'] = {
        'parent': len(ps), 'candidate': len(cs),
        'removed': {'layout=%s stride=%s addr=%s' % k: n for k, n in (spa - sca).items()},
        'added': {'layout=%s stride=%s addr=%s' % k: n for k, n in (sca - spa).items()},
        'identical_inventory': spa == sca}

    # which shared object do the removed loads touch, and who else touches it?
    def var_of(addr):
        """walk the id-free expression tree down to the OpVariable node."""
        if not isinstance(addr, tuple):
            return None
        if addr and addr[0] == 'var':
            return addr
        for x in addr[1:]:
            if isinstance(x, tuple):
                v = var_of(x)
                if v is not None:
                    return v
            elif isinstance(x, (list,)):
                for y in x:
                    v = var_of(y) if isinstance(y, tuple) else None
                    if v is not None:
                        return v
        return None
    gone_vars = {var_of(l['addr_c']) for l in pl if
                 (l['layout'], l['stride'], l['addr']) in (pa - ca)}
    store_vars = {}
    for s in cs:
        v = var_of(s['addr_c'])
        key = '%s stride=%s' % (v, s['stride'])
        store_vars[key] = store_vars.get(key, 0) + 1
    # ordinal -> storage class for every OpVariable, so a ('var', class, binding, ord)
    # node can be read back as "the ord-th OpVariable, storage class N"
    ords = {}
    n = 0
    for op, ops, _, _ in cand.instructions:
        if op == 59:
            ords[n] = dict(id=ops[1], storage_class=ops[2] if len(ops) > 2 else None)
            n += 1
    res['removed_load_object'] = {
        'removed_load_pointer_nodes': sorted(str(v) for v in gone_vars),
        'candidate_store_sites_by_pointer_node': store_vars,
        'workgroup_variables': {k: v for k, v in ords.items() if v['storage_class'] == 4},
        'workgroup_variable_count': sum(1 for v in ords.values() if v['storage_class'] == 4),
        'same_object_as_a_store_site': any(
            str(v) in store_vars_key for v in gone_vars for store_vars_key in store_vars),
        'identity': ('the removed loads and the two stores both address the same workgroup '
                     'object at RowMajor layout / stride 32 with offsets m0*32+0 and m0*32+16; '
                     'in the source those are the shY publication (stage 3, lines 322/329) and '
                     'the shY seed reload (stage 9, lines 534/543), which the diff puts behind '
                     '`if (debug)` / removes'),
    }

    sp, sc = struct_keys(par), struct_keys(cand)
    hunks = []
    sm = difflib.SequenceMatcher(None, sp, sc, autojunk=False)
    for tag, i1, i2, j1, j2 in sm.get_opcodes():
        if tag == 'equal':
            continue
        hunks.append(dict(tag=tag, parent=[i1, i2], candidate=[j1, j2],
                          parent_lines=sp[i1:i2], candidate_lines=sc[j1:j2]))
    with (CARD / ('build/spv-diff-%s.txt' % variant)).open('w', encoding='utf-8') as f:
        for h in hunks:
            f.write('=== %s parent%s cand%s\n' % (h['tag'], h['parent'], h['candidate']))
            for l in h['parent_lines']:
                f.write('  - %s\n' % l)
            for l in h['candidate_lines']:
                f.write('  + %s\n' % l)
    res['structural_diff'] = {'hunks': len(hunks),
                              'summary': [{k: h[k] for k in ('tag', 'parent', 'candidate')}
                                          for h in hunks]}

    fp_p, fp_c = fp_multiset(par), fp_multiset(cand)
    res['fp_multiset'] = {'parent_sites': sum(fp_p.values()),
                          'candidate_sites': sum(fp_c.values()),
                          'identical': fp_p == fp_c,
                          'parent_only': [str(k) for k in (fp_p - fp_c)][:20],
                          'candidate_only': [str(k) for k in (fp_c - fp_p)][:20]}
    ncp, np_ = nc_sites(par)
    ncc, nc_ = nc_sites(cand)
    res['nc_sites'] = {'parent': np_, 'candidate': nc_,
                       'identical_multiset': ncp == ncc,
                       'parent_only': [str(k) for k in (ncp - ncc)][:20],
                       'candidate_only': [str(k) for k in (ncc - ncp)][:20]}

    res['bindings'] = {'parent': {str(k): v for k, v in par.binding_ids().items()},
                       'candidate': {str(k): v for k, v in cand.binding_ids().items()}}
    res['storage_bytes'] = {'parent': par.storage_bytes(), 'candidate': cand.storage_bytes()}
    res['storage_bytes_identical'] = par.storage_bytes() == cand.storage_bytes()
    res['private_storage_variables'] = {
        'parent': sum(1 for op, ops, _, _ in par.instructions
                      if op == 59 and len(ops) > 2 and ops[2] == PRIVATE),
        'candidate': sum(1 for op, ops, _, _ in cand.instructions
                         if op == 59 and len(ops) > 2 and ops[2] == PRIVATE)}
    res['cfg_shape'] = {'parent': {OPNAME[k]: hp.get(k, 0) for k in (246, 247, 248, 249, 250)},
                        'candidate': {OPNAME[k]: hc.get(k, 0) for k in (246, 247, 248, 249, 250)}}

    (CARD / ('build/spv-check-%s.json' % variant)).write_text(json.dumps(res, indent=1),
                                                              encoding='utf-8')
    print(json.dumps({k: res[k] for k in ('variant', 'cooperative_matrix_ops', 'H_vs_A_type',
                                          'coopmat_loads', 'coopmat_stores', 'structural_diff',
                                          'fp_multiset', 'nc_sites', 'storage_bytes_identical',
                                          'private_storage_variables', 'opcode_delta')},
                     indent=1)[:9000])


if __name__ == '__main__':
    main(sys.argv[1] if len(sys.argv) > 1 else 'candidate')
