"""Task 4 probe: per-channel residual-gain regression.

model  out = ffn + r*c_in          (r = residual tap actually applied)
Compare r_teacher (FP16) with r_gpu (INT8 chain) per channel.
Also: the same for the *teacher* alone gives the FP16 model's own r.
"""
import json
import numpy as np
from pathlib import Path

B = Path("@NR_WORKSPACE@")
W = B / "work/dsh-fullnet-static-20261029"
CHAIN = W / "C00-color-baseline/runs/diag-chain62-66"
TEACH = (B / "work/codex-astra-fp16-int8-audit-20261024/int8-quality-first-20261029"
         / "real-int8-24/dump-fp16")
C = 64


def load(p):
    return np.fromfile(p, dtype="<f2").astype(np.float64)


def regress(out, inp, C=C):
    """per-channel least-squares slope of out on inp (no intercept)."""
    O = out.reshape(-1, C)
    I = inp.reshape(-1, C)
    num = (O * I).sum(0)
    den = (I * I).sum(0)
    return num / np.maximum(den, 1e-30)


def regress_ic(out, inp, C=C):
    O = out.reshape(-1, C)
    I = inp.reshape(-1, C)
    r = np.zeros(C)
    for c in range(C):
        A = np.stack([I[:, c], np.ones(len(I))], 1)
        sol, *_ = np.linalg.lstsq(A, O[:, c], rcond=None)
        r[c] = sol[0]
    return r


def main():
    t = {62: load(TEACH / "dump_block62_out.bin"),
         63: load(TEACH / "dump_block63_out.bin"),
         64: load(TEACH / "dump_block64_out.bin"),
         65: load(TEACH / "dump_block65_out.bin")}
    t_in = {62: load(TEACH / "dump_ioport_block62_blend.bin"), 63: t[62], 64: t[63], 65: t[64]}
    g = {b: load(CHAIN / f"chain-block{b}_out.bin") for b in (62, 63, 64, 65)}
    g_in = {62: None, 63: g[62], 64: g[63], 65: g[64]}

    out = {}
    for b in (63, 64, 65):
        rt = regress(t[b], t_in[b])
        rg = regress(g[b], g_in[b])
        rti = regress_ic(t[b], t_in[b])
        rgi = regress_ic(g[b], g_in[b])
        out[f"b{b}"] = {
            "r_teacher_noicpt_64": [round(float(x), 4) for x in rt],
            "r_gpu_noicpt_64": [round(float(x), 4) for x in rg],
            "r_teacher_icpt_64": [round(float(x), 4) for x in rti],
            "r_gpu_icpt_64": [round(float(x), 4) for x in rgi],
            "r_teacher_mean": float(rt.mean()), "r_gpu_mean": float(rg.mean()),
            "r_teacher_16per_dev": float(np.abs(np.tile(rt[:16], 4) - rt).max()),
            "r_gpu_16per_dev": float(np.abs(np.tile(rg[:16], 4) - rg).max()),
            "r_teacher_icpt_mean": float(rti.mean()), "r_gpu_icpt_mean": float(rgi.mean()),
        }
    # teacher b62 has the true input; measure FP16 model's residual gain there too
    rt62 = regress(t[62], t_in[62])
    out["b62_teacher"] = {"r_teacher_noicpt_64": [round(float(x), 4) for x in rt62],
                          "r_teacher_mean": float(rt62.mean())}
    print(json.dumps(out, indent=1))


if __name__ == "__main__":
    main()
