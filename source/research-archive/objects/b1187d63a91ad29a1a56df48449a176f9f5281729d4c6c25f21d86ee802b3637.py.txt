# -*- coding: utf-8 -*-
"""Generate ROI-SOURCE.md from COLOR-BASELINE.json (no numbers typed by hand)."""
import json
from pathlib import Path

OUT = Path(__file__).resolve().parent.parent
d = json.loads((OUT / "COLOR-BASELINE.json").read_text(encoding="utf-8"))
ROIS = d["observation_protocol"]["roi"]
DESC = {
    "sky-bright": "bright sky gap between the overpass beams (upper middle); the brightest "
                  "non-sign surface in 34189 and the main saturated-highlight population",
    "building-bright": "lit white building face immediately right of the sky gap, left of the "
                       "FUYUTSUKI sign; the brightest **unsaturated** surface (no channel clamped)",
    "ncpd-sign": "the glowing blue NCPD letters and their halo on the sign face",
    "blue-ground": "blue-lit pavement / sidewalk in the bottom-left foreground",
    "blue-wall": "blue-lit graffiti wall under the overpass (strongest blue-dominant surface; "
                 "added as a fifth probe beyond the four requested families)",
    "neutral-wall": "grey concrete face of the right-hand structure (nearest to R=G=B in 34189)",
}
lines = []
A = lines.append

A("# ROI-SOURCE — fixed colour ROIs for C00\n")
A(f"Generated: {d['generated']}  |  card: C00-color-baseline  |  workspace: `{d['workspace']}`\n")
A("## 1. What is fixed here\n")
A("Six axis-aligned rectangles, determined **once** on frame **34189** at **1280x720**, then applied")
A("unchanged to every frame and to **both** INT8 arms (XE and the older c64). No per-frame, per-arm")
A("or per-image ROI adaptation is performed anywhere in `tools/metrics.py`.\n")
A("Coordinate convention: half-open pixel ranges `[x0, x1) x [y0, y1)`, origin at the top-left")
A("corner, x to the right, y downwards, in the same 1280x720 frame as the float buffer's plane 0.\n")
A("## 2. Source image and why these coordinates transfer to the float data\n")
A("- `Q/visual-comparison/DISPLAY-CONTRACT.json` states: frame 34189, source = application RGB")
A("  `[0,:,:,:3]` float32, mapping = identical `clip(0,1)`, `round(value*255)`, no independent")
A("  exposure / tone mapping / gamma.")
A("- Independent check (`tools/verify_display.py`): `Q/visual-comparison/fp16.png` equals")
A("  `round(clip(network-output.bin plane0, 0, 1) * 255)` on **2,764,794 of 2,764,800** components;")
A("  the 6 differing components differ by exactly 1 LSB (tie rounding only). So coordinates read off")
A("  the published PNG address the same pixels as the float reference buffer.")
A("- The candidate arm is *not* used for ROI placement, so the mask cannot be biased toward either arm.\n")
A("## 3. ROI table (reference statistics on frame 34189)\n")
A("| ROI | box [x0,y0,x1,y1] | size px | ref mean RGB | ref luma | R-G | B-G | ref clip-any | description |")
A("|---|---|---|---|---|---|---|---|---|")
f0 = next(f for f in d["frames"] if f["frame"] == 34189)
for name, meta in f0["arms"]["xe"]["roi_meta"].items():
    g = f0["arms"]["xe"]["roi"][name]
    box = meta["box"]
    A(f"| `{name}` | [{box[0]},{box[1]},{box[2]},{box[3]}] | {meta['npx']} | "
      f"({g['R']['ref_mean']:.3f}, {g['G']['ref_mean']:.3f}, {g['B']['ref_mean']:.3f}) | "
      f"{meta['ref_Y_mean']:.3f} | {g['RG']['ref_mean']:+.3f} | {g['BG']['ref_mean']:+.3f} | "
      f"{meta['ref_clip_frac_any']*100:.2f}% | {DESC[name]} |")
A("\n`ref clip-any` = fraction of ROI pixels where at least one FP16 channel is at the 1.0 clamp.\n")
sky = f0["arms"]["xe"]["roi_meta"]["sky-bright"]
A(f"Note: a rectangle cannot separate bare sky from the silhouetted towers inside the gap - "
  f"{sky['ref_bucket_frac']['dark']*100:.0f}% of `sky-bright` is below luma 0.25 and is included "
  f"identically in every arm; {sky['ref_bucket_frac']['bright']*100:.0f}% is above luma 0.70.\n")
A("## 4. Same boxes on the other two frames — content drift is measured, not hidden\n")
A("The two motion sequences move the camera, so a screen-space box does not keep the same physical")
A("surface. The table reports what the box actually contains on each frame; arm comparisons stay")
A("valid because **both arms use the identical mask**, but cross-frame ROI comparisons do not.\n")
A("| ROI | frame | ref luma | R-G | B-G | dark / mid / bright frac | clip-any |")
A("|---|---|---|---|---|---|---|")
for fr in d["frames"]:
    for name, meta in fr["arms"]["xe"]["roi_meta"].items():
        g = fr["arms"]["xe"]["roi"][name]
        bf = meta["ref_bucket_frac"]
        A(f"| `{name}` | {fr['frame']} | {meta['ref_Y_mean']:.3f} | {g['RG']['ref_mean']:+.3f} | "
          f"{g['BG']['ref_mean']:+.3f} | {bf['dark']*100:.0f} / {bf['mid']*100:.0f} / "
          f"{bf['bright']*100:.0f} % | {meta['ref_clip_frac_any']*100:.2f}% |")
A("\nReading the drift: `blue-ground` keeps its blue-dominant character on all three frames")
A("(B-G +0.100 / +0.157 / +0.176), `sky-bright` and `building-bright` stay bright but lose a large")
A("part of their luma on 41444 (camera moves the sky gap out of the box), and `neutral-wall` is truly")
A("neutral only on 34189 (+0.003 R-G); on 41444 the box lands on a reddish surface (+0.034 R-G).")
A("`ncpd-sign` keeps blue dominance but drops to luma 0.081 on 41444 and 0.087 on 43654.\n")
A("## 5. What is missing (why this is not a per-surface mask)\n")
A("- No segmentation, object-id, depth or albedo buffer was captured, so the ROIs are rectangles, not")
A("  per-pixel surface masks; a rectangle can mix a surface with its background.")
A("- No neutral reference patch exists in all three frames (concrete in this scene carries blue")
A("  ambient and orange paint), so `neutral-wall` is only neutral *on 34189* and cannot serve as a")
A("  white-balance anchor.")
A("- The output colour space / transfer function of `frame_out` is unverified, so ROI numbers are raw")
A("  RGB differences; no Lab, DeltaE or luminance in cd/m2 is claimed.")
A("- The ROI set was placed by visual inspection of the published 1280x720 PNG plus 32x32 block")
A("  statistics of the FP16 float buffer; there is no independent ground-truth annotation to check it")
A("  against, and no second annotator.")
A("- Motion frames would need per-frame feature tracking to keep the same surface; that is")
A("  deliberately **not** done here because the card requires one fixed mask.\n")
A("## 6. Reproduce\n")
A("```")
A("PY = @NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/"
  "int8-quality-first-20261029/tool-env/Scripts/python.exe")
A(f"& $PY -X utf8 {OUT}/tools/scan_roi.py        # 32x32 block scan used to place the boxes")
A(f"& $PY -X utf8 {OUT}/tools/verify_display.py   # PNG == round(clip(plane0)*255) check + crops")
A(f"& $PY -X utf8 {OUT}/tools/roi_refine.py       # zoom crops of each ROI family")
A(f"& $PY -X utf8 {OUT}/tools/roi_final_probe.py  # per-frame content stats of the final boxes")
A("```\n")
(OUT / "ROI-SOURCE.md").write_text("\n".join(lines) + "\n", encoding="utf-8")
print("wrote ROI-SOURCE.md", len(lines), "lines")
