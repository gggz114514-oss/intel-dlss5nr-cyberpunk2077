"""Collect card JOBS.json into one serial GPU queue (does not run anything)."""
from __future__ import annotations

import json
import sys
from pathlib import Path

sys.path.insert(0, str(Path(__file__).resolve().parent.parent / "tools"))
import planlib  # noqa: E402

W = Path(r"@NR_WORKSPACE@/work/dsh-fullnet-static-20261029")
PRIORITY = {
    "S01-c32": 10,
    "S01-c32-tail": 11,
    "S05-c128": 20,
    "S06-c256": 30,
    "S07-split16": 40,
    "S08-vit": 50,
    "S02-f01": 60,
    "S12-proj": 70,
    "S03-tail": 80,
    "S13-quality": 90,
}


def main() -> None:
    jobs = []
    for card in sorted(W.glob("S*/JOBS.json")):
        card_name = card.parent.name
        try:
            items = json.loads(card.read_text(encoding="utf-8"))
        except Exception as e:  # noqa: BLE001
            print("skip unreadable", card, e)
            continue
        if isinstance(items, dict):
            items = items.get("jobs", [])
        for j in items:
            plan = Path(j["plan"])
            out = Path(j["out"])
            problems = []
            if not plan.is_file():
                problems.append("plan_missing")
            else:
                try:
                    p = planlib.Plan(plan)
                    problems.extend(p.validate())
                except Exception as e:  # noqa: BLE001
                    problems.append(f"plan_invalid: {e}")
            if out.exists():
                problems.append("out_dir_not_fresh")
            jobs.append({
                "card": card_name,
                "priority": PRIORITY.get(card_name, 999),
                "sequence": j.get("sequence"),
                "frame": j.get("frame"),
                "plan": plan.as_posix(),
                "out": out.as_posix(),
                "plan_sha256": planlib.sha256_file(plan) if plan.is_file() else None,
                "expected_output_sha256": j.get("expected_output_sha256"),
                "dump": bool(j.get("dump", False)),
                "note": j.get("note"),
                "preflight_problems": problems,
                "ready": not problems,
            })
    jobs.sort(key=lambda j: (j["priority"], str(j["frame"])))
    out = {
        "schema": "dsh-gpu-queue/1",
        "status": "NOT_RUN",
        "runner": "@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/phase-diagnosis-20261025/direct-clock-build/Release/clock_runner.exe",
        "lock": "@NR_GPU_WORKSPACE@/artifacts/parallel-port-20261013/GPU.lock",
        "authorization": "NOT_ALLOCATED; see tickets/VK-FULLNET-STATIC-GPU-SLOT-20261029.md",
        "jobs": jobs,
        "ready": sum(j["ready"] for j in jobs),
        "blocked": sum(not j["ready"] for j in jobs),
    }
    (W / "gpu/JOBS-QUEUE.json").write_text(json.dumps(out, indent=2), encoding="utf-8")
    print(json.dumps({k: out[k] for k in ("ready", "blocked")}, indent=2))
    for j in jobs:
        flag = "OK " if j["ready"] else "BAD"
        print(f"{flag} p{j['priority']:3d} {j['card']:14s} {j['sequence']}-{j['frame']} {j['plan_sha256'][:12] if j['plan_sha256'] else '--------'} {j['preflight_problems']}")


if __name__ == "__main__":
    main()
