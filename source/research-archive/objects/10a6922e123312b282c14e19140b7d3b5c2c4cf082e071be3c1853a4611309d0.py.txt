"""n31: resolve the predicate behind every conditional branch after 0x4A90.

For each conditional branch in the post-entry region, walk back to the last write
of the flag register (cmp/mov/and/or with a flag destination) and print it, so
the branch can be classified against the fixture constants
(pc.p0.z = debug = 0, nrf_ctl(8) = 0, flags&4 = 4).  CPU only.
"""
import re
import sys
from pathlib import Path

CAMP = Path('@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/fullnet-continuous-20261027')
sys.path.insert(0, str(CAMP))
import ISA  # noqa: E402

ISA_PATH = CAMP / 'queue/n29-f01-alu-chain/native/captures/native-native-cand/ISA/CS_asma0888c5a3c6703a9_simd32_0000000000000001_0_orig.asm'
ENTRY_END = 0x4A90

WRITE = re.compile(r'\(([a-z]+)\)(f\d)\.(\d)')
READ = re.compile(r'\((?:W&|~)*(f\d)\.(\d)\)')
MOVFLAG = re.compile(r'f\d\.\d<\d+>:')


def flag_writes(ins):
    out = []
    for m in WRITE.finditer(ins['raw']):
        out.append((int(m.group(2)[1]), int(m.group(3))))
    if ins['op'].startswith('mov') and ins['ops'] and MOVFLAG.match(ins['ops'][0]):
        m = re.match(r'f(\d)\.(\d)<', ins['ops'][0])
        out.append((int(m.group(1)), int(m.group(2))))
    return out


def pred_of(ins):
    m = READ.search(ins['raw'].split('//')[0])
    if m:
        return (int(m.group(1)[1]), int(m.group(2)))
    return None


def target_addr(ops, isa_by_label):
    for o in ops:
        m = re.match(r'L(\d+)', o)
        if m:
            return int(m.group(1))
    return None


def main():
    isa = ISA.parse(ISA_PATH)
    by_addr = {i['addr']: i for i in isa}
    by_flag = {}
    for k, ins in enumerate(isa):
        for fw in flag_writes(ins):
            by_flag.setdefault(fw, []).append(k)
    print('%-6s %-6s %-8s %-7s %-6s %s' % ('addr', 'span', 'pred', 'target', 'def', 'definer'))
    for k, ins in enumerate(isa):
        if ins['addr'] < ENTRY_END or ins['op'] not in ('jmpi', 'goto'):
            continue
        pr = pred_of(ins)
        tgt = target_addr(ins['ops'], None)
        span = (tgt - ins['addr']) if tgt else None
        if pr is None:
            print('%04X   %-6s %-8s %-7s %-6s %s' % (ins['addr'], span, '-', tgt, '-',
                                                    ins['raw'].strip()[:90]))
            continue
        lst = [x for x in by_flag.get(pr, []) if x < k]
        d = lst[-1] if lst else None
        txt = isa[d]['raw'].strip() if d is not None else '(none)'
        print('%04X   %-6s f%d.%-4d %-7s %-6s %s' % (
            ins['addr'], span, pr[0], pr[1], tgt,
            ('%04X' % isa[d]['addr']) if d is not None else '-', txt[:105]))


if __name__ == '__main__':
    main()
