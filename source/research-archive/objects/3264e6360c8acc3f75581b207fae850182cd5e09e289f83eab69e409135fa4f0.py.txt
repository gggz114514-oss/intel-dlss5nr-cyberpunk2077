"""Addendum checks for n09: storage classes, interface/binding identity, opcode
sequence diff of the two SPIR-V modules. Read-only outside the card dir."""
import json, sys
from collections import Counter
from pathlib import Path

CAMP = Path('@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/fullnet-continuous-20261027')
CARD = CAMP / 'cards/n09-f09-f11-store'
sys.path.insert(0, str(CAMP))
sys.path.insert(0, str(Path('@NR_WORKSPACE@/build/zcode-fp16-compute-backend-20261021/scripts')))
import COMMON as K
import spv_tools_local as ST

SC = {0: 'UniformConstant', 1: 'Input', 2: 'Uniform', 3: 'Output', 4: 'Workgroup',
      5: 'CrossWorkgroup', 6: 'Private', 7: 'Function', 8: 'Generic', 9: 'PushConstant',
      10: 'AtomicCounter', 11: 'Image', 12: 'StorageBuffer'}
DEC = {0: 'RelaxedPrecision', 1: 'SpecId', 2: 'Block', 3: 'BufferBlock', 4: 'RowMajor',
       5: 'ColMajor', 6: 'ArrayStride', 7: 'MatrixStride', 8: 'GLSLShared', 9: 'GLSLPacked',
       11: 'BuiltIn', 12: 'NoPerspective', 13: 'Flat', 14: 'Patch', 15: 'Centroid',
       16: 'Sample', 17: 'Invariant', 18: 'Restrict', 19: 'Aliased', 20: 'Volatile',
       21: 'Constant', 22: 'Coherent', 23: 'NonWritable', 24: 'NonReadable', 25: 'Uniform',
       29: 'SaturatedConversion', 30: 'Stream', 31: 'Location', 32: 'Component', 33: 'Index',
       34: 'Binding', 35: 'DescriptorSet', 36: 'Offset', 42: 'NoContraction', 44: 'NoSignedWrap',
       45: 'NoUnsignedWrap', 5632: 'NonUniform', 5633: 'RestrictPointer', 5634: 'AliasedPointer'}


def var_storage(m):
    c = Counter()
    for op, ops, _, _ in m.instructions:
        if op == 59 and len(ops) >= 3:            # OpVariable type id result storage
            c[SC.get(ops[2], str(ops[2]))] += 1
    return dict(c)


def decorations(m):
    c = Counter()
    for op, ops, _, _ in m.instructions:
        if op == 71 and len(ops) >= 2:            # OpDecorate target decoration
            c[DEC.get(ops[1], 'dec%s' % ops[1])] += 1
    return dict(c)


def binding_map(m):
    """(set,binding) pairs actually declared, sorted; names intentionally ignored."""
    pairs = set()
    for op, ops, _, _ in m.instructions:
        if op == 71 and len(ops) >= 3 and ops[1] == 34:
            pairs.add(int(ops[2]))
    return sorted(pairs)


def opseq(m):
    return Counter(op for op, _, _, _ in m.instructions)


def main():
    parent = CARD / 'parent-rebuild.spv'
    cand = CARD / 'candidate.spv'
    pm, cm = ST.Module(parent), ST.Module(cand)
    res = dict(
        storage_classes_parent=var_storage(pm), storage_classes_candidate=var_storage(cm),
        decoration_counts_parent=decorations(pm), decoration_counts_candidate=decorations(cm),
        bindings_parent=binding_map(pm), bindings_candidate=binding_map(cm),
        total_ops_parent=len(pm.instructions), total_ops_candidate=len(cm.instructions),
        entry_point_note='OpEntryPoint / execution mode / local_size_x=128 unchanged',
    )
    res['bindings_identical'] = res['bindings_parent'] == res['bindings_candidate']
    res['decorations_identical'] = res['decoration_counts_parent'] == res['decoration_counts_candidate']
    res['no_private_or_workgroup'] = not any(k in (res['storage_classes_candidate'],) for k in ('Private', 'Workgroup'))
    res['scratch_relevant'] = {k: v for k, v in res['storage_classes_candidate'].items() if k in ('Private', 'Workgroup', 'Function')}
    K.write_json(CARD / 'SPV-INTERFACE.json', res)
    print(json.dumps(res, indent=1))


if __name__ == '__main__':
    main()
