"""Extract per-sample frame times from the receipt of each of the 18 runs and compare the arms."""
import json, sys
from collections import defaultdict
from pathlib import Path
sys.path.insert(0, str(Path(__file__).parent))
import COMMON as K

root = K.C / 'queue' / 'vk102-compare' / 'run'
by_arm = defaultdict(list)
hashes = defaultdict(set)
for run in sorted(root.glob('perm-*')):
    arm = run.name.split('-')[-1]
    pair = int(run.name.split('-')[1])
    rc = K.read_json(run / 'port_network_receipt.json')
    fh = K.read_json(run / 'FRAME-HASHES.json') if (run / 'FRAME-HASHES.json').exists() else {}
    for k, v in (fh or {}).items():
        if isinstance(v, str):
            hashes[arm].add(v)
    # find the per-sample timings
    smp = rc.get('samples') or rc.get('frames') or []
    times = []
    for s in smp:
        g = s.get('gpu_ms') if isinstance(s, dict) else None
        if g:
            times.append(g)
    if not times:
        # fall back to a top-level list of times
        for key in ('gpu_ms', 'frame_ms', 'times_ms'):
            v = rc.get(key)
            if isinstance(v, list) and v:
                times = [x for x in v if isinstance(x, (int, float))]
                break
    if times:
        by_arm[arm].append((pair, sum(times) / len(times)))
print('arms:', {k: len(v) for k, v in by_arm.items()})
print('hashes per arm:', {k: len(v) for k, v in hashes.items()})

T = 2.570581835636305
def closeness(x, y):
    n = min(len(x), len(y))
    diffs = [x[i][1] - y[i][1] for i in range(n)]
    m = sum(diffs) / n
    sd = (sum((v - m) ** 2 for v in diffs) / (n - 1)) ** 0.5
    half = T * sd / (n ** 0.5)
    same = all(d > 0 for d in diffs) or all(d < 0 for d in diffs)
    return dict(saving_ms=m, CI95_ms=[m - half, m + half], groups=n,
                status='FASTER_95' if (m - half > 0 and same) else ('SLOWER_95' if (m + half < 0 and same) else 'INDISTINCT'),
                means=dict(x=sum(v for _, v in x[:n]) / n, y=sum(v for _, v in y[:n]) / n))

out = {}
order = [a for a in ('A', 'C', 'V') if a in by_arm]
for i in range(len(order)):
    for j in range(i + 1, len(order)):
        a, b = order[i], order[j]
        c = closeness(by_arm[a], by_arm[b])
        out['%s_minus_%s' % (a, b)] = c
        print('%s - %s: %s saving=%+.6f CI95=[%+.6f, %+.6f] means %.3f vs %.3f'
              % (a, b, c['status'], c['saving_ms'], c['CI95_ms'][0], c['CI95_ms'][1],
                 c['means']['x'], c['means']['y']))
K.write_json(K.C / 'evidence' / 'VK102-COMPARISON.json',
             dict(comparisons=out, frame_hashes={k: sorted(v) for k, v in hashes.items()},
                  method='same runner/fixture/pool/warm1/reps30; six balanced permutations; all '
                         'samples retained; t(df=5) over permutation means; vk102 compared on timing '
                         'only because it is a different model version'))
