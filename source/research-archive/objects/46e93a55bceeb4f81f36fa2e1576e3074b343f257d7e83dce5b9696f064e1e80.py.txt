"""S07 diagnostic: why does the FP16 contract reference differ from dump Y by ~0.6%?"""
import sys
from pathlib import Path

import numpy as np

B = Path("@NR_WORKSPACE@")
W = B / "work/dsh-fullnet-static-20261029"
sys.path.insert(0, str(W / "tools"))
sys.path.insert(0, str(W / "S07-split16/tools"))
from planlib import Plan  # noqa: E402
import model as M  # noqa: E402

D = B / "work/codex-astra-fp16-int8-audit-20261024/int8-quality-first-20261029/real-int8-24"


def main():
    p = Plan(D / "c64-family-static/combo-34189.txt")
    d0 = p.dispatch_by_name("split16_b23_l0")
    d1 = p.dispatch_by_name("split16_b23_l1")
    push0 = [int(x) for x in d0.push]
    p2, p3 = push0[8:12], push0[12:16]
    pk = M.Pack(Path(p.buffers[d0.binds[10].buffer].file).read_bytes(),
                Path(p.buffers[d1.binds[10].buffer].file).read_bytes(),
                Path(p.buffers[d0.binds[0].buffer].file), p2, p3)
    td = D / "dump-fp16"
    X = M.load_half(td / "dump_block22_out.bin")[:1056]
    F = M.load_half(td / "dump_split16_b23_F.bin")[:1056]
    Y = M.load_half(td / "dump_split16_b23_Y.bin")[:1056]

    A = (F @ pk.Wc.T)                       # fp32 dot
    seed = M.f16(X * pk.rs)                 # half product widened
    cands = {
        "h1 f16(A+seed)": M.f16(A + seed),
        "h2 f16(f16(A)+seed)": M.f16(M.f16(A) + seed),
        "h3 f16(A + X*rs fp32)": M.f16(A + X * pk.rs),
        "h4 f16(A) only": M.f16(A),
        "h5 f16(A+X)": M.f16(A + X),
        "h6 no-transpose F@Wc": M.f16(F @ pk.Wc + seed),
    }
    for k, v in cands.items():
        print(f"{k:28s} relL2={M.rel_l2(v, Y, slice(None)):.6f}  eq={np.mean(v==Y):.4f}")
    # structure of h1 error
    e = (cands["h1 f16(A+seed)"] - Y)
    per_ch = np.linalg.norm(e, axis=0) / np.maximum(np.linalg.norm(Y, axis=0), 1e-9)
    per_tok = np.linalg.norm(e, axis=1) / np.maximum(np.linalg.norm(Y, axis=1), 1e-9)
    print("per-channel relL2: max %.4f mean %.4f  n(>0.05)=%d" % (per_ch.max(), per_ch.mean(), (per_ch > 0.05).sum()))
    print("per-token relL2:  max %.4f mean %.4f" % (per_tok.max(), per_tok.mean()))
    order = np.argsort(-per_ch)[:12]
    print("worst channels:", order.tolist())
    print("their relL2:", np.round(per_ch[order], 4).tolist())
    print("rs[worst]:", np.round(pk.rs[order], 4).tolist())
    # absolute error scale vs value scale
    print("mean |e| =", float(np.abs(e).mean()), " mean |Y| =", float(np.abs(Y).mean()))
    # is the error a pure gain per channel?
    gains = (Y * cands["h1 f16(A+seed)"]).sum(0) / np.maximum((cands["h1 f16(A+seed)"] ** 2).sum(0), 1e-12)
    print("per-channel best gain: min %.5f max %.5f std %.5f" % (gains.min(), gains.max(), gains.std()))
    g = np.argsort(-np.abs(gains - 1))[:10]
    print("worst gain channels:", g.tolist(), np.round(gains[g], 4).tolist())


if __name__ == "__main__":
    main()
