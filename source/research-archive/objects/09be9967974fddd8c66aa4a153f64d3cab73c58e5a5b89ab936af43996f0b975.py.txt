def tokens(p):
    return [a if a else b for a, b in TOKEN_RE.findall(Path(p).read_text(encoding='utf-8-sig',
                                                                        errors='replace'))]


def parse_plan(p):
    t = tokens(p)
    i = 0
    assert t[0] == 'PORT_NETWORK_1', t[:4]
    rows, nb, npass, nout = map(int, t[1:5])
    i = 5
    for _ in range(nb):
        assert t[i] == 'BUFFER', t[i:i + 4]
        i += 4
    passes = []
    prev = -1
    for _ in range(npass):
        assert t[i] == 'PASS', t[i:i + 4]
        row = int(t[i + 1]); name = t[i + 2]; spv = t[i + 3]
        subgroup = int(t[i + 4]); shared = int(t[i + 5])
        nbind = int(t[i + 6]); npush = int(t[i + 7])
        gx, gy, gz = int(t[i + 8]), int(t[i + 9]), int(t[i + 10])
        assert row >= prev
        prev = row
        i += 11
        binds = []
        for _ in range(nbind):
            assert t[i] == 'BIND', t[i:i + 5]
            binds.append(int(t[i + 1]))
            i += 5
        assert t[i] == 'PUSH', t[i:i + 2]
        i += 1 + npush
        passes.append(dict(ordinal=len(passes), row=row, name=name, spv=spv,
                           subgroup=subgroup, shared=shared, nbind=nbind, npush=npush,
                           grid=[gx, gy, gz], binds=binds))
    return passes


def pipeline_vector(passes):
    """first-appearance order of clock_runner.cpp:354 dedup keys."""
    order, key_to_idx = [], {}
    for p in passes:
        key = (p['spv'].replace('\\', '/').lower(), p['npush'], p['subgroup'], tuple(p['binds']))
        if key not in key_to_idx:
            key_to_idx[key] = len(order)
            order.append(key)
    vec = [key_to_idx[(p['spv'].replace('\\', '/').lower(), p['npush'], p['subgroup'],
                      tuple(p['binds']))] for p in passes]
    return vec, order
