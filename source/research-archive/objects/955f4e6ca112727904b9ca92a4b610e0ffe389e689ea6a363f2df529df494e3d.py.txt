"""n04b-f05-c128-feed build: parent reproduction, candidate compile, SPV validate,
NoContraction correspondence, plan replacement, CPU plan check.

Writes only inside this card directory.  No GPU work: glslang + spv_validate_cpu +
clock_runner --cpu-check only.
"""
import sys, json, shutil, subprocess
from pathlib import Path

C = Path(r'@NR_WORKSPACE@\work\codex-astra-fp16-int8-audit-20261024\fullnet-continuous-20261027')
CARD = C / 'cards' / 'n04b-f05-c128-feed'
sys.path.insert(0, str(C))
import COMMON as K

GROUP_SHA = '20f4074fc70c32c0e74f0ad000053492e5f5c420259da6044e7a00cfc3ee3f4f'
ORDS = [19, 22, 25, 28, 31, 34, 195, 198, 201, 204, 207, 210]


def nc_profile(path):
    m = K.ST.Module(Path(path))
    ids = K.ST.no_contraction_ids(m)
    seq = []
    for op, ops, idx, cnt in m.instructions:
        if not ops:
            continue
        rid = ops[0] if (19 <= op <= 39 or op in (7, 11, 248) or op >= 4000 and op in (4456, 4457, 4458)) else (ops[1] if len(ops) > 1 else None)
        if rid in ids:
            seq.append((op, len(ops)))
    return dict(path=str(path), sha256=K.sha(path), nc_count=len(ids),
                nc_op_sequence=seq, fma_sites=len(m.ext_inst_fma_ordinals()),
                storage=m.storage_bytes())


def main():
    card = CARD
    build = card / 'build'
    build.mkdir(exist_ok=True)
    row = K.row_by_ordinal(19)
    src_parent = Path(row['source'])
    spv_parent = Path(row['spv'])
    assert K.sha(spv_parent) == GROUP_SHA, K.sha(spv_parent)
    incs = K.include_dirs_for(19, src_parent)

    # 1. parent byte reproduction ------------------------------------------
    par = build / 'parent-raw.spv'
    K.compile_glsl(src_parent, par, incs)
    parent_match = K.sha(par) == GROUP_SHA

    # 2. candidate ---------------------------------------------------------
    cand_src = card / 'candidate.comp'
    cand_spv = card / 'candidate.spv'
    K.compile_glsl(cand_src, cand_spv, incs)

    # 3. SPV validation ----------------------------------------------------
    v_par = v_can = None
    try:
        v_par = K.validate_spv(par)
    except Exception as exc:                     # noqa: BLE001
        v_par = 'FAILED: %s' % exc
    try:
        v_can = K.validate_spv(cand_spv)
    except Exception as exc:                     # noqa: BLE001
        v_can = 'FAILED: %s' % exc

    # 4. NoContraction correspondence -------------------------------------
    ncp = nc_profile(par)
    ncc = nc_profile(cand_spv)
    nc_ok = (ncp['nc_op_sequence'] == ncc['nc_op_sequence'])

    # 5. plan --------------------------------------------------------------
    plan = card / 'plan.txt'
    nrepl = K.make_plan(cand_spv, plan, spv_shas={GROUP_SHA})
    cpu = None
    cpu_err = None
    try:
        cpu = K.cpu_check(plan)
    except Exception as exc:                     # noqa: BLE001
        cpu_err = str(exc)

    shutil.copyfile(par, card / 'parent-rebuild.spv')

    out = {
        'card': 'n04b-f05-c128-feed',
        'shader_group': {'spv_sha256_baseline': GROUP_SHA, 'pass_ordinals': ORDS, 'pass_count': len(ORDS)},
        'parent': {
            'plan': K.ident(K.BASELINE),
            'spv': K.ident(spv_parent),
            'source': K.ident(src_parent),
            'rebuild_spv': K.ident(card / 'parent-rebuild.spv'),
            'rebuild_recipe': 'raw glslang -V --target-env vulkan1.3, no decoration transfer',
            'rebuild_byte_identical': parent_match,
            'include_dirs': [str(d) for d in incs],
        },
        'candidate': {
            'source': K.ident(cand_src),
            'spv': K.ident(cand_spv),
            'validate_cpu': 'OK' if v_can is None else v_can,
        },
        'nc': {
            'float_NC_checked': ncc['nc_count'],
            'parent_nc_count': ncp['nc_count'],
            'candidate_nc_count': ncc['nc_count'],
            'nc_op_sequence_identical': nc_ok,
            'fma_sites_parent': ncp['fma_sites'],
            'fma_sites_candidate': ncc['fma_sites'],
            'rule': 'every NoContraction-decorated site of the candidate must match the parent one-for-one in instruction order; no site added or removed',
        },
        'storage_bytes': {'parent': ncp['storage'], 'candidate': ncc['storage']},
        'plan': {
            'path': str(plan), 'sha256': K.sha(plan),
            'spv_paths_replaced': nrepl,
            'parent_plan': K.ident(K.BASELINE),
        },
        'cpu_check': {'argv': [str(K.RUNNER), '--plan', str(plan), '--cpu-check'],
                      'stdout': cpu, 'error': cpu_err},
    }
    K.write_json(card / 'BUILD.json', out)
    print(json.dumps({k: out[k] for k in ('shader_group', 'nc', 'storage_bytes', 'cpu_check')}, indent=1))
    print('parent match', parent_match, 'cand sha', out['candidate']['spv']['sha256'])
    print('validate parent:', 'OK' if v_par is None else v_par[:200])
    print('validate cand  :', out['candidate']['validate_cpu'][:200])
    print('plan repl', nrepl, 'plan sha', out['plan']['sha256'])


if __name__ == '__main__':
    main()
