import json, re
from pathlib import Path
ISA = Path(r"@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/fullnet-mechanism-transfer-20261026/f05-k-consumer-tile/c256/captures/native-f20-pixel-owner/ISA")
rows = []
for p in sorted(ISA.glob("*.asm")):
    t = p.read_text(encoding="utf-8", errors="replace")
    rows.append((t.count("dpas"), str(p.name), t.count("send"), t.count("mov"),
                 len(t.splitlines())))
rows.sort()
for dpas, name, send, mov, lines in rows:
    print("%5d dpas %5d send %5d mov %7d lines  %s" % (dpas, send, mov, lines, name))
