import re, collections
from pathlib import Path
F = Path(r"@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/fullnet-mechanism-transfer-20261026/f05-k-consumer-tile/c256/captures/native-f20-pixel-owner/ISA/CS_asmfd78c91bb8744f0f_simd32_000000000000000e_13_orig.asm")
lines = F.read_text(encoding="utf-8", errors="replace").splitlines()
FENCE = next(i for i, l in enumerate(lines) if "fence.slm" in l)

# destination register of an instruction: first "rN.<sub>" token before the ':' and
# immediately after the execution-size field
DST = re.compile(r"\)\s+(?:null\s+)?(r\d+)\.(\d+)")
mx_dst = 0
for l in lines:
    m = DST.search(l)
    if m:
        mx_dst = max(mx_dst, int(m.group(1)))
allr = [int(x) for l in lines for x in re.findall(r"\br(\d+)[.\s]", l)]
print("max dest register:", mx_dst, " max any register:", max(allr))
print("registers >=100 touched:", sorted({x for x in allr if x >= 100}))

DPAS = re.compile(r"dpas\.[\d x]+\((\d+)\|M0\)\s+(r\d+)\.\d+:(\w+)\s+(?:null|\S+)\s+(r\d+[.\d]*):(\w+)\s+(r\d+[.\d]*):(\w+)")
qkv, att = [], []
for i, l in enumerate(lines):
    m = DPAS.search(l)
    if m:
        (qkv if i < FENCE else att).append((i, m.group(2), m.group(4), m.group(6)))

print()
print("QKV dpas dst tuples:", collections.Counter(d for _, d, _, _ in qkv))
print("QKV distinct dst:", len({d for _, d, _, _ in qkv}))
print("ATT dpas dst tuples:", collections.Counter(d for _, d, _, _ in att))
print("ATT distinct A:", len({a for _, _, a, _ in att}), " B:", len({b for _, _, _, b in att}))
print("QKV distinct A:", len({a for _, _, a, _ in qkv}), " B:", len({b for _, _, _, b in qkv}))
print()
print("first 10 QKV dpas:", qkv[:10])
print("last 6 QKV dpas:", qkv[-6:])
