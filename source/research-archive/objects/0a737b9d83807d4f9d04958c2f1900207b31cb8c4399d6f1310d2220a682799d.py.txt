"""Reproduce LIFT-ATTRIBUTION.json on the captured WG to prove the CPU model identity."""
import json
import numpy as np
import f01_model as M

DIAG = (r"@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/"
        r"int8-v15-offline-20261028/first-difference/f01-feature-capture/run/c32_f1_f2_diag_f1.bin")
HALF = (r"@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/"
        r"int8-v15-offline-20261028/first-difference/f01-feature-capture/run/f01-half-debug.bin")

d32 = np.fromfile(DIAG, dtype="<u4")
feat32 = d32[57440:57440 + 1024].view("<f4").reshape(64, 16).copy()
assert np.all(feat32 == feat32.astype(np.float16).astype(np.float32)), "features must be exact halves"
x = feat32.astype(np.float16)

Wl, We, Wc, rs = M.WLIFT_AS_F32, M.WEXP_AS_F32, M.WCON_AS_F32, M.RS_AS_F32
lift16, act16, y16 = M.ffn_teacher(x, Wl, We, Wc, rs)

# per-subgroup static amax scale (the old dynamic contract)
amax = np.array([np.abs(x[q * 16:(q + 1) * 16].astype(np.float32)).max() for q in range(4)],
                np.float32)
sf_sg = (amax / np.float32(127.0)).astype(np.float32)
print("feature_absmax_per_channel", [float(v) for v in np.abs(x.astype(np.float32)).max(0)])
print("feature_scale_per_sg     ", [float(v) for v in sf_sg])

qw1, sw1 = M.qmat(Wl)          # int8 weights, per-output-channel
invF = (np.float32(127.0) / amax).astype(np.float32)


def qa_for(scale_vec):
    z = (x.astype(np.float32) * scale_vec[:, None].repeat(16, axis=0)).astype(np.float32)
    return M.rne_clamp127(z).astype(np.int32)


def rel(a, b):
    a = np.asarray(a, np.float64); b = np.asarray(b, np.float64)
    return float(np.linalg.norm(a - b) / np.linalg.norm(b))


def met(a, b):
    a = np.asarray(a); b = np.asarray(b)
    d = np.abs(a.astype(np.float64) - b.astype(np.float64))
    return {"relL2": rel(a, b), "maxerr": float(d.max()), "neq": int((a != b).sum())}


rows = {}
# activation-only: int8 activation code, exact float32 weights
qa = qa_for(invF)
qa_real = qa.astype(np.float32) * sf_sg[:, None].repeat(16, axis=0)
lift_act = (qa_real @ Wl.T).astype(np.float16)
rows["activation_only"] = {"lift_vs_FP16": met(lift_act, lift16)}
# weight-only: exact half features, int8 weights
acc_w = (x.astype(np.float32) @ (qw1.astype(np.float32) * sw1[:, None]).T)
lift_w = acc_w.astype(np.float16)
rows["weight_only"] = {"lift_vs_FP16": met(lift_w, lift16)}
# both: int8 code x int8 weight, dequant with sw*sF per token
acc = qa @ qw1.T
mult = (sw1[None, :] * sf_sg[:, None].repeat(16, axis=0)).astype(np.float32)
lift_b = (acc.astype(np.float32) * mult).astype(np.float16)
rows["both"] = {"lift_vs_FP16": met(lift_b, lift16)}

for k, lf in [("activation_only", lift_act), ("weight_only", lift_w), ("both", lift_b)]:
    # teacher uses its own lift; input lift is replaced, so recompute directly
    l2 = lf.astype(np.float32)
    e2 = (l2 @ We.T).astype(np.float16)
    a2 = np.array([M.act_h(v) for v in e2.ravel()], np.float16).reshape(e2.shape)
    yc = a2.astype(np.float32) @ Wc.T
    seed = (l2 * rs[None, :].astype(np.float32)).astype(np.float16)
    y2 = (yc + seed.astype(np.float32)).astype(np.float16)
    rows[k]["propagated_FP16_FFN_Y"] = met(y2, y16)

print(json.dumps(rows, indent=2))

# --- direct GPU-dump identity for the INT8 arm (captured WG) -------------------
h = np.fromfile(HALF, dtype="<f2")
gpu_x = h[22528:22528 + 2048].reshape(64, 32)
gpu_e = h[0:8192].reshape(64, 128)
gpu_y = h[8192:10240].reshape(64, 32)
acc = qa @ qw1.T
mult = (sw1[None, :] * sf_sg[:, None].repeat(16, axis=0)).astype(np.float32)
lift_b2 = (acc.astype(np.float32) * mult).astype(np.float16)
e8, y8, _, _ = M.ffn_int8(lift_b2, M._SC0, )
print("gpu_vs_model", json.dumps({
    "lift": met(lift_b2, gpu_x),
    "expand": met(e8, gpu_e),
    "contract_Y": met(y8, gpu_y),
}, indent=2))
