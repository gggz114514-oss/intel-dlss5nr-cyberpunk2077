def sha(path):
    return hashlib.sha256(Path(path).read_bytes()).hexdigest()


def f32(x):
    return np.float32(x)


# ---------------------------------------------------------------- addressing --
def io_at(px, py, c, W, H, C, layout):
    px = np.asarray(px, np.int64); py = np.asarray(py, np.int64); c = np.asarray(c, np.int64)
    if layout == 2:
        return (py * W + px) * C + c
    if layout == 3:
        sub = ((c & 1) | (((c >> 3) & 1) << 1) | (((c >> 1) & 1) << 2) | (((c >> 2) & 1) << 3))
        return (c // 16) * (W * H * 16) + (py * W + px) * 16 + sub
    t = (py * W + px) if layout == 1 else (((py // 4) * (W // 4) + px // 4) * 16 + 4 * (py % 4) + px % 4)
    return ((t // 16) * (C // 16) + c // 16) * 256 + (t % 16) * 16 + c % 16


def ds_pixel(u, mode):
    """Verbatim port of the parent `uvec2 ds_pixel(uint u)` (vectorised)."""
    u = np.asarray(u, np.int64)
    if mode == 0:
        tile, slot = u // 16, u % 16
        tiles_x = OW // 4
        return 4 * (tile % tiles_x) + slot % 4, 4 * (tile // tiles_x) + slot // 4
    return u % OW, u // OW


# ------------------------------------------------------------------ pooling --
def pooled_values(flat16, mode=SRC_LAYOUT):
    """Exact four-point half pool: precise half adds, one half *0.25, as the parent."""
    u = np.arange(NPIX, dtype=np.int64)
    px, py = ds_pixel(u, mode)
    c = np.arange(K, dtype=np.int64)
    idx = np.empty((NPIX, K, 4), np.int64)
    for j, (dx, dy) in enumerate(((0, 0), (1, 0), (0, 1), (1, 1))):
        idx[:, :, j] = io_at((2 * px + dx)[:, None], (2 * py + dy)[:, None],
                             c[None, :], IW, IH, K, SRC_LAYOUT)
    g = flat16[idx]
    top = g[:, :, 0] + g[:, :, 1]
    bottom = g[:, :, 2] + g[:, :, 3]
    s = top + bottom
    return (s * np.float16(0.25)).astype(np.float16)


def parent_fp16_out(v_half, w_kn):
    """f16 x f16 with fp32 accumulation (parent MMA), fp32 value + one half narrow."""
    acc = v_half.astype(np.float32) @ w_kn.astype(np.float32)
    return acc, acc.astype(np.float16)


def dst_base_by_pixel(mode):
    u = np.arange(NPIX, dtype=np.int64)
    m0 = u - (u % 8)
    fx, fy = ds_pixel(m0, mode)
    bases = [io_at(fx, fy, n0, OW, OH, N, mode) for n0 in range(0, N, 16)]
    return m0, fx, fy, bases


def store_output(out_half, mode):
    """Addresses of the deployed fragment stores (one coopMatStore per N16 fragment)."""
    m0, fx, fy, bases = dst_base_by_pixel(mode)
    m = np.arange(NPIX, dtype=np.int64) % 8
    buf = np.full(OW * OH * N, np.float16(0), np.float16)
    hits = np.zeros(OW * OH * N, np.int32)
    for gi, n0 in enumerate(range(0, N, 16)):
        addr = bases[gi][:, None] + m[:, None] * 16 + np.arange(16, dtype=np.int64)[None, :]
        buf[addr.ravel()] = out_half[:, n0:n0 + 16].ravel()
        hits += np.bincount(addr.ravel(), minlength=hits.size).astype(np.int32)
    return buf, hits


def build_w_canonical(w_tiled):
    """fp16 DS_WB tile -> canonical [n][k]; the tiled form is [N16][K16][16N][16K]."""
    w = np.empty((N, K), np.float16)
    for n in range(N):
        for k in range(K):
            w[n, k] = w_tiled[((n // 16) * (K // 16) + k // 16) * 256 + (n % 16) * 16 + (k % 16)]
    return w


def rne_clip(x, lo=-127.0, hi=127.0):
    return np.clip(np.rint(x), lo, hi)


def analyse():
    EVID.mkdir(parents=True, exist_ok=True)
    PACKS.mkdir(parents=True, exist_ok=True)
    out = {"card": "r03-ioport4", "kind": "CPU_STRUCT", "gpu": "not run",
           "geometry": {"IW": IW, "IH": IH, "OW": OW, "OH": OH, "K": K, "N": N,
                        "valid_W": VALID_W, "valid_H": VALID_H,
                        "src_layout": SRC_LAYOUT, "dst_layout": DST_LAYOUT,
                        "grid": [GX, GY, 1], "local_size_x": LOCAL_SIZE,
                        "wg_per_frame": GX * GY, "pixels_per_wg": 8, "channels_per_wg": 32}}

    # ---- 1. layout proofs (integer only) ---------------------------------
    proof = {"fragment_store_identity": {}, "dispatch_store_coverage": {},
             "dst_layout_branches": {}}
    u = np.arange(NPIX, dtype=np.int64)
    m = u % 8
    c = np.arange(N, dtype=np.int64)
    for mode, label in ((0, "layout0_spatial4x4"), (1, "layout1_raster")):
        px_u, py_u = ds_pixel(u, mode)
        m0 = u - m
        fx, fy = ds_pixel(m0, mode)
        ok = 0
        bad = []
        for n0 in range(0, N, 16):
            dst = io_at(fx, fy, n0, OW, OH, N, mode)[:, None]
            for cc in range(n0, n0 + 16):
                got = io_at(px_u, py_u, cc, OW, OH, N, mode)
                exp = dst[:, 0] + m * 16 + (cc - n0)
                eq = got == exp
                ok += int(eq.sum())
                if not eq.all() and len(bad) < 8:
                    bad.append([int(u[~eq][0]), int(cc)])
        proof["fragment_store_identity"][label] = {
            "pairs_checked": int(NPIX * N), "pairs_equal": int(ok),
            "mismatch_examples": bad,
            "rule": "io_at(ds_pixel(m0+m), n0+n) == io_at(ds_pixel(m0), n0) + m*16 + n",
            "verdict": bool(ok == NPIX * N),
        }
        hits = np.zeros(OW * OH * N, np.int32)
        for gi, n0 in enumerate(range(0, N, 16)):
            base = io_at(fx, fy, n0, OW, OH, N, mode)
            addr = base[:, None] + m[:, None] * 16 + np.arange(16, dtype=np.int64)[None, :]
            hits += np.bincount(addr.ravel(), minlength=hits.size).astype(np.int32)
        proof["dispatch_store_coverage"][label] = {
            "allocation_elements": OW * OH * N,
            "written_once": int((hits == 1).sum()),
            "unwritten": int((hits == 0).sum()),
            "multi_written": int((hits > 1).sum()),
            "verdict": bool((hits == 1).all()),
        }
        proof["dst_layout_branches"][label] = {
            "ds_pixel_mode": mode,
            "formula": ("4*(tile%tiles_x)+slot%4, 4*(tile/tiles_x)+slot/4; tile=u/16, slot=u%16, tiles_x=OW/4"
                        if mode == 0 else "u%OW, u/OW"),
            "deployed": bool(DST_LAYOUT == mode),
            "owner_comment": ("layout0 stays inside one half of a 4x4 tile; layout1 stays in a raster "
                              "token16 tile including zero-filled padding"),
        }

    # input buffer addressing bijection
    t = np.arange(IW * IH, dtype=np.int64)
    ci = np.arange(K, dtype=np.int64)
    in_idx = io_at((t % IW)[:, None], (t // IW)[:, None], ci[None, :], IW, IH, K, SRC_LAYOUT)
    proof["input_bijection"] = {
        "pairs_checked": int(IW * IH * K), "distinct": int(np.unique(in_idx).size),
        "min": int(in_idx.min()), "max": int(in_idx.max()),
        "allocation_half": IW * IH * K, "verdict": bool(np.unique(in_idx).size == IW * IH * K),
        "formula": "io_at layout0: ((t/16)*(C/16)+c/16)*256+(t%16)*16+c%16, t=((py/4)*(W/4)+px/4)*16+4*(py%4)+px%4",
    }
    proof["valid_region"] = {
        "guard": "py<OH && px<valid_W && py<valid_H",
        "OH": OH, "valid_W": VALID_W, "valid_H": VALID_H, "OW": OW,
        "guard_rejects_pixels": 0,
        "reason": "valid_W==OW and valid_H==OH in the deployed push, so all 64512 pooled pixels are computed",
        "padding_policy": ("the M8 fragment store writes all 8 rows; with GX*8 == OW*OH every row is a real "
                           "pixel, so no zero-filled padding is emitted for layout0 at this grid"),
        "input_tail_reads": "2*px+1 <= 671 < IW and 2*py+1 <= 383 < IH: no out-of-range input read",
    }

    # ---- 2. captures + pooling -------------------------------------------
    cap = {}
    for tag, base in (("g3_int8_c32_body", CAP_G3), ("g1_fp16_c32_body", CAP_G1)):
        p = base / "dump_skip4.bin"
        cap[tag] = {"path": p.as_posix(), "bytes": p.stat().st_size, "sha256": sha(p),
                    "half": p.stat().st_size // 2,
                    "matches_input_allocation": bool(p.stat().st_size == IW * IH * K * 2)}
        cap[tag]["_data"] = np.fromfile(p, dtype="<f2")
    cap_meta = {k: {kk: vv for kk, vv in v.items() if kk != "_data"} for k, v in cap.items()}
    v_g3 = pooled_values(cap["g3_int8_c32_body"]["_data"])
    v_g1 = pooled_values(cap["g1_fp16_c32_body"]["_data"])

    # ---- 3. weight tile + parent emulation validated on the captured pair --
    wpad = np.fromfile(FP16_PACK, dtype="<f2")
    w_tiled = wpad[WB_HALF:WB_HALF + N * K].copy()
    w_can = build_w_canonical(w_tiled)
    rt = np.empty_like(w_tiled)
    for n in range(N):
        for k in range(K):
            rt[((n // 16) * (K // 16) + k // 16) * 256 + (n % 16) * 16 + (k % 16)] = w_can[n, k]
    roundtrip_ok = bool(np.array_equal(rt, w_tiled))

    ref1 = np.fromfile(CAP_G1 / "dump_block4_out.bin", dtype="<f2")
    _, out1 = parent_fp16_out(v_g1, w_can.T)
    buf1, hits1 = store_output(out1, DST_LAYOUT)
    d1 = np.abs(buf1.astype(np.float32) - ref1.astype(np.float32))
    den1 = max(float(np.abs(ref1.astype(np.float32)).max()), 1e-30)
    emu = {
        "capture": {"path": (CAP_G1 / "dump_block4_out.bin").as_posix(),
                    "bytes": (CAP_G1 / "dump_block4_out.bin").stat().st_size,
                    "sha256": sha(CAP_G1 / "dump_block4_out.bin"),
                    "half": int(ref1.size),
                    "matches_output_allocation": bool(ref1.size == OW * OH * N)},
        "store_coverage": {"written_once": int((hits1 == 1).sum()), "unwritten": int((hits1 == 0).sum()),
                           "multi_written": int((hits1 > 1).sum()),
                           "verdict": bool((hits1 == 1).all())},
        "elements": int(ref1.size),
        "max_abs": float(d1.max()), "rel_of_max": float(d1.max() / den1),
        "rms": float(np.sqrt((d1.astype(np.float64) ** 2).mean())),
        "nrmsr_of_max": float(np.sqrt((d1.astype(np.float64) ** 2).mean()) / den1),
        "bit_equal_fraction": float(np.equal(buf1, ref1).mean()),
        "parent_absmax": den1,
        "verdict": "LAYOUT_AND_MATH_CONFIRMED" if float(d1.max() / den1) < 2e-3 else "FAIL",
        "note": ("the parent MMA accumulates in fp32 in an implementation-defined order; this emulation "
                 "uses one fp32 order, so bit identity is not expected. The check confirms the "
                 "ds_pixel/io_at/fragment-store mapping and the orientation of the weight tile."),
    }
    # orientation control: read the 16x16 tile transposed (wrong) and rebuild
    w_alt = np.empty((N, K), np.float16)
    for n in range(N):
        for k in range(K):
            w_alt[n, k] = w_tiled[((n // 16) * (K // 16) + k // 16) * 256 + (k % 16) * 16 + (n % 16)]
    _, out_alt = parent_fp16_out(v_g1, w_alt.T)
    buf_alt, _ = store_output(out_alt, DST_LAYOUT)
    d_alt = np.abs(buf_alt.astype(np.float32) - ref1.astype(np.float32))
    emu["orientation_control"] = {"max_abs": float(d_alt.max()),
                                  "rel_of_max": float(d_alt.max() / den1),
                                  "note": "transposed inner-tile reading; must be far worse than the accepted mapping"}
    proof["layout_validation"] = emu
    proof["weight_canonical_roundtrip"] = roundtrip_ok
    proof["cpu_only"] = True
    proof["verdict"] = bool(
        proof["fragment_store_identity"]["layout0_spatial4x4"]["verdict"]
        and proof["fragment_store_identity"]["layout1_raster"]["verdict"]
        and proof["dispatch_store_coverage"]["layout0_spatial4x4"]["verdict"]
        and proof["dispatch_store_coverage"]["layout1_raster"]["verdict"]
        and proof["input_bijection"]["verdict"]
        and emu["store_coverage"]["verdict"]
        and emu["verdict"].startswith("LAYOUT")
        and roundtrip_ok)
    (EVID / "LAYOUT-PROOF.json").write_text(json.dumps(proof, indent=1) + "\n", encoding="utf-8")

    # ---- 4. static steps -------------------------------------------------
    vmax = float(np.abs(v_g3.astype(np.float32)).max())
    sV = f32(f32(vmax) / f32(127.0))
    invV = f32(f32(127.0) / f32(vmax))
    wamax = np.abs(w_can.astype(np.float32)).max(axis=1)
    sW = np.array([f32(f32(a) / f32(127.0)) if a > 0 else f32(1.0) for a in wamax], np.float32)
    invW = np.array([f32(f32(127.0) / f32(a)) if a > 0 else f32(1.0) for a in wamax], np.float32)

    qv = rne_clip(v_g3.astype(np.float32) * invV).astype(np.int32)
    qw = np.empty((N, K), np.int32)
    for n in range(N):
        qw[n] = rne_clip(w_can[n].astype(np.float32) / sW[n]).astype(np.int32)
    sat_v = int(np.count_nonzero(np.abs(qv) >= 127))
    esc = np.array([f32(f32(sV) * sW[n]) for n in range(N)], np.float32)

    # ---- 5. INT8 oracle vs parent emulation on the deployed arm ----------
    acc_i = qv.astype(np.int64) @ qw.T.astype(np.int64)
    int_exact = bool(int(np.abs(acc_i).max()) < 2 ** 24)
    deq = (acc_i.astype(np.float32) * esc[None, :]).astype(np.float32)
    out_i8 = deq.astype(np.float16)
    _, out_p = parent_fp16_out(v_g3, w_can.T)
    diff = np.abs(out_i8.astype(np.float32) - out_p.astype(np.float32))
    dpmax = float(np.abs(out_p.astype(np.float32)).max())
    per_ch = diff.max(axis=0)
    nr = {
        "elements": int(diff.size), "max_abs": float(diff.max()),
        "rel_of_max": float(diff.max() / max(dpmax, 1e-30)),
        "rms": float(np.sqrt((diff.astype(np.float64) ** 2).mean())),
        "nrmsr_of_max": float(np.sqrt((diff.astype(np.float64) ** 2).mean()) / max(dpmax, 1e-30)),
        "bit_equal_fraction": float(np.equal(out_i8, out_p).mean()),
        "per_channel_max_abs_max": float(per_ch.max()),
        "per_channel_max_abs_mean": float(per_ch.mean()),
        "parent_absmax": dpmax,
        "candidate_absmax": float(np.abs(out_i8.astype(np.float32)).max()),
    }
    # diagnostic against the module gate wording of the parent package
    # (abs 0.01 + rel 0.02 elementwise, NRMSE 0.005) -- measured against the parent
    # emulation on the same captured input, NOT a GPU gate.
    tol = 0.01 + 0.02 * np.abs(out_p.astype(np.float32))
    nr["gate_style"] = {
        "rule": "|candidate-parent| <= 0.01 + 0.02*|parent| elementwise; NRMSE <= 0.005",
        "violations": int((diff > tol).sum()),
        "fraction_within": float((diff <= tol).mean()),
        "worst_ratio": float((diff / tol).max()),
        "nrmse_ok": bool(nr["nrmsr_of_max"] <= 0.005),
        "verdict": "WITHIN_GATE_STYLE_BOUNDS_ON_CPU" if (diff <= tol).all() and nr["nrmsr_of_max"] <= 0.005
                   else "OUTSIDE_GATE_STYLE_BOUNDS",
        "note": "diagnostic only: the real gate needs the allocated GPU slot",
    }
    # ABI self-check: the kernel's own load addressing must equal the pack layout
    abi = {"B": 0, "A": 0}
    for n in range(N):
        for k in range(K):
            n0 = (n // 16) * 16
            off = ((n0 // 16) * (K // 32)) * 512
            kernel_off = off + (n - n0) * 32 + k
            pack_off = (n // 16) * (K // 32) * 512 + (k // 32) * 512 + (n % 16) * 32 + (k % 32)
            abi["B"] += int(kernel_off == pack_off)
    abi["B_pairs"] = N * K
    abi["B_verdict"] = abi["B"] == N * K
    abi["A_note"] = ("coopMatLoad(a_i8, 0, pc.p1.x=32, RowMajor) reads element (m,k) at m*32+k, "
                     "which is exactly the parent's a_half[i] index i=m*K+k used by the pool write")
    abi["scale_deq_block_index"] = {str(n0): [256 + (n0 // 16) * 128, 256 + (n0 // 16) * 128 + 128]
                                    for n0 in (0, 32)}

    # ---- 6. packs --------------------------------------------------------
    i8 = np.zeros(N * K, np.uint8)
    for n in range(N):
        for k in range(K):
            off = (n // 16) * (K // 32) * (32 * 16) + (k // 32) * (32 * 16) + (n % 16) * 32 + (k % 32)
            i8[off] = np.uint8(qw[n, k] & 0xFF)
    wpath = PACKS / "r03_ioport4_ds_w.i8.bin"
    wpath.write_bytes(i8.tobytes())
    # scale table, 1280 float32:
    #   0..3 header, 4..67 dequant steps, 68..131 qx output multipliers,
    #   256..639 dequant step fragments (per N16 block, col-major ld 8),
    #   768..1151 qx multiplier fragments
    oamax = np.abs(out_p.astype(np.float32)).max(axis=0)
    sout = np.array([f32(f32(a) / f32(127.0)) if a > 0 else f32(1.0) for a in oamax], np.float32)
    inv_out = np.array([f32(f32(127.0) / f32(a)) if a > 0 else f32(1.0) for a in oamax], np.float32)
    sc = np.zeros(1280, np.float32)
    sc[0] = sV; sc[1] = invV; sc[2] = f32(sW.max()); sc[3] = f32(f32(1.0) / f32(sW.max()))
    sc[4:68] = esc
    sc[68:132] = inv_out
    for blk in range(4):
        for cc in range(16):
            n = blk * 16 + cc
            for r in range(8):
                sc[256 + blk * 128 + cc * 8 + r] = esc[n]
                sc[768 + blk * 128 + cc * 8 + r] = inv_out[n]
    spath = PACKS / "r03_ioport4_ds_scales.f32.bin"
    spath.write_bytes(sc.tobytes())
    abi["scale_fragment_values_match"] = bool(all(
        sc[256 + (n0 // 16) * 128 + cc * 8 + r] == esc[n0 + cc]
        for n0 in (0, 32) for cc in range(16) for r in range(8)))
    abi["qx_fragment_values_match"] = bool(all(
        sc[768 + (n0 // 16) * 128 + cc * 8 + r] == inv_out[n0 + cc]
        for n0 in (0, 32) for cc in range(16) for r in range(8)))

    # qx arm reference: same int32 accumulator and dequant, then the output multiplier + RNE/clamp
    qx_f = (acc_i.astype(np.float32) * esc[None, :]).astype(np.float32) * inv_out[None, :]
    qx_codes = np.clip(np.rint(qx_f), -127.0, 127.0).astype(np.int32)
    qx_real = (qx_codes.astype(np.float32) * sout[None, :]).astype(np.float16)
    qd = np.abs(qx_real.astype(np.float32) - out_p.astype(np.float32))
    qx_nr = {
        "elements": int(qd.size), "max_abs": float(qd.max()),
        "rel_of_max": float(qd.max() / max(dpmax, 1e-30)),
        "rms": float(np.sqrt((qd.astype(np.float64) ** 2).mean())),
        "nrmsr_of_max": float(np.sqrt((qd.astype(np.float64) ** 2).mean()) / max(dpmax, 1e-30)),
        "code_saturation_pm127": int(np.count_nonzero(np.abs(qx_codes) >= 127)),
        "codes": int(qx_codes.size),
        "distinct_codes": int(np.unique(qx_codes).size),
        "parent_absmax": dpmax,
    }

    # ---- 7. evidence -----------------------------------------------------
    contract = {
        "card": "r03-ioport4",
        "position": {"pass": "ioport_block4_0", "plan_order": 5,
                     "spv_parent": "candidates/amd-map-20261015/round4-integrated/candidate/spv/ds-f33-k32.spv",
                     "parent_sha256": "101924e70cb6eb62384e0e9bba809192e43b1123a1c513197053d041d0c68c30",
                     "grid": [GX, GY, 1],
                     "push": "p0=672,384,336,192 p1=32,64,16768,0 p2=0,0,0,0 p3=0,0,336,192",
                     "bindings_parent": {"0": "ioport_4_ioport_pack 37632B", "2": "skip4 16515072B",
                                         "3": "block4_out 8257536B", "1,4,5,6,7": "4B dummy"}},
