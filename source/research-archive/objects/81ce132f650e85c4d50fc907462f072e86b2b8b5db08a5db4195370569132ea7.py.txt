#!/usr/bin/env python3
"""Assemble the A1 candidate receipts (CANDIDATE.json, jobs.json) from measured files."""
import hashlib
import json
from pathlib import Path

W = Path(r"@NR_WORKSPACE@/work/dsh-xmx-handoff-validation-20261024")
CARD = W / "candidates/a1-d034-gm-drop"
BUILD = W / "build/a1-d034-gm-drop"
ROOT = Path(r"@NR_WORKSPACE@")
RUNNER = ROOT / "build/gpa-b321-prep/port_network_model_runner.exe"
PLAN232 = ROOT / "candidates/dsh-fp16-same-family-20261023/combo-d01-d034-s32k11-v1/plan-combo-d01-d034-s32k11-v1.txt"
DSH232 = Path(r"@NR_WORKSPACE@/work/dsh-fp16-same-family-20261023/main")


def sha(p):
    return hashlib.sha256(Path(p).read_bytes()).hexdigest()


def sz(p):
    return Path(p).stat().st_size


def main():
    spv = json.loads((CARD / "evidence" / "SPV-DELTA.json").read_text(encoding="utf-8"))
    wg = json.loads((BUILD / "a1-wg.json").read_text(encoding="utf-8"))
    edit = json.loads((BUILD / "a1-edit.json").read_text(encoding="utf-8"))
    pdelta = json.loads((CARD / "evidence" / "PLAN-DELTA.json").read_text(encoding="utf-8"))

    widths = {}
    for p in ("c64", "c128", "c256"):
        src = CARD / "source" / p
        parent_src = CARD / "source_parent" / p
        entry = next(src.glob("fused_d034_*.comp"))
        parent_entry = next(parent_src.glob("fused_d034_*.comp"))
        widths[p] = {
            "parent_source": {"path": str(parent_entry), "sha256": sha(parent_entry)},
            "candidate_source": {"path": str(entry), "sha256": sha(entry),
                                 "removed_store_lines": [s["line"] for s in edit["removed"][p]["stores"]]},
            "parent_spv": {"path": str(CARD / "spv" / f"parent_{p}.spv"),
                           "sha256": sha(CARD / "spv" / f"parent_{p}.spv"),
                           "bytes": sz(CARD / "spv" / f"parent_{p}.spv")},
            "candidate_spv": {"path": str(CARD / "spv" / f"a1_{p}.spv"),
                              "sha256": sha(CARD / "spv" / f"a1_{p}.spv"),
                              "bytes": sz(CARD / "spv" / f"a1_{p}.spv")},
            "compile_repro": {
                "argv": [str(ROOT / "src/project/tools/glslang-16.5.0/bin/glslang.exe"),
                         "-V", "--target-env", "vulkan1.3", f"-I<{p} source dir>",
                         "-o", "<out>.spv", entry.name],
                "glslang": "16.5.0 (SPIR-V 1.6 target env vulkan1.3)",
                "parent_reproduced_byte_identical": spv["widths"][p]["parent"]["sha256"]
                == spv["widths"][p]["parent"]["sha256"] and
                spv["widths"][p]["parent"]["sha256"] == sha(CARD / "spv" / f"parent_{p}.spv"),
            },
            "spv_delta": {k: spv["widths"][p][k] for k in
                          ("missing_histogram", "added_empty", "missing_instruction_count",
                           "missing_words", "word_delta_consistent", "missing_float_typed",
                           "gM_store_total_parent", "gM_store_total_child",
                           "loads_parent", "loads_child", "barriers_parent", "barriers_child",
                           "verdict")},
        }

    doc = {
        "card": "a1-d034-gm-drop",
        "date": "2026-10-24",
        "what": "D034 fused kernels (C64/C128/C256) with the four unread gM mirror stores removed; "
                "everything else (mLocal store, barrier, half narrowing, K order, owner, grid, "
                "descriptors, shared size) unchanged.",
        "parent_identity": {
            "plan": str(PLAN232), "plan_sha256": sha(PLAN232), "passes": 232,
            "spvs": {p: widths[p]["parent_spv"] for p in widths},
        },
        "candidate_identity": {
            "plan": str(CARD / "plan-a1-232.txt"),
            "plan_sha256": sha(CARD / "plan-a1-232.txt"),
            "passes": 232,
            "plan_delta": {"changed_lines": pdelta["changed_line_count"],
                           "spv_swaps": pdelta["spv_swaps"],
                           "all_other_lines_identical": True},
            "spvs": {p: widths[p]["candidate_spv"] for p in widths},
        },
        "widths": widths,
        "validators": {
            "spv_validate_cpu": {p: "result=0" for p in widths},
            "plan_lint": {"candidate_problems": 60, "parent_problems": 60,
                          "identical_problem_set": True,
                          "note": "all 60 are the pre-existing 'POLICY last = passes+1' "
                                  "sentinel convention warning, present in the untouched 232 parent"},
            "plan_pool_upload": {"verdict": "PASS", "mode": "production"},
        },
        "dynamic_removed_cost": {
            "affected_passes": wg["affected_passes"],
            "workgroups_per_frame": wg["sum_workgroups"],
            "workgroups_by_spv": wg["by_spv"],
            "coopmat_store_executions_removed_per_frame": wg["removed_coopmat_store_exec_per_frame"],
            "half_written_removed_per_frame": wg["removed_half_per_frame"],
            "bytes_removed_per_frame_source_level": wg["removed_bytes_per_frame"],
            "method": "one gM store per workgroup per affected pass; 8x16 half group store = 128 half "
                      "= 256 B; workgroup counts read from the 232 plan dispatch fields",
            "not_claimed": "not measured DRAM bytes, not native spill, not a millisecond estimate",
        },
        "qualification": {
            "CPU_STRUCT": True, "OFFLINE_COMPILE": True, "SPV_IDENTITY": True,
            "NATIVE": "not run (no GPU slot allocated)",
            "GATE": "not run (no GPU slot allocated)",
            "ORDINARY": "not run (no GPU slot allocated)",
            "QUALITY": "not applicable to this card",
        },
        "diagnostic_caveat": "plans that use --dump-intermediates add the transient M buffer to "
                             "OUTPUT and set resident=1, so the M dump of this candidate is NOT "
                             "byte-equal to the 232 parent's M dump. The M dump stays out of the "
                             "equivalence gate; gates compare frame outputs, not the removed mirror.",
        "gpu_next": "run/a1-d034-gm-drop/jobs.json (gate fixed/fb2 for both arms, ordinary 232 vs A1)",
    }
    (CARD / "CANDIDATE.json").write_text(json.dumps(doc, indent=1) + "\n", encoding="utf-8", newline="\n")

    # ---- jobs: gate + ordinary, both arms ----
    outroot = ROOT / "run/dsh-xmx-handoff-validation-20261024"
    jobs = []
    for arm, plan, passes in (("p232", PLAN232, 232), ("cand-a1", CARD / "plan-a1-232.txt", 232)):
        for kind, reps, extra, gold in (("fixed", 1, [], "1755e9c2fadba97f08cb5b02b7a4294cde3675379bdec75352287505f7e129d6"),
                                        ("fb2", 2, ["--feedback", "--save-frames"],
                                         "9c3e511e97a5879247bc2965792c96dd62928dddfd85a48bbf25af1e807d3536")):
            out = outroot / f"a1-{arm}-{kind}"
            args = ["--plan", str(plan), "--out", str(out), "--pool", "--warm", "0",
                    "--reps", str(reps)] + extra
            jobs.append({"label": f"a1-{arm}-{kind}", "kind": "gate",
                         "purpose": f"numeric gate {kind} ({arm})", "exe": str(RUNNER),
                         "out": str(out), "fresh_out": True, "plan": str(plan),
                         "expected_passes": passes,
                         "gold": gold if arm == "p232" else None,
                         "gold_rule": "candidate must reproduce the parent arm's frame bytes "
                                      "exactly (same gold for both arms)",
                         "args": args})
    for pair in (1, 2, 3):
        for arm, plan in (("p232", PLAN232), ("cand-a1", CARD / "plan-a1-232.txt")):
            out = outroot / f"a1-{arm}-ord{pair}"
            jobs.append({"label": f"a1-{arm}-ord{pair}", "kind": "ordinary",
                         "purpose": f"ordinary whole-graph {arm} pair {pair}", "exe": str(RUNNER),
                         "out": str(out), "fresh_out": True, "plan": str(plan),
                         "expected_passes": 232, "gold": None,
                         "args": ["--plan", str(plan), "--out", str(out), "--pool",
                                  "--warm", "1", "--reps", "30", "--feedback"]})
    (W / "run/a1-d034-gm-drop").mkdir(parents=True, exist_ok=True)
    (W / "run/a1-d034-gm-drop" / "jobs.json").write_text(
        json.dumps({"label": "a1-d034-gm-drop", "plan": str(CARD / "plan-a1-232.txt"),
                    "parent_plan": str(PLAN232), "runner": str(RUNNER),
                    "runner_sha256": sha(RUNNER), "jobs": jobs,
                    "protocol": "same runner/input/pool/warm/feedback as the 232 combo batch; "
                                "3 process pairs x 30 samples, pair = statistical unit"},
                   indent=1) + "\n", encoding="utf-8", newline="\n")
    print("CANDIDATE.json + jobs.json written")
    print("cand plan", doc["candidate_identity"]["plan_sha256"])
    print("jobs", len(jobs))


if __name__ == "__main__":
    main()
