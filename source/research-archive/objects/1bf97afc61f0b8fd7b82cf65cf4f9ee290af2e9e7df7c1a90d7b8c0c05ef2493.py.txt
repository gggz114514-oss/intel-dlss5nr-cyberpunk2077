"""Generate SOURCE-DIFF.patch and PLAN-DIFF.json for n03b-f04-c256 (CPU only)."""
import difflib, json, sys
from pathlib import Path

sys.path.insert(0, r'@NR_WORKSPACE@\work\codex-astra-fp16-int8-audit-20261024\fullnet-continuous-20261027')
import COMMON as K

CARD = Path(__file__).resolve().parent
PARENT = K.D / 'backend-uniform-owner/d034-family/c256/candidate.comp'
CAND = CARD / 'candidate.comp'
BEST, BEST_SHA, _chain = K.current_best()

p = PARENT.read_text(encoding='utf-8-sig').splitlines(keepends=True)
c = CAND.read_text(encoding='utf-8-sig').splitlines(keepends=True)
diff = list(difflib.unified_diff(p, c,
                                fromfile='a/d034-family/c256/candidate.comp (parent, sha256 725c5906...)',
                                tofile='b/cards/n03b-f04-c256/candidate.comp (candidate)',
                                n=6))
(CARD / 'SOURCE-DIFF.patch').write_text(''.join(diff), encoding='utf-8')
added = sum(1 for l in diff if l.startswith('+') and not l.startswith('+++'))
removed = sum(1 for l in diff if l.startswith('-') and not l.startswith('---'))

# plan diff: only SPV path substitutions may differ
base = Path(BEST).read_text(encoding='utf-8-sig').splitlines()
cand = (CARD / 'plan.txt').read_text(encoding='utf-8-sig').splitlines()
bl = [i for i, (a, b) in enumerate(zip(base, cand)) if a != b]
report = dict(parent_plan=dict(path=str(BEST), sha256=BEST_SHA, lines=len(base),
                               role='current accepted chain head'),
              candidate_plan=dict(path=str(CARD / 'plan.txt'), sha256=K.sha(CARD / 'plan.txt'),
                                  lines=len(cand)),
              same_line_count=len(base) == len(cand),
              differing_lines=len(bl),
              changed_line_numbers=[i + 1 for i in bl][:20],
              sample=dict(before=base[bl[0]].strip()[:240], after=cand[bl[0]].strip()[:240]) if bl else None,
              other_lines_identical=(len(base) == len(cand) and len(bl) == 15))
NEW_SPV = str(CARD / 'candidate.spv').replace('\\', '/')


def norm(line):
    """Replace the quoted SPV path field (field 3) with a placeholder."""
    parts = line.split('"')
    if len(parts) >= 4:
        parts[3] = '<SPV>'
    return '"'.join(parts)


report_extra = dict(
    every_changed_line_is_the_c256_spv=bool(bl) and all(
        norm(base[i]) == norm(cand[i]) and 'c256/candidate.spv' in base[i]
        for i in bl))
report.update(report_extra)
K.write_json(CARD / 'PLAN-DIFF.json', report)
print(json.dumps(dict(patch_added=added, patch_removed=removed, plan=report), indent=2)[:3000])
