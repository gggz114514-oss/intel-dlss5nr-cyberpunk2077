Self-contained word-level SPIR-V parser: no spirv-dis/spirv-tools on this host.
Answers three questions for the parent module:

  1. where is the contract seed computed, widened and first consumed (program order,
     with word offsets), i.e. is the widen already sunk to the dequant epilogue?
  2. is the `NRH_C seedN = ycN` copy still a real instruction (OpCopyObject) or has
     the front end renamed it away (SSA identity)?
  3. what is the exact per-opcode multiset, so parent/child can be diffed as
     (opcode, word count) histograms.
"""

RESULT_AT_ZERO = set(range(19, 34)) | {7, 11, 33, 248, 249, 250, 251, 252,
                                       253, 254, 255}
# opcodes with no result id at all
NO_RESULT = {0, 3, 4, 5, 6, 8, 10, 14, 15, 16, 17, 71, 72, 73, 74, 75,
             62, 63, 64, 246, 247, 249, 250, 251, 252, 253, 254, 255,
             5368, 5369}
# Extension opcodes keep the generic (result type, result id) shape of the core.


class Inst:
    __slots__ = ("op", "ops", "pos", "n", "idx")

    def __init__(self, op, ops, pos, n, idx):
        self.op, self.ops, self.pos, self.n, self.idx = op, ops, pos, n, idx

    @property
    def name(self):
        return OPNAME.get(self.op, "Op%s" % self.op)

    @property
    def rid(self):
        if self.op in NO_RESULT or not self.ops:
            return None
        if self.op in RESULT_AT_ZERO:
            return self.ops[0]
        if len(self.ops) < 2:
            return None
        return self.ops[1]

    @property
    def rtype(self):
        if self.op in NO_RESULT or not self.ops:
            return None
        if self.op in RESULT_AT_ZERO:
            return None
        return self.ops[0]

    def text(self):
        parts = [str(x) for x in self.ops]
        return "%-30s %s" % (self.name, " ".join(parts))


class Module:
    def __init__(self, path):
        self.path = path
        data = open(path, "rb").read()
        assert len(data) % 4 == 0, path
        self.nbytes = len(data)
        self.words = list(struct.unpack("<%dI" % (len(data) // 4), data))
        assert self.words[0] == 0x07230203, hex(self.words[0])
        self.version, self.generator, self.bound, self.schema = self.words[1:5]
        self.insts = []
        self.blocks = []       # (function index, label inst index)
        i, idx = 5, 0
        while i < len(self.words):
            w = self.words[i]
            op, n = w & 0xFFFF, w >> 16
            assert n >= 1 and i + n <= len(self.words), (i, n)
            self.insts.append(Inst(op, self.words[i + 1:i + n], i, n, idx))
            i += n
            idx += 1
        self.by_id = {}
        for ins in self.insts:
            rid = ins.rid
            if rid is not None and rid != 0:
                self.by_id.setdefault(rid, ins)
        self.uses = {}
        for ins in self.insts:
            for p, w in enumerate(ins.ops):
                if rid_is_skipped(ins.op, p):
                    continue
                self.uses.setdefault(w, []).append(ins)

    def funcs(self):
        out, cur = [], None
        for ins in self.insts:
            if ins.op == 54:
                cur = {"fn": ins, "params": [], "insts": []}
                out.append(cur)
            elif cur is not None:
                if ins.op == 55:
                    cur["params"].append(ins)
                if ins.op != 56:
                    cur["insts"].append(ins)
        return out

    def exts(self):
        return [(i.ops[1].to_bytes(4, "little").split(b"\0")[0].decode(),
                 i.ops[0]) for i in self.insts if i.op == 11]

    def hist(self, insts=None):
        h = {}
        for ins in (insts if insts is not None else self.insts):
            h[(ins.op, ins.n)] = h.get((ins.op, ins.n), 0) + 1
        return h


def rid_is_skipped(op, pos):
    """True for operand positions that are not ID *uses* (result id, literals)."""
    if op in NO_RESULT:
        return False
    if op in RESULT_AT_ZERO:
        return pos == 0
    if pos == 1:
        return True
    if pos == 0:
        return True          # result type id - not a value use
    return False


def load(path):
    return Module(path)
