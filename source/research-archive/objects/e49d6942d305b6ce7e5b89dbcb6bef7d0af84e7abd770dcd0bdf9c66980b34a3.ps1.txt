$ErrorActionPreference = 'Stop'
$gameRoot = '@NR_GAME@/bin/x64'
if (Get-Process -Name Cyberpunk2077 -ErrorAction SilentlyContinue) { throw 'Cyberpunk is already running' }
$lockPath = '@NR_GPU_WORKSPACE@/artifacts/parallel-port-20261013/GPU.lock'
$stream = [IO.File]::Open($lockPath,[IO.FileMode]::Open,[IO.FileAccess]::ReadWrite,[IO.FileShare]::None)
$owned = $false
$record = $null
function Write-LockRecord {
    $bytes = [Text.Encoding]::UTF8.GetBytes(($record | ConvertTo-Json -Depth 6))
    $stream.Position=0; $stream.SetLength(0); $stream.Write($bytes,0,$bytes.Length); $stream.Flush($true)
}
try {
    $reader = [IO.StreamReader]::new($stream,[Text.Encoding]::UTF8,$true,4096,$true)
    $previous = $reader.ReadToEnd() | ConvertFrom-Json
    $reader.Dispose()
    if ($previous.state -ne 'FINISHED') { throw "GPU queue is not finished: $($previous.owner) / $($previous.state)" }
    $record = @{owner='CP2077-OPTIMIZED-USER-TEST-20261028';holder_pid=$PID;gpu_pid=0;state='RUNNING';started_utc=[DateTime]::UtcNow.ToString('o');lock_mode='Existing GPU.lock FileShare.None handle held for game lifetime';authorization='User 2026-10-28: reproduce optimized Small game opening-animation GPU crash, one run only'}
    Write-LockRecord
    $owned=$true
    $gameProcess = Start-Process -FilePath (Join-Path $gameRoot 'Cyberpunk2077.exe') -WorkingDirectory $gameRoot -PassThru
    $record.gpu_pid=$gameProcess.Id
    Write-LockRecord
    Write-Host 'Keep this window open while testing. The existing GPU lock is held until the game exits.'
    $gameProcess.WaitForExit()
    $record.game_exit_code=$gameProcess.ExitCode
} finally {
    if ($owned) {
        $record.state='FINISHED'; $record.finished_utc=[DateTime]::UtcNow.ToString('o')
        Write-LockRecord
        $record | ConvertTo-Json -Depth 6 | Set-Content -LiteralPath (Join-Path $PSScriptRoot 'USER-TEST-GPU-RECEIPT.json') -Encoding UTF8
    }
    $stream.Dispose()
}
