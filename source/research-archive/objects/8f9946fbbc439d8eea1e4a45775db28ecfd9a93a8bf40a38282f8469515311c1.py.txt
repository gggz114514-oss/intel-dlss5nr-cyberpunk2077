"""Decisive control: pad the ADOPTED F05 arena back to 24576 B and see if the win disappears.

n36 won +0.117 and +0.126 ms with a change that did two things at once: it cut the arena
24576 -> 16384 B (residency 2 -> 4) and it moved the V projection after score. F10 and F20 then
cut shared memory for residency alone and won nothing, so it is unclear whether n36's gain came
from residency at all. This probe keeps n36's restructured code and only restores the array size,
which separates the two effects.
"""
import sys
from pathlib import Path
sys.path.insert(0, str(Path(__file__).parent))
import COMMON as K
from audit_card import pass_lines

card = K.C / 'cards' / 'n36-f05-arena-shrink'
out = K.C / 'queue' / 'control-f05-pad'
out.mkdir(parents=True, exist_ok=True)
cur = K.read_json(K.C / 'CURRENT.json')
head = Path(cur['best_plan']['path'])
recs = pass_lines(head)

groups = {}
for r in K.rows():
    if r['family'] != 'F05':
        continue
    groups.setdefault(recs[r['ordinal']]['spv'], []).append(r['ordinal'])
print('F05 groups at head:', {Path(k).name: len(v) for k, v in groups.items()})

nb36 = K.read_json(card / 'BUILD.json')
n05a = nb36
hdr = sorted({str(Path(h['path']).parent) for h in n05a['include_headers']})
R = K.read_json(K.C / 'evidence' / 'RECIPES.json')

repoint = {}
for spv, ords in groups.items():
    sha = K.sha(Path(spv))
    name = Path(spv).name
    if name.startswith('candidate-c'):
        src = card / name.replace('.spv', '.comp')
        tag = name.split('-')[-1].split('.')[0]
        hdrs = nb36['groups'][tag]['include_headers']
        inc = sorted({str(Path(h['path']).parent) for h in hdrs}) + [str(card)]
    else:
        src = Path(R[sha]['source'])
        inc = [R[sha]['compile']['pinned_include_dir']] + list(R[sha]['compile']['include_dirs'])
    rep = out / ('repro-%s.spv' % Path(spv).stem)
    K.compile_glsl(src, rep, include_dirs=inc, defines=())
    print('  %-28s reproduced: %s' % (name, K.sha(rep) == sha))
    text = src.read_text(encoding='utf-8')
    assert 'qkv[8192]' in text, (name, 'expected the shrunk declaration')
    padc = out / ('pad-%s.comp' % Path(spv).stem)
    padc.write_text(text.replace('qkv[8192]', 'qkv[12288]'), encoding='utf-8')
    pspv = out / ('pad-%s.spv' % Path(spv).stem)
    K.compile_glsl(padc, pspv, include_dirs=inc, defines=())
    K.validate_spv(pspv)
    repoint[spv] = dict(ords=ords, pad=pspv.as_posix())

lines = head.read_text(encoding='utf-8-sig').splitlines()
n = 0
for spv, info in repoint.items():
    for o in info['ords']:
        line = recs[o]['line']
        toks = line.split()
        toks[-6] = '24576'
        new = ' '.join(toks).replace(Path(spv).as_posix(), info['pad'])
        idx = next(i for i, l in enumerate(lines) if l == line)
        lines[idx] = new
        n += 1
plan = out / 'plan.txt'
plan.write_text('\n'.join(lines) + '\n', encoding='utf-8')
print('rows repointed:', n, 'plan sha', K.sha(plan))
print(K.cpu_check(plan).strip()[-90:])
