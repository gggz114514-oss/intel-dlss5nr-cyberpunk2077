"""scratch: final sanity check of the card deliverables."""
import json
from pathlib import Path

d = Path(".").resolve()
for f in ("BUILD.json", "PROOF.json", "COST-LEDGER.json", "A-AUDIT.json", "A-ARITH.json",
          "C-REDUNDANCY.json", "C-CONSUMERS.json", "A-LIVE.json", "A-DISTANCE.json"):
    p = d / f
    j = json.loads(p.read_text(encoding="utf-8"))
    print("%-18s %8d B  keys=%d" % (f, p.stat().st_size, len(j)))

b = json.loads((d / "BUILD.json").read_text(encoding="utf-8"))
print("\nstatus      :", b["status"])
print("plan sha    :", b["plan"]["sha256"])
print("passes repl :", b["plan"]["passes_replaced"])
print("float_NC    :", b["float_NC_checked"], "parent", b["float_NC_parent"])
print("parent repro:", b["parent_rebuild"]["byte_match"], b["parent_rebuild"]["sha256"][:16])
print("candidate   :", b["candidate"]["spv_sha256"][:16], b["candidate"]["spv_bytes"], "B")
print("cpu_check   :", b["cpu_check"].strip())
print("dispatch    :", b["dispatch_authorization"])

a = json.loads((d / "A-AUDIT.json").read_text(encoding="utf-8"))
print("\nhistory     :", a["histogram"], "live", a["live"], "/", a["total"])
r = json.loads((d / "C-REDUNDANCY.json").read_text(encoding="utf-8"))
print("redundancy  : sites", r["n_sites"], "distinct defs", r["n_distinct_producers"],
      "census said", r["census_redundant"], "actual", r["actual_same_value_pairs"])
p = json.loads((d / "PROOF.json").read_text(encoding="utf-8"))
print("proof counts:", p["counts"])
print("distances   :", p["distances"]["first_consumer"]["parent"])
