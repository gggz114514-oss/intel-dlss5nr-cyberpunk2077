"""n06d emit: SOURCE-DIFF.patch, COST-LEDGER.json, shT/barrier footprint accounting.

CPU only. Reads the shipped candidate.comp / candidate.spv / BUILD.json / PROOF.json.

import spvstat

BUILD = K.read_json(CARD / 'BUILD.json')

BASE_ISA = NATIVE['base']


def write_patch():
    a = PARENT_SRC.read_text(encoding='utf-8-sig').splitlines(keepends=True)
    b = (CARD / 'candidate.comp').read_text(encoding='utf-8-sig').splitlines(keepends=True)
    diff = list(difflib.unified_diff(
        a, b, fromfile='a/fullnet-mechanism-transfer-20261026/f06-scale-fragments/'
                       'family/c256/candidate.comp',
        tofile='b/cards/n06d-f06-c256-atile/candidate.comp', n=4))
    (CARD / 'SOURCE-DIFF.patch').write_text(''.join(diff), encoding='utf-8')
    hunks = sum(1 for l in diff if l.startswith('@@'))
    adds = sum(1 for l in diff if l.startswith('+') and not l.startswith('+++'))
    dels = sum(1 for l in diff if l.startswith('-') and not l.startswith('---'))
    return dict(path=str(CARD / 'SOURCE-DIFF.patch'), hunks=hunks, added_lines=adds,
                removed_lines=dels, sha256=K.sha(CARD / 'SOURCE-DIFF.patch'),
                parent_source_sha256=K.sha(PARENT_SRC),
                candidate_source_sha256=K.sha(CARD / 'candidate.comp'))


def shT_footprint():
    pm = spvstat.Mod(CARD / 'parent-rebuild.spv')
    cm = spvstat.Mod(CARD / 'candidate.spv')
    return dict(parent=dict(workgroup_bytes=32768, arrays=1, barriers=pm.hist.get(224, 0),
                            instr=len(pm.instr), words=len(pm.words),
                            loops=pm.hist.get(246, 0)),
                candidate=dict(workgroup_bytes=32768, arrays=1, barriers=cm.hist.get(224, 0),
                               instr=len(cm.instr), words=len(cm.words),
                               loops=cm.hist.get(246, 0)),
                note='shT is the same half[16384] Workgroup array in both arms (32768 B); '
                     'only the slot a given element lands in changed')
