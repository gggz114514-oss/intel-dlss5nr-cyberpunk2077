"""Scan a native capture directory: instruction counts per ISA file, and pick the kernel matching a target count."""
import json, sys
from pathlib import Path
sys.path.insert(0, str(Path(__file__).parent))
import COMMON as K
import ISA

def scan(d):
    out = []
    for p in sorted(Path(d).rglob('*.asm')):
        try:
            s = ISA.summary(p)
        except Exception as e:
            out.append(dict(path=str(p), error=str(e)))
            continue
        out.append(dict(path=str(p), sha256=K.sha(p), instructions=s['instructions'],
                        sends=s['sends'], dpas=s['dpas'], syncs=s['syncs'],
                        max_r=s['max_r'], first_dpas=s['first_dpas'],
                        msgs=s['message_comments'][:6]))
    return out

if __name__ == '__main__':
    d = sys.argv[1]
    rows = scan(d)
    rows.sort(key=lambda r: -r.get('instructions', 0))
    out = Path(d) / 'ISA-SCAN.json'
    K.write_json(out, rows)
    for r in rows[:14]:
        print('%-52s instr=%5s send=%4s dpas=%4s sync=%3s maxr=%4s' % (
            Path(r['path']).name[:52], r.get('instructions'), r.get('sends'), r.get('dpas'),
            r.get('syncs'), r.get('max_r')))
    print('total', len(rows), '->', out)
