import sys, json, collections
from pathlib import Path
C = Path("@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/fullnet-continuous-20261027")
sys.path.insert(0, str(C))
import COMMON as K
import spv_tools_local as ST

CARD = C / "cards" / "n05a-f05-c256-mreuse"
P = Path(r"@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/fullnet-mechanism-transfer-20261026/f05-k-consumer-tile/c256/c256/candidate.spv")
Cd = CARD / "candidate.spv"

NAMES = {61: "OpLoad", 62: "OpStore", 65: "OpAccessChain", 80: "OpCompositeConstruct",
         81: "OpCompositeExtract", 82: "OpCompositeInsert", 86: "OpFAdd", 88: "OpFMul",
         12: "OpExtInst", 42: "OpConstantNull", 43: "OpConstant", 124: "OpAny",
         128: "OpSNegate", 129: "OpIAdd", 130: "OpFAdd?", 132: "OpIAdd2", 133: "OpIMul",
         134: "OpUDiv", 136: "OpUMod", 137: "OpSRem", 194: "OpShiftLeftLogical",
         199: "OpNot", 205: "OpULessThan", 218: "OpSelect", 245: "OpPhi",
         249: "OpBranch", 250: "OpBranchConditional", 248: "OpLabel",
         4456: "OpCooperativeMatrixConvertKHR", 4457: "OpCooperativeMatrixLoadKHR",
         4458: "OpCooperativeMatrixStoreKHR", 4459: "OpCooperativeMatrixMulAddKHR",
         4460: "OpCooperativeMatrixLengthKHR"}

def hist(m):
    c = collections.Counter()
    for op, ops, _, _ in m.instructions:
        c[op] += 1
    return c

pm = ST.Module(P)
cm = ST.Module(Cd)
ph, ch = hist(pm), hist(cm)
keys = sorted(set(ph) | set(ch), key=lambda o: -(ph.get(o, 0) + ch.get(o, 0)))
print("%-38s %7s %7s %7s" % ("opcode", "parent", "cand", "delta"))
rows = []
for o in keys:
    a, b = ph.get(o, 0), ch.get(o, 0)
    if a == b and NAMES.get(o) is None:
        continue
    rows.append((NAMES.get(o, "op%d" % o), a, b, b - a))
for n, a, b, d in rows:
    if a != b:
        print("%-38s %7d %7d %+7d" % (n, a, b, d))
print()
print("total instr: parent %d cand %d (%+d)" % (sum(ph.values()), sum(ch.values()),
                                                sum(ch.values()) - sum(ph.values())))
print("spv bytes : parent %d cand %d" % (P.stat().st_size, Cd.stat().st_size))
print()
for lab, m in (("parent", pm), ("cand", cm)):
    print(lab, "storage_bytes:", json.dumps(m.storage_bytes()))
    print(lab, "bindings:", m.binding_ids())
