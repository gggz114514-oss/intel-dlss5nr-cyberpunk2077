def sha(p):
    return hashlib.sha256(pathlib.Path(p).read_bytes()).hexdigest()


def compile_glsl(comp, spv, include_dirs):
    argv = [str(GLSLANG), '-V', '--target-env', 'vulkan1.3']
    for d in include_dirs:
        argv.append('-I' + str(d))
    argv += [str(comp), '-o', str(spv)]
    r = subprocess.run(argv, capture_output=True, text=True,
                       encoding='utf-8', errors='replace')
    if r.returncode != 0:
        raise SystemExit('compile failed: %s\n%s\n%s' % (comp, r.stdout, r.stderr))
    return argv


def main():
    out = {}
    (CARD / 'build').mkdir(parents=True, exist_ok=True)
    for g, spec in GROUPS.items():
        rec = dict(group=g, passes=spec['passes'])
        rec['spv'] = dict(path=str(spec['spv']), sha256=sha(spec['spv']))
        rec['source'] = dict(path=str(spec['source']), sha256=sha(spec['source']))
        assert rec['spv']['sha256'] == spec['spv_sha'], (g, 'parent spv sha changed')
        assert rec['source']['sha256'] == spec['source_sha'], (g, 'parent source sha changed')
        rec['headers'] = []
        for name, want in spec['headers'].items():
            hit = None
            for d in spec['includes']:
                if (d / name).exists():
                    hit = d / name
                    break
            assert hit is not None, (g, 'header not found', name)
            got = sha(hit)
            rec['headers'].append(dict(name=name, path=str(hit), sha256=got,
                                       expected=want, ok=got == want))
            assert got == want, (g, 'header sha changed', name, got, want)
        raw = CARD / 'build' / ('parent-%s-raw.spv' % g)
        rec['argv'] = compile_glsl(spec['source'], raw, spec['includes'] + [spec['source'].parent])
        rec['raw_sha256'] = sha(raw)
        rec['raw_match'] = rec['raw_sha256'] == spec['spv_sha']
        if rec['raw_match']:
            shutil.copyfile(raw, CARD / ('parent-rebuild-%s.spv' % g))
            rec['parent_rebuild'] = dict(path=str(CARD / ('parent-rebuild-%s.spv' % g)),
                                         sha256=sha(CARD / ('parent-rebuild-%s.spv' % g)))
        out[g] = rec
        print('%-5s raw=%s match=%s' % (g, rec['raw_sha256'][:16], rec['raw_match']))
    (CARD / 'build/repro.json').write_text(json.dumps(out, indent=2), encoding='utf-8')
    ok = all(r['raw_match'] for r in out.values())
    print('ALL THREE PARENTS REPRODUCED BYTE-EXACTLY' if ok else 'REPRODUCTION FAILED')
    return 0 if ok else 1


if __name__ == '__main__':
    sys.exit(main())
