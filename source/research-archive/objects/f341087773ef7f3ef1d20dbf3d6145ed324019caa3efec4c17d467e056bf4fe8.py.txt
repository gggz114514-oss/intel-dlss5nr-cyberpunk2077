"""Split the F20 (ord 230) ISA capture into regions delimited by sync.bar and
tally opcode classes + SLM/UGM message kinds per region.

Read-only analysis: the ISA capture is never modified.
"""
import re, sys, json, collections
from pathlib import Path

sys.stdout.reconfigure(encoding='utf-8')

ISA = Path(r'@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024'
           r'/fullnet-mechanism-transfer-20261026/f10-weight-return-separation-20261027'
           r'/captures/native-f20-pixel-owner/ISA'
           r'/CS_asm76902ef344609c8a_simd32_0000000000000027_38_orig.asm')

LINE = re.compile(r'^/\* \[(?P<addr>[0-9A-F]+)\]  \*/ (?P<pre>.*?)\s{2,}(?P<op>[a-z][a-z0-9._]*)\s+'
                  r'(?P<rest>.*)$')


def parse(path):
    ins = []
    for ln, raw in enumerate(path.read_text(encoding='utf-8', errors='replace').splitlines(), 1):
        m = LINE.match(raw.rstrip())
        if not m:
            continue
        rest = m.group('rest')
        msg = ''
        if '//' in rest:
            msg = rest.split('//', 1)[1].strip()
        ins.append(dict(ln=ln, addr=int(m.group('addr'), 16), op=m.group('op'),
                        pred=m.group('pre').strip(), rest=rest, msg=msg))
    return ins


def main():
    ins = parse(ISA)
    print('instructions parsed:', len(ins), 'last addr', hex(ins[-1]['addr']))

    # region boundaries
    bars = [i for i, x in enumerate(ins) if x['op'] == 'sync.bar']
    print('sync.bar count:', len(bars), 'at', [hex(ins[i]['addr']) for i in bars])
    bounds = [0] + bars + [len(ins)]
    regions = []
    for k in range(len(bounds) - 1):
        lo, hi = bounds[k], bounds[k + 1]
        chunk = ins[lo:hi]
        ops = collections.Counter(x['op'] for x in chunk)
        msgs = collections.Counter(x['msg'] for x in chunk if x['msg'])
        regions.append(dict(idx=k, first_ln=ins[lo]['ln'], last_ln=ins[hi - 1]['ln'],
                            addr0=hex(ins[lo]['addr']), addr1=hex(ins[hi - 1]['addr']),
                            n=len(chunk), ops=dict(ops), msgs=dict(msgs)))
    for r in regions:
        print('\n== region %d  lines %d..%d  addr %s..%s  insns %d'
              % (r['idx'], r['first_ln'], r['last_ln'], r['addr0'], r['addr1'], r['n']))
        top = sorted(r['ops'].items(), key=lambda kv: -kv[1])[:12]
        print('   ops:', ', '.join('%s=%d' % kv for kv in top))
        slm = {k: v for k, v in r['msgs'].items() if 'slm' in k}
        ugm = {k: v for k, v in r['msgs'].items() if 'ugm' in k or 'gtwy' in k}
        print('   slm:', json.dumps(slm, indent=None))
        print('   ugm:', json.dumps(ugm, indent=None))
    Path('cards/n28-f20-barrier-alu/tools/isa-regions.json').write_text(
        json.dumps(regions, indent=2), encoding='utf-8')


if __name__ == '__main__':
    main()
