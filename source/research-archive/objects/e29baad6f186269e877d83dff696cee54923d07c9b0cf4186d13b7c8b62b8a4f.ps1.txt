$ErrorActionPreference='Stop'
$game='@NR_GAME@/bin/x64'
$package='@NR_WORKSPACE@/work/github-experimental-game-install-20261024/package'
try { & (Join-Path $PSScriptRoot 'Launch-Game.ps1') }
finally {
 if (!(Get-Process Cyberpunk2077 -ErrorAction SilentlyContinue)) {
  if (Test-Path -LiteralPath "$game/plugins/dsh_diag_control.ini") {
   Remove-Item -LiteralPath "$game/plugins/dsh_diag_control.ini" -Force -ErrorAction SilentlyContinue
  }
  Copy-Item -LiteralPath "$game/plugins/CyberpunkNRBridge.log" -Destination "$PSScriptRoot/after-CyberpunkNRBridge.log" -Force
  Copy-Item -LiteralPath "$package/config/component_status.json" -Destination "$PSScriptRoot/after-component_status.json" -Force
  Copy-Item -LiteralPath "$package/config/panel_state.json" -Destination "$PSScriptRoot/after-panel_state.json" -Force
  Copy-Item -LiteralPath "$PSScriptRoot/before/game/plugins/CyberpunkNRBridge.asi" -Destination "$game/plugins/CyberpunkNRBridge.asi" -Force
  Copy-Item -LiteralPath "$PSScriptRoot/before/game/nrvk_bridge.dll" -Destination "$game/nrvk_bridge.dll" -Force
  Copy-Item -LiteralPath "$PSScriptRoot/before/game/plugins/nr_bridge.ini" -Destination "$game/plugins/nr_bridge.ini" -Force
  Copy-Item -LiteralPath "$PSScriptRoot/before/config/nr_panel.ini" -Destination "$package/config/nr_panel.ini" -Force
  $m=Get-Content -Raw (Join-Path $PSScriptRoot 'INSTALL.json')|ConvertFrom-Json
  $entries=@()
  $targets=@{ 'CyberpunkNRBridge.asi'="$game/plugins/CyberpunkNRBridge.asi"; 'nrvk_bridge.dll'="$game/nrvk_bridge.dll"; 'nr_bridge.ini'="$game/plugins/nr_bridge.ini"; 'nr_panel.ini'="$package/config/nr_panel.ini" }
  foreach($name in $targets.Keys){$h=(Get-FileHash -LiteralPath $targets[$name] -Algorithm SHA256).Hash.ToLowerInvariant();$entries+=[ordered]@{name=$name;expected=$m.original_sha256.PSObject.Properties[$name].Value;actual=$h;match=($h -eq $m.original_sha256.PSObject.Properties[$name].Value)}}
  [IO.File]::WriteAllText((Join-Path $PSScriptRoot 'RESTORE-VERIFY.json'),([ordered]@{checked_utc=[DateTime]::UtcNow.ToString('o');files=$entries;all_match=(@($entries|Where-Object{-not $_.match}).Count -eq 0)}|ConvertTo-Json -Depth 6),[Text.UTF8Encoding]::new($false))
  [IO.File]::WriteAllText((Join-Path $PSScriptRoot 'RESTORED.txt'),'DSH diagnosis game run ended; original four files restored and hash-checked.',[Text.UTF8Encoding]::new($false))
 } else { throw 'Game still alive: do not restore loaded files' }
}
