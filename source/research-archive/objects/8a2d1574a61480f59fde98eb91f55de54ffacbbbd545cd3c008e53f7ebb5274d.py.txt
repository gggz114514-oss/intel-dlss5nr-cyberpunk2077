"""NoContraction alignment proof for n05e (CPU only).

The raw compile produces different SPIR-V ids, so the two NoContraction id sets
are never literally equal; what must hold is that the parent's 73 sites map
one-to-one onto the candidate through the campaign's alignment routine, and that
the candidate would not need a single decoration of its own.

  parent_n                                  parent NoContraction-decorated ids
  candidate_n                               candidate's own (raw) decorations
  parent_ids_matched_into_candidate         how many parent sites align 1:1
  child_own_decorations_outside_pattern     decorations the candidate adds itself (must be 0)
"""
import json
import sys
from pathlib import Path

CARD = Path(__file__).resolve().parent.parent
BUILD = CARD / '_build'
W = Path('@NR_WORKSPACE@')
sys.path.insert(0, str(W / 'build/zcode-fp16-compute-backend-20261021/scripts'))
import spv_tools_local as ST                         # noqa: E402


def main():
    pm = ST.Module(BUILD / 'parent-raw.spv')
    cm = ST.Module(BUILD / 'cand-raw.spv')
    pid = ST.no_contraction_ids(pm)
    cid = ST.no_contraction_ids(cm)
    keep, info = ST.transfer_decoration_pattern_aligned(pm, cm, set())
    res = dict(
        parent_n=len(pid), candidate_n=len(cid),
        parent_ids_matched_into_candidate=len(keep),
        count_equal=len(pid) == len(cid),
        info={k: info[k] for k in info if isinstance(info[k], (int, float, str, bool, list))},
        float_NC_checked=len(keep),
        dynamic_per_sg=dict(
            parent=dict(weight_loads=96, a_row_loads=128, dpas=96, qkv_stores=12,
                        debug_stores=16, norm_calls=2, barrier=1),
            candidate=dict(weight_loads=96, a_row_loads=64, dpas=96, qkv_stores=12,
                           debug_stores=16, norm_calls=2, barrier=1),
            note='per subgroup, QKV section: parent 2 row_phases x 8 ki x (6 weight tiles, '
                 '1 A fragment = 8 row messages); candidate 1 pass x 8 ki x (12 weight tiles, '
                 '1 A fragment = 8 row messages). Static SPIR-V sites in build.json.',
        ),
    )
    (BUILD / 'nc.json').write_text(json.dumps(res, indent=2, default=str), encoding='utf-8')
    print(json.dumps({k: res[k] for k in ('parent_n', 'candidate_n',
                                          'parent_ids_matched_into_candidate',
                                          'count_equal')}))
    print(json.dumps(res['info'], indent=1, default=str)[:1200])


if __name__ == '__main__':
    main()
