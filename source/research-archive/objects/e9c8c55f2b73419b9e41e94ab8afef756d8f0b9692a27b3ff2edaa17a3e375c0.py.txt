#!/usr/bin/env python3
"""s37-fnorm localization step 1: compare compiled float arithmetic
(parent scalar norm tree vs candidate fragment XOR-chain tree).
CPU-only; reads the two sealed SPVs, no GPU, no writes outside this dir.
"""
import sys
from collections import Counter

sys.path.insert(0, r"@NR_WORKSPACE@/build/zcode-fp16-compute-backend-20261021/scripts")
import u09_build as U
S = U.S

CAND = (r"@NR_WORKSPACE@/work/zcode-fp16-fullnet-handoff-20261024"
        r"/candidates/s37-fnorm/spv")


def analyze(path, label):
    m = S.parse(path)
    types = {}      # id -> ('f16'|'f32'|'v2f16'|'v4f16'|'m8n16f16'|'m8n16f32'|...)
    consts = {}     # id -> value word
    extnames = {}   # import id -> name string
    ops_out = []
    for inst in m:
        nm = inst.name
        ops = inst.ops
        if nm == "OpTypeFloat":
            types[ops[0]] = "f%d" % ops[1]
        elif nm == "OpTypeVector":
            base = types.get(ops[1], "t%d" % ops[1])
            types[ops[0]] = "v%d%s" % (ops[2], base)
        elif nm == "OpTypeMatrix":
            base = types.get(ops[1], "t%d" % ops[1])
            types[ops[0]] = "mat%dx%s" % (ops[2], base)
        elif nm == "OpTypeInt":
            types[ops[0]] = "i%d" % ops[1]
        elif nm == "OpTypeCooperativeMatrixKHR":
            base = types.get(ops[1], "t%d" % ops[1])
            types[ops[0]] = "coop(%s)" % base
        elif nm == "OpConstant":
            consts[ops[0]] = ops[1]
        elif nm == "OpExtInstImport":
            extnames[ops[0]] = S.words_to_str(ops[1:]) if hasattr(S, "words_to_str") else "?"
        elif nm in ("OpFMul", "OpFAdd", "OpFDiv", "OpFSub", "OpFNegate",
                    "OpExtInst", "OpMatrixTimesScalar", "OpVectorTimesScalar",
                    "OpFConvert", "OpBitcast", "OpGroupNonUniformShuffleXor",
                    "OpGroupNonUniformShuffle", "OpGroupNonUniformBroadcast",
                    "OpCompositeConstruct", "OpCompositeExtract", "OpCompositeInsert"):
            ops_out.append((inst.idx, nm, inst.type, ops))
    hist = Counter()
    for idx, nm, rtype, ops in ops_out:
        t = types.get(rtype, "t%s" % rtype)
        if nm == "OpExtInst":
            g = extnames.get(ops[0], "?")
            hist[("OpExtInst:%s/%s" % (g, ops[2]), t)] += 1
        else:
            hist[(nm, t)] += 1
    print("=== %s ===" % label)
    for (nm, t), v in sorted(hist.items()):
        print("  %-34s %-14s x%d" % (nm, t, v))
    print()
    return m, types, consts, extnames, ops_out


pm, pt, pc, pe, po = analyze(CAND + "/parent_repro.spv", "PARENT (scalar norm)")
cm, ct, cc, ce, co = analyze(CAND + "/s37_fnorm.spv", "CANDIDATE (fragment norm)")
