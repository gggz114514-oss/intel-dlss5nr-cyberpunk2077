"""Minimal plan reader + buffer-15/270/277 ownership report for the F2 pass."""
from __future__ import annotations

import hashlib
import json
import sys
from pathlib import Path

B = Path(r"@NR_WORKSPACE@")
PLANS = {
    "fp16_incumbent": B / "work/codex-astra-fp16-int8-audit-20261024/fullnet-continuous-20261027/backend-merged/plan.txt",
    "int8_corrected": B / "work/codex-astra-fp16-int8-audit-20261024/int8-v15-offline-20261028/corrected-20261029/plan.txt",
    "candidate_34189": B / "work/codex-astra-fp16-int8-audit-20261024/int8-quality-first-20261029/real-int8-24/c64-family-static/combo-34189.txt",
}


def parse(path: Path) -> dict:
    import shlex
    bufs, passes, cur = [], [], None
    for line in path.read_text(encoding="utf-8", errors="replace").splitlines():
        t = line.strip()
        if not t:
            continue
        tok = shlex.split(t)
        k = tok[0]
        if k == "BUFFER":
            bufs.append({"index": len(bufs), "name": tok[1], "bytes": int(tok[2]),
                         "file": tok[3] if len(tok) > 3 else ""})
        elif k == "PASS":
            cur = {"ordinal": len(passes), "group": int(tok[1]), "name": tok[2],
                   "spv_path": tok[3], "n": [int(x) for x in tok[4:11]],
                   "bind": [], "push": []}
            passes.append(cur)
        elif k == "BIND" and cur is not None:
            cur["bind"].append({"binding": int(tok[1]), "buffer": int(tok[2]),
                                "offset": int(tok[3]), "bytes": int(tok[4])})
        elif k == "PUSH" and cur is not None:
            cur["push"] = [int(x) & 0xFFFFFFFF for x in tok[1:]]
        elif k == "OUTPUT" and cur is not None:
            cur.setdefault("outputs", []).append(int(tok[1]))
    return {"path": str(path), "sha256": hashlib.sha256(path.read_bytes()).hexdigest(),
            "buffers": bufs, "passes": passes}


if __name__ == "__main__":
    out = {}
    for key, p in PLANS.items():
        pl = parse(p)
        out[key] = {"sha256": pl["sha256"], "n_passes": len(pl["passes"]), "n_buffers": len(pl["buffers"])}
        # F2 pass(es)
        f2 = [ps for ps in pl["passes"] if ps["name"] == "f2c32_block66_fused"]
        out[key]["f2"] = [{"ordinal": ps["ordinal"], "group": ps["group"], "spv": ps["spv_path"],
                           "n": ps["n"], "push": ps["push"],
                           "bind": ps["bind"], "outputs": ps.get("outputs")} for ps in f2]
        # who binds buffer 15 / 270 / 277
        for b in (15, 270, 277):
            hits = []
            for ps in pl["passes"]:
                for bd in ps["bind"]:
                    if bd["buffer"] == b:
                        hits.append({"ordinal": ps["ordinal"], "pass": ps["name"], "binding": bd["binding"],
                                     "bytes": bd["bytes"], "spv": Path(ps["spv_path"]).name})
            out[key][f"buf{b}"] = {"name": pl["buffers"][b]["name"], "bytes": pl["buffers"][b]["bytes"],
                                   "file": pl["buffers"][b]["file"], "bound_by": hits}
    dst = Path(__file__).with_name("EVIDENCE") / "planbits.json"
    dst.write_text(json.dumps(out, indent=1), encoding="utf-8")
    for key, v in out.items():
        print("==", key, v["sha256"][:16], "passes", v["n_passes"], "buffers", v["n_buffers"])

            print("   F2 ord", f["ordinal"], "grp", f["group"], Path(f["spv"]).name, "n", f["n"])
