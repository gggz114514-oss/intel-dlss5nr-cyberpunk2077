"""Print ISA windows by (ordinal, address) with file + line numbers. Read-only."""
import json, os, re, sys

CARD = os.path.dirname(os.path.abspath(__file__))
EVID = os.path.join(os.path.dirname(os.path.dirname(CARD)), "evidence")
IDX = json.load(open(os.path.join(EVID, "NATIVE-INDEX.json"), encoding="utf-8"))
COV = {r["ordinal"]: r for r in IDX["coverage"]}
H2F = json.load(open(os.path.join(CARD, "_hash2asm.json"), encoding="utf-8"))
LINE_RE = re.compile(r"^/\* \[([0-9A-Fa-f]+)\]\s+\*/\s*(.*)$")


def resolve(ordinal):
    for c in (H2F.get(str(ordinal)) or []):
        if os.path.exists(c):
            return c
    return None


def load(path):
    rows = []
    for i, raw in enumerate(open(path, encoding="utf-8", errors="replace"), 1):
        m = LINE_RE.match(raw)
        rows.append((i, m.group(1) if m else None, raw.rstrip("\n")))
    return rows


def window(ordinal, addr, before=6, after=10, grep=None):
    p = resolve(ordinal)
    rows = load(p)
    idx = [k for k, (ln, a, _) in enumerate(rows) if a and a.upper() == addr.upper()]
    print("### ord=%d %s %s" % (ordinal, COV[ordinal]["family"], COV[ordinal]["name"]))
    print("### ISA:", p)
    if not idx:
        print("  address not found")
        return
    for k in idx[:1]:
        for j in range(max(0, k - before), min(len(rows), k + after + 1)):
            ln, a, txt = rows[j]
            mark = ">>" if j == k else "  "
            print("%s L%-6d [%s] %s" % (mark, ln, a or "----", txt.strip()))


if __name__ == "__main__":
    # argv: ordinal addr [before] [after]
    o = int(sys.argv[1]); a = sys.argv[2]
    b = int(sys.argv[3]) if len(sys.argv) > 3 else 6
    af = int(sys.argv[4]) if len(sys.argv) > 4 else 10
    window(o, a, b, af)
