"""Compare the refreshed stall profile (current head, after n27+n29) with the earlier one."""
import json, sys
from pathlib import Path
sys.path.insert(0, str(Path(__file__).parent))
import COMMON as K


def load(name):
    p = list((K.C / 'queue' / name).rglob('hardware-counters.json'))[0]
    d = K.read_json(p)
    names = [m['name'] for m in d['metrics']]
    idx = {n: i for i, n in enumerate(names)}
    ps = d['frames'][0]['passes']
    gpu = [x['values'][idx['GpuTime']] for x in ps]
    tot = sum(gpu) / 1e6

    def w(n):
        return sum(x['values'][idx[n]] * g for x, g in zip(ps, gpu)) / sum(gpu)

    return dict(tot=tot, w=w, ps=ps, idx=idx, gpu=gpu)


a = load('counters-xmx2')
b = load('counters-head2')
print('frame sum: before %.3f ms -> now %.3f ms  (delta %+.3f)' % (a['tot'], b['tot'], b['tot'] - a['tot']))
for n in ('XVE_ACTIVE', 'XVE_STALL_ALUWR', 'XVE_STALL_SENDWR', 'XVE_STALL_BARRIER',
          'XVE_STALL_PIPESTALL', 'XVE_STALL_PS_DEPENDENCY'):
    va, vb = a['w'](n), b['w'](n)
    print('  %-24s %6.2f%% -> %6.2f%%  (%+.2f pts)   ms %.3f -> %.3f  (%+.3f)'
          % (n, va, vb, vb - va, va * a['tot'] / 100, vb * b['tot'] / 100,
             vb * b['tot'] / 100 - va * a['tot'] / 100))
sa = sum(x['values'][a['idx']['XVE_INST_EXECUTED_XMX_FP16']] for x in a['ps'])
sb = sum(x['values'][b['idx']['XVE_INST_EXECUTED_XMX_FP16']] for x in b['ps'])
print('  XMX FP16 slots: %.4e -> %.4e' % (sa, sb))
for name in ('f1c32_image_pre', 's37_fold_body_r1a_v1'):
    for tag, z in (('before', a), ('now', b)):
        hit = [x for x in z['ps'] if x['name'] == name]
        if hit:
            g = hit[0]['values'][z['idx']['GpuTime']] / 1e6
            print('  %-26s %-6s gpu %.3f ms  ALUWR %5.2f%% (%.3f ms)  BAR %5.2f%%' % (
                name, tag, g, hit[0]['values'][z['idx']['XVE_STALL_ALUWR']],
                g * hit[0]['values'][z['idx']['XVE_STALL_ALUWR']] / 100,
                hit[0]['values'][z['idx']['XVE_STALL_BARRIER']]))
K.write_json(K.C / 'evidence' / 'STALL-BUDGET-HEAD2.json',
             dict(frame_ms=b['tot'], active=b['w']('XVE_ACTIVE'),
                  aluwr_ms=b['w']('XVE_STALL_ALUWR') * b['tot'] / 100,
                  sendwr_ms=b['w']('XVE_STALL_SENDWR') * b['tot'] / 100,
                  barrier_ms=b['w']('XVE_STALL_BARRIER') * b['tot'] / 100,
                  pipestall_ms=b['w']('XVE_STALL_PIPESTALL') * b['tot'] / 100))
