import sys
from pathlib import Path
import numpy as np

sys.path.insert(0, str(Path(__file__).parent))
import c32blk as M
from validate import bk, PUSH, ptr

B = Path(r"@NR_WORKSPACE@")
W = B / "work/dsh-fullnet-static-20261029"
src = W / "C00-color-baseline/runs/diag65-69"
tea = B / "work/codex-astra-fp16-int8-audit-20261024/int8-quality-first-20261029/real-int8-24/dump-fp16"

x, valid = M.load_block(src / "c00-block66_out.bin", -1, -1)
blk = M.Block(bk("half", 67), bk("i8", 67), bk("sc", 67), bk("aux", 67), ptr(PUSH))
y, _ = blk.forward(x, valid)
idx, vv = M.token_index(-1, -1)
gpu = np.fromfile(src / "c00-block67_out.bin", dtype="<f2")[idx]
tch = np.fromfile(tea / "dump_block67_out.bin", dtype="<f2")[idx]
m = vv
pred = y[m].astype(np.float64)
g = gpu[m].astype(np.float64)
t = tch[m].astype(np.float64)
print("pred vs gpu ", M.rel_l2(pred, g), M.corr(pred, g))
print("pred vs tea ", M.rel_l2(pred, t), M.corr(pred, t))
print("gpu  vs tea ", M.rel_l2(g, t), M.corr(g, t))
print("rms pred/gpu/tea", np.sqrt((pred**2).mean()), np.sqrt((g**2).mean()), np.sqrt((t**2).mean()))
# per-channel
for c in range(32):
    r = M.rel_l2(pred[:, c], g[:, c])
    cc = M.corr(pred[:, c], g[:, c])
    if c < 8 or r > 0.7:
        print(f"ch {c:2d} relL2 {r:7.4f} corr {cc:6.3f} rms pred/gpu {np.sqrt((pred[:,c]**2).mean()):8.2f} {np.sqrt((g[:,c]**2).mean()):8.2f}")
# best fit scale
a = float((pred * g).sum() / (pred * pred).sum())
print("best scale pred->gpu", a)
# position structure: relL2 for slot groups
for slot in range(16):
    sel = np.zeros(m.sum(), bool)
print("per-slot (4x4) relL2:")
tok_idx = np.where(m)[0]
slots = ((tok_idx % 64) % 16)
for s in range(16):
    sel = slots == s
    print(f"  slot {s:2d} n={sel.sum():6d} relL2 {M.rel_l2(pred[sel], g[sel]):7.4f}")
