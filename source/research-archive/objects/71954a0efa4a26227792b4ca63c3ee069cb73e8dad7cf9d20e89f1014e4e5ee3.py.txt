#!/usr/bin/env python3
"""Locate the vit ws[] header inside the two block31 packs and compare it.

The ws[] table is not at byte 0 of the pack (byte 0 is int8 weight codes), so this scans
for the value X5 reported for the fixed pack (ws[8] = 8.642212, ws[3] = 0.115711) and then
prints the same 16-float window from the G3-bound pack. That is the only honest way to
confirm or refute the claim without knowing the header offset a priori.
"""
import hashlib
import sys
from pathlib import Path

import numpy as np

B = Path("@NR_WORKSPACE@")
G3 = B / ("candidates/codex-sol-int8-long-008-20261019/vit21-blocks31-38-qkv-v2"
          "/pack/vit_block31_weights.int8.bin")
V1 = B / ("candidates/codex-sol-int8-long-008-20261019/vit21-blocks31-38-v1"
          "/pack/vit_block31_weights.int8.bin")
TARGET = np.float32(8.642212)
DENORM = 1.1754944e-38


def scan(path, limit=2 << 20):
    raw = np.fromfile(path, dtype="<u1", count=limit)
    f = raw[: (raw.size // 4) * 4].view("<f4")
    hits = np.nonzero(np.abs(f - TARGET) < np.float32(1e-4))[0]
    return f, hits


def main():
    f1, h1 = scan(V1)
    print(f"v1 pack {V1.name} sha {hashlib.sha256(V1.read_bytes()).hexdigest()[:16]} "
          f"hits for ws[8]=8.642212 in first 2 MiB: {h1[:5].tolist()}")
    if not h1.size:
        print("could not locate the table; claim left UNVERIFIED")
        return 2
    base = int(h1[0]) - 8                      # window start = hit - 8 floats
    g3 = np.fromfile(G3, dtype="<f4", count=base + 16)
    print(f"\nwindow at f32 index {base} (byte {base * 4})")
    for label, f in (("X5 v1 (fixed)", f1), ("G3-bound", g3)):
        w = f[base:base + 16]
        den = int(np.sum((np.abs(w) < DENORM) & (w != 0)))
        print(f"{label:14s} ws[0:16] = {np.array2string(w, precision=6)}")
        print(f"{'':14s} denormals {den}  ws[3]={w[3]:.6g} ws[8]={w[8]:.6g} "
              f"ws[10]={w[10]:.6g}  1/ws[8]={float(1.0 / w[8]) if w[8] else float('nan'):.6g}  "
              f"ws[8]*ws[3]={float(w[8]) * float(w[3]):.6g}")
    return 0


if __name__ == "__main__":
    sys.exit(main())
