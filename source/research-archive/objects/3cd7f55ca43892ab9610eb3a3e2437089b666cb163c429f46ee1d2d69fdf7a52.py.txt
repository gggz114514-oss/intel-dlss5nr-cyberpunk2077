import re,json
from pathlib import Path
from review_native import inst,parents,D
result={}
for arm,base in parents.items():
 rows=inst(next(base.glob('*38_orig.asm')))
 ids=[i for i,(_,x) in enumerate(rows) if 'dpas.' in x]
 ids=ids[16:32 if arm=='parent' else 48]
 pairs=[]
 for pos in range(0,len(ids),2):
  first,second=ids[pos:pos+2];ops=[]
  for ix in [first,second]:
   pc,line=rows[ix];regs=re.findall(r'r(\d+)(?:\.\d+)?:',line)
   assert len(regs)==4,(pc,line)
   b=int(regs[2]);a=int(regs[3]);send=None
   for k in range(ix-1,-1,-1):
    m=re.search(r'send\.ugm.*?\)\s+r(\d+)\s',rows[k][1])
    if m and int(m[1])==b and 'rd:8;' in rows[k][1]:
     send=k;break
   assert send is not None,(pc,line)
   ops.append(dict(dpas_pc=pc,C=int(regs[1]),B_first=b,B_last=b+7,A=a,send_pc=rows[send][0],send_index=send,dpas_index=ix))
  a,b=ops
  pairs.append(dict(pair=pos//2,ops=ops,same_A=a['A']==b['A'],both_B_issued_before_first_DPAS=b['send_index']<first,
   return_register_overlap=list(range(max(a['B_first'],b['B_first']),min(a['B_last'],b['B_last'])+1))))
 result[arm]=pairs
(D/'WEIGHT-OVERLAP.json').write_text(json.dumps(result,indent=2),encoding='utf-8')
for arm,pairs in result.items():
 print(arm,'pairs',len(pairs),'same_A',sum(p['same_A'] for p in pairs),'both-issued-before',sum(p['both_B_issued_before_first_DPAS'] for p in pairs),'overlap',sum(bool(p['return_register_overlap']) for p in pairs))
 print([(p['ops'][0]['dpas_pc'],p['ops'][1]['dpas_pc'],p['both_B_issued_before_first_DPAS'],p['return_register_overlap']) for p in pairs])
