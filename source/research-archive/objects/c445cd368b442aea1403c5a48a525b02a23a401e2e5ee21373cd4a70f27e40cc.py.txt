"""n33 task B6: finish the candidate SPV's NoContraction set (CPU only).

The campaign recipe copies the parent's 185 NoContraction sites onto a fresh
compile by in-order subsequence matching of each function.  This candidate is a
statement reorder, so `main` is a block permutation and the in-order match fails.

What the recipe's 185 actually consist of (measured in _b4/_b5):
    parent raw (glslang, `precise` only)  183 sites = 88 OpFMul + 86 OpFAdd
                                          + 8 OpVectorTimesScalar + 1 OpDot
    + 2 sites the historical decoration parent added on the exp helper's
      FrexpStruct (function 9, `nr_exp_h2`), which this card does not touch.

So the candidate gets the same 185 by construction:
    its own 183 `precise` sites (identical opcode histogram: the reorder does not
    add or remove an FP expression)  +  the 2 FrexpStruct sites of the exp helper,
    transferred by *function identity* (nr_exp_h2 is byte-identical between parent
    and candidate; asserted below).

Evidence: build/nc-final.json.
"""
import sys, json
from pathlib import Path
from collections import Counter

HERE = Path(__file__).parent
CAMP = HERE.parent.parent
sys.path.insert(0, str(CAMP))
import COMMON as K  # noqa: E402

ST = K.ST
BUILD = HERE / "build"
PARENT_SPV = Path("@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/"
                  "c32-rest-20261026/body-pv-den-overlap/candidate.spv")
GLSL = {37: "FMin", 40: "FMax", 43: "FClamp", 45: "SClamp", 31: "Sqrt", 32: "InverseSqrt",
        4: "FAbs", 48: "Fma", 50: "FrexpStruct"}


def sha(p):
    import hashlib
    return hashlib.sha256(Path(p).read_bytes()).hexdigest()


def nc_hist(m):
    dec = set(ST.no_contraction_ids(m))
    c = Counter()
    for op, ops, *_ in m.instructions:
        rid = ST.result_id_of(op, ops)
        if rid is not None and rid in dec:
            c[op] += 1
    return c


def skel(m, region):
    return [(m.instructions[i][0], tuple(m.instructions[i][1])) for i in region]


def main():
    pm = ST.Module(PARENT_SPV)
    cm = ST.Module(BUILD / "candidate-raw.spv")
    pr, cr = ST.function_regions(pm), ST.function_regions(cm)
    assert len(pr) == len(cr) == 10, (len(pr), len(cr))

    p_dec = set(ST.no_contraction_ids(pm))
    c_dec = set(ST.no_contraction_ids(cm))
    print("parent NC %d   candidate raw NC %d" % (len(p_dec), len(c_dec)))
    p_hist, c_hist = nc_hist(pm), nc_hist(cm)
    print("parent    histogram", dict(p_hist))
    print("candidate histogram", dict(c_hist))
    # the only difference may be the OpExtInst sites the historical parent added
    expect = Counter(p_hist)
    expect[12] -= 2
    if not expect[12]:
        del expect[12]
    assert expect == c_hist, (dict(expect), dict(c_hist),
                              "the reorder changed the set of decorated FP ops")

    # the exp helper: same function, byte-identical skeleton
    same = [fi for fi, (a, b) in enumerate(zip(pr, cr)) if skel(pm, a) == skel(cm, b)]
    print("functions with an identical instruction skeleton: %s" % same)

    # decorated FrexpStruct sites of the parent, and their function
    extra = []
    for idx, (op, ops, *_) in enumerate(pm.instructions):
        rid = ST.result_id_of(op, ops)
        if rid is not None and rid in p_dec and rid not in c_dec and op == 12:
            fi = [f for f, r in enumerate(pr) if idx in r][0]
            extra.append((fi, pr[fi].index(idx), ops))
    print("parent sites decorated on OpExtInst (not in the candidate's own set): %s"
          % [(fi, pos, GLSL.get(o[3])) for fi, pos, o in extra])
    assert len(extra) == 2
    for fi, pos, _ in extra:
        assert fi in same, "exp helper is not byte-identical, cannot transfer by function identity"

    keep = set(c_dec)
    for fi, pos, _ in extra:
        idx = cr[fi][pos]
        rid = ST.result_id_of(cm.instructions[idx][0], cm.instructions[idx][1])
        assert rid is not None
        keep.add(rid)
    print("keep set: %d ids (%d own + %d exp-helper)" % (len(keep), len(c_dec), len(extra)))
    assert len(keep) == len(p_dec) == 185

    out = HERE / "candidate.spv"
    ST.set_no_contraction(cm, keep, out)
    nc2 = ST.no_contraction_ids(ST.Module(out))
    K.validate_spv(out)
    print("candidate.spv %d B  NC %d  sha %s  validate OK"
          % (out.stat().st_size, len(nc2), sha(out)[:16]))

    rec = dict(parent_spv=str(PARENT_SPV), parent_spv_sha=sha(PARENT_SPV),
               parent_bytes=PARENT_SPV.stat().st_size, parent_nc=len(p_dec),
               candidate_raw=str(BUILD / "candidate-raw.spv"),
               candidate_raw_sha=sha(BUILD / "candidate-raw.spv"),
               candidate_raw_nc=len(c_dec),
               candidate_spv=str(out), candidate_spv_sha=sha(out),
               candidate_bytes=out.stat().st_size, candidate_nc=len(nc2),
               nc_histogram={str(k): v for k, v in nc_hist(pm).items()},
               identical_skeleton_functions=same,
               exp_helper_sites_transferred=[dict(function=fi, position=pos,
                                                  glsl450=GLSL.get(o[3]))
                                             for fi, pos, o in extra],
               method=("own `precise` sites (183, identical opcode histogram to the parent's raw "
                       "compile) + 2 exp-helper FrexpStruct sites transferred by function identity; "
                       "the campaign's in-order pattern transfer cannot place a block-permuted main"))
    (BUILD / "nc-final.json").write_text(json.dumps(rec, indent=1), encoding="utf-8")
    print("wrote", BUILD / "nc-final.json")


if __name__ == "__main__":
    main()
