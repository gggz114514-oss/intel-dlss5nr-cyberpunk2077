"""n06d delivery self-check: every deliverable present, hashes consistent, and the
read-only world untouched (baseline plan, chain head, COMMON.py, other cards).

Run: <python> scratch/verify_delivery.py
"""
import hashlib
import json
import sys
from pathlib import Path

CARD = Path(__file__).resolve().parent.parent
CAMP = CARD.parent.parent
sys.path.insert(0, str(CAMP))
import COMMON as K

REQUIRED = ['CARD.txt', 'ANALYSIS.md', 'SOURCE-DIFF.patch', 'PROOF.json', 'COST-LEDGER.json',
            'NATIVE-PREDICTION.md', 'BUILD.json', 'candidate.comp', 'candidate.spv',
            'parent-rebuild.spv', 'plan.txt', 'GPU-REQUEST.json', 'RESUME.txt']
CURRENT_SHA = '77d24b8389ab113ca7cfdb31f2cc8f6d79b0308775affa4a5c664aaf67d58745'
COMMON_SHA = '0438138ccae004ab9fc0da8b9cd771f9f76b98327d5007b8c0b25ed9b1185596'


def sha(p):
    return hashlib.sha256(Path(p).read_bytes()).hexdigest()


def main():
    ok, fail = [], []

    def check(name, cond, detail=''):
        (ok if cond else fail).append('%s %s' % (name, detail))

    for n in REQUIRED:
        check('exists:' + n, (CARD / n).exists())

    b = K.read_json(CARD / 'BUILD.json')
    pr = K.read_json(CARD / 'PROOF.json')
    cl = K.read_json(CARD / 'COST-LEDGER.json')
    g = K.read_json(CARD / 'GPU-REQUEST.json')

    check('parent-rebuild == baseline bytes',
          sha(CARD / 'parent-rebuild.spv') == b['parent']['sha256'])
    check('candidate.spv sha', sha(CARD / 'candidate.spv') == b['candidate']['sha256'])
    check('candidate.comp sha', sha(CARD / 'candidate.comp') == b['candidate']['source_sha256'])
    check('candidate source sha in PROOF',
          pr['identity']['candidate_source']['sha256'] == b['candidate']['source_sha256'])
    check('candidate spv sha in PROOF',
          pr['identity']['candidate_spv']['sha256'] == b['candidate']['sha256'])
    check('plan sha', sha(CARD / 'plan.txt') == b['plan']['sha256'] == g['plan']['sha256'])
    check('reciprocal diff sha', sha(CARD / 'SOURCE-DIFF.patch') == cl['diff']['sha256'])
    check('15 PASS replaced', b['plan']['replaced_passes'] == 15 == b['plan']['diff_lines'])
    check('cpu-check ok', b['cpu_check'].startswith('CPU_PLAN_OK'))
    check('cpu-check == chain head', b['cpu_check_identical_to_chain_head'] is True)
    check('validate rc0', b['candidate']['validate_returncode'] == 0)
    check('float_NC_checked present', 'float_NC_checked' in b['spirv'])
    check('no_contraction 0/0', b['spirv']['parent']['no_contraction'] == 0 ==
          b['spirv']['candidate']['no_contraction'])
    check('interface all equal', b['spirv']['interface_all_equal'] is True)
    check('gpu-request keys',
          set(['card', 'shader_group', 'plan', 'arms', 'gates', 'native', 'resources',
               'ordinary', 'hypothesis', 'native_observables']) <= set(g.keys()))
    check('gpu-request 15 ordinals', len(g['shader_group']['pass_ordinals']) == 15)
    check('gpu-request baseline sha', g['shader_group']['spv_sha256_baseline'] ==
          [b['parent']['sha256']])
    check('proof publish mismatch 0',
          pr['publish']['elementwise']['value_mismatches'] == 0 and
          pr['publish']['elementwise']['write_address_mismatches'] == 0)
    check('proof read mismatch 0', pr['read']['value_mismatches'] == 0 and
          pr['read']['tuples'] == 131072)
    check('proof hazards zero', pr['domains']['hazards']['WAW']['count'] == 0 and
          pr['domains']['hazards']['WAR']['count'] == 0 and
          pr['domains']['hazards']['RAW']['count'] == 1)
    check('proof workgroup memory 32768 both',
          pr['spirv']['workgroup_memory']['parent']['bytes'] == 32768 ==
          pr['spirv']['workgroup_memory']['candidate']['bytes'])
    check('proof A stride 256->16', [r['stride'] for r in
          pr['spirv']['coop_matrix_loads']['parent']] == [256, 256, 256, 16, 16] and
          [r['stride'] for r in pr['spirv']['coop_matrix_loads']['candidate']] ==
          [256, 256, 16, 16, 16])
    check('proof bytes equal', pr['bytes']['per_sg']['A_read_bytes_equal'] is True)

    # read-only world
    check('baseline plan untouched', K.sha(K.BASELINE) == K.BASELINE_SHA)
    head, head_sha, _ = K.current_best()
    check('chain head untouched', head_sha == b['chain_head']['sha256'] and sha(head) == head_sha)
    check('CURRENT.json untouched', sha(CAMP / 'CURRENT.json') == CURRENT_SHA)
    check('COMMON.py untouched', sha(CAMP / 'COMMON.py') == COMMON_SHA)

    # this card must not have created anything outside its own directory
    check('no plan in queue dir for this card',
          not (CAMP / 'queue' / CARD.name / 'plan.txt').exists(),
          '(the plan lives inside the card dir only)')

    print('OK   %d checks' % len(ok))
    for o in ok:
        print('  ok   ' + o)
    if fail:
        print('FAIL %d checks' % len(fail))
        for f in fail:
            print('  FAIL ' + f)
        raise SystemExit(1)
    print('\nDELIVERY OK: %d files checked, plan replaced 15 PASS, cpu-check identical to the '
          'chain head, 0 publish / 0 read mismatches' % len(REQUIRED))


if __name__ == '__main__':
    main()
