"""n33 task C2: per-group consumer lists + rewrite table for REDUNDANCY-PROOF.md.

For every one of the 24 expensive ops of kernel 7a38a17947f21ae6:
  * source operand + its reaching definition (the instruction that produced the
    value the op consumes)
  * the live consumers of the result, bounded by the next write of that dword
    (after that write the register is recycled and later readers are not
    consumers of this value)
  * for every pair inside a census group (same operand text): every write of the
    source dword strictly between the two sites.
Also emits the verdict table used in REDUNDANCY-PROOF.md.
"""
import sys, json
from pathlib import Path
from collections import defaultdict

HERE = Path(__file__).parent
sys.path.insert(0, str(HERE))
import _isa  # noqa: E402
from _t02_live import reachable  # noqa: E402
from _t01_identity import classify, reg_base, reaching  # noqa: E402

OUT = HERE / "C-CONSUMERS.json"


def build():
    ins, labels = _isa.parse()
    seen = reachable(ins, labels)
    for i, it in enumerate(ins):
        it["live"] = seen[i]
        it["w"], it["r"] = classify(it)
    by_dw = defaultdict(list)
    for it in ins:
        for f, no, dw, n in it["w"]:
            for d in range(dw, dw + n):
                by_dw[(f, no, d)].append(it["idx"])
    for v in by_dw.values():
        v.sort()
    return ins, by_dw


def main():
    ins, by_dw = build()
    n = len(ins)
    live = [i for i in range(n) if ins[i]["live"]]

    exp = [k for k in live if ins[k]["op"] in ("math.rsqt", "math.inv")]
    rows = []
    for k in exp:
        key = reg_base(ins[k]["ops"][0])
        prod = reaching(by_dw, key, k)
        # consumers of the result dword until it is rewritten
        nxt_write = None
        for w in by_dw.get(key, []):
            if w > k:
                nxt_write = w
                break
        cons = []
        for j in live:
            if j <= k or (nxt_write is not None and j >= nxt_write):
                continue
            for f, no, dw, cnt in ins[j]["r"]:
                if any((f, no, d) == tuple(key) for d in range(dw, dw + cnt)):
                    cons.append(j)
                    break
        rows.append(dict(idx=k, addr=hex(ins[k]["addr"]), op=ins[k]["op"],
                         src_text=ins[k]["ops"][0], src_key=list(key),
                         producer=prod, producer_addr=hex(ins[prod]["addr"]) if prod else None,
                         producer_text=ins[prod]["text"][:72] if prod else None,
                         next_write=nxt_write,
                         next_write_addr=hex(ins[nxt_write]["addr"]) if nxt_write else None,
                         next_write_text=ins[nxt_write]["text"][:72] if nxt_write else None,
                         consumers=[dict(idx=c, addr=hex(ins[c]["addr"]), op=ins[c]["op"],
                                         text=ins[c]["text"][:64]) for c in cons],
                         n_consumers=len(cons)))

    # ---- group by operand text (census criterion)
    groups = defaultdict(list)
    for r in rows:
        groups[r["src_text"]].append(r)

    table = []
    for txt, rs in groups.items():
        rs.sort(key=lambda r: r["idx"])
        op = rs[0]["op"]
        entry = dict(op=op, src_text=txt, sites=[r["idx"] for r in rs],
                     producers=[r["producer"] for r in rs],
                     n_sites=len(rs), n_distinct_producers=len({r["producer"] for r in rs}))
        pairs = []
        for i in range(len(rs)):
            for j in range(i + 1, len(rs)):
                a, b = rs[i], rs[j]
                key = tuple(a["src_key"])
                w = [x for x in by_dw.get(key, []) if a["idx"] < x < b["idx"]]
                pairs.append(dict(a=a["idx"], b=b["idx"], n_writes=len(w),
                                  writes=w[:6],
                                  same_producer=(a["producer"] == b["producer"])))
        entry["pairs"] = pairs
        entry["min_writes_between"] = min((p["n_writes"] for p in pairs), default=None)
        entry["shared_definition_pairs"] = sum(1 for p in pairs if p["same_producer"])
        table.append(entry)

    table.sort(key=lambda e: -e["n_sites"])
    print("%-9s %-22s %5s %9s %6s %9s" % ("op", "src text", "sites", "distinct", "minW", "shared"))
    for e in table:
        print("%-9s %-22s %5d %9d %6d %9d"
              % (e["op"], e["src_text"][:22], e["n_sites"], e["n_distinct_producers"],
                 e["min_writes_between"], e["shared_definition_pairs"]))

    print("\n== per site: producer / next write / consumers ==")
    for r in rows:
        print("%5d %-9s %-18s prod %5s  nextW %5s ncons %d : %s"
              % (r["idx"], r["op"], r["src_text"][:18], r["producer"], r["next_write"],
                 r["n_consumers"],
                 ",".join("%d%s" % (c["idx"], "(" + c["op"] + ")") for c in r["consumers"][:6])))

    json.dump(dict(sites=rows, groups=table), open(OUT, "w"), indent=1)
    print("\nwrote", OUT)


if __name__ == "__main__":
    main()
