"""Record L34 (verification gates are blind to address literals; plans do not pin SPV bytes),
refine L25 (fewer chains is not automatically bad), and check the 16 KB declaration."""
import json, struct, sys
from pathlib import Path
sys.path.insert(0, str(Path(__file__).parent))
import COMMON as K

# 0. verify the module really declares 16384 B of workgroup memory
spv = K.C / 'cards' / 'n36-f05-arena-shrink' / 'candidate-c256.spv'
d = spv.read_bytes()
w = struct.unpack('<%dI' % (len(d) // 4), d[:len(d) // 4 * 4])
declares = 8192 in w          # 8192 halves * 2 B = 16384 B
print('F05 C256 candidate declares qkv[8192] (=16384 B):', declares)

p = K.C / 'evidence' / 'LESSONS.json'
L = K.read_json(p)
ids = [x['id'] for x in L['append_after_L7']]

if 'L34' not in ids:
    L['append_after_L7'].append({
        'id': 'L34',
        'claim': 'The standard verification gates are blind to a wrong address literal, and a plan does not pin the candidate bytes it dispatches',
        'evidence': [
            'the first n36 candidate passed storage-byte accounting, spv_validate, parent-byte reproduction, the coopmat and NoContraction multisets and the hazard model, yet failed the fixed gold frame: one plane-base literal +4096u inside the V section had not been updated when the section stride changed 6144 -> 4096, so the PV read went out of the section',
            'the hazard and offset models used the intended offsets, not the literal in the source text, which is why they were all silent',
            'the fix (16 sites, 4096u -> 0u) left the plan file unchanged, because the plan stores SPV paths and not their hashes: the plan sha stayed 21292385 while the candidate sha changed to 236aad96, so dispatch identity must be checked by SPV sha',
            'the dedicated gate that caught it strips the section base and checks each access literal against the allowed set and the section bounds (_addr_audit.py in the n36 card)'
        ],
        'consequence': 'Any card that changes an arena stride or plane layout must run an address-literal audit in addition to the usual gates, and must record the SPV hashes it dispatches rather than the plan hash.'
    })
    print('L34 added')

for x in L['append_after_L7']:
    if x['id'] == 'L25':
        x['claim'] = ('Instruction-level parallelism is worth more than instruction count when it is lost at the same time as '
                      'something else is gained; a drop in independent chains is a cost to weigh, not a veto')
        x['evidence'].append('n36 tier2 cut the QKV-section independent chains 6 -> 4 yet still measured +0.126122 ms, '
                             'because the occupancy gain (2 -> 4 resident workgroups) dominated; so the earlier wording '
                             '("never collapse independent chains") was too strong')
        x['consequence'] = ('Keep the rule "report chains before/after and treat a collapse as a cost", but do not refuse '
                            'a card on chain count alone: n28 lost because it collapsed 4 chains to 1 with nothing gained, '
                            'while n36 tier2 collapsed 6 to 4 and won 0.126 ms on occupancy.')
        print('L25 refined')
K.write_json(p, L)
K.write_json(K.C / 'evidence' / 'ARENA-16384-CHECK.json', dict(
    candidate=K.sha(spv), declares_8192_halves=declares,
    note='SPIR-V array length confirms 16384 B of workgroup memory; the ISA-side Workgroup Memory attribute '
         'was not printed by this IGC build, so the module declaration is the available evidence'))
print('recorded')
