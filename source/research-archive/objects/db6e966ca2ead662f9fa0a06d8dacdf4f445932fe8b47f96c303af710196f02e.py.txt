def sha(p: Path) -> str:
    h = hashlib.sha256()
    with open(p, "rb") as f:
        for c in iter(lambda: f.read(1 << 20), b""):
            h.update(c)
    return h.hexdigest()


def main():
    rc = D / "F2-ROOT-CAUSE.txt"
    txt = rc.read_text(encoding="utf-8")
    if "ADDENDUM 2026-10-29 22:45" not in txt:
        rc.write_text(txt + ADD, encoding="utf-8")
    lines = []
    for x in sorted(D.glob("*")):
        if x.is_file() and x.name != "SHA256SUMS.txt":
            lines.append(f"{sha(x)}  {x.name}")
    for sub in ("SOURCE", "EVIDENCE", "build"):
        for x in sorted((D / sub).glob("*")):
            if x.is_file():
                lines.append(f"{sha(x)}  {sub}/{x.name}")
    (D / "SHA256SUMS.txt").write_text("\n".join(lines) + "\n", encoding="utf-8")
    print("files:", len(lines))
    for l in lines:
        if any(k in l for k in ("teacher-block65", "FOLLOWUP", "motion-frames", "STATUS", "ROOT-CAUSE")):
            print(" ", l)


if __name__ == "__main__":
    main()
