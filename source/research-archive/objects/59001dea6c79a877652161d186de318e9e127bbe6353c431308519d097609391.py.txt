"""Identify the SPIR-V call graph around the entry gather helpers."""
import sys, collections, json
from pathlib import Path

CARD = Path(__file__).resolve().parents[1]
sys.path.insert(0, str(CARD.parents[1]))
import COMMON as K
import spv_tools_local as ST

KEY = {57: 'call', 61: 'load', 132: 'imul', 128: 'iadd', 139: 'smod', 138: 'srem',
       137: 'umod', 134: 'udiv', 169: 'select', 175: 'sgt', 179: 'slt', 128: 'iadd',
       164: 'land', 62: 'store', 65: 'accesschain', 71: 'extract', 77: 'shuffle',
       79: 'construct', 248: 'bitfield?', 124: 'fma?', 5: 'name'}


def analyse(path):
    m = ST.Module(Path(path))
    reg = ST.function_regions(m)
    out = []
    for r in reg:
        c = collections.Counter()
        callees = collections.Counter()
        for i in r:
            op, ops = m.instructions[i][0], m.instructions[i][1]
            c[op] += 1
            if op == 57 and isinstance(ops, list):
                callees[ops[-1] if len(ops) > 1 else ops[0]] += 1
        out.append(dict(start=r[0], size=len(r), fid=m.instructions[r[0]][1][0],
                        calls=dict(callees), n_call=sum(callees.values()),
                        n_smod=c.get(139, 0), n_srem=c.get(138, 0), n_load=c.get(61, 0),
                        n_imul=c.get(132, 0), n_iadd=c.get(128, 0), n_sel=c.get(169, 0),
                        n_store=c.get(62, 0), n_ac=c.get(65, 0)))
    return out


def main():
    res = {}
    for name in ('parent-rebuild.spv', 'candidate.spv'):
        res[name] = analyse(CARD / name)
    K.write_json(CARD / 'tools' / 'spv-callgraph.json', res)
    for name, funcs in res.items():
        print('==', name)
        for f in funcs:
            print('  fid%-4s start%-5d size%-5d calls%-2d %s smod=%d srem=%d imul=%d iadd=%d sel=%d ac=%d load=%d' % (
                f['fid'], f['start'], f['size'], f['n_call'],
                'callees=' + str(f['calls'])[:60], f['n_smod'], f['n_srem'],
                f['n_imul'], f['n_iadd'], f['n_sel'], f['n_ac'], f['n_load']))


if __name__ == '__main__':
    main()
