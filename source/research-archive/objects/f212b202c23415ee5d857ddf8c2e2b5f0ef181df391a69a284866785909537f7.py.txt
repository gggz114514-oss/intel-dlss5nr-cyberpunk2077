#!/usr/bin/env python3
"""Rebuild the C2 candidate on the *current* G3 baseline.

Why: the first C2 plan was derived from the diagnostic plan
`C/c32-i8-debug-eact-v1/plans/plan-c32-i8-diag-b1-eact.txt`, which binds the OLD
scale pack `C/pack/c32_b1_scales.f32.bin` (sha f3053c8c..., sS*invS = 127.000002).
Current G3 binds the corrected pack `C/c32-scale-from-x-v2/pack/c32_b1_scales.f32.bin`
(sha ecdcea95..., sS*invS = 1.00000002).  A candidate measured on the old pack cannot
stand in for the current G3 arm.

This script only rewrites, for the seven `*_swin_body_1` passes that bind
`c32_body1_i8.spv`, the SPV path and the declared shared size.  Everything else -
BUFFER/OUTPUT/HISTORY/POLICY/BIND/PUSH and every other PASS - is byte-identical to G3.
"""
import hashlib
import json
import sys
from pathlib import Path

ROOT = Path(r"@NR_WORKSPACE@")
W = ROOT / "work/dsh-xmx-handoff-validation-20261024"
CARD = W / "candidates/c2-c32-regk32"
G3 = ROOT / "candidates/int8-coverage-v2-20261020/plan-g3-production.txt"
CAND_SPV = CARD / "spv/c2_body1_regk32.spv"
NEW_SHARED = 10240
OLD_SPV_NAME = "c32_body1_i8.spv"


def sha(p):
    return hashlib.sha256(Path(p).read_bytes()).hexdigest()


def main():
    lines = G3.read_text(encoding="utf-8").splitlines()
    out, changed, i = [], [], 1
    out.append(lines[0])
    ordinal = 0
    while i < len(lines):
        l = lines[i]
        if l.startswith("PASS "):
            f = l.split()
            nbind = int(f[6])
            block = lines[i:i + nbind + 2]
            if Path(f[3].strip('"')).name == OLD_SPV_NAME:
                old_shared = f[5]
                f[3] = f'"{CAND_SPV.as_posix()}"'
                f[5] = str(NEW_SHARED)
                new_pass_line = " ".join(f)
                changed.append({"ordinal": ordinal, "name": f[2].strip('"'),
                                "spv_old": OLD_SPV_NAME, "spv_new": CAND_SPV.name,
                                "shared_old": int(old_shared), "shared_new": NEW_SHARED})
                block[0] = new_pass_line
            out.extend(block)
            i += nbind + 2
            ordinal += 1
            continue
        out.append(l)
        i += 1
    dst = CARD / "plan-c2-regk32-g3.txt"
    dst.write_text("\n".join(out) + "\n", encoding="utf-8", newline="\n")
    rep = {"source_plan": str(G3), "source_sha256": sha(G3),
           "candidate_plan": str(dst), "candidate_plan_sha256": sha(dst),
           "passes": ordinal, "changed_passes": len(changed),
           "spv": str(CAND_SPV), "spv_sha256": sha(CAND_SPV),
           "shared_new": NEW_SHARED, "changes": changed,
           "baseline_note": "G3 scale pack C/c32-scale-from-x-v2/pack/c32_b1_scales.f32.bin "
                            "sha ecdcea9556212561..., sS*invS = 1.00000002 (corrected); the "
                            "diagnostic pack C/pack/c32_b1_scales.f32.bin sha f3053c8c..., "
                            "sS*invS = 127.000002 is NOT the current baseline"}
    (CARD / "evidence" / "PLAN-G3-REBUILD.json").write_text(
        json.dumps(rep, indent=1) + "\n", encoding="utf-8", newline="\n")
    print(json.dumps({"passes": ordinal, "changed": len(changed),
                      "plan_sha": rep["candidate_plan_sha256"]}, indent=1))
    for c in changed:
        print("  ", c["ordinal"], c["name"], c["shared_old"], "->", c["shared_new"])


if __name__ == "__main__":
    main()
