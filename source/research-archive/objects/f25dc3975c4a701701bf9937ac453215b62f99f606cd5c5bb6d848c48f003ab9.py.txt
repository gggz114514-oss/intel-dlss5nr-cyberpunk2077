"""Local, card-private Intel ISA (GTPin --dump_isa asm5) parser.

Same line grammar as the campaign's ISA.py, plus the missing piece this card
needs: in these dumps the dependency token group `{...}` sits *before* the
`// message` comment, so the campaign parser (which only accepts `{...}` at
end-of-line) reads deps as None.  Here both orders are accepted.

CPU only, read-only on the dump.
"""
import re
from pathlib import Path

LINE = re.compile(r'^/\*\s*\[(?P<addr>[0-9A-Fa-f]+)\]\s*\*/\s*(?P<rest>.*)$')
TOKGRP = re.compile(r'\{\s*(?P<deps>[^}]*)\}')
COMMENT = re.compile(r'//\s*(?P<c>.*)$')


def parse(path):
    out = []
    for raw in Path(path).read_text(encoding='utf-8-sig', errors='replace').splitlines():
        m = LINE.match(raw)
        if not m:
            continue
        rest = m.group('rest')
        deps = None
        dm = TOKGRP.search(rest)
        if dm:
            deps = dm.group('deps').strip()
            rest = rest[:dm.start()] + rest[dm.end():]
        comment = None
        cm = COMMENT.search(rest)
        if cm:
            comment = cm.group('c').strip()
            rest = rest[:cm.start()]
        toks = rest.split()
        if not toks:
            continue
        flags = ''
        if toks[0].startswith('(') and toks[0].endswith(')'):
            flags = toks[0]
            toks = toks[1:]
        if not toks:
            continue
        op = toks[0]
        ops = [t for t in toks[1:] if not (t.startswith('(') and t.endswith(')'))]
        out.append(dict(addr=int(m.group('addr'), 16), flags=flags, op=op, ops=ops,
                        deps=deps, comment=comment, raw=raw.rstrip()))
    return out


def reg(op):
    m = re.match(r'^r(\d+)', op or '')
    return int(m.group(1)) if m else None


def rd_of(i):
    m = re.search(r'rd:(\d+)', i['comment'] or '')
    return int(m.group(1)) if m else 0


def msg_of(i):
    c = i['comment'] or ''
    return c.split(';')[-1].strip()


def tokens(i):
    return [int(x) for x in re.findall(r'\$(\d+)', i['deps'] or '')]
