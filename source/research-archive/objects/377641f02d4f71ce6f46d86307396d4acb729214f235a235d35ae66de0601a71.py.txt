#!/usr/bin/env python3
"""Write build-receipt.json, CANDIDATE.json and evidence/NUMERIC-CONTRACT.json.

All hashes/sizes are measured here, not transcribed.
"""
import hashlib
import json
import platform
import subprocess
import sys
from datetime import datetime
from pathlib import Path

CARD = Path(r"@NR_WORKSPACE@/work/dsh-xmx-handoff-validation-20261024/candidates/b3-c32-seed")
BUILD = Path(r"@NR_WORKSPACE@/work/dsh-xmx-handoff-validation-20261024/build/b3-c32-seed")
RUN = Path(r"@NR_WORKSPACE@/work/dsh-xmx-handoff-validation-20261024/run/b3-c32-seed")
GLSLANG = Path(r"@NR_WORKSPACE@/src/project/tools/glslang-16.5.0/bin/glslang.exe")
VALIDATOR = Path(r"@NR_WORKSPACE@/candidates/amd-map-20261015/s4-final-output-fusion-fix/production-repair/spv_validate_cpu.exe")
PYTHON = Path(r"@NR_WORKSPACE@/run/python312/python.exe")

def sha(p):
    return hashlib.sha256(Path(p).read_bytes()).hexdigest()


def info(p):
    p = Path(p)
    st = p.stat()
    return {"path": str(p).replace("\\", "/"), "sha256": sha(p), "bytes": st.st_size,
            "mtime": datetime.fromtimestamp(st.st_mtime).astimezone().isoformat(timespec="seconds")}


def read(p):
    return Path(p).read_text(encoding="utf-8", errors="replace").strip()

            "instruction_delta": {

                "OpName": -4, "OpVariable": -4, "OpLoad": -8, "OpStore": -8,

    print("parent identical:", identical_parent, " copies match:", spv_copies_match,
          " includes identical:", src_includes_match)

    main()
