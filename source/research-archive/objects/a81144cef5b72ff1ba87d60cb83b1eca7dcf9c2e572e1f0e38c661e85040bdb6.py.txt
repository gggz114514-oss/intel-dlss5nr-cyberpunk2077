"""Analyse the three-arm vk102 batch by hand: the vk102 arm cannot pass the gold gate because it is
a different model version, so only its timing is used (the identity protocol requires exactly that).
"""
import json, sys
from collections import defaultdict
from pathlib import Path
sys.path.insert(0, str(Path(__file__).parent))
import COMMON as K

d = K.read_json(K.C / 'queue' / 'vk102-compare' / 'LUNA-ORDINARY.json')
jobs = d['jobs'] if isinstance(d, dict) and 'jobs' in d else d
by_arm = defaultdict(list)
hashes = defaultdict(set)
for j in jobs:
    if j.get('status') != 'COMPLETED':
        continue
    samples = j.get('samples') or []
    gpu = [s.get('gpu_ms') for s in samples if s.get('gpu_ms')]
    if gpu:
        by_arm[j['arm']].append(sum(gpu) / len(gpu))
    if j.get('frame'):
        hashes[j['arm']].add(j['frame'])
print('samples per arm:', {k: len(v) for k, v in by_arm.items()})
print('distinct frame hashes per arm:', {k: len(v) for k, v in hashes.items()})
for k, v in hashes.items():
    print('  %s: %s' % (k, sorted(v)[:2]))

T = 2.570581835636305
def compare(x, y, name):
    n = min(len(x), len(y))
    diffs = [x[i] - y[i] for i in range(n)]
    m = sum(diffs) / n
    sd = (sum((v - m) ** 2 for v in diffs) / (n - 1)) ** 0.5
    half = T * sd / (n ** 0.5)
    same = all(v > 0 for v in diffs) or all(v < 0 for v in diffs)
    print('%s: %s saving=%.6f CI95=[%.6f, %.6f] groups=%d means %.3f -> %.3f' % (
        name, 'FASTER_95' if (m - half > 0 and same) else 'INDISTINCT',
        m, m - half, m + half, n, sum(y) / n, sum(x) / n))
    return dict(name=name, saving_ms=m, CI95_ms=[m - half, m + half], groups=n,
                mean_x=sum(x) / n, mean_y=sum(y) / n)

A, C, V = by_arm['A'], by_arm['C'], by_arm['V']
out = []
if len(V) and len(A):
    out.append(compare(A, V, 'A_minus_V (handoff vs vk102)'))
if len(V) and len(C):
    out.append(compare(C, V, 'C_minus_V (head vs vk102)'))
if len(A) and len(C):
    out.append(compare(A, C, 'A_minus_C (handoff vs head)'))
K.write_json(K.C / 'evidence' / 'VK102-COMPARISON.json',
             dict(arms={k: dict(n=len(v), mean_ms=sum(v) / len(v)) for k, v in by_arm.items()},
                  comparisons=out, frame_hashes_per_arm={k: sorted(v) for k, v in hashes.items()},
                  method='same runner/fixture/pool/warm1/reps30; six balanced permutations; '
                         'all samples retained; t(df=5) over permutation means. The vk102 arm is a '
                         'different model version so it is compared on timing only.'))
