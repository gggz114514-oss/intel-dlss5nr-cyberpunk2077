from pathlib import Path
import json,re,hashlib
D=Path('@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/fullnet-mechanism-transfer-20261026')
S=D/'f05-four-weight-window-20261027';T=D/'f10-weight-return-separation-20261027'
read=lambda p:json.loads(p.read_text(encoding='utf-8-sig'))
result=[]
for w,i in [('c64',5),('c128',9)]:
 r={'width':w,'pipeline':i}
 for arm,root in [('parent',T),('candidate',S)]:
  p=next((root/'captures').glob(f'*/ISA/*_{i}_orig.asm'))
  ins=[(int(m[1],16),l) for l in p.read_text().splitlines() if (m:=re.search(r'\[([0-9A-F]+)\]',l))]
  dpas=[(pc,l) for pc,l in ins if 'dpas.' in l];end=dpas[5][0];first=dpas[0][0]
  rows=[l for pc,l in ins if pc<=end and ('load.ugm' in l or 'dpas.' in l or 'sync.' in l)]
  (S/f'{w}-{arm}-feed.asm').write_text('\n'.join(l for pc,l in ins if pc<=end+64),encoding='utf-8')
  stats=read(next((root/'run').glob('*resources*/pipeline-statistics.json')))
  st={s['name']:s['value'] for s in next(x for x in stats['pipelines'] if x['index']==i)['executables'][0]['statistics']}
  r[arm]={'native':str(p),'sha256':hashlib.sha256(p.read_bytes()).hexdigest(),'statistics':st,'first_six_dpas':[l for pc,l in dpas[:6]],'feed_rows':rows,'weights_before_first':sum(pc<first and 'load.ugm.d32x8.a32' in l for pc,l in ins)}
  assert st['Scratch Memory Size']==0
 result.append(r)
(S/'NATIVE-REVIEW.json').write_text(json.dumps(result,indent=2),encoding='utf-8')
for r in result:
 print(r['width'],[(a,r[a]['statistics'],r[a]['weights_before_first']) for a in ['parent','candidate']])
 print('\n'.join(r['candidate']['feed_rows']))
