"""n39-f06-c256-slack: reproduce the parent SPIR-V bytes of the two shaders this card touches.

Parent identity (chain head, GROUPS.md "链头 sha" == baseline sha for both groups):
  F06 C256  15 PASS (38,41,44,47,50,53,56,170,173,176,179,182,185,188,191)
            svp 881dff8c5145...  src c3efee68...  pinned header c256_common.glsl 315e7c34
  F18 C256  1  PASS (168)
            svp 9057283fe0ac...  src 7ef6e586...  pinned header b3_03_swin_c256_d2_common.glsl 315e7c34
Recipe for both = evidence/RECIPES.json: glslang 16.5.0 `-V --target-env vulkan1.3`, no -D,
one pinned include dir, decoration_parent=null, nc_recipe.mode = raw (no NoContraction patch).

Writes build/repro.json + parent-rebuild.spv + parent-f18-rebuild.spv.
"""
import json
import pathlib
import shutil
import sys

sys.path.insert(0, str(pathlib.Path(__file__).resolve().parents[2]))
import COMMON as K                                     # noqa: E402

CARD = pathlib.Path(__file__).resolve().parent
GROUPS = [dict(label='f06-c256', parent_sha='881dff8c51454f79181a567d6b976ea21ec7c81174bf075ca7be723b09c4a3f0',
               out='parent-rebuild.spv', ordinal=38, expect_rows=15),
          dict(label='f18-c256', parent_sha='9057283fe0ac4fda674b56c9d68aff7ae050b10c8897a5c517f5a710971d6e05',
               out='parent-f18-rebuild.spv', ordinal=168, expect_rows=1)]


def main():
    recipes = K.read_json(K.C / 'evidence' / 'RECIPES.json')
    recipes = recipes.get('shaders') or recipes
    out = dict(groups={})
    ok = True
    for g in GROUPS:
        rec = recipes[g['parent_sha']]
        src = pathlib.Path(rec['source'])
        pinned = pathlib.Path(rec['compile']['pinned_include_dir'])
        assert K.sha(src) == rec['source_sha256']
        for name, want in rec['compile']['header_pins'].items():
            assert K.sha(pinned / name) == want, name
        raw = CARD / 'build' / ('%s-parent-raw.spv' % g['label'])
        raw.parent.mkdir(exist_ok=True)
        K.compile_glsl(src, raw, [pinned] + [pathlib.Path(d) for d in rec['compile']['include_dirs']])
        res = dict(parent_sha_expected=g['parent_sha'], source=K.ident(src),
                   recipe=dict(include_dirs=[str(pinned)] + rec['compile']['include_dirs'],
                               defines=rec['compile']['defines'],
                               decoration_parent=rec['decoration_parent'],
                               nc_recipe=rec['nc_recipe'],
                               header_pins=rec['compile']['header_pins']),
                   raw=K.ident(raw), raw_match=K.sha(raw) == g['parent_sha'])
        if res['raw_match']:
            shutil.copyfile(raw, CARD / g['out'])
            res['parent_rebuild'] = K.ident(CARD / g['out'])
            res['recipe_used'] = 'raw (no decoration transfer, no NoContraction patch)'
        rows = [r for r in K.rows() if r['spv_sha256'] == g['parent_sha']]
        res['rows'] = dict(count=len(rows), ordinals=[r['ordinal'] for r in rows],
                           families=sorted({r['family'] for r in rows}),
                           spv_paths=sorted({r['spv'] for r in rows}),
                           plan_field_2=sorted({r['header_fields'][1] for r in rows}),
                           grids=sorted({tuple(r['header_fields'][4:]) for r in rows}))
        assert res['rows']['count'] == g['expect_rows'], res['rows']['count']
        out['groups'][g['label']] = res
        ok = ok and res['raw_match']
        print('%-9s raw_match=%s  rows=%d  sha=%s' % (g['label'], res['raw_match'],
                                                     res['rows']['count'],
                                                     res.get('parent_rebuild', {}).get('sha256', '')[:16]))
    # the two rows the card's "17" counts but this card cannot change
    r193 = K.row_by_ordinal(193)
    true193 = K.read_json(K.C / 'evidence' / 'TRUE-SLM.json')
    t = next(r for r in true193['rows'] if r['ordinal'] == 193)
    out['seventeenth_row'] = dict(
        ordinal=193, family=r193['family'], name=r193['name'],
        spv_sha256=r193['spv_sha256'], plan_field_2=r193['header_fields'][1],
        module_workgroup_bytes=t['true'],
        note='the card counts 17 rows for the 32768 B bucket, but ordinal 193 carries a STALE '
             'plan field (32768) while its module declares 16384 B (L35, evidence/TRUE-SLM.json); '
             'it is already at 4 residents and there is nothing to remove, so the package this '
             'card can actually dispatch is 15 F06 + 1 F18 = 16 rows')
    K.write_json(CARD / 'build' / 'repro.json', out)
    print('package rows = 16 (+1 stale row 193, true 16384 B)')
    return 0 if ok else 1


if __name__ == '__main__':
    sys.exit(main())
