"""Generate candidate.comp for n05d-f05-c64-mbatch from the byte-identified parent.

The parent file (parent-rebuild.comp, sha 2c2dc9db...) is the current accepted
F05 C64 source.  This script replaces exactly one region -- the QKV projection
block, from the `// ---- QKV projection` marker through the `} // row_phase values
retire before next phase` line -- and leaves every other byte untouched, then
writes SOURCE-DIFF.patch from the two texts.

CPU only; no GPU, no Vulkan, no plan, no baseline write.
"""
import difflib
import sys
from pathlib import Path

CARD = Path(__file__).resolve().parent.parent

START = '    // ---- QKV projection: unchanged (one A fragment feeds all six accumulators) ----'
END = '    } // row_phase values retire before next phase'

NEW = r'''    // ---- QKV projection: M-batch weight-tile reuse, D2 form (Q/K pass, V pass) --
    // n05d-f05-c64-mbatch (2026-10-27).  The parent walks the same 48 weight tiles
    // twice, once per row_phase, with six f32 accumulators live (6 x 8 = 48 GRF; the
    // measured QKV-window peak of this kernel is 70/128 GRF, 29 GRF of it non-
    // accumulator).  Holding all twelve accumulators at once (12 x 8 = 96 GRF) costs
    // 96 + 29 = 125 GRF before the second M fragment's A return (+8..12 GRF) is
    // counted, so D1 does not fit in 128 GRF and is not implemented (BUDGET-C64.md).
    //
    // D2 splits by channel group: pass 1 keeps the eight Q/K accumulators of both M
    // blocks (64 GRF) and loads each Q/K weight tile once for two independent MMAs;
    // pass 2 does the same for V with four accumulators.  Q and K cannot be split
    // apart because the sealed norm tree consumes q0,q1,k0,k1 together
    // (nr_norm_qk_pair), so (Q,K) | V is the only legal boundary.
    //
    //   per subgroup, full 64-token window        parent    candidate
    //     weight tile loads (d32x8,  rd:8)             48           24   -24 (-50%)
    //     A row loads       (d32x8t, rd:1)        8 pieces    16 pieces  +8 pieces
    //                                           64 messages 128 messages +64 messages
    //     DPAS (8x16x16) in this section               48           48     +0
    //     peak MAT_C accumulators live                  6            8    +2 (16 GRF)
    //
    // C64 owner recomputation (this is not the C128 mapping): the parent's M block is
    // m0q = (sg % 4u) * 8u + row_phase * 32u, so its two owner phases are the token
    // blocks [(sg%4)*8, +8) and [(sg%4)*8+32, +8) of the 64-token window; K = 64 means
    // the k axis has exactly four steps k0 = 0,16,32,48 (not eight), the weight tile
    // index is ((96h+t)/16)*4 + k0/16 (the *4u is that four-step K axis), and the A row
    // stride is 64 half (not 128).  m0a is the old row_phase 0 block and m0b the old
    // row_phase 1 block, so every accumulator consumes k0 in the same order with the
    // same operand values and the same store addresses: the result is bit-exact.
    const uint m0a = (sg % 4u) * 8u;
    const uint m0b = m0a + 32u;
    { // ---- pass 1: Q and K of both M blocks, each weight tile consumed once ----
        MAT_C accQ0a = MAT_C(0.0);
        MAT_C accQ1a = MAT_C(0.0);
        MAT_C accK0a = MAT_C(0.0);
        MAT_C accK1a = MAT_C(0.0);
        MAT_C accQ0b = MAT_C(0.0);
        MAT_C accQ1b = MAT_C(0.0);
        MAT_C accK0b = MAT_C(0.0);
        MAT_C accK1b = MAT_C(0.0);
        for (uint k0 = 0u; k0 < 64u; k0 += 16u) {
            MAT_A a0; coopMatLoad(a0, gY, (w * 64u + m0a) * 64u + k0, 64u, LROW);
            MAT_A a1; coopMatLoad(a1, gY, (w * 64u + m0b) * 64u + k0, 64u, LROW);
            {
                MAT_B b0, b1;
                coopMatLoad(b0, pack, wq + (((96u * h + 0u) / 16u) * 4u + k0 / 16u) * 256u, 16u, LCOL);
                coopMatLoad(b1, pack, wq + (((96u * h + 16u) / 16u) * 4u + k0 / 16u) * 256u, 16u, LCOL);
                accQ0a = coopMatMulAdd(a0, b0, accQ0a);
                accQ0b = coopMatMulAdd(a1, b0, accQ0b);
                accQ1a = coopMatMulAdd(a0, b1, accQ1a);
                accQ1b = coopMatMulAdd(a1, b1, accQ1b);
            }
            {
                MAT_B b0, b1;
                coopMatLoad(b0, pack, wq + (((96u * h + 32u) / 16u) * 4u + k0 / 16u) * 256u, 16u, LCOL);
                coopMatLoad(b1, pack, wq + (((96u * h + 48u) / 16u) * 4u + k0 / 16u) * 256u, 16u, LCOL);
                accK0a = coopMatMulAdd(a0, b0, accK0a);
                accK0b = coopMatMulAdd(a1, b0, accK0b);
                accK1a = coopMatMulAdd(a0, b1, accK1a);
                accK1b = coopMatMulAdd(a1, b1, accK1b);
            }
        }
        // Retire one M block at a time, in the parent's row_phase order (m0a first),
        // so at most four half fragments are live and the publication order of Q, K
        // and V matches the parent exactly.
        {
            MAT_H q0=MAT_H(accQ0a),q1=MAT_H(accQ1a);
            MAT_H k0=MAT_H(accK0a),k1=MAT_H(accK1a);
            if(debug) { coopMatStore(q0,dbg,45056u+m0a*64u+32u*h,64u,LROW); coopMatStore(q1,dbg,45056u+m0a*64u+32u*h+16u,64u,LROW); }
            if(debug) { coopMatStore(k0,dbg,49152u+m0a*64u+32u*h,64u,LROW); coopMatStore(k1,dbg,49152u+m0a*64u+32u*h+16u,64u,LROW); }
            nr_norm_qk_pair(q0,q1,k0,k1,float16_t(auxf[uint(pc.p5.x)+h]),lane);
            coopMatStore(q0,qkv,((gl_SubgroupID / 4u) * 6144u) + 0u+m0a*32u,32u,LROW);
            coopMatStore(q1,qkv,((gl_SubgroupID / 4u) * 6144u) + 0u+m0a*32u+16u,32u,LROW);
            if(debug) { coopMatStore(q0,dbg,57344u+m0a*64u+32u*h,64u,LROW); coopMatStore(q1,dbg,57344u+m0a*64u+32u*h+16u,64u,LROW); }
            coopMatStore(k0,qkv,((gl_SubgroupID / 4u) * 6144u) + 2048u+(m0a/8u*2u+0u)*128u,8u,LCOL);
            coopMatStore(k1,qkv,((gl_SubgroupID / 4u) * 6144u) + 2048u+(m0a/8u*2u+1u)*128u,8u,LCOL);
            if(debug) { coopMatStore(k0,dbg,61440u+m0a*64u+32u*h,64u,LROW); coopMatStore(k1,dbg,61440u+m0a*64u+32u*h+16u,64u,LROW); }
        }
        {
            MAT_H q0=MAT_H(accQ0b),q1=MAT_H(accQ1b);
            MAT_H k0=MAT_H(accK0b),k1=MAT_H(accK1b);
            if(debug) { coopMatStore(q0,dbg,45056u+m0b*64u+32u*h,64u,LROW); coopMatStore(q1,dbg,45056u+m0b*64u+32u*h+16u,64u,LROW); }
            if(debug) { coopMatStore(k0,dbg,49152u+m0b*64u+32u*h,64u,LROW); coopMatStore(k1,dbg,49152u+m0b*64u+32u*h+16u,64u,LROW); }
            nr_norm_qk_pair(q0,q1,k0,k1,float16_t(auxf[uint(pc.p5.x)+h]),lane);
            coopMatStore(q0,qkv,((gl_SubgroupID / 4u) * 6144u) + 0u+m0b*32u,32u,LROW);
            coopMatStore(q1,qkv,((gl_SubgroupID / 4u) * 6144u) + 0u+m0b*32u+16u,32u,LROW);
            if(debug) { coopMatStore(q0,dbg,57344u+m0b*64u+32u*h,64u,LROW); coopMatStore(q1,dbg,57344u+m0b*64u+32u*h+16u,64u,LROW); }
            coopMatStore(k0,qkv,((gl_SubgroupID / 4u) * 6144u) + 2048u+(m0b/8u*2u+0u)*128u,8u,LCOL);
            coopMatStore(k1,qkv,((gl_SubgroupID / 4u) * 6144u) + 2048u+(m0b/8u*2u+1u)*128u,8u,LCOL);
            if(debug) { coopMatStore(k0,dbg,61440u+m0b*64u+32u*h,64u,LROW); coopMatStore(k1,dbg,61440u+m0b*64u+32u*h+16u,64u,LROW); }
        }
    }
    { // ---- pass 2: V of both M blocks, each weight tile consumed once ----
        MAT_C accV0a = MAT_C(0.0);
        MAT_C accV1a = MAT_C(0.0);
        MAT_C accV0b = MAT_C(0.0);
        MAT_C accV1b = MAT_C(0.0);
        for (uint k0 = 0u; k0 < 64u; k0 += 16u) {
            MAT_A a0; coopMatLoad(a0, gY, (w * 64u + m0a) * 64u + k0, 64u, LROW);
            MAT_A a1; coopMatLoad(a1, gY, (w * 64u + m0b) * 64u + k0, 64u, LROW);
            MAT_B b0, b1;
            coopMatLoad(b0, pack, wq + (((96u * h + 64u) / 16u) * 4u + k0 / 16u) * 256u, 16u, LCOL);
            coopMatLoad(b1, pack, wq + (((96u * h + 80u) / 16u) * 4u + k0 / 16u) * 256u, 16u, LCOL);
            accV0a = coopMatMulAdd(a0, b0, accV0a);
            accV0b = coopMatMulAdd(a1, b0, accV0b);
            accV1a = coopMatMulAdd(a0, b1, accV1a);
            accV1b = coopMatMulAdd(a1, b1, accV1b);
        }
        {
            MAT_H narrowed;
            narrowed = MAT_H(accV0a);
            coopMatStore(narrowed, qkv, ((gl_SubgroupID / 4u) * 6144u) + 4096u + m0a * 32u, 32u, LROW);
            narrowed = MAT_H(accV1a);
            coopMatStore(narrowed, qkv, ((gl_SubgroupID / 4u) * 6144u) + 4096u + m0a * 32u + 16u, 32u, LROW);
        }
        {
            MAT_H narrowed;
            narrowed = MAT_H(accV0b);
            coopMatStore(narrowed, qkv, ((gl_SubgroupID / 4u) * 6144u) + 4096u + m0b * 32u, 32u, LROW);
            narrowed = MAT_H(accV1b);
            coopMatStore(narrowed, qkv, ((gl_SubgroupID / 4u) * 6144u) + 4096u + m0b * 32u + 16u, 32u, LROW);
        }
    } // both M blocks retire before the publication barrier'''


def main():
    src = (CARD / 'parent-rebuild.comp').read_text(encoding='utf-8')
    lines = src.splitlines()
    i = next(k for k, l in enumerate(lines) if l == START)
    j = next(k for k, l in enumerate(lines) if l == END)
    out = lines[:i] + NEW.splitlines() + lines[j + 1:]
    text = '\n'.join(out) + ('\n' if src.endswith('\n') else '')
    (CARD / 'candidate.comp').write_text(text, encoding='utf-8')

    diff = difflib.unified_diff(
        lines, out, fromfile='a/candidate.comp', tofile='b/candidate.comp',
        n=3, lineterm='')
    dtext = '\n'.join(diff) + '\n'
    (CARD / 'SOURCE-DIFF.patch').write_text(dtext, encoding='utf-8')

    added = sum(1 for l in dtext.splitlines() if l.startswith('+') and not l.startswith('+++'))
    removed = sum(1 for l in dtext.splitlines() if l.startswith('-') and not l.startswith('---'))
    hunks = sum(1 for l in dtext.splitlines() if l.startswith('@@'))
    print('parent lines      :', len(lines))
    print('candidate lines   :', len(out))
    print('replaced region   : lines %d..%d (%d lines)' % (i + 1, j + 1, j - i + 1))
    print('diff hunks        :', hunks, ' +%d -%d' % (added, removed))
    print('candidate bytes   :', len(text.encode('utf-8')))
    if len(text.encode('utf-8')) < 20000:
        sys.exit('unexpected output size')


if __name__ == '__main__':
    main()
