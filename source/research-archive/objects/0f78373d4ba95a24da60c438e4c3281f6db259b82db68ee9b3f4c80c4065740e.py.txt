"""N02B F20 NoContraction site parity check (CPU only).

The contract requires the float FMA / NoContraction site set to correspond
one-to-one with the parent.  Both modules carry 45 NoContraction decorations
transferred from the recorded decoration parent; this script proves the
*decoration sites* are the same set of floating-point instructions, not just the
same count: every decorated instruction is fingerprinted by
(function index, opcode, ID-free operand signature) and the two multisets must
be equal.

Writes build/nc-check.json.
"""
import collections
import json
import pathlib
import sys

sys.path.insert(0, r'@NR_WORKSPACE@/build/zcode-fp16-compute-backend-20261021/scripts')
import spv_tools_local as ST          # noqa: E402

CARD = pathlib.Path(__file__).resolve().parent


def site_fingerprints(m):
    """[(function_index, position_in_function, opcode, operand_signature)] for every
    NoContraction-decorated result instruction."""
    decorated = set(ST.no_contraction_ids(m))
    regions = ST.function_regions(m)
    out = []
    for fi, region in enumerate(regions):
        for pos, idx in enumerate(region):
            op, ops, _, _ = m.instructions[idx]
            rid = ST.result_id_of(op, ops)
            if rid is not None and rid in decorated:
                out.append((fi, pos, op, str(m.operand_signature(rid, 3))))
    return out


def main():
    par = ST.Module(CARD / 'parent-rebuild.spv')
    cand = ST.Module(CARD / 'candidate.spv')
    sp = site_fingerprints(par)
    sc = site_fingerprints(cand)
    # (function, opcode, signature) must match as a multiset; positions shift only
    # inside the function this card edits (dead integer instructions were removed).
    key = lambda t: (t[0], t[2], t[3])
    bp = collections.Counter(key(t) for t in sp)
    bc = collections.Counter(key(t) for t in sc)
    res = {
        'card': 'n02b-f20-store',
        'parent_decorated_sites': len(sp),
        'candidate_decorated_sites': len(sc),
        'per_function_parent': dict(sorted(collections.Counter(t[0] for t in sp).items())),
        'per_function_candidate': dict(sorted(collections.Counter(t[0] for t in sc).items())),
        'identical_signature_multiset': bp == bc,
        'parent_only': [str(k) for k in (bp - bc)],
        'candidate_only': [str(k) for k in (bc - bp)],
        'opcodes_decorated': sorted({t[2] for t in sp} | {t[2] for t in sc}),
        'position_shift_within_function': {
            'parent_positions': sorted(t[1] for t in sp),
            'candidate_positions': sorted(t[1] for t in sc),
        },
        'method': ('ST.decoration_pattern over every function region; each decorated instruction is '
                   'identified by (function index, opcode, ID-free operand signature). Positions may '
                   'shift inside the edited function because dead integer instructions were removed, '
                   'but the (function, opcode, signature) multiset must be identical.'),
    }
    assert res['identical_signature_multiset'], res['parent_only'] + res['candidate_only']
    assert res['parent_decorated_sites'] == res['candidate_decorated_sites'] == 45
    (CARD / 'build/nc-check.json').write_text(json.dumps(res, indent=1), encoding='utf-8')
    print(json.dumps({k: res[k] for k in ('parent_decorated_sites', 'candidate_decorated_sites',
                                          'per_function_parent', 'identical_signature_multiset',
                                          'opcodes_decorated')}, indent=1))


if __name__ == '__main__':
    main()
