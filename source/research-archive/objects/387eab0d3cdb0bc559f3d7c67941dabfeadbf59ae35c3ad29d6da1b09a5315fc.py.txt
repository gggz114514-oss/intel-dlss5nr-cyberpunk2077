def sha(p):
    return hashlib.sha256(pathlib.Path(p).read_bytes()).hexdigest()


def run(argv):
    r = subprocess.run([str(x) for x in argv], capture_output=True, text=True,
                       encoding='utf-8', errors='replace')
    return r.returncode, (r.stdout or '').strip(), (r.stderr or '').strip()


def compile_glsl(src, out, incs):
    argv = [GLSLANG, '-V', '--target-env', 'vulkan1.3']
    for d in incs:
        argv.append('-I' + str(d))
    argv += [str(src), '-o', str(out)]
    return argv, run(argv)


def build_spv(src, raw, final, incs):
    """compile + NoContraction pattern transfer from the recorded decoration parent."""
    argv, (rc, so, se) = compile_glsl(src, raw, incs)
    assert rc == 0, 'glslang failed\n%s\n%s' % (so, se)
    pm = ST.Module(DECOR_PARENT)
    cm = ST.Module(raw)
    keep, info = ST.transfer_decoration_pattern_aligned(pm, cm, set())
    ST.set_no_contraction(cm, keep, final)
    return dict(argv=[str(x) for x in argv], glslang_rc=rc, glslang_stderr=se[-400:],
                raw_sha256=sha(raw), raw_bytes=pathlib.Path(raw).stat().st_size,
                nc_count=len(keep), nc_parent=len(ST.no_contraction_ids(pm)),
                nc_info={k: info[k] for k in ('alignment', 'functions_mapped',
                                              'child_functions_without_parent_counterpart',
                                              'parent_decorations_total')},
                spv_sha256=sha(final), spv_bytes=pathlib.Path(final).stat().st_size)


def unified_diff(old_bytes, new_bytes, a_name, b_name):
    import difflib
    a = old_bytes.decode('utf-8-sig').replace('\r\n', '\n').split('\n')
    b = new_bytes.decode('utf-8-sig').replace('\r\n', '\n').split('\n')
    lines = [l.rstrip('\n') for l in difflib.unified_diff(a, b, fromfile=a_name,
                                                          tofile=b_name, n=4)]
    return '\n'.join(lines) + '\n'


def plan_audit(card_plan, base_plan, base_sha, chain):
    base_text = pathlib.Path(base_plan).read_text(encoding='utf-8-sig')
    base_lines = base_text.splitlines()
    cand_lines = pathlib.Path(card_plan).read_text(encoding='utf-8-sig').splitlines()
    assert len(base_lines) == len(cand_lines), 'plan line count changed'
    diff_lines = [i for i, (a, b) in enumerate(zip(base_lines, cand_lines)) if a != b]
    assert len(diff_lines) == 1, 'plan differs on %d lines: %s' % (len(diff_lines), diff_lines)
    i = diff_lines[0]
    assert base_lines[i] == K.pass_lines_of(base_plan)[ORDINAL]['line'], 'not the ord-230 line'
    recs = K.pass_lines_of(card_plan)
    lines = cand_lines
    i0 = lines.index(recs[ORDINAL]['line'])
    binds, push = [], None
    for l in lines[i0 + 1:]:
        if l.startswith('PASS '):
            break
        if l.startswith('BIND '):
            binds.append([int(x) for x in l.split()[1:]])
        elif l.startswith('PUSH '):
            push = [int(x) for x in l.split()[1:]]
    return dict(path=str(card_plan), sha256=sha(card_plan),
                parent_plan={'path': str(base_plan), 'sha256': base_sha},
                chain_head_label=chain[-1]['label'] if chain else None,
                replaced_passes=1, diff_lines=diff_lines, line_count=len(cand_lines),
                changed_line_ordinal=ORDINAL, old_line=base_lines[i], new_line=cand_lines[i],
                pass_151_occurrences=sum(1 for l in lines if l.startswith(PASS_LABEL + ' ')),
                bind=[{'set': b[0], 'buffer_id': b[1], 'offset': b[2], 'size_bytes': b[3]}
                      for b in binds],
                push_dwords=push,
                push_pc={('p%d' % (k // 4)): push[k:k + 4] for k in range(0, len(push), 4)},
                push_dwords_as_i32=[x - (1 << 32) if x >= (1 << 31) else x for x in push])


def make_candidate(name, anchors, patch_name):
    """anchors: [(old_bytes, new_bytes, count)] -> writes <name>.comp, spv, plan, patch."""
    src = PARENT_SRC.read_bytes()
    cand = src
    for old, new, cnt in anchors:
        assert src.count(old) == cnt, '%s: anchor count %d != %d (%r)' % (
            name, src.count(old), cnt, old[:60])
        cand = cand.replace(old, new)
    assert cand != src
    (CARD / (name + '.comp')).write_bytes(cand)
    patch = unified_diff(src, cand, 'a/f20-coarse-owner-20261026/candidate.comp',
                         'b/cards/n17b-f20-transport/%s.comp' % name)
    (CARD / patch_name).write_text(patch, encoding='utf-8')
    info = {'source': str(CARD / (name + '.comp')), 'sha256': sha(CARD / (name + '.comp')),
            'bytes': len(cand),
            'patch': {'path': str(CARD / patch_name), 'sha256': sha(CARD / patch_name),
                      'removed_lines': sum(1 for l in patch.splitlines()
                                           if l.startswith('-') and not l.startswith('---')),
                      'added_lines': sum(1 for l in patch.splitlines()
                                         if l.startswith('+') and not l.startswith('+++'))}}
    return info


def main():
    (CARD / 'build').mkdir(exist_ok=True)
    assert sha(PARENT_SRC) == PARENT_SRC_SHA, 'parent source sha changed'
    assert sha(PARENT_SPV) == PARENT_SPV_SHA, 'baseline parent SPV sha changed'

                   'evidence': 'evidence/RECIPES.json entry 215aca67... (nc_recipe mode=parent)'}}

    preb = build_spv(PARENT_SRC, CARD / 'build/parent-raw.spv',
                     CARD / 'parent-rebuild.spv', incs)
    preb['byte_match'] = preb['spv_sha256'] == PARENT_SPV_SHA
    assert preb['byte_match'], 'parent did not reproduce byte for byte'
    (CARD / 'parent.comp').write_bytes(PARENT_SRC.read_bytes())
    parent['rebuild'] = preb

    # ---------------- 2. the two variants ----------------
    probe_info = make_candidate('probe-candidate',
                                [(ol, ol.replace(b'coopMatStore', b'if (debug) coopMatStore'), 1)
                                 for ol in P_PUB_LINES] +
                                [(P_LOAD_OLD, P_LOAD_NEW, 1)],
                                'PROBE-SOURCE-DIFF.patch')
    y_info = make_candidate('candidate',
                            [(Y_DECL_OLD, Y_DECL_NEW, 1),
                             (Y_STORE_OLD, Y_STORE_NEW, 1),
                             (Y_SEED0_OLD, Y_SEED0_NEW, 1),
                             (Y_SEED1_OLD, Y_SEED1_NEW, 1)],
                            'SOURCE-DIFF.patch')

    out = {'card': 'n17b-f20-transport', 'family': 'F20', 'ordinal': ORDINAL,
           'gpu_used': False, 'deployed': False, 'parent': parent,
           'float_NC_checked': preb['nc_count']}
    out['float_NC_note'] = (
        'the same %d NoContraction decorations are transferred from the recorded decoration '
        'parent onto the same FP instructions; every mapped pair is verified by ID-free operand '
        'signature and the float/GLSL.std.450 expression multiset is compared site by site '
        '(build/spv-check-*.json)' % preb['nc_count'])

    base_plan, base_sha, chain = K.current_best()
    for key, name, info in (('probe', 'probe-candidate', probe_info), ('y', 'candidate', y_info)):
        info['build'] = build_spv(CARD / (name + '.comp'), CARD / ('build/%s-raw.spv' % name),
                                  CARD / (name + '.spv'), incs)
        info['spv'] = str(CARD / (name + '.spv'))
        info['nc_equal_to_parent'] = info['build']['nc_count'] == preb['nc_count']
        rc, so, se = run([VALIDATE, CARD / (name + '.spv')])
        info['validate'] = {'rc': rc, 'stdout': so, 'stderr': se}
        assert rc == 0, '%s failed spv_validate_cpu: %s %s' % (name, so, se)
        plan_path = CARD / ('probe-plan.txt' if key == 'probe' else 'plan.txt')
        n = K.make_plan(CARD / (name + '.spv'), plan_path, parent_plan=base_plan,
                        spv_shas={PARENT_SPV_SHA})
        assert n == 1, 'replaced %d passes, expected 1' % n
        info['plan'] = plan_audit(plan_path, base_plan, base_sha, chain)
        rc, so, se = run([RUNNER, '--plan', plan_path, '--cpu-check'])
        info['cpu_check'] = {'argv': [str(RUNNER), '--plan', str(plan_path), '--cpu-check'],
                             'rc': rc, 'stdout': so, 'stderr': se}
        assert rc == 0, 'cpu-check failed for %s' % name
        out[key] = info

    rc, so, se = run([RUNNER, '--plan', base_plan, '--cpu-check'])
    parent_check = {'argv': [str(RUNNER), '--plan', str(base_plan), '--cpu-check'],
                    'rc': rc, 'stdout': so, 'stderr': se}
    out['parent_cpu_check'] = parent_check
    for key in ('probe', 'y'):
        out[key]['cpu_check']['identical_to_parent'] = out[key]['cpu_check']['stdout'] == so
    out['chain_head'] = {'plan': str(base_plan), 'sha256': base_sha,
                         'labels': [c['label'] for c in chain]}

    K.write_json(CARD / 'BUILD.json', out)
    K.write_json(CARD / 'PROBE-BUILD.json',
                 {'card': 'n17b-f20-transport', 'role': 'PROBE (sensitivity ablation, NOT a candidate)',
                  'family': 'F20', 'ordinal': ORDINAL, 'gpu_used': False, 'deployed': False,
                  'parent': parent, 'probe': out['probe'],
                  'float_NC_checked': preb['nc_count'],
                  'chain_head': out['chain_head'], 'parent_cpu_check': parent_check})
    print(json.dumps({'parent_rebuild': preb['spv_sha256'],
                      'probe': {'src': probe_info['sha256'], 'spv': probe_info['build']['spv_sha256'],
                                'plan': probe_info['plan']['sha256'],
                                'nc': probe_info['build']['nc_count'],
                                'cpu': probe_info['cpu_check']['stdout']},
                      'y': {'src': y_info['sha256'], 'spv': y_info['build']['spv_sha256'],
                            'plan': y_info['plan']['sha256'],
                            'nc': y_info['build']['nc_count'],
                            'cpu': y_info['cpu_check']['stdout']},

if __name__ == '__main__':
    main()
