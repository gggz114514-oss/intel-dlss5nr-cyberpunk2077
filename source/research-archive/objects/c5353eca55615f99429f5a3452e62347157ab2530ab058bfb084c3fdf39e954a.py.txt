import json, os
ROOT = r"@NR_WORKSPACE@\work\codex-astra-fp16-int8-audit-20261024\fullnet-continuous-20261027"
s = json.load(open(os.path.join(ROOT, "cards", "n16-crossdispatch", "tools", "isa-stats.json"), encoding="utf-8"))
for pid in sorted(s, key=int):
    e = s[pid]
    loads = {k: v for k, v in e["counts"].items() if k.startswith("load.ugm")}
    stores = {k: v for k, v in e["counts"].items() if k.startswith("store.ugm")}
    slm = {k: v for k, v in e["counts"].items() if ".slm" in k or k.startswith(("barrier", "fence"))}
    print("%2s %-5s %-26s instr=%5d dpas=%4d" % (pid, ",".join(e["families"]), e["first_pass"], e["instructions"], e["dpas"]))
    print("    L:", ", ".join("%s=%d" % (k.split(".")[2] if len(k.split(".")) > 2 else k, v) for k, v in sorted(loads.items(), key=lambda kv: -kv[1])))
    print("    S:", ", ".join("%s=%d" % (k.split(".")[2] if len(k.split(".")) > 2 else k, v) for k, v in sorted(stores.items(), key=lambda kv: -kv[1])))
    if slm:
        print("    M:", ", ".join("%s=%d" % (k, v) for k, v in sorted(slm.items(), key=lambda kv: -kv[1])))
