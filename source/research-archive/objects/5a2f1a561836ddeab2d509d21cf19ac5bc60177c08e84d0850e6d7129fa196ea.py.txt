import json, collections, sys
LED = r"@NR_WORKSPACE@/work/dsh-fullnet-static-20261029/ledger"
d = json.load(open(LED + "/FULLNET-ROWS.json", encoding="utf-8"))
rows = d["rows"]


def edges(row):
    for e in row.get("quant_edges") or []:
        yield e


unk = []
for r in rows:
    for i, e in enumerate(edges(r)):
        if e.get("scale_A_source") == "UNKNOWN" or e.get("scale_B_source") == "UNKNOWN":
            unk.append((r, i, e))

print("UNKNOWN edges:", len(unk))
byfam = collections.Counter()
byname = collections.Counter()
for r, i, e in unk:
    byfam[r["family"]] += 1
    byname[r["name"]] += 1
print("by family:", dict(byfam))
print("by dispatch:")
for k, v in byname.items():
    print(f"  {v:3d}  {k}")

print("\n=== detail ===")
for r, i, e in unk:
    print(
        f'ord={r["ordinal"]:3d} {r["name"]:28s} fam={r["family"]:6s} edge#{i} ea={e.get("scale_A_source")} eb={e.get("scale_B_source")} stage={r.get("stage")}'
    )
    print(f'    edge="{e.get("edge")}" A={e.get("A_dtype")} B={e.get("B_dtype")} N={e.get("N")} K={e.get("K")}')
    print(f'    ev={str(e.get("evidence"))[:300]}')
    print(f'    spv={r["spv"]["sha256"][:16]} {r["spv"]["basename"]} int8mma={r["spv"]["int8_mma"]} fp16={r["spv"]["fp16_mma"]} 1/127={r["spv"].get("has_1_over_127")} 127={r["spv"].get("has_127")}')
