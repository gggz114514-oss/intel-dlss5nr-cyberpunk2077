import json, sys
p = r'@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/fullnet-continuous-20261027/cards/nat-index/evidence/NATIVE-INDEX.json'
want = sys.argv[1]
d = json.load(open(p, encoding='utf-8-sig'))
hits = []
seen = set()
for cap in d.get('captures', []):
    for rec in cap.get('records', []):
        if rec.get('spv_sha256_at_capture') == want and rec.get('identity') == 'CURRENT':
            key = rec['sha256']
            if key in seen:
                continue
            seen.add(key)
            hits.append((rec['isa'], rec['sha256'], rec.get('bytes'), rec.get('isa_scan'),
                         rec.get('driver_stats'), rec.get('baseline_pipelines')))
print('unique CURRENT captures:', len(hits))
for h in hits[:4]:
    print(json.dumps(dict(isa=h[0], sha256=h[1], bytes=h[2], isa_scan=h[3],
                          driver_stats=h[4], pipelines=h[5]), indent=1))
