import struct, collections
p = r"@NR_WORKSPACE@/candidates/int8-coverage-v2-20261020/B/spv/c64_d1.spv"
data=open(p,'rb').read(); n=len(data)//4
w=struct.unpack("<%dI"%n, data[:n*4]); i=5; c=collections.Counter()
first=[]
while i<n:
    wc=w[i]>>16; op=w[i]&0xFFFF
    if wc==0: break
    c[op]+=1
    if op in (5359,4457) and len(first)<3:
        first.append((op, list(w[i+1:i+wc])))
    i+=wc
print("opcounts", dict(sorted(c.items(), key=lambda kv:-kv[1])[:12]))
print("coopmat types", c.get(5358), c.get(4456), "loadNV", c.get(5356), "loadKHR", c.get(4459))
print("first muladds", first)
