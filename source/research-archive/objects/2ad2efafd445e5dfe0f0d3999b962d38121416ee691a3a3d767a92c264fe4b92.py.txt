import json, re, sys
from pathlib import Path
H = Path('@NR_USER_HOME@/Documents/Codex/2026-10-24/nr-fp16-int8-astra-audit-20261024/handoff/dsh-fullnet-20261027')
rows = json.load(open(H/'CURRENT-232-ROWS.json', encoding='utf-8-sig'))
out = {}
for r in rows:
    if r['family'] not in ('F17', 'F18', 'F03', 'F07'):
        continue
    src = Path(r['source'])
    txt = src.read_text(encoding='utf-8-sig', errors='replace')
    shared = [l.strip() for l in txt.splitlines() if re.match(r'\s*shared\s', l)]
    incs = [l.split('"')[1] for l in txt.splitlines() if l.strip().startswith('#include "')]
    out[r['ordinal']] = dict(fam=r['family'], name=r['name'], src=str(src),
                             spv_sha=r['spv_sha256'][:16], lines=len(txt.splitlines()),
                             shared=shared, includes=incs)
    print(r['ordinal'], r['family'], r['name'], r['spv_sha256'][:12])
    for s in shared:
        print('    ', s)
    for i in incs:
        p = src.parent / i
        print('     inc', i, p.exists())
