"""n37-f10-arena-shrink: parent byte reproduction (CPU only).

Parent = the chain head's F10 shader, i.e. n10-f10-next/candidate.spv
(sha 228dee8a437dd0827c4c7bc770707e645c753d493252fdda0907283889a9eb0e), which is the
PASS row of ordinals 62,66,...,164 in the current best plan
(accepted/n36-f05-arena-c64c128-20261027/plan.txt).

Recipe: evidence/RECIPES.json for the handoff baseline 6b7d2bd9 = glslang 16.5.0
-V --target-env vulkan1.3, no defines, one pinned include dir, decoration_parent null,
nc_recipe mode raw.  The same recipe reproduces the chain-head source byte-exactly
(verified here, both the chain head and the handoff baseline).

Writes build/repro.json.  No GPU.
"""
import hashlib
import json
import pathlib
import subprocess
import sys

W = pathlib.Path('@NR_WORKSPACE@')
A = W / 'work/codex-astra-fp16-int8-audit-20261024'
C = A / 'fullnet-continuous-20261027'
CARD = C / 'cards/n37-f10-arena-shrink'
GLSLANG = W / 'src/project/tools/glslang-16.5.0/bin/glslang.exe'
PIN = C / 'cards/idt-identity-recipe/evidence/headers/6b7d2bd95f0c'
HEADERS = {'register_handoff.glsl': 'dd8307b1c59c76e41d705442207fafd1c87e2a87fe57d04f343af5c27caa5f26',
           'split16_common.glsl': '6766f92732bb2e85e4eeef734aa371423bd8ea2bb318ec5605e85ee999e457a3'}

PARENT_SOURCE = C / 'cards/n10-f10-next/candidate.comp'
PARENT_SPV = C / 'cards/n10-f10-next/candidate.spv'
PARENT_SHA = '228dee8a437dd0827c4c7bc770707e645c753d493252fdda0907283889a9eb0e'
HANDOFF_SOURCE = (A / 'fullnet-mechanism-transfer-20261026/f10-weight-return-separation-20261027'
                  / 'candidate.comp')
HANDOFF_SPV = (A / 'fullnet-mechanism-transfer-20261026/f10-weight-return-separation-20261027'
               / 'candidate.spv')
HANDOFF_SHA = '6b7d2bd95f0c08044075dd0dceefc470bffc501d0c7406525b84fe023c2a4c48'


def sha(p):
    return hashlib.sha256(pathlib.Path(p).read_bytes()).hexdigest()


def compile_to(src, out):
    argv = [str(GLSLANG), '-V', '--target-env', 'vulkan1.3', '-I' + str(PIN),
            str(src), '-o', str(out)]
    r = subprocess.run(argv, capture_output=True, text=True, encoding='utf-8',
                       errors='replace')
    return argv, r.returncode, ((r.stdout or '') + (r.stderr or '')).strip()


def main():
    out = {}
    b = CARD / 'build'
    b.mkdir(parents=True, exist_ok=True)

    hdr = {n: dict(path=str(PIN / n), sha256=sha(PIN / n), expected=HEADERS[n],
                   match=sha(PIN / n) == HEADERS[n]) for n in HEADERS}
    assert all(h['match'] for h in hdr.values()), 'pinned header hash mismatch'

    arms = [('parent_chainhead', PARENT_SOURCE, PARENT_SPV, PARENT_SHA, 'parent-rebuild.spv'),
            ('handoff_baseline', HANDOFF_SOURCE, HANDOFF_SPV, HANDOFF_SHA,
             'handoff-rebuild.spv')]
    for name, src, spv, want, dst in arms:
        argv, rc, log = compile_to(src, b / dst)
        got = sha(b / dst)
        out[name] = dict(source=dict(path=str(src), sha256=sha(src)),
                         target=dict(path=str(spv), sha256=sha(spv),
                                     recorded_sha256=want),
                         compile_argv=argv, exit=rc, log_tail=log[-300:],
                         rebuild=dict(path=str(b / dst), sha256=got),
                         raw_match=(got == want),
                         note='glslang 16.5.0 -V --target-env vulkan1.3, pinned include dir, '
                              'no defines, no decoration transfer')
        if name == 'parent_chainhead':
            # contract deliverable: parent-rebuild.spv at the card root
            (CARD / 'parent-rebuild.spv').write_bytes((b / dst).read_bytes())
            out[name]['card_root_copy'] = dict(path=str(CARD / 'parent-rebuild.spv'),
                                               sha256=sha(CARD / 'parent-rebuild.spv'))
        print('%-17s raw_match=%s sha=%s' % (name, got == want, got[:16]))

    out['headers'] = hdr
    out['recipe'] = dict(
        card_recipe='evidence/RECIPES.json[6b7d2bd9...] (39/39 verified family recipe)',
        defines=[], include_dirs=[str(PIN)],
        decoration_parent=None, nc_recipe_mode='raw',
        independent_recheck='evidence/PARENT-BYTE-VERIFIED.json')
    ok = all(out[k]['raw_match'] for k, _, _, _, _ in arms)
    (b / 'repro.json').write_text(json.dumps(out, indent=2), encoding='utf-8')
    print('PARENT BYTE REPRODUCTION', 'PASS' if ok else 'FAIL')
    return 0 if ok else 1


if __name__ == '__main__':
    sys.exit(main())
