import json, os, sys
ROOT = r"@NR_WORKSPACE@\work\codex-astra-fp16-int8-audit-20261024\fullnet-continuous-20261027\evidence"

def trunc(x, n=1200):
    s = json.dumps(x, ensure_ascii=False)
    return s[:n]

n = json.load(open(os.path.join(ROOT, "NATIVE-INDEX.json"), encoding="utf-8"))
print("NATIVE keys:", list(n))
print("baseline keys:", list(n["baseline"]))
print("pipeline0:", trunc(n["baseline"]["pipelines"]["0"], 2500))
print("other top keys sample:")
for k in list(n):
    if k != "baseline":
        print(" ", k, type(n[k]).__name__, len(n[k]) if isinstance(n[k], (list, dict)) else n[k])

g = json.load(open(os.path.join(ROOT, "GLSL-INDEX.json"), encoding="utf-8"))
print("\nGLSL keys:", list(g)[:20])
for k in list(g)[:6]:
    v = g[k]
    print(" ", k, type(v).__name__, len(v) if isinstance(v, (list, dict)) else v)
    if isinstance(v, dict):
        print("   subkeys:", list(v)[:10])
        kk = list(v)[0]
        print("   first:", kk, trunc(v[kk], 800))
    if isinstance(v, list) and v:
        print("   first:", trunc(v[0], 800))
