"""S06 block15 deliverables: packs, quality, calibration, jobs, cost, candidate.

Everything numeric is recomputed from pack bytes read back from disk.
"""
import json
import sys
from pathlib import Path

import numpy as np

sys.path.insert(0, str(Path(__file__).parent))
import c256b15 as L

W = L.W
PKG = W / "PACK"
PKG.mkdir(exist_ok=True)
FRAMES = [("34189", L.D / "dump-fp16"),
          ("41444", L.D / "teacher-diagnostics/teacher-41444"),
          ("43654", L.D / "teacher-diagnostics/teacher-43654")]

search = json.loads((W / "evidence/06-calibration-search.json").read_text())
diag = np.load(W / "evidence/06-diag.npy")
SX_MSE = search["new"]["scales"]["sX"]
SA = search["new"]["scales"]["sA"]
SM = search["new"]["scales"]["sM"]
# per-input-channel absmax over the three calibration frames (equalisation input)
idx0, _ = L.window_to_canonical()
v0 = idx0 >= 0
am = np.zeros(256)
for _n, _td in FRAMES:
    _Xc = L.load_x_canonical(_td / "dump_block14_out.bin")
    _Xw = np.zeros((L.N_TOK, 256), dtype=np.float32)
    _Xw[v0] = _Xc[idx0[v0]]
    am = np.maximum(am, np.abs(_Xw[v0]).max(axis=0))
SX_ABS = float(np.max(am / diag) / 127.0)

idx, oob = L.window_to_canonical()
valid = idx >= 0
NW = L.N_TOK
We, Wm, Wc, rs = L.decode_weights()
raw_old = np.fromfile(L.PACK_OLD, dtype=np.uint8)
qWm = L.from_pack_bytes(raw_old[L.WM_OFF:L.SC_OFF].view(np.int8), 256, 128)
qWc = L.from_pack_bytes(raw_old[L.WC_OFF:L.WE_OFF].view(np.int8), 256, 256)
sc_old = np.frombuffer(raw_old[L.SCAL_OFF:L.SCAL_OFF + 128].tobytes(), dtype="<f4")
sWm = np.frombuffer(raw_old[L.SWM_OFF:L.SWM_OFF + 1024].tobytes(), dtype="<f4").astype(np.float64)
sWc = np.frombuffer(raw_old[L.SWC_OFF:L.SWC_OFF + 1024].tobytes(), dtype="<f4").astype(np.float64)
wt = We.astype(np.float64) * diag[None, :]
sWe64 = np.maximum(np.abs(wt).max(axis=1) / 127.0, 1e-12)
qWe = np.clip(np.rint(wt / sWe64[:, None]), -127.0, 127.0).astype(np.int8)
sWe = sWe64.astype(np.float32)


def build_pack(sX, sA, sM, qWe_local=qWe, sWe_local=sWe):
    out = raw_old.copy()
    out[L.WC_OFF:L.WE_OFF] = L.to_pack_bytes(qWc, 256).view(np.uint8)
    out[L.WE_OFF:L.WM_OFF] = L.to_pack_bytes(qWe_local, 256).view(np.uint8)
    out[L.WM_OFF:L.SC_OFF] = L.to_pack_bytes(qWm, 128).view(np.uint8)
    out[L.EW_OFF:L.EW_OFF + 4096] = np.frombuffer((np.float64(sX) * sWe_local.astype(np.float64)).astype("<f4").tobytes(), dtype=np.uint8)
    out[L.MW_OFF:L.MW_OFF + 1024] = np.frombuffer((np.float64(sA) * sWm).astype("<f4").tobytes(), dtype=np.uint8)
    out[L.ALPHA_OFF:L.ALPHA_OFF + 1024] = np.frombuffer((np.float64(sM) * sWc).astype("<f4").tobytes(), dtype=np.uint8)
    out[L.SWE_OFF:L.SWE_OFF + 4096] = np.frombuffer(sWe_local.astype("<f4").tobytes(), dtype=np.uint8)
    out[L.SWM_OFF:L.SWM_OFF + 1024] = np.frombuffer(sWm.astype("<f4").tobytes(), dtype=np.uint8)
    out[L.SWC_OFF:L.SWC_OFF + 1024] = np.frombuffer(sWc.astype("<f4").tobytes(), dtype=np.uint8)
    scal = np.zeros(32, dtype="<f4")
    scal[0] = np.float32(1.0 / sX); scal[1] = np.float32(1.0 / sA); scal[2] = np.float32(1.0 / sM)
    scal[3] = sX; scal[4] = sA; scal[5] = sM
    scal[6] = float(np.max(am / diag)); scal[7] = float(np.max(np.abs(qWe_local.astype(np.float32) * sWe_local[:, None])) * sX)
    scal[8] = float(sM * 127.0); scal[9:12] = [256.0, 128.0, 256.0]; scal[12] = 1.0
    out[L.SCAL_OFF:L.SCAL_OFF + 128] = np.frombuffer(scal.tobytes(), dtype=np.uint8)
    out[L.PAD_OFF:L.PAD_OFF + 1024] = np.frombuffer((np.float32(1.0 / sX) / diag).astype("<f4").tobytes(), dtype=np.uint8)
    out[L.PAD_OFF + 1024:L.PAD_OFF + L.PAD_BYTES] = 0
    assert out[L.QX_OFF:L.QX_OFF + L.QX_BYTES].tobytes() == raw_old[L.QX_OFF:L.QX_OFF + L.QX_BYTES].tobytes()
    assert out[L.QA_OFF:].tobytes() == raw_old[L.QA_OFF:].tobytes()
    return out


PACKS = {
    "A_mse": dict(sX=SX_MSE, sA=SA, sM=SM, file=PKG / "c256_b15_int8_qxeq.bin"),
    "B_absmax": dict(sX=SX_ABS, sA=SA, sM=SM, file=PKG / "c256_b15_int8_qxeq_absmax.bin"),
}
for k, v in PACKS.items():
    raw = build_pack(v["sX"], v["sA"], v["sM"])
    v["file"].write_bytes(raw.tobytes())
    v["sha256"] = L.sha256(v["file"])
    v["bytes"] = v["file"].stat().st_size
    v["raw"] = raw


def decode_pack(raw):
    codes = dict(
        wc=L.from_pack_bytes(raw[L.WC_OFF:L.WE_OFF].view(np.int8), 256, 256),
        we=L.from_pack_bytes(raw[L.WE_OFF:L.WM_OFF].view(np.int8), 1024, 256),
        wm=L.from_pack_bytes(raw[L.WM_OFF:L.SC_OFF].view(np.int8), 256, 128))
    sc = np.frombuffer(raw[L.SCAL_OFF:L.SCAL_OFF + 128].tobytes(), dtype="<f4")
    d = dict(invX=float(sc[0]), invA=float(sc[1]), invM=float(sc[2]),
             sX=float(sc[3]), sA=float(sc[4]), sM=float(sc[5]))
    d["ew"] = np.frombuffer(raw[L.EW_OFF:L.EW_OFF + 4096].tobytes(), dtype="<f4")
    d["mw"] = np.frombuffer(raw[L.MW_OFF:L.MW_OFF + 1024].tobytes(), dtype="<f4")
    d["alpha"] = np.frombuffer(raw[L.ALPHA_OFF:L.ALPHA_OFF + 1024].tobytes(), dtype="<f4")
    rsv = np.frombuffer(raw[L.RS_OFF:L.RS_OFF + 1024].tobytes(), dtype="<f4").astype(np.float16)
    tab = np.frombuffer(raw[L.PAD_OFF:L.PAD_OFF + 1024].tobytes(), dtype="<f4")
    return codes, d, rsv, tab


def run(raw, Xw, use_tab):
    codes, d, rsv, tab = decode_pack(raw)
    inv_tab = tab.astype(np.float64) if use_tab else np.full(256, d["invX"], dtype=np.float64)
    qx = np.clip(np.rint(Xw.astype(np.float64) * inv_tab[None, :]), -127.0, 127.0)
    acc = qx @ codes["we"].astype(np.float64).T
    v = np.float32(acc.astype(np.float32) * d["ew"][None, :])
    ha = L.act_half32(L.narrow_half(v))
    qa = np.clip(np.rint((ha * d["invA"]).astype(np.float64)), -127.0, 127.0).astype(np.int64)
    qa_f = qa.astype(np.float64)
    mreal = np.empty((NW, 256), dtype=np.float32)
    for g in range(L.MID_GROUPS):
        sl = slice(32 * g, 32 * g + 32)
        r = qa_f[:, 128 * g:128 * g + 128] @ codes["wm"][sl].astype(np.float64).T
        mreal[:, sl] = r.astype(np.float32) * d["mw"][sl][None, :]
    qm = np.clip(np.rint((mreal * np.float32(d["invM"])).astype(np.float64)), -127.0, 127.0).astype(np.int64)
    acc3 = qm @ codes["wc"].astype(np.float64).T
    real = acc3.astype(np.float32) * d["alpha"][None, :]
    seed = L.narrow_half(Xw * rsv[None, :])
    y = np.float32(real + seed).astype(np.float16)
    st = dict(qx_sat=int((np.abs(qx) >= 127).sum()), qx_n=int(qx.size),
              qa_sat=int((np.abs(qa) >= 127).sum()), qa_n=int(qa.size),
              qm_sat=int((np.abs(qm) >= 127).sum()), qm_n=int(qm.size),
              qx_zero=int((qx == 0).sum()), qa_zero=int((qa == 0).sum()), qm_zero=int((qm == 0).sum()),
              x_dequant_max=float(np.abs(np.where(valid[:, None], np.float32(qx) * d["sX"] * (tab if use_tab else 1.0), 0)).max()))
    return dict(y=y.astype(np.float32), qx=qx, qa=qa, qm=qm, mreal=mreal, a_dequant=(qa * d["sA"]).astype(np.float32),
                st=st, d=d)


arms = {"old": raw_old, "A_mse": PACKS["A_mse"]["raw"], "B_absmax": PACKS["B_absmax"]["raw"]}
use_tab = {"old": False, "A_mse": True, "B_absmax": True}
agg = {k: dict(se=0.0, seA=0.0, seM=0.0, max_abs=0.0, per={}) for k in arms}
ref = dict(seY=0.0, seA=0.0, seM=0.0, n=0)
for name, td in FRAMES:
    Xc = L.load_x_canonical(td / "dump_block14_out.bin")
    Y = np.fromfile(td / "dump_swin_15_Y.bin", dtype="<f2").reshape(NW, 256).astype(np.float32)
    Xw = np.zeros((NW, 256), dtype=np.float32)
    Xw[valid] = Xc[idx[valid]]
    # FP16 reference A/M for stage attribution
    Araw = (Xw.astype(np.float64) @ We.astype(np.float64).T).astype(np.float32)
    Aref = L.narrow_half(Araw)
    Aact = L.act_half32(Aref)
    Mref = np.empty((NW, 256), dtype=np.float32)
    for g in range(L.MID_GROUPS):
        sl = slice(32 * g, 32 * g + 32)
        Mref[:, sl] = (Aact[:, 128 * g:128 * g + 128].astype(np.float64) @ Wm[sl].astype(np.float64).T).astype(np.float32)
    vY = Y[valid].astype(np.float64)
    ref["seY"] += float(np.sum(vY * vY)); ref["n"] += vY.size
    ref["seA"] += float(np.sum(Aact[valid].astype(np.float64) ** 2))
    ref["seM"] += float(np.sum(Mref[valid].astype(np.float64) ** 2))
    for k in arms:
        r = run(arms[k], Xw, use_tab[k])
        dY = r["y"][valid].astype(np.float64) - vY
        dA = (r["a_dequant"][valid].astype(np.float64) - Aact[valid].astype(np.float64))
        dM = (r["mreal"][valid].astype(np.float64) - Mref[valid].astype(np.float64))
        agg[k]["se"] += float(np.sum(dY * dY))
        agg[k]["seA"] += float(np.sum(dA * dA))
        agg[k]["seM"] += float(np.sum(dM * dM))
        agg[k]["max_abs"] = max(agg[k]["max_abs"], float(np.abs(dY).max()))
        agg[k]["per"][name] = dict(rel_l2=float(np.sqrt(np.mean(dY ** 2) / np.mean(vY ** 2))),
                                   rms=float(np.sqrt(np.mean(dY ** 2))),
                                   max_abs=float(np.abs(dY).max()),
                                   stage_rel_l2_A=float(np.sqrt(np.mean(dA ** 2) / np.mean(Aact[valid].astype(np.float64) ** 2))),
                                   stage_rel_l2_M=float(np.sqrt(np.mean(dM ** 2) / np.mean(Mref[valid].astype(np.float64) ** 2))),
                                   **r["st"])
    print(name, {k: round(agg[k]["per"][name]["rel_l2"], 6) for k in arms}, flush=True)

summary = {}
for k in arms:
    summary[k] = dict(
        pooled_rel_l2=float(np.sqrt(agg[k]["se"] / ref["seY"])),
        pooled_stage_A=float(np.sqrt(agg[k]["seA"] / ref["seA"])),
        pooled_stage_M=float(np.sqrt(agg[k]["seM"] / ref["seM"])),
        max_abs=agg[k]["max_abs"], per_frame=agg[k]["per"])
summary["improvement"] = dict(
    A_vs_old=1 - summary["A_mse"]["pooled_rel_l2"] / summary["old"]["pooled_rel_l2"],
    B_vs_old=1 - summary["B_absmax"]["pooled_rel_l2"] / summary["old"]["pooled_rel_l2"])
(L.W / "evidence/08-quality.json").write_text(json.dumps(
    dict(summary=summary, packs={k: dict(file=str(v["file"]), sha256=v["sha256"], bytes=v["bytes"],
                                         sX=v["sX"], sA=v["sA"], sM=v["sM"]) for k, v in PACKS.items()},
         diagonal=dict(values=sorted(float(x) for x in np.unique(diag)),
                       counts={str(float(x)): int((diag == x).sum()) for x in np.unique(diag)}),
         sWe=dict(min=float(sWe.min()), max=float(sWe.max())),
         sWm=dict(min=float(sWm.min()), max=float(sWm.max())),
         sWc=dict(min=float(sWc.min()), max=float(sWc.max()))), indent=1), encoding="utf-8")
print(json.dumps({k: {kk: vv for kk, vv in summary[k].items() if kk != "per_frame"} for k in summary}, indent=1))
