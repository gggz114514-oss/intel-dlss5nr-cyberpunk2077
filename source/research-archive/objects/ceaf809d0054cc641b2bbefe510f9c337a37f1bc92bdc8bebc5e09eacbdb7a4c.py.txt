"""S07 error budget: where does the split16 b23 l0/l1 error come from?

  deltaA(chain) = (Fi - F16) @ Wc          -- upstream expand/activation error
  deltaA(quant) = (qF/sF - Fi) @ Wc        -- contract-input quantisation only
  deltaA(Wc)    = Fi @ (Wcq*wscL - Wc)     -- contract weight quantisation only
  deltaY        = deltaA                   -- the seed is exact by construction
"""
import json
import sys
from pathlib import Path

import numpy as np

B = Path("@NR_WORKSPACE@")
W = B / "work/dsh-fullnet-static-20261029"
sys.path.insert(0, str(W / "tools"))
sys.path.insert(0, str(W / "S07-split16/tools"))
sys.path.insert(0, str(W / "S07-split16/tools"))
from calibrate import Chain, TEACHERS  # noqa: E402
import model as M  # noqa: E402


def main():
    ch = Chain()
    rep = {}
    arms = {"old": dict(ch.old)}
    cal = json.loads((W / "S07-split16/CALIBRATION.json").read_text()) if (W / "S07-split16/CALIBRATION.json").exists() else None
    if cal:
        arms["new_scalar"] = cal["new"]["scales"]
    frames = []
    for fr in ch.frames:
        row = {"frame": fr["tag"]}
        for name, s in arms.items():
            F, A, Y, qx, qa, qf = ch.forward(fr, s)
            F16 = M.f16(fr["X"] @ ch.pk.W1.T)
            # fp16 reference chain, same code path as model.fp16_forward
            F16, Y16 = M.fp16_forward(ch.pk, fr["X"])
            A16 = F16 @ ch.pk.Wc.T
            nF = np.linalg.norm(F16[:1056])
            nA = np.linalg.norm(A16[:1056])
            nY = np.linalg.norm(fr["Y"][:1056])
            d_chain = (F[:1056] - F16[:1056]) @ ch.pk.Wc.T
            d_quant = ((qf[:1056] * s["sF"]) - F[:1056]) @ ch.pk.Wc.T
            d_w = F[:1056] @ (ch.pk.Wcq_m * (ch.pk.wscL * (s["sF"] / ch.old["sF"]))[:, None] - ch.pk.Wc).T
            row[name] = {
                "F_rel_l2": M.rel_l2(F, fr["F"]),
                "Y_rel_l2": M.rel_l2(Y, fr["Y"]),
                "Y_abs_rms": float(np.sqrt(np.mean((Y[:1056] - fr["Y"][:1056]) ** 2))),
                "deltaA_chain_abs": float(np.linalg.norm(d_chain) / nY),
                "deltaA_quantF_abs": float(np.linalg.norm(d_quant) / nY),
                "deltaA_weight_abs": float(np.linalg.norm(d_w) / nY),
                "A_over_Y": float(nA / nY),
                "F_quant_only": float(np.linalg.norm(qf[:1056] * s["sF"] - F[:1056]) / nF),
                "F_chain_only": float(np.linalg.norm(F[:1056] - F16[:1056]) / nF),
            }
        frames.append(row)
        print(json.dumps(row, indent=1), flush=True)
    rep["frames"] = frames
    rep["note"] = ("deltaY equals deltaA because the residual seed X*rs is exact; the three terms are "
                   "measured separately on the same teacher F, so they do not add exactly.")
    (W / "S07-split16/BUDGET.json").write_text(json.dumps(rep, indent=1))
    print("wrote BUDGET.json")


if __name__ == "__main__":
    main()
