"""Resolve the small plan properly: PASS <row> "<name>" "<spv>" ... with over-escaped separators."""
import hashlib, re, json, sys
from collections import Counter
from pathlib import Path
sys.path.insert(0, str(Path(__file__).parent))
import COMMON as K

SMALL = Path('@NR_WORKSPACE@/work/vk102-small-310-9-16-20261019')
plan = SMALL / 'package' / 'manifest' / 'port_network_runtime.txt'
raw = plan.read_text(encoding='utf-8-sig', errors='replace')

rows = []
for line in raw.splitlines():
    if not line.startswith('PASS '):
        continue
    parts = line.split('"')
    if len(parts) < 4:
        continue
    name, path = parts[1].strip(), parts[3].strip()
    fixed = re.sub(r'\\{2,}', r'\\', path)
    rows.append((name, fixed))
print('PASS lines parsed:', len(rows), '| first:', rows[0][0], '->', rows[0][1][:70])

small = {}
for name, path in rows:
    f = Path(path)
    if f.exists():
        small.setdefault(path, (hashlib.sha256(f.read_bytes()).hexdigest(), 0))
        sha, n = small[path]
        small[path] = (sha, n + 1)
print('resolved files: %d of %d distinct paths' % (len(small), len({p for _, p in rows})))

base = {}
for r in K.rows():
    base.setdefault(r['spv_sha256'], set()).add(r['family'])
opt = K.read_json(K.C / 'backend-merged' / 'MANIFEST.json')
opt_by_sha = {}
for r in opt['rows']:
    opt_by_sha.setdefault(r['spv_sha256'], r['family'])

shas = Counter({sha: n for sha, n in small.values()})
hit = sum(n for s, n in shas.items() if s in base)
print('\nsmall PASS lines whose shader sha is also in the 232-row baseline: %d / %d' % (hit, len(rows)))
for s, n in shas.most_common(18):
    fam = ','.join(sorted(base.get(s, []))) or 'NOT-IN-BASELINE'
    print('  %s %3d PASS  %s' % (s[:12], n, fam))
K.write_json(K.C / 'evidence' / 'SMALL-TRANSFER.json',
             dict(small_passes=len(rows), resolved=len(small), baseline_hit=hit,
                  shas=dict(shas), families={k: sorted(v) for k, v in base.items()}))
