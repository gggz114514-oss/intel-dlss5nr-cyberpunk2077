_spv_cache = {}


def spv_sha(p):
    if p not in _spv_cache:
        _spv_cache[p] = planlib.sha256_file(p) if Path(p).is_file() else None
    return _spv_cache[p]


TOL_BUFFER = __import__("re").compile(r'^BUFFER\s+"?([^"\s]+)"?\s+(\d+)\s+"?([^"\s]*)"?\s*$')
TOL_PASS = __import__("re").compile(r'^PASS\s+(\d+)\s+"?([^"\s]+)"?\s+"?([^"\s]+)"?')
TOL_BIND = __import__("re").compile(r'^BIND\s+(\d+)\s+(\d+)\s+(\d+)\s+(\d+)\s*$')
TOL_OUT = __import__("re").compile(r'^OUTPUT\s+(\d+)\s+(\S+)\s+"?([^"\s]+)"?\s*$')


def tolerant_summary(plan_path: Path, why: str) -> dict:
    """Fallback for plans planlib rejects (e.g. unquoted SPV paths)."""
    import hashlib
    import re
    lines = plan_path.read_text(encoding="utf-8").splitlines()
    buffers, passes, outputs, history = [], [], [], []
    for line in lines:
        if not line or line.startswith("#"):
            continue
        m = TOL_BUFFER.match(line)
        if m:
            buffers.append({"idx": len(buffers), "name": m.group(1), "bytes": int(m.group(2)),
                            "file": m.group(3)})
            continue
        m = TOL_PASS.match(line)
        if m:
            passes.append({"ordinal": len(passes), "group": int(m.group(1)),
                           "name": m.group(2), "spv": m.group(3), "push": None})
            continue
        if line.startswith("PUSH ") and passes:
            passes[-1]["push"] = line.split()[1:]
            continue
        m = TOL_OUT.match(line)
        if m:
            outputs.append({"buffer": int(m.group(1)), "format": m.group(2), "file": m.group(3)})
            continue
        if line.startswith("HISTORY "):
            history.append(line.split()[1:])
    spvs = sorted({p["spv"] for p in passes})
    spv_rows = [{"path": s.replace("\\", "/"), "sha256": spv_sha(s)} for s in spvs]
    file_rows = [{"name": b["name"], "bytes": b["bytes"], "file": b["file"].replace("\\", "/"),
                  "sha256": planlib.sha256_file(b["file"]) if Path(b["file"]).is_file() else None}
                 for b in buffers if b["file"] != "-"]
    h = hashlib.sha256()
    for r in spv_rows:
        h.update((r["sha256"] or "MISSING").encode())
    for r in sorted(file_rows, key=lambda r: r["name"]):
        h.update((r["sha256"] or "MISSING").encode())
    name_by_idx = {b["idx"]: b for b in buffers}
    return {
        "plan": str(plan_path).replace("\\", "/"),
        "plan_sha256_raw": planlib.sha256_file(plan_path),
        "plan_sha256_text": None,
        "parser": "tolerant-regex",
        "parser_note": why,
        "header": None,
        "buffers": len(buffers),
        "dispatches": len(passes),
        "outputs": [{"buffer": o["buffer"], "name": name_by_idx[o["buffer"]]["name"],
                     "bytes": name_by_idx[o["buffer"]]["bytes"], "format": o["format"],
                     "file": o["file"]} for o in outputs],
        "dispatch0_push": passes[0]["push"] if passes else None,
        "history_lines": [" ".join(hh) for hh in history],
        "distinct_spv": len(spv_rows),
        "spv_set": spv_rows,
        "file_buffers": file_rows,
        "spv_pack_identity_sha256": h.hexdigest(),
        "validation_problems": [],
        "capture_bindings": {b["name"]: b["file"].replace("\\", "/") for b in buffers
                             if b["file"].replace("\\", "/").lower().endswith(("pre.bin", "post.bin"))},
    }


def arm_summary(plan_path: Path) -> dict:
    try:
        pl = planlib.Plan(plan_path)
    except planlib.PlanError as exc:
        return tolerant_summary(plan_path, f"planlib rejected: {exc}")
    problems = pl.validate()
    spvs = sorted({p.spv for p in pl.passes})
    spv_rows = [{"path": s.replace("\\", "/"), "sha256": spv_sha(s)} for s in spvs]
    files = [b for b in pl.buffers if b.file != "-"]
    file_rows = [{"name": b.name, "bytes": b.bytes, "file": b.file.replace("\\", "/"),
                  "sha256": planlib.sha256_file(b.file) if Path(b.file).is_file() else None}
                 for b in files]
    # hash the whole SPV/pack identity as one digest
    import hashlib
    h = hashlib.sha256()
    for r in spv_rows:
        h.update((r["sha256"] or "MISSING").encode())
    for r in sorted(file_rows, key=lambda r: r["name"]):
        h.update((r["sha256"] or "MISSING").encode())
    return {
        "plan": str(plan_path).replace("\\", "/"),
        "plan_sha256_raw": planlib.sha256_file(plan_path),
        "plan_sha256_text": pl.sha256(),
        "parser": "planlib",
        "parser_note": None,
        "header": list(pl.header),
        "buffers": len(pl.buffers),
        "dispatches": len(pl.passes),
        "outputs": [{"buffer": o[0], "name": pl.buffers[o[0]].name, "bytes": pl.buffers[o[0]].bytes,
                     "format": o[1], "file": o[2]} for o in pl.outputs],
        "dispatch0_push": pl.passes[0].push if pl.passes else None,
        "history_lines": [" ".join(h[0]) for h in pl.history],
        "distinct_spv": len(spv_rows),
        "spv_set": spv_rows,
        "file_buffers": file_rows,
        "spv_pack_identity_sha256": h.hexdigest(),
        "validation_problems": problems,
        "capture_bindings": {
            b.name: b.file.replace("\\", "/") for b in pl.buffers
            if b.file.replace("\\", "/").lower().endswith(("pre.bin", "post.bin"))
        },
    }
