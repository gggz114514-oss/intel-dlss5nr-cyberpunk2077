import json,pathlib,statistics,math,hashlib
A=pathlib.Path(__file__).parent;D=A/'controlled-gpu-20261025'
read=lambda p:json.loads(p.read_text(encoding='utf-8-sig'))
base=read(A.parent/'zcode-original-combo1-diagnostic-repeats-20261025/stalls-fixed-1/pipeline-statistics.json')
bs={p['index']:{s['name']:s['value'] for s in p['executables'][0]['statistics']} for p in base['pipelines']}
out={'scope':'Main auditor raw-receipt analysis; CI unit=process pair, all samples retained','candidates':{}}
for key in ['f05-c256-qkv6','f08-group1']:
 c={'frame_sha256':{},'resource_changes':[],'ordinary':[]}
 for mode in ['fixed','fb2']:
  c['frame_sha256'][mode]={f.name:hashlib.sha256(f.read_bytes()).hexdigest() for f in sorted((D/(key+'-'+mode)).glob('*.bin'))}
 for p in read(D/(key+'-resources')/'pipeline-statistics.json')['pipelines']:
  vals={s['name']:s['value'] for s in p['executables'][0]['statistics']}
  if vals!=bs[p['index']]:c['resource_changes'].append({'pipeline':p['index'],'parent':bs[p['index']],'candidate':vals})
 for pair in range(1,4):
  arms={}
  for arm in ['parent','candidate']:
   files=list(D.glob(key+f'-ordinary-{pair}-{arm}*/port_network_receipt.json'))
   if len(files)==1:
    r=read(files[0]);arms[arm]={'receipt':str(files[0]),'samples':r['gpu_samples_ms'],'mean_ms':statistics.mean(r['gpu_samples_ms']),'normal_close':r['normal_close'],'nan':r['nan'],'inf':r['inf'],'feedback':r['feedback'],'per_pass_timing':r['per_pass_timing']}
  if len(arms)==2:c['ordinary'].append({'pair':pair,'arms':arms,'saving_ms':arms['parent']['mean_ms']-arms['candidate']['mean_ms']})
 if len(c['ordinary'])==3:
  ds=[r['saving_ms'] for r in c['ordinary']];mean=statistics.mean(ds);half=4.302652729911275*statistics.stdev(ds)/math.sqrt(3);ci=[mean-half,mean+half]
  c['performance']={'pair_savings_ms':ds,'mean_saving_ms':mean,'t95_process_pairs':ci,'decision':'FASTER' if ci[0]>0 else 'SLOWER' if ci[1]<0 else 'INDISTINCT','timing_caveat':'First pair execution interrupted by wrapper gate misclassification; retain all samples and elapsed-order evidence. Only three independent process pairs.'}
 out['candidates'][key]=c
(A/'MAIN-AUDITOR-CONTROLLED-RESULTS-20261025.json').write_text(json.dumps(out,indent=2),encoding='utf-8')
