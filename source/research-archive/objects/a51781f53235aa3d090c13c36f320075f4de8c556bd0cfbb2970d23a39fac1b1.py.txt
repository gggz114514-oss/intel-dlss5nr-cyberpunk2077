"""Compare the three arms using each run's own gpu_mean_ms (30 reps per run, 6 runs per arm)."""
import json, sys
from collections import defaultdict
from pathlib import Path
sys.path.insert(0, str(Path(__file__).parent))
import COMMON as K

root = K.C / 'queue' / 'vk102-compare' / 'run'
by_arm = defaultdict(list)
info = {}
for run in sorted(root.glob('perm-*')):
    arm = run.name.split('-')[-1]
    pair = int(run.name.split('-')[1])
    rc = K.read_json(run / 'port_network_receipt.json')
    by_arm[arm].append((pair, rc['gpu_mean_ms']))
    info[arm] = dict(passes=rc.get('gpu_passes'), nan=rc.get('nan'), inf=rc.get('inf'))
print('runs per arm:', {k: len(v) for k, v in by_arm.items()}, 'passes:', {k: v['passes'] for k, v in info.items()})

T = 2.570581835636305
out = {}
names = {'A': 'handoff baseline', 'C': 'current head (all optimisations)', 'V': 'vk102'}
for x, y in (('A', 'C'), ('V', 'A'), ('V', 'C')):
    xs = sorted(by_arm[x])
    ys = sorted(by_arm[y])
    n = min(len(xs), len(ys))
    diffs = [xs[i][1] - ys[i][1] for i in range(n)]
    m = sum(diffs) / n
    sd = (sum((v - m) ** 2 for v in diffs) / (n - 1)) ** 0.5
    half = T * sd / (n ** 0.5)
    same = all(d > 0 for d in diffs) or all(d < 0 for d in diffs)
    mx = sum(v for _, v in xs[:n]) / n
    my = sum(v for _, v in ys[:n]) / n
    status = 'FASTER_95' if (m - half > 0 and same) else ('SLOWER_95' if (m + half < 0 and same) else 'INDISTINCT')
    out['%s_minus_%s' % (x, y)] = dict(saving_ms=m, CI95_ms=[m - half, m + half], groups=n, status=status,
                                       mean_x_ms=mx, mean_y_ms=my, label='%s vs %s' % (names[x], names[y]))
    print('%-9s %-4s mean %7.3f ms   %-4s mean %7.3f ms   diff %+7.4f  CI[%+.4f, %+.4f]  %s'
          % (x, m and names[x][:9], mx, y, my, m, m - half, m + half, status))
pct = 100 * out['V_minus_C']['saving_ms'] / out['V_minus_C']['mean_y_ms']
K.write_json(K.C / 'evidence' / 'VK102-COMPARISON.json',
             dict(comparisons=out, arm_info=info,
                  head_vs_vk102_percent=pct,
                  method='same runner/fixture/pool/warm1/reps30; six balanced permutations of '
                         'vk102/handoff/head; all samples retained; t(df=5) over permutation means; '
                         'vk102 is a different model version so it is compared on timing only'))
print('\nhead vs vk102: %.3f ms faster = %.2f%%' % (out['V_minus_C']['saving_ms'], pct))
