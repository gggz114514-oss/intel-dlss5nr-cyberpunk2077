"""Three-arm same-batch comparison: vk102 (V), campaign handoff baseline (A), current head (C).

vk102 is a different model version, so its output cannot be gated against our gold frames; its arm
only has to run clean and agree with itself across samples (the two current arms already have their
gold proofs from the gates stage). Same runner, fixture, pool, warm 1, reps 30, feedback, six
balanced permutations, all samples retained, t(df=5) over permutation means.
"""
import json, sys
from pathlib import Path
sys.path.insert(0, str(Path(__file__).parent))
import COMMON as K
import STAGE
import ANALYZE

PERM = [('V', 'A', 'C'), ('V', 'C', 'A'), ('A', 'V', 'C'), ('A', 'C', 'V'), ('C', 'V', 'A'), ('C', 'A', 'V')]
VK = Path('@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/'
          'vk102-current-paired-20261026/vk102-path-normalized.txt')
cur = K.read_json(K.C / 'CURRENT.json')
arms = {'V': (VK, K.sha(VK)), 'A': (K.BASELINE, K.sha(K.BASELINE)),
        'C': (Path(cur['best_plan']['path']), cur['best_plan']['sha256'])}
out = K.C / 'queue' / 'vk102-compare'
out.mkdir(parents=True, exist_ok=True)

jobs = []
for pi, order in enumerate(PERM):
    for aid in order:
        p, s = arms[aid]
        jobs.append(K.ordinary_job('perm-%d-%s' % (pi, aid), p, s,
                                   out / 'run' / ('perm-%d-%s' % (pi, aid)), aid, pi))
spec = dict(stage='ordinary', owner='LUNA-CONTINUOUS-VK102',
            authorization='Main authorized the three-arm vk102 comparison only: vk102, handoff '
                          'baseline and current head, same batch. vk102 is a different model version '
                          'so only its timing is compared; the two current arms carry their gold proofs.',
            output_root=str(out), identities=[str(v[0]) for v in arms.values()] + [str(K.RUNNER)],
            jobs=jobs, rules='three-arm same-batch; all samples retained')
K.write_json(out / 'spec-ordinary.json', spec)
manifest, ps1 = STAGE.prepare(spec)
STAGE.run(manifest, ps1)
s = STAGE.summarize(spec['output_root'])
K.write_json(out / 'STAGE-SUMMARY.json', s)
print('ordinary', s['status'])
if s['status'] != 'COMPLETED':
    print([(j['id'], j['status'], j['frame']) for j in s['jobs']])
    sys.exit(1)
rep = ANALYZE.report(out / 'LUNA-ORDINARY.json',
                     [dict(arms=('V', 'A')), dict(arms=('V', 'C')), dict(arms=('A', 'C'))],
                     out / 'TIMING.json')
for c in rep['comparisons']:
    print('%s: %s saving=%.6f CI95=[%.6f, %.6f] groups=%d' % (
        c['name'], c['status'], c['saving_ms'], c['CI95_ms'][0], c['CI95_ms'][1], c['groups']))
