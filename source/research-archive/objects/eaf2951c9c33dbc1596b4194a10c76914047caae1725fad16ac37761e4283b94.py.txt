def sha(rel: str) -> str | None:
    p = HERE / rel
    return planlib.sha256_file(p) if p.is_file() else None


def job(name: str, plan_rel: str, out: str, dump: list[str], purpose: str, status: str) -> dict:
    return {
        "id": name,
        "frame": 34189,
        "base_plan": str(CAND).replace("\\", "/"),
        "base_plan_sha256": planlib.sha256_file(CAND),
        "plan": str(HERE / plan_rel).replace("\\", "/"),
        "plan_sha256": sha(plan_rel),
        "out_dir": str(HERE / "runs" / out).replace("\\", "/"),
        "dump_files": dump,
        "purpose": purpose,
        "status": status,
    }
