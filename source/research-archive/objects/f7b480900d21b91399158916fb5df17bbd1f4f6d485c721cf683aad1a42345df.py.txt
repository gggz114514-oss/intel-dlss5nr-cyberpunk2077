"""n31: SPIR-V instruction-mix delta parent -> candidate (CPU only).

Confirms the edit is exactly what the source diff says: OpSelect drops, and no
float arithmetic / NoContraction / memory opcode changes (CONTRACT §2/§3).
"""

from collections import Counter

CARD = Path(__file__).resolve().parents[1]

         42: 'OpConstantTrue', 43: 'OpConstantFalse', 44: 'OpConstantComposite',
         55: 'OpFunctionParameter', 59: 'OpVariable', 61: 'OpLoad', 62: 'OpStore',
         65: 'OpAccessChain', 71: 'OpDecorate', 80: 'OpCompositeExtract',
         81: 'OpCompositeInsert', 86: 'OpBitcast', 87: 'OpSNegate', 124: 'OpBitcast2',
         126: 'OpKill', 128: 'OpIAdd', 129: 'OpFAdd', 130: 'OpISub', 131: 'OpFSub',
         132: 'OpIMul', 133: 'OpFMul', 134: 'OpUDiv', 135: 'OpSDiv', 136: 'OpFDiv',
         137: 'OpUMod', 138: 'OpSRem', 139: 'OpSMod', 141: 'OpFRem',
         142: 'OpFMod', 144: 'OpVectorShuffle', 169: 'OpSelect', 170: 'OpIEqual',
         171: 'OpINotEqual', 172: 'OpUGreaterThan', 173: 'OpSGreaterThan',
         177: 'OpFOrdGreaterThan', 187: 'OpFOrdLessThan', 190: 'OpFOrdEqual',
         191: 'OpFUnordNotEqual', 194: 'OpShiftRightLogical', 195: 'OpShiftRightArithmetic',
         196: 'OpShiftLeftLogical', 197: 'OpBitwiseOr', 198: 'OpBitwiseXor',
         199: 'OpBitwiseAnd', 205: 'OpFNegate', 227: 'OpSelect2', 245: 'OpPhi',

         113: 'OpSConvert', 114: 'OpFConvert', 115: 'OpQuantizeToF16',
         57: 'OpFunction', 56: 'OpFunctionEnd', 54: 'OpFunctionCall',
         19: 'OpTypeVoid', 23: 'OpTypeVector', 21: 'OpTypeInt',
         22: 'OpTypeFloat', 24: 'OpTypeMatrix', 25: 'OpTypeImage'}
FLOAT_OPS = (129, 131, 133, 136, 141, 142, 205, 177, 187, 190, 191, 115)


def opcodes(path):
    data = Path(path).read_bytes()
    words = struct.unpack('<%dI' % (len(data) // 4), data)
    assert words[0] == 0x07230203, hex(words[0])
    i, out = 5, Counter()
    while i < len(words):
        op = words[i] & 0xFFFF
        wc = words[i] >> 16
        if wc == 0:
            break
        out[op] += 1
        i += wc
    return out


def main():
    parent = CARD.parents[1] / 'cards/n29-f01-alu-chain/candidate.spv'
    cand = CARD / 'candidate.spv'
    a, b = opcodes(parent), opcodes(cand)
    rows = []
    for op in sorted(set(a) | set(b)):
        if a[op] != b[op]:
            rows.append(dict(opcode=op, name=NAMES.get(op, 'op%d' % op),
                             parent=a[op], candidate=b[op], delta=b[op] - a[op]))
    fa = sum(a[o] for o in FLOAT_OPS)
    fb = sum(b[o] for o in FLOAT_OPS)
    res = dict(parent=str(parent), candidate=str(cand),
               parent_instructions=sum(a.values()), candidate_instructions=sum(b.values()),
               changed=rows,
               float_arith_parent=fa, float_arith_candidate=fb, float_arith_equal=(fa == fb),
               opselect_parent=a[169], opselect_candidate=b[169])
    (CARD / 'tools/spv-delta.json').write_text(json.dumps(res, indent=1), encoding='utf-8')
    print('parent inst %d -> candidate %d (%+d)'
          % (res['parent_instructions'], res['candidate_instructions'],
             res['candidate_instructions'] - res['parent_instructions']))
    for r in rows:
        print('  %-24s %5d -> %5d  (%+d)' % (r['name'], r['parent'], r['candidate'], r['delta']))
    print('float arithmetic opcodes: %d -> %d  equal=%s' % (fa, fb, fa == fb))
    print('OpSelect: %d -> %d (%+d)' % (a[169], b[169], b[169] - a[169]))


if __name__ == '__main__':
    sys.exit(main())
