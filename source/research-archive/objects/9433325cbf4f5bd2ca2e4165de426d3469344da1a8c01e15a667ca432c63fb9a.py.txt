"""n35-f18-latency: build plan.txt (replace ord 166 only) and run the CPU check.

The parent is the current chain head (accepted/n29-f01-alu-chain-20261027/plan.txt);
only the SPV path of the single PASS with baseline sha c27fa8c9... is replaced, and
the script asserts that the replacement count equals that PASS count (1).
"""
import json, sys
from pathlib import Path

CAMP = Path(__file__).resolve().parent.parent.parent
sys.path.insert(0, str(CAMP))
import COMMON as K

HERE = Path(__file__).resolve().parent
SHAS = {'c27fa8c9e2e1f9fce6d40bbbe0ee15660b86eabf84712145c499e7433f52c683'}


def main():
    base, base_sha, _ = K.current_best()
    cand = HERE / 'candidate.spv'
    plan = HERE / 'plan.txt'
    n = K.make_plan(cand, plan, parent_plan=base, spv_shas=SHAS)
    rows = [r for r in K.rows() if r['spv_sha256'] in SHAS]
    out = dict(parent_plan=dict(path=str(base), sha256=base_sha),
               plan=dict(path=str(plan), sha256=K.sha(plan)),
               replaced=n, expected=len(rows), ordinals=[r['ordinal'] for r in rows],
               names=[r['name'] for r in rows])
    # structural check: only that one PASS line differs
    import difflib
    a = Path(base).read_text(encoding='utf-8-sig').splitlines()
    b = plan.read_text(encoding='utf-8-sig').splitlines()
    diff = [l for l in difflib.unified_diff(a, b, lineterm='', n=0) if l[:1] in '+-' and not l.startswith(('+++', '---'))]
    out['diff_lines'] = len(diff)
    out['diff_sample'] = diff[:4]
    try:
        cc = K.cpu_check(plan)
    except Exception as ex:
        cc = 'EXCEPTION: %s' % ex
    out['cpu_check'] = cc
    (HERE / 'PLAN-BUILD.json').write_text(json.dumps(out, indent=1, ensure_ascii=False), encoding='utf-8')
    print(json.dumps({k: v for k, v in out.items() if k != 'cpu_check'}, indent=1, ensure_ascii=False))
    print('---- cpu_check ----')
    print(cc)


if __name__ == '__main__':
    main()
