import sys
from pathlib import Path
import numpy as np

sys.path.insert(0, str(Path(__file__).parent))
import c32blk as M
from validate import bk, PUSH, ptr

B = Path(r"@NR_WORKSPACE@")
W = B / "work/dsh-fullnet-static-20261029"
src = W / "C00-color-baseline/runs/diag65-69"
tea = B / "work/codex-astra-fp16-int8-audit-20261024/int8-quality-first-20261029/real-int8-24/dump-fp16"
f2fix = W / "C00-integer-ref/runs/f2fixed-diag"

idx, vv = M.token_index(-1, -1)
gpu = np.fromfile(src / "c00-block67_out.bin", dtype="<f2")[idx][vv].astype(np.float64)
blk = M.Block(bk("half", 67), bk("i8", 67), bk("sc", 67), bk("aux", 67), ptr(PUSH))

cands = {
    "int8_block66": src / "c00-block66_out.bin",
    "teacher_block66": tea / "dump_block66_out.bin",
    "f2fixed_block66": f2fix / "f2fixed-block66_out.bin",
    "int8_block65": src / "c00-block65_out.bin",   # wrong size, skip
}
for name, p in cands.items():
    if not p.is_file():
        continue
    try:
        x, valid = M.load_block(p, -1, -1)
    except AssertionError as e:
        print(name, "size mismatch", e)
        continue
    y, _ = blk.forward(x, valid)
    pred = y[vv].astype(np.float64)
    print(f"{name:16s} vs gpu67 relL2={M.rel_l2(pred, gpu):.4f} corr={M.corr(pred, gpu):.4f}")

# Also: how does gpu67 compare to teacher67 / f2fixed-derived? and to c00-block66 content statistics
print("input stats: int8 absmax", np.abs(np.fromfile(src / 'c00-block66_out.bin', dtype='<f2').astype(np.float32)).max())
print("input stats: teacher absmax", np.abs(np.fromfile(tea / 'dump_block66_out.bin', dtype='<f2').astype(np.float32)).max())
print("input stats: f2fixed absmax", np.abs(np.fromfile(f2fix / 'f2fixed-block66_out.bin', dtype='<f2').astype(np.float32)).max())
