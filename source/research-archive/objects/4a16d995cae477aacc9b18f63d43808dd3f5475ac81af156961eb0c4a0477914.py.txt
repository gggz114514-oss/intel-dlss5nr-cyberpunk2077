from pathlib import Path

R=Path(__file__).resolve().parent;P=R/'f20-score-owner'
def ident(p):return dict(path=str(p),sha256=hashlib.sha256(p.read_bytes()).hexdigest())
def native(base):
 p=next((base/'captures').rglob('*38_orig.asm'))
 rows=[l for l in p.read_text().splitlines() if re.match(r'/\* \[[0-9A-F]+\]',l)]
 return dict(**ident(p),instruction_count=len(rows),counts={x:sum(x in l for l in rows) for x in ['dpas.','send.slm','send.ugm','sync.bar','mov','add','mul','sel']},scratch_instructions=[l for l in rows if re.search(r'\.ss\b',l)])
N=R/sys.argv[2]
if sys.argv[1]=='native':
 out=dict(parent=native(P),candidate=native(N),limits='Static counts include debug branches. Not time or dynamic work. Max-r is not allocated GRF. Physical peak unknown.')
 assert not out['candidate']['scratch_instructions']
 for j in json.loads((N/'LUNA-GATES.json').read_text())['jobs']:
  d=Path(j['runner_output']);assert ident(d/'frame_out.bin')['sha256']==j['expected_fixed_frame_out']
  if j['mode']=='fb2':assert ident(d/'frame_0.bin')['sha256']==j['expected_frame0']
 (N/'NATIVE-REVIEW.json').write_text(json.dumps(out,indent=2),encoding='utf-8');print(json.dumps(out,indent=2))
elif sys.argv[1]=='prepare':
 assert (N/'NATIVE-REVIEW.json').exists()
 t=json.loads((P/'LUNA-ORDINARY.json').read_text());t['output_root']=str(N)

 if pack:t['identities'].append(ident(Path(pack['path'])))
 proto=t['jobs'][0];jobs=[]
 for pair,order in enumerate([['parent','candidate'],['candidate','parent'],['parent','candidate']]):
  for arm in order:
   j=json.loads(json.dumps(proto));j['pair']=pair;j['candidate']=arm;j['id']=f'ordinary-{pair}-{arm}'
   dest=N/'run'/j['id'];plan=(P if arm=='parent' else N)/'plan.txt';j['runner_output']=str(dest);j['plan']=ident(plan)
   j['argv'][j['argv'].index('--plan')+1]=str(plan);j['argv'][j['argv'].index('--out')+1]=str(dest);jobs.append(j)
 t['jobs']=jobs;t['rules']=['Six serial ordinary jobs latest score-owner vs candidate. Three alternating pairs, warm1/reps30/feedback. All samples. No profiler. Unique lock. Error/nongold stop, no retry.']
 (N/'LUNA-ORDINARY.json').write_text(json.dumps(t,indent=2),encoding='utf-8')

 t['identities']=[v for v in t['identities'] if not v['path'].endswith('.spv')]+[ident(N/'candidate.spv')]
 if pack:t['identities'].append(ident(Path(pack['path'])))
 for j in t['jobs']:j['plan']=ident(N/'plan.txt')
 (N/'LUNA-RESOURCES.json').write_text(json.dumps(t,indent=2),encoding='utf-8')
elif sys.argv[1]=='resource':
 p=N/'run/resource-exact-four/pipeline-statistics.json';data=json.loads(p.read_text());parent=json.loads((P/'run/resource-score-owner/pipeline-statistics.json').read_text())
 assert data['pass_pipeline_indices']==parent['pass_pipeline_indices']
 def stats(x):return {s['name']:s['value'] for s in x['executables'][0]['statistics']}
 for x,y in zip(data['pipelines'],parent['pipelines']):
  assert x['index']==y['index'];assert stats(x)['Scratch Memory Size']<=stats(y)['Scratch Memory Size']
 pp=next(x for x in data['pipelines'] if x['index']==38);st=stats(pp)
 expected_shared=json.loads((N/'PROOF.json').read_text()).get('shared_bytes',25088)
 assert st['Scratch Memory Size']==0 and st['Workgroup Memory Size']==expected_shared
 out=dict(input=ident(p),pipeline=pp,all_pipeline_scratch_not_increased=True,timing_not_used=True)
 (N/'RESOURCE-REVIEW.json').write_text(json.dumps(out,indent=2),encoding='utf-8');print(N.name,st)
elif sys.argv[1]=='ordinary':
 t=json.loads((N/'LUNA-ORDINARY.json').read_text());means={};samples={}
 for j in t['jobs']:
  d=Path(j['runner_output']);r=json.loads((d/'port_network_receipt.json').read_text());assert ident(d/'frame_out.bin')['sha256']==j['expected_frame_out']

 (N/'ORDINARY-REVIEW.json').write_text(json.dumps(out,indent=2),encoding='utf-8');print(json.dumps({k:v for k,v in out.items() if k!='all_180_samples'},indent=2))
