p = CARD / 'NATIVE-PREDICTION.md'
t = p.read_text(encoding='utf-8')
if 'MEASURED — the main agent' not in t:
    p.write_text(t.rstrip() + '\n' + PRED, encoding='utf-8')
    print('appended measured section to NATIVE-PREDICTION.md')

r = CARD / 'RESUME.txt'
s = r.read_text(encoding='utf-8')
old = 'STATUS: BUILT / AWAITING-GPU  (candidate.spv + plan.txt + cpu_check done, no GPU touched)'
new = ('STATUS: GATES PASSED + NATIVE LANDED, AWAITING COUNTERS/ORDINARY.\n'
       '  The main agent ran this card on 2026-10-28T02:47Z against accepted/n29-f01-alu-chain:\n'
       '  both gold frames byte-exact in both arms (fixed 1755e9c2..., fb2 9c3e511e...), and the\n'
       '  captured ISA shows sync.bar 12 -> 11, the tail store 4 -> 1, tail mul/add 129/128 ->\n'
       '  33/35, stage9+tail 733 -> 528 instructions, 4694 -> 4484 total, max_r r127 and no spill\n'
       '  in both arms.  One cost came out worse than predicted: the tail weight read became 32\n'
       '  narrow load.ugm.d16u32 instead of 1 wide d32x16t (same bytes, +28 messages); the named\n'
       '  fix is in BUILD.json -> gpu_evidence.named_fix_for_the_weight_messages.  Details and the\n'
       '  full measured table: NATIVE-PREDICTION.md section 6.  Still missing: the two-arm\n'
       '  counters (BARRIER / ALUWR per PASS) and the ordinary 6 x 30 batch.')
if old in s and 'GATES PASSED + NATIVE LANDED' not in s:
