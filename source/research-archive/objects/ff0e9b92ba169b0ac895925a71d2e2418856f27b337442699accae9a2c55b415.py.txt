import sys
from pathlib import Path
import numpy as np

sys.path.insert(0, str(Path(__file__).parent))
import c32blk as M
from validate import bk, PUSH, ptr

B = Path(r"@NR_WORKSPACE@")
W = B / "work/dsh-fullnet-static-20261029"
src = W / "C00-color-baseline/runs/diag65-69"
tea = B / "work/codex-astra-fp16-int8-audit-20261024/int8-quality-first-20261029/real-int8-24/dump-fp16"

x, valid = M.load_block(src / "c00-block66_out.bin", -1, -1)
blk = M.Block(bk("half", 67), bk("i8", 67), bk("sc", 67), bk("aux", 67), ptr(PUSH))
y, _ = blk.forward(x, valid)
idx, vv = M.token_index(-1, -1)
gpu = np.fromfile(src / "c00-block67_out.bin", dtype="<f2")[idx][vv].astype(np.float64)
tch = np.fromfile(tea / "dump_block67_out.bin", dtype="<f2")[idx][vv].astype(np.float64)
pred = y[vv].astype(np.float64)


def cmat(a, b):
    a = a - a.mean(0)
    b = b - b.mean(0)
    a /= np.linalg.norm(a, axis=0, keepdims=True) + 1e-30
    b /= np.linalg.norm(b, axis=0, keepdims=True) + 1e-30
    return a.T @ b


C = cmat(pred, gpu)
best = C.argmax(0)
print("best pred channel for each GPU channel:", best.tolist())
print("diag corr:", np.round(np.diag(C), 3).tolist())
print("max offdiag corr per gpu ch:", np.round(C.max(0), 3).tolist())
print("scalar fit (gpu = a*pred):", np.round([float((pred[:, c] * gpu[:, c]).sum() / (pred[:, c] ** 2).sum()) for c in range(32)], 3).tolist())
T = cmat(tch, gpu)
print("teacher-vs-gpu diag:", np.round(np.diag(T), 3).tolist())
print("teacher best channel:", T.argmax(0).tolist())
