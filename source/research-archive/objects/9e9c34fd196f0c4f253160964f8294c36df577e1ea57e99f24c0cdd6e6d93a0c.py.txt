"""Score the S08 arm on frame 34189 with the project's own 24-frame metric (no GPU needed)."""
import hashlib
import json
from pathlib import Path

import numpy as np

Q = Path("@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/int8-quality-first-20261029")
D = Q / "real-int8-24"
O = D / "c64-family-static"
RUN = Path("@NR_WORKSPACE@/work/dsh-fullnet-static-20261029/gpu/runs/s08-b31-local")

jobs = json.loads((D / "JOBS.json").read_text(encoding="utf-8"))
job = next(j for j in jobs if isinstance(j, dict) and j.get("frame") == 34189 and j.get("reference"))
print("job keys", list(job.keys()))
ref = np.fromfile(job["reference"], dtype="<f4").reshape(2, 720, 1280, 4)
print("reference", job["reference"])

out = dict(frame=34189, sequence=job.get("sequence"), reference=job["reference"],
           metric="replayed-network.bin[0,:,:,:3] vs reference[0,:,:,:3], PSNR = -10*log10(MSE), peak=1",
           arms={})
for arm, path in (("old_combo_34189", O / "run-34189"), ("s08_b31_pack", RUN)):
    f = path / "replayed-network.bin"
    x = np.fromfile(f, dtype="<f4").reshape(ref.shape)
    assert np.isfinite(x).all()
    e = x[0, :, :, :3].astype(float) - ref[0, :, :, :3].astype(float)
    mse = float(np.mean(e * e))
    out["arms"][arm] = dict(mse=mse, psnr=float(-10 * np.log10(mse)),
                            mae=float(np.abs(e).mean()), max_abs=float(np.abs(e).max()),
                            sha256=hashlib.sha256(f.read_bytes()).hexdigest())
q24 = json.loads((O / "QUALITY-24.json").read_text(encoding="utf-8"))
row = next(r for r in q24["rows"] if r["frame"] == 34189)
out["QUALITY24_row_34189"] = {k: (v["psnr"] if isinstance(v, dict) else v) for k, v in row.items()}
out["matches_project_baseline"] = abs(out["arms"]["old_combo_34189"]["psnr"] - row["static_c64"]["psnr"]) < 1e-9
a = out["arms"]["old_combo_34189"]["psnr"]
b = out["arms"]["s08_b31_pack"]["psnr"]
S = Path("@NR_WORKSPACE@/work/dsh-fullnet-static-20261029/S08-vit")
out["delta_db"] = b - a
print(json.dumps(out, indent=1))
(S / "GPU-ARM-SCORE-34189.json").write_text(json.dumps(out, indent=1) + "\n", encoding="utf-8")
