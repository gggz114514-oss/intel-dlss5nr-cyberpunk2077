function Save-Json($path,$value){$json=$value|ConvertTo-Json -Depth 32;[IO.File]::WriteAllText($path,$json,[Text.UTF8Encoding]::new($false))}
function Hash-Bytes([byte[]]$bytes){$sha=[Security.Cryptography.SHA256]::Create();try{([BitConverter]::ToString($sha.ComputeHash($bytes))).Replace('-','').ToLowerInvariant()}finally{$sha.Dispose()}}

function Get-Frames([string]$outDir){$fs=@();if(Test-Path -LiteralPath $outDir){Get-ChildItem -LiteralPath $outDir -File -Filter 'frame_*.bin'|Sort-Object Name|ForEach-Object {$fs+=,[ordered]@{path=$_.FullName;sha256=(Hash-File $_.FullName)}}};,$fs}
