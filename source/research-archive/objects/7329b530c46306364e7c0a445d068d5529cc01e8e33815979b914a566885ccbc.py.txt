from pathlib import Path
import numpy as np,json
D=Path(__file__).resolve().parent/'optional-mma';results=[]
for mode,files in [('fixed',['frame_out.bin']),('fb8',[f'frame_{i}.bin' for i in range(8)])]:
 for file in files:
  a=np.fromfile(D/'run'/('parent-'+mode)/file,dtype='<f4').astype(np.float64)
  b=np.fromfile(D/'run'/('candidate-'+mode)/file,dtype='<f4').astype(np.float64)
  assert a.size==b.size and np.isfinite(a).all() and np.isfinite(b).all()
  delta=b-a;rmse=float(np.sqrt(np.mean(delta*delta)))
  results.append(dict(mode=mode,file=file,elements=a.size,different=int(np.count_nonzero(delta)),max_abs=float(np.max(np.abs(delta))),rmse=rmse,psnr_peak1=float(-20*np.log10(rmse)) if rmse else None,parent_absmax=float(np.max(np.abs(a))),candidate_absmax=float(np.max(np.abs(b)))))
(D/'SCREEN-REVIEW.json').write_text(json.dumps(dict(results=results,limits='Raw full output float buffer, no application crop or visual/temporal acceptance. Eight repeated-input feedback frames only; not representative moving sequences.'),indent=2))
print(json.dumps(results,indent=2))
