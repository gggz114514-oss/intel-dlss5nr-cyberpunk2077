"""Build CAMPAIGN-SYNTHESIS.md: what paid, what did not, and the structural walls."""
import json, sys
from collections import Counter, defaultdict
from pathlib import Path
sys.path.insert(0, str(Path(__file__).parent))
import COMMON as K

led = K.read_json(K.C / 'CAMPAIGN.json')
cur = K.read_json(K.C / 'CURRENT.json')
les = K.read_json(K.C / 'evidence' / 'LESSONS.json')

cards = []
for c in led['cards']:
    name = c.get('card')
    p = K.C / 'cards' / name / 'RESULTS.json'
    if not p.exists():
        p = K.C / 'queue' / name / 'RESULTS.json'
    r = K.read_json(p) if p.exists() else {}
    t = c.get('timing') or {}
    cards.append(dict(name=name, status=r.get('status') or c.get('status'), triage=r.get('triage'),
                      saving=t.get('saving_ms'), ci=t.get('CI95_ms'), kind=t.get('status'),
                      family=r.get('family')))

measured = [c for c in cards if c['saving'] is not None]
acc = [c for c in measured if c['kind'] == 'FASTER_95' and str(c['status']).startswith('ADOPTED')]
neg = [c for c in measured if c['kind'] == 'SLOWER_95']
ns = [c for c in measured if c['kind'] not in ('FASTER_95', 'SLOWER_95')]

lines = ['# 全后端连续改造 — 总结（截至 %s 的链头）' % cur['best_plan']['sha256'][:12], '',
         '## 1. 结果', '',
         '| 项 | 值 |', '|---|---|',
         '| 接受链 | %d 步（含 1 次消融 + 1 次恢复，均未改变最终链头） |' % len(cur['chain']),
         '| 端到端（handoff vs 链头，三臂平衡批 540 样本） | **+0.884494 ms** CI95[0.873210, 0.895778] |',
         '| 链头与 handoff 的差异 | 83 个 PASS 行（= 各步并集），非 PASS 行零改动 |',
         '| 已测卡 | %d 张（含包/普查/阻断），其中 %d 张有同批计时 |' % (len(cards), len(measured)),
         '| 采纳 | %d 张：%s |' % (len(acc), '、'.join(c['name'] for c in acc)),
         '| 显著变慢 | %d 张 |' % len(neg),
         '| 不显著（INDISTINCT） | %d 张 |' % len(ns),
         '| 跨族结论 | %d 条（evidence/LESSONS.json L1–L20） |' % (7 + len(les['append_after_L7'])), '']

lines += ['## 2. 什么付钱、什么不付钱（按机制类，实测）', '',
          '| 机制类 | 代表卡 | 结果 | 结论 |', '|---|---|---|---|',
          '| 删重复内存消息（寄存器不增） | n05a F05 C256 | **+0.521** | 最强的一类，但**值宽依赖**：同一机制在 C128/C64 成包为 0 |',
          '| 删共享内存往返 + barrier | n09b F09 | **+0.171** | 第二强；关键在"往返落在关键路径上" |',
          '| 删 barrier / arena 合并 | n10 F10 | **+0.094** | 第三强；且必须保住 F10 的三块独立 B 返回区 |',
          '| 删标量链（取模/地址重算） | n18 F01 | **+0.079** | 只在**单派发且大**的核上兑现 |',
          '| 片段共享（一片服务两组） | n09c F09 | **+0.061** | 与前两项同族叠加有效 |',
          '| 权重对跨 M 块复用 | n06/n06b F06 | **+0.040 / +0.018** | 有效但小 |',
          '| 纯重排（软件流水） | n10b F10 | −0.005 ns | 删 0 指令 ⇒ 收益只能来自依赖顺序，实测无 |',
          '| 删消息但被并发隐藏 | n06d F06 C256 / n17c F20 | −0.002 / +0.012 ns | 消息减半也不动帧（L12） |',
          '| 抬活值（多 acc / owner 扩大） | n08 F08 / n13 F15 / n05e F05 | **−0.18 / −0.11 / −0.35** | 占用率吃掉一切 |',
          '| 片段形状加宽 | n02d F02 / n23 F05 | 均 A 类 | 驱动不因此发更宽消息（L14/L20） |',
          '| 改 grid / 跨 PASS 折叠 | n24 F18 | **+0.010 ns** | 机制确证（−38.7% 指令），但站点隔离份额只有 2.4% |', '']

lines += ['## 3. 三堵结构墙', '',
          '- **寄存器 128 GRF**：F18 的 4× 权重冗余消除要 +32 GRF（现 r124）；F05 C128 的 12-acc 要 +96（实测 −0.349）。'
          '删死代码只能腾 ~10 个（L17）。',
          '- **操作数读 512 B/条**：全库 2038 捕获里宽 opcode 只出现在 exec=1 广播读；按 lane 的操作数读恒为 512 B。'
          'F05 C128 的 96 条/WG 是**已证下界**（48 KB ÷ 512 B）。',
          '- **单 PASS 边界**：跨 PASS 折叠可做到逐字节等价且 −38.7% 指令，但收益取决于该段隔离份额；'
          '已测的唯一站点只有 2.4%。', '']

lines += ['## 4. 20 条跨族结论（按主题）', '',
          '| 主题 | 条目 |', '|---|---|',
          '| **静态账会骗人** | L11 静态 32-lane 造出幽灵字节；L14 片段形状改变不产生宽消息；L18 源码冗余可能已被 IGC 折叠；L20 宽 opcode 是广播读、操作数读恒 512 B |',
          '| **测量** | L12 隔离计时不预测整帧敏感性；L13 批内 CI 低估批间漂移（<0.03ms 会翻转符号）；L8 成包结果不可预测（7 包只有 1 个过阈值） |',
          '| **寄存器** | L4 抬活值吃掉收益（三次负例）；L17 死代码只还它自己inflate的宽度 |',
          '| **位移/布局** | L1 源码级地址折叠不落地；L16 地址/窄读轴 21 族 0 站点；L15 F18 的 4× 冗余受 r124 挡住；L20 F05 消息下界 |',
          '| **工程** | L9 预算先于实现；L10 卡片前提可能过期（先读父 ISA）；L7 编译期宏决定模块内容；L6 F08 顺序依赖未解释（n26 在查） |', '']

lines += ['## 5. 族级地图', '', '见 `FAMILY-COVERAGE.md`（含每族隔离时长、链上已改行数、已测卡结果）。', '',
          '## 6. 还剩什么', '',
          '- **F15（2.2%）**：den/PV 已 BLOCKED（源码冗余已被折叠）；只剩 sk 的 SLM 布局与 query 跨度两条，先验低。',
          '- **F04 C64（7 PASS）**：n25 在查。',
          '- **F07（1.2%）**：n27 在查。',
          '- **F08 的 WG 顺序依赖**：n26 在查——它同时决定跨派发改造在该族是否安全。',
          '- **F03/F12/F17（合计 0.9%）**：未开卡，作用域低于检出限，只适合成包。',
          '- **INT8 分支**：完全未碰（施工书 §7），需要单独的口径与资产。']
(K.C / 'CAMPAIGN-SYNTHESIS.md').write_text('\n'.join(lines) + '\n', encoding='utf-8')
print('\n'.join(lines[:26]))
