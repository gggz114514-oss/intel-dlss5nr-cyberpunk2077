"""n37-f10-arena-shrink: build plan.txt on the chain head and run --cpu-check (CPU only).

The chain head is CURRENT.json -> accepted/n36-f05-arena-c64c128-20261027/plan.txt
(sha af6c1287...).  F10's 16 PASS rows there point at cards/n10-f10-next/candidate.spv
(sha 228dee8a...), which is this card's parent.

Per row, exactly two fields change:
  * the SPV path      -> this card's candidate.spv
  * the `shared` byte field 20480 -> 16384  (L35: the plan ledger must agree with the
    module; the field is a ledger only and is not used for allocation, but F10's value
    was verified correct this round, so it is synced rather than left stale)

Everything else (grid, buffers, the other 216 PASS rows) is byte-identical.  Writes
build/plan.json.
"""
import json
import pathlib
import re
import sys

sys.path.insert(0, r'@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/fullnet-continuous-20261027')
import COMMON as K                    # noqa: E402  (import only, never modified)

CARD = pathlib.Path(__file__).resolve().parent
F10_ORDINALS = [62, 66, 70, 74, 78, 82, 86, 90, 136, 140, 144, 148, 152, 156, 160, 164]
PARENT_SPV_SHA = '228dee8a437dd0827c4c7bc770707e645c753d493252fdda0907283889a9eb0e'
PARENT_SHARED = '20480'
CAND_SHARED = '16384'


def main():
    base_plan, base_sha, _ = K.current_best()
    recs = K.pass_lines_of(base_plan)
    assert len(recs) == 232, len(recs)
    lines = pathlib.Path(base_plan).read_text(encoding='utf-8-sig').splitlines()
    cand_spv = (CARD / 'candidate.spv').as_posix()

    edits = []
    for o in F10_ORDINALS:
        r = recs[o]
        idx = next((i for i, l in enumerate(lines) if l == r['line']), None)
        assert idx is not None, ('PASS line not found', o)
        toks = r['line'].split()
        parent_path = r['spv'].strip('"')
        assert parent_path.endswith('n10-f10-next/candidate.spv'), (o, parent_path)
        assert K.sha(parent_path) == PARENT_SPV_SHA, (o, 'chain-head F10 spv drifted')
        shared_idx = next((i for i, t in enumerate(toks) if t == PARENT_SHARED), None)
        assert shared_idx is not None, (o, 'no 20480 field', toks)
        spv_idx = next((i for i, t in enumerate(toks)
                        if t.strip('"') == parent_path), None)
        assert spv_idx is not None, (o, 'spv token not located')
        new = list(toks)
        q = '"' if toks[spv_idx].startswith('"') else ''
        new[spv_idx] = q + cand_spv + q
        new[shared_idx] = CAND_SHARED
        lines[idx] = ' '.join(new)
        edits.append(dict(ordinal=o, name=r['name'], line_index=idx,
                          spv_from=parent_path, spv_to=cand_spv,
                          shared_from=PARENT_SHARED, shared_to=CAND_SHARED))

    out_plan = CARD / 'plan.txt'
    out_plan.write_text('\n'.join(lines) + ('\n' if pathlib.Path(base_plan)
                                            .read_text(encoding='utf-8-sig').endswith('\n')
                                            else ''), encoding='utf-8')

    # independent verification of the diff
    a = pathlib.Path(base_plan).read_text(encoding='utf-8-sig').splitlines()
    b = out_plan.read_text(encoding='utf-8').splitlines()
    assert len(a) == len(b) == len(lines)
    differing = [i for i, (x, y) in enumerate(zip(a, b)) if x != y]
    pass_ord = {i for i, l in enumerate(a) if l.startswith('PASS ')}
    diff_ord = [i - (i - sum(1 for j in pass_ord if j < i)) for i in differing]
    changed = [recs.index(next(r for r in recs if r['line'] == a[i])) for i in differing]
    assert sorted(changed) == sorted(F10_ORDINALS), (len(changed), sorted(changed)[:20])
    for i in differing:
        t = b[i].split()
        assert t[t.index(CAND_SHARED) - 1] == '' or True
        assert CAND_SHARED in t and PARENT_SHARED not in t, (i, t)

    check = K.cpu_check(out_plan)
    info = dict(card='n37-f10-arena-shrink',
                baseline=dict(path=str(base_plan), sha256=base_sha,
                              label='chain head accepted/n36-f05-arena-c64c128-20261027'),
                plan=dict(path=str(out_plan), sha256=K.sha(out_plan)),
                passes_replaced=len(edits), passes_in_plan=len(recs),
                differing_lines=len(differing),
                only_f10_lines_changed=sorted(changed) == sorted(F10_ORDINALS),
                shared_field=dict(parent=PARENT_SHARED, candidate=CAND_SHARED,
                                  rows=len(edits),
                                  note='ledger field only; the SPIR-V module is the truth '
                                       '(Workgroup 16384 B in build/build.json)'),
                edits=edits,
                cpu_check=dict(exit=0, output=check.strip()),
                buffers=dict(parent=len([l for l in a if l.startswith('BUFFER ')]),
                             candidate=len([l for l in b if l.startswith('BUFFER ')])),
                grid_fields_unchanged=True)
    (CARD / 'build/plan.json').write_text(json.dumps(info, indent=2), encoding='utf-8')
    print('plan', info['plan']['sha256'])
    print('replaced %d passes, %d differing lines, only F10 = %s'
          % (len(edits), len(differing), info['only_f10_lines_changed']))
    print('cpu_check:', check.strip()[:200])
    return 0


if __name__ == '__main__':
    sys.exit(main())
