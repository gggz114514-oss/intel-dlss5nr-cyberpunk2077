D = B / "work/codex-astra-fp16-int8-audit-20261024/int8-quality-first-20261029/real-int8-24"


def bk(n, blk):
    return {
        "half": B / f"candidates/amd-map-20261015/s1-integrated/candidate/packs/swin_{blk}_weights.fp16.bin",
        "i8": B / f"work/dsh-int8-fullnet-impl-20261024/candidates/c32-body1-unified/pack/c32_block{blk}_unified_weights.i8.bin",
        "sc": D / f"c32-minmax/b{blk}-scales.bin",
        "aux": B / f"src/project/artifacts/parallel-port-20261013/swin/network/block{blk}/pack/c32_aux.bin",
    }[n]


PUSH = ["168", "4294967295", "0", "0", "9344", "5248", "13440", "0", "16512", "5120",
        "4096", "0", "0", "0", "96", "4294967295"]
PUSH68 = PUSH[:15] + ["0"]
PUSH69 = ["168", "0", "0", "0", "9344", "5248", "13440", "0", "16512", "5120",
          "4096", "0", "0", "0", "96", "4294967295"]


def ptr(p):
    return [int(x) for x in p]


def run_case(inp, outp, blk, push, label):
    t0 = time.time()
    x, valid = M.load_block(inp, dump_shift_x, dump_shift_y)
    blkobj = M.Block(bk("half", blk), bk("i8", blk), bk("sc", blk), bk("aux", blk), push)
    y, _ = blkobj.forward(x, valid)
    idx, vv = M.token_index(dump_shift_x, dump_shift_y)
    ref = np.fromfile(outp, dtype="<f2")[idx]
    m = vv
    a = y[m].astype(np.float64)
    b = ref[m].astype(np.float64)
    r = {
        "label": label, "block": blk, "shift": [dump_shift_x, dump_shift_y],
        "relL2": M.rel_l2(a, b), "corr": M.corr(a, b),
        "rms": float(np.sqrt(((a - b) ** 2).mean())),
        "absmax_pred": float(np.abs(a).max()), "absmax_gpu": float(np.abs(b).max()),
        "valid_tokens": int(m.sum()), "secs": round(time.time() - t0, 1),
    }
    return r, y, idx, vv
