"""Emit SOURCE-DIFF.patch (parent source -> candidate source) for the three groups."""
import difflib, sys
from pathlib import Path

HERE = Path(__file__).parent
CAMP = HERE.parent.parent
sys.path.insert(0, str(CAMP))
import COMMON as K

PAIRS = {
    "c64": (K.D / "f05-kv-consumer-layout/family/c64/candidate.comp", HERE / "candidate-c64.comp"),
    "c128": (K.D / "f05-kv-consumer-layout/family/c128/candidate.comp", HERE / "candidate-c128.comp"),
    "c256": (CAMP / "cards/n05a-f05-c256-mreuse/candidate.comp", HERE / "candidate-c256.comp"),
}
out = []
for g, (a, b) in PAIRS.items():
    la = a.read_text(encoding="utf-8-sig").splitlines(keepends=True)
    lb = b.read_text(encoding="utf-8-sig").splitlines(keepends=True)
    d = list(difflib.unified_diff(la, lb, fromfile="a/%s/%s" % (g, a.name),
                                 tofile="b/%s/%s" % (g, b.name), n=3))
    out.append("".join(d))
    added = sum(1 for x in d if x.startswith("+") and not x.startswith("+++"))
    removed = sum(1 for x in d if x.startswith("-") and not x.startswith("---"))
    print("%-5s diff: +%d -%d lines" % (g, added, removed))
(HERE / "SOURCE-DIFF.patch").write_text("".join(out), encoding="utf-8")
print("wrote", HERE / "SOURCE-DIFF.patch", (HERE / "SOURCE-DIFF.patch").stat().st_size, "B")
