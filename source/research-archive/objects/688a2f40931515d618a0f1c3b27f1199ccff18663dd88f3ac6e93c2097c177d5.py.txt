"""n06-f06-output build: parent byte reproduction, candidate, SPIR-V checks, plan, cpu-check.

CPU only. Never touches the GPU, the baseline plan, or any other card directory.
"""
import json, sys, subprocess
from pathlib import Path

CARD = Path(r'@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/fullnet-continuous-20261027/cards/n06-f06-output')
CAMP = CARD.parent.parent
sys.path.insert(0, str(CAMP))
import COMMON as K
import spv_tools_local as ST

PARENT_SRC = Path(r'@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/fullnet-mechanism-transfer-20261026/f06-scale-fragments/family/c128/candidate.comp')
PARENT_SPV = Path(r'@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/fullnet-mechanism-transfer-20261026/f06-scale-fragments/family/c128/candidate.spv')
PARENT_SHA = '2f2d55739e76f9e946f2c1a55050947bcf15ed98b7fe25f2084af0350d440f00'
PINNED_INC = Path(r'@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/fullnet-continuous-20261027/cards/idt-identity-recipe/evidence/headers/2f2d55739e76')
DEFINES = []


def run(argv):
    r = subprocess.run([str(a) for a in argv], capture_output=True, text=True,
                       encoding='utf-8', errors='replace')
    return r


def compile_(src, out):
    argv = [K.GLSLANG, '-V', '--target-env', 'vulkan1.3']
    for d in DEFINES:
        argv.append('-D' + d)
    argv.append('-I' + str(PINNED_INC))
    argv += [str(src), '-o', str(out)]
    r = run(argv)
    if r.returncode != 0:
        raise SystemExit('compile failed: %s\n%s\n%s' % (argv, r.stdout, r.stderr))
    return dict(argv=[str(a) for a in argv], stderr=r.stderr.strip(),
                sha256=K.sha(out), bytes=Path(out).stat().st_size)


def spv_stats(path):
    m = ST.Module(Path(path))
    ops = {}
    for op, o, _, _ in m.instructions:
        ops[op] = ops.get(op, 0) + 1

    def n(x):
        return ops.get(x, 0)

    fma = len(m.ext_inst_fma_ordinals())
    nc = sorted(ST.no_contraction_ids(m))
    # OpCooperativeMatrix* = 4457/4458/4459/4460/4461/4462 (KHR) in SPIR-V 1.6
    coop = {k: n(k) for k in (4456, 4457, 4458, 4459, 4460, 4461, 4462)}
    int_cf = {str(k): n(k) for k in (128, 130, 132, 134, 135, 137, 138, 139)}
    return dict(path=str(path), sha256=K.sha(path), words=len(m.words),
                instructions=len(m.instructions), bound=m.bound,
                fma_ext_inst=fma, no_contraction=len(nc), no_contraction_ids=nc,
                coop=coop, int_cf=int_cf, opcodes=ops)


def main():
    out = {}
    # ---------------------------------------------------------- parent rebuild
    pre = CARD / 'parent-rebuild.spv'
    rep = compile_(PARENT_SRC, pre)
    rep['source'] = dict(path=str(PARENT_SRC), sha256=K.sha(PARENT_SRC))
    rep['target'] = dict(path=str(PARENT_SPV), sha256=K.sha(PARENT_SPV))
    rep['byte_match'] = rep['sha256'] == PARENT_SHA
    assert rep['byte_match'], rep
    out['parent_rebuild'] = rep

    # ---------------------------------------------------------- candidate
    csrc = CARD / 'candidate.comp'
    cspv = CARD / 'candidate.spv'
    cand = compile_(csrc, cspv)
    cand['source'] = dict(path=str(csrc), sha256=K.sha(csrc))
    out['candidate'] = cand

    # ---------------------------------------------------------- validation
    vp = run([K.VALIDATE, cspv])
    vpar = run([K.VALIDATE, pre])
    out['spv_validate_cpu'] = dict(
        candidate=dict(rc=vp.returncode, stdout=vp.stdout.strip(), stderr=vp.stderr.strip()),
        parent=dict(rc=vpar.returncode, stdout=vpar.stdout.strip(), stderr=vpar.stderr.strip()))
    assert vp.returncode == 0

    # ---------------------------------------------------------- plan
    plan = CARD / 'plan.txt'
    replaced = K.make_plan(cspv, plan, spv_shas={PARENT_SHA})
    out['plan'] = dict(path=str(plan), sha256=K.sha(plan), replaced_passes=replaced,
                       parent_plan=dict(path=str(K.BASELINE), sha256=K.sha(K.BASELINE)))

    # ---------------------------------------------------------- cpu check
    cp = run([K.RUNNER, '--plan', str(plan), '--cpu-check'])
    cpb = run([K.RUNNER, '--plan', str(K.BASELINE), '--cpu-check'])
    out['cpu_check'] = dict(argv=[str(K.RUNNER), '--plan', str(plan), '--cpu-check'],
                            rc=cp.returncode, stdout=cp.stdout.strip(), stderr=cp.stderr.strip(),
                            baseline_stdout=cpb.stdout.strip())
    assert cp.returncode == 0

    # ---------------------------------------------------------- SPIR-V stats
    out['spv_parent'] = spv_stats(pre)
    out['spv_candidate'] = spv_stats(cspv)
    out['float_NC_checked'] = out['spv_parent']['no_contraction']

    (CARD / 'build').mkdir(exist_ok=True)
    K.write_json(CARD / 'build' / 'spv-stats.json',
                 dict(parent=out['spv_parent'], candidate=out['spv_candidate']))
    print(json.dumps({k: v for k, v in out.items()
                      if k not in ('spv_parent', 'spv_candidate')}, indent=2))
    K.write_json(CARD / 'build' / 'build-raw.json', out)
    return out


if __name__ == '__main__':
    main()
