"""Independently verify the lineage audit's headline claim.

Claim: applying the original 76-row TEMPLATE to the current optimised head selects 108 PASS
against small's 118, with ten rows each losing one dispatch (nine wide Swin rows 4->3 and the F20
tail 3->2), and every one of the 76 template rows still exists.

This reads the TEMPLATE straight out of the build script's AST (never executes it), counts the
current head's PASS lines per row, and compares.
"""
import ast, sys
from collections import Counter, defaultdict
from pathlib import Path
sys.path.insert(0, str(Path(__file__).parent))
import COMMON as K
from audit_card import pass_lines

SRC = Path('@NR_WORKSPACE@/work/vk102-small-310-9-16-20261019/scripts/'
           '04_build_small_package.py')
tree = ast.parse(SRC.read_text(encoding='utf-8', errors='replace'))
templates = None
for node in tree.body:
    if isinstance(node, ast.Assign) and any(isinstance(t, ast.Name) and t.id == 'TEMPLATE' for t in node.targets):
        templates = ast.literal_eval(node.value)
print('TEMPLATE rows:', len(templates), '| first:', list(templates.items())[:4])

cur = K.read_json(K.C / 'CURRENT.json')
head = pass_lines(Path(cur['best_plan']['path']))
by_row = defaultdict(list)
for rec in head:
    by_row[int(rec['line'].split()[1])].append(rec)
print('head rows present:', len(by_row), 'total PASS:', len(head))

small_plan = Path('@NR_WORKSPACE@/work/vk102-small-310-9-16-20261019/'
                  'package/manifest/port_network_runtime.txt')
small_passes = [l for l in small_plan.read_text(encoding='utf-8-sig', errors='replace').splitlines()
                if l.startswith('PASS ')]
by_small_row = Counter(int(l.split()[1]) for l in small_passes)
print('small PASS:', len(small_passes), 'over', len(by_small_row), 'rows')

missing = [s for s in templates if s not in by_row]
new_total = 0
deltas = []
for s_row, f_row in templates.items():
    n_new = len(by_row.get(f_row, []))
    n_old = by_small_row.get(s_row, 0)
    new_total += n_new
    if n_old and n_new != n_old:
        deltas.append((s_row, f_row, n_old, n_new))
print('\ntemplate rows missing from the head:', len(missing), missing[:5])
print('selected PASS with the template:', new_total)
print('rows whose dispatch count changed: %d' % len(deltas))
for d in sorted(deltas, key=lambda x: x[0]):
    print('   small row %-3d -> full row %-3d   %d -> %d' % d)
print('\nsum of deltas: %+d (small %d -> %d)' % (new_total - len(small_passes), len(small_passes), new_total))
