"""Three-arm same-batch comparison: vk102 (V), campaign handoff baseline (A), current head (C).

vk102 is a different model version, so its output cannot be matched against our gold frames; per
the vk102 identity protocol the requirement for that arm is a clean run (no NaN/Inf, expected
output size) and self-consistency across its own samples. The two current arms must reproduce the
known gold hashes, which the gates stage already proves separately.

Six balanced permutations, 30 samples per arm per permutation, all samples kept, two-sided
t(df=5) over the six permutation means.
"""
import json, sys
from pathlib import Path
sys.path.insert(0, str(Path(__file__).parent))
import COMMON as K
import STAGE
import ANALYZE

PERM = [('V', 'A', 'C'), ('V', 'C', 'A'), ('A', 'V', 'C'), ('A', 'C', 'V'), ('C', 'V', 'A'), ('C', 'A', 'V')]
VK = Path('@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/'
          'vk102-current-paired-20261026/vk102-path-normalized.txt')
cur = K.read_json(K.C / 'CURRENT.json')
arms = {'V': (VK, K.sha(VK)), 'A': (K.BASELINE, K.sha(K.BASELINE)),
        'C': (Path(cur['best_plan']['path']), cur['best_plan']['sha256'])}
out = K.C / 'queue' / 'vk102-compare'
out.mkdir(parents=True, exist_ok=True)

jobs = []
for pi, order in enumerate(PERM):
    for aid in order:
        p, s = arms[aid]
        jobs.append(K.ordinary_job('perm-%d-%s' % (pi, aid), p, s,
                                   out / 'run' / ('perm-%d-%s' % (pi, aid)), aid, pi))
spec = dict(stage='ordinary', owner='LUNA-CONTINUOUS-VK102',
            authorization='Main authorized the three-arm vk102 comparison only: vk102, handoff '
                          'baseline and current head, same batch.',
            output_root=str(out), identities={k: dict(path=str(v[0]), sha256=v[1]) for k, v in arms.items()})
K.write_json(out / 'GPU-REQUEST.json', spec)
print('arms:', {k: v[1][:12] for k, v in arms.items()})
print('jobs:', len(jobs))
import run_external
run_external.main(['vk102-compare', str(out / 'GPU-REQUEST.json'), 'ordinary'])
