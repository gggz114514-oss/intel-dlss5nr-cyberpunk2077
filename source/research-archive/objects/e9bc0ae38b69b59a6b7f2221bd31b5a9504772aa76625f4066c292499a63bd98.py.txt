"""QKV weight-feed conflict table for the F05 C64 group (card n04a-f05-c64-feed).

Outputs one row per B-weight SEND and per B-DPAS in the QKV projection region,
plus a register-role map that shows where a register carries B data and where
it carries an address payload.  Read-only.
"""
import json
import re
import sys
from pathlib import Path

sys.path.insert(0, str(Path(__file__).parent))
from isa_scan import parse, ASM

ins = parse(ASM)
by_off = {i['off']: i for i in ins}
order = [i['off'] for i in ins]

WEIGHT = 'load.ugm.d32x8.a32.bss[a0.2]'
DPAS = 'dpas.8x8'


def fields(head):
    return [f for f in re.split(r'\s{2,}', head) if f]


def reg_of(field):
    m = re.match(r'^r(\d+)', field.strip())
    return int(m.group(1)) if m else None


rows = []
for i in ins:
    if WEIGHT in i['tail']:
        f = fields(i["full"])
        dst = reg_of(f[1]); pay = reg_of(f[2])
        rows.append(dict(kind='send', off=i['off'], dst=dst, payload=pay,
                         execmask=i['execmask'], pred=i['pred'],
                         toks=re.findall(r'\$(\d+)', i['tail'])))
    elif i['mnemonic'].startswith('dpas'):
        f = fields(i["full"])
        acc = reg_of(f[1]); b = reg_of(f[3]); a = reg_of(f[4])
        rows.append(dict(kind='dpas', off=i['off'], acc=acc, b=b, a=a,
                         execmask=i['execmask'], pred=i['pred'],
                         toks=re.findall(r'\$(\d+)', i['tail'])))
rows.sort(key=lambda r: r['off'])

# ---- which instruction defines each payload register just before use? ----
def defs_of(reg, before_off, window=40):
    out = []
    pat = re.compile(r'\br' + str(reg) + r'\.\d+<')
    for i in ins:
        if i['off'] >= before_off:
            break
        if i['off'] < before_off - window * 16:
            continue
        if pat.search(i['head']) and not i['mnemonic'].startswith('dpas'):
            out.append((i['off'], i['mnemonic'], i['head'].strip()[:70],
                        re.findall(r'\$(\d+)', i['tail'])))
    return out[-1] if out else None

print('%-6s %-5s %-4s %-4s %-7s %-4s %s' % ('off', 'kind', 'dst', 'pay/B', 'acc', 'mask', 'note'))
for r in rows:
    if r['kind'] == 'send':
        d = defs_of(r['payload'], r['off'])
        print('%-6s %-5s r%-3d r%-3d %-7s %-4s addr_def=%s' % (
            hex(r['off']), 'send', r['dst'], r['payload'], '', r['execmask'],
            (hex(d[0]) + ':' + d[1]) if d else '?'))
    else:
        print('%-6s %-5s %-4s r%-3d r%-3d %-4s' % (
            hex(r['off']), 'dpas', '', r['b'], r['acc'], r['execmask']))

# ---- register role map inside the QKV region ----
qkv_lo, qkv_hi = 0x0C00, 0x2200
roles = {}
for r in rows:
    if not (qkv_lo <= r['off'] < qkv_hi):
        continue
    if r['kind'] == 'send':
        roles.setdefault(r['dst'], {}).setdefault('Bdata', []).append(r['off'])
        roles.setdefault(r['payload'], {}).setdefault('addr', []).append(r['off'])
    else:
        roles.setdefault(r['b'], {}).setdefault('Buse', []).append(r['off'])
print('\nregister roles in QKV region (0x%X..0x%X):' % (qkv_lo, qkv_hi))
for reg in sorted(roles):
    print('  r%-3d %s' % (reg, json.dumps({k: [hex(x) for x in v]
                                           for k, v in roles[reg].items()})))
