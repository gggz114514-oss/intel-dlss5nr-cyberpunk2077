"""n31: detect the `(lower?x:peer)+(lower?peer:x)` identity selects in the ISA.

(a,b) + (b,a) is the signature of an operand swap that is an identity for a
commutative FP add:  (lower ? x : peer) + (lower ? peer : x) == x + peer.
CPU only.
"""
import sys
from collections import Counter
from pathlib import Path

CAMP = Path('@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/fullnet-continuous-20261027')
sys.path.insert(0, str(CAMP))
import ISA

ISA_PATH = CAMP / 'queue/n29-f01-alu-chain/native/captures/native-native-cand/ISA/CS_asma0888c5a3c6703a9_simd32_0000000000000001_0_orig.asm'
ENTRY_END = 0x4A90


def dst(op):
    return op.split('<')[0]


def srcs(ins):
    return [o.split('<')[0] for o in ins['ops'][1:]]


def main():
    isa = ISA.parse(ISA_PATH)
    last = {}
    pairs = []
    for k, ins in enumerate(isa):
        if ins['addr'] < ENTRY_END:
            continue
        d = dst(ins['ops'][0]) if ins['ops'] else ''
        if ins['op'] in ('add', 'add3'):
            o = ins['ops']
            if len(o) == 3:
                a, b = dst(o[1]), dst(o[2])
                if a.startswith('acc') and b.startswith('acc'):
                    sa, sb = last.get(a), last.get(b)
                    if (sa is not None and sb is not None
                            and isa[sa]['op'] == 'sel' and isa[sb]['op'] == 'sel'):
                        oa, ob = srcs(isa[sa]), srcs(isa[sb])
                        if len(oa) >= 2 and len(ob) >= 2 and oa[0] == ob[1] and oa[1] == ob[0]:
                            pairs.append((ins['addr'], isa[sa]['addr'], isa[sb]['addr'],
                                          oa[0], oa[1],
                                          (isa[sa]['flags'] or '') + '|' + (isa[sb]['flags'] or '')))
        if d:
            last[d] = k
    print('identity (operand-swap) sel pairs: %d  -> %d sels' % (len(pairs), 2 * len(pairs)))
    blk = Counter()
    for addr, sa, sb, x, y, fl in pairs:
        b = 'A' if addr < 0x11DE0 else ('B' if addr < 0x129D0 else ('C' if addr < 0x13190 else 'D'))
        blk[b] += 1
    print('per block:', dict(blk))
    preds = Counter()
    for addr, sa, sb, x, y, fl in pairs:
        for f in fl.split('|'):
            if f.startswith('(') and f.endswith(')'):
                preds[f] += 1
    print('predicates:', dict(preds))
    print('span: %04X..%04X' % (min(p[0] for p in pairs), max(p[0] for p in pairs)))
    print('sample adds:', [hex(p[0]) for p in pairs[:10]])
    # do the four reduction blocks each hold 7 step-pairs (4 accumulate + 3 butterfly)?
    print('math.inv:', [hex(i["addr"]) for i in isa if i['op'] == 'math.inv' and i['addr'] >= ENTRY_END])


if __name__ == '__main__':
    main()
