"""Per-k-step instruction budget of the F13 expand kernel (static, CPU only)."""
import sys, collections, json
sys.path.insert(0, r'@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/fullnet-continuous-20261027/cards/n11-vit-ffn/_tools')
from isa_scan import parse

p = sys.argv[1]
ins = parse(p)
lo, hi = int(sys.argv[2], 16), int(sys.argv[3], 16)


def cls(i):
    t = i['text']
    if i['op'].startswith('send'):
        return 'send.wide' if '(16|M0)' in t else 'send.narrow'
    if i['op'].startswith('dpas'):
        return 'dpas'
    if i['op'] == 'sync.nop':
        return 'sync'
    if i['op'] == 'mov':
        import re
        m = re.search(r'\((\d+)\|M0\)', t)
        return 'mov%s' % (m.group(1) if m else '?')
    if i['op'] in ('add', 'add3', 'shl', 'mul', 'mad', 'sel', 'cmp', 'and', 'or', 'jmpi', 'bfn.0xEA'):
        return 'int'
    return i['op']


for rng in [(lo, hi)]:
    c = collections.Counter()
    n = 0
    for i in ins:
        if rng[0] <= i['off'] <= rng[1]:
            c[cls(i)] += 1
            n += 1
    print('window %05X..%05X  instructions=%d' % (rng[0], rng[1], n))
    for k, v in c.most_common():
        print('   %-12s %4d  %5.1f%%' % (k, v, 100.0 * v / n))
