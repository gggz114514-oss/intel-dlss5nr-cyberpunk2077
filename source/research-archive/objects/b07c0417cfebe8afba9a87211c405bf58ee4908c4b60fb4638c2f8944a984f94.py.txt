"""n27-f07 proofs: exhaustive SLM permutation/equivalence + SPIR-V structural audit.

PROOF-A (mapping)   the reblocked staging index map is a bijection on [0,16384) and every
                    A-fragment element read by the candidate equals the element the parent
                    read (exhaustive: 8 jb x 16 kb x 8 rows x 16 cols = 16384 elements).
PROOF-B (arithmetic) complete opcode-histogram delta; the non-integer opcode sequence
                    (integer/address set {128,130,132,134,135,137,138,139} removed) must be
                    identical, i.e. no float op is added, removed or reordered;
                    NoContraction sites compared one by one by (opcode, operand shape).
PROOF-C (resources) entry point, execution modes, local size, binding set, workgroup
                    variable count unchanged.
"""
import json, sys
from pathlib import Path
from collections import Counter

C = Path(r'@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/fullnet-continuous-20261027')
sys.path.insert(0, str(C))
import COMMON as K

CARD = C / 'cards/n27-f07'
PARENT = Path(K.row_by_ordinal(59)['spv'])
CAND = CARD / 'candidate.spv'
INT_OPS = {128, 130, 132, 134, 135, 137, 138, 139}
RESULT_AT_ZERO = set(K.ST.RESULT_AT_ZERO_OPS)


# ------------------------------------------------------------------ PROOF-A
def mapping_proof():
    N = 16384
    dst = {}
    for row in range(64):
        for k in range(256):
            jb, r = row >> 3, row & 7
            kb, c = k >> 4, k & 15
            dst[(row, k)] = ((jb * 16 + kb) * 8 + r) * 16 + c
    bijection = sorted(dst.values()) == list(range(N))
    bad = 0
    for jb in range(8):
        for kb in range(16):
            for r in range(8):
                for c in range(16):
                    if dst[(jb * 8 + r, kb * 16 + c)] != (jb * 16 + kb) * 128 + r * 16 + c:
                        bad += 1
    blocks = sorted({dst[(jb * 8 + r, kb * 16 + c)] // 128
                     for jb in range(8) for kb in range(16) for r in range(8) for c in range(16)})
    return dict(
        staging_elements=N, bijection=bijection, distinct_images=len(set(dst.values())),
        fragment_elements_checked=8 * 16 * 8 * 16, fragment_mismatches=bad,
        fragment_block_is_contiguous_256B=(bad == 0),
        fragment_blocks=len(blocks), fragment_block_ids_are_0_to_127=(blocks == list(range(128))),
        parent_row_stride_halfs=256, candidate_row_stride_halfs=16,
        parent_bytes_per_A_fragment_read=8 * 32, candidate_bytes_per_A_fragment_read=256,
        formula='dst(row,k) = ((row/8*16 + k/16)*8 + row%8)*16 + k%16',
        statement='bijective on 16384 elements; fragment (jb,kb) occupies exactly '
                  '[(jb*16+kb)*128, +128) halfs = one contiguous 256 B block; every '
                  'A-fragment element identity holds',
    )


# ------------------------------------------------------------------ PROOF-B
def shape(ops, op):
    """id-free operand shape for comparison across renumbered modules."""
    rest = ops[2:] if op in RESULT_AT_ZERO else ops[1:]
    return tuple('%d' % x if x < 64 else 'id' for x in rest)


def spv_proof():
    pm, cm = K.ST.Module(PARENT), K.ST.Module(CAND)
    hp = Counter(i[0] for i in pm.instructions)
    hc = Counter(i[0] for i in cm.instructions)
    delta = {str(k): [hp.get(k, 0), hc.get(k, 0)] for k in sorted(set(hp) | set(hc))
             if hp.get(k, 0) != hc.get(k, 0)}
    delta_int_only = all(int(k) in INT_OPS for k in delta)
    seq_p = [i[0] for i in pm.instructions if i[0] not in INT_OPS]
    seq_c = [i[0] for i in cm.instructions if i[0] not in INT_OPS]
    # NoContraction sites
    def nc_sites(m):
        dec = {ops[0] for op, ops, _, _ in m.instructions
               if op == 71 and len(ops) > 1 and ops[1] == K.ST.NO_CONTRACTION}
        out = []
        for idx, (op, ops, _, _) in enumerate(m.instructions):
            if K.ST.result_id_of(op, ops) in dec:
                out.append(dict(index=idx, op=op, shape=shape(ops, op)))
        return out
    ncp, ncc = nc_sites(pm), nc_sites(cm)
    # binding set (decoration id -> binding); compare the multiset of (binding, storage)
    def binds(m):
        b = []
        for op, ops, _, _ in m.instructions:
            if op == 71 and ops[1] == 33:
                b.append(ops[2])
        return sorted(b)
    def storage(m):
        c = Counter(ops[2] for op, ops, _, _ in m.instructions if op == 59)
        return {str(k): v for k, v in sorted(c.items())}
    FLOAT_ARITH = {129: 'OpFAdd', 131: 'OpFSub', 133: 'OpFMul'}
    arith_delta = {k: [hp.get(k, 0), hc.get(k, 0)] for k in sorted(set(FLOAT_ARITH) | INT_OPS)
                   if hp.get(k, 0) != hc.get(k, 0)}
    scaffold = {5, 43, 59, 61, 62, 199, 176, 194, 246, 248, 249, 250, 16, 17, 71}
    return dict(
        parent_instructions=len(pm.instructions), candidate_instructions=len(cm.instructions),
        parent_words=len(pm.words), candidate_words=len(cm.words),
        opcode_histogram_delta=delta,
        arithmetic_opcode_delta=arith_delta,
        float_arithmetic_unchanged=all(int(k) not in set(FLOAT_ARITH) for k in arith_delta),
        arithmetic_delta_is_integer_only=all(int(k) in INT_OPS for k in arith_delta),
        non_arithmetic_delta_opcodes=sorted(int(k) for k in delta if int(k) not in set(FLOAT_ARITH) | INT_OPS),
        integer_control_ops_total=[sum(hp.get(k, 0) for k in INT_OPS), sum(hc.get(k, 0) for k in INT_OPS)],
        non_integer_opcode_sequence_identical=(seq_p == seq_c),
        non_integer_instruction_count=[len(seq_p), len(seq_c)],
        no_contraction_count=[len(ncp), len(ncc)],
        no_contraction_sites_identical=[(s['op'], s['shape']) for s in ncp] ==
                                       [(s['op'], s['shape']) for s in ncc],
        no_contraction_sites=ncp,
        bindings_identical=(binds(pm) == binds(cm)), bindings=binds(cm),
        storage_class_counts=[storage(pm), storage(cm)],
        workgroup_variable_count=[sum(1 for op, ops, _, _ in pm.instructions if op == 59 and ops[2] == 4),
                                  sum(1 for op, ops, _, _ in cm.instructions if op == 59 and ops[2] == 4)],
        staging_elements_per_thread=[16384 // 256, 1024 // 256 * 16],
        note='the only arithmetic delta is OpIAdd/OpUDiv from the index permutation; the extra '
             'OpLoad/OpStore come from expressing the copy as 16 contiguous halfs per chunk '
             '(same 64 halfs per thread as the parent), which is the intended change.',
    )


if __name__ == '__main__':
    a = mapping_proof()
    b = spv_proof()
    out = dict(parent=dict(path=str(PARENT), sha256=K.sha(PARENT)),
               candidate=dict(path=str(CAND), sha256=K.sha(CAND)),
               mapping=a, spv=b)
    K.write_json(CARD / 'PROOF.json', out)
    print(json.dumps(out, indent=1))
