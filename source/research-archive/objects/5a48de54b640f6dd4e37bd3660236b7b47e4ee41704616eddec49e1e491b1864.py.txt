"""nat-index probe: how are native ISA captures laid out and how do we tie them to a plan?

Read-only. Prints counts only.
"""
import json, re, sys, os
from pathlib import Path
from collections import Counter, defaultdict

W = Path('@NR_WORKSPACE@')
A = W / 'work/codex-astra-fp16-int8-audit-20261024'

ISA_RE = re.compile(r'^CS_asm([0-9a-f]{16})_simd(\d+)_([0-9a-f]{16})_(\d+)_orig\.(asm|txt)$', re.I)

def walk_asm(root):
    out = []
    for dirpath, dirnames, filenames in os.walk(root):
        for fn in filenames:
            if fn.lower().endswith(('.asm', '.isa')):
                out.append(Path(dirpath) / fn)
    return out

def main():
    asms = walk_asm(W)
    print('total asm:', len(asms))
    bad = [p for p in asms if not ISA_RE.match(p.name)]
    print('non-matching names:', len(bad))
    for p in bad[:10]:
        print('   ', p)

    sessions = defaultdict(list)
    for p in asms:
        m = ISA_RE.match(p.name)
        if m:
            sessions[p.parent].add if False else sessions[p.parent].append((int(m.group(4)), m.group(1)))
    print('session dirs:', len(sessions))
    print('sessions by dir name:', Counter(d.name for d in sessions).most_common(10))

    # pipeline-index sets
    sizes = Counter(len(v) for v in sessions.values())
    print('pipeline-count histogram (files per session dir):', sorted(sizes.items())[:15], '...')
    dup = {d: v for d, v in sessions.items() if len(set(i for i, _ in v)) != len(v)}
    print('sessions with duplicate pipeline index:', len(dup))

    # roots
    roots = Counter()
    for d in sessions:
        rel = d.relative_to(W)
        roots[str(rel.parts[0])] += 1
    print('session dirs by top-level root:', roots.most_common())

if __name__ == '__main__':
    main()
