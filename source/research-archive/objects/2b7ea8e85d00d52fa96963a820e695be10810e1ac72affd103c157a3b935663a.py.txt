"""Static output colour-correction study on the current 24-frame candidate.

Fit y' = M y + b (3x3 + bias) on two sequences, test on the third.  This is a
12-parameter static transform; if it folds into the final 4x32 exit weights it
is a weight-only change, not a new pass.
"""
from __future__ import annotations

import json
from pathlib import Path

import numpy as np

Q = Path(r"@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/int8-quality-first-20261029")
OUT = Path(r"@NR_WORKSPACE@/work/dsh-fullnet-static-20261029/reports")


def load(p: Path) -> np.ndarray:
    return np.fromfile(p, dtype="<f4").reshape(2, 720, 1280, 4)


def main() -> None:
    jobs = json.loads((Q / "real-int8-24/JOBS.json").read_text())
    rows = []
    for j in jobs:
        if j["dump"]:
            continue
        f = j["frame"]
        ref = load(Path(j["reference"]))[0, :, :, :3].astype(np.float64).reshape(-1, 3)
        cand = load(Q / f"real-int8-24/c64-family-static/run-{f}/replayed-network.bin")[0, :, :, :3].astype(np.float64).reshape(-1, 3)
        rows.append({"frame": f, "seq": j["sequence"], "ref": ref, "cand": cand})

    def psnr(a, b):
        e = a - b
        return float(-10 * np.log10(np.mean(e * e)))

    def fit(train):
        x = np.concatenate([r["cand"] for r in train])
        y = np.concatenate([r["ref"] for r in train])
        a = np.concatenate([x, np.ones((len(x), 1))], axis=1)
        c = np.linalg.lstsq(a, y, rcond=None)[0]
        return c

    def apply(c, x):
        return np.concatenate([x, np.ones((len(x), 1))], axis=1) @ c

    out = {"scope": "24 frames, candidate vs captured FP16 application RGB; static 3x3+bias output correction",
           "note": "leave-one-sequence-out is a scene holdout, not a new-scene holdout; all sequences are daytime street",
           "folds": [], "full_fit": None}
    seqs = sorted({r["seq"] for r in rows})
    for held in seqs:
        tr = [r for r in rows if r["seq"] != held]
        te = [r for r in rows if r["seq"] == held]
        c = fit(tr)
        base = float(np.mean([psnr(r["ref"], r["cand"]) for r in te]))
        corr = float(np.mean([psnr(r["ref"], apply(c, r["cand"])) for r in te]))
        out["folds"].append({"held_out": held, "base_mean_db": base, "corrected_mean_db": corr,
                             "gain_db": corr - base, "matrix": c.tolist()})
    c_all = fit(rows)
    base_all = float(np.mean([psnr(r["ref"], r["cand"]) for r in rows]))
    corr_all = float(np.mean([psnr(r["ref"], apply(c_all, r["cand"])) for r in rows]))
    out["full_fit"] = {"base_mean_db": base_all, "corrected_mean_db": corr_all,
                       "gain_db": corr_all - base_all, "matrix": c_all.tolist(),
                       "warning": "in-sample; do not use the full-fit gain as a generalization estimate"}
    (OUT / "OUTPUT-CORRECTION-STUDY.json").write_text(json.dumps(out, indent=2), encoding="utf-8")
    print(json.dumps(out, indent=2)[:2000])


if __name__ == "__main__":
    main()
