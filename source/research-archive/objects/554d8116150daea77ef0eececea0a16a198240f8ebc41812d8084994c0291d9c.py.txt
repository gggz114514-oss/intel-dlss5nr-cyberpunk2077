def main(path):
    data = Path(path).read_bytes()
    words = list(struct.unpack('<%dI' % (len(data) // 4), data))
    i = 5
    insts = []
    while i < len(words):
        w = words[i]
        op, count = w & 0xffff, w >> 16
        insts.append((op, words[i + 1:i + count]))
        i += count
    from collections import Counter
    c = Counter(op for op, _ in insts)
    unknown = sorted(k for k in c if k not in KNOWN)
    print('module', path)
    print('unknown opcodes:', unknown)
    for op in unknown:
        ex = [o for o2, o in insts if o2 == op][:3]
        print('  op%-5d count=%-4d examples=%s' % (op, c[op], [list(e) for e in ex]))
    print('--- known group/quad/bit ops present ---')
    for op in sorted(c):
        if op in KNOWN and ('Group' in KNOWN[op] or 'Quad' in KNOWN[op] or 'Ballot' in KNOWN[op]
                            or 'Control' in KNOWN[op] or 'BitField' in KNOWN[op]
                            or 'Cooperative' in KNOWN[op]):
            print('  %-5d %-32s %d' % (op, KNOWN[op], c[op]))


if __name__ == '__main__':
    main(sys.argv[1])
