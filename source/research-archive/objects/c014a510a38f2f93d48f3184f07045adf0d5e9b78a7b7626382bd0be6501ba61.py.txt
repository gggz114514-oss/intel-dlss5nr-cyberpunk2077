#!/usr/bin/env python3
"""s37fn5..s37fnA: per-site-class explicit-fusion variants.

The fragment pair add covers the parent's a-sites on lanes 0..7 (cols c,
c+16 for c<8) and b-sites on lanes 8..15. The parent's gate-compiled fusion
pattern is mixed (uniform first/none/second all fail the gate), so express
each class independently with a lane-conditional select:
  a-form(lane<8):  FIRST  fma(h0,h0,H(h1*h1)) | SECOND fma(h1,h1,H(h0*h0)) | NONE H(h0*h0)+H(h1*h1)
  b-form(lane>=8): same three forms.
Six remaining combos are built and gated against the gold frame.
"""
import hashlib
import shutil
import subprocess
import sys
from pathlib import Path

WS = Path(r"@NR_WORKSPACE@")
CAND_SRC = WS / "work/zcode-fp16-fullnet-handoff-20261024/candidates/s37-fnorm/source/s37_fold_body_r1a.comp"
OUT = WS / "work/zcode-fp16-hotspot-r2-20261024/s37fn-matrix"
INC = CAND_SRC.parent
COMPILER = WS / "src/project/tools/glslang-16.5.0/bin/glslang.exe"
VALIDATOR = WS / "candidates/amd-map-20261015/s4-final-output-fusion-fix/production-repair/spv_validate_cpu.exe"

OLD = """            precise float16_t z0[4], z1[4];
            for (uint ci = 0u; ci < 4u; ++ci) {
                z0[ci] = half0[ci] * half0[ci];
                z1[ci] = half1[ci] * half1[ci];
            }
            // per-component pair = sq[c] + sq[c+16] (local: same lane, same
            // row, both fragments). Packed 2 uint32 for XOR shuffle chain.
            uint p01 = packFloat2x16(f16vec2(z0[0] + z1[0], z0[1] + z1[1]));
            uint p23 = packFloat2x16(f16vec2(z0[2] + z1[2], z0[3] + z1[3]));"""

# per-class forms: F=first fused (second pre-rounded), S=second fused, N=none
FORMS = {
    "F": "fma({A}, {A}, z1[{c}])",
    "S": "fma({B}, {B}, z0[{c}])",
    "N": "z0[{c}] + z1[{c}]",
}


def expr(form, ci):
    return FORMS[form].format(A=f"half0[{ci}]", B=f"half1[{ci}]", c=ci)


def pair_block(fa, fb):
    lines = ["            precise float16_t z0[4], z1[4];",
             "            for (uint ci = 0u; ci < 4u; ++ci) {",
             "                z0[ci] = half0[ci] * half0[ci];",
             "                z1[ci] = half1[ci] * half1[ci];",
             "            }",
             "            // per-class explicit fusion (a-sites lanes 0..7, b-sites lanes 8..15)",
             "            precise float16_t pv01[4], pv23[4];",
             "            {",
             "                precise float16_t pa[4], pb[4];"]
    for ci in range(4):
        lines.append(f"                pa[{ci}] = {expr(fa, ci)};")
    for ci in range(4):
        lines.append(f"                pb[{ci}] = {expr(fb, ci)};")
    lines += ["                uint sel = (lane & 15u) < 8u ? 1u : 0u;",
              "                for (uint ci = 0u; ci < 4u; ++ci) {",
              "                    pv01[ci] = sel != 0u ? pa[ci] : pb[ci];",
              "                }",
              "            }",
              "            uint p01 = packFloat2x16(f16vec2(pv01[0], pv01[1]));",
              "            uint p23 = packFloat2x16(f16vec2(pv01[2], pv01[3]));"]
    # drop the unused pv23 bookkeeping
    txt = "\n".join(lines)
    txt = txt.replace("precise float16_t pv01[4], pv23[4];", "precise float16_t pv01[4];")
    return txt


def sha(p):
    return hashlib.sha256(Path(p).read_bytes()).hexdigest()


def compile_one(src, out_spv, label):
    r = subprocess.run([str(COMPILER), "-V", "--target-env", "vulkan1.3",
                        "-I" + str(INC), "-o", str(out_spv), str(src)],
                       capture_output=True, text=True)
    if r.returncode != 0:
        print(label, "COMPILE FAIL:\n", r.stderr[-1500:])
        return False
    v = subprocess.run([str(VALIDATOR), str(out_spv)], capture_output=True, text=True)
    ok = "result=0" in (v.stdout + v.stderr)
    print(label, sha(out_spv)[:16], "validate", "OK" if ok else "FAIL")
    return ok


def main():
    (OUT / "source").mkdir(parents=True, exist_ok=True)
    (OUT / "spv").mkdir(exist_ok=True)
    for f in INC.iterdir():
        if f.suffix in (".comp", ".glsl"):
            shutil.copy2(f, OUT / "source" / f.name)
    base = CAND_SRC.read_text(encoding="utf-8")
    assert base.count(OLD) == 1

    D01 = WS / "candidates/dsh-fp16-same-family-20261023/combo-d01-d034-s32k11-v1/plan-combo-d01-d034-s32k11-v1.txt"
    base_plan = Path(D01).read_text(encoding="utf-8").replace("\r\n", "\n").split("\n")

    combos = [("FN", "F", "N"), ("NF", "N", "F"), ("FS", "F", "S"),
              ("SF", "S", "F"), ("NS", "N", "S"), ("SN", "S", "N")]
    for name, fa, fb in combos:
        txt = base.replace(OLD, pair_block(fa, fb))
        p = OUT / "source" / f"s37fn{name}.comp"
        p.write_text(txt, encoding="utf-8")
        spv = OUT / "spv" / f"s37fn{name}.spv"
        if not compile_one(p, spv, name + ":"):
            continue
        lines = list(base_plan)
        done = 0
        for i, ln in enumerate(lines):
            if ln.startswith('PASS 151 "s37_fold_body_r1a_v1"'):
                w = ln.split()
                w[3] = str(spv).replace("\\", "/")
                lines[i] = " ".join(w)
                done += 1
        assert done == 1
        plan = OUT / f"plan-s37fn{name}-232.txt"
        Path(plan).write_text("\n".join(lines), encoding="utf-8")
        print(" ", plan.name, sha(plan)[:16])


if __name__ == "__main__":
    main()
