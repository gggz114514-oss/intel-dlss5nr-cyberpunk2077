#!/usr/bin/env python3
"""Ad-hoc: dump the loop structure (labels, merges, phis) of the R03 SPVs."""
import sys
from pathlib import Path
sys.path.insert(0, str(Path(__file__).resolve().parent))
import spv_read as S  # noqa: E402

CARD = Path("@NR_WORKSPACE@/work/dsh-xmx-handoff-validation-20261024/candidates/r03-ioport4")

for name in ("parent_repro", "r03_ioport4_i8"):
    m = S.Module(CARD / "spv" / (name + ".spv"))
    print("==", name)
    for i, (op, a, wc) in enumerate(m.ins):
        if op in (246, 245) or (op == 250):
            nm = m.name(op)
            extra = ""
            if op == 245:
                base = m.defining(a[2][0] if False else a[2]) if len(a) > 2 else None
                extra = " phi(%s) <- %s" % (a[2], m.name(base[0]) if base else "?")
            print("   idx=%d %s ops=%s%s" % (i, nm, a, extra))
