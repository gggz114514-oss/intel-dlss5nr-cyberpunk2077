"""compare the three pass-0 ISA variants that claim the same SPV sha."""
import re
from pathlib import Path
from collections import Counter

INSN = re.compile(r'^\s*/\*\s*\[([0-9A-Fa-f]+)\]\s*\*/\s*(.*)$')
BASE = Path('@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024')
FILES = [
    BASE / 'fullnet-mechanism-transfer-20261026/f10-weight-return-separation-20261027/captures/native-f20-pixel-owner/ISA/CS_asm5bd64f2b0906027c_simd32_0000000000000001_0_orig.asm',
    BASE / 'f20-tail-diagnosis-20261026/captures/r0-null/ISA/CS_asm3937f576576af5b6_simd32_0000000000000001_0_orig.asm',
    BASE / 'fullnet-mechanism-transfer-20261026/f05-kv-ready-split-20261027/captures/dry-clone/ISA/CS_asm0f0860f340d97767_simd32_0000000000000001_0_orig.asm',
]


def stats(p):
    n = 0
    mn = Counter()
    sends = Counter()
    blocks = 0
    for line in open(p, encoding='utf-8', errors='replace'):
        if line[:1] == 'L' and re.match(r'^L\d+:', line):
            blocks += 1
        m = INSN.match(line)
        if not m:
            continue
        n += 1
        parts = m.group(2).split()
        op = parts[1] if parts and parts[0].startswith('(') else (parts[0] if parts else '?')
        mn[op] += 1
        if 'send' in op:
            tail = m.group(2).split('//')[-1].strip()
            sends[tail[:24]] += 1
    return n, blocks, mn, sends


for p in FILES:
    n, blocks, mn, sends = stats(p)
    print('\n', p.parent.parent.parent.parent.name, p.name[7:23])
    print('   insn=%d blocks=%d bytes=%d' % (n, blocks, p.stat().st_size))
    print('   top ops:', mn.most_common(8))
    print('   sends:', sends.most_common(4))
