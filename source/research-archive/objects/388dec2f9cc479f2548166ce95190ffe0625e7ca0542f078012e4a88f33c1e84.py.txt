import sys
sys.path.insert(0, r"@NR_WORKSPACE@/work/dsh-xmx-handoff-validation-20261024/candidates/b3-c32-seed/tools")
from b3_spv import Module
M = Module(r"@NR_WORKSPACE@/work/dsh-xmx-handoff-validation-20261024/build/b3-c32-seed/parent_repro.spv")
names = {}
for ins in M.insts:
    if ins.op == 5:
        s = b"".join(x.to_bytes(4,"little") for x in ins.ops[1:]).split(b"\0")[0].decode()
        names[ins.ops[0]] = s
print("OpName count", len(names))
import re
for k,v in names.items():
    if re.search(r"seed|yc|widen|deq|narrow|seedH", v):
        print("  id=%-6d %s" % (k,v))
print("--- first 25 names ---")
for i,(k,v) in enumerate(names.items()):
    if i<25: print("  id=%-6d %r" % (k,v))
