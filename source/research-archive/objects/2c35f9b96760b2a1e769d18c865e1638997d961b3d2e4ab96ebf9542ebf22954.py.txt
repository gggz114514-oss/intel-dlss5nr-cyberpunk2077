"""make_candidate.py -- emit the C2 candidate source from the frozen parent.

The change set is deliberately small and enumerable:

  1. parent line 40  `shared int8_t sE8[8192];`            -> comment (deleted)
  2. after line 41   `#include "c32_i8_common.glsl"`       -> + include c2 helper
  3. parent lines 260..497 (one contiguous region)         -> register K32 body

Nothing else in c32_body1_i8.comp changes.  c32_i8_common.glsl and
register_handoff.glsl are copied byte-identically (see build-receipt.json).

Region 3 replaces, in this order:
  * `// ---- C32-I8 expand: ...` + the `NRH_C seed0..seed3` copies   (kept verbatim)
  * the eight feature-group blocks with 16 `nr_i8_put_e` stores
  * `subgroupMemoryBarrierShared(); subgroupBarrier();`  (the E publication)
  * the `// ---- C32-I8 contract: ...` block with its kg loop and 4
    `coopMatLoad(xb8, sE8, ...)` reads
with four K32 step blocks that each produce four CH8x16 fragments, quantise
each once, join them with c2_i8_join_B and consume the resulting B32x16 with
the same four A blocks and the same accumulator order as the parent.
"""
import hashlib
import io
import json
from pathlib import Path

CARD = Path(r"@NR_WORKSPACE@/work/dsh-xmx-handoff-validation-20261024"
            r"/candidates/c2-c32-regk32")
PARENT_COMP = CARD / "source/parent/c32_body1_i8.comp"
CAND = CARD / "source/candidate/c32_body1_i8.comp"
HELPER = CARD / "source/candidate/c2_i8_regk32.glsl"
PARENT_COMMON = CARD / "source/parent/c32_i8_common.glsl"
CAND_COMMON = CARD / "source/candidate/c32_i8_common.glsl"

HELPER_TEXT = r"""// C2 (c2-c32-regk32): register K32 for the C32 int8 contract.
//   The 128 hidden channels are produced 32 at a time as four CH8x16 fragments,
//   quantised once each (c2_i8_pack), joined into one B32x16 code fragment
//   (c2_i8_join_B) and consumed by the four contract accumulators before the
//   next 32 features start.  No E byte is staged in shared memory, so the
//   expand/contract publication barrier disappears with the stores.
//   The permutation is the closed form of the measured Intel B580 int8 B map
//   (source: candidates/dsh-int8-same-family-20261023/abi-probe/evidence/
//   I8-ABI-MEASURED.json, generated helper generated_i8_adapter.glsl).  It is
//   proved element-for-element, for all 32 lanes x 16 components, against both
//   the sealed measurement and the parent's shared coopMatLoad layout in
//   evidence/COORD-PROOF.json.
#ifndef C2_I8_REG_K32_GLSL
#define C2_I8_REG_K32_GLSL

// One code word per CH8x16 fragment: byte b carries component b of this lane.
// nr_i8_code is the parent's own quantiser -- same expression, same operands,
// same two rounding points, so the four codes are bit-identical to the four
// int8 bytes the parent stored in sE8.
uint c2_i8_pack(NRH_CH e, float inv) {
    return (uint(int(nr_i8_code(e[0], inv)) & 0xFF))
         | (uint(int(nr_i8_code(e[1], inv)) & 0xFF) << 8u)
         | (uint(int(nr_i8_code(e[2], inv)) & 0xFF) << 16u)
         | (uint(int(nr_i8_code(e[3], inv)) & 0xFF) << 24u);
}

// Byte b of a packed word as a signed int8 value (exact inverse of the insert).
int c2_i8_byte(uint p, uint byteix) {
    uint v = (p >> (8u * byteix)) & 0xFFu;
    return v >= 128u ? int(v) - 256 : int(v);
}

// Destination component c of lane L reads CH word w[c >> 2] from lane
// ((L & 1) << 4) | (L >> 2) plus 8 * (c & 1), byte (c & 2) | ((L >> 1) & 1).
uint c2_i8_src_lane(uint lane, uint khi) {
    return (((lane & 1u) << 4u) | (lane >> 2u)) + (khi << 3u);
}

// Four CH8x16 code words (features 0..7, 8..15, 16..23, 24..31) -> one B32x16.
// Two shuffles per word carry both K halves; the bytes are selected locally.
I8B c2_i8_join_B(uint w0, uint w1, uint w2, uint w3) {
    uint lane = gl_SubgroupInvocationID;
    uint lo = (lane >> 1u) & 1u;
    uint s0 = c2_i8_src_lane(lane, 0u);
    uint s1 = c2_i8_src_lane(lane, 1u);
    uint a0 = subgroupShuffle(w0, s0);
    uint b0 = subgroupShuffle(w0, s1);
    uint a1 = subgroupShuffle(w1, s0);
    uint b1 = subgroupShuffle(w1, s1);
    uint a2 = subgroupShuffle(w2, s0);
    uint b2 = subgroupShuffle(w2, s1);
    uint a3 = subgroupShuffle(w3, s0);
    uint b3 = subgroupShuffle(w3, s1);
    I8B q;
    q[0]  = int8_t(c2_i8_byte(a0, lo));
    q[1]  = int8_t(c2_i8_byte(b0, lo));
    q[2]  = int8_t(c2_i8_byte(a0, 2u + lo));
    q[3]  = int8_t(c2_i8_byte(b0, 2u + lo));
    q[4]  = int8_t(c2_i8_byte(a1, lo));
    q[5]  = int8_t(c2_i8_byte(b1, lo));
    q[6]  = int8_t(c2_i8_byte(a1, 2u + lo));
    q[7]  = int8_t(c2_i8_byte(b1, 2u + lo));
    q[8]  = int8_t(c2_i8_byte(a2, lo));
    q[9]  = int8_t(c2_i8_byte(b2, lo));
    q[10] = int8_t(c2_i8_byte(a2, 2u + lo));
    q[11] = int8_t(c2_i8_byte(b2, 2u + lo));
    q[12] = int8_t(c2_i8_byte(a3, lo));
    q[13] = int8_t(c2_i8_byte(b3, lo));
    q[14] = int8_t(c2_i8_byte(a3, 2u + lo));
    q[15] = int8_t(c2_i8_byte(b3, 2u + lo));
    return q;
}
#endif
"""


def fragment(frag, var, mbase, act=True):
    """one expand quantiser fragment -> activated half CH8x16, parent verbatim."""
    dbg = "t0*128u+%du" % (8 * frag)
    out = ["                {",
           "                    NRH_C ec%d = nr_i8_deq(a%d, %du, C32_I8_MULT_E);"
           % (frag % 2, frag % 2, mbase),
           "                    %s = nrh_narrow(ec%d);" % (var, frag % 2)]
    if act:
        for i in range(4):
            out.append("                    %s[%d] = nr_act_h(%s[%d]);" % (var, i, var, i))
    out.append("                    if (debug) coopMatStore(%s, dbg, %s, 128u, LCOL);" % (var, dbg))
    out.append("                }")
    return out


def step(p):
    """one K32 step: 4 fragments -> pack -> join -> 4 contract MMAs."""
    out = ["            // ---- K32 step %d: features %d..%d, four CH8x16 fragments -------"
           % (p, 32 * p, 32 * p + 31),
           "            {",
           "                NRH_CH e0;",
           "                NRH_CH e1;",
           "                NRH_CH e2;",
           "                NRH_CH e3;"]
    for half in (0, 1):                     # two 16-feature expand groups
        # `wa` is scoped exactly like the parent's group block, so it stays an
        # SSA value instead of being materialised as a function variable
        out += ["                {",
                "                    I8A wa;",
                "                    a0 = I8C(0);",
                "                    a1 = I8C(0);",
                "                    coopMatLoad(wa, w8, %du, 32u, LCOL);" % (1024 * (4 * p + 2 * half)),
                "                    a0 = coopMatMulAdd(wa, xb8, a0);",
                "                    coopMatLoad(wa, w8, %du, 32u, LCOL);" % (1024 * (4 * p + 2 * half + 1)),
                "                    a1 = coopMatMulAdd(wa, xb8, a1);",
                "                }"]
        out += fragment(4 * p + 2 * half, "e%d" % (2 * half), 32 * p + 16 * half)
        out += fragment(4 * p + 2 * half + 1, "e%d" % (2 * half + 1), 32 * p + 16 * half + 8)
    out += ["                // consume this K32 now, then release the fragments",
            "                I8B eb8 = c2_i8_join_B(c2_i8_pack(e0, c2_inv_e), c2_i8_pack(e1, c2_inv_e),",
            "                                       c2_i8_pack(e2, c2_inv_e), c2_i8_pack(e3, c2_inv_e));",
            "                {",
            "                    I8A wa;"]
    for o in range(4):
        out += ["                    coopMatLoad(wa, w8, 16384u + (%du*4u+%du)*1024u, 32u, LCOL);"
                % (o, p),
                "                    ca%d = coopMatMulAdd(wa, eb8, ca%d);" % (o, o)]
    out += ["                }"]
    out.append("            }")
    return out


def region():
    out = ["        // ---- C2: expand with register K32; no shared E staging -----------",
           "        NRH_C seed0 = yc0;",
           "        NRH_C seed1 = yc1;",
           "        NRH_C seed2 = yc2;",
           "        NRH_C seed3 = yc3;",
           "        // one invE fetch for all 16 code words (the parent reloaded it per call)",
           "        const float c2_inv_e = fsc[3];",
           "        I8C ca0 = I8C(0);",
           "        I8C ca1 = I8C(0);",
           "        I8C ca2 = I8C(0);",
           "        I8C ca3 = I8C(0);",
           "        {",
           "            I8B xb8;",
           "            coopMatLoad(xb8, sX8, sg * 512u, 32u, LCOL);",
           "            I8C a0;",
           "            I8C a1;"]
    for p in range(4):
        out += step(p)
    out += ["        }",
            # parent epilogue, verbatim
            "        yc0 = nr_i8_deq_add(ca0, 0u, C32_I8_MULT_C, seed0);",
            "        yc1 = nr_i8_deq_add(ca1, 8u, C32_I8_MULT_C, seed1);",
            "        yc2 = nr_i8_deq_add(ca2, 16u, C32_I8_MULT_C, seed2);",
            "        yc3 = nr_i8_deq_add(ca3, 24u, C32_I8_MULT_C, seed3);"]
    return out


def main():
    src = io.open(PARENT_COMP, encoding="utf-8").read().split("\n")
    assert src[39] == "shared int8_t sE8[8192];", src[39]
    assert src[40] == '#include "c32_i8_common.glsl"', src[40]
    assert src[259].startswith("        // ---- C32-I8 expand:"), src[259]
    assert src[496] == "        }", repr(src[496])
    assert src[497] == "        y0 = nrh_narrow(yc0);", src[497]

    new = list(src)
    new[39] = "// C2: shared E staging deleted -- E code words never leave registers."
    new = new[:41] + ['#include "c2_i8_regk32.glsl"'] + new[41:]
    # one extra line above shifts everything below it by one
    assert new[259] == "        }", repr(new[259])                      # line 259
    assert new[260].startswith("        // ---- C32-I8 expand:"), new[260]
    assert new[497] == "        }", repr(new[497])                      # line 497
    assert new[498] == "        y0 = nrh_narrow(yc0);", new[498]
    head = new[:260]
    tail = new[498:]
    body = region()
    cand = head + body + tail
    text = "\n".join(cand)

    HELPER.write_text(HELPER_TEXT, encoding="utf-8", newline="\n")
    CAND.write_text(text, encoding="utf-8", newline="\n")

    # c32_i8_common.glsl: drop the shared-E writer, which has no call site left
    # (C2 quantises straight into registers).  Everything else is verbatim.
    common = io.open(PARENT_COMMON, encoding="utf-8").read().split("\n")
    assert common[52].startswith("void nr_i8_put_e("), common[52]
    assert common[59] == "}", common[59]
    ccommon = (common[:51]
               + ["// C2: nr_i8_put_e deleted together with sE8 -- the E code word now",
                  "// goes straight into a register (c2_i8_pack in c2_i8_regk32.glsl)."]
               + common[60:])
    CAND_COMMON.write_text("\n".join(ccommon), encoding="utf-8", newline="\n")

    rec = {
        "card": "c2-c32-regk32",
        "parent_comp": str(PARENT_COMP),
        "parent_comp_sha256": hashlib.sha256(PARENT_COMP.read_bytes()).hexdigest(),
        "parent_lines": len(src),
        "candidate_comp": str(CAND),
        "candidate_comp_sha256": hashlib.sha256(CAND.read_bytes()).hexdigest(),
        "candidate_lines": len(cand),
        "helper": str(HELPER),
        "helper_sha256": hashlib.sha256(HELPER.read_bytes()).hexdigest(),
        "edits": [
            {"file": "c32_body1_i8.comp", "parent_line": 40, "op": "delete",
             "was": "shared int8_t sE8[8192];",
             "now": "// C2: shared E staging deleted -- E code words never leave registers.",
             "effect": "workgroup E staging removed (-8192 B LDS)"},
            {"file": "c32_body1_i8.comp", "parent_line": 41, "op": "insert_after",
             "now": '#include "c2_i8_regk32.glsl"',
             "effect": "closed-form pack + i8_join_B helper"},
            {"file": "c32_body1_i8.comp", "parent_line": [260, 497], "op": "replace",
             "lines_in": 238, "lines_out": len(body),
             "removed": ["8 expand feature-group blocks",
                         "16 nr_i8_put_e calls (64 int8 shared stores per lane)",
                         "subgroupMemoryBarrierShared() + subgroupBarrier() (E publication)",
                         "contract kg loop with 4 coopMatLoad(xb8, sE8, ...) reads"],
             "added": ["4 K32 step blocks: 4 CH8x16 fragments each",
                       "16 c2_i8_pack calls (64 nr_i8_code, unchanged count)",
                       "4 c2_i8_join_B calls (8 subgroupShuffle + 16 byte extracts each)",
                       "16 contract coopMatMulAdd, same (o,kg) A blocks, same order"],
             "kept_verbatim": ["NRH_C seed0..seed3 copies",
                               "16 expand coopMatLoad(A) + 16 coopMatMulAdd + 16 nr_i8_deq",
                               "64 nr_act_h applications",
                               "16 debug coopMatStore of the activated E fragments",
                               "nr_i8_deq_add epilogue and the y narrow"]},
        ],
    }
    (CARD / "build").mkdir(parents=True, exist_ok=True)
    (CARD / "build/CHANGE-SET.json").write_text(json.dumps(rec, indent=1) + "\n", encoding="utf-8")
    print(json.dumps({k: rec[k] for k in ("parent_lines", "candidate_lines",
                                         "candidate_comp_sha256", "helper_sha256")}, indent=1))


if __name__ == "__main__":
    main()
