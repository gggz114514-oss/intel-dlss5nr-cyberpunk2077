"""n38-f20-arena-shrink: per-plane SLM lifetime table, retireable windows, byte budget.

Read-only analysis + machine-checked counts.  No GPU, no deployment.

F20 = ord 230 `s37_fold_body_r1a_v1`, source `f20-coarse-owner-20261026/candidate.comp`
(sha 3e0879b3...), SPV 215aca67..., chain head sha 215aca67... (GROUPS.md: F20 chain head
== baseline sha, so the parent of this card is the handoff row itself).

Outputs
  build/lifetime.json : declarations (source text + parent SPIR-V, variable by variable),
                        the site table (every site pinned to a source line by an anchor that
                        must be present), per-site per-SG element footprints, the value-window
                        timeline per arm, and the RAW / WAR / WAW enumeration with
                        same-owner / cross-owner counts plus the named closer of every pair.
  stdout              : the human summary that SLM-LIFETIME.md quotes.

Method
  Every shared-memory access in the module is enumerated from the source text and its
  footprint is computed as a set of flat element indices per subgroup (8 subgroups of 32
  lanes, local_size_x = 256).  A pair (producer, consumer) is a hazard when the two sites
  touch the same physical array element; `same_owner` counts (sg, sg') with sg == sg',
  `cross_owner` counts sg != sg'.  A cross-owner hazard is legal only when a barrier or a
  payload dependency closes it; every closer is named in CLOSERS, keyed by the exact
  (writer_site, reader_site) pair, and its source anchor is asserted to exist, so a stale
  table cannot silently pass.

  Hazards are enumerated inside each value window (RAW) and between adjacent value windows
  of the same physical array (WAR/WAW), because that is exactly the set of orderings the
  hardware has to observe.
"""

def rows(spec, sg):
    return spec(sg) if callable(spec) else spec


def rows_eight(sg):
    return [8 * sg + i for i in range(8)]


def rows_expand(sg):
    """stage 1: mA = (sg/4 + 4r)*8, mB = mA + 16, r = 0,1."""
    out = []
    for r in (0, 1):
        mA = (sg // 4 + 4 * r) * 8
        out += [mA + i for i in range(8)]
        out += [mA + 16 + i for i in range(8)]
    return out


def rows_split(sg):
    """tail: t = sg*32 + lane -> rows 0..31 for sg 0, 32..63 for sg 1, nothing else."""
    return [32 * sg + i for i in range(32)] if sg < 2 else []


def cols_expand(sg):
    return [(sg % 4) * 32 + i for i in range(32)]


def foot(value, sg, rowset, colset):
    p = PLANE[value]
    base = ARENA_PLANE_BASE.get(value, 0)
    cols = colset(sg) if callable(colset) else colset
    return {(base + r * p['stride'] + c) for r in rowset for c in cols}

DECL = re.compile(r'shared\s+float16_t\s+(\w+)\s*\[\s*(\d+)\s*\]')


def sha(p):
    return hashlib.sha256(pathlib.Path(p).read_bytes()).hexdigest()


def declarations(text):
    out = {}
    for m in DECL.finditer(text):
        out[m.group(1)] = dict(halves=int(m.group(2)), bytes=int(m.group(2)) * 2,
                               line=text[:m.start()].count('\n') + 1)
    return out


def check_anchors(text, arm):
    lines = text.splitlines()
    bad = []
    for s in SITES:
        if s['arm'] not in ('both', arm):
            continue
        anchors = (s.get('anchor_by_arm') or {}).get(arm) or s['anchor']
        for a, ln in zip(anchors, s['lines']):
            ok = (0 < ln <= len(lines)) and (a in lines[ln - 1])
            if not ok:
                bad.append(dict(site=s['id'], line=ln, expected=a,
                                actual=(lines[ln - 1].strip()[:100]
                                        if 0 < ln <= len(lines) else '')))
    return bad


def arm_sites(arm):
    out = []
    for s in SITES:
        if s['arm'] not in ('both', arm):
            continue
        arr = (s.get('array_by_arm') or {}).get(arm) or s.get('array')
        vals = [s['value']] + list(s.get('extra_planes') or [])
        fps = {sg: set() for sg in range(NSG)}
        for v in vals:
            for sg in range(NSG):
                fps[sg] |= foot(v, sg, rows(s['rows'], sg), s['cols'])
        out.append(dict(id=s['id'], stage=s['stage'], kind=s['kind'], value=s['value'],
                        extra_planes=list(s.get('extra_planes') or []), array=arr,
                        owner=s['owner'], debug=bool(s.get('debug')), lines=s['lines'],
                        note=s['note'],
                        footprint_per_sg={sg: len(fps[sg]) for sg in range(NSG)},
                        footprint_total=sum(len(f) for f in fps.values()),
                        distinct_elements=len(set().union(*fps.values())),
                        _fps=fps))
    return out


def pair_counts(w, r):
    """(same_owner, cross_owner, distinct) over all (writer sg, reader sg) pairs."""
    if w['array'] != r['array']:
        return None
    same = cross = 0
    union = set()
    pairs = []
    for a in range(NSG):
        for b in range(NSG):
            inter = w['_fps'][a] & r['_fps'][b]
            if not inter:
                continue
            union |= inter
            if a == b:
                same += len(inter)
            else:
                cross += len(inter)
                pairs.append([a, b])
    if not union:
        return None
    return dict(writer_site=w['id'], reader_site=r['id'], array=w['array'],
                writer_stage=w['stage'], reader_stage=r['stage'],
                same_owner=same, cross_owner=cross, distinct=len(union),
                cross_owner_sg_pairs=pairs)

    for (arr, wname) in WINDOWS[arm]:
        vals = WINDOW_VALUES[(arr, wname)]
        inside = [s for s in sites if s['array'] == arr and s['value'] in vals]
        for w in [s for s in inside if s['kind'] == 'W']:
            for r in [s for s in inside if s['kind'] == 'R']:
                pc = pair_counts(w, r)
                if pc is None:
                    continue
                pc.update(window='%s/%s' % (arr, wname), relation='RAW',
                          closer=CLOSERS.get((w['id'], r['id'])))
                rows_out.append(pc)
    for a, b in zip(WINDOWS[arm], WINDOWS[arm][1:]):
        (arr1, w1), (arr2, w2) = a, b
        if arr1 != arr2:
            continue
        s1 = [s for s in sites if s['array'] == arr1 and s['value'] in WINDOW_VALUES[(arr1, w1)]]
        s2 = [s for s in sites if s['array'] == arr2 and s['value'] in WINDOW_VALUES[(arr2, w2)]]
        for r in [s for s in s1 if s['kind'] == 'R']:
            for w in [s for s in s2 if s['kind'] == 'W']:
                pc = pair_counts(w, r)
                if pc is None:
                    continue
                pc.update(window='%s: %s -> %s' % (arr1, w1, w2), relation='WAR',
                          closer=CLOSERS.get((w['id'], r['id'])))
                rows_out.append(pc)
        for w1s in [s for s in s1 if s['kind'] == 'W']:
            for w2s in [s for s in s2 if s['kind'] == 'W']:
                pc = pair_counts(w2s, w1s)
                if pc is None:
                    continue
                pc.update(window='%s: %s -> %s' % (arr1, w1, w2), relation='WAW',
                          closer=CLOSERS.get((w2s['id'], w1s['id'])))
                rows_out.append(pc)
    for r in rows_out:
        if r['closer'] is None:
            missing.append((r['writer_site'], r['reader_site'], r['relation']))
    assert not missing, ('closers not declared', arm, missing)
    for s in sites:
        s.pop('_fps')
    return sites, rows_out


def main():
    parent_text = PARENT_SRC.read_text(encoding='utf-8-sig')
    arms = {'parent': dict(src=PARENT_SRC, text=parent_text, sha=sha(PARENT_SRC))}
    cand_src = CARD / 'candidate.comp'
    if cand_src.exists():
        arms['candidate'] = dict(src=cand_src, text=cand_src.read_text(encoding='utf-8-sig'),
                                 sha=sha(cand_src))

    res = dict(card='n38-f20-arena-shrink', family='F20', ordinal=230, nsubgroups=NSG,
               gpu_used=False,
               parent=dict(source=str(PARENT_SRC), source_sha256=sha(PARENT_SRC),
                           source_sha_expected=PARENT_SRC_SHA,
                           spv=str(PARENT_SPV), spv_sha256=sha(PARENT_SPV),
                           spv_sha_expected=PARENT_SPV_SHA))
    assert res['parent']['source_sha256'] == PARENT_SRC_SHA
    assert res['parent']['spv_sha256'] == PARENT_SPV_SHA

    # ---- parent SPIR-V: Workgroup storage, variable by variable ----------------
    pm = ST.Module(PARENT_SPV)
    by_id = pm.id_to_instruction()
    wg = []
    for op, ops, _, _ in pm.instructions:
        if op == 59 and ops[2] == 4:
            ptr_ops = by_id[ops[0]][1]
            wg.append(dict(name=pm.name_of(ops[1]), bytes=pm.type_size_bytes(ptr_ops[2])))
    res['parent_spv_workgroup'] = dict(variables=wg,
                                       total_bytes=sum(v['bytes'] for v in wg),
                                       storage_bytes=pm.storage_bytes()['Workgroup'])
    assert res['parent_spv_workgroup']['total_bytes'] == 24576

    out, hazards = {}, {}
    for arm, rec in arms.items():
        decl = declarations(rec['text'])
        bad = check_anchors(rec['text'], arm)
        sites, hz = run_arm(arm)
        out[arm] = dict(source=str(rec['src']), source_sha256=rec['sha'],
                        declarations=decl,
                        declared_bytes=sum(d['bytes'] for d in decl.values()),
                        anchor_failures=bad, sites=sites,
                        barrier_calls=rec['text'].count('barrier();'))
        hazards[arm] = hz
        txt = rec['text']
        for h in hz:
            a = h['closer'].get('anchor') or ''
            if a:
                assert a in txt, ('closer anchor missing', arm, h['writer_site'],
                                  h['reader_site'], a[:60])
    res['arms'] = out
    res['hazards'] = hazards

    # ---- the aliased plane: prove the offset expression is character-identical --
    # The candidate carries a longer header, so the comparison is by content, not by line:
    # every parent line that mentions the retired array, with `shY` renamed to `shX`, must
    # appear verbatim in the candidate.  (The order-insensitive multiset form of the same
    # statement is rule I2 of _addr_audit.py.)
    aliased = []
    if 'candidate' in arms:
        plines = parent_text.splitlines()
        clines = arms['candidate']['text'].splitlines()
        for ln, l in enumerate(plines, 1):
            if 'shY' not in l:
                continue
            if l.lstrip().startswith('shared '):
                # the declaration is REMOVED, not renamed: the whole point of the card
                aliased.append(dict(parent_line=ln, parent=l.strip(), kind='declaration',
                                    candidate_has_declaration=any(
                                        'shared float16_t shY' in c for c in clines),
                                    equal_after_rename=not any(
                                        'shared float16_t shY' in c for c in clines)))
                continue
            renamed = l.replace('shY', 'shX')
            hits = [i + 1 for i, c in enumerate(clines) if c == renamed]
            aliased.append(dict(parent_line=ln, parent=l.strip(), kind='access',
                                renamed=renamed.strip(), candidate_lines=hits,
                                found=bool(hits), equal_after_rename=bool(hits)))
        assert aliased and all(a['equal_after_rename'] for a in aliased), \
            'aliased line not identical: %s' % [a for a in aliased if not a['equal_after_rename']]
        assert sum(1 for a in aliased if a['kind'] == 'access') == 5, aliased
    res['aliased_plane_lines'] = aliased

    # ---- byte budget ----------------------------------------------------------
    budget = {arm: dict(arrays=out[arm]['declarations'],
                        total_bytes=out[arm]['declared_bytes']) for arm in arms}
    for p in (65536, 69632, 73727):
        budget.setdefault('residency', {})['P=%d' % p] = {
            arm: p // budget[arm]['total_bytes'] for arm in arms}
    if 'candidate' in arms:
        budget['delta_bytes'] = budget['candidate']['total_bytes'] - budget['parent']['total_bytes']
        budget['saving_pct'] = round(100.0 * -budget['delta_bytes'] / budget['parent']['total_bytes'], 2)
    res['byte_budget'] = budget
    res['occupancy_threshold'] = dict(
        pool_bracket='P in [65536, 73728)  (n36 two-probe bracket: F05 24576->2 / 49152->1 and '
                     'F06 32768->2 / 49152->1; evidence/OCCUPANCY-ELASTICITY*.json)',
        max_bytes_for_three=65536 // 3,
        parent_bytes=budget['parent']['total_bytes'],
        parent_residency=65536 // budget['parent']['total_bytes'],
        candidate_bytes=budget.get('candidate', {}).get('total_bytes'),
        candidate_residency=(65536 // budget['candidate']['total_bytes']
                             if 'candidate' in arms else None),
        note='24576 is the SPIR-V and captured-ISA truth; the plan ledger field says 29184 = '
             '24576 + 4608 (stale, see n30 STAGE-BLOCK-MATRIX.md 1.1)')
    if 'candidate' in arms:
        assert budget['candidate']['total_bytes'] <= res['occupancy_threshold']['max_bytes_for_three']

    (CARD / 'build').mkdir(exist_ok=True)
    (CARD / 'build/lifetime.json').write_text(json.dumps(res, indent=1), encoding='utf-8')

    print('parent declarations:', {k: (v['halves'], v['bytes'])
                                   for k, v in out['parent']['declarations'].items()})
    print('parent declared bytes:', out['parent']['declared_bytes'],
          '| SPIR-V Workgroup:', res['parent_spv_workgroup']['total_bytes'],
          '| variables:', [(v['name'], v['bytes']) for v in wg])
    if 'candidate' in arms:
        print('candidate declarations:', {k: (v['halves'], v['bytes'])
                                          for k, v in out['candidate']['declarations'].items()})
        print('candidate declared bytes:', out['candidate']['declared_bytes'],
              'delta', budget['delta_bytes'], 'saving %', budget['saving_pct'])
        print('aliased lines identical after rename:',
              all(a['equal_after_rename'] for a in aliased))
    print('anchor failures:', {a: out[a]['anchor_failures'] for a in arms})
    print('residency:', json.dumps(budget['residency']))
    for arm in arms:
        print('-' * 26, 'hazards', arm)
        for h in hazards[arm]:
            print('%-11s -> %-11s %-4s %-26s same=%-5d cross=%-5d %-18s %s' %
                  (h['writer_site'], h['reader_site'], h['relation'], h['window'],
                   h['same_owner'], h['cross_owner'], h['closer']['kind'], h['closer'].get('at')))


if __name__ == '__main__':
    main()
