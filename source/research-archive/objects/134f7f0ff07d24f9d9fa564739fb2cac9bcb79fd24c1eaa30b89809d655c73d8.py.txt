from pathlib import Path
import json,hashlib,shutil,configparser,datetime
base=Path('@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024')
game=Path('@NR_GAME@/bin/x64')
pkg=Path('@NR_WORKSPACE@/work/github-experimental-game-install-20261024/package')
small=base/'game-latest-20261028/small'
def sha(p):return hashlib.sha256(p.read_bytes()).hexdigest()
m=json.loads((small/'manifest/port_network_manifest.json').read_text())
assert (m['rows'],m['passes'],m['buffers'])==(76,108,222)
assert sha(small/'bin/nrvk_bridge.dll')=='337b43490b1ea9136ecd6ca300056e469d076430a1c15c455132b7981e94e316'
checked=0
for section in ('network_shaders','bridge_shaders','cold'):
 for a in m[section].values():
  p=small/'manifest'/a['path']
  assert p.stat().st_size==a['bytes'] and sha(p)==a['sha256'],str(p)
  checked+=1
stamp=datetime.datetime.now(datetime.timezone.utc).strftime('%Y%m%dT%H%M%SZ')
out=base/('small-install-'+stamp);out.mkdir();(out/'before').mkdir()
paths={'nrvk_bridge.dll':game/'nrvk_bridge.dll','nr_bridge.ini':game/'plugins/nr_bridge.ini','nr_panel.ini':pkg/'config/nr_panel.ini','panel-config.json':pkg/'panel/panel-config.json','nr_control_panel.py':pkg/'panel/nr_control_panel.py'}
before={k:sha(p) for k,p in paths.items()}
for k,p in paths.items():shutil.copy2(p,out/'before'/k)
asi=sha(game/'plugins/CyberpunkNRBridge.asi')
full=sha(base/'game-latest-20261028/full/bin/nrvk_bridge.dll')
shutil.copy2(small/'bin/nrvk_bridge.dll',paths['nrvk_bridge.dll'])
cfg=configparser.ConfigParser();cfg.read(paths['nr_bridge.ini'],encoding='utf-8-sig')
cfg['bridge']['manifest_dir']=(small/'manifest').as_posix()
with paths['nr_bridge.ini'].open('w',encoding='utf8') as f:cfg.write(f)
cfg=configparser.ConfigParser();cfg.read(paths['nr_panel.ini'],encoding='utf-8-sig')
cfg['NR']['backend']='small';cfg['Backends']['small']=(small/'manifest').as_posix()
with paths['nr_panel.ini'].open('w',encoding='utf8') as f:cfg.write(f)
p=paths['panel-config.json'];settings=json.loads(p.read_text(encoding='utf-8-sig'));settings['manifest']=(small/'manifest').as_posix();p.write_text(json.dumps(settings,ensure_ascii=False,indent=2),encoding='utf8')
p=paths['nr_control_panel.py'];s=p.read_text(encoding='utf8');s=s.replace('Small（原版）','Small（优化版）');p.write_text(s,encoding='utf8')
assert sha(paths['nrvk_bridge.dll'])==sha(small/'bin/nrvk_bridge.dll')
assert sha(game/'plugins/CyberpunkNRBridge.asi')==asi
assert sha(base/'game-latest-20261028/full/bin/nrvk_bridge.dll')==full
receipt={'status':'installed_not_yet_game_verified','checked_manifest_assets':checked,'before':before,'after':{k:sha(p) for k,p in paths.items()},'preserved_asi':asi,'preserved_full_dll':full,'small_manifest':sha(small/'manifest/port_network_manifest.json'),'small_runtime':sha(small/'manifest/port_network_runtime.txt'),'selected':'small','backup':str(out/'before')}
(out/'INSTALL.json').write_text(json.dumps(receipt,indent=2),encoding='utf8')
print(json.dumps(receipt,indent=2))
