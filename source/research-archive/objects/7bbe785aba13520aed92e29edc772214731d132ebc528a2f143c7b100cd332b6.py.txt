def g_of(site):
    return groups[site['src_text'][0]]


def rewrites_for(site):
    g = g_of(site)
    for s in g['sites']:
        if s['addr'] == site['addr']:
            return s['rewrites_before_next']
    return []


def table(site_list, opname):
    lines = ['| # | site | op | source operand (text) | reaching def | def instruction | first consumer | writes to that register before the next same-text site |',
             '|---|---|---|---|---|---|---|---|']
    for i, s in enumerate(site_list, 1):
        deftext = (s['producer_text'] or '').split('*/')[-1].strip()[:46].replace('|', '\\|')
        lines.append('| %d | `%s` | `%s` | `%s` | `%s` | `%s` | `%s` (%s) | %s |'
                     % (i, s['addr'], opname, s['src_text'][0], s['producer_addr'],
                        deftext,
                        s['consumer_addrs'][0] if s['consumer_addrs'] else '-',
                        s['consumer_ops'][0] if s['consumer_ops'] else '-',
                        rewrites_for(s)))
    return '\n'.join(lines)


def group_table():
    lines = ['| source operand text | sites | distinct reaching defs | site addresses | writes to that register between adjacent sites |',
             '|---|---|---|---|---|']
    for txt, g in sorted(groups.items(), key=lambda kv: (-kv[1]['n'], kv[0])):
        rows = ['`%s` (prod `%s`)' % (s['addr'], s['producer']) for s in g['sites']]
        rw = [('%s' % s['rewrites_before_next']) if s['rewrites_before_next'] else '-'
              for s in g['sites']]
        lines.append('| `%s` | %d | **%d** | %s | %s |'
                     % (txt, g['n'], g['distinct_producers'],
                        '<br>'.join(rows), '<br>'.join(rw)))
    return '\n'.join(lines)


def crosscheck_table():
    lines = ['| op | source text (census group) | sites | distinct reaching defs | same-value pairs | distinct leaf signatures | writes to the source dword between adjacent sites |',
             '|---|---|---|---|---|---|---|']
    for g in X['groups']:
        rw = []
        for w in g['rewrites']:
            rw.append('%s->%s:%d' % (w['a_addr'], w['b_addr'], len(w['rewrites'])))
        lines.append('| `%s` | `%s` | %d | **%d** | **%d** | %d | %s |'
                     % (g['op'], g['src_text'], g['copies'], g['distinct_producers'],
                        len(g['same_value_pairs']), g['distinct_leaf_signatures'],
                        ' '.join(rw) if rw else '-'))
    return '\n'.join(lines)


def aluwr_table():
    """how much of the measured PASS ALUWR the expensive ops explain"""
    w_alu = A['w_alu']
    aluwr = A['aluwr_pass']
    rows = [
        ('24 条贵操作（16 rsqt + 8 inv，全 always-run）', 720.0,
         A['expensive']['ll_ms_all']),
        ('其中「暴露」的 15 条（普查口径 distance<=2）', 720.0 * 15 / 24,
         A['expensive']['ll_ms_all'] * 15 / 24),
        ('本卡依赖图口径 <=2 的 17 条', 720.0 * 17 / 24,
         A['expensive']['ll_ms_all'] * 17 / 24),
        ('其中 3-10 的 6 条', 720.0 * 6 / 24, A['expensive']['ll_ms_all'] * 6 / 24),
        ('其中 >=20 的 1 条', 720.0 * 1 / 24, A['expensive']['ll_ms_all'] * 1 / 24),
    ]
    lines = ['| 集合 | L24 当量 | 摊到的 ALUWR (ms/PASS) | 占该 PASS ALUWR %.6f | 占全帧 17.142 ms |'
             % aluwr,
             '|---|---|---|---|---|']
    for name, w, ms in rows:
        lines.append('| %s | %.1f | **%.6f** | %.1f%% | %.3f%% |'
                     % (name, w, ms, 100.0 * ms / aluwr, 100.0 * ms / 17.142))
    return '\n'.join(lines)

## 2. 普查的 8 个「同源」分组，逐组判定

__TAB_GRP__

* **不同达到定义数 = 站点数**：每个文本分组内，站点数 n 与不同定义数完全相等
  （7 组如此，`r20.0` 只有 1 个站点）；
* **相邻站点之间的写入数 >= 1**：除最后一站外，每个站点的 `rewrites_before_next`
  都 >= 1（最小 1，最大 65）。即「同一个 `rX.d` 在两次使用之间被改写过」，
  这正是普查 `proof_requirement` 要排除的情形。

### 2.1 一个具体例子（`r18.0<1;1,0>:f`，4 个站点，4 个不同定义）

```
/* [A000] */ mov r18.0:f  r17.0:hf      <- r17 来自 a0 间接读 (a0 = r29+0x500)
/* [A080] */ math.rsqt r18.0:f r18.0:f  <- 站点 1
/* [BDE0] */ mov r18.0:f  r20.0:hf      <- r20 来自另一条 a0 间接读 (a0 = r1.16+0x480)
/* [BE00] */ math.rsqt r18.0:f r18.0:f  <- 站点 2
/* [C320] */ mov r18.0:f  r20.0:hf      <- 同上形状但 a0 与 r16 不同
/* [C340] */ math.rsqt r18.0:f r18.0:f  <- 站点 3
/* [C820] */ mov r18.0:f  r1.0:hf       <- r1 来自 r16 = 另一条 a0 间接读
/* [C870] */ math.rsqt r18.0:f r18.0:f  <- 站点 4
```

四个站点用的是**同一个物理寄存器号**，但输入分别是 `r17`、`r20`（第一次）、`r20`（第二次，源不同）、`r1`，
且 4 次 a0 间接读的地址不同。**不是同一个值。**

### 2.2 方法自检（为什么这不是「全都不同」的废话）

签名函数先在**不建模间接地址**的版本上跑，它把 3 个站点（`0xa490 / 0xa980 / 0xaeb0`）
和 2 个站点（`0xbe00 / 0xc340`）判成「表达式树相同」——因为 `mov r32.0:f r1.0:hf`
的 `r1.0` 源自 `mov r1.0:uw r30.0:uw`，而 `r30` 源自 `r[a0.0]`（寄存器间接读），
不建模 a0 时这些读看起来都是「无源」的常量。
把 a0 的值（间接地址）纳入签名后，两组**全部裂开**，剩余同值组 = **0**。

=> 签名的区分力是经过反例验证的，不是恒真。`REDUNDANCY.json.sites[]` 里每个站点都带
`producer_addr / producer_op / consumer_addrs`，可逐条复核。

## 3. 源码层论证：这些值本来就来自不同数据

ISA 的 16 条 `math.rsqt` = `full_norm()` 的 2 条 `inversesqrt` x **8 个调用点**（IGC 内联）。
8 个调用点的实参（`candidate.comp:738..783`）：

| 调用 | 实参 | 分量 | 结论 |
|---|---|---|---|
| 1..4 | `full_norm(q00[c],q01[c],k00[c],k01[c])` | c=0..3 | 4 组不同分量 |
| 5..8 | `full_norm(q10[c],q11[c],k10[c],k11[c])` | c=0..3 | 另一组累加器 |

`q00/q01/k00/k01` 与 `q10/q11/k10/k11` 是**不同的 DPAS 累加器**（`z00/z01/z10/z11`，A 操作数
`ya00/ya01` 与 `ya10/ya11` 不同、B 操作数（权重）相同），即**不同 token 行的投影**。
16 条开方的被开方数因此是 16 个不同的规约结果，没有「同一值算两次」。

8 条 `math.inv` 同理：4 个 softmax 分母组 x 2 个分量（`candidate.comp:959/996/1037/1074`），
四组分别规约 `p00..p03` 的 `[0],[1]`、`[2],[3]`、`p10..p13` 的 `[0],[1]`、`[2],[3]` ——
**互不相交的分量**，8 个分母值互不相同。

## 4. 消费者清单与「读同一值」判定

| 贵操作 | 站点数 | 消费者（每个站点） | 是否有两个站点被同一消费者读取 |
|---|---|---|---|
| `math.rsqt` | 16 | 每条只被紧随其后的 `mov (:f->:hf)` 读取，之后进入该分量的 scale 乘法 | **否**（16 个消费者地址两两不同） |
| `math.inv` | 8 | 每条只被紧随其后的 `mov (:f->:hf)` 读取，之后进入该组的 `p...=p...*inv` 乘法 | **否**（8 个消费者地址两两不同） |

消费者地址见第 1 节表的最后一列与 `REDUNDANCY.json.sites[].consumer_addrs`；
`cards/n33-f02-latency/_c2_consumers.py` 的消费者计数口径（每个生产者的 live 读者数）在两套实现下都为 1。
「让一组算一次、多处复用」的前提是**多个消费者读同一个生产者**；本核每个贵操作
只有 1 个私有消费者，结构上不存在可共享的读点。

## 5. 判定与后果（分诊 BLOCKED）

| 项 | 普查声称 | 本卡实测 | 依据 |
|---|---|---|---|
| `math.rsqt` 冗余 | 9 / 16 | **0** | 16 个达到定义互异 + 16 个消费者互异 + 8 个调用点实参互异 |
| `math.inv` 冗余 | 5 / 8 | **0** | 8 个达到定义互异 + 4 组规约的分量互斥 |
| 可删后条数 | rsqt->4 / inv->3 | **16 / 8（不可删）** | 删任一条都会用另一个分量的值替换本分量的值 |
| 每 lane 独立链数 | 不得下降 | **__NCH__ -> __NCH__**（无改动） | 无候选，图不变 |

分诊：**A 类（实现未落地 / 无工作可做）**。这不是「机制无效」，而是**本核没有冗余副本**：
普查的 14 条「冗余」是 GRF 号复用的假象，两套独立实现（n34 表达式树签名、n33 reaching-def/leaf）
与源码侧三方一致给出 0。

**两条路都已关闭**：

* **删除路线**（本节）：0 条真冗余，删不了；
* **延迟隐藏路线**：n32 已实测（L26）源码重排改不动 IGC 的基本块内调度；本核另有一条更硬的结构性障碍——
  16 条 `math.rsqt` 全在 `full_norm` 一个 SPIR-V 函数体内（模块里只有 **2 条 OpExtInst Inversesqrt + 8 个调用点**），
  重排 8 个调用点在 **SPIR-V 层面根本不改变 rsqt 的位置**，见 `SPV-STRUCTURE.json`。

**上限**：见 §0.1——24 条贵操作合计 __LLMS__ ms/PASS（该 PASS ALUWR 的 18.0 %，全帧 0.087 %）；
15 条暴露点全藏住也只值 __EXMS15__ ms/PASS（17 条口径 __EXMS__）。无论哪条路，都低于 L13 的 0.03 ms 门 3.2 倍（0.03 / 0.0093）。
