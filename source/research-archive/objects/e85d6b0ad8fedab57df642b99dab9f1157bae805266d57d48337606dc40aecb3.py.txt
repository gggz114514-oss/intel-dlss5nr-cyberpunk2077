"""Dump ISA windows around the expensive ops of the three F05 kernels.

- math.rsqt / math.inv / sel census with a +/-N window
- mov type split (:hf<-:f narrow, :f<-:hf widen, :hf<-:hf, :f<-:f)
Reuses _regions.py for the debug-skip classification.
"""
import sys, re
from collections import Counter
from pathlib import Path

HERE = Path(__file__).parent
sys.path.insert(0, str(HERE))
import _regions as R

WIN = int(sys.argv[1]) if len(sys.argv) > 1 else 5
KIND = sys.argv[2] if len(sys.argv) > 2 else "rsqt"


def show(ins, i, lo, hi):
    for j in range(max(0, i - lo), min(len(ins), i + hi + 1)):
        it = ins[j]
        mark = ">>" if j == i else "  "
        print("%s [%06X] %-4s %s" % (mark, it["addr"], "L" if it["live"] else "D", it["text"]))


def castsplit(ins):
    c = Counter()
    for it in ins:
        if not it["live"]:
            continue
        t = it["text"]
        if it["op"] == "mov":
            m = re.search(r":([a-z0-9]+)\s*<\s*-\s*:([a-z0-9]+)", t)
            if m:
                c["mov.%s<-%s" % (m.group(1), m.group(2))] += 1
            else:
                c["mov (no type pair)"] += 1
        elif ":" in t and "<-" in t:
            m = re.search(r":([a-z0-9]+)\s*<\s*-\s*:([a-z0-9]+)", t)
            if m:
                c["%s.%s<-%s" % (it["op"], m.group(1), m.group(2))] += 1
    return c


for w in (64, 128, 256):
    ins, labels = R.load(w)
    ins, guard = R.classify(w, ins, labels)
    live = [i for i in ins if i["live"]]
    print("=" * 78)
    print("c%d  live=%d  guard=%s" % (w, len(live), guard))
    print("-- cast split (live):")
    for k, v in castsplit(ins).most_common(14):
        print("   %-24s %5d" % (k, v))
    tgt = [i for i in live if i["op"] in ("math.rsqt", "math.inv")]
    print("-- %s sites: %d" % (KIND, len(tgt)))
    if KIND in ("rsqt", "inv", "all"):
        for it in tgt:
            show(ins, ins.index(it), WIN, WIN)
            print()
    if KIND == "sel":
        sels = [i for i in live if i["op"] == "sel"]
        print("-- sel sites: %d (first 6)" % len(sels))
        for it in sels[:6]:
            show(ins, ins.index(it), 2, 3)
            print()
