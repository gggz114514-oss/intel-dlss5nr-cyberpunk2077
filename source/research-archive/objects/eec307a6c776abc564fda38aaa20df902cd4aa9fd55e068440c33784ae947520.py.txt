"""nat-index probe 3: pipeline<->pass mapping stability + capture->plan resolution (merged)."""
import json, os, re, sys, hashlib
from pathlib import Path
from collections import Counter, defaultdict

W = Path('@NR_WORKSPACE@')
A = W / 'work/codex-astra-fp16-int8-audit-20261024'
H = Path('@NR_USER_HOME@/Documents/Codex/2026-10-24/nr-fp16-int8-astra-audit-20261024/handoff/dsh-fullnet-20261027')
BASELINE = A / 'fullnet-mechanism-transfer-20261026/f10-weight-return-separation-20261027/plan.txt'
BASE_SHA = 'a024c3c5aa1d1432ab3f0591bf03a897880d86be2c99d6de1e1492a2a7b2befe'

ISA_RE = re.compile(r'^CS_asm([0-9a-f]{16})_simd(\d+)_([0-9a-f]{16})_(\d+)_orig\.asm$', re.I)


def sha(p):
    h = hashlib.sha256()
    with open(p, 'rb') as f:
        for b in iter(lambda: f.read(1 << 20), b''):
            h.update(b)
    return h.hexdigest()


def main():
    rows = json.loads((H / 'CURRENT-232-ROWS.json').read_text(encoding='utf-8-sig'))
    print('rows:', len(rows))
    print('baseline plan sha on disk:', sha(BASELINE), 'matches constant:', sha(BASELINE) == BASE_SHA)

    # ---- pipeline-statistics with 232 pass indices
    ppis = Counter()
    ppi_files = {}
    for root in (W / 'work', W / 'run'):
        for dp, dn, fn in os.walk(root):
            if 'pipeline-statistics.json' in fn:
                p = Path(dp) / 'pipeline-statistics.json'
                try:
                    j = json.loads(p.read_text(encoding='utf-8-sig', errors='ignore'))
                except Exception:
                    continue
                ppi = j.get('pass_pipeline_indices')
                if isinstance(ppi, list) and len(ppi) == 232:
                    key = tuple(ppi)
                    ppis[key] += 1
                    ppi_files.setdefault(key, []).append(str(p))
    print('\npipeline-statistics with 232 pass indices:', sum(ppis.values()),
          ' distinct ppi vectors:', len(ppis))
    for k, n in ppis.most_common(5):
        print('  n=%d  first=%s  npipelines=%d' % (n, list(k[:12]), len(set(k))))
        print('      e.g.', ppi_files[k][0])

    # choose dominant
    best = ppis.most_common(1)[0][0]
    print('\n--- dominant mapping: pipeline -> pass ordinals ---')
    by_pipe = defaultdict(list)
    for o, pi in enumerate(best):
        by_pipe[pi].append(o)
    print('pipelines:', len(by_pipe), 'passes:', len(best))
    print('passes per pipeline histogram:', Counter(len(v) for v in by_pipe.values()))
    for pi in sorted(by_pipe):
        ords = by_pipe[pi]
        fams = sorted({rows[o]['family'] for o in ords})
        print('  p%-2d n=%-3d fams=%-16s first=%s' % (pi, len(ords), ','.join(fams)[:16],
                                                      rows[ords[0]]['name']))

    # ---- capture dirs & plan resolution
    print('\n--- capture dirs ---')
    asm_dirs = {}
    for root in (W / 'work', W / 'run', W / 'candidates'):
        if not root.exists():
            continue
        for dp, dn, fn in os.walk(root):
            hits = [f for f in fn if ISA_RE.match(f)]
            if hits:
                asm_dirs[Path(dp)] = sorted(int(ISA_RE.match(f).group(4)) for f in hits)
    print('asm session dirs:', len(asm_dirs))
    print('distinct pipeline-index sets:', Counter(tuple(v) for v in asm_dirs.values()).most_common(5))

    # plans referenced from directories (heuristic: a plan*.txt in the capture's card dir)
    unres = []
    for d in sorted(asm_dirs):
        card = d.parent.parent
        cands = sorted(card.glob('*.txt'))
        plans = [c for c in cands if c.name.startswith('plan')]
        if not plans:
            unres.append(d)
    print('session dirs whose card dir has no plan*.txt:', len(unres))
    for d in unres[:25]:
        print('   ', d.relative_to(W))


if __name__ == '__main__':
    main()
