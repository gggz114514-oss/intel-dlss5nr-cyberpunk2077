"""Dump the block62..66 dispatch chain with resolved buffer names + scale taps.

Read-only over B. Writes only under C00-c64-b65/.
"""
import sys, json, hashlib
from pathlib import Path

B = Path("@NR_WORKSPACE@")
W = B / "work/dsh-fullnet-static-20261029"
sys.path.insert(0, str(W / "tools"))
import planlib

PLAN = W / "C00-color-baseline/DIAG-chain62-66-34189.txt"


def main():
    plan = planlib.Plan(PLAN)
    probs = plan.validate()
    out = {"plan": str(PLAN), "sha256": planlib.sha256_file(PLAN),
           "header": plan.header, "validate_problems": probs,
           "n_buffers": len(plan.buffers), "n_passes": len(plan.passes)}
    rows = []
    for p in plan.passes:
        if not any(k in p.name for k in ("c64_b62", "c64_b63", "c64_b64", "c64_b65", "c64_b66",
                                          "block62_swin_body", "block63_swin_body",
                                          "block64_swin_body", "block65_swin_body",
                                          "block66_swin_body", "f2c32_block66")):
            continue
        r = {"ordinal": p.ordinal, "group": p.group, "name": p.name,
             "spv": p.spv, "spv_sha256": (planlib.sha256_file(p.spv) if Path(p.spv).exists() else "MISSING"),
             "spv_bytes": (Path(p.spv).stat().st_size if Path(p.spv).exists() else None),
             "tail": p.tail_tokens,
             "binds": [{"binding": b.binding, "buffer": b.buffer,
                        "name": plan.buffers[b.buffer].name,
                        "buf_bytes": plan.buffers[b.buffer].bytes,
                        "file": plan.buffers[b.buffer].file,
                        "offset": b.offset, "bytes": b.bytes} for b in p.binds],
             "push": p.push}
        rows.append(r)
    out["passes"] = rows
    # named buffer index lookup for the whole block65 kit
    out["block65_buffers"] = {
        n: {"index": plan.buffers_by_name[n].index, "bytes": plan.buffers_by_name[n].bytes,
            "file": plan.buffers_by_name[n].file}
        for n in plan.buffers_by_name if ("b65" in n or "block65" in n or "swin_65" in n)}
    print(json.dumps(out, indent=1))


if __name__ == "__main__":
    main()
