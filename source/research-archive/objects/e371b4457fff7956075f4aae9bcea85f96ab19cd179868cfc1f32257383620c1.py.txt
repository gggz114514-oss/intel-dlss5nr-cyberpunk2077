"""Dump the memory-side SPIR-V instructions (OpVariable/OpLoad/OpStore/OpAccessChain)
of parent vs candidate so the +5/+6/+6 delta is explained instruction by instruction.

Prints only compact lines: id, opcode, storage class / pointer id + its OpName.
"""
import sys
from pathlib import Path

CARD = Path(__file__).resolve().parent.parent
W = Path('@NR_WORKSPACE@')
sys.path.insert(0, str(W / 'build/zcode-fp16-compute-backend-20261021/scripts'))
import spv_tools_local as ST

ST_NAMES = {0: 'UniformConstant', 1: 'Input', 2: 'Uniform', 3: 'Output', 4: 'Workgroup',
            5: 'CrossWorkgroup', 6: 'Private', 7: 'Function', 8: 'Generic', 9: 'PushConstant',
            10: 'AtomicCounter', 11: 'Image', 12: 'StorageBuffer'}
NAMES = {59: 'Var', 60: 'ImageTexelPtr', 61: 'Load', 62: 'Store', 63: 'CopyMem',
         65: 'AccessChain', 66: 'InBoundsAccessChain', 67: 'PtrAccessChain',
         71: 'Decorate', 74: 'GroupDecorate', 5: 'Name', 6: 'MemberName'}


def dump(path):
    m = ST.Module(path)
    names = {}
    for op, o, _, _ in m.instructions:
        if op == 5:
            i = o[0]
            parts = []
            for w in o[1:]:
                parts.append(w.to_bytes(4, 'little'))
            names[i] = b''.join(parts).split(b'\0')[0].decode('utf-8', 'replace')
    out = []
    for op, o, _, _ in m.instructions:
        if op == 59:
            out.append('Var id=%d ptrtype=%d storage=%s name=%r' % (o[1], o[0], ST_NAMES[o[2]], names.get(o[1], '')))
        elif op == 61:
            out.append('Load id=%d ptr=%d ptrname=%r' % (o[1], o[2], names.get(o[2], '')))
        elif op == 62:
            out.append('Store ptr=%d ptrname=%r' % (o[0], names.get(o[0], '')))
        elif op == 65:
            out.append('AccessChain id=%d base=%d basename=%r' % (o[1], o[2], names.get(o[2], '')))
    return out, names


for tag in ('parent', 'candidate'):
    p = CARD / ('parent-rebuild.spv' if tag == 'parent' else 'candidate.spv')
    lines, names = dump(p)
    print('=====', tag, p.name, len(lines))
    for l in lines:
        print(' ', l)
