"""n39-f06-c256-slack: assemble BUILD.json and GPU-REQUEST.json from the machine artifacts.

Everything here is read from build/*.json so no hash is typed by hand, and the last step
re-verifies that every path referenced still exists with the recorded sha256.

Writes BUILD.json + GPU-REQUEST.json.
"""
import json
import pathlib
import sys

CARD = pathlib.Path(__file__).resolve().parent
sys.path.insert(0, str(CARD.parents[1]))
import COMMON as K                      # noqa: E402


def read(rel):
    return json.loads((CARD / rel).read_text(encoding='utf-8'))


def main():
    repro = read('build/repro.json')
    build = read('build/build.json')
    plan = read('build/plan.json')
    audit = read('build/addr-audit.json')
    negc = read('build/audit-negcontrol.json')
    isa = read('build/isa-parent.json')
    proof = read('PROOF.json')

    f06p = repro['groups']['f06-c256']
    f18p = repro['groups']['f18-c256']
    b06 = build['pairs']['f06-c256']
    b18 = build['pairs']['f18-c256']
    pkg = plan['plans']['package16']
    pad = plan['plans']['padback']
    f06only = plan['plans']['f06only15']

    def ident(rel):
        return K.ident(CARD / rel)

    build_json = dict(
        card='n39-f06-c256-slack',
        family='F06 C256 (15 PASS) + F18 C256 (1 PASS)',
        status='BUILT_AWAITING_GPU',
        chain_head=plan['chain_head'],
        parent=dict(
            plan=plan['chain_head'],
            f06=dict(spv=dict(path=K.row_by_ordinal(38)['spv'],
                              sha256=f06p['parent_sha_expected']),
                     source=f06p['source'], rows=f06p['rows']),
            f18=dict(spv=dict(path=K.row_by_ordinal(168)['spv'],
                              sha256=f18p['parent_sha_expected']),
                     source=f18p['source'], rows=f18p['rows']),
            parent_rebuild=dict(f06=ident('parent-rebuild.spv'), f18=ident('parent-f18-rebuild.spv'),
                                byte_match=bool(f06p['raw_match'] and f18p['raw_match']),
                                recipe=dict(f06=f06p['recipe_used'], f18=f18p['recipe_used'])),
            recipe_detail=dict(f06=f06p['recipe'], f18=f18p['recipe']),
        ),
        candidate=dict(
            f06=dict(spv=ident('candidate.spv'), source=ident('candidate.comp'),
                     workgroup_bytes=b06['candidate']['workgroup_bytes'],
                     workgroup_vars=b06['candidate']['workgroup_vars'],
                     instructions=b06['candidate']['instructions'],
                     barriers=b06['candidate']['barriers']),
            f18=dict(spv=ident('candidate-f18.spv'), source=ident('candidate-f18.comp'),
                     workgroup_bytes=b18['candidate']['workgroup_bytes'],
                     workgroup_vars=b18['candidate']['workgroup_vars'],
                     instructions=b18['candidate']['instructions'],
                     barriers=b18['candidate']['barriers']),
            padback=dict(spv=ident('candidate-padback.spv'),
                         source=ident('candidate-padback.comp'),
                         workgroup_bytes=build['controls']['padback-vs-candidate']['candidate']['workgroup_bytes'],
                         purpose='pad-back control: 32768 B declared again, copy still deleted'),
        ),
        workgroup_bytes=dict(parent_f06=b06['workgroup_bytes']['parent'],
                             candidate_f06=b06['workgroup_bytes']['candidate'],
                             parent_f18=b18['workgroup_bytes']['parent'],
                             candidate_f18=b18['workgroup_bytes']['candidate'],
                             reduction_bytes=b06['workgroup_bytes']['reduction']),
        float_NC_checked=b06['float_NC_checked'],
        float_NC_note='the recipe is raw mode: neither module carries any NoContraction '
                      'decoration, so the checked set is empty on both sides and trivially equal',
        fp_multiset_identical=b06['fp_multiset']['identical'],
        dpas=dict(parent=b06['parent']['dpas'], candidate=b06['candidate']['dpas']),
        a_operand_pointer_roots=dict(parent=b06['parent']['cm_load_pointer_roots'],
                                     candidate=b06['candidate']['cm_load_pointer_roots']),
        plans={name: dict(path=v['path'], sha256=v['sha256'], replaced=v['replaced'],
                          differing_lines=v['differing_lines'],
                          cpu_check=v['cpu_check'].get('stdout', '').strip()[:200],
                          cpu_check_ok=v['cpu_check']['ok'])
               for name, v in plan['plans'].items()},
        cpu_check=pkg['cpu_check']['stdout'].strip(),
        address_audit={k: audit[k]['verdict'] for k in audit},
        address_audit_negcontrol=dict(caught=negc['caught'], tested=negc['tested']),
        proof=dict(checks=len(proof['checks']), passed=proof['passed'], failed=proof['failed']),
        parent_isa=dict(source=isa['parent_f06_c256']['path'],
                        sha256=isa['parent_f06_c256']['sha256'],
                        instructions=isa['parent_f06_c256']['instructions'],
                        dpas=isa['parent_f06_c256']['dpas'],
                        sends=isa['parent_f06_c256']['sends'],
                        syncs=isa['parent_f06_c256']['syncs'],
                        max_r=isa['parent_f06_c256']['max_r'],
                        messages=isa['parent_f06_c256']['messages']),
        precedent_f05_c256=isa['precedent_f05_c256']['messages'],
        no_gpu_no_lock='no Vulkan task, no GPU lock, no deployment: this card is CPU only',
    )
    K.write_json(CARD / 'BUILD.json', build_json)

    ords = f06p['rows']['ordinals'] + f18p['rows']['ordinals']
    spv_per_ordinal = {str(o): dict(candidate_spv=(CARD / 'candidate.spv').as_posix())
                       for o in f06p['rows']['ordinals']}
    spv_per_ordinal[str(f18p['rows']['ordinals'][0])] = dict(
        candidate_spv=(CARD / 'candidate-f18.spv').as_posix())

    req = dict(
        card='n39-f06-c256-slack',
        family='F06 C256 (+F18 C256)',
        status='BUILT_AWAITING_GPU',
        one_line='Delete the 32768 B Workgroup staging array (a byte-for-byte copy of the '
                 'read-only gT/gM tile) and load the DPAS A operand straight from the global '
                 'tile at the same offsets: SPIR-V Workgroup 32768 -> 0 B, so SLM stops being '
                 'the residency constraint (2 -> >= 4 workgroups per SM), with the same 1024 A '
                 'messages, the same values and one fewer barrier.',
        triage_if_negative='A if the captured ISA still reports Workgroup Memory 32768 B, or '
                           'load.slm.d32x8t is still 1024, or the A reads did not appear as '
                           'load.ugm.d32x8t (NATIVE-PREDICTION.md H0/O1/O2) -- and then no '
                           'millisecond inference is allowed. B if the A-read move to L1/L2 costs '
                           'more than the residency step buys (L2 read amplification ~8x, or a '
                           'frame that is slower with both gold gates green). C/D only if H0, O1-O10 '
                           'and both gold gates all land and the frame does not respond.',
        delivery='single tier: 16 rows = 15 F06 C256 + 1 F18 C256 (both point at their own '
                 'candidate SPV). plan-f06only.txt is the 15-row attribution subset and '
                 'plan-padback.txt is the pad-back control arm (32768 B declared again, copy '
                 'still deleted). ordinal 193 is NOT part of the package: its plan field says '
                 '32768 B but its module declares 16384 B (L35 / evidence/TRUE-SLM.json), so the '
                 'card text "17 rows" is 16 rows in reality.',
        shader_group=dict(
            spv_sha256_baseline=[f06p['parent_sha_expected'], f18p['parent_sha_expected']],
            pass_ordinals=ords,
            group='F06 C256 (%d PASS, grid 11x6 or 11x7, 1067 workgroups/frame) + F18 C256 (%d PASS, '
                  '66 workgroups/frame)' % (len(f06p['rows']['ordinals']), len(f18p['rows']['ordinals'])),
            candidate_spv=(CARD / 'candidate.spv').as_posix(),
            candidate_spv_sha256=K.sha(CARD / 'candidate.spv'),
            candidate_spv_f18=(CARD / 'candidate-f18.spv').as_posix(),
            candidate_spv_f18_sha256=K.sha(CARD / 'candidate-f18.spv'),
            candidate_spv_padback=(CARD / 'candidate-padback.spv').as_posix(),
            candidate_spv_padback_sha256=K.sha(CARD / 'candidate-padback.spv'),
            parent_spv_sha256=dict(f06=f06p['parent_sha_expected'], f18=f18p['parent_sha_expected']),
            parent_source_sha256=dict(f06=f06p['source']['sha256'], f18=f18p['source']['sha256']),
            candidate_source_sha256=dict(f06=K.sha(CARD / 'candidate.comp'),
                                         f18=K.sha(CARD / 'candidate-f18.comp')),
            workgroup_memory_bytes=dict(baseline_f06=32768, candidate_f06=0,
                                        baseline_f18=32768, candidate_f18=0),
            float_NC_checked=b06['float_NC_checked'],
            grids=sorted({tuple(r['header_fields'][4:7]) for r in K.rows()
                          if r['spv_sha256'] == f06p['parent_sha_expected']}),
            passes_replaced=16,
        ),
        plan=dict(path=pkg['path'], sha256=pkg['sha256'],
                  baseline_chain_head_plan=plan['chain_head']['path'],
                  baseline_chain_head_sha256=plan['chain_head']['sha256']),
        aux_plans=dict(padback=dict(path=pad['path'], sha256=pad['sha256'], replaced=pad['replaced']),
                       f06only=dict(path=f06only['path'], sha256=f06only['sha256'],
                                    replaced=f06only['replaced'])),
        rebase=dict(mode='per_ordinal'),
        spv_per_ordinal=spv_per_ordinal,
        arms=[dict(id='base', plan='baseline', role='baseline',
                   plan_path=plan['chain_head']['path'],
                   plan_sha256=plan['chain_head']['sha256']),
              dict(id='cand', plan='candidate', role='candidate',
                   plan_path=pkg['path'], plan_sha256=pkg['sha256']),
              dict(id='padback', plan='candidate', role='control',
                   plan_path=pad['path'], plan_sha256=pad['sha256']),
              dict(id='f06only', plan='candidate', role='attribution', optional=True,
                   plan_path=f06only['path'], plan_sha256=f06only['sha256'])],
        gates=['fixed', 'fb2'],
        native=True,
        resources=True,
        ordinary=dict(design='two-arm-6-alternating', reps=30, warm=1, feedback=True),
        hypothesis='If the captured ISA reports Workgroup Memory 0 B for the F06 rows with '
                   'load.slm.d32x8t 1024 -> 0, load.ugm.d32x8t +1024, store.slm.d16u32 16 -> 0, '
                   'fence.slm 1 -> 0, dpas still 256, max_r <= 127 and scratch 0, then the '
                   'staging copy is gone and the occupancy counters should step the group from 2 '
                   'to at least 4 resident workgroups; any frame delta is then the measured value '
                   'of that residency step minus the cost of serving the A operand from L1/L2 '
                   'instead of SLM. If the ISA still says 32768 B that is class A and no '
                   'millisecond inference is permitted.',
        native_observables=[
            'HARD GATE O0 (already offline): module Workgroup 32768 -> 0 B, Workgroup variables '
            '1 -> 0; if the dispatched SPV is not 8f592cf9576f137c... that is class A.',
            'O1 send.slm load.slm.d32x8t.a32 1024 -> 0.  Still 1024 means the A operand still '
            'reads SLM -> class A.',
            'O2 send.ugm load.ugm.d32x8t.* 128 -> ~1152 (+1024).  The F05 C256 kernel in the '
            'accepted chain already runs this shape (512 load.ugm.d32x8t, 0 load.slm.d32x8t).',
            'O3 send.slm store.slm.d16u32.a32 16 -> 0; O4 send.ugm load.ugm.d16u32.a32.bss[a0.0] '
            '18 -> 0 (at most 2); O5 fence.slm.none.group 1 -> 0; O6 gtwy signal barrier 2 -> 1.',
            'O7 dpas 256 -> 256 and O8 weight loads 256 -> 256 and O9 output stores 128/16 -> '
            '128/16 must all be unchanged; anything else means the maths moved.',
            'O10 total send 1597 -> 1561 +/- 12.',
            'O11 instructions 7087 -> 6800..7000; O12 mov <= 1320; O13 first_dpas 356 -> '
            '250..330; O14 max_r <= 127; O15 scratch 0; O16 syncs 1265 -> ~1250.',
            'O17 resident workgroups per SM 2 -> >= 4 (occupancy counters).  If O1-O6 land but '
            'the residency stays 2, the gain claim is not testable and it is class B (resource '
            'bound elsewhere), not "the mechanism failed".',
            'O19 if the resource counters expose L2/DRAM read bytes, they should stay near the '
            'parent (the 32 KiB tile is read once per workgroup); an ~8x amplification means L1 '
            'is not absorbing the 8-subgroup replication -> class B, next step is the half '
            'staging variant in RESUME.txt 5.',
        ],
        gates_detail=dict(
            fixed='--pool --warm 0 --reps 1 -> last frame '
                  '1755e9c2fadba97f08cb5b02b7a4294cde3675379bdec75352287505f7e129d6',
            fb2='--pool --warm 0 --reps 2 --feedback --save-frames -> last frame '
                '9c3e511e97a5879247bc2965792c96dd62928dddfd85a48bbf25af1e807d3536',
            ordinary='--pool --warm 1 --reps 30 --feedback -> last frame '
                     '2b63d4ba1967eebcda61698d538963f59bccaf6d690ef8168f1695a46c5d658d'),
        triage_rules=dict(
            A_implementation_not_landed='ISA Workgroup Memory still 32768 B, or '
                                        'load.slm.d32x8t still 1024, or store.slm/fence.slm still '
                                        'present. No millisecond inference.',
            B_side_effect='max_r > 127, or scratch > 0, or class-A checks land but occupancy stays '
                          '2, or an ~8x L2 read amplification appears. Price it, then try the half '
                          'staging variant.',
            C_no_net_gain='H0/O1-O10 land, both gold gates pass, occupancy >= 4 and ordinary still '
                          'fails to reach 0.03 ms with 6/6 sign agreement -> D INDISTINCT per L13, '
                          'ask for a second batch before closing the axis.',
            E_numeric_failure='gold frame differs: run _addr_audit.py and its 6 negative controls '
                              'first, then check the tile base w*16384 on every A load and on the '
                              'debug echo. Do NOT suspect rounding, K order or reduction trees: '
                              'FP multiset identical, NC 0 = 0, dpas 256 = 256.'),
        expected_frame_gain_ms=dict(
            low=0.02, point=0.10, high=0.30,
            calibration='the same 2 -> 4 residency step measured +0.116668 ms on F05 C256 '
                        '(36 PASS, 2.95 ms of family time) and the reverse probe on THIS group '
                        'costs -0.363969 ms for 2 -> 1 workgroups (evidence/OCCUPANCY-ELASTICITY.json). '
                        'The curve is concave (L38), and this card also deletes real work (32 KiB '
                        'copy + 1 barrier), so the band is wide on purpose.'),
        notes=dict(
            parent_identity='chain head af6c1287...; both parents are handoff rows whose chain-head '
                            'sha equals the baseline sha (GROUPS.md), reproduced byte-exactly from '
                            'source before any edit (build/repro.json).',
            seventeenth_row='ordinal 193 (c3x_body_c128) carries plan field 32768 but its module '
                            'declares 16384 B (L35, evidence/TRUE-SLM.json); it is already at 4 '
                            'residents and is deliberately left untouched.',
            no_gpu_no_lock='no Vulkan task was run by this card and no GPU lock was taken; '
                           'nothing was deployed.',
            padback_rationale='the pad-back arm restores only the declaration (one store per thread '
                              'plus a dead debug read keep the array alive) so that the residency '
                              'step is isolated from the deleted copy work, per L38.',
        ),
    )
    K.write_json(CARD / 'GPU-REQUEST.json', req)

    # ---- consistency: every referenced path exists with the recorded sha
    refs = [(build_json['parent']['parent_rebuild']['f06'], None),
            (build_json['parent']['parent_rebuild']['f18'], None),
            (build_json['candidate']['f06']['spv'], None),
            (build_json['candidate']['f06']['source'], None),
            (build_json['candidate']['f18']['spv'], None),
            (build_json['candidate']['f18']['source'], None),
            (build_json['candidate']['padback']['spv'], None),
            (build_json['candidate']['padback']['source'], None)]
    bad = []
    for rec, _ in refs:
        p = pathlib.Path(rec['path'])
        if not p.exists():
            bad.append(('missing', rec['path']))
        elif K.sha(p) != rec['sha256']:
            bad.append(('sha', rec['path']))
    for name, rec in build_json['plans'].items():
        if K.sha(pathlib.Path(rec['path'])) != rec['sha256']:
            bad.append(('plan sha', name))
    print('BUILD.json + GPU-REQUEST.json written; %d checks' % (len(refs) + len(build_json['plans'])))
    print('inconsistent references:', bad or 'none')
    return 0 if not bad else 1


if __name__ == '__main__':
    sys.exit(main())
