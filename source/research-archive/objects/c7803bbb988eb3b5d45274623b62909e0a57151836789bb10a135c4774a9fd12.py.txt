import re, collections, json
from pathlib import Path
F = Path(r"@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/fullnet-mechanism-transfer-20261026/f05-k-consumer-tile/c256/captures/native-f20-pixel-owner/ISA/CS_asmfd78c91bb8744f0f_simd32_000000000000000e_13_orig.asm")
lines = F.read_text(encoding="utf-8", errors="replace").splitlines()

def desc(l):
    if "//" not in l:
        return None
    cmt = l.split("//", 1)[1].strip()
    m = re.search(r"rd:\d+; ([a-z0-9_.]+)", cmt)
    return m.group(1) if m else None

c = collections.Counter()
recs = []
for i, l in enumerate(lines):
    if not re.search(r"\bsend\b", l):
        continue
    d = desc(l)
    c[d] += 1
    # destination register(s): first rd:1 -> "send.x (1|M0) rDst rSrc null:0 off desc"
    m = re.search(r"send\.(\w+)\s+\((\d+)\|M0\)\s+(r\d+)\s+(r\d+)", l)
    recs.append(dict(line=i, kind=(m.group(1) if m else "?"),
                     dst=(m.group(3) if m else None),
                     desc=d, text=l.strip()))
print("== send descriptors ==")
for k, v in c.most_common():
    print("%5d %s" % (v, k))
print()
print("n sends parsed:", len(recs))

# weight loads: load.ugm.d32x8t / d32x4 (global). Identify by dst registers and
# by the "dup address" structure: count distinct (desc, imm) pairs.
imm = re.compile(r"\[A\+0x([0-9A-Fa-f]+)\]|0x([0-9A-Fa-f]{6,})")
def key(r):
    return (r["desc"], r["text"].split("//")[1].strip())
kc = collections.Counter(key(r) for r in recs if r["desc"] and r["desc"].startswith("load.ugm"))
print()
print("== global load (load.ugm) descriptor histogram ==")
for k, v in collections.Counter(r["desc"] for r in recs if r["desc"] and r["desc"].startswith("load.ugm")).most_common():
    print("%5d %s" % (v, k))
print()
print("== the 6 QKV weight-load descriptors and how often each appears ==")
for d in sorted(set(r["desc"] for r in recs if r["desc"] and r["desc"].startswith("load.ugm"))):
    print("%5d %s" % (collections.Counter(r["desc"] for r in recs if r["desc"]).get(d, 0), d))
json.dump(recs, open("_build/isa_sends.json", "w"), indent=1)
