def sha(p):
    return hashlib.sha256(open(p, "rb").read()).hexdigest()


def info(p):
    return {"path": p.replace("\\", "/"), "bytes": os.path.getsize(p), "sha256": sha(p)}

def build_receipt():
    parent_src = CARD + "/source/parent"
    cand_src = CARD + "/source/candidate"
    out = CARD + "/spv/r04_c64_pv_i8.spv"
    rep = CARD + "/spv/parent_repro.spv"
    val = subprocess.run([VALIDATOR, out], capture_output=True, text=True)
    valp = subprocess.run([VALIDATOR, rep], capture_output=True, text=True)
    lint = subprocess.run([PYPATH, LINT, CARD + "/plan-r04-g3-base.txt"],
                          capture_output=True, text=True)
    return {
        "status": "OFFLINE_COMPILE + CPU ONLY; no GPU, no pipeline, no GPU.lock",
        "qualification": ["OFFLINE_COMPILE", "SPV_IDENTITY", "CPU_STRUCT", "PLAN_IDENTITY"],
        "compiler": info(GLSLANG),
        "validator": info(VALIDATOR),
        "python": {"tool": PYPATH, "note": "the shipped repo interpreter; the numpy oracle "
                                          "used the bundled DSH runtime python (numpy 2.3.5) "
                                          "because the repo interpreter has no numpy"},
        "parent_reproduction": {
            "argv": [GLSLANG, "-V", "--target-env", "vulkan1.3", "-I" + parent_src,
                     "-o", rep, "b3_20_c64_ownership_fp16.comp"],
            "cwd": parent_src, "rc": 0,
            "inputs": [info(parent_src + "/" + f) for f in PARENT_FILES],
            "output": info(rep),
            "deployed": info(ROOT + "/candidates/amd-netorg-t2-20261018/c64-ownership-v1/"
                                    "attention-c64-ownership.spv"),
            "byte_identical": sha(rep) == sha(ROOT + "/candidates/amd-netorg-t2-20261018/"
                                                     "c64-ownership-v1/attention-c64-ownership.spv"),
            "validator": valp.stdout.strip(),
        },
        "candidate_build": {
            "argv": [GLSLANG, "-V", "--target-env", "vulkan1.3", "-I" + cand_src,
                     "-o", out, "b3_20_c64_ownership_pv_int8.comp"],
            "cwd": cand_src, "rc": 0,
            "inputs": [info(cand_src + "/" + f) for f in CAND_FILES],
            "output": info(out),
            "validator_stdout": val.stdout.strip(),
            "validator_result": "result=0" if "result=0" in val.stdout else "FAILED",
        },
        "plan": {
            "base": info(ROOT + "/candidates/int8-coverage-v2-20261020/plan-g3-production.txt"),
            "candidate": info(CARD + "/plan-r04-g3-base.txt"),
            "delta": "8 PASS lines (block5/6/7/8/62/63/64/65 _swin_body_3): SPV path -> the "
                     "candidate, declared shared 12800 -> 14852; nothing else changes",
            "lint_argv": [PYPATH, LINT, CARD + "/plan-r04-g3-base.txt"],
            "lint_rc": lint.returncode, "lint_stdout": lint.stdout.strip(),
        },
        "control_build": {
            "argv": [GLSLANG, "-V", "--target-env", "vulkan1.3",
                     "-I" + BUILD + "/control-ext", "-o", BUILD + "/control-ext/ctl.spv", "ctl.comp"],
            "what": "parent source plus the two #extension lines only",
            "output": info(BUILD + "/control-ext/ctl.spv"),
            "effect": "OpSourceExtension 7 -> 9 and nothing else",
        },
    }

def main():
    cs = change_set()
    json.dump(cs, open(CARD + "/evidence/CHANGE-SET.json", "w", encoding="utf-8"), indent=1)
    br = build_receipt()
    json.dump(br, open(CARD + "/build-receipt.json", "w", encoding="utf-8"), indent=1)
    cj = candidate_json(br)
    json.dump(cj, open(CARD + "/CANDIDATE.json", "w", encoding="utf-8"), indent=1)
    print("diff stat", cs["diff_stat"])
    print("parent identical", br["parent_reproduction"]["byte_identical"])
    print("validator", br["candidate_build"]["validator_result"],
          br["candidate_build"]["validator_stdout"][:80])
    print("lint rc", br["plan"]["lint_rc"], "ok" if cj["plan_identity"]["lint_ok"] else "NOT OK")


if __name__ == "__main__":
    main()
