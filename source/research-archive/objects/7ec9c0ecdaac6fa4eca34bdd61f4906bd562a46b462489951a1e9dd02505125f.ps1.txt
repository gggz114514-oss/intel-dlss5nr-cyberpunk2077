param([string]$ManifestPath)
$ErrorActionPreference='Stop';$task=Get-Content $ManifestPath -Raw|ConvertFrom-Json;$root=$task.output_root;$base=Split-Path $ManifestPath -Parent;$template=Join-Path (Split-Path $base -Parent) 'execute-nativeonly-pilot.ps1';$source=Join-Path (Split-Path $base -Parent) 'LUNA-REMAINING-NATIVE-JOBS.json';$sourceHash=(Get-FileHash $source -Algorithm SHA256).Hash.ToLowerInvariant();$results=[Collections.ArrayList]::new();$rp=Join-Path $root 'LUNA-REMAINING-NATIVE-RESULTS.json';$sp=Join-Path $root 'LUNA-REMAINING-NATIVE-STATUS.json';$stop=$null
function SaveJson($p,$v){[IO.File]::WriteAllText($p,($v|ConvertTo-Json -Depth 45),[Text.UTF8Encoding]::new($false))}
function SaveProgress($state,$next){SaveJson $rp ([ordered]@{status=$state;jobs=@($results);executed_jobs=@($results|Where-Object {$_.status -ne 'SKIPPED_AFTER_STOP'}).Count;stop_reason=$stop;source_manifest=$source;source_manifest_sha256=$sourceHash;authorization='Parent authorized the exact 48 remaining NativeOnly captures; no performance interpretation.'});SaveJson $sp ([ordered]@{status=$state;updated_at=[DateTime]::UtcNow.ToString('o');planned_jobs=@($task.jobs).Count;executed_jobs=@($results).Count;next_job=$next;stop_reason=$stop;lock_state='released between jobs';results_path=$rp})}
try{
 if(@(Get-Process|Where-Object {$_.ProcessName -match 'Cyberpunk|REDlauncher|port_network|profile_zprof|gtpin|clock_runner|clock_query'}).Count){throw 'Active process before FFN native execution'}
 foreach($i in $task.identities){if((Get-FileHash $i.path -Algorithm SHA256).Hash.ToLowerInvariant() -ne $i.sha256){throw ('Identity mismatch '+$i.path)}}
 foreach($j in $task.jobs){if((Get-FileHash $j.plan.path -Algorithm SHA256).Hash.ToLowerInvariant() -ne $j.plan.sha256){throw ('Plan mismatch '+$j.id)};if(Test-Path $j.runner_output){throw ('Runner output not fresh '+$j.id)};if(Test-Path $j.capture_output){throw ('Capture output not fresh '+$j.id)};$mi=[Array]::IndexOf($j.argv,'--msg_file');if($mi -lt 0 -or (Test-Path $j.argv[$mi+1])){throw ('Message output not fresh '+$j.id)}}
 $lock='@NR_GPU_WORKSPACE@\artifacts\parallel-port-20261013\GPU.lock';$before=Get-Content $lock -Raw|ConvertFrom-Json;if($before.state -ne 'FINISHED'){throw ('GPU lock unavailable '+$before.owner)};SaveProgress 'RUNNING' $task.jobs[0].id
 foreach($job in $task.jobs){if($stop){[void]$results.Add([ordered]@{id=$job.id;status='SKIPPED_AFTER_STOP';reason=$stop;actual_argv=$job.argv});SaveProgress 'STOPPED' $null;continue};if(@(Get-Process|Where-Object {$_.ProcessName -match 'Cyberpunk|REDlauncher|port_network|profile_zprof|gtpin|clock_runner|clock_query'}).Count){throw ('Active process before '+$job.id)};$lockState=Get-Content $lock -Raw|ConvertFrom-Json;if($lockState.state -ne 'FINISHED'){throw ('GPU lock unavailable before '+$job.id)}
  $sub=Join-Path $root $job.id;New-Item -ItemType Directory -Path $sub|Out-Null;$one=$task|ConvertTo-Json -Depth 60|ConvertFrom-Json;$one.output_root=$sub;$one.jobs=@($job);$oneManifest=Join-Path $sub ($job.id+'.json');SaveJson $oneManifest $one;$start=[DateTime]::UtcNow.ToString('o');$child=Start-Process -FilePath 'powershell.exe' -ArgumentList @('-NoProfile','-ExecutionPolicy','Bypass','-File',$template,'-ManifestPath',$oneManifest) -Wait -PassThru -NoNewWindow;$end=[DateTime]::UtcNow.ToString('o');$oneResult=Join-Path $sub 'LUNA-NATIVEONLY-PILOT-RESULTS.json';if(-not (Test-Path $oneResult)){throw ('Missing native result '+$job.id)};$rr=Get-Content $oneResult -Raw|ConvertFrom-Json;$entry=$rr.jobs[0];$good=($child.ExitCode -eq 0 -and $rr.status -eq 'COMPLETED' -and $entry.exit_code -eq 0 -and $entry.normal_close -eq $true -and $entry.frame_matches_gold -eq $true);$entry|Add-Member -NotePropertyName orchestration_started_utc -NotePropertyValue $start -Force;$entry|Add-Member -NotePropertyName orchestration_ended_utc -NotePropertyValue $end -Force;$entry|Add-Member -NotePropertyName executor_result_path -NotePropertyValue $oneResult -Force;$entry|Add-Member -NotePropertyName status -NotePropertyValue $(if($good){'COMPLETED'}else{'FAILED'}) -Force;[void]$results.Add($entry);if(-not $good){$stop='NativeOnly exit/receipt/frame gate failed: '+$job.id;SaveProgress 'STOPPED' $null;break};SaveProgress 'RUNNING' 'next listed job'
 }
 SaveProgress $(if($stop){'STOPPED'}else{'COMPLETED'}) $null
}catch{$stop=$_.Exception.Message;SaveProgress 'FAILED_OR_STOPPED' $null;throw}
