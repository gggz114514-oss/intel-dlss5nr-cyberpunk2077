def sha(p):
    h = hashlib.sha256()
    with open(p, "rb") as f:
        for b in iter(lambda: f.read(1 << 20), b""):
            h.update(b)
    return h.hexdigest()


def ident(p):
    p = Path(p)
    return {"path": p.as_posix(), "bytes": p.stat().st_size, "sha256": sha(p)}


def bits(v):
    return "0x%04x" % struct.unpack("<H", struct.pack("<e", v))[0]

Wc = [[ref.w[16384 + ((o // 8) * 4 + e // 32) * 1024 + (e % 32) * 32 + (o % 8)] for e in range(128)]
      for o in range(32)]


def act_h(x):
    t = max(-4.0, min(4.0, x))
    u = h16((-0.055908203125) * abs(t) + 0.447265625)
    v = h16(t * u + 0.89453125)
    return h16(x * v)


def stage(qx, x):
    acc = [sum(W[f][k] * qx[k] for k in range(32)) for f in range(128)]
    e = [act_h(h16(f32(a * ref.mult_e[f]))) for f, a in enumerate(acc)]
    qe = [code(e[f], ref.inve) for f in range(128)]
    y = []
    for o in range(32):
        a = sum(Wc[o][k] * qe[k] for k in range(128))
        y.append(h16(f32(f32(a * ref.mult_c[o]) + float(h16(x[o] * ref.rs[o])))))
    return e, y

def segment_parent(off, rank):
    """Parent: expand and contract interleave per feature group (4 expand then
    4 contract, 8 groups; source kernel.comp:243-713).  In word order the
    instruction gap alternates 450 / 81, which identifies the blocks."""
    if off >= 40000:
        return "nrf_lift (K=16 fp16 MMA)"
    if off <= 22543:
        return "expand (8 groups x 4)" if (rank // 4) % 2 == 0 else "contract (8 groups x 4)"
    if off < 24890:
        return "QKV projection (3 x 8)"
    if off < 30939:
        return "score S^T = K.Q^T (8 blocks x 2)"
    if off < 34513:
        return "PV context (4 P blocks x 4)"
    return "output projection (4 fragments x 2)"


def group(offs, seg):
    out = []
    for rank, s in enumerate(offs):
        name = seg(s["word_offset"], rank)
        if out and out[-1]["segment"] == name:
            out[-1]["sites"].append(s["word_offset"])
        else:
            out.append({"segment": name, "sites": [s["word_offset"]]})
    return out
