#!/usr/bin/env python3
"""Strict plan checks that `plan_lint.py` does not actually perform (TOOL-DEFECTS.md D1).

1. BIND bounds: parse all five fields `BIND <binding> <resource> <offset> <bytes>` and
   require `offset + bytes <= declared buffer size`, and that the binding slots of a PASS
   are the set the SPIR-V module actually declares (0..nbind-1).
2. Declared shared bytes must equal the module's sum of Workgroup variable sizes, for
   every PASS whose SPV exists on disk (cached per file).

usage: check_plan_strict.py <plan> <out.json>
"""
import json
import re
import sys
from pathlib import Path

sys.path.insert(0, r"@NR_WORKSPACE@/work/dsh-xmx-handoff-validation-20261024/build/r12-split16-l2")
import spv_r12 as R  # noqa: E402

TOKEN = re.compile(r'"[^"]*"|\S+')


def main(argv):
    plan = Path(argv[0])
    lines = plan.read_text(encoding="utf-8").splitlines()
    h = lines[0].split()
    nb, npass, nout = int(h[2]), int(h[3]), int(h[4])
    bufs = []
    for ln in lines[1:1 + nb]:
        m = re.match(r'BUFFER "([^"]+)" (\d+) "([^"]*)"', ln)
        bufs.append({"name": m.group(1), "bytes": int(m.group(2))})
    rest = lines[1 + nb:]
    cache = {}
    problems = []
    checked = 0
    shared_ok = shared_bad = 0
    i = 0
    passes = []
    for pi in range(npass):
        t = TOKEN.findall(rest[i])
        row, name, spv = int(t[1]), t[2].strip('"'), t[3].strip('"')
        sg, sh, nbind, npush = int(t[4]), int(t[5]), int(t[6]), int(t[7])
        for b in range(nbind):
            w = rest[i + 1 + b].split()
            if w[0] != "BIND" or len(w) != 5:
                problems.append("pass %d (%s): bad BIND arity %s" % (pi, name, w))
                continue
            slot, res, off, ln = int(w[1]), int(w[2]), int(w[3]), int(w[4])
            if res >= nb:
                problems.append("pass %d (%s): resource %d >= %d" % (pi, name, res, nb))
            elif off + ln > bufs[res]["bytes"]:
                problems.append("pass %d (%s): BIND slot %d res %d %d+%d > declared %d (%s)"
                                % (pi, name, slot, res, off, ln, bufs[res]["bytes"],
                                   bufs[res]["name"]))
            if slot != b:
                problems.append("pass %d (%s): binding slot %d out of order at position %d"
                                % (pi, name, slot, b))
            checked += 1
        f = Path(spv)
        if f.exists():
            if str(f) not in cache:
                cache[str(f)] = R.Mod(f).workgroup_bytes()
            wgb = cache[str(f)]
            if sh != wgb:
                shared_bad += 1
                problems.append("pass %d (%s): declared shared %d != module workgroup bytes %d (%s)"
                                % (pi, name, sh, wgb, f.name))
            else:
                shared_ok += 1
        passes.append({"plan_order": row, "pass": name, "spv": spv, "shared": sh,
                       "nbind": nbind, "npush": npush})
        i += 1 + nbind + 1
    out = {"plan": str(plan), "bindings_checked": checked, "passes": npass,
           "buffers": nb, "modules_cached": len(cache),
           "shared_matches_module_workgroup_bytes": shared_ok,
           "shared_mismatch": shared_bad,
           "problems": problems[:40], "problem_count": len(problems),
           "ok": not problems}
    Path(argv[1]).write_text(json.dumps(out, indent=1) + "\n", encoding="utf-8", newline="\n")
    print("bindings", checked, "modules", len(cache), "shared ok", shared_ok,
          "bad", shared_bad, "problems", len(problems))
    for p in problems[:10]:
        print(" ", p)
    return 0 if not problems else 1


if __name__ == "__main__":
    sys.exit(main(sys.argv[1:]))
