"""Reverse occupancy probe on F05: pad the qkv staging array 24576 B -> 49152 B.

All three F05 shader groups (C64 8 rows, C128 12 rows, C256 16 rows = 36 PASS) declare the same
`shared float16_t qkv[12288]` (24576 B). Doubling it should take the SM from two resident
workgroups to one, with zero change to the maths.

Reproduction is checked per group first: the untouched source must recompile to the exact SPIR-V
currently dispatched.
"""
import json, re, sys
from pathlib import Path
sys.path.insert(0, str(Path(__file__).parent))
import COMMON as K
from audit_card import pass_lines

cur = K.read_json(K.C / 'CURRENT.json')
recs = pass_lines(Path(cur['best_plan']['path']))
rows = K.rows()
sel = [r for r in rows if recs[r['ordinal']]['line'].split()[-6] == '24576' and r['family'] == 'F05']
groups = {}
for r in sel:
    groups.setdefault(Path(recs[r['ordinal']]['spv']).as_posix(), []).append(r['ordinal'])
R = K.read_json(K.C / 'evidence' / 'RECIPES.json')
out = K.C / 'queue' / 'probe-occ-f05'
out.mkdir(parents=True, exist_ok=True)

built = {}
for spv_path, ords in groups.items():
    spv = Path(spv_path)
    sha = K.sha(spv)
    # source + compile flags: recipe for the baseline groups, the n05a card for the adopted C256 group
    if sha in R:
        src = Path(R[sha]['source'])
        inc = [R[sha]['compile']['pinned_include_dir']] + R[sha]['compile']['include_dirs']
        defs = R[sha]['compile'].get('defines', ())
    else:
        src = K.C / 'cards' / 'n05a-f05-c256-mreuse' / 'candidate.comp'
        n05a = K.read_json(K.C / 'cards' / 'n05a-f05-c256-mreuse' / 'BUILD.json')
        inc = [n05a['include_headers']] if isinstance(n05a.get('include_headers'), str) else list(n05a.get('include_headers') or [])
        inc = [str(x) for x in inc]
        defs = ()
    tag = sha[:8]
    K.compile_glsl(src, out / ('parent-%s.spv' % tag), include_dirs=inc, defines=defs)
    ok = K.sha(out / ('parent-%s.spv' % tag)) == sha
    print('%s rows %2d  parent reproduced: %s' % (tag, len(ords), ok))
    text = src.read_text(encoding='utf-8')
    assert 'shared float16_t qkv[12288];' in text, (tag, 'qkv decl not found')
    padded = text.replace('shared float16_t qkv[12288];', 'shared float16_t qkv[24576];')
    (out / ('pad-%s.comp' % tag)).write_text(padded, encoding='utf-8')
    K.compile_glsl(out / ('pad-%s.comp' % tag), out / ('pad-%s.spv' % tag), include_dirs=inc, defines=defs)
    K.validate_spv(out / ('pad-%s.spv' % tag))
    built[spv_path] = dict(ords=ords, pad=(out / ('pad-%s.spv' % tag)).as_posix(), parent_ok=ok)

# plan: 36 rows repointed, shared field 24576 -> 49152
head = Path(cur['best_plan']['path'])
lines = head.read_text(encoding='utf-8-sig').splitlines()
n = 0
for spv_path, info in built.items():
    for o in info['ords']:
        line = recs[o]['line']
        toks = line.split()
        toks[-6] = '49152'
        new_line = ' '.join(toks).replace(Path(spv_path).as_posix(), info['pad'])
        idx = next(i for i, l in enumerate(lines) if l == line)
        lines[idx] = new_line
        n += 1
plan = out / 'plan.txt'
plan.write_text('\n'.join(lines) + '\n', encoding='utf-8')
print('rows repointed:', n, 'plan sha', K.sha(plan))
print(K.cpu_check(plan).strip()[-90:])
