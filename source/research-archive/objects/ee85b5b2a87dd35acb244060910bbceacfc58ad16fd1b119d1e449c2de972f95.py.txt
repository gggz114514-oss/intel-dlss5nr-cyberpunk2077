"""Chain vs side-branch breakdown of the F01 entry region (read-only ISA)."""
import collections
import json
import sys
from pathlib import Path

CARD = Path(__file__).resolve().parents[1]
sys.path.insert(0, str(CARD))
from tools.isa_chain import load, slice_of, int_tail, gathers, region_of  # noqa: E402


def klass(op):
    if op.startswith('send'):
        return 'send'
    if op in ('mul', 'macl', 'mad', 'mach'):
        return 'mul/macl/mad'
    if op in ('add', 'add3', 'sub', 'addc'):
        return 'add/add3'
    if op in ('shl', 'shr', 'asr', 'and', 'or', 'xor') or op.startswith('bfn'):
        return 'bit (shl/shr/and/xor/bfn)'
    if op in ('sel', 'cmp', 'min', 'max'):
        return 'sel/cmp'
    if op.startswith('math') or op in ('rndd', 'frc', 'rndu', 'rnde'):
        return 'math/round'
    if op.startswith('mov'):
        return 'mov'
    if op.startswith('sync') or op in ('join', 'jmpi', 'goto'):
        return 'ctrl/sync'
    return op


def main():
    isa, by_dw = load()
    reg = [i for i in isa if region_of(i['addr'])]
    gs = gathers(isa)
    chain, tail = set(), set()
    for g in gs:
        nodes = slice_of(isa, by_dw, g['idx'])
        for k in nodes:
            chain.add(isa[k]['addr'])
        t, _ = int_tail(isa, nodes, g['idx'])
        for k in t:
            tail.add(isa[k]['addr'])
    blocks = collections.defaultdict(lambda: [0, 0, 0])
    for i in reg:
        b = i['addr'] >> 8
        blocks[b][0] += 1
        if i['addr'] in chain:
            blocks[b][1] += 1
        if i['addr'] in tail:
            blocks[b][2] += 1
    print('block   instr  chain inttail')
    for b in sorted(blocks):
        n, c, t = blocks[b]
        print('0x%02X00  %5d  %5d  %5d' % (b, n, c, t))
    print('total   %5d  %5d  %5d' % (len(reg), len(chain), len(tail)))
    cc, cs = collections.Counter(), collections.Counter()
    for i in reg:
        if i['addr'] in chain:
            cc[klass(i['op'])] += 1
        else:
            cs[klass(i['op'])] += 1
    print('\n%-26s %6s %6s' % ('class', 'chain', 'side'))
    for k in sorted(set(cc) | set(cs), key=lambda x: -(cc[x] + cs[x])):
        print('%-26s %6d %6d' % (k, cc[k], cs[k]))
    out = dict(region=len(reg), chain=len(chain), int_tail=len(tail), side=len(reg) - len(chain),
               blocks={hex(b << 8): dict(instr=v[0], chain=v[1], int_tail=v[2])
                       for b, v in sorted(blocks.items())},
               chain_by_class=dict(cc), side_by_class=dict(cs))
    (CARD / 'tools' / 'chain-blocks.json').write_text(json.dumps(out, indent=1), encoding='utf-8')


if __name__ == '__main__':
    main()
