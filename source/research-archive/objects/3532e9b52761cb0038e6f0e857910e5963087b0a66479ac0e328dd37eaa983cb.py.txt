"""n21 step-9: final self-audit (card-local, read only + no writes outside the card)."""
import sys, json, re, subprocess
from pathlib import Path

CARD = Path(__file__).resolve().parent
C = CARD.parent.parent
sys.path.insert(0, str(C))
import COMMON as K

REQUIRED = ['ANALYSIS.md', 'PROOF.json', 'SOURCE-DIFF.patch', 'candidate.comp', 'BUILD.json',
            'candidate.spv', 'plan.txt', 'COST-LEDGER.json', 'NATIVE-PREDICTION.md',
            'GPU-REQUEST.json', 'RESUME.txt', 'parent-rebuild.spv']
problems = []
print('%-24s %10s  %s' % ('file', 'bytes', 'sha256'))
for name in REQUIRED:
    p = CARD / name
    if not p.exists():
        problems.append('missing ' + name)
        print('%-24s %10s  MISSING' % (name, '-'))
        continue
    print('%-24s %10d  %s' % (name, p.stat().st_size, K.sha(p)))

for name in ('BUILD.json', 'PROOF.json', 'COST-LEDGER.json', 'GPU-REQUEST.json'):
    try:
        json.loads((CARD / name).read_text(encoding='utf-8'))
    except Exception as e:
        problems.append('%s: %s' % (name, e))

b = json.loads((CARD / 'BUILD.json').read_text(encoding='utf-8'))
checks = {
    'candidate.spv sha': (b['candidate']['spv']['sha256'], K.sha(CARD / 'candidate.spv')),
    'candidate.comp sha': (b['candidate']['source']['sha256'], K.sha(CARD / 'candidate.comp')),
    'parent-rebuild sha': (b['parent_rebuild']['sha256'], K.sha(CARD / 'parent-rebuild.spv')),
    'plan sha': (b['plan']['sha256'], K.sha(CARD / 'plan.txt')),
    'parent source copy sha': (b['candidate']['source_copy_of_parent']['sha256'], K.sha(CARD / 'parent.comp')),
}
for k, (want, got) in checks.items():
    ok = want == got
    print('%-24s %s' % (k, 'OK' if ok else 'MISMATCH want=%s got=%s' % (want, got)))
    if not ok:
        problems.append(k)

# parent-rebuild must equal the baseline row SPV
row = K.row_by_ordinal(194)
print('parent-rebuild == baseline row spv:', K.sha(CARD / 'parent-rebuild.spv') == row['spv_sha256'],
      row['spv_sha256'])

# candidate must differ from the parent and reference no dbg
print('candidate.spv != parent spv:', K.sha(CARD / 'candidate.spv') != row['spv_sha256'])
print("candidate.comp contains 'dbg':", 'dbg' in (CARD / 'candidate.comp').read_text(encoding='utf-8'))

# plan: exactly one changed line vs the chain head, replaced == 1
head, head_sha, _ = K.current_best()
a = Path(head).read_text(encoding='utf-8-sig').splitlines()
b2 = (CARD / 'plan.txt').read_text(encoding='utf-8').splitlines()
diff = [(i, x) for i, (x, y) in enumerate(zip(a, b2)) if x != y]
print('plan lines', len(a), len(b2), 'changed', len(diff), 'ord194 line no(1-based)', diff[0][0] + 1 if diff else None)
print('plan references candidate:', (CARD / 'candidate.spv').as_posix() in (CARD / 'plan.txt').read_text(encoding='utf-8'))
print('ord 193 line untouched:', a[193] == b2[193])

cp = K.cpu_check(CARD / 'plan.txt')
print('cpu-check:', cp.strip())

# SOURCE-DIFF.patch must be 9 deletions / 0 insertions
patch = (CARD / 'SOURCE-DIFF.patch').read_text(encoding='utf-8')
minus = [l for l in patch.splitlines() if l.startswith('-') and not l.startswith('---')]
plus = [l for l in patch.splitlines() if l.startswith('+') and not l.startswith('+++')]
print('patch: -%d +%d' % (len(minus), len(plus)))
if len(minus) != 9 or len(plus) != 0:
    problems.append('patch shape')

# no writes outside the card: check mtimes of the baseline plan and the baseline SPV are old
print('baseline plan sha unchanged:',
      K.sha(head) == head_sha, head_sha)
print('baseline row spv sha unchanged:', K.sha(row['spv']) == row['spv_sha256'])

print('\nPROBLEMS:', problems if problems else 'none')
