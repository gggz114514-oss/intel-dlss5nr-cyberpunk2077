def sha(p):
    h = hashlib.sha256()
    try:
        with open(p, "rb") as f:
            for b in iter(lambda: f.read(1 << 20), b""):
                h.update(b)
        return h.hexdigest()
    except OSError:
        return None

def probe(ps):
    for k, v in PROBE.items():
        if k == ps.spv.replace("\\", "/"):
            return v
    return {}


def packs(ps):
    out = []
    for b in ps.binds:
        bufr = buf[b.buffer]
        if bufr.file in ("-", ""):
            continue
        out.append({"binding": b.binding, "buffer": bufr.name,
                    "file": os.path.basename(bufr.file), "path": bufr.file,
                    "bind_bytes": b.bytes, "file_bytes": bufr.bytes,
                    "sha256": sha(bufr.file)})
    return out
