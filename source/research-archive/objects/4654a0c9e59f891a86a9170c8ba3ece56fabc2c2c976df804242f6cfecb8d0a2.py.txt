import json, statistics, re
from pathlib import Path
D=Path(__file__).resolve().parent
def read(p): return json.loads(p.read_text(encoding='utf-8-sig'))
basepath=D/'run/recovery-baseline/port_network_receipt.json'
if not basepath.exists(): basepath=D/'run/baseline/port_network_receipt.json'
base=read(basepath)
orders={'f02':[1,2,3,4,227,228,229],'f20':[230]}
out={'status':'PILOT_ONLY_NOT_ACCEPTANCE','baseline':str(basepath),'jobs':[]}
for label in ['f02-whole','f20-whole','f02-local-null','f20-local-null','f02-maxbuckets','f20-maxbuckets']:
 p=D/'run'/label/'port_network_receipt.json'
 if not p.exists(): continue
 r=read(p); idx=orders[label[:3]]
 def samples(x):
  rows={row['index']:row for row in x['per_pass_gpu']}
  return [sum(rows[i]['samples_ms'][j] for i in idx) for j in range(len(x['gpu_samples_ms']))]
 b=statistics.median(samples(base)); s=samples(r); med=statistics.median(s)
 phases=[]
 for p in (D/'captures'/label).rglob('phaseclock.txt'):
  for line in p.read_text().splitlines():
   c=line.split()
   if len(c)>12 and c[0]=='main' and re.fullmatch('[0-9a-f]{16}',c[1]):
    phases.append({'freq':int(c[4]),'cycles':int(c[5]),'avg':int(c[6]),'dispatch':int(c[10])})
 out['jobs'].append({'id':label,'target_median_ms':med,'baseline_median_ms':b,'ratio_to_baseline':med/b,'samples_ms':s,'phase_records':phases,'freq_set':sorted(set(x['freq'] for x in phases)),'average_counter_median':statistics.median(x['avg'] for x in phases) if phases else None})
(D/'MAIN-CALIBRATION-REVIEW.json').write_text(json.dumps(out,indent=2),encoding='utf-8')
for j in out['jobs']: print({k:v for k,v in j.items() if k not in ['phase_records','samples_ms']})
