import json, hashlib, os
base = r"@NR_WORKSPACE@/work/dsh-fullnet-static-20261029/ledger"
for n in ["FULLNET-ROWS.json","DYNAMIC-SCALE-REMAINDERS.json","FP16-MMA-REMAINDERS.json","SCALE-CLASS-SUMMARY.json"]:
    p = os.path.join(base,n); d = json.load(open(p, encoding="utf-8"))
    h = hashlib.sha256(open(p,'rb').read()).hexdigest()
    n_rows = len(d.get("rows", d.get("positions", []))) if n!="SCALE-CLASS-SUMMARY.json" else d["totals"]["dispatches"]
    print(f"{n:32s} valid_json=True items={n_rows:4d} sha={h}")
print("recompute matches:", json.load(open(os.path.join(base,"SCALE-CLASS-SUMMARY.json"), encoding="utf-8"))["recompute_check"]["matches"])
