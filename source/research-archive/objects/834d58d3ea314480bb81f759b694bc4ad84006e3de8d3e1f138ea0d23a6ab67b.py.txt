"""Record the occupancy probe and the specific target it identifies."""
import json, sys
from pathlib import Path
sys.path.insert(0, str(Path(__file__).parent))
import COMMON as K

probe = K.read_json(K.C / 'evidence' / 'OCCUPANCY-PROBE.json')
rows = probe['rows']
by_slm = {}
for r in rows:
    by_slm.setdefault(r['slm'], []).append(r)
summary = {str(k): dict(n=len(v), occ=sum(x['occ'] for x in v) / len(v),
                        active=sum(x['active'] for x in v) / len(v))
           for k, v in sorted(by_slm.items())}

p = K.C / 'evidence' / 'LESSONS.json'
d = K.read_json(p)
ids = [x['id'] for x in d['append_after_L7']]
if 'L29' not in ids:
    d['append_after_L7'].append({
        'id': 'L29',
        'claim': 'Occupancy is capped by registers in every kernel and by shared memory above 16 KB; the only untried way to raise it is to shrink a workgroups working set, which is a shape change',
        'evidence': [
            'a probe over the frame joined each PASSes measured occupancy/XVE activity with its own resources: 199 of 232 passes sit at max_r 127 and the remaining 33 report r255 (a special register), so effectively every kernel is at the 128-register ceiling',
            'occupancy falls with shared memory above 16 KB: 8 KB -> 86.1%, 20 KB -> 81.7%, 24 KB -> 64.4%, 32 KB -> 41.7%',
            'XVE activity does not follow occupancy monotonically (the 0 KB group has 79.9% occupancy but 32.0% activity, while 8 KB has 86.1% and 46.2%), so occupancy is necessary but not sufficient',
            'the 17 passes at 32 KB are the clearest gap: 41.7% occupancy against 86.1% achieved elsewhere in the same frame',
            'shrinking the staging buffer by segmenting it was already tried twice (n06c) and lost 0.0246 ms because it added staging messages and barriers; halving the tokens per workgroup (32 KB -> 16 KB, twice the workgroups) has NOT been tried and needs a paired grid change (L6)'
        ],
        'consequence': 'If occupancy is pursued, the concrete target is the 32 KB group: halve tokens per workgroup to reach ~16 KB and roughly double residency, with the SPIR-V and the launch grid changed together and a post-dispatch write-set proof that the coverage is unchanged.'
    })
    K.write_json(p, d)
    print('L29 added')
K.write_json(K.C / 'evidence' / 'OCCUPANCY-SUMMARY.json', dict(by_slm=summary, n=len(rows)))
print(json.dumps(summary, indent=1)[:600])
