"""Run an ad-hoc probe plan against the chain head (gates + ordinary).

Usage: python run_probe.py <name> <plan.txt> [<expected replaced>]

The probe is explicitly not a candidate: it exists to measure the cost of one named
change (e.g. removing a single fragment transport). All samples are kept.
"""
import sys
from pathlib import Path
sys.path.insert(0, str(Path(__file__).parent))
import COMMON as K
import STAGE
import ANALYZE

name = sys.argv[1]
plan = Path(sys.argv[2])
stage = sys.argv[3] if len(sys.argv) > 3 else 'all'
base, base_sha, chain = K.current_best()
psha = K.sha(plan)
out = K.C / 'queue' / name
out.mkdir(parents=True, exist_ok=True)
run = out / 'run'

arms = [('base', base, base_sha), ('probe', plan, psha)]
jobs = []
if stage in ('all', 'gates'):
    for aid, p, s in arms:
        for mode in ('fixed', 'fb2'):
            jobs.append(K.gate_job('gates-%s-%s' % (aid, mode), p, s,
                                   run / ('gates-%s-%s' % (aid, mode)), mode, aid))
spec = dict(stage='gates', owner='LUNA-CONTINUOUS-PROBE',
            authorization='Main (this session) authorized probe %s gates only; this probe may change numerics by design and is never adopted.' % name,
            output_root=str(out / 'gates'), identities=[str(base), str(plan), str(K.RUNNER)],
            jobs=jobs, rules='probe; not a candidate; gold failures expected if the probe removes work')
sp, s_ = None, None
if jobs:
    sp = out / 'spec-gates.json'
    K.write_json(sp, spec)
    manifest, ps1 = STAGE.prepare(spec)
    r = STAGE.run(manifest, ps1)
    s = STAGE.summarize(spec['output_root'])
    K.write_json(Path(spec['output_root']) / 'STAGE-SUMMARY.json', s)
    print('gates', s['status'], [(j['id'], j['status'], j['frame']) for j in s['jobs']])

if stage in ('all', 'ordinary'):
    jobs = []
    for pi in range(6):
        order = [arms[0], arms[1]] if pi % 2 == 0 else [arms[1], arms[0]]
        for aid, p, s_ in order:
            jobs.append(K.ordinary_job('ord-%d-%s' % (pi, aid), p, s_,
                                       run / ('ord-%d-%s' % (pi, aid)), aid, pi))
    spec = dict(stage='ordinary', owner='LUNA-CONTINUOUS-PROBE',
                authorization='Main (this session) authorized probe %s ordinary only; never adopted.' % name,
                output_root=str(out / 'ordinary'), identities=[str(base), str(plan), str(K.RUNNER)],
                jobs=jobs, rules='probe; all samples retained')
    sp = out / 'spec-ordinary.json'
    K.write_json(sp, spec)
    manifest, ps1 = STAGE.prepare(spec)
    r = STAGE.run(manifest, ps1)
    s = STAGE.summarize(spec['output_root'])
    K.write_json(Path(spec['output_root']) / 'STAGE-SUMMARY.json', s)
    print('ordinary', s['status'], [(j['id'], j['status'], j['frame']) for j in s['jobs']])
    if s['status'] == 'COMPLETED':
        rep = ANALYZE.report(Path(spec['output_root']) / 'LUNA-ORDINARY.json',
                             [dict(arms=('base', 'probe'))], out / 'TIMING.json',
                             extra=dict(probe=name))
        c = rep['comparisons'][0]
        print('PROBE %s: %s saving=%.6f CI95=[%.6f, %.6f] groups=%d' % (
            name, c['status'], c['saving_ms'], c['CI95_ms'][0], c['CI95_ms'][1], c['groups']))
        print('pairs', [(g['pair'], round(g['base'], 5), round(g['probe'], 5)) for g in c['group_means']])
