"""Debug the NoContraction function alignment between the parent rebuild and the
candidate raw compile.  CPU only."""
import sys, difflib
from pathlib import Path
CARD = Path(__file__).parent
sys.path.insert(0, str(CARD.parent.parent))
sys.path.insert(0, str(CARD))
import COMMON as K
ST = K.ST

CARD = Path(__file__).parent
P = CARD / 'parent-rebuild.spv'
C = CARD / 'candidate-raw.spv'

pm, cm = ST.Module(P), ST.Module(C)
pr, cr = ST.function_regions(pm), ST.function_regions(cm)
print('parent regions', len(pr), [len(r) for r in pr])
print('child  regions', len(cr), [len(r) for r in cr])
psk = ST._skeleton_of(pm, pr)
csk = ST._skeleton_of(cm, cr)
for i in range(len(pr)):
    pat = ST.decoration_pattern(pm, pr[i])
    print('parent fn', i, 'instrs', len(psk[i]), 'decorated', len(pat))
for ci in csk:
    print('child fn', ci, 'instrs', len(csk[ci]))
# best alignment for parent fn 0
want = psk[0]
pattern = ST.decoration_pattern(pm, pr[0])
for ci, have in csk.items():
    blocks = difflib.SequenceMatcher(None, want, have, autojunk=False).get_matching_blocks()
    matched = sum(b.size for b in blocks)
    pos = {}
    for b in blocks:
        for k in range(b.size):
            pos[b.a + k] = b.b + k
    missing = [p for p in pattern if p not in pos]
    print('child fn %d: matched=%d missing_decorated=%d first_missing=%s'
          % (ci, matched, len(missing), missing[:6]))
