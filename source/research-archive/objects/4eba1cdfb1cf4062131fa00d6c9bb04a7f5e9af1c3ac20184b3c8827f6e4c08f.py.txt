#!/usr/bin/env python3
"""Emit the named GPU test ticket from the jobs every card defined.

Writes `tickets/VK-XMX-HANDOFF-GPU-SLOT-20261024.md`.  The ticket is a *request*:
nothing here was executed, and the ticket says so.
"""
import hashlib
import json
from pathlib import Path

W = Path(r"@NR_WORKSPACE@/work/dsh-xmx-handoff-validation-20261024")
ROOT = Path(r"@NR_WORKSPACE@")
LOCK = r"@NR_GPU_WORKSPACE@/artifacts/parallel-port-20261013/GPU.lock"


def sha(p):
    return hashlib.sha256(Path(p).read_bytes()).hexdigest()


def main():
    rows, total = [], 0
    for jobs in sorted((W / "run").glob("*/jobs.json")):
        jd = json.loads(jobs.read_text(encoding="utf-8"))
        jd["jobs"] = jd.get("jobs") or jd.get("gpu_jobs") or []
        card = jobs.parent.name
        jobs_list = []
        for j in jd["jobs"]:
            if not isinstance(j, dict):
                continue
            j = dict(j)
            j.setdefault("label", j.get("id") or "job")
            j.setdefault("args", j.get("argv") or [])
            if "plan" not in j:
                argv = j.get("args") or []
                if "--plan" in argv:
                    k = argv.index("--plan")
                    if k + 1 < len(argv):
                        j["plan"] = argv[k + 1]
            k = str(j.get("kind", "")).lower()
            j["kind"] = ("gate" if "gate" in k else
                         ("ordinary" if "ordinar" in k else ("capture" if "captur" in k else "other")))
            jobs_list.append(j)
        jd["jobs"] = jobs_list
        gate = [j for j in jobs_list if j.get("kind") == "gate"]
        ordn = [j for j in jobs_list if j.get("kind") == "ordinary"]
        total += len([j for j in jd["jobs"] if isinstance(j, dict)])
        rows.append((card, jd, gate, ordn, jobs))
    lines = ["# 申请 GPU 时段：DSH XMX 交接与资源寿命验证队列（2026-10-24）", "",
             "申请方：本线 DSH 长任务（工作包 "
             "`work/dsh-xmx-handoff-validation-20261024`）。",
             f"不是新锁、不是新队列：沿用唯一锁 `{LOCK}`。", "",
             "## 现状（申请方实测）", "",
             "- 锁文件末态 `state=FINISHED`（2026-10-24T01:25:20+09:00 由 "
             "`CP2077-EXPERIMENTAL-USER-TEST-20261024` 释放），未持有。",
             "- 本机 `Cyberpunk2077` 未运行（进程数 0，作业核对时点）。",
             "- 尽管锁空闲，本工作包**没有本次时段的明确分配**，因此**一个 GPU 作业都没有跑**："
             "没有创建 device/pipeline、没有 native capture、没有 gate、没有 ordinary、没有改锁。",
             "",
             "## 申请内容", "",
             f"全部队列共 **{total} 个作业**，全部已写成可提交的 argv，"
             "按卡分组、按 `gate → ordinary` 串行。", "",
             "| 卡 | 作业数 | gate | ordinary | 计划 | jobs.json sha256 |",
             "|---|---:|---:|---:|---|---|"]
    for card, jd, gate, ordn, jobs in rows:
        _p = jd.get("plan") or jd.get("parent_plan") or ""
        if not isinstance(_p, str):
            _p = ""
        plan = Path(_p)
        lines.append(f"| `{card}` | {len([j for j in jd['jobs'] if isinstance(j, dict)])} | {len(gate)} | {len(ordn)} | "
                     f"`{plan.name}` | `{sha(jobs)[:16]}…` |")
    lines += ["", "## 逐作业清单", ""]
    for card, jd, gate, ordn, jobs in rows:
        lines.append(f"### `{card}`")
        lines.append("")
        lines.append("| label | kind | plan | passes | gold |")
        lines.append("|---|---|---|---:|---|")
        for j in jd["jobs"]:
            if not isinstance(j, dict) or "label" not in j:
                continue
            g = j.get("gold")
            g = f"`{g[:16]}…`" if g else "两臂帧恒等（不预设哈希）"
            plan = j.get("plan") or jd.get("plan") or jd.get("parent_plan") or "?"
            if not isinstance(plan, str):
                plan = "?"
            lines.append(f"| `{j['label']}` | {j['kind']} | `{Path(plan).name}` | "
                         f"{j.get('expected_passes')} | {g} |")
        lines.append("")
    lines += ["## 提交方式", "",
              "沿用私有 wrapper 副本（只改 `$prepPath`/owner 标签/授权文字，锁语义未改）：",
              "`build/dsh-int8-same-family-20261023/run_locked_ps51.ps1` "
              "(SHA `931ceb039b41660115f0316a3a3ae41afc9072aa220d6226065aa820aad04f73`)。", ""]
    return lines


def write():
    lines = main()
    (W / "tickets").mkdir(parents=True, exist_ok=True)
    out = W / "tickets" / "VK-XMX-HANDOFF-GPU-SLOT-20261024.md"
    out.write_text("\n".join(lines) + "\n", encoding="utf-8", newline="\n")
    print("wrote", out, len(lines), "lines")


if __name__ == "__main__":
    write()
