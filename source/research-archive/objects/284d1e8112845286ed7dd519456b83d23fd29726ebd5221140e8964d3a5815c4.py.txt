# S09 read-only scanner: list attention-relevant dispatch rows of CURRENT-ROWS.json
import json, sys, os, hashlib, collections

ROWS = r"@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/int8-quality-first-20261029/dsh-static-handoff/CURRENT-ROWS.json"
out = sys.argv[1] if len(sys.argv) > 1 else None

with open(ROWS, "r", encoding="utf-8") as f:
    j = json.load(f)

print("plan:", j.get("plan"))
print("plan_sha256:", j.get("sha256"))
print("n_rows:", len(j["rows"]))
print("unique_spvs:", j.get("unique_spvs"))

rows = j["rows"]
pat = ("attn", "attention", "qkv", "qk", "pv")
def is_attn(r):
    n = r["name"].lower()
    b = os.path.basename(r["spv"]).lower()
    return any(p in n for p in pat) or any(p in b for p in pat)

sel = [r for r in rows if is_attn(r)]
print("attn_like_rows:", len(sel))
print()
hdr = f"{'ord':>3} {'name':<42} {'pass':>4} {'spv_basename':<52} {'sha8':<8} bytes"
print(hdr)
for r in rows:
    mark = "*" if is_attn(r) else " "
    print(f"{mark}{r['ordinal']:>3} {r['name']:<42} {r['pass_number']:>4} {os.path.basename(r['spv']):<52} {r['sha256'][:8]:<8}")

if out:
    keep = {
        "plan": j.get("plan"),
        "plan_sha256": j.get("sha256"),
        "n_rows": len(rows),
        "attn_rows": sel,
        "unique_spvs": j.get("unique_spvs"),
        "buffers": j.get("buffers"),
    }
    with open(out, "w", encoding="utf-8") as f:
        json.dump(keep, f, indent=1)
    print("\nwrote", out)
