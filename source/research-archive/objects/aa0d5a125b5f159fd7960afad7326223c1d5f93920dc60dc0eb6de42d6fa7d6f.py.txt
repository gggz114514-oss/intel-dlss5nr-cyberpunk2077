"""Build plan.txt for n32-f05-aluwr: replace the 36 F05 PASS lines' SPV path only.

Parent plan = current chain head (accepted/n29-f01-alu-chain-20261027/plan.txt,
sha d8875665...).  Nothing else in the plan is touched; the count of replaced PASS
lines must equal 36 and every other field must be byte-identical.
"""
import json, sys, re, hashlib
from pathlib import Path

HERE = Path(__file__).parent
CAMP = HERE.parent.parent
sys.path.insert(0, str(CAMP))
import COMMON as K

HEAD, HEAD_SHA, _ = K.current_best()
assert K.sha(HEAD) == HEAD_SHA
MAP = {
    "c64": [7, 10, 13, 16, 215, 218, 221, 224],
    "c128": [19, 22, 25, 28, 31, 34, 195, 198, 201, 204, 207, 210],
    "c256": [37, 40, 43, 46, 49, 52, 55, 58, 169, 172, 175, 178, 181, 184, 187, 190],
}
NEW = {g: (HERE / ("candidate-%s.spv" % g)).as_posix() for g in MAP}
recs = K.pass_lines_of(HEAD)
assert len(recs) == 232, len(recs)
lines = HEAD.read_text(encoding="utf-8-sig").splitlines()
out = list(lines)
touched, per_group = 0, {}
for g, ords in MAP.items():
    per_group[g] = []
    for o in ords:
        r = recs[o]
        idx = next(i for i, l in enumerate(lines) if l == r["line"])
        toks = out[idx].split()


        def norm(t):
            return str(t).strip('"').replace("\\", "/")
        pos = [i for i, t in enumerate(toks) if norm(t) == norm(r["spv"])]
        assert len(pos) == 1, (o, r["spv"], pos)
        q = '"' if toks[pos[0]].startswith('"') else ""
        toks[pos[0]] = q + NEW[g] + q
        out[idx] = " ".join(toks)
        touched += 1
        per_group[g].append(dict(ordinal=o, name=r["name"], old=r["spv"], new=NEW[g]))
assert touched == 36, touched
plan = HERE / "plan.txt"
if HEAD.read_text(encoding="utf-8-sig").endswith("\n"):
    plan.write_text("\n".join(out) + "\n", encoding="utf-8")
else:
    plan.write_text("\n".join(out), encoding="utf-8")

# ---- self checks
recs2 = K.pass_lines_of(plan)
assert len(recs2) == 232
diff = [i for i in range(232) if recs2[i]["line"] != recs[i]["line"]]
assert diff == sorted(sum(MAP.values(), [])), (len(diff), diff[:5])
for i in diff:
    a, b = recs[i]["line"].split(), recs2[i]["line"].split()
    assert len(a) == len(b)
    assert sum(1 for x, y in zip(a, b) if x != y) == 1, (i, a, b)
print("plan written:", plan, K.sha(plan))
print("PASS lines changed:", len(diff), "= 8 + 12 + 16")
print("cpu_check: running ...")
try:
    txt = K.cpu_check(plan)
except Exception as ex:                     # noqa: BLE001
    txt = "FAILED: %s" % ex
print(txt.strip()[:1200])
(HERE / "build" / "plan-build.json").write_text(json.dumps(
    dict(parent_plan=str(HEAD), parent_plan_sha=HEAD_SHA, plan=str(plan),
         plan_sha=K.sha(plan), changed_lines=len(diff), per_group=per_group,
         cpu_check=txt.strip()), indent=1), encoding="utf-8")
