def _str(words):
    bs = b''.join(int(w).to_bytes(4, 'little') for w in words)
    return bs.split(b'\x00')[0].decode('utf-8', 'replace')


def names(m):
    out = {}
    for op, ops, _, _ in m.instructions:
        if op == 5:
            out[ops[0]] = _str(ops[1:])
    return out


def funcs(m):
    """[(func_id, name, [instruction indices])]"""
    nm = names(m)
    regions = ST.function_regions(m)
    out = []
    for reg in regions:
        fid = None
        for i in reg:
            op, ops, _, _ = m.instructions[i]
            if op == 54:
                fid = ops[1]
                break
        out.append((fid, nm.get(fid, '?'), reg))
    return out


def hist(insts, m, idxs):
    c = Counter()
    for i in idxs:
        c[m.instructions[i][0]] += 1
    return c


def report(path, tag):
    m = ST.Module(Path(path))
    print('==', tag, Path(path).name, 'instr', len(m.instructions), 'words', len(m.words))
    nm = names(m)
    print('   globals named:', {k: v for k, v in nm.items() if k < 100})
    tot = Counter()
    for fid, name, reg in funcs(m):
        c = hist(None, m, reg)
        tot.update(c)
        key = {OPNAME.get(k, k): v for k, v in sorted(c.items())
               if k in (57, 61, 62, 65, 71, 72, 4457, 4458, 4459, 128, 130, 132, 134, 135, 137, 138, 139, 112, 113, 79, 80, 81, 82, 33, 43, 44)}
        print('   func', fid, repr(name), 'n=%d' % len(reg), key)
    print('   TOTAL', {OPNAME.get(k, k): v for k, v in sorted(tot.items()) if k in (57, 61, 62, 65, 71, 72, 4457, 4458, 4459, 128, 130, 132, 134, 135, 137, 138, 139, 112, 113, 79, 80, 81, 82)})
    nc = ST.no_contraction_ids(m)
    print('   NoContraction ids', len(nc), sorted(nc)[:20])
    # operand signature of each NC id
    for cid in sorted(nc):
        op, ops = m.defining_instruction(cid)
        print('     NC id', cid, OPNAME.get(op, op), m.operand_signature(cid, 3))
    print('   fma sites', len(m.ext_inst_fma_ordinals()))
    # storage bytes
    try:
        print('   storage_bytes', m.storage_bytes())
    except Exception as e:
        print('   storage_bytes failed', e)
    return m
