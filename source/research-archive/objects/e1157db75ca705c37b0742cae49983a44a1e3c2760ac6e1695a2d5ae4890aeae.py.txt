#!/usr/bin/env python3
"""F10 controlled-control ordinary timing: 3 pairs x 30 samples, alternating
arm order (p->c, c->p, p->c), warm 1 + feedback, pair = statistical unit.
All samples kept. Results to this line's evidence dir (does not touch the
Z package's ORDINARY-RESULTS.json)."""
import json
import math
import shutil
import subprocess
import time
from pathlib import Path

WS = Path(r"@NR_WORKSPACE@")
RUNNER = WS / "build/gpa-b321-prep/port_network_model_runner.exe"
PARENT = WS / "work/zcode-fp16-fullnet-handoff-20261024/candidates/combo1-gates-passed/plan-combo1-232.txt"
CAND = WS / "work/codex-astra-fp16-int8-audit-20261024/f10-controlled-source-20261025/plan-b1-earlier-232.txt"
OUT = WS / "work/zcode-combo1-profile-20261025/run"
T = 4.302652729911275


def arm(label, plan):
    out = OUT / label
    if out.exists():
        shutil.rmtree(out)
    r = subprocess.run([str(RUNNER), "--plan", str(plan), "--out", str(out),
                        "--pool", "--warm", "1", "--reps", "30", "--feedback"],
                       capture_output=True, text=True, cwd=str(WS))
    rec = out / "port_network_receipt.json"
    if not rec.exists():
        return None, (r.stdout + r.stderr)[-200:]
    j = json.loads(rec.read_text(encoding="utf-8"))
    return j.get("gpu_mean_ms"), j


def stats(ds):
    n = len(ds)
    m = sum(ds) / n
    sd = (sum((d - m) ** 2 for d in ds) / (n - 1)) ** 0.5
    half = T * sd / n ** 0.5
    return {"mean_ms": m, "ci95_ms": [m - half, m + half], "samples": ds}


def main():
    pairs, detail = [], []
    for i in (1, 2, 3):
        order = "p,c" if i % 2 == 1 else "c,p"
        res = {}
        for who in order.split(","):
            plan = PARENT if who == "p" else CAND
            label = f"f10ctl-{'p' if who=='p' else 'c'}-ord{i}"
            ms, rec = arm(label, plan)
            res[who] = (ms, rec)
        if res['p'][0] is None or res['c'][0] is None:
            detail.append({"pair": i, "error": "arm failed", "order": order})
            continue
        d = res['p'][0] - res['c'][0]
        pairs.append(d)
        detail.append({"pair": i, "order": order,
                       "parent_ms": res['p'][0], "cand_ms": res['c'][0], "delta": d,
                       "parent": {k: res['p'][1].get(k) for k in ("gpu_passes", "nan", "inf", "normal_close")},
                       "cand": {k: res['c'][1].get(k) for k in ("gpu_passes", "nan", "inf", "normal_close")}})
        print(f"pair {i} ({order}): p={res['p'][0]:.4f} c={res['c'][0]:.4f} delta={d:+.4f}", flush=True)
    out = {"pairs": detail}
    if len(pairs) == 3:
        s = stats(pairs)
        out["paired_stats"] = s
        out["verdict"] = ("FASTER(95%)" if s["ci95_ms"][0] > 0 else
                          ("SLOWER(95%)" if s["ci95_ms"][1] < 0 else "INDISTINCT"))
        print(f"verdict: {out['verdict']} mean={s['mean_ms']:+.4f} CI=[{s['ci95_ms'][0]:+.4f},{s['ci95_ms'][1]:+.4f}]")
    Path("evidence/f10ctl-ordinary.json").write_text(json.dumps(out, indent=1), encoding="utf-8")
    print("saved evidence/f10ctl-ordinary.json")


if __name__ == "__main__":
    main()
