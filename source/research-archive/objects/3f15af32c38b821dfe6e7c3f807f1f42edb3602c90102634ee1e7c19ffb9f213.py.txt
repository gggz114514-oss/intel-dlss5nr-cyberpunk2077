"""inspect the produced index (nat-index card)."""
import json, sys
from pathlib import Path

P = Path(__file__).parent / 'evidence' / 'NATIVE-INDEX.json'
j = json.loads(P.read_text(encoding='utf-8'))
rows = {c['ordinal']: c for c in j['coverage']}

print('== baseline capture records ==')
for c in j['captures']:
    if 'f10-weight-return-separation' in c['isa_dir']:
        print(c['capture_root'])
        print('  plan:', json.dumps(c['plan']))
        for e in c['records']:
            print('   p%-2d %-8s %s  passes=%s' % (e['pipeline_index'], e['identity'],
                                                  e['isa']['path'].split('\\')[-1][:30],
                                                  (e.get('pass_ordinals') or [])[:3]))

print('\n== pass coverage not CURRENT ==')
for o, c in rows.items():
    if c['status'] != 'CURRENT':
        print(o, c['name'], c['family'], c['status'], 'cur=%d hist=%d' % (len(c['current_isa']), len(c['historical_isa'])))

print('\n== captures with UNKNOWN identity records ==')
from collections import Counter
cnt = Counter()
for c in j['captures']:
    n = sum(1 for e in c['records'] if e['identity'] == 'UNKNOWN')
    if n:
        cnt[(c['capture_root'], c['mapping_mode'], (c['plan'] or {}).get('n_passes'))] = n
for k, v in list(cnt.items())[:30]:
    print('  ', v, k)

print('\n== mapping modes ==')
print(Counter(c['mapping_mode'] for c in j['captures']))
print('\n== plan n_passes ==')
print(Counter((c['plan'] or {}).get('n_passes') for c in j['captures']))
print('\n== plan sha source ==')
print(Counter((c['plan'] or {}).get('sha_source') for c in j['captures']))
print('\n== driver stats coverage ==')
print(Counter(bool(e.get('driver_stats')) for c in j['captures'] for e in c['records']))
