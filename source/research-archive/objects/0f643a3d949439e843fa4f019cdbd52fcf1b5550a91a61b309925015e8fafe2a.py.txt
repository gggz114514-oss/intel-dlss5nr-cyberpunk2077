"""Trace the true return-register window of every `send` in the F02 body dump.

Ground truth for "how many bytes does this load return": the dependency token
`$N` on the send and `$N.dst` on its consumers tells us exactly which GRFs the
message wrote -- the `rd:N` field alone is ambiguous for block messages.

CPU only.
"""
import re, sys, collections
sys.path.insert(0, '@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/fullnet-continuous-20261027/cards/n01b-f02-load-side')
import isaparse as P


def main(path):
    isa = P.parse(path)
    for k, i in enumerate(isa):
        i['idx'] = k
    sends = {}
    for i in isa:
        if i['op'].startswith('send'):
            for t in P.tokens(i):
                sends[t] = i
    use = collections.defaultdict(list)
    for i in isa:
        for t in P.tokens(i):
            if '.dst' in (i['deps'] or ''):
                use[t].append(i)
    rows = []
    for t, s in sorted(sends.items()):
        dregs = set()
        for i in use.get(t, []):
            r = P.reg(i['ops'][0]) if i['ops'] else None
            if r is not None:
                dregs.add(r)
        rows.append(dict(token=t, idx=s['idx'], addr=s['addr'], op=s['op'],
                         dst=P.reg(s['ops'][0]) if s['ops'] else None,
                         rd=P.rd_of(s), msg=P.msg_of(s),
                         consumers=[dict(addr=i['addr'], op=i['op'], dst=P.reg(i['ops'][0]) if i['ops'] else None)
                                    for i in use.get(t, [])],
                         consumer_dst=sorted(dregs)))
    return rows, isa


if __name__ == '__main__':
    rows, isa = main(sys.argv[1])
    print('sends', len(rows))
    print(collections.Counter((r['op'], r['rd']) for r in rows))
    print()
    for r in rows:
        if r['op'] == 'send.ugm' and r['rd'] == 8:
            print('0x%05X dst=r%s rd=%d ncons=%d dstregs=%s ops=%s %s'
                  % (r['addr'], r['dst'], r['rd'], len(r['consumers']),
                     r['consumer_dst'], sorted({c['op'] for c in r['consumers']}), r['msg']))
    print('---- slm')
    for r in rows:
        if r['op'] == 'send.slm' and r['rd'] == 8:
            print('0x%05X dst=r%s rd=%d ncons=%d dstregs=%s ops=%s %s'
                  % (r['addr'], r['dst'], r['rd'], len(r['consumers']),
                     r['consumer_dst'], sorted({c['op'] for c in r['consumers']}), r['msg']))
    print('---- ugm rd:1 (first 6 of %d)' % sum(1 for r in rows if r['op'] == 'send.ugm' and r['rd'] == 1))
    for r in [r for r in rows if r['op'] == 'send.ugm' and r['rd'] == 1][:6]:
        print('0x%05X dst=r%s ncons=%d dstregs=%s ops=%s %s'
              % (r['addr'], r['dst'], len(r['consumers']), r['consumer_dst'],
                 sorted({c['op'] for c in r['consumers']}), r['msg']))
    print('---- ugm rd:4')
