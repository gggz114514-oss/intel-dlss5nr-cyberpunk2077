"""S01 follow-up: pre-register the ablation predictions and build their plans.

(a) CPU prediction of the FFN-Y error for the three packs (same eval tiles as before);
(b) self-contained one-frame (34189) combo plans for the two ablation packs.
"""
import sys, json
from pathlib import Path
import numpy as np

sys.path.insert(0, str(Path(__file__).resolve().parent))
import c32lib as L
sys.path.insert(0, r"@NR_WORKSPACE@/work/dsh-fullnet-static-20261029/tools")
from planlib import Plan, sha256_file  # noqa: E402

HERE = Path(__file__).resolve().parent.parent
D = L.D
FRAMES = [("34189", D / "dump-fp16"), ("41444", D / "teacher-diagnostics/teacher-41444"),
          ("43654", D / "teacher-diagnostics/teacher-43654")]

# ---- reference: minmax pack (scalar X and scalar E) and its float reference
info_min = L.block_bindings(D / "c32-minmax/local-b4.txt", ordinal=4)
we_min, wc_min = L.load_w8(info_min["w8"])
fsc_min = L.load_fsc(info_min["fsc"])
wb = np.fromfile(info_min["wb"], dtype="<f2")
rs = wb[info_min["rs_base"] + 8:info_min["rs_base"] + 40].astype(np.float16)
sE_min = np.float32(fsc_min[2])
Wc_real = wc_min.astype(np.float32) * (fsc_min[132:164] / sE_min)[:, None]
tap_e_min = fsc_min[4:132]
invX_min = np.full(32, np.float32(fsc_min[1]))

PACKS = {
    "frozen_minmax": (D / "c32-minmax/b4-scales.bin", D / "c32-minmax/../c32-minmax/b4-scales.bin"),
}
rng = np.random.default_rng(777)
tiles = np.sort(rng.choice(168 * 96, size=2000, replace=False))
TT, CC = np.meshgrid(tiles, np.arange(16), indexing="ij")
TT = TT.ravel(); CC = CC.ravel()

variants = {
    "frozen_minmax": (info_min["w8"], D / "c32-minmax/b4-scales.bin"),
    "x_plus_e_full": (HERE / "PACK/c32_block4_xe_weights.i8.bin", HERE / "PACK/c32_block4_xe_scales.f32.bin"),
    "ablate_e_only": (HERE / "PACK/ablate-e-only-weights.i8.bin", HERE / "PACK/ablate-e-only-scales.f32.bin"),
    "ablate_e_half": (HERE / "PACK/ablate-e-half-weights.i8.bin", HERE / "PACK/ablate-e-half-scales.f32.bin"),
}
rows = []
for name, (w8p, scp) in variants.items():
    we, wc = L.load_w8(w8p)
    fsc = L.load_fsc(scp)
    invx = (fsc[L.FSC_INVX:L.FSC_INVX + 32] if fsc.size >= L.FSC_INVX + 32 else invX_min)
    inve = (fsc[L.FSC_INVE:L.FSC_INVE + 128] if fsc.size >= L.FSC_INVE + 128
            else np.full(128, np.float32(fsc[L.FSC_INV_E])))
    per = []
    for tag, folder in FRAMES:
        x_flat = np.fromfile(folder / "dump_block3_out.bin", dtype="<f2")
        x = L.gather(x_flat, TT, CC)
        xq = L.code8_ch(x, invx)
        acc = np.rint(we.astype(np.float64) @ xq.astype(np.float64)).astype(np.int32)
        a_h = L.act_h(L.to_half(acc.astype(np.float32) * fsc[4:132][:, None]))
        seed = L.widen(L.to_half(x * rs[:, None]))
        y_ref = Wc_real @ a_h.astype(np.float32) + seed
        eq = L.code8_ch(a_h, inve)
        sat = int(np.count_nonzero(np.abs(eq) >= 127))
        y = (np.rint(wc.astype(np.float64) @ eq.astype(np.float64)).astype(np.int32)
             .astype(np.float32) * fsc[132:164][:, None] + seed)
        per.append(dict(frame=tag, rel_l2=L.rel_l2(y, y_ref), saturated=sat, codes=int(eq.size)))
    rows.append(dict(variant=name, weights=str(w8p).replace("\\", "/"),
                     weights_sha256=sha256_file(w8p), scales=str(scp).replace("\\", "/"),
                     scales_sha256=sha256_file(scp), per_frame=per,
                     mean_rel_l2=float(np.mean([p["rel_l2"] for p in per]))))
    print(f"{name:16s} mean={rows[-1]['mean_rel_l2']:.5f} "
          + " ".join(f"{p['frame']}:{p['rel_l2']:.4f}/sat{p['saturated']}" for p in per))

# ---- one-frame ablation plans
out = {}
for name in ("ablate_e_only", "ablate_e_half"):
    w8p = HERE / "PACK" / (name.replace("_", "-") + "-weights.i8.bin")
    scp = HERE / "PACK" / (name.replace("_", "-") + "-scales.f32.bin")
    p = Plan(D / "c64-family-static/combo-34189.txt")
    p.set_spv(4, (HERE / "SOURCE/c32_b4_xe.spv").as_posix(), name="block4_swin_body_1")
    p.set_buffer("c32_block4_unified_weights", file=w8p.as_posix())
    p.set_buffer("c32_block4_unified_scales", file=scp.as_posix(), bytes_=1808)
    p.assert_valid()
    dest = HERE / f"candidate-34189-{name.replace('_', '-')}.txt"
    p.write(dest, check=False)
    Plan(dest).assert_valid()
    out[name] = dict(plan=str(dest).replace("\\", "/"), sha256=sha256_file(dest),
                     weights_sha256=sha256_file(w8p), scales_sha256=sha256_file(scp))
    print(name, out[name]["sha256"][:16])

(HERE / "CPU" / "ablation_preregistration.json").write_text(json.dumps(
    dict(metric="same FFN-Y model-relative metric as CALIBRATION.json", variants=rows,
         plans=out, eval_tiles=2000, tile_seed=777), indent=2))
