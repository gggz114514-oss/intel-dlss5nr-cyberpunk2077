"""S07 probe: dump split16 dispatch rows from the current candidate plan."""
import json
import sys
from pathlib import Path

B = Path("@NR_WORKSPACE@")
W = B / "work/dsh-fullnet-static-20261029"
sys.path.insert(0, str(W / "tools"))
from planlib import Plan, sha256_file  # noqa: E402

PLAN = B / "work/codex-astra-fp16-int8-audit-20261024/int8-quality-first-20261029/real-int8-24/c64-family-static/combo-34189.txt"

NAMES = [
    "split16_b23_l0_qx_producer",
    "split16_b23_l0",
    "split16_b23_l1",
    "split16_b23_l2",
    "split16_b23_l3",
    "split16_b30_l0",
    "split16_b30_l1",
    "split16_b30_l2",
    "b30_qt_producer",
    "split16_b30_l3",
    "split16_b30_l4",
]


def main() -> None:
    p = Plan(PLAN)
    print(f"plan sha={p.sha256()} header={p.header} problems={p.validate()}")
    for name in NAMES:
        hits = p.find_dispatches(name)
        if not hits:
            print(f"!! no dispatch named {name}")
            continue
        for d in hits:
            print(f"\n=== ordinal {d.ordinal} group {d.group} name {d.name}")
            print(f"    spv {d.spv}")
            print(f"    spv_sha {sha256_file(d.spv) if Path(d.spv).is_file() else 'MISSING'}")
            print(f"    tail {d.tail_tokens}")
            for b in d.binds:
                buf = p.buffers[b.buffer]
                print(f"    BIND {b.binding:2d} -> buf[{b.buffer}] {buf.name} off={b.offset} bytes={b.bytes} (bufbytes={buf.bytes}) file={buf.file}")
            print(f"    PUSH n={len(d.push)}: {d.push}")
    # policy rows for buffers touched by these dispatches
    touched = set()
    for name in NAMES:
        for d in p.find_dispatches(name):
            for b in d.binds:
                touched.add(b.buffer)
    pol = {x.buffer: x for x in p.policies}
    print("\n=== POLICY for touched buffers")
    for i in sorted(touched):
        x = pol.get(i)
        print(f"  buf[{i}] {p.buffers[i].name}: {None if x is None else (x.resident, x.first, x.last, x.clear_first)}")
    print("\n=== OUTPUT lines")
    for o in p.outputs:
        print("  ", o[:3])


if __name__ == "__main__":
    main()
