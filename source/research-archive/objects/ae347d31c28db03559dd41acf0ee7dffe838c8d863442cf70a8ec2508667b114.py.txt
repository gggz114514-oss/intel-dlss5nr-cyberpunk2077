"""Dump an ISA window (by address range) from the F02 body dump. CPU only."""
import sys
sys.path.insert(0, '@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/fullnet-continuous-20261027/cards/n01b-f02-load-side')
import isaparse as P

path = sys.argv[1]
lo = int(sys.argv[2], 16)
hi = int(sys.argv[3], 16)
isa = P.parse(path)
for k, i in enumerate(isa):
    if lo <= i['addr'] <= hi:
        print('%5d 0x%05X  %s' % (k, i['addr'], i['raw'][:150]))
