#!/usr/bin/env python3
"""Build the R12 plan candidate from the G3 production plan.

The only plan-level change is the SPV path and the declared shared bytes of the 16
`split16_bNN_l2` PASS lines whose PV segment moves to INT8:
  * 15 ordinary positions -> candidates/r12-split16-l2/spv/r12_split16_l2_pv_int8.spv
  * block30               -> candidates/r12-split16-l2/spv/r12_split16_b30_l2_pv_int8.spv
No BIND, PUSH, OUTPUT, POLICY or RESOURCE line changes: the candidate adds no descriptor
binding and no push word (all new scales are derived inside the workgroup).

usage: make_plan.py <g3-plan> <out-plan> <out-delta.json>
"""
import hashlib
import json
import re
import sys
from pathlib import Path

TOKEN = re.compile(r'"[^"]*"|\S+')
CARD = "@NR_WORKSPACE@/work/dsh-xmx-handoff-validation-20261024/candidates/r12-split16-l2"
SPV15 = CARD + "/spv/r12_split16_l2_pv_int8.spv"
SPV30 = CARD + "/spv/r12_split16_b30_l2_pv_int8.spv"
NAMES = {"split16_b%d_l2" % b for b in list(range(23, 31)) + list(range(40, 48))}
SHARED = 30724


def main(argv):
    src, dst, rep = Path(argv[0]), Path(argv[1]), Path(argv[2])
    text = src.read_text(encoding="utf-8")
    lines = text.splitlines(keepends=True)
    h = lines[0].split()
    nb, npass, nout = int(h[2]), int(h[3]), int(h[4])
    delta = []
    i = 1 + nb
    for _ in range(npass):
        t = TOKEN.findall(lines[i].rstrip("\r\n"))
        name = t[2].strip('"')
        if name in NAMES:
            row, spv, sg, sh, nbind, npush = (int(t[1]), t[3].strip('"'), int(t[4]),
                                              int(t[5]), int(t[6]), int(t[7]))
            new_spv = SPV30 if name == "split16_b30_l2" else SPV15
            old_line = lines[i]
            eol = "\r\n" if old_line.endswith("\r\n") else "\n"
            new_line = old_line.replace('"%s"' % spv, '"%s"' % new_spv).replace(
                " %d %d %d " % (sg, sh, nbind), " %d %d %d " % (sg, SHARED, nbind))
            assert new_line != old_line, name
            lines[i] = new_line
            delta.append({"plan_order": row, "pass": name, "spv_old": spv, "spv_new": new_spv,
                          "shared_old": sh, "shared_new": SHARED, "nbind": nbind,
                          "npush": npush, "grid": [int(x) for x in t[8:11]],
                          "line_old": old_line.rstrip("\r\n"), "line_new": new_line.rstrip("\r\n")})
        i += 1 + int(t[6]) + 1
    out = "".join(lines)
    dst.write_text(out, encoding="utf-8", newline="")
    rec = {"base_plan": str(src), "base_sha256": hashlib.sha256(src.read_bytes()).hexdigest(),
           "candidate_plan": str(dst),
           "candidate_sha256": hashlib.sha256(dst.read_bytes()).hexdigest(),
           "candidate_bytes": dst.stat().st_size,
           "changed_passes": len(delta), "shared_old": 24576, "shared_new": SHARED,
           "bind_push_output_policy_changes": 0,
           "deltas": delta}
    rep.write_text(json.dumps(rec, indent=1) + "\n", encoding="utf-8", newline="\n")
    print("changed", len(delta), "PASS lines; wrote", dst, rec["candidate_sha256"][:16])
    return 0 if len(delta) == 16 else 1


if __name__ == "__main__":
    sys.exit(main(sys.argv[1:]))
