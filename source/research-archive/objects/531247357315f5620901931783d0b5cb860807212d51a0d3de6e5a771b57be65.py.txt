import json, os, re, hashlib

ROOT = r"@NR_WORKSPACE@\work\codex-astra-fp16-int8-audit-20261024\fullnet-continuous-20261027"
PLAN = r"@NR_WORKSPACE@\work\codex-astra-fp16-int8-audit-20261024\fullnet-mechanism-transfer-20261026\f10-weight-return-separation-20261027\plan.txt"
OUT = os.path.join(ROOT, "cards", "n16-crossdispatch", "tools")

def parse_plan(path):
    lines = open(path, encoding="utf-8-sig").read().splitlines()
    head = lines[0].split()
    rows, nbuf, npass, nout = (int(x) for x in head[1:5])
    bufs, passes = [], []
    i = 1
    for _ in range(nbuf):
        t = lines[i].split()
        name = t[1].strip('"')
        bufs.append({"name": name, "bytes": int(t[2]), "path": t[3].strip('"')})
        i += 1
    for _ in range(npass):
        t = lines[i].split()
        e = {"ordinal": len(passes), "row": int(t[1]), "name": t[2].strip('"'), "spv": t[3].strip('"'),
             "subgroup": int(t[4]), "shared": int(t[5]), "nbinds": int(t[6]),
             "npush": int(t[7]), "dx": int(t[8]), "dy": int(t[9]), "dz": int(t[10]),
             "binds": []}
        i += 1
        for _ in range(e["nbinds"]):
            b = lines[i].split()
            e["binds"].append({"slot": int(b[1]), "res": int(b[2]),
                               "off": int(b[3]), "bytes": int(b[4])})
            i += 1
        i += 1
        passes.append(e)
    return bufs, passes

bufs, passes = parse_plan(PLAN)
fg = json.load(open(os.path.join(ROOT, "evidence", "FAMILY-GROUPS.json"), encoding="utf-8"))
rows = {}
for sha, g in fg["groups"].items():
    for o in g["ordinals"]:
        rows[o] = g

with open(os.path.join(OUT, "pass-table.tsv"), "w", encoding="utf-8") as f:
    f.write("ord\tfamily\tname\tgrid\tsg\tshared\tspv_sha\tbinds\n")
    for p in passes:
        g = rows[p["ordinal"]]
        bs = ";".join("%d:%s:%d@%d" % (b["slot"], bufs[b["res"]]["name"], bufs[b["res"]]["bytes"], b["off"]) for b in p["binds"])
        f.write("%d\t%s\t%s\t%dx%dx%d\t%d\t%d\t%s\t%s\n" % (
            p["ordinal"], g["family"], p["name"], p["dx"], p["dy"], p["dz"],
            p["subgroup"], p["shared"], g["spv_sha256"][:12], bs))

print("wrote pass-table.tsv", len(passes))
for p in passes:
    g = rows[p["ordinal"]]
    print("%3d %-4s %-28s grid=%dx%dx%d sg=%d shm=%d spv=%s" % (
        p["ordinal"], g["family"], p["name"], p["dx"], p["dy"], p["dz"], p["subgroup"], p["shared"], g["spv_sha256"][:8]))
