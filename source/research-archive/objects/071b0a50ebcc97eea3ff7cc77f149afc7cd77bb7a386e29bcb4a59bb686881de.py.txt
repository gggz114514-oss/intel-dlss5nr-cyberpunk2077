import json,struct,statistics as s
from pathlib import Path
D=Path(__file__).resolve().parent
def read(p):return json.loads(p.read_text(encoding='utf-8-sig'))
tails=[]
for rep,name in [(0,'stage-norm-publish'),(2,'ffn-only')]:
 p=D/'run'/f'c32-family-r{rep}-{name}'/'clock_order_4.bin'
 values=[]
 for i,(a,b,c,d) in enumerate(struct.iter_unpack('<IIII',p.read_bytes())):
  if a or b or c or d:values.append({'x':i//4%84,'y':i//4//84,'sg':i%4,'ticks':((c+(d<<32))-(a+(b<<32)))%2**64})
 row={'rep':rep,'phase':name,'mean':s.mean(x['ticks'] for x in values),'median':s.median(x['ticks'] for x in values),'top':sorted(values,key=lambda x:x['ticks'],reverse=True)[:8]};tails.append(row)
print('tails',tails)
(D/'MAIN-C32-FAMILY-TAILS.json').write_text(json.dumps(tails,indent=2),encoding='utf-8')
rows=[]
names=['control','ffn-groups','ffn-expand','ffn-activation-join','ffn-contract','ffn-null']
for rep in range(3):
 arms={}
 for name in names:
  p=D/'run'/f'c32-ffn-r{rep}-{name}';r=read(p/'port_network_receipt.json');t=next(x for x in r['per_pass_gpu'] if x['index']==1)
  vals=[]
  for a,b,c,d in struct.iter_unpack('<IIII',(p/'clock.bin').read_bytes()):
   if a or b or c or d:vals.append(((c+(d<<32))-(a+(b<<32)))%2**64)
  arms[name]={'ticks':s.mean(vals) if vals else 0,'median':s.median(vals) if vals else 0,'max':max(vals) if vals else 0,'ms':s.median(t['samples_ms'])}
 row={'rep':rep,'arms':arms,'closure':sum(arms[n]['ticks'] for n in names[2:5])/arms['ffn-groups']['ticks'],'perturbation':{n:arms[n]['ms']/arms['control']['ms']-1 for n in names[1:]}};rows.append(row)
 print('ffn',rep,{n:round(v['ticks'],1) for n,v in arms.items()},'closure',row['closure'])
(D/'MAIN-C32-FFN-REVIEW.json').write_text(json.dumps(rows,indent=2),encoding='utf-8')
