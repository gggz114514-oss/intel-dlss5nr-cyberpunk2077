"""Embed tools/gather-table.md (UTF-16LE, written by a shell redirect) into
CHAIN-AUDIT.md at the <!--GATHER-TABLE--> placeholder.  Encoding-safe: reads and
writes explicit UTF-8/UTF-16, no Chinese literals in this file."""
import sys
from pathlib import Path

CARD = Path(__file__).resolve().parents[1]
DOC = CARD / 'CHAIN-AUDIT.md'
TABLE = CARD / 'tools' / 'gather-table.md'
PLACEHOLDER = '<!--GATHER-TABLE-->'


def read_any(p):
    raw = p.read_bytes()
    if raw[:2] in (b'\xff\xfe', b'\xfe\xff'):
        return raw.decode('utf-16')
    return raw.decode('utf-8-sig')


def main():
    doc = DOC.read_text(encoding='utf-8')
    table = read_any(TABLE).strip()
    if PLACEHOLDER not in doc:
        print('placeholder already replaced; nothing to do')
        return 0
    DOC.write_text(doc.replace(PLACEHOLDER, table), encoding='utf-8')
    print('embedded %d lines of table into %s' % (table.count('\n') + 1, DOC.name))
    return 0


if __name__ == '__main__':
    sys.exit(main())
