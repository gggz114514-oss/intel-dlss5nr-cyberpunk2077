"""Baseline native-ISA facts for the F11 shader group (read-only evidence).

Reads the accepted native capture of the current baseline plan and derives the
per-category instruction accounting used by COST-LEDGER/NATIVE-PREDICTION.
No GPU is touched: this only parses existing .asm text.
"""
import hashlib, json, re, sys
from collections import Counter
from pathlib import Path

CAMP = Path('@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/fullnet-continuous-20261027')
CARD = CAMP / 'cards/n09-f09-f11-store'
sys.path.insert(0, str(CAMP))
import COMMON as K

ASM = Path('@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/'
           'fullnet-mechanism-transfer-20261026/f10-kv-retired-input/family/captures/'
           'native-f20-pixel-owner/ISA/CS_asm585aeef5165a31b8_simd32_0000000000000014_19_orig.asm')
LEDGER = Path('@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/'
              'fullnet-mechanism-transfer-20261026/remaining-amd-production-audit-20261027/INTEL-ROWS.json')

LINE = re.compile(r'^/\* \[([0-9A-F]{4})\]\s*\*/\s*(?:\((W)\)\s*)?([a-zA-Z][a-zA-Z0-9._]*)')


def main():
    text = ASM.read_text(encoding='utf-8', errors='replace')
    lines = text.splitlines()
    ops = Counter()
    seq = []
    for ln in lines:
        m = LINE.match(ln)
        if not m:
            continue
        off, w, op = m.group(1), m.group(2), m.group(3)
        ops[op] += 1
        seq.append((off, op, ln.strip()))
    stores = [s for s in seq if s[1] == 'send.ugm' and 'store.ugm' in s[2]]
    loads = [s for s in seq if s[1] == 'send.ugm' and 'load.ugm' in s[2]]
    first_send = min(int(s[0], 16) for s in loads + stores)
    last_dpas = max(int(s[0], 16) for s in seq if s[1].startswith('dpas'))
    ic = len(seq)
    ledger = K.read_json(LEDGER)
    row = [r for r in ledger if r['ordinal'] == 63][0]
    res = dict(
        isa_path=str(ASM), isa_sha256=hashlib.sha256(ASM.read_bytes()).hexdigest(),
        isa_bytes=ASM.stat().st_size,
        ledger_statistics=row['native']['statistics'],
        ledger_native_sha256=row['native']['native_sha256'],
        ledger_capture_plan=row['native']['capture_plan'],
        parsed_instruction_count=ic,
        opcode_counts=dict(ops.most_common()),
        dpas_count=ops.get('dpas.8x8', 0),
        load_messages=len(loads), store_messages=len(stores),
        store_forms=[s[2] for s in stores],
        store_offsets=[s[0] for s in stores],
        first_send_offset=hex(first_send), last_dpas_offset=hex(last_dpas),
        prologue_instructions_before_first_send=len([s for s in seq if int(s[0], 16) < first_send]),
        sync_counts={k: v for k, v in ops.items() if k.startswith('sync')},
        per_pass_gputime_ms=None,
        group_timing_ms=None,
    )
    per = []
    for r in ledger:
        if r['family'] == 'F11':
            t = r.get('current_baseline_timing_ms') or {}
            per.append(dict(ordinal=r['ordinal'], name=r['name'],
                            fixed=(t.get('fixed-frame0') or {}).get('median'),
                            fb2=(t.get('fb2-frame1') or {}).get('median')))
    res['per_pass_gputime_ms'] = per
    fx = [p['fixed'] for p in per if p['fixed']]
    fb = [p['fb2'] for p in per if p['fb2']]
    res['group_timing_ms'] = dict(
        fixed_sum=round(sum(fx), 6), fixed_min=min(fx), fixed_max=max(fx),
        fb2_sum=round(sum(fb), 6), fb2_min=min(fb), fb2_max=max(fb),
        n=len(per),
        scope='GpuTime single-hardware-pass, warm0/short-feedback per dispatch; SUM over 15 dispatches is a '
              'context number, not an ordinary-arm frame time',
    )
    K.write_json(CARD / 'NATIVE-BASELINE.json', res)
    print(json.dumps({k: res[k] for k in ('parsed_instruction_count', 'dpas_count', 'load_messages',
                                          'store_messages', 'prologue_instructions_before_first_send',
                                          'first_send_offset', 'last_dpas_offset', 'sync_counts',
                                          'isa_sha256', 'ledger_statistics', 'group_timing_ms')}, indent=1))
    print('opcode counts:', json.dumps(dict(ops.most_common())))


if __name__ == '__main__':
    main()
