# ---- source-level proof that only the QKV projection block changed -----------
ptxt = PARENT_SRC.read_text(encoding="utf-8-sig").splitlines()
ctxt = CAND_SRC.read_text(encoding="utf-8-sig").splitlines()
sm = difflib.SequenceMatcher(a=ptxt, b=ctxt, autojunk=False)
ops = []
for tag, i1, i2, j1, j2 in sm.get_opcodes():
    if tag == "equal":
        continue
    ops.append(dict(tag=tag, parent_lines=[i1 + 1, i2], cand_lines=[j1 + 1, j2],
                    parent_first=ptxt[i1] if i1 < len(ptxt) else None,
                    cand_first=ctxt[j1] if j1 < len(ctxt) else None))
# the QKV block in the parent is delimited by the row_phase loop and its comment
p_qkv_start = next(i for i, l in enumerate(ptxt) if "QKV 64" in l) + 1
p_qkv_end = next(i for i, l in enumerate(ptxt) if "row_phase values retire" in l) + 1
only_qkv = all(op["tag"] == "insert"                                          # pure insertion (comment)
               or (op["parent_lines"][0] >= p_qkv_start and op["parent_lines"][1] <= p_qkv_end)
               for op in ops)

# ---- per-statement equality of the QKV block --------------------------------
def qkv_stmts(lines, lo, hi):
    out = []
    for l in lines[lo:hi]:
        s = l.strip()
        if re.search(r"(coopMatLoad|coopMatMulAdd|coopMatStore)\(", s):
            out.append(re.sub(r"\bm0[ab]\b", "m0", s))
    return out

p_stmts = qkv_stmts(ptxt, p_qkv_start - 1, p_qkv_end)
c_qkv_start = next(i for i, l in enumerate(ctxt) if "N05-a: the two row_phase" in l)
c_qkv_end = next(i for i, l in enumerate(ctxt) if "both M blocks retire" in l)
c_stmts = qkv_stmts(ctxt, c_qkv_start, c_qkv_end)

CALL = re.compile(r"(coopMatLoad|coopMatStore)\(([^,()]+,)([^()]*)\)")
def addr_expr(s):
    """(kind, operand-buffer + address + stride + layout), variable-name independent."""
    out = []
    for m in CALL.finditer(s):
        args = [a.strip() for a in m.group(3).split(",")]
        if any("coopMat" in a for a in args):
            continue
        out.append((m.group(1),) + tuple(re.sub(r"\s+", "", re.sub(r"\bm0[ab]\b", "m0", a)) for a in args))
    return out

p_addr, c_addr = [], []
for s in p_stmts:
    p_addr += addr_expr(s)
for s in c_stmts:
    c_addr += addr_expr(s)
p_addr_set, c_addr_set = set(p_addr), set(c_addr)
missing = sorted(c_addr_set - p_addr_set)
mult_ok = not missing
mma = lambda L: sum(s.count("coopMatMulAdd(") for s in L)
p_mma, c_mma = mma(p_stmts), mma(c_stmts)

def dsc(l):
    m = re.search(r"rd:\d+; ([a-z0-9_.]+)", l.split("//", 1)[1]) if "//" in l else None
    return m.group(1) if m else None
def region(a, b):
    import collections
    return dict(send=collections.Counter(dsc(lines[i]) for i in range(a, b) if re.search(r"\bsend\b", lines[i])),
                dpas=sum(1 for i in range(a, b) if re.search(r"\bdpas\b", lines[i])))
rq, ra = region(0, FENCE), region(FENCE, len(lines))
parent_native = dict(isa=str(ISA), dpas_total=rq["dpas"] + ra["dpas"],

                     send_total=sum(rq["send"].values()) + sum(ra["send"].values()),
                     qkv=dict(rq["send"]), attn=dict(ra["send"]),
                     fence_lines=[FENCE])
