"""Extra census for ALUWR-AUDIT.md: sel breakdown, cast (pack/unpack) census,
and the per-call norm-section instruction budget.

sel sites are attributed by their immediate operands:
  * one immediate  6.2e-05  -> `max(v, eps)` clamp before math.rsqt   (norm)
  * two immediates         -> `clamp(y, lo, hi)` of the sealed exp bit construction
  * otherwise              -> operand swap / misc
"""
import sys, re
from collections import Counter
from pathlib import Path
HERE = Path(__file__).parent
sys.path.insert(0, str(HERE))
import _regions as R

for w in (64, 128, 256):
    ins, labels = R.load(w)
    ins, guard = R.classify(w, ins, labels)
    live = [i for i in ins if i["live"]]
    sel = [i for i in live if i["op"] == "sel"]
    imm = Counter()
    for i in sel:
        imms = re.findall(r"(?:^|\s)(-?\d+\.?\d*(?:e-?\d+)?):(?:f|hf|d|uw|w|ud)\b", i["args"])
        if any(x.startswith("6.2e-05") or x == "6.2e-05" for x in imms):
            imm["norm max(v,eps) clamp"] += 1
        elif len(imms) >= 2:
            imm["exp clamp(lo,hi)"] += 1
        else:
            imm["other sel (%d imm)" % len(imms)] += 1
    # pack/unpack round trips: mov to/from the same .hf dword in adjacent sequence
    pack = sum(1 for i in live if i["op"] == "mov" and re.search(r":uw\s+r\d+\.\d+<2;1,0>:uw", i["args"]))
    hf2f = sum(1 for i in live if i["op"] == "mov" and re.search(r":f\s+r\d+\.\d+<1;1,0>:hf", i["args"]))
    f2hf = sum(1 for i in live if i["op"] == "mov" and re.search(r":hf\s+r\d+\.\d+<1;1,0>:f", i["args"]))
    print("=" * 74)
    print("c%d live %d  sel %d" % (w, len(live), len(sel)))
    for k, v in imm.most_common():
        print("   %-28s %4d" % (k, v))
    print("   pack hf-pair (mov :uw from hf pair) %d ; widen f<-hf %d ; narrow hf<-f %d"
          % (pack, hf2f, f2hf))
    # norm section = first math.rsqt .. last math.rsqt of the first phase pair
    rs = [i["addr"] for i in live if i["op"] == "math.rsqt"]
    print("   rsqt %d  span %X..%X = %d instructions"
          % (len(rs), rs[0], rs[-1],
             len([i for i in live if rs[0] <= i["addr"] <= rs[-1]])))
    # how many instructions from the first instruction of a call to its rsqt
    print("   instructions per norm call (mean between rsqt pairs): %.1f"
          % ((rs[-1] - rs[0]) / 16 / 16.0))
