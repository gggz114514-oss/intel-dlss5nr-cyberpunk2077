#!/usr/bin/env python3
"""Compare parent-diag vs s37fn-diag1 debug dumps, staged by kernel section.

Usage:
  compare_diag_dumps.py <parent_out_dir> <diag1_out_dir> [--body]

Stages (ioport_70_ioport_body_debug, 25088 halfs):
  A tot      [0..127]       per-row half total (scalar tree vs XOR chain)
  B rawQ     [128..2175]    raw Q before norm (shArena read vs fragment)
  C rawK     [2176..4223]   raw K before norm
  D normQ    [10240..12287] normalized+scaled Q
  E normK    [12288..14335] normalized+scaled K
  F shP      [14336..18431] post-exp P
  G ctx      [18432..20479] PV context
  H shX      [20480..22527] final X (pre-fold-store)
--body also diffs dump_ioport_70_ioport_body.bin (33030144 halfs) with a ULP
histogram to cross-check the recorded gate failure (747354 / 96% 1ULP).
"""
import sys
import numpy as np
from pathlib import Path

STAGES = [
    ("A_tot", 0, 128),
    ("B_rawQ", 128, 2176),
    ("C_rawK", 2176, 4224),
    ("D_normQ", 10240, 12288),
    ("E_normK", 12288, 14336),
    ("F_shP", 14336, 18432),
    ("G_ctx", 18432, 20480),
    ("H_shX", 20480, 22528),
]


def ulp_key(bits):
    bits = bits.astype(np.uint32)
    sign = (bits & 0x8000) != 0
    mag = (bits & 0x7FFF).astype(np.uint32)
    return np.where(sign, 0x8000 - mag, 0x8000 + mag)


def load(path):
    a = np.fromfile(path, dtype="<u2")
    return a


def diff_stage(name, pa, ca, lo, hi):
    p = pa[lo:hi]
    c = ca[lo:hi]
    same = p == c
    n = hi - lo
    nd = int((~same).sum())
    if nd == 0:
        print(f"{name:8s} [{lo:5d}..{hi:5d}]  IDENTICAL ({n})")
        return
    idx = np.nonzero(~same)[0]
    first = idx[0]
    pk = ulp_key(p); ck = ulp_key(c)
    ulp = np.abs(pk.astype(np.int64) - ck.astype(np.int64))
    hist = {}
    for v, cnt in zip(*np.unique(ulp[idx], return_counts=True)):
        hist[int(v)] = int(cnt)
    top = sorted(hist.items())[:6]
    print(f"{name:8s} [{lo:5d}..{hi:5d}]  DIFF {nd}/{n} ({100.0*nd/n:.3f}%)  "
          f"first@abs={lo+first}  ULP-top={top}")
    for k in idx[:5]:
        print(f"    e={lo+k}: parent={p[k].astype('>u2').tobytes().hex()} "
              f"({np.float16(view16(p[k]))!r})  cand={c[k].astype('>u2').tobytes().hex()} "
              f"({np.float16(view16(c[k]))!r})  ulp={int(ulp[k])}")


def view16(u16):
    return np.frombuffer(np.array([u16], dtype="<u2").tobytes(), dtype="<f2")[0]


def main():
    pdir, cdir = Path(sys.argv[1]), Path(sys.argv[2])
    dbg_name = "dump_ioport_70_ioport_body_debug.bin"
    pa = load(pdir / dbg_name)
    ca = load(cdir / dbg_name)
    assert pa.size == ca.size == 25088, (pa.size, ca.size)
    print(f"parent={pdir}  diag1={cdir}")
    for name, lo, hi in STAGES:
        diff_stage(name, pa, ca, lo, hi)
    if "--body" in sys.argv:
        pb = load(pdir / "dump_ioport_70_ioport_body.bin")
        cb = load(cdir / "dump_ioport_70_ioport_body.bin")
        assert pb.size == cb.size, (pb.size, cb.size)
        diff_stage("BODY", pb, cb, 0, pb.size)


if __name__ == "__main__":
    main()
