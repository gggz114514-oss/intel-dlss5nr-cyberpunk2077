param([ValidateSet('install','restore')][string]$Mode='install')
$ErrorActionPreference='Stop'
$root=$PSScriptRoot
if(Get-Process Cyberpunk2077 -ErrorAction SilentlyContinue){throw 'Exit Cyberpunk normally first'}
$before=Get-Content -LiteralPath "$root/BEFORE.json" -Raw | ConvertFrom-Json
$package=Get-Content -LiteralPath "$root/PACKAGE.json" -Raw | ConvertFrom-Json
function Hash([string]$path){(Get-FileHash -LiteralPath $path -Algorithm SHA256).Hash.ToLowerInvariant()}
foreach($p in $before.PSObject.Properties){if((Hash "$root/before/$($p.Name)") -ne $p.Value.sha256){throw 'Backup hash mismatch'}}
foreach($p in $package.preserved.PSObject.Properties){if((Hash $p.Name) -ne $p.Value){throw 'Accepted artifact changed'}}
$boot=$before.'nrvk_bridge.dll'.path
if((Hash $boot) -notin @($before.'nrvk_bridge.dll'.sha256,$package.capture_dll_sha256)){throw 'Unexpected installed DLL'}
if($Mode -eq 'restore'){
    foreach($p in $before.PSObject.Properties){
        Copy-Item -LiteralPath "$root/before/$($p.Name)" -Destination $p.Value.path -Force
        if((Hash $p.Value.path) -ne $p.Value.sha256){throw 'Restore verification failed'}
    }
    @{status='ORIGINAL_FILES_BYTE_RESTORED';time=[DateTime]::UtcNow.ToString('o')} | ConvertTo-Json | Set-Content -LiteralPath "$root/RESTORED.json" -Encoding utf8
    Write-Output 'ORIGINAL_FILES_BYTE_RESTORED'
    return
}
Copy-Item -LiteralPath "$root/bin/nrvk_bridge.dll" -Destination $boot -Force
$manifest=($root -replace '\\','/')+'/manifest'
$bridge=Get-Content -LiteralPath "$root/before/nr_bridge.ini" -Raw
$bridge=$bridge -replace '(?m)^manifest_dir\s*=.*$',("manifest_dir = "+$manifest)
Set-Content -LiteralPath $before.'nr_bridge.ini'.path -Value $bridge -Encoding utf8
$panel=Get-Content -LiteralPath "$root/before/nr_panel.ini" -Raw
$panel=$panel -replace '(?m)^backend\s*=.*$','backend = 102'
$panel=$panel -replace '(?m)^102\s*=.*$',("102 = "+$manifest)
Set-Content -LiteralPath $before.'nr_panel.ini'.path -Value $panel -Encoding utf8
$settings=Get-Content -LiteralPath "$root/before/panel-config.json" -Raw | ConvertFrom-Json
$settings.vk102_manifest=$manifest
$settings | ConvertTo-Json -Depth 8 | Set-Content -LiteralPath $before.'panel-config.json'.path -Encoding utf8
$after=@{}
foreach($p in $before.PSObject.Properties){$after[$p.Name]=Hash $p.Value.path}
if($after['nrvk_bridge.dll'] -ne $package.capture_dll_sha256){throw 'Mounted DLL hash mismatch'}
@{status='MOUNTED_GAME_NOT_STARTED';before=$before;after=$after;preserved=$package.preserved;selected='102 / Full capture';capture_armed=$false} | ConvertTo-Json -Depth 8 | Set-Content -LiteralPath "$root/INSTALL.json" -Encoding utf8
Write-Output 'CAPTURE_MOUNTED; game not started; capture not armed'
