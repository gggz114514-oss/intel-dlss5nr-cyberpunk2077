"""Card-private Intel ISA (GTPin --dump_isa asm5) parser for n02c-f02-fb-bias.

Grammar identical to the campaign's ISA.py plus the fix n01b established: the
dependency token group `{...}` precedes the `// message` comment, so a parser
that only accepts `{...}` at end-of-line loses deps.  CPU only, read-only.
"""
import re
from pathlib import Path

LINE = re.compile(r'^/\*\s*\[(?P<addr>[0-9A-Fa-f]+)\]\s*\*/\s*(?P<rest>.*)$')
TOKGRP = re.compile(r'\{\s*(?P<deps>[^}]*)\}')
COMMENT = re.compile(r'//\s*(?P<c>.*)$')


def parse(path):
    out = []
    for raw in Path(path).read_text(encoding='utf-8-sig', errors='replace').splitlines():
        m = LINE.match(raw)
        if not m:
            continue
        rest = m.group('rest')
        deps = None
        dm = TOKGRP.search(rest)
        if dm:
            deps = dm.group('deps').strip()
            rest = rest[:dm.start()] + rest[dm.end():]
        comment = None
        cm = COMMENT.search(rest)
        if cm:
            comment = cm.group('c').strip()
            rest = rest[:cm.start()]
        toks = rest.split()
        if not toks:
            continue
        flags = ''
        if toks[0].startswith('(') and toks[0].endswith(')'):
            flags = toks[0]
            toks = toks[1:]
        if not toks:
            continue
        op = toks[0]
        mw = re.match(r'^(?:send\.\w+|dpas[\w.]*|mov|add|mul|mad|sel)\s+\((\d+)\|M0\)',
                      rest)
        width = int(mw.group(1)) if mw else None
        ops = [t for t in toks[1:] if not (t.startswith('(') and t.endswith(')'))]
        out.append(dict(addr=int(m.group('addr'), 16), flags=flags, op=op, ops=ops,
                        deps=deps, comment=comment, width=width, raw=raw.rstrip()))
    return out


def reg(op):
    m = re.match(r'^r(\d+)', op or '')
    return int(m.group(1)) if m else None


def rd_of(i):
    m = re.search(r'rd:(\d+)', i['comment'] or '')
    return int(m.group(1)) if m else 0


def wr_of(i):
    m = re.search(r'wr:(\d+)\+(\d+)', i['comment'] or '')
    return (int(m.group(1)), int(m.group(2))) if m else (0, 0)


def msg_of(i):
    c = i['comment'] or ''
    return c.split(';')[-1].strip()


def desc_of(i):
    """The raw message-descriptor hex (second 0x token of a send)."""
    hexes = [t for t in i['ops'] if re.match(r'^0x[0-9A-Fa-f]+$', t)]
    return hexes[-1] if hexes else None


def tokens(i):
    return [int(x) for x in re.findall(r'\$(\d+)', i['deps'] or '')]


def lane_width(i):
    """Execution size of the instruction, from the `<n|M0>` region token."""
    if i.get('width') is not None:
        return i['width']
    m = re.search(r'\((\d+)\|M0\)', i['raw'])
    return int(m.group(1)) if m else None
