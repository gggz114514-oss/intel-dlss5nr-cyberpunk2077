#!/usr/bin/env python3
"""Ad-hoc: locate the extra OpUDiv instructions in the qx arm."""
import sys
from pathlib import Path
sys.path.insert(0, str(Path(__file__).resolve().parent))
import spv_read as S  # noqa: E402

CARD = Path("@NR_WORKSPACE@/work/dsh-xmx-handoff-validation-20261024/candidates/r03-ioport4")
for name in ("parent_repro", "r03_ioport4_i8", "r03_ioport4_i8_qx"):
    m = S.Module(CARD / "spv" / (name + ".spv"))
    hits = [(i, a, wc) for i, (op, a, wc) in enumerate(m.ins) if op == 134]
    print("==", name, "UDiv", len(hits))
    for i, a, wc in hits:
        print("   idx=%d wc=%d ops=%s" % (i, wc, a))
        for v in a[2:4]:
            d = m.defining(v)
            if d:
                print("       id %d <- %s %s" % (v, m.name(d[0]), d[1]))
            else:
                c = m.consts.get(v)
                print("       id %d const %s" % (v, c))
