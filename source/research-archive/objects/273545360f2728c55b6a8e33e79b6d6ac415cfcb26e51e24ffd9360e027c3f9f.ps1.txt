$ErrorActionPreference='Stop'
$gameRoot='@NR_GAME@/bin/x64'
$pkg='@NR_WORKSPACE@/work/github-experimental-game-install-20261024/package'
$lock='@NR_GPU_WORKSPACE@/artifacts/parallel-port-20261013/GPU.lock'
$run=$PSScriptRoot
if(Get-Process Cyberpunk2077 -ErrorAction SilentlyContinue){throw 'Cyberpunk is already running'}
$stream=[IO.File]::Open($lock,[IO.FileMode]::Open,[IO.FileAccess]::ReadWrite,[IO.FileShare]::None)
$owned=$false;$record=$null
function SaveLock { $bytes=[Text.Encoding]::UTF8.GetBytes(($record|ConvertTo-Json -Depth 8));$stream.Position=0;$stream.SetLength(0);$stream.Write($bytes,0,$bytes.Length);$stream.Flush($true) }
try {
 $reader=[IO.StreamReader]::new($stream,[Text.Encoding]::UTF8,$true,4096,$true);$previous=$reader.ReadToEnd()|ConvertFrom-Json;$reader.Dispose()
 if($previous.state -ne 'FINISHED'){throw "GPU lock not FINISHED: $($previous.owner)/$($previous.state)"}
 $lockBefore=Join-Path $run 'GPU.lock.before.json';[IO.File]::WriteAllText($lockBefore,($previous|ConvertTo-Json -Depth 8),[Text.UTF8Encoding]::new($false))
 $m=Get-Content -Raw (Join-Path $run 'INSTALL.json')|ConvertFrom-Json
 $asi=Join-Path $run 'bin\CyberpunkNRBridge-diag.asi'
 $dll=Join-Path $run 'bin\nrvk_bridge-diag.dll'
 Copy-Item -LiteralPath $asi -Destination (Join-Path $gameRoot 'plugins\CyberpunkNRBridge.asi') -Force
 Copy-Item -LiteralPath $dll -Destination (Join-Path $gameRoot 'nrvk_bridge.dll') -Force
 Copy-Item -LiteralPath (Join-Path $run 'install-nr_bridge.ini') -Destination (Join-Path $gameRoot 'plugins\nr_bridge.ini') -Force
 Copy-Item -LiteralPath (Join-Path $run 'install-nr_panel.ini') -Destination (Join-Path $pkg 'config\nr_panel.ini') -Force
 $diagIni = "[diag]`r`nmode=0`r`nwindow_id=0`r`narm_reuse=0`r`nflush_seq=0`r`nout_dir=@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/prep-gpu-handoff-20261028/dsh-time-diagnosis-20261028`r`n"
 [IO.File]::WriteAllText((Join-Path $gameRoot 'plugins\dsh_diag_control.ini'), $diagIni, [Text.UTF8Encoding]::new($false))
 foreach($name in @('CyberpunkNRBridge.asi','nrvk_bridge.dll','nr_bridge.ini','nr_panel.ini')){
  if($name -eq 'CyberpunkNRBridge.asi'){$target=Join-Path $gameRoot 'plugins\CyberpunkNRBridge.asi'}
  elseif($name -eq 'nrvk_bridge.dll'){$target=Join-Path $gameRoot 'nrvk_bridge.dll'}
  elseif($name -eq 'nr_bridge.ini'){$target=Join-Path $gameRoot 'plugins\nr_bridge.ini'}
  else{$target=Join-Path $pkg 'config\nr_panel.ini'}
  $h=(Get-FileHash -LiteralPath $target -Algorithm SHA256).Hash.ToLowerInvariant()
  if($h -ne $m.installed_sha256.PSObject.Properties[$name].Value){throw "Installed file identity mismatch: $name"}
 }
 $record=[ordered]@{owner='CP2077-DSH-DIAGNOSIS-20261028';holder_pid=$PID;gpu_pid=$null;state='RUNNING';started_utc=[DateTime]::UtcNow.ToString('o');authorization=$m.authorization;lock_mode='FileShare.None held for game lifetime';previous_owner=$previous.owner;previous_state=$previous.state;arm='diag-d1-d2';asi_sha256=$m.diag_asi_sha256;full_dll_sha256=$m.diag_dll_sha256;expected_build=$m.expected_build;expected_backend_route='small'}
 SaveLock;$owned=$true
 Add-Type @"
using System;
using System.Runtime.InteropServices;
public class DefaultDesktopProc {
    [StructLayout(LayoutKind.Sequential, CharSet = CharSet.Unicode)]
    public struct STARTUPINFO {
        public int cb;
        public string lpReserved;
        public string lpDesktop;
        public string lpTitle;
        public int dwX, dwY, dwXSize, dwYSize, dwXCountChars, dwYCountChars, dwFillAttribute, dwFlags;
        public short wShowWindow, cbReserved2;
        public IntPtr lpReserved2, hStdInput, hStdOutput, hStdError;
    }
    [StructLayout(LayoutKind.Sequential)]
    public struct PROCESS_INFORMATION {
        public IntPtr hProcess, hThread;
        public int dwProcessId, dwThreadId;
    }
    [DllImport("kernel32.dll", SetLastError = true, CharSet = CharSet.Unicode)]
    public static extern bool CreateProcessW(
        string lpApplicationName, string lpCommandLine, IntPtr lpProcessAttributes, IntPtr lpThreadAttributes,
        bool bInheritHandles, uint dwCreationFlags, IntPtr lpEnvironment, string lpCurrentDirectory,
        ref STARTUPINFO lpStartupInfo, out PROCESS_INFORMATION lpProcessInformation);
    [DllImport("kernel32.dll")]
    public static extern bool CloseHandle(IntPtr hObject);
}
"@
 $exePath = Join-Path $gameRoot 'Cyberpunk2077.exe'
 $si = New-Object DefaultDesktopProc+STARTUPINFO
 $si.cb = [Runtime.InteropServices.Marshal]::SizeOf($si)
 $si.lpDesktop = 'WinSta0\Default'
 $si.dwFlags = 1
 $si.wShowWindow = 1
 $pi = New-Object DefaultDesktopProc+PROCESS_INFORMATION
 if (-not [DefaultDesktopProc]::CreateProcessW($exePath, "`"$exePath`"", [IntPtr]::Zero, [IntPtr]::Zero, $false, 0, [IntPtr]::Zero, $gameRoot, [ref]$si, [ref]$pi)) {
  throw "CreateProcessW on WinSta0\Default failed: $([Runtime.InteropServices.Marshal]::GetLastWin32Error())"
 }
 [void][DefaultDesktopProc]::CloseHandle($pi.hThread)
 [void][DefaultDesktopProc]::CloseHandle($pi.hProcess)
 $proc=Get-Process -Id $pi.dwProcessId
 $record.gpu_pid=$proc.Id;SaveLock;Write-Host "Game PID=$($proc.Id); arm=diag-d1-d2; ASI SHA256=$($m.diag_asi_sha256); DLL SHA256=$($m.diag_dll_sha256); expected build=$($m.expected_build); route=small"
 $proc.WaitForExit();$record.game_exit_code=$proc.ExitCode
} finally {
 if($owned){$record.state='FINISHED';$record.gpu_pid=$null;$record.finished_utc=[DateTime]::UtcNow.ToString('o');SaveLock;[IO.File]::WriteAllText((Join-Path $run 'USER-TEST-GPU-RECEIPT.json'),($record|ConvertTo-Json -Depth 8),[Text.UTF8Encoding]::new($false))}
 $stream.Dispose()
}
