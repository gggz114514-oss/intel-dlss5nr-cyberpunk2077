def parse_plan(path):
    lines = open(path, encoding="utf-8-sig").read().splitlines()
    head = lines[0].split()
    nbuf, npass, nout = (int(x) for x in head[2:5])
    bufs, passes = [], []
    i = 1
    for _ in range(nbuf):
        t = lines[i].split()
        bufs.append({"name": t[1].strip('"'), "bytes": int(t[2])})
        i += 1
    for _ in range(npass):
        t = lines[i].split()
        e = {"ordinal": len(passes), "name": t[2].strip('"'), "spv": t[3].strip('"'),
             "shared": int(t[5]), "nbinds": int(t[6]), "npush": int(t[7]),
             "dx": int(t[8]), "dy": int(t[9]), "dz": int(t[10]), "binds": []}
        i += 1
        for _ in range(e["nbinds"]):
            b = lines[i].split()
            e["binds"].append({"slot": int(b[1]), "res": int(b[2]), "off": int(b[3]), "bytes": int(b[4])})
            i += 1
        p = lines[i].split()
        e["push"] = [int(x) for x in p[1:]]
        i += 1
        passes.append(e)
    return bufs, passes
