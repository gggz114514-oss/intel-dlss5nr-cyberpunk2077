"""Quick structural probe of parent vs candidate SPIR-V (CPU only, no disassembler)."""
import sys
from pathlib import Path
CARD = Path(__file__).resolve().parent.parent
sys.path.insert(0, str(CARD / 'scratch'))
import spvstat

def probe(tag, path):
    m = spvstat.Mod(path)
    h = m.hist
    print('== %s %s' % (tag, path))
    print('  instr %d words %d' % (len(m.instr), len(m.words)))
    for op in (61, 62, 12, 4457, 4458, 4459, 245, 246, 249, 250, 247):
        print('  %-6s %-32s %d' % (op, m.opname(op), h.get(op, 0)))
    # cooperative matrix load calls: operands = result type, result, ptr, stride const, layout
    idx = 0
    for op, o in m.instr:
        if op == 4457:
            print('  coopLoad@%4d  ptr=%s stride_const=%s colmajor=%s  words=%s'
                  % (idx, o[2], o[3], o[4], list(o)))
        if op == 4458:
            print('  coopStore@%4d ptr=%s stride_const=%s colmajor=%s'
                  % (idx, o[2], o[3], o[4]))
        idx += 1
    # loop structure
    print('  loop headers:', [ (i, list(o)) for i,(op,o) in enumerate(m.instr) if op == 245 ][:8])

for tag, p in (('parent', sys.argv[1]), ('cand', sys.argv[2])):
    probe(tag, p)
