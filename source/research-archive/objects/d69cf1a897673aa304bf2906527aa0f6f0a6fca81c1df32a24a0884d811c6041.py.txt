"""Exhaustive CPU proof of the n19 blend-address collapse (integer only).

Old (baseline) and new (candidate) index formulas are evaluated side by side for
*every* combination of the inputs that the 8 literal call sites can produce.
No GPU, no SPIR-V: pure integer arithmetic, so this is a complete proof over the
reachable domain (the domain bound is itself proved from the source text).

Old:
  sx = slot % 4; sy = slot / 4
  px = (2*wgx + (sg&1))*4 + sx;  py = (2*wgy + (sg>>1))*4 + sy
  t  = ((py/4)*(W/4) + px/4)*16 + 4*(py%4) + px%4
  sa = p4z + ((t/16)*(32/16) + ch/16)*256 + (t%16)*16 + ch%16
  ct = (2*(sg>>1) + (sy>>1))*4 + 2*(sg&1) + (sx>>1);  shk = ct*32 + ch
New:
  ax = 2*wgx + (sg&1); ay = 2*wgy + (sg>>1)
  sa = p4z + (ay*(W/4) + ax)*512 + (ch/16)*256 + slot*16 + ch%16
  shk = 256*(sg>>1) + 64*(sg&1) + ch + 128*(j>=4) + 32*(j&1)
"""
import json
import sys
from pathlib import Path

CAMP = Path('@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/fullnet-continuous-20261027')
CARD = CAMP / 'cards/n19-f19-root'

U32 = 0xFFFFFFFF


def old_sa(sg, slot, ch, wgx, wgy, W, p4z):
    sx, sy = slot % 4, slot // 4
    px = (2 * wgx + (sg & 1)) * 4 + sx
    py = (2 * wgy + (sg >> 1)) * 4 + sy
    t = ((py // 4) * (W // 4) + px // 4) * 16 + 4 * (py % 4) + px % 4
    return p4z + ((t // 16) * (32 // 16) + ch // 16) * 256 + (t % 16) * 16 + ch % 16


def old_shk(sg, slot, ch):
    sx, sy = slot % 4, slot // 4
    ct = (2 * (sg >> 1) + (sy >> 1)) * 4 + 2 * (sg & 1) + (sx >> 1)
    return ct * 32 + ch


def new_sa(sg, slot, ch, wgx, wgy, W, p4z):
    ax = 2 * wgx + (sg & 1)
    ay = 2 * wgy + (sg >> 1)
    return p4z + (ay * (W // 4) + ax) * 512 + (ch // 16) * 256 + slot * 16 + ch % 16


def new_shk(sg, slot, ch, j):
    return 256 * (sg >> 1) + 64 * (sg & 1) + ch + 128 * (1 if j >= 4 else 0) + 32 * (j & 1)


def main():
    checked = 0
    bad = []
    wgs = [0, 1, 2, 3, 20, 41, 83]
    Ws = [16, 32, 64, 128, 4096]
    p4zs = [0, 7, 1024]
    for lane in range(32):
        hi = lane >> 4
        ch = 2 * (lane & 15)
        for sg in range(4):
            for j in range(8):
                slot = 2 * j + hi           # <- the only slot values the 8 sites create
                for wgx in wgs:
                    for wgy in wgs:
                        for W in Ws:
                            for p4z in p4zs:
                                a = old_sa(sg, slot, ch, wgx, wgy, W, p4z)
                                b = new_sa(sg, slot, ch, wgx, wgy, W, p4z)
                                checked += 1
                                if a != b:
                                    bad.append(('sa', lane, sg, j, wgx, wgy, W, p4z, a, b))
                a, b = old_shk(sg, slot, ch), new_shk(sg, slot, ch, j)
                checked += 1
                if a != b:
                    bad.append(('shk', lane, sg, j, a, b))
    # the domain precondition itself: every literal call site passes slot < 16
    slots = sorted({2 * j + hi for j in range(8) for hi in (0, 1)})
    domain_ok = slots == list(range(16))
    # where the identity WOULD break: slot >= 16 (no reachable call site produces it)
    outside = []
    for slot in (16, 17, 31):
        for ch in (0, 2, 30):
            a = old_shk(0, slot, ch)
            b = new_shk(0, slot, ch, slot // 2)
            outside.append(dict(slot=slot, ch=ch, old=a, new=b, equal=a == b))
    res = dict(comparisons=checked, mismatches=len(bad), first_mismatches=bad[:5],
               reachable_slots=slots, domain_is_0_15=domain_ok,
               outside_domain_note=outside,
               note='slot = 2*j + (lane>>4) for the 8 literal call sites -> [0,16); '
                    'sy=slot/4 in [0,4) and sx=slot%4 in [0,4) make every /4 %4 /16 %16 '
                    'in nrf2_io_at0 an exact identity.')
    (CARD / 'proof-address.json').write_text(json.dumps(res, indent=2), encoding='utf-8')
    print(json.dumps({k: v for k, v in res.items() if k != 'outside_domain_note'}, indent=2))
    sys.exit(0 if (not bad and domain_ok) else 1)


if __name__ == '__main__':
    main()
