"""Address / statement provenance proof (CPU only).

Normalises the candidate's QKV block by rewriting m0a and m0b back to m0, then
checks that the multiset of statement lines is exactly two copies of the parent's
QKV block (the parent executed that block twice, once per row_phase).  Anything
that does not match is reported by line.
"""
import json
import re
from collections import Counter
from pathlib import Path

CARD = Path(__file__).resolve().parent.parent
BUILD = CARD / '_build'


def block(path, lo_pat, hi_pat):
    lines = Path(path).read_text(encoding='utf-8').splitlines()
    lo = next(i for i, l in enumerate(lines) if re.search(lo_pat, l))
    hi = next(i for i, l in enumerate(lines) if re.search(hi_pat, l))
    return lines[lo:hi + 1]


def norm(lines, subs):
    out = []
    for l in lines:
        s = l
        for a, b in subs:
            s = s.replace(a, b)
        s = re.sub(r'\s+', ' ', s).strip()
        if not s or s.startswith('//'):
            continue
        out.append(s)
    return out


def unify(lines):
    """Collapse the two M instances (鈥 / 鈥) and the two A fragments back to the
    parent's single-instance names, so a pure rename does not count as a diff."""
    out = []
    for l in lines:
        s = l
        for t in ('Q0', 'Q1', 'K0', 'K1', 'V0', 'V1'):
            s = s.replace('acc%s%s' % (t, 'a'), 'acc' + t).replace('acc%s%s' % (t, 'b'), 'acc' + t)
        s = s.replace('MAT_A a0; coopMatLoad(a0,', 'MAT_A a; coopMatLoad(a,')
        s = s.replace('MAT_A a1; coopMatLoad(a1,', 'MAT_A a; coopMatLoad(a,')
        s = s.replace('coopMatMulAdd(a0,', 'coopMatMulAdd(a,').replace('coopMatMulAdd(a1,',
                                                                     'coopMatMulAdd(a,')
        s = re.sub(r'\bm0[ab]\b', 'm0', s)
        s = re.sub(r'\s+', ' ', s).strip()
        out.append(s)
    return out


def main():
    par = block(CARD / 'parent-rebuild.comp', r'^    // AN6:', r'row_phase values retire')
    cand = block(CARD / 'candidate.comp', r'^    // AN6:', r'both M blocks retire')
    p = norm(par, [])
    c = unify(norm(cand, []))
    pc, cc = Counter(p), Counter(c)
    only_parent = {k: v for k, v in (pc - cc).items()}
    only_cand = {k: v for k, v in (cc - pc).items()}
    # every parent statement must appear exactly twice in the candidate, except
    # the loop header / declarations that the candidate re-expresses
    res = dict(
        parent_statements=len(p), candidate_statements=len(c),
        distinct_parent_statements=len(pc),
        matched_exactly_twice=sum(1 for k, v in pc.items() if cc.get(k, 0) == 2 * v),
        parent_statements_absent_in_candidate=sorted(k for k in pc if cc.get(k, 0) == 0),
        count_multiples={k: [v, cc.get(k, 0)] for k, v in sorted(pc.items())
                         if cc.get(k, 0) != 2 * v},
        stores=dict(
            parent=[x for x in p if x.startswith('coopMatStore')],
            candidate_normalised=[x for x in c if x.startswith('coopMatStore')],
        ),
    )
    res['store_multisets_equal_after_doubling'] = (
        Counter(res['stores']['parent'] * 2) == Counter(res['stores']['candidate_normalised']))
    (BUILD / 'addresses.json').write_text(json.dumps(res, indent=2), encoding='utf-8')
    print('parent statements', len(p), 'candidate', len(c))
    print('matched exactly twice:', res['matched_exactly_twice'], '/', len(pc))
    print('absent in candidate:', res['parent_statements_absent_in_candidate'])
    print('multiplicity != 2 (parent -> candidate):')
    for k, v in res['count_multiples'].items():
        print('   %d -> %d : %s' % (v[0], v[1], k))
    print('store multiset equal after doubling:', res['store_multisets_equal_after_doubling'])
    print('parent stores:', len(res['stores']['parent']),
          'candidate stores:', len(res['stores']['candidate_normalised']))


if __name__ == '__main__':
    main()
