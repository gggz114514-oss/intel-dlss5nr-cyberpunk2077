"""Score the 3 patched runs that appeared in S08-vit/runs/ (read-only) with the project metric.

Each run dir carries its own port_network_receipt.json, so the plan identity is read from the
receipt rather than assumed.
"""
import hashlib
import json
from pathlib import Path

import numpy as np

R = Path("@NR_WORKSPACE@")
S = R / "work/dsh-fullnet-static-20261029/S08-vit"
Q = R / "work/codex-astra-fp16-int8-audit-20261024/int8-quality-first-20261029"
D = Q / "real-int8-24"
O = D / "c64-family-static"

jobs = json.loads((D / "JOBS.json").read_text(encoding="utf-8"))
refs = {j["frame"]: j for j in jobs if isinstance(j, dict) and j.get("reference")}
q24 = {r["frame"]: r for r in json.loads((O / "QUALITY-24.json").read_text(encoding="utf-8"))["rows"]}

out = {"runs": {}, "note": "scoring only; S08 did not create these runs"}
rows = []
for run in sorted((S / "runs").glob("run-*")):
    f = int(run.name.split("-")[1])
    fout = run / "replayed-network.bin"
    rec = json.loads((run / "port_network_receipt.json").read_text(encoding="utf-8"))
    ref = np.fromfile(refs[f]["reference"], dtype="<f4").reshape(2, 720, 1280, 4)
    x = np.fromfile(fout, dtype="<f4").reshape(ref.shape)
    e = x[0, :, :, :3].astype(float) - ref[0, :, :, :3].astype(float)
    mse = float(np.mean(e * e))
    psnr = float(-10 * np.log10(mse))
    old = q24[f]["static_c64"]["psnr"]
    rows.append(dict(frame=f, psnr=psnr, old_combo_psnr=old, delta_db=psnr - old,
                     fine_c32=q24[f]["fine_c32"]["psnr"],
                     output_sha256=hashlib.sha256(fout.read_bytes()).hexdigest(),
                     status=rec.get("status"), gpu_passes=rec.get("gpu_passes"),
                     allocation_mode=rec.get("allocation_mode")))
    out["runs"][str(f)] = rows[-1]
    print(f, round(psnr, 6), "old", round(old, 6), "delta", round(psnr - old, 6), flush=True)

if rows:
    out["summary"] = dict(
        frames=len(rows), mean_psnr=float(np.mean([r["psnr"] for r in rows])),
        mean_old=float(np.mean([r["old_combo_psnr"] for r in rows])),
        mean_delta_db=float(np.mean([r["delta_db"] for r in rows])),
        all_improved=bool(all(r["delta_db"] > 0 for r in rows)),
        min_delta_db=float(min(r["delta_db"] for r in rows)),
        worst_frame=int(min(rows, key=lambda r: r["delta_db"])["frame"]),
        scope="3 calibration frames only (34189/41444/43654); the other 21 are not measured here")
    print(json.dumps(out["summary"], indent=1))
(S / "END-TO-END-3FRAME.json").write_text(json.dumps(out, indent=1) + "\n", encoding="utf-8")
