"""F02 body ISA census for card n02c-f02-fb-bias.  CPU only, read-only.

Prints:
  * opcode census (dpas.* grouped)
  * every send form: op / rd / message / lane-width class / execution byte payload
  * the fb-bias read sites (load.ugm.d32x16t) with their addresses and neighbours
  * the barrier / fence structure around the bias reads
"""
import sys, re, collections, json
from pathlib import Path
sys.path.insert(0, str(Path(__file__).parent))
import isaparse as P

ELEM_BYTES = {'d8': 1, 'd16': 2, 'd32': 4, 'd64': 8, 'd8u32': None, 'd16u32': None}


def payload_bytes(msg):
    """Bytes transferred by one block-read/store message, from its mnemonic.

    `load.ugm.d32x16t` -> d32 = 32-bit elements, x16 = 16 of them -> 64 B.
    Calibrated on three independently known fragment sizes in this family
    (NRH_B 16x16 half = 512 B -> d64x64t, NRH_CH 8x16 half contiguous = 256 B ->
    d32x64t, NRH_C 8x16 f32 row = 64 B -> d32x16t).
    """
    m = re.search(r'\.(d\d+)x(\d+)t?\b', msg)
    if not m:
        return None
    elem, n = m.group(1), int(m.group(2))
    eb = ELEM_BYTES.get(elem)
    return None if eb is None else eb * n


def width_class(i):
    w = P.lane_width(i)
    if w is None:
        return '?'
    return 'uniform' if w == 1 else 'simd%d' % w


def census(path):
    isa = P.parse(path)
    ops = collections.Counter()
    for i in isa:
        o = i['op'].split('.')[0]
        ops[o] += 1
    sends = [i for i in isa if i['op'].startswith('send')]
    forms = collections.Counter(
        (i['op'], width_class(i), P.rd_of(i), P.msg_of(i)) for i in sends)
    return isa, ops, sends, forms


def main(path):
    isa, ops, sends, forms = census(path)
    print('file:', Path(path).name)
    print('instructions', len(isa))
    print()
    print('--- opcode census (mnemonic stem)')
    for k, v in ops.most_common():
        print('%6d  %s' % (v, k))
    print()
    print('--- send forms (count, op, width, rd, message, bytes/msg, bytes total)')
    tot = collections.Counter()
    for (op, wc, rd, msg), v in forms.most_common():
        b = payload_bytes(msg)
        print('%5d  %-9s %-8s rd:%-2d %-46s %s' %
              (v, op, wc, rd, msg, ('%4d B x %3d = %7d B' % (b, v, b * v)) if b else '-'))
        tot[(op, wc)] += v
    print()
    print('--- send totals by op/width')
    for k, v in tot.most_common():
        print('%5d  %s %s' % (v, k[0], k[1]))
    print()
    print('--- fb bias reads: load.ugm.d32x16t')
    hits = [i for i in isa if 'load.ugm.d32x16t' in (i['comment'] or '')]
    print('count', len(hits))
    for i in hits:
        print('  [%04x] %-30s %s' % (i['addr'], i['flags'], i['raw'].split('*/')[-1].strip()))
    print()
    print('--- barriers / fences / gateway')
    for i in isa:
        if i['op'] in ('sync.bar', 'sync.nop', 'sync.wg', 'sync.allwr') or \
           'fence' in (i['comment'] or '') or i['op'] == 'send.gtwy':
            print('  [%04x] %-24s %s' % (i['addr'], i['op'], i['comment'] or ''))


if __name__ == '__main__':
    main(sys.argv[1])
