"""S06 block15: produce candidate-34189.txt from combo-34189 with the minimal delta.

Delta = (1) buffer "c256_b15_int8" -> the equalised pack (same byte count),
        (2) dispatch ordinal 50 (c256_b15_qx_producer) -> the per-channel SPV.
Nothing else changes; every other BIND/PUSH/POLICY/OUTPUT byte is preserved.
"""
import json
import sys
from pathlib import Path

sys.path.insert(0, str(Path(__file__).parent))
import c256b15 as L
sys.path.insert(0, str(L.W.parent / "tools"))
import planlib

BASE = L.Q / "real-int8-24/c64-family-static/combo-34189.txt"
BASE_SHA = "2b5481d5ce6b8ec8c1f62e68d2057b39d5267825d655e844a2ee2c5500f1c690"
NEW_SPV = L.W / "SOURCE/c256_b15_qx_producer_eq.spv"
NEW_PACK = L.W / "PACK/c256_b15_int8_qxeq.bin"
OUT = L.W / "candidate-34189.txt"

assert L.sha256(BASE) == BASE_SHA, "base plan identity changed"
p = planlib.Plan(BASE)
assert p.sha256() == BASE_SHA
before = p.sha256()

d0 = p.dispatch(50)
assert d0.name == "c256_b15_qx_producer", d0.name
old_spv_sha = planlib.sha256_file(d0.spv)
assert old_spv_sha == "36707a220903a6bb7fff3669fa5ae158da6ff76003c5f133e70f326c81ab007d"

log = []
log.append(p.set_buffer("c256_b15_int8", file=NEW_PACK.as_posix(), bytes_=NEW_PACK.stat().st_size))
log.append(p.set_spv(50, NEW_SPV.as_posix(), name="c256_b15_qx_producer"))
p.assert_valid()
p.write(OUT, check=False)
sha = planlib.sha256_file(OUT)

# structural check: only the intended lines differ
a = BASE.read_text(encoding="utf-8").splitlines()
b = OUT.read_text(encoding="utf-8").splitlines()
assert len(a) == len(b), (len(a), len(b))
diff = [(i, a[i], b[i]) for i in range(len(a)) if a[i] != b[i]]
print("changed lines:", len(diff))
for i, x, y in diff:
    print(f"  line {i+1}:\n    - {x}\n    + {y}")
res = dict(base=str(BASE), base_sha256=BASE_SHA, out=str(OUT), out_sha256=sha,
           lines=len(b), changed_lines=len(diff),
           changes=[dict(line=i + 1, before=x, after=y) for i, x, y in diff],
           spv=dict(ordinal=50, name="c256_b15_qx_producer",
                    old_spv=d0.spv, old_spv_sha256=old_spv_sha,
                    new_spv=NEW_SPV.as_posix(), new_spv_sha256=planlib.sha256_file(NEW_SPV)),
           pack=dict(buffer="c256_b15_int8", index=533, bytes=int(NEW_PACK.stat().st_size),
                     old=str(L.PACK_OLD), old_sha256=L.PACK_OLD_SHA,
                     new=str(NEW_PACK), new_sha256=planlib.sha256_file(NEW_PACK)))
(L.W / "evidence/10-plan.json").write_text(json.dumps(res, indent=1), encoding="utf-8")
print(json.dumps({k: v for k, v in res.items() if k != "changes"}, indent=1))
