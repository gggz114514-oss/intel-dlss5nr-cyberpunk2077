"""Emit evidence/FAMILY-GROUPS.json: shader groups with ordinals, pass labels and plan indices."""
import json, sys, re
from pathlib import Path
sys.path.insert(0, str(Path(__file__).parent))
import COMMON as K

rs = K.rows()
plan = K.BASELINE.read_text(encoding='utf-8-sig').splitlines()
passes = []
for l in plan:
    if l.startswith('PASS '):
        q = re.findall(r'"([^"]*)"', l)
        if len(q) == 2:
            name, spv = q
        else:
            toks = l.split()
            name, spv = toks[2], toks[3]
        passes.append(dict(plan_index=int(l.split()[1]), name=name, spv=spv.replace('/', '\\')))
assert len(passes) == 232, len(passes)

assert len(passes) == 232 == len(rs), (len(passes), len(rs))
order_mismatch = [(r['ordinal'], r['name'], passes[r['ordinal']]['name'], passes[r['ordinal']]['spv'], r['spv'])
                  for r in rs if passes[r['ordinal']]['name'] != r['name'] or passes[r['ordinal']]['spv'] != r['spv']]

groups = {}
for r in rs:
    key = r['spv_sha256']
    g = groups.setdefault(key, dict(spv=r['spv'], spv_sha256=key, source=r['source'],
                                    source_sha256=r['source_sha256'], family=r['family'],
                                    ordinals=[], pass_labels=[], plan_indices=[], names=[], grids=[]))
    g['ordinals'].append(r['ordinal'])
    g['pass_labels'].append(r.get('pass_label'))
    g['names'].append(r['name'])
    g['grids'].append(r.get('grid'))
    g['plan_indices'].append(passes[r['ordinal']]['plan_index'])

# sanity: every plan PASS line is claimed exactly once
claimed = [pi for g in groups.values() for pi in g['plan_indices'] if pi is not None]
dupes = {x: claimed.count(x) for x in set(claimed) if claimed.count(x) > 1}
missing = sorted(set(range(232)) - set(claimed))
out = dict(baseline_plan=str(K.BASELINE), baseline_sha=K.sha(K.BASELINE), rows=len(rs),
           unique_shaders=len(groups),
           plan_pass_count=len(passes), claimed=len(claimed),
           plan_index_unique=len(set(claimed)) == len(claimed),
           plan_index_min=min(claimed), plan_index_max=max(claimed),
           order_mismatch=order_mismatch,
           plan_pass_labels=sorted(claimed), groups=groups)
K.write_json(K.C / 'evidence' / 'FAMILY-GROUPS.json', out)
print('groups', len(groups), 'claimed', len(claimed), 'dupes', len(dupes), 'missing', len(missing))
for k, g in sorted(groups.items(), key=lambda kv: kv[1]['ordinals'][0]):
    print('%-4s %-30s n=%2d ord=%s' % (g['family'], Path(g['spv']).parent.name[:28], len(g['ordinals']),
                                       g['ordinals'][:6]))
