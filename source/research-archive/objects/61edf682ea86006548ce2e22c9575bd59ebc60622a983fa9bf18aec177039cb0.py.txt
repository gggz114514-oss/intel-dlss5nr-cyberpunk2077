# -*- coding: utf-8 -*-
"""Scan 32x32 block statistics to locate the four ROI families on the
1280x720 FP16 reference image (frame 34189 plane0, verified == visual-comparison/fp16.png)."""
import sys
from pathlib import Path

import numpy as np

sys.path.insert(0, str(Path(__file__).resolve().parent))
import pngio  # noqa: E402

Q = Path(r"@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/int8-quality-first-20261029")
ref = np.fromfile(Q / "dataset/day-street-static-formal-01/0-34189/network-output.bin",
                  dtype="<f4").reshape(2, 720, 1280, 4)[0, :, :, :3].astype(np.float64)

T = 32
H, W = 720 // T, 1280 // T
b = ref[:H * T, :W * T].reshape(H, T, W, T, 3).mean(axis=(1, 3))
L = b @ np.array([0.2126, 0.7152, 0.0722])


def show(title, mask, key, n=8):
    ys, xs = np.where(mask)
    if len(ys) == 0:
        print(f"--- {title}: none")
        return []
    ord_ = np.argsort(key[ys, xs])[::-1][:n]
    print(f"--- {title}")
    out = []
    for i in ord_:
        y, x = int(ys[i]), int(xs[i])
        r, g, bl = b[y, x]
        print(f"  block x={x*T:4d} y={y*T:3d}  mean=({r:.3f},{g:.3f},{bl:.3f}) "
              f"L={L[y,x]:.3f} R-G={r-g:+.3f} B-G={bl-g:+.3f}")
        out.append((x * T, y * T))
    return out


# 1) bright / sky  (upper region between overpass beams)
m = np.zeros((H, W), bool)
m[2:8, 13:24] = True
show("bright-sky candidates (luma)", m, L)

# 2) bright building facade (right of the FUYUTSUKI sign)
m = np.zeros((H, W), bool)
m[3:9, 17:23] = True
show("facade candidates (luma)", m, L)

# 3) NCPD sign glow (blue dominant)
m = np.zeros((H, W), bool)
m[7:10, 13:20] = True
show("NCPD sign candidates (B-R)", m, b[:, :, 2] - b[:, :, 0])

# 4) blue-lit ground / wall
m = np.zeros((H, W), bool)
m[10:21, 1:11] = True
show("blue ground left candidates (B-R)", m, b[:, :, 2] - b[:, :, 0])
m = np.zeros((H, W), bool)
m[10:17, 19:27] = True
show("blue region under overpass candidates (B-R)", m, b[:, :, 2] - b[:, :, 0])

# 5) neutral concrete wall (low chroma, mid luma, wall-like areas)
m = np.zeros((H, W), bool)
m[6:13, 26:39] = True          # right concrete beam / column
m[1:3, 0:40] = True            # ceiling beams
m[9:20, 0:3] = True            # left wall
chroma = np.abs(b[:, :, 0] - b[:, :, 1]) + np.abs(b[:, :, 2] - b[:, :, 1])
key = np.where((L > 0.08) & (L < 0.55), -chroma, -9)
show("neutral wall candidates (lowest chroma)", m, key)

for name, (x0, y0, x1, y1) in {
    "sky-gap": (560, 150, 720, 215),
    "sky-gap2": (600, 120, 700, 190),
    "white-building": (860, 150, 935, 265),
    "facade-sign-right": (790, 160, 850, 240),
    "ncpd-letters": (455, 245, 620, 298),
    "blue-sidewalk": (100, 600, 300, 672),
    "blue-wall-underpass": (640, 360, 800, 440),
    "blue-floor-right": (950, 450, 1120, 520),
    "concrete-column-right": (1185, 255, 1265, 395),
    "ceiling-beam": (540, 30, 700, 90),
    "road-neutral": (850, 630, 1000, 700),
}.items():
    c = ref[y0:y1, x0:x1].reshape(-1, 3)
    m3 = c.mean(0)
    print(f"{name:24s} box=({x0},{y0},{x1},{y1}) mean=({m3[0]:.3f},{m3[1]:.3f},{m3[2]:.3f}) "
          f"L={float(m3 @ np.array([0.2126,0.7152,0.0722])):.3f} "
          f"R-G={m3[0]-m3[1]:+.3f} B-G={m3[2]-m3[1]:+.3f} "
          f"clip_any={float((c>=1.0).any(1).mean()):.4f}")
