"""n18-f01-entry build: parent-byte reproduction + candidate compile.

Read-only on the campaign infrastructure: imports COMMON (no modification) and
uses the recorded nc_recipe for F01 (decoration migration from the parent body
f01.spv + NoContraction set of 189 ids).

Usage:
  python build.py parent     -> parent-rebuild.spv + parent-rebuild.json
  python build.py candidate  -> candidate.spv (+ raw/decorated) + candidate-build.json
"""
import sys, json, shutil
from pathlib import Path

CARD = Path(__file__).resolve().parents[1]
K = None
sys.path.insert(0, str(CARD.parents[1]))
import COMMON as K
import spv_tools_local as ST

ORD = 0
RECIPES = K.read_json(CARD.parents[1] / 'evidence' / 'RECIPES.json')
SHA = K.row_by_ordinal(ORD)['spv_sha256']
RECIPE = RECIPES[SHA]
PARENT_SPV = Path(RECIPE['nc_recipe']['parent'])          # c32-no-gain-diagnosis/combined-existing/f01.spv
PARENT_SRC = Path(RECIPE['source'])
INCLUDES = [Path(p) for p in RECIPE['compile']['include_dirs']]
PINNED = Path(RECIPE['compile']['pinned_include_dir'])


def includes_for(comp):
    """Recipe include dirs first (pinned header), then COMMON's discovery."""
    dirs = [comp.parent, PINNED]
    for d in INCLUDES:
        if d not in dirs:
            dirs.append(d)
    for d in K.include_dirs_for(ORD, comp):
        if Path(d) not in dirs:
            dirs.append(Path(d))
    return dirs


def decorate(raw, out):
    pm = ST.Module(PARENT_SPV)
    cm = ST.Module(raw)
    keep, info = ST.transfer_decoration_pattern_aligned(pm, cm, set())
    ST.set_no_contraction(cm, keep, out)
    return dict(
        parent_nc=len(ST.no_contraction_ids(pm)),
        nc_keep=len(keep),
        functions_mapped=info.get('functions_mapped'),
        child_functions_without_parent_counterpart=len(
            info.get('child_functions_without_parent_counterpart') or []),
        parent_decorations_total=info.get('parent_decorations_total'),
        child_own_decorations_outside_pattern=info.get('child_own_decorations_outside_pattern'),
    )


def build(kind):
    if kind == 'parent':
        src, raw, out = PARENT_SRC, CARD / 'parent-raw.spv', CARD / 'parent-rebuild.spv'
    else:
        src, raw, out = CARD / 'candidate.comp', CARD / 'candidate-raw.spv', CARD / 'candidate.spv'
    incs = includes_for(src)
    K.compile_glsl(src, raw, incs)
    dec = decorate(raw, out)
    res = dict(
        kind=kind, source=K.ident(src), includes=[str(p) for p in incs],
        header_pins={p.name: K.sha(p) for p in incs for p in sorted(p.glob('*.glsl'))} if False else None,
        raw=K.ident(raw), decorated=K.ident(out), decor=dec,
        expected=(K.row_by_ordinal(ORD)['spv_sha256'] if kind == 'parent' else None),
    )
    res['raw_match'] = res['raw']['sha256'] == res['expected']
    res['decorated_match'] = res['decorated']['sha256'] == res['expected']
    json_path = CARD / ('parent-rebuild.json' if kind == 'parent' else 'candidate-build.json')
    K.write_json(json_path, res)
    print(json.dumps({k: res[k] for k in ('kind', 'raw', 'decorated', 'decor', 'raw_match', 'decorated_match')}, indent=1))
    return res


if __name__ == '__main__':
    build(sys.argv[1])
