"""S06 post-GPU step 4: qX dequant audit in the real domain + document emit."""
import json
import sys
from pathlib import Path

import numpy as np

sys.path.insert(0, str(Path(__file__).parent))
import c256b15 as L

W = L.W
CAP = W.parent / "gpu/runs/input-capture/xq-c256-b15.bin"
idx, oob = L.window_to_canonical()
valid = idx >= 0
Yteach = np.fromfile(L.D / "dump-fp16/dump_swin_15_Y.bin", dtype="<f2").reshape(L.N_TOK, 256).astype(np.float32)
Xw = np.zeros((L.N_TOK, 256), dtype=np.float32)
Xw[valid] = L.load_x_canonical(CAP)[idx[valid]]
raw_old = np.fromfile(L.PACK_OLD, dtype=np.uint8)
raw_A = np.fromfile(W / "PACK/c256_b15_int8_qxeq.bin", dtype=np.uint8)
diag = np.load(W / "evidence/06-diag.npy")
ev = {}

for tag, raw, tab in (("old", raw_old, False), ("A_mse", raw_A, True)):
    codes, d = L.decode_pack(raw)
    inv = d["tab"].astype(np.float64) if tab else np.full(256, d["invX"], dtype=np.float64)
    qx = np.clip(np.rint(Xw.astype(np.float64) * inv[None, :]), -127.0, 127.0)
    # dequantise: qx * step, step = 1/inv  (for the equalised arm that is sX*D_c)
    step = (1.0 / inv).astype(np.float64)
    xdq = qx * step[None, :]
    dq = xdq[valid].astype(np.float64) - Xw[valid].astype(np.float64)
    ev[tag] = dict(
        qx_dequant_rel_l2_vs_real_input=float(np.sqrt(np.mean(dq ** 2) / np.mean(Xw[valid].astype(np.float64) ** 2))),
        qx_dequant_max_abs=float(np.abs(dq).max()),
        qx_sat=int((np.abs(qx) >= 127).sum()),
        qx_rms_code=float(np.sqrt(np.mean(qx.astype(np.float64) ** 2))),
        per_channel_step=[float(x) for x in step[:6]],
        step_ratio_vs_uniform=float(np.max(step) / np.min(step)),
    )
ev["note"] = ("step = 1/inv is sX for the shipped pack and sX*D_c for the equalised pack, i.e. exactly the "
              "dequantisation the expand epilogue performs through ew")
(W / "evidence/19-qx-audit.json").write_text(json.dumps(ev, indent=1), encoding="utf-8")
print(json.dumps(ev, indent=1))
