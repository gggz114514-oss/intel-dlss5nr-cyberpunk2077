"""Pick the F05 C128 / C64 native asm and print DPAS + register facts.

CPU only: reads the existing n04b (C128) and n04a (C64) native captures under queue/.
Writes nothing outside this card dir.
"""
import re, sys, json
from pathlib import Path

Q = Path('@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/'
         'fullnet-continuous-20261027/queue')

# pipeline index in the capture file name: 9 = F05 C128 (START-HERE: "C128 F05是pipeline9")
C128 = Q / 'n04b-f05-c128-feed/native/captures/native-native-base/ISA/CS_asmf98d5ab4e239dd54_simd32_000000000000000a_9_orig.asm'
C64  = Q / 'n04a-f05-c64-feed/native/captures/native-native-base/ISA'

def parse(path):
    out = []
    rx = re.compile(r'/\* \[([0-9A-Fa-f]+)\]\s*\*/\s*(\([^)]*\))?\s*([a-zA-Z0-9_.]+)\s*(.*)$')
    for line in Path(path).read_text(encoding='utf-8', errors='replace').splitlines():
        m = rx.match(line.strip())
        if not m:
            continue
        pc = int(m.group(1), 16)
        op = m.group(3)
        rest = m.group(4)
        # strip trailing annotation comments
        body = rest.split('//')[0]
        out.append(dict(pc=pc, op=op, body=body, line=line.strip(), pred=(m.group(2) or '')))
    return out

REG = re.compile(r'\b(r|acc)(\d+)(?:\.(\d+))?')

def regs_of(body):
    out = []
    # drop the send descriptor / immediate hex tokens
    b = re.sub(r'0x[0-9A-Fa-f]+', ' ', body)
    for m in REG.finditer(b):
        kind, num = m.group(1), int(m.group(2))
        if kind == 'acc':
            out.append(('acc', num))
        else:
            out.append(('r', num))
    return out

def main():
    for tag, path in (('C128', C128), ('C64', C64)):
        p = Path(path)
        if p.is_dir():
            files = sorted(p.glob('*.asm'))
            # choose by pipeline index 5 (C64 F05 = pipeline 5)
            cand = [f for f in files if re.search(r'_5_orig\.asm$', f.name)]
            if not cand:
                print(tag, 'no pipeline-5 file; files=', [f.name for f in files][:5])
                continue
            p = cand[0]
        ins = parse(p)
        dpas = [i for i in ins if i['op'].startswith('dpas')]
        print('== %s %s' % (tag, p.name))
        print('instructions=%d dpas=%d' % (len(ins), len(dpas)))
        # destination register of each dpas, in program order
        seen = []
        for d in dpas:
            rs = regs_of(d['body'])
            seen.append((d['pc'], [x for x in rs if x[0] == 'r'][:1], d['body'].strip()[:80]))
        for pc, dst, b in seen:
            print('  %04X %s' % (pc, b))
        maxr = 0
        for i in ins:
            for k, n in regs_of(i['body']):
                if k == 'r' and n > maxr:
                    maxr = n
        print('  max register index used = r%d' % maxr)

if __name__ == '__main__':
    main()
