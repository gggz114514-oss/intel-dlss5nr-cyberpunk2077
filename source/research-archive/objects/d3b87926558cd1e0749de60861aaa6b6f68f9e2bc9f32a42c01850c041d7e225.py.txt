"""Re-measure n32 with the corrected index (d_arith), as n33 pointed out.

d_first is useless when the first consumer is a same-expression cast: the distance is 1 for both
arms. d_arith skips cast/move consumers and measures the distance to the first real arithmetic
instruction that reads the result. If d_arith grew for n32, then source reordering DID move the
arithmetic schedule and L26 must be softened.
"""
import re, sys
from collections import Counter
from pathlib import Path
sys.path.insert(0, str(Path(__file__).parent))
import COMMON as K
import ISA

REG = re.compile(r'\br(\d+)(?:\.(\d+))?')
ARITH = ('mul', 'mad', 'add', 'add3', 'fma', 'macl', 'dpas')


def reg(o):
    m = REG.match(o or '')
    return (int(m.group(1)), m.group(2)) if m else None


def analyse(path):
    isa = ISA.parse(path)
    ops = [[o for o in i['ops'] if o] for i in isa]
    kinds = [i['op'] for i in isa]
    first = []
    arith = []
    for i, k in enumerate(kinds):
        if k != 'math.rsqt':
            continue
        regs = [reg(o) for o in ops[i] if reg(o)]
        if len(regs) < 2:
            continue
        dst = regs[0]
        d1 = da = None
        for j in range(i + 1, min(i + 200, len(kinds))):
            hit = any(reg(o) == dst for o in ops[j])
            if not hit:
                continue
            if d1 is None:
                d1 = j - i
            if kinds[j].split('.')[0] in ARITH or kinds[j].startswith('mul') or kinds[j].startswith('mad'):
                da = j - i
                break
        first.append(d1)
        arith.append(da)
    return first, arith


def pid(name):
    m = re.search(r'_(\d+)_orig\.asm$', name)
    return int(m.group(1)) if m else -1


base = K.C / 'queue' / 'n32-run' / 'run' / 'captures-base'
cand = K.C / 'queue' / 'n32-run' / 'run' / 'captures-cand'
fb = {pid(p.name): p for p in base.rglob('*.asm')}
fc = {pid(p.name): p for p in cand.rglob('*.asm')}
for i in sorted(set(fb) & set(fc)):
    if fb[i].read_bytes() == fc[i].read_bytes():
        continue
    print('pipeline', i)
    for tag, p in (('base', fb[i]), ('cand', fc[i])):
        d1, da = analyse(p)
        print('  %-4s d_first %s' % (tag, dict(sorted(Counter(x for x in d1 if x is not None).items()))))
        print('       d_arith %s' % dict(sorted(Counter(x for x in da if x is not None).items())))
