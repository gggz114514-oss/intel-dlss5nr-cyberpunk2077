"""Register write/read timeline for the QKV projection region (card n04a-f05-c64-feed).

For every register that appears as a B-weight return bank or as an address
payload of a B-weight send, print the ordered events in the QKV projection
region: which instruction writes it, which SEND/DPAS reads it, and the SWSB
tokens attached.  Read-only.
"""
import re
import sys
from pathlib import Path

sys.path.insert(0, str(Path(__file__).parent))
from isa_scan import parse, ASM

ins = parse(ASM)
LO, HI = 0x0C00, 0x2200
regions = [('row_phase0', 0x0C00, 0x2200), ('row_phase1', 0x4A00, 0x6200)]

WEIGHT = 'load.ugm.d32x8.a32.bss[a0.2]'


def fields(s):
    return [f for f in re.split(r'\s{2,}', s) if f]


def first_reg(field):
    m = re.match(r'^r(\d+)', field.strip())
    return (int(m.group(1)) if m else None)


def events(lo, hi, regs):
    ev = []
    for i in ins:
        if not (lo <= i['off'] < hi):
            continue
        f = fields(i['full'])
        toks = re.findall(r'\$(\d+)', i['tail'])
        mn = i['mnemonic']
        if WEIGHT in i['tail']:
            dst = first_reg(f[1]); pay = first_reg(f[2])
            if dst in regs:
                ev.append((i['off'], 'SEND->B r%d' % dst, 'rd:8', toks))
            if pay in regs:
                ev.append((i['off'], 'addr(payload) r%d' % pay, '', toks))
            continue
        if mn.startswith('dpas'):
            b = first_reg(f[3]); acc = first_reg(f[1])
            if b in regs:
                ev.append((i['off'], 'DPAS reads B r%d' % b, 'acc r%d' % acc, toks))
            continue
        # ordinary instruction: a write if its first operand is rNN.d<
        dst = first_reg(f[1]) if len(f) > 1 else None
        if dst in regs and mn not in ('send.ugm', 'send.slm', 'sync.nop'):
            width = (i['execmask'] or 0)
            ev.append((i['off'], 'write r%d  %s' % (dst, i['full'][:64]), 'w%d' % width, toks))
    return sorted(ev)


REG = {7, 8, 18, 22}          # address payload AND B bank (the coupling set)
if __name__ == '__main__':
    which = sys.argv[1] if len(sys.argv) > 1 else 'row_phase0'
    lo, hi = dict((n, (a, b)) for n, a, b in regions)[which]
    for reg in sorted(REG):
        print('=== r%d (%s) ===' % (reg, which))
        for e in events(lo, hi, {reg}):
            print('  %-7s %-46s %-10s %s' % (hex(e[0]), e[1], e[2], e[3]))
