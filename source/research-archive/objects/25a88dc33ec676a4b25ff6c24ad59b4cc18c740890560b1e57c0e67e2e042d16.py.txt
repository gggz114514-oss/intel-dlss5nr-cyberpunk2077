from pathlib import Path
import json,re,hashlib
N=Path(__file__).resolve().parent;R=N.parent
out={}
for arm,P in [('dualweight',R/'f20-peer8-dualweight'),('fouracc',N)]:
 p=next((P/'captures'/('native-'+P.name)/'ISA').glob('*38_orig.asm'));rows=[]
 for l in p.read_text().splitlines():
  m=re.match(r'/\* \[([0-9A-F]+)\]\s*\*/\s*(.*)',l)
  if m:rows.append((m[1],m[2]))
 dp=[i for i,(_,l) in enumerate(rows) if 'dpas.' in l];assert len(dp)==80
 expand=[]
 for i in dp[:16]:
  pc,l=rows[i];expand.append(dict(pc=pc,text=l))
 pairs=[]
 for k in range(16,48,2):
  ops=[]
  for i in dp[k:k+2]:
   pc,l=rows[i];regs=re.findall(r'r(\d+)(?:\.\d+)?:',l);assert len(regs)==4;b=int(regs[2])
   for t in range(i-1,-1,-1):
    m=re.search(r'send\.ugm.*?\)\s+r(\d+)\s',rows[t][1])
    if m and int(m[1])==b and 'rd:8;' in rows[t][1]:break
   else:raise AssertionError('Missing weight send')
   ops.append(dict(dpas=pc,index=i,send=rows[t][0],send_index=t,B=b,A=int(regs[3])))
  a,b=ops;pairs.append(dict(ops=ops,both_early=b['send_index']<a['index'],overlap=max(a['B'],b['B'])<=min(a['B']+7,b['B']+7)))
 out[arm]=dict(path=str(p),sha256=hashlib.sha256(p.read_bytes()).hexdigest(),instruction_count=len(rows),scratch=[dict(pc=pc,text=l) for pc,l in rows if re.search(r'\.ss\b',l)],barriers=sum('sync.bar' in l for _,l in rows),expansion=expand,contract=pairs,contract_early=sum(p['both_early'] for p in pairs),contract_overlap=sum(p['overlap'] for p in pairs))

for base in [0,8]:
 first=v['expansion'][base:base+4];second=v['expansion'][base+4:base+8]
 dest=[]

  assert 'null:f' in x['text'],x
  dest.append(int(re.findall(r'r(\d+):f',x['text'])[0]))
 assert len(set(dest))==4 and all(abs(a-b)>=8 for i,a in enumerate(dest) for b in dest[i+1:]),dest
 reads=[int(re.findall(r'r(\d+):f',x['text'])[1]) for x in second]
 assert set(reads)==set(dest),(dest,reads)
 banks.append(dict(first_K_C_banks=dest,second_K_C_sources=reads))
v['four_independent_accumulators']=banks
(N/'NATIVE-REVIEW.json').write_text(json.dumps(out,indent=2))
print({k:{x:(len(v[x]) if x=='scratch' else v[x]) for x in ['instruction_count','scratch','barriers','contract_early','contract_overlap']} for k,v in out.items()});print(banks)
