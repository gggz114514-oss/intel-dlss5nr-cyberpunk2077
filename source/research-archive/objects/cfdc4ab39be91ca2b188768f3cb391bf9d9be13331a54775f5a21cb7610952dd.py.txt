rows = {}
for sha, g in fg["groups"].items():
    for o in g["ordinals"]:
        rows[o] = dict(g, ordinal=o)

def row(o):
    g = rows[o]
    return {"ordinals": g["ordinals"], "name": g["names"][0], "family": g["family"],
            "spv": g["spv"], "spv_sha256": g["spv_sha256"]}

def msgs(pid, prefix):
    out = {}
    for k, v in isa[str(pid)]["counts"].items():
        if k.startswith(prefix):
            parts = k.split(".")
            shape = ".".join(parts[2:]) if len(parts) > 2 else k
            out[shape] = out.get(shape, 0) + v
    return out
