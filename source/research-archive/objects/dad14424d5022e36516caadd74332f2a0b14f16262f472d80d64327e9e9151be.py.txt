import json, os, re, collections

ROOT = r"@NR_WORKSPACE@\work\codex-astra-fp16-int8-audit-20261024\fullnet-continuous-20261027"
ISA_DIRS = [
    os.path.join(ROOT, "queue", "n09c-f09-b-share", "native", "captures", "native-native-base", "ISA"),
    r"@NR_WORKSPACE@\work\codex-astra-fp16-int8-audit-20261024\fullnet-mechanism-transfer-20261026\f10-weight-return-separation-20261027\captures\native-f20-pixel-owner\ISA",
]
OUT = os.path.join(ROOT, "cards", "n16-crossdispatch", "tools")

n = json.load(open(os.path.join(ROOT, "evidence", "NATIVE-INDEX.json"), encoding="utf-8"))
pm = n["pipeline_map"]["vector"]
pipes = n["baseline"]["pipelines"]

# map kernel hash -> isa file (first dir wins)
byhash = {}
for d in ISA_DIRS:
    if not os.path.isdir(d):
        continue
    for f in os.listdir(d):
        m = re.match(r"CS_asm([0-9a-f]{16})_", f)
        if m and m.group(1) not in byhash:
            byhash[m.group(1)] = os.path.join(d, f)

msg_re = re.compile(r"//\s*[^;]*;\s*([A-Za-z0-9_.\[\]+x]+)")
stats = {}
for pid, p in pipes.items():
    kh = p["kernel_hash"]
    path = byhash.get(kh)
    if not path:
        print("MISSING ISA", pid, kh)
        continue
    counts = collections.Counter()
    instr = 0
    dpas = 0
    for line in open(path, encoding="utf-8", errors="replace"):
        s = line.strip()
        if not s or s.startswith("//") or s.startswith("L"):
            continue
        instr += 1
        if " dpas" in s:
            dpas += 1
        if "send." in s:
            m = msg_re.search(s)
            if not m:
                counts["send.unknown"] += 1
                continue
            name = m.group(1)
            for pref in ("load.slm", "store.slm", "load.ugm", "store.ugm", "load.ugm.atomic",
                         "load.slm.atomic", "store.ugm.atomic", "atomic", "barrier", "fence", "load.ugm"):
                if name.startswith(pref):
                    counts[name] += 1
                    break
            else:
                counts["send.other"] += 1
    stats[pid] = {"kernel_hash": kh, "isa": path, "instructions": instr, "dpas": dpas,
                  "messages": sum(v for k, v in counts.items() if k.startswith(("load.", "store.", "atomic"))),
                  "counts": dict(counts), "passes": p["passes"], "first_pass": p["first_pass"],
                  "families": p["families"], "shared_bytes": p["shared_bytes"]}

json.dump(stats, open(os.path.join(OUT, "isa-stats.json"), "w", encoding="utf-8"), indent=1)
for pid in sorted(stats, key=int):
    s = stats[pid]
    c = s["counts"]
    nl = sum(v for k, v in c.items() if k.startswith("load.ugm"))
    ns = sum(v for k, v in c.items() if k.startswith("store.ugm"))
    sl = sum(v for k, v in c.items() if k.startswith("load.slm"))
    ss = sum(v for k, v in c.items() if k.startswith("store.slm"))
    print("%2s %-4s %-24s instr=%5d dpas=%4d ugmL=%4d ugmS=%3d slmL=%3d slmS=%3d send=%4d" % (
        pid, ",".join(s["families"]), s["first_pass"], s["instructions"], s["dpas"], nl, ns, sl, ss, s["messages"]))
