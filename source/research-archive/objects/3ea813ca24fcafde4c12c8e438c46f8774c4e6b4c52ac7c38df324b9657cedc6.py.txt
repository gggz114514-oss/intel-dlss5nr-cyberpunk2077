"""Structural fingerprint of every ISA in one capture (read only, CPU only).

Purpose: identify the F16 projection kernel by *geometry derived from the current
source* (per k0 step: 1x (16|M0) d32x8 weight load, 2x 8-row A tiles = 16x
(1|M0) d32x8t loads, 2 dpas.8x8, 2 conditional branches), instead of trusting a
pipeline index printed inside the capture file name (which belongs to an older
plan and is not the current identity, contract 8.7).
"""
import sys, json, collections, glob, os
sys.path.insert(0, r'@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/fullnet-continuous-20261027/cards/n11b-vit-projection-hoist/_tools')
import isa_scan as S

ISA_DIR = r'@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/fullnet-mechanism-transfer-20261026/f08-register-chain-20261027/captures/native-f20-pixel-owner/ISA'


def fp(path):
    ins = S.parse(path)
    sends = [i for i in ins if i['op'].startswith('send')]
    def cnt(sub, width=None):
        return len([i for i in sends if sub in i['comment'] and (width is None or i['width'] == width)])
    dpas = [i['off'] for i in ins if i['op'].startswith('dpas')]
    mov32 = len([i for i in ins if i['op'] == 'mov' and i['width'] == 32])
    jmpi = [i for i in ins if i['op'].startswith('jmpi')]
    cond = len([i for i in jmpi if i['pred'] and '&' in i['pred']])
    return dict(file=os.path.basename(path), total=len(ins), dpas=len(dpas), mov32=mov32,
                mov32_per_dpas=(round(mov32 / len(dpas), 3) if dpas else None),
                wide_B=cnt('load.ugm.d32x8.', 16), narrow_A=cnt('load.ugm.d32x8t', 1),
                stores=cnt('store.ugm'), other_ld=cnt('load.ugm.d16'),
                jmpi_cond=cond, sync=len([i for i in ins if i['op'].startswith('sync')]))


if __name__ == '__main__':
    rows = [fp(p) for p in sorted(glob.glob(ISA_DIR + '/*.asm'))]
    out = {"capture_dir": ISA_DIR, "kernels": rows}
    # kernels whose per-step shape matches the current u03-f18-v2 projection source:
    # dpas = 128 (2/step x 64), wide_B = 64 (1/step), narrow_A = 1024 (16/step)
    proj = [r for r in rows if r['dpas'] == 128 and r['wide_B'] == 64 and r['narrow_A'] == 1024]
    out['projection_geometry_candidates'] = proj
    op = os.path.join(r'@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/fullnet-continuous-20261027/cards/n11b-vit-projection-hoist', '_build-fingerprints.json')
    open(op, 'w', encoding='utf-8').write(json.dumps(out, indent=1))
    print(json.dumps(out['projection_geometry_candidates'], indent=1))
    print('%d kernels scanned' % len(rows))
