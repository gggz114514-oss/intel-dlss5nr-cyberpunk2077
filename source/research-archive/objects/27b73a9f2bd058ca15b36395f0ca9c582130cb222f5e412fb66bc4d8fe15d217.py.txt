import sys, json, collections
from pathlib import Path
C = Path("@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/fullnet-continuous-20261027")
sys.path.insert(0, str(C))
import COMMON as K

CARD = C / "cards" / "n05a-f05-c256-mreuse"
PLAN = CARD / "plan.txt"

bl = K.BASELINE.read_text(encoding="utf-8-sig").splitlines()
cd = PLAN.read_text(encoding="utf-8").splitlines()
print("line counts:", len(bl), len(cd))
assert len(bl) == len(cd)
diff = [(i, a, b) for i, (a, b) in enumerate(zip(bl, cd)) if a != b]
print("differing lines:", len(diff))
for i, a, b in diff[:3]:
    print(i, "\nB:", a, "\nC:", b)
# every differing line must be a PASS line whose only change is the spv path
import re
bad = 0
for i, a, b in diff:
    if not a.startswith("PASS ") or not b.startswith("PASS "):
        bad += 1
        print("NON-PASS DIFF", i, a, b)
        continue
    pa = a.split('"')
    pb = b.split('"')
    if len(pa) != 5 or len(pb) != 5 or pa[0] != pb[0] or pa[1] != pb[1] or pa[2] != pb[2] or pa[4] != pb[4]:
        bad += 1
        print("UNEXPECTED PASS DIFF", i, a, b)
print("unexpected diffs:", bad)

GRID = {}
rows = K.rows()
for r in rows:
    if r["family"] == "F05" and r["spv_sha256"] == "9d61c4e3825265c42e7815372f6411a5e4777ab6267170a79a40dbfe862f66d8":
        GRID[r["ordinal"]] = r

tot_wg = 0
print()
print("ord | pass | grid(x,y,z) | WGs | SGs | threads | label")
for o in sorted(GRID):
    r = GRID[o]
    gx, gy, gz = [int(x) for x in r["grid"]]
    wg = gx * gy * gz
    tot_wg += wg
    print("%3d | %5s | %2d,%d,%d | %4d | %5d | %7d | %s" % (
        o, r["pass_label"], gx, gy, gz, wg, wg * 8, wg * 256, r["name"]))
print("total WGs over the 16 dispatches:", tot_wg)
print("total SG instances:", tot_wg * 8)
print("total threads:", tot_wg * 256)

# per-SG dynamic counts
per_sg = dict(parent=dict(A=64, B=192, DPAS=192, acc_qk=4, acc_v=2),
              cand=dict(A=64, B=96, DPAS=192, acc_qk=8, acc_v=4))
print()
print(json.dumps(per_sg, indent=2))
for k, v in per_sg.items():
    print(k, "B-tile logical bytes per SG:", v["B"] * 512,
          "A-tile logical bytes per SG:", v["A"] * 256)
print("frame B-tile loads removed:", (192 - 96) * tot_wg * 8)
print("frame B logical bytes removed:", (192 - 96) * 512 * tot_wg * 8)
