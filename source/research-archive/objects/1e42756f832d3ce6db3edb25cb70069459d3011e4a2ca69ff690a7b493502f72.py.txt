"""Record the token-halving analysis and the cheaper reverse probe it suggests."""
import json, sys

import COMMON as K

p = K.C / 'evidence' / 'LESSONS.json'
d = K.read_json(p)
ids = [x['id'] for x in d['append_after_L7']]
if 'L30' not in ids:
    d['append_after_L7'].append({
        'id': 'L30',
        'claim': 'Halving tokens per workgroup in the 32 KB kernel would raise occupancy 1.8x but costs +34.8% messages; a cheaper way to learn the occupancy slope is to make residency worse on purpose',
        'evidence': [
            '17 PASSes declare 32768 B of shared memory (F06 15 + F18 2) with 2189 workgroups per frame; their measured occupancy is 41.7% against 86.1% for the 8 KB group in the same frame',
            'the kernel message mix per thread is load.slm.d32x8t 1024 (staging reads, proportional to tokens) plus per-workgroup traffic: load.ugm.d32x8 256 (weights), load.ugm.d32x8t 128, store.ugm.d32x8t 128, store.ugm.d32x64t 16 and small items',
            'scaling that mix to half the tokens per workgroup (twice the workgroups) leaves staging totals unchanged but doubles everything per-workgroup: total messages 3.50e6 -> 4.71e6 per frame, +34.8%',
            'the uncertainty is genuine: this same kernel was shown insensitive to message count when n06d cut its messages 54% and the frame did not move (-0.0025 ms), while its occupancy gap is the largest in the frame'
        ],
        'consequence': 'Before rewriting the kernel shape, measure the slope the cheap way: pad the shared array so residency halves with zero change to the maths, and see how much the frame slows. That gives the occupancy elasticity from one GPU slot without touching the index arithmetic.'
    })
    K.write_json(p, d)
    print('L30 added')
print('analysis recorded')
