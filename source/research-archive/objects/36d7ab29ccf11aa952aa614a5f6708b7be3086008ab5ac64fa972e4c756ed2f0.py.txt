import json, os, re, collections

D = r"@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/int8-quality-first-20261029/real-int8-24"
o = json.load(open(os.path.join(D, "DUMP-ALIGNMENT-AUDIT.json"), encoding="utf-8"))
print("n entries", len(o))
print("keys", sorted({k for e in o for k in e}))
for e in o:
    e.setdefault("n", None)
    e.setdefault("fp_std", None)
    e.setdefault("i8_std", None)
    e.setdefault("rel", None)

blocks = [e for e in o if re.fullmatch(r"dump_block\d+_out\.bin", e["name"])]
blocks.sort(key=lambda e: int(re.findall(r"\d+", e["name"])[0]))
print("\n%-22s %9s %9s %9s %7s %s" % ("name", "n", "fp_std", "i8_std", "rel", "status"))
for e in blocks:
    f = lambda x, p=6: ("%.*f" % (p, x)) if isinstance(x, float) else str(x)
    print("%-22s %9s %9s %9s %7s %s" % (e["name"].replace("dump_", "").replace("_out.bin", ""), e["n"], f(e["fp_std"]), f(e["i8_std"]), f(e["rel"], 4), e.get("status")))

print("\n--- dead / size_mismatch entries (first 60) ---")
for e in o:
    if e.get("status") != "live":
        print("%-40s status=%-14s n=%s rel=%s" % (e["name"], e.get("status"), e["n"], e.get("rel")))
