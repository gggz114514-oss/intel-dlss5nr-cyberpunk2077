"""R01 F1 - reproduce GATE-f1-scalex2 on the captured dump, CPU only.

Reads the frozen capture dump (read-only) and the frozen pack files, replays
ref_c32.Ref.stage_fragment, and reports the mismatch structure so the first
difference can be attributed.
"""
import hashlib
import json
import struct
import sys
from pathlib import Path

ROOT = Path("@NR_WORKSPACE@")
DUMP = ROOT / "run/int8-coverage-v2-20261020/C/c32-i8-diag-f1-scalex2-001/diag_f1_debug.bin"
WEIGHTS = ROOT / "candidates/int8-coverage-v2-20261020/C/pack/c32_f1_weights.i8.bin"
SCALES = ROOT / "candidates/int8-coverage-v2-20261020/C/c32-scale-from-x-v2/pack/c32_f1_scales.f32.bin"
FP16 = ROOT / "candidates/amd-map-20261015/s1-integrated/candidate/packs/ioport_0_ioport_pack.fp16.bin"
RS_BASE = 17280
RS_OFFSET = 8

sys.path.insert(0, str(ROOT / "work/int8-coverage-v2-20261020/C/tools"))
from ref_c32 import Ref, half_array, code  # noqa: E402

dbg = half_array(DUMP)
ref = Ref(WEIGHTS, SCALES, FP16, RS_BASE, RS_OFFSET)

out = {
    "dump": str(DUMP),
    "dump_bytes": DUMP.stat().st_size,
    "dump_sha256": hashlib.sha256(DUMP.read_bytes()).hexdigest(),
    "dump_halves": len(dbg),
    "weights_sha256": hashlib.sha256(WEIGHTS.read_bytes()).hexdigest(),
    "scales_sha256": hashlib.sha256(SCALES.read_bytes()).hexdigest(),
    "fp16_sha256": hashlib.sha256(FP16.read_bytes()).hexdigest(),
    "scales": {"sX": ref.sx, "invX": ref.invx, "sE": ref.se, "invE": ref.inve},
    "rs_32": ref.rs,
    "rs_distinct": len(set(ref.rs)),
}

e_bad = y_bad = 0
first_e = first_y = None
max_e = max_y = 0.0
per_token_e = [0] * 64
per_token_y = [0] * 64
per_feature_e = [0] * 128
e_mism = []
y_mism = []
rows = []
for t in range(64):
    x = [dbg[22528 + t * 32 + c] for c in range(32)]
    e_ref, y_ref = ref.stage_fragment(x, t)
    bad_e = 0
    for f in range(128):
        g = dbg[t * 128 + f]
        d = abs(float(g) - float(e_ref[f]))
        max_e = max(max_e, d)
        if g != e_ref[f]:
            e_bad += 1
            bad_e += 1
            per_feature_e[f] += 1
            if first_e is None:
                first_e = {"token": t, "feature": f, "gpu": g, "ref": e_ref[f], "absdiff": d}
            e_mism.append({"t": t, "f": f, "gpu": g, "ref": e_ref[f]})
    per_token_e[t] = bad_e
    bad_y = 0
    for o in range(32):
        g = dbg[8192 + t * 32 + o]
        d = abs(float(g) - float(y_ref[o]))
        max_y = max(max_y, d)
        if g != y_ref[o]:
            y_bad += 1
            bad_y += 1
            if first_y is None:
                first_y = {"token": t, "channel": o, "gpu": g, "ref": y_ref[o], "absdiff": d}
            y_mism.append({"t": t, "o": o, "gpu": g, "ref": y_ref[o]})
    per_token_y[t] = bad_y
    rows.append({"t": t, "e_bad": bad_e, "y_bad": bad_y, "x_amax": max(abs(float(v)) for v in x)})

out.update({
    "e_neq": e_bad, "e_n": 8192, "e_max_abs": max_e, "e_first_error": first_e,
    "y_neq": y_bad, "y_n": 2048, "y_max_abs": max_y, "y_first_error": first_y,
    "per_token": rows,
    "per_feature_e_bad_nonzero": {str(i): per_feature_e[i] for i in range(128) if per_feature_e[i]},
    "tokens_with_e_bad": [t for t in range(64) if per_token_e[t]],
    "tokens_with_y_bad": [t for t in range(64) if per_token_y[t]],
})
print(json.dumps({k: out[k] for k in
                  ("e_neq", "y_neq", "e_max_abs", "y_max_abs", "e_first_error", "y_first_error",
                   "tokens_with_e_bad", "tokens_with_y_bad")}, indent=1))
print("rs distinct:", out["rs_distinct"])
print("scales:", out["scales"])

Path(__file__).with_name("f1_probe1.out.json").write_text(
    json.dumps({**out, "e_mism_sample": e_mism[:64], "y_mism_sample": y_mism[:64]}, indent=1),
    encoding="utf-8")
