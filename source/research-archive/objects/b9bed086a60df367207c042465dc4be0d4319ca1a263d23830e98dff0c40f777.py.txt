from pathlib import Path

p = Path(__file__).with_name('run_external.py')
t = p.read_text(encoding='utf-8')
old = "    jobs = [K.native_job('native-%s' % a, p, s, run / ('native-%s' % a), a) for a, p, s in arms]"
new = ("    jobs = [K.native_job('native-%s' % a, p, s, run / ('native-%s' % a),\n"
       "                          run / ('captures-%s' % a), run / ('native-%s.log' % a), a)\n"
       "            for a, p, s in arms]")
assert t.count(old) == 1, t.count(old)
p.write_text(t.replace(old, new, 1), encoding='utf-8')
print('patched native_job call')
