"""Task 1c: isolate the residual branch.  blockN_out = blockN_in + branch.
   block62 input is the TRUE FP16 blend dump, so e62 is a clean single-block
   branch error with identical input.
"""
import json
import numpy as np
from pathlib import Path

B = Path("@NR_WORKSPACE@")
W = B / "work/dsh-fullnet-static-20261029"
CHAIN = W / "C00-color-baseline/runs/diag-chain62-66"
TEACH = (B / "work/codex-astra-fp16-int8-audit-20261024/int8-quality-first-20261029"
         / "real-int8-24/dump-fp16")


def load(p):
    return np.fromfile(p, dtype="<f2").astype(np.float32)


def rms(a):
    return float(np.sqrt(np.mean(a.astype(np.float64) ** 2)))


def relL2(a, ref):
    return float(np.linalg.norm((a - ref).astype(np.float64)) / max(np.linalg.norm(ref.astype(np.float64)), 1e-30))


def slope(a, ref):
    a = a.astype(np.float64); ref = ref.astype(np.float64)
    return float(np.dot(a, ref) / max(np.dot(ref, ref), 1e-30))


def main():
    out = {}
    # --- single block 62: identical true input ---
    i62 = load(TEACH / "dump_ioport_block62_blend.bin")
    t62 = load(TEACH / "dump_block62_out.bin")
    c62 = load(CHAIN / "chain-block62_out.bin")
    d62 = t62 - i62
    e62 = c62 - i62
    out["b62_branch"] = {
        "rms_in": rms(i62),
        "rms_teach_branch": rms(d62), "rms_gpu_branch": rms(e62),
        "branch_ratio": rms(e62) / rms(d62),
        "branch_slope": slope(e62, d62),
        "branch_relL2": relL2(e62, d62),
        "branch_corr": float(np.corrcoef(e62, d62)[0, 1]),
        "out_slope_vs_teach": slope(c62, t62),
        "out_ratio": rms(c62) / rms(t62),
    }
    # --- chained deltas ---
    chain = {b: load(CHAIN / f"chain-block{b}_out.bin") for b in (62, 63, 64, 65)}
    teach = {b: load(TEACH / f"dump_block{b}_out.bin") for b in (62, 63, 64, 65)}
    for b in (63, 64, 65):
        dt = teach[b] - teach[b - 1]
        et = chain[b] - chain[b - 1]
        out[f"b{b}_branch"] = {
            "rms_teach_branch": rms(dt), "rms_gpu_branch": rms(et),
            "branch_ratio": rms(et) / rms(dt),
            "branch_slope": slope(et, dt),
            "branch_relL2": relL2(et, dt),
            "branch_corr": float(np.corrcoef(et, dt)[0, 1]),
            "out_slope_vs_teach": slope(chain[b], teach[b]),
            "out_ratio": rms(chain[b]) / rms(teach[b]),
            "out_relL2": relL2(chain[b], teach[b]),
            "note": "delta is chain-difference, NOT a pure branch (in != teacher in)",
        }
    # --- 16-periodicity of the per-channel *slope*, block62 (clean input) ---
    C = 64
    E = e62.reshape(-1, C).astype(np.float64)
    D = d62.reshape(-1, C).astype(np.float64)
    ch_slope = (E * D).sum(0) / np.maximum((D * D).sum(0), 1e-30)
    ch_rr = np.sqrt((E * E).mean(0)) / np.maximum(np.sqrt((D * D).mean(0)), 1e-30)
    out["b62_per_channel"] = {
        "branch_slope_64": [round(float(x), 4) for x in ch_slope],
        "branch_rms_ratio_64": [round(float(x), 4) for x in ch_rr],
        "slope_16period_dev": float(np.abs(np.tile(ch_slope[:16], 4) - ch_slope).max()),
        "slope_32period_dev": float(np.abs(np.tile(ch_slope[:32], 2) - ch_slope).max()),
    }
    print(json.dumps(out, indent=1))


if __name__ == "__main__":
    main()
