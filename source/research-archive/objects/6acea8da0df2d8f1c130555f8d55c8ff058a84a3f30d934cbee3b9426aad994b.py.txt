"""Read-only ISA histogram/section extractor for the F14 (vit qkv) pipeline.

Usage:
  python isa_stat.py <isa.asm> [hist]
  python isa_stat.py <isa.asm> grep <regex> [ctx]
  python isa_stat.py <isa.asm> window <start_line> <end_line>

No GPU, no build; reads a captured .asm only.
"""
import re
import sys
from collections import Counter

INSN = re.compile(r'^\s*/\*\s*\[([0-9A-Fa-f]+)\]\s*\*/\s*(.*)$')
OPC = re.compile(r'(?:\(W[^)]*\)\s+)?([a-z][a-z0-9_.]*)\s')


def lines(path):
    with open(path, 'r', encoding='utf-8', errors='replace') as f:
        return f.read().splitlines()


def parse(line):
    m = INSN.match(line)
    if not m:
        return None
    body = m.group(2)
    o = OPC.search(body)
    if not o:
        return None
    return m.group(1).upper(), o.group(1)


def hist(path):
    c = Counter()
    for ln in lines(path):
        p = parse(ln)
        if p:
            c[p[1]] += 1
    tot = sum(c.values())
    print('total instructions: %d  distinct opcodes: %d' % (tot, len(c)))
    for k, v in c.most_common():
        print('%6d  %s' % (v, k))


def main():
    path = sys.argv[1]
    cmd = sys.argv[2] if len(sys.argv) > 2 else 'hist'
    if cmd == 'hist':
        hist(path)
    elif cmd == 'grep':
        rx = re.compile(sys.argv[3])
        ctx = int(sys.argv[4]) if len(sys.argv) > 4 else 0
        ls = lines(path)
        hits = [i for i, l in enumerate(ls) if rx.search(l)]
        # merge windows
        shown = set()
        for i in hits:
            for j in range(max(0, i - ctx), min(len(ls), i + ctx + 1)):
                shown.add(j)
        for j in sorted(shown):
            print('%5d %s' % (j + 1, ls[j]))
            if ctx and j + 1 < len(ls) and (j + 1) not in shown:
                pass
        print('-- %d matching lines' % len(hits))
    elif cmd == 'window':
        a = int(sys.argv[3])
        b = int(sys.argv[4])
        ls = lines(path)
        for j in range(a - 1, min(b, len(ls))):
            print('%5d %s' % (j + 1, ls[j]))
    else:
        raise SystemExit('unknown command %r' % cmd)


if __name__ == '__main__':
    main()
