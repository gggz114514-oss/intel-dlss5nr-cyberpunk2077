"""Patch: treat a JSON null 'ordinary' block as absent (cards may declare no ordinary arm)."""
from pathlib import Path

files = ['submit_card.py', 'package.py', 'card_pipeline.py', 'adopt.py', 'audit_card.py']
patched = []
for f in files:
    p = Path(__file__).with_name(f)
    if not p.exists():
        continue
    t = p.read_text(encoding='utf-8')
    t2 = t.replace("req.get('ordinary', {})", "(req.get('ordinary') or {})")
    if t2 != t:
        p.write_text(t2, encoding='utf-8')
        patched.append(f)
print('patched:', patched)

# show remaining occurrences for review
for f in files:
    p = Path(__file__).with_name(f)
    if not p.exists():
        continue
    for i, line in enumerate(p.read_text(encoding='utf-8').splitlines(), 1):
        if "'ordinary'" in line:
            print('%s:%d %s' % (f, i, line.strip()))
