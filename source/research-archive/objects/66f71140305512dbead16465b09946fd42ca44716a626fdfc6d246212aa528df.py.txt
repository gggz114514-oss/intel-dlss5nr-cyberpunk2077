"""Find which captured ISA differs between two capture dirs (the shader under test)."""
import json, sys
from pathlib import Path
sys.path.insert(0, str(Path(__file__).parent))
import COMMON as K
import ISA

a = Path(sys.argv[1]); b = Path(sys.argv[2])
fa = {p.name: p for p in a.rglob('*.asm')}
fb = {p.name: p for p in b.rglob('*.asm')}
print('files', len(fa), len(fb), 'common', len(set(fa) & set(fb)))
rows = []
for n in sorted(set(fa) & set(fb)):
    sa, sb = K.sha(fa[n]), K.sha(fb[n])
    rows.append(dict(name=n, same=sa == sb, sha_a=sa, sha_b=sb))
diff = [r for r in rows if not r['same']]
print('identical', len(rows) - len(diff), 'differing', len(diff))
for r in diff:
    pa, pb = fa[r['name']], fb[r['name']]
    ta, tb = ISA.summary(pa), ISA.summary(pb)
    r['base'] = dict(instructions=ta['instructions'], sends=ta['sends'], dpas=ta['dpas'],
                     syncs=ta['syncs'], max_r=ta['max_r'], first_dpas=ta['first_dpas'])
    r['cand'] = dict(instructions=tb['instructions'], sends=tb['sends'], dpas=tb['dpas'],
                     syncs=tb['syncs'], max_r=tb['max_r'], first_dpas=tb['first_dpas'])
    print('%s\n  base %s\n  cand %s' % (r['name'], r['base'], r['cand']))
out = K.C / 'queue' / 'ISA-DIFF-TMP.json'
K.write_json(out, dict(a=str(a), b=str(b), rows=rows, differing=diff))
print('->', out)
