$ErrorActionPreference='Stop'
function SaveJson($p,$v){[IO.File]::WriteAllText($p,($v|ConvertTo-Json -Depth 40),[Text.UTF8Encoding]::new($false))}
function HashFile([string]$p){(Get-FileHash -LiteralPath $p -Algorithm SHA256).Hash.ToLowerInvariant()}
function HashBytes([byte[]]$b){$h=[Security.Cryptography.SHA256]::Create();try{([BitConverter]::ToString($h.ComputeHash($b))).Replace('-','').ToLowerInvariant()}finally{$h.Dispose()}}
function SaveLock($o){$b=[Text.Encoding]::UTF8.GetBytes(($o|ConvertTo-Json -Depth 20));$script:ls.Position=0;$script:ls.SetLength(0);$script:ls.Write($b,0,$b.Length);$script:ls.Flush()}

  if($ls -and $state){$state.gpu_pid=$null;$state.state='FINISHED';$state.finished_utc=[DateTime]::UtcNow.ToString('o');$state.exit_code=1;$state.failure=$failure;try{SaveLock $state;SaveJson (Join-Path $root 'GPU.lock.released.json') $state;SaveJson (Join-Path $root 'LOCK-EVIDENCE.json') ([ordered]@{path=$lock;owner=$state.owner;acquired_utc=$state.started_utc;exclusive_acquired=$true;previous_owner=$state.previous_owner;previous_state=$state.previous_state;previous_lock_copy=$state.previous_lock_copy;previous_lock_sha256=$state.previous_lock_sha256;release_pending=$false;released_utc=$state.finished_utc;release_state='FINISHED';final_lock_owner=$state.owner;final_lock_state='FINISHED'})}catch{}}
