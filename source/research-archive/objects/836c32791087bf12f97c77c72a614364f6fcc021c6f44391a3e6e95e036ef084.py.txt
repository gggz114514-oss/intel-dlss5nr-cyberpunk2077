"""Confirmation batch: handoff baseline vs the current accepted chain head, one batch.

This is the "combination must be re-measured on the latest accepted baseline" rule:
it checks that the four adopted steps compound to the same order as their individual
same-batch deltas, rather than assuming additivity.
"""
import json, sys
from pathlib import Path
sys.path.insert(0, str(Path(__file__).parent))
import COMMON as K
import STAGE
import ANALYZE

name = sys.argv[1] if len(sys.argv) > 1 else 'confirm-chain'
stage = sys.argv[2] if len(sys.argv) > 2 else 'ordinary'
head, head_sha, chain = K.current_best()
out = K.C / 'queue' / name
out.mkdir(parents=True, exist_ok=True)
run = out / 'run'

arms = [('handoff', K.BASELINE, K.BASELINE_SHA), ('head', head, head_sha)]
jobs = []
if stage == 'gates':
    for aid, p, s in arms:
        for mode in ('fixed', 'fb2'):
            jobs.append(K.gate_job('%s-%s-%s' % (stage, aid, mode), p, s,
                                   run / ('%s-%s-%s' % (stage, aid, mode)), mode, aid))
elif stage == 'ordinary':
    for pi in range(6):
        order = [arms[0], arms[1]] if pi % 2 == 0 else [arms[1], arms[0]]
        for aid, p, s in order:
            jobs.append(K.ordinary_job('ord-%d-%s' % (pi, aid), p, s,
                                       run / ('ord-%d-%s' % (pi, aid)), aid, pi))
spec = dict(stage=stage, owner='LUNA-CONTINUOUS-CONFIRM',
            authorization='Main (this session) authorized the chain confirmation batch only: handoff baseline vs current accepted chain head, gold required.',
            output_root=str(out / stage), identities=[str(K.BASELINE), str(head), str(K.RUNNER)],
            jobs=jobs, rules='chain confirmation; all samples retained, no rerun after nongold')
sp = out / ('spec-%s.json' % stage)
K.write_json(sp, spec)
manifest, ps1 = STAGE.prepare(spec)
r = STAGE.run(manifest, ps1)
s = STAGE.summarize(spec['output_root'])
K.write_json(Path(spec['output_root']) / 'STAGE-SUMMARY.json', s)
print(stage, s['status'], [(j['id'], j['status'], j['frame']) for j in s['jobs']])
if stage == 'ordinary' and s['status'] == 'COMPLETED':
    rep = ANALYZE.report(Path(spec['output_root']) / 'LUNA-ORDINARY.json',
                         [dict(arms=('handoff', 'head'))], out / 'TIMING.json',
                         extra=dict(chain=chain))
    c = rep['comparisons'][0]
    print('chain confirmation: %s saving=%.6f CI95=[%.6f, %.6f] groups=%d' % (
        c['status'], c['saving_ms'], c['CI95_ms'][0], c['CI95_ms'][1], c['groups']))
    print('pairs', [(g['pair'], round(g['handoff'], 5), round(g['head'], 5)) for g in c['group_means']])
    print('sum of same-batch increments = %.6f ms' % (K.read_json(K.C / 'CURRENT.json')['cumulative_ms_in_chain']))
