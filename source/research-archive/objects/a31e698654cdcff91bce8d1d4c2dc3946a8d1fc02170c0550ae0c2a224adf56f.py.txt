"""Assemble the read-only native baseline account for the F14 pipeline.

Takes the CURRENT-identity ISA that the campaign's own nat-index maps to
pipeline 24 (PASS 58 / vit_block31_qkv, SPV sha 72e2df7c...) and records the
counters this card predicts against.  Read-only: nothing is captured here, no
GPU is touched.
"""
import hashlib
import json
import re
from collections import Counter
from pathlib import Path

HERE = Path(__file__).resolve().parent
CARD = HERE.parent
EVID = CARD / 'evidence'
EVID.mkdir(exist_ok=True)
IDX = (CARD.parent.parent / 'cards' / 'nat-index' / 'evidence' / 'NATIVE-INDEX.json')
SPV_SHA = '72e2df7c7b1e542f375f5c7c2bdc572433a08c5f1a1a0d706026efa54b9a1e87'
PIPE = 24

INSN = re.compile(r'^\s*/\*\s*\[([0-9A-Fa-f]+)\]\s*\*/\s*(.*)$')
OPC = re.compile(r'(?:\(W[^)]*\)\s+)?(?:\([^)]*\)\s+)?([a-z][a-z0-9_.]*)')


def main():
    idx = json.loads(IDX.read_text(encoding='utf-8-sig'))
    BASE = 'a024c3c5aa1d1432ab3f0591bf03a897880d86be2c99d6de1e1492a2a7b2befe'
    cands = []
    for cap in idx['captures']:
        for r in cap['records']:
            if r.get('pipeline_in_capture') == PIPE and r.get('spv_sha256_at_capture') == SPV_SHA:
                plan = cap.get('plan') or {}
                cands.append((plan.get('sha256') == BASE, plan.get('sha256'), cap, r))
    cands.sort(key=lambda t: (not t[0], t[2].get('capture') or ''))
    assert cands, 'no capture of pipeline %d' % PIPE
    plan_ok, plan_sha, plan, rec = cands[0]
    isa = Path(rec['isa'])
    lines = isa.read_text(encoding='utf-8', errors='replace').splitlines()
    hist = Counter()
    n = 0
    for ln in lines:
        m = INSN.match(ln)
        if not m:
            continue
        o = OPC.search(m.group(2))
        if o:
            hist[o.group(1)] += 1
            n += 1
    indirect = [ln.strip() for ln in lines if re.search(r'>:[a-z]+ +r\[a0', ln)]
    a0_setup = [ln.strip() for ln in lines if re.search(r'a0\.0<1>:(uw|ud) +r2\.(0|16)', ln)]
    out = dict(
        schema='n12-vit-qkv/isa-account/v1',
        provenance='read-only: campaign nat-index record for pipeline %d whose capture plan sha256 '
                    'equals the F10 baseline plan (preferred) -- this card captured nothing' % PIPE,
        capture=plan.get('capture'),
        capture_root=plan.get('capture_root'),
        capture_plan_sha256=plan_sha,
        capture_plan_equals_baseline=plan_ok,
        all_isa_copies_identical=dict(
            note='every capture of this SPV in the nat-index reports the same native hash '
                 '2ef42bf87fecdff2 and instruction_lines 7633',
            captures_total=len(cands)),
        isa=dict(path=str(isa), sha256=hashlib.sha256(isa.read_bytes()).hexdigest(),
                 bytes=isa.stat().st_size, native_hash=rec.get('native_hash'),
                 dispatch=rec.get('dispatch'), subgroup=rec.get('subgroup')),
        spv=rec.get('spv'), spv_sha256_at_capture=rec.get('spv_sha256_at_capture'),
        instruction_lines=n,
        opcode_histogram=dict(hist.most_common()),
        counters=dict(
            dpas=hist.get('dpas.8x8', 0),
            send_ugm=hist.get('send.ugm', 0),
            send_gtwy=hist.get('send.gtwy', 0),
            sync_nop=hist.get('sync.nop', 0),
            sync_allrd=hist.get('sync.allrd', 0),
            goto=hist.get('goto', 0),
            join=hist.get('join', 0),
            jmpi=hist.get('jmpi', 0),
            math_rsqt=hist.get('math.rsqt', 0),
            sel=hist.get('sel', 0),
            cmp=hist.get('cmp', 0),
            mov=hist.get('mov', 0),
            add=hist.get('add', 0),
            mul=hist.get('mul', 0),
            shl=hist.get('shl', 0),
        ),
        cross_lane_reads=dict(
            note='GRF-indirect reads r[a0] are the compiler lowering of subgroupShuffle: the value '
                 'is staged in a GRF and read back at the source lane index',
            count=len(indirect), examples=indirect[:4]),
        a0_address_setups=len(a0_setup),
        driver_pipeline_statistics=dict(
            instruction_count=7633, basic_blocks=312, scratch_bytes=0,
            workgroup_memory_bytes=0, source='queue/n08-f08-wg-map/resources/run/resources-base/'
                                             'pipeline-statistics.json pipelines[24]'),
    )
    (EVID / 'ISA-ACCOUNT.json').write_text(json.dumps(out, indent=2), encoding='utf-8')
    print(json.dumps({k: v for k, v in out.items() if k != 'opcode_histogram'}, indent=2)[:4000])


if __name__ == '__main__':
    main()
