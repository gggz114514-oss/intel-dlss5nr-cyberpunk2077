"""Exhaustive integer/identity checks behind PROOF.json (card n04a-f05-c64-feed).

CPU only, no GPU.  Every assertion here is re-runnable and deterministic.
"""
import json
import sys
from pathlib import Path

sys.path.insert(0, r'@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/fullnet-continuous-20261027')
import COMMON as K

CARD = K.C / 'cards' / 'n04a-f05-c64-feed'
GROUP = [7, 10, 13, 16, 215, 218, 221, 224]
NS = (0, 16, 32, 48, 64, 80)
KS = (0, 16, 32, 48)
HMAX = 65536


def parent_index(wq, h, n, k0):
    return wq + (((96 * h + n) // 16) * 4 + k0 // 16) * 256


def candidate_index(wq, h, n, k0):
    wq_row = wq + ((96 * h) // 16) * 1024
    wk = wq_row + (k0 // 16) * 256
    return wk + 1024 * (n // 16)


def main():
    out = {}
    # 1. element index identity, exhaustive over the reachable head range
    bad = 0
    for wq in (0, 1, 7, 12345, 4294967000):
        for h in range(HMAX):
            for n in NS:
                for k0 in KS:
                    if parent_index(wq, h, n, k0) != candidate_index(wq, h, n, k0):
                        bad += 1
    out['index_identity'] = dict(
        pairs_checked=5 * HMAX * len(NS) * len(KS), mismatches=bad,
        domain=dict(wq='sampled incl. near-2^32', h='0..%d' % (HMAX - 1),
                    n=list(NS), k0=list(KS)),
        statement='((96h+n)/16*4 + k0/16)*256 == 6144h + 1024*(n/16) + 256*(k0/16)')

    # 2. exactness of the division (no truncation anywhere reachable)
    trunc = 0
    for h in range(HMAX):
        for n in NS:
            if (96 * h + n) % 16 or (96 * h) % 16:
                trunc += 1
    out['division_exact'] = dict(residues_nonzero=trunc,
                                 statement='96h and n are multiples of 16 for every reachable h,n')

    # 3. per-head pack ranges do not overlap and cover 24 tiles
    ranges = [(6 * h * 1024, 6 * h * 1024 + 6144) for h in range(64)]
    overlap = 0
    for a, b in zip(ranges, ranges[1:]):
        if a[1] > b[0]:
            overlap += 1
    idx = set()
    for h in (0, 1, 13, 63):
        for n in NS:
            for k0 in KS:
                idx.add(candidate_index(0, h, n, k0))
    out['pack_ranges'] = dict(overlapping_adjacent_heads=overlap,
                              distinct_indices_4_heads=len(idx),
                              expected=4 * len(NS) * len(KS),
                              statement='head h owns pack[6144h, 6144h+6144) = 24 tiles of 256 halves;'
                                        ' the 24 (n,k0) indices per head are distinct')

    # 4. accumulator component map is a bijection
    seen = set()
    for lane in range(32):
        for ci in range(4):
            row = 2 * ci + lane // 16
            col = lane % 16
            assert (row, col) not in seen
            seen.add((row, col))
    out['acc_component_map'] = dict(pairs=len(seen), distinct=len(seen),
                                    statement='(lane,ci)->(2*ci+lane/16, lane%16) covers 8x16 once')

    # 5. plan replacement count and identity
    row = K.row_by_ordinal(7)
    plan_text = (CARD / 'plan.txt').read_text(encoding='utf-8-sig')
    base_text = K.BASELINE.read_text(encoding='utf-8-sig')
    cand_posix = str(CARD / 'candidate.spv').replace('\\', '/')
    parent_posix = row['spv'].replace('\\', '/')
    expected = base_text.replace(parent_posix, cand_posix)
    out['plan'] = dict(
        group_size=len(GROUP),
        candidate_path_occurrences=plan_text.count(cand_posix),
        baseline_path_occurrences_in_plan=plan_text.count(parent_posix),
        baseline_path_occurrences_in_baseline=base_text.count(parent_posix),
        plan_equals_baseline_with_only_this_path_swapped=(expected == plan_text),
        changed_characters=len(plan_text) - len(expected),
        sha256=K.sha(CARD / 'plan.txt'))

    # 6. source / spv identity
    out['identity'] = dict(
        parent_spv=dict(path=str(row['spv']), sha256=K.sha(row['spv'])),
        parent_source=dict(path=str(row['source']), sha256=K.sha(row['source'])),
        parent_rebuild_sha256=K.sha(CARD / 'parent-rebuild.spv'),
        parent_rebuild_matches=K.sha(CARD / 'parent-rebuild.spv') == row['spv_sha256'],
        candidate_spv=dict(path=str(CARD / 'candidate.spv'), sha256=K.sha(CARD / 'candidate.spv')),
        candidate_source=dict(path=str(CARD / 'candidate.comp'), sha256=K.sha(CARD / 'candidate.comp')),
        base_plan_sha256=K.sha(K.BASELINE))

    K.write_json(CARD / 'build' / 'proof-check.json', out)
    print(json.dumps(out, indent=1)[:2400])
    assert bad == 0 and trunc == 0 and overlap == 0 and len(seen) == 128


if __name__ == '__main__':
    main()
