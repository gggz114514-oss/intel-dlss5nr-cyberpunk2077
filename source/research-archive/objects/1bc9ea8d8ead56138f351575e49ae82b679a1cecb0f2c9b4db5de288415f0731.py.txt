#!/usr/bin/env python3
"""One machine-generated coverage summary from the census + position map.

Nothing in the summary is typed by hand: every count comes from
`tools/plan_mma_census.py` (operand-level A/B dtype per PASS) over the plan actually on
disk, and the 129-position table comes from `tools/build_position_map.py`.

usage: summarize_coverage.py <census.json> <plan-substring> <position-map.json> --out dir
"""
import hashlib
import json
import sys
from collections import defaultdict
from pathlib import Path


def main(argv):
    out = None
    if "--out" in argv:
        i = argv.index("--out")
        out = Path(argv[i + 1])
        argv = argv[:i] + argv[i + 2:]
    census_path, needle, posmap_path = argv[0], argv[1], argv[2]
    c = [p for p in json.loads(Path(census_path).read_text(encoding="utf-8"))["plans"]
         if needle in p["plan"]][0]
    pm = json.loads(Path(posmap_path).read_text(encoding="utf-8"))
    rows = [r for r in c["rows"] if r["fp16_mma"] > 0]
    agg = defaultdict(lambda: [0, 0, []])
    for r in rows:
        a = agg[r["spv"]]
        a[0] += 1
        a[1] += r["fp16_mma"]
        a[2].append(r["pass_name"])
    doc = {
        "plan": {"path": c["plan"], "sha256": c["plan_sha256"], "header": c["header"],
                 "passes": c["passes"], "buffers": c["buffers"], "policies": c["policies"]},
        "fp16_static_mma": c["fp16_static_mma_total"],
        "int8_static_mma": c["int8_static_mma_total"],
        "passes_with_fp16": c["passes_with_fp16"],
        "positions": {"original": pm["g3"]["positions"],
                      "closed": pm["closed_positions"], "open": pm["open_positions"],
                      "fp16_sites_removed": pm["fp16_sites_removed"]},
        "residual_by_module": [{"module": k, "fp16": v[1], "passes": v[0],
                                "pass_names": v[2]}
                               for k, v in sorted(agg.items(), key=lambda kv: -kv[1][1])],
    }
    L = [f"# 覆盖率总结（机器生成，勿手改）", "",
         f"plan `{Path(c['plan']).name}` `{c['plan_sha256']}`",
         f"header `{c['header']}`　passes {c['passes']}　buffers {c['buffers']}　"
         f"policies {c['policies']}", "",
         f"**FP16 静态矩阵 {c['fp16_static_mma_total']} 条 / {c['passes_with_fp16']} 个 PASS**"
         f"　INT8 静态矩阵 {c['int8_static_mma_total']} 条", "",
         f"原 129 位置：**已关闭 {pm['closed_positions']}，未关闭 {pm['open_positions']}**，"
         f"共移除 {pm['fp16_sites_removed']} 条 FP16 矩阵", "",
         "| FP16 | PASS 数 | 模块 |", "|---:|---:|---|"]
    for e in doc["residual_by_module"]:
        L.append(f"| {e['fp16']} | {e['passes']} | `{e['module']}` |")
    L += ["", "状态词口径：静态零残余 ≠ 数值通过 ≠ 质量通过。本表只描述静态 dtype 覆盖。"]
    text = "\n".join(L) + "\n"
    if out:
        out.mkdir(parents=True, exist_ok=True)
        (out / "SUMMARY.json").write_text(json.dumps(doc, indent=1, ensure_ascii=False) + "\n",
                                          encoding="utf-8", newline="\n")
        (out / "SUMMARY.md").write_text(text, encoding="utf-8", newline="\n")
    print(text)
    return 0


if __name__ == "__main__":
    sys.exit(main(sys.argv[1:]))
