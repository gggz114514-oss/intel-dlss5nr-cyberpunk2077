from pathlib import Path
import json, hashlib, re
R=Path('@NR_WORKSPACE@')
O=R/'work/codex-astra-fp16-int8-audit-20261024'
mp=R/'candidates/dsh-int8-same-family-20261023/abi-probe/evidence/I8-ABI-MEASURED.json'
ap=mp.parent.parent/'generated/generated_i8_adapter.glsl'
m=json.loads(mp.read_text(encoding='utf-8'))
maps={k:{tuple(a):tuple(b) for a,b in v['lane_comp']} for k,v in m['maps'].items()}
src=ap.read_text(encoding='utf-8')
tables={}
for name in ('FRAG','LANE','BYTE'):
    body=re.search(r'const uint I8B_'+name+r'\[16\]\[32\] = \{(.*?)\n\};',src,re.S).group(1)
    tables[name]=[list(map(int,re.findall(r'(\d+)u',row))) for row in re.findall(r'\{([^{}]+)\}',body)]
assert len(maps['B'])==512 and len(set(maps['B'].values()))==512
mismatch=[]; seen=set(); table_diffs=0; signed_checks=0
for (lane,ci),expected in maps['B'].items():
    frag=ci>>2
    sl=(lane>>2)+((lane&1)<<4)+((ci&1)<<3)
    byte=(((ci>>1)&1)<<1)+((lane>>1)&1)
    row,col=maps['CH'][(sl,byte)]
    actual=(frag*8+row,col)
    if actual!=expected:mismatch.append([lane,ci,actual,expected])
    table_diffs+=int(any(tables[n][ci][lane]!=v for n,v in [('FRAG',frag),('LANE',sl),('BYTE',byte)]))
    seen.add(actual)
    for code in range(-127,128):
        word=(code&255)<<(8*byte)
        decoded=(word>>(8*byte))&255
        if decoded>=128:decoded-=256
        assert decoded==code
        signed_checks+=1
assert not mismatch and table_diffs==0 and len(seen)==512
# Two shuffles per packed fragment; each result supplies two destination bytes.
groups={}
for lane in range(32):
    pairs={(ci>>2,(lane>>2)+((lane&1)<<4)+((ci&1)<<3)) for ci in range(16)}
    groups[lane]=len(pairs)
assert set(groups.values())=={8}
# Integer-stage oracle, reconstruct B through actual mapping and compare with SLM layout.

E=((np.arange(128*16,dtype=np.int32).reshape(128,16)*43+11)%255)-127
ref=np.zeros((32,16),dtype=np.int32); got=ref.copy()
for kg in range(4):
    slab=E[kg*32:(kg+1)*32]
    words={}
    for frag in range(4):
        for sl in range(32):
            word=0
            for byte in range(4):
                r,c=maps['CH'][(sl,byte)]
                word|=(int(slab[frag*8+r,c])&255)<<(byte*8)
            words[frag,sl]=word
    direct=np.empty((32,16),dtype=np.int32)
    for (lane,ci),(r,c) in maps['B'].items():
        frag=ci>>2;sl=(lane>>2)+((lane&1)<<4)+((ci&1)<<3)
        byte=(((ci>>1)&1)<<1)+((lane>>1)&1)
        code=(words[frag,sl]>>(8*byte))&255
        direct[r,c]=code if code<128 else code-256
    assert np.array_equal(slab,direct)
    ref+=A[:,kg*32:(kg+1)*32]@slab
    got+=A[:,kg*32:(kg+1)*32]@direct
assert np.array_equal(ref,got)
