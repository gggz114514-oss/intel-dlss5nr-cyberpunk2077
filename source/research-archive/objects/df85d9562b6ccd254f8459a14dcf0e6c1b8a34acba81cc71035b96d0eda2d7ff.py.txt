def toks(line):
    try:
        return shlex.split(line)
    except ValueError:
        return line.split()

class Plan:
    def __init__(self, path):
        self.path = path
        self.buffers = []      # list of dict
        self.passes = []       # list of dict
        self.outputs = []      # list of dict
        self.bind_lines = []
        for ln in open(path, encoding="utf-8", errors="replace"):
            s = ln.strip()
            if not s:
                continue
            t = toks(s)
            k = t[0]
            if k == "PORT_NETWORK_1":
                self.header = [int(x) for x in t[1:5]]
            elif k == "BUFFER":
                self.buffers.append({"name": t[1], "bytes": int(t[2]), "file": t[3] if len(t) > 3 else "-"})
            elif k == "PASS":
                self.passes.append({"group": int(t[1]), "name": t[2], "spv": t[3], "args": t[4:]})
            elif k == "BIND":
                self.bind_lines.append([int(t[1]), int(t[2]), int(t[3]), int(t[4])])
            elif k == "OUTPUT":
                self.outputs.append({"buffer_index": int(t[1]), "format": t[2], "file": t[3]})
            elif k == "POLICY":
                pass
    def bname(self, i):
        return self.buffers[i]["name"] if 0 <= i < len(self.buffers) else "?%d" % i
    def sha(self):
        h = hashlib.sha256()
        h.update(open(self.path, "rb").read())
        return h.hexdigest()

# diff each arm vs FP16: number of changed SPV paths and their names

report = {}

    n = min(len(base.passes), len(pl.passes))
    changed = []
    for i in range(n):
        a, b = base.passes[i], pl.passes[i]
        if os.path.basename(a["spv"]) != os.path.basename(b["spv"]) or a["args"] != b["args"]:

    print("\n%s : %d/%d dispatches differ" % (k, len(changed), n))

        print("   ord=%-3d %-28s %s -> %s  argsame=%s" % c)
    if len(changed) > 25:
        print("   ... %d more" % (len(changed) - 25))
