"""Final self-check of the card directory: JSON parses, every recorded sha matches the
file on disk, the plan is the one that was cpu-checked, and the deliverable list is complete.
"""
import json, hashlib
from pathlib import Path

HERE = Path(__file__).parent
CAMP = HERE.parent.parent


def sha(p):
    return hashlib.sha256(Path(p).read_bytes()).hexdigest()


ok = True
for j in ("BUILD.json", "PROOF.json", "COST-LEDGER.json", "GPU-REQUEST.json", "A-AUDIT.json"):
    try:
        json.loads((HERE / j).read_text(encoding="utf-8-sig"))
        print("JSON OK   ", j)
    except Exception as e:                     # noqa: BLE001
        ok = False
        print("JSON FAIL ", j, e)

b = json.loads((HERE / "BUILD.json").read_text(encoding="utf-8"))
print("\n-- recorded sha vs disk")
print("plan", b["identity"]["candidate_plan"]["sha256"], "==", sha(HERE / "plan.txt"),
      b["identity"]["candidate_plan"]["sha256"] == sha(HERE / "plan.txt"))
for g in b["shader_groups"]:
    w = g["width"]
    spv = HERE / ("candidate-c%d.spv" % w)
    src = HERE / ("candidate-c%d.comp" % w)
    a = g["candidate_spv"]["sha256"] == sha(spv)
    c = g["candidate_source"]["sha256"] == sha(src)
    print("c%-3d spv %s  src %s" % (w, a, c))
    ok = ok and a and c
    par = Path(g["parent_spv"]["path"])
    print("      parent file sha ok:", sha(par) == g["parent_spv"]["sha256"])

req = json.loads((HERE / "GPU-REQUEST.json").read_text(encoding="utf-8"))
print("\n-- GPU-REQUEST")
print("plan sha matches plan.txt:", req["plan"]["sha256"] == sha(HERE / "plan.txt"))
print("pass_ordinals:", len(req["shader_group"]["pass_ordinals"]), "unique:",
      len(set(req["shader_group"]["pass_ordinals"])))
print("baseline spv list:", len(req["shader_group"]["spv_sha256_baseline"]))

need = ["ALUWR-AUDIT.md", "ANALYSIS.md", "PROOF.json", "SOURCE-DIFF.patch", "BUILD.json",
        "plan.txt", "COST-LEDGER.json", "NATIVE-PREDICTION.md", "GPU-REQUEST.json", "RESUME.txt",
        "candidate-c64.comp", "candidate-c64.spv", "candidate-c128.comp", "candidate-c128.spv",
        "candidate-c256.comp", "candidate-c256.spv"]
print("\n-- deliverables")
for n in need:
    e = (HERE / n).exists()
    ok = ok and e
    print("  %-24s %s %8d B" % (n, "OK " if e else "MISSING", (HERE / n).stat().st_size if e else 0))
print("\nALL CHECKS PASS:", ok)
