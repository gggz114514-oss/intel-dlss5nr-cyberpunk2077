"""n29-f01-alu-chain build: parent-byte reproduction + candidate compile.

Read-only on the campaign infrastructure: imports COMMON (never modifies it) and
uses the recorded F01 nc_recipe (decoration migration from the parent body
c32-no-gain-diagnosis-20261026/combined-existing/f01.spv + the 189-id
NoContraction set), which is the same recipe n18 recorded for this PASS.

Usage:
  python build.py parent     -> parent-rebuild.spv + parent-rebuild.json
  python build.py candidate  -> candidate.spv (+ raw) + candidate-build.json
"""
import json
import sys
from pathlib import Path

CARD = Path(__file__).resolve().parents[1]
sys.path.insert(0, str(CARD.parents[1]))
import COMMON as K                      # noqa: E402
import spv_tools_local as ST            # noqa: E402

ORD = 0
RECIPES = K.read_json(CARD.parents[1] / 'evidence' / 'RECIPES.json')
SHA = K.row_by_ordinal(ORD)['spv_sha256']            # 31c2fd9a... (handoff row, pre-n18)
RECIPE = RECIPES[SHA]
PARENT_SPV = Path(RECIPE['nc_recipe']['parent'])     # combined-existing/f01.spv
INCLUDES = [Path(p) for p in RECIPE['compile']['include_dirs']]
PINNED = Path(RECIPE['compile']['pinned_include_dir'])

# the *card* parent is n18's adopted candidate, not the handoff row
CARD_PARENT_SRC = CARD.parents[0] / 'n18-f01-entry' / 'candidate.comp'
CARD_PARENT_SPV = CARD.parents[0] / 'n18-f01-entry' / 'candidate.spv'


def includes_for(comp):
    dirs = [Path(comp).parent, PINNED]
    for d in INCLUDES:
        if d not in dirs:
            dirs.append(d)
    for d in K.include_dirs_for(ORD, comp):
        if Path(d) not in dirs:
            dirs.append(Path(d))
    return dirs


def decorate(raw, out):
    pm = ST.Module(PARENT_SPV)
    cm = ST.Module(raw)
    keep, info = ST.transfer_decoration_pattern_aligned(pm, cm, set())
    ST.set_no_contraction(cm, keep, out)
    return dict(
        parent_nc=len(ST.no_contraction_ids(pm)),
        nc_keep=len(keep),
        functions_mapped=info.get('functions_mapped'),
        child_functions_without_parent_counterpart=len(
            info.get('child_functions_without_parent_counterpart') or []),
        parent_decorations_total=info.get('parent_decorations_total'),
        child_own_decorations_outside_pattern=info.get('child_own_decorations_outside_pattern'),
    )


def build(kind):
    if kind == 'parent':
        src, raw, out = CARD_PARENT_SRC, CARD / 'parent-raw.spv', CARD / 'parent-rebuild.spv'
        expected = K.sha(CARD_PARENT_SPV)
    else:
        src, raw, out = CARD / 'candidate.comp', CARD / 'candidate-raw.spv', CARD / 'candidate.spv'
        expected = None
    incs = includes_for(src)
    K.compile_glsl(src, raw, incs)
    dec = decorate(raw, out)
    res = dict(kind=kind, source=K.ident(src), includes=[str(p) for p in incs],
               raw=K.ident(raw), decorated=K.ident(out), decor=dec, expected=expected)
    res['raw_match'] = (res['raw']['sha256'] == expected) if expected else None
    res['decorated_match'] = (res['decorated']['sha256'] == expected) if expected else None
    K.write_json(CARD / ('parent-rebuild.json' if kind == 'parent' else 'candidate-build.json'), res)
    print(json.dumps({k: res[k] for k in ('kind', 'raw', 'decorated', 'decor',
                                          'raw_match', 'decorated_match')}, indent=1))
    return res


if __name__ == '__main__':
    build(sys.argv[1])
