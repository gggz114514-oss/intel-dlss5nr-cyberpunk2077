"""Compact SPIR-V block dump (control flow + cooperative-matrix ops) for n11."""
import struct, sys
from pathlib import Path

NAMES = {
    248: 'OpLabel', 249: 'OpBranch', 250: 'OpBranchConditional', 246: 'OpLoopMerge',
    247: 'OpSelectionMerge', 245: 'OpPhi', 71: 'OpDecorate', 59: 'OpVariable',
    61: 'OpLoad', 62: 'OpStore', 65: 'OpAccessChain',
    4456: 'CoopLoad', 4457: 'CoopStore', 4458: 'CoopMulAdd', 4460: 'CoopLength',
    43: 'OpConstant',
}
COOP = {4456, 4457, 4458, 4460}


def load(path):
    w = list(struct.unpack('<%dI' % (Path(path).stat().st_size // 4), Path(path).read_bytes()))
    i, out = 5, []
    while i < len(w):
        op, n = w[i] & 0xFFFF, w[i] >> 16
        out.append((op, w[i + 1:i + n], i))
        i += n
    return out


def dump(path):
    print('==', path)
    for op, a, off in load(path):
        if op == 248:
            print('L%-4d:' % a[0], end=' ')
        elif op == 249:
            print('BR->%d' % a[0])
        elif op == 250:
            print('BRCOND cond=%d T->%d F->%d' % (a[0], a[1], a[2]))
        elif op == 246:
            print('LOOPMERGE merge=%d cont=%d' % (a[0], a[1]))
        elif op == 247:
            print('SELMERGE merge=%d' % a[0])
        elif op in COOP:
            print('%s(len %d) %s' % (NAMES[op], len(a), a[:4]), end=' ')
    print()


for p in sys.argv[1:]:
    dump(p)
