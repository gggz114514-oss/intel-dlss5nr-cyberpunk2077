def dyn():
    """Per-subgroup dynamic counts.

    parent    : 4 j  x [1 M block] x [2 weight n-blocks] x 8 k
                A = 4*8 = 32, B = 4*2*8 = 64, seed = 4*2 = 8, store = 4*2 = 8, DPAS = 64
    candidate : 2 jp x [2 M blocks] x [2 weight n-blocks] x 8 k
                A = 2*2*8 = 32 (one per M block and k), B = 2*2*8 = 32 (one per n-block
                and k, shared by both M blocks), seed = 2*4 = 8, store = 2*4 = 8, DPAS = 64
    """
    d = {}
    for name, a_tiles, b_tiles, seeds, stores in (
            ('parent', 4 * 8, 4 * 2 * 8, 4 * 2, 4 * 2),
            ('candidate', 2 * 2 * 8, 2 * 2 * 8, 2 * 4, 2 * 4)):
        d[name] = dict(dpas=64, a_tile_loads=a_tiles, b_tile_loads=b_tiles,
                       seed_loads=seeds, coop_loads=a_tiles + b_tiles + seeds,
                       stores=stores)
    for name in d:
        d[name].update(
            wg_b_tile_loads=d[name]['b_tile_loads'] * 8,
            wg_a_tile_loads=d[name]['a_tile_loads'] * 8,
            wg_seed_loads=d[name]['seed_loads'] * 8,
            wg_sends=8 * (d[name]['a_tile_loads'] * 8 + d[name]['b_tile_loads']
                          + d[name]['seed_loads'] * 8 + d[name]['stores'] * 8),
            wg_load_bytes=8 * (d[name]['a_tile_loads'] * A_TILE
                               + d[name]['b_tile_loads'] * B_TILE
                               + d[name]['seed_loads'] * A_TILE),
        )
    return d


def native_census(path):
    isa = ISA.parse(path)
    ops = collections.Counter(i['op'] for i in isa)
    forms = collections.Counter()
    for i in isa:
        if i['op'].startswith('send'):
            m = re.search(r'(load|store)\.ugm\.([A-Za-z0-9]+)\.([A-Za-z0-9]+)', i['comment'] or '')
            forms[(m.group(1) + '.' + m.group(2)) if m else 'other'] += 1
    regs = set()
    for i in isa:
        for o in i['ops']:
            for m in re.finditer(r'\br(\d+)', o):
                regs.add(int(m.group(1)))
    return dict(path=str(path), sha256=K.sha(path), instructions=len(isa),
                dpas=ops['dpas.8x8'], sends=sum(v for k, v in ops.items() if k.startswith('send')),
                send_forms=dict(forms), mov=ops['mov'], add=ops['add'], mul=ops['mul'],
                sel=ops['sel'], cmp=ops['cmp'], sync_nop=ops['sync.nop'],
                sync_allwr=ops['sync.allwr'], sync_allrd=ops['sync.allrd'],
                barriers=ops['sync.bar'], max_r=max(regs), distinct_r=len(regs),
                first_dpas=hex(min(i['addr'] for i in isa if i['op'].startswith('dpas'))))
