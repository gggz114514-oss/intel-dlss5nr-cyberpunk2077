import re, collections
from pathlib import Path
F = Path(r"@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/fullnet-mechanism-transfer-20261026/f05-k-consumer-tile/c256/captures/native-f20-pixel-owner/ISA/CS_asmfd78c91bb8744f0f_simd32_000000000000000e_13_orig.asm")
lines = F.read_text(encoding="utf-8", errors="replace").splitlines()
FENCE = next(i for i, l in enumerate(lines) if "fence.slm" in l)
DPAS = re.compile(r"dpas\.[\d x]+\((\d+)\|M0\)\s+(\S+)\s+(\S+)\s+(\S+)\s+(\S+)")
recs = []
for i, l in enumerate(lines):
    m = DPAS.search(l)
    if m:
        recs.append((i, m.group(2).split(":")[0], m.group(4).split(":")[0], m.group(5).split(":")[0]))

qkv = [r for r in recs if r[0] < FENCE]
att = [r for r in recs if r[0] >= FENCE]
print("QKV dpas dst regs (distinct, in order):")
seen = []
for _, d, a, b in qkv:
    if d not in seen:
        seen.append(d)
print("  ", seen)
print("   distinct dst:", len(seen))
print("QKV A regs distinct:", len(set(r[2] for r in qkv)), " B regs distinct:", len(set(r[3] for r in qkv)))
print("ATT dpas dst regs:", sorted(set(r[1] for r in att), key=lambda s: int(s[1:])))
print("ATT A regs:", sorted(set(r[2] for r in att), key=lambda s: int(s[1:])))
print("ATT B regs:", sorted(set(r[3] for r in att), key=lambda s: int(s[1:])))

# register footprint
mx = 0
for l in lines:
    for g in re.findall(r"\br(\d+)\.", l):
        mx = max(mx, int(g))
print("max register index touched:", mx)
# dpas dst contiguous tuples for QKV first 12
print()
print("first 14 QKV dpas (line, dst, A, B):")
for r in qkv[:14]:
    print("   ", r)
print("last 8 QKV dpas:")
for r in qkv[-8:]:
    print("   ", r)
