"""Show the ISA window around the fb-bias reads so the scheduling/interleaving
with dpas can be judged.  CPU only.

Usage: window.py <asm> [start_hex] [end_hex]
"""
import sys
from pathlib import Path
sys.path.insert(0, str(Path(__file__).parent))
import isaparse as P

path = sys.argv[1]
a0 = int(sys.argv[2], 16) if len(sys.argv) > 2 else 0xb600
a1 = int(sys.argv[3], 16) if len(sys.argv) > 3 else 0xc300
isa = P.parse(path)
for i in isa:
    if a0 <= i['addr'] <= a1:
        print('[%04x] %s' % (i['addr'], i['raw'].split('*/')[-1].strip()))
