"""Ablation: revert selected accepted steps in the chain head and measure the difference.

Usage: python ablation.py <name> <step_index,step_index,...>

The chain head stays the base arm; the ablation plan is the chain head with the selected
steps' PASS lines pointing back at their pre-step SPVs. A positive saving means those
steps are worth that many ms (i.e. the chain head is faster than the ablation).
"""
import json, re, sys
from pathlib import Path
sys.path.insert(0, str(Path(__file__).parent))
import COMMON as K
from audit_card import pass_lines

name = sys.argv[1]
steps = [int(x) for x in sys.argv[2].split(',')]
cur = K.read_json(K.C / 'CURRENT.json')
chain = cur['chain']
head = Path(cur['best_plan']['path'])
lines = head.read_text(encoding='utf-8-sig').splitlines()
recs = pass_lines(head)

reverted = []
for si in steps:
    step = chain[si]
    prev = Path(chain[si - 1]['path'])
    own = Path(step['path'])
    a = pass_lines(prev)
    b = pass_lines(own)
    n = 0
    for ra, rb in zip(a, b):
        if ra['line'] == rb['line']:
            continue
        # locate the same PASS in the head by name (later steps may have moved other lines)
        idx = next((i for i, l in enumerate(lines) if l.split()[1:3][-1] and ra['name'] in l), None)
        if idx is None:
            raise SystemExit('cannot locate pass %s in head' % ra['name'])
        lines[idx] = ra['line']
        n += 1
    reverted.append(dict(step=si, label=step.get('label'), passes=n))
print('reverted', reverted)

out = K.C / 'queue' / name
out.mkdir(parents=True, exist_ok=True)
plan = out / 'plan.txt'
plan.write_text('\n'.join(lines) + ('\n' if head.read_text(encoding='utf-8-sig').endswith('\n') else ''),
                encoding='utf-8')
# sanity: the ablation plan must differ from the head exactly in the reverted passes
ha = pass_lines(head)
ab = pass_lines(plan)
changed = [x['name'] for x, y in zip(ha, ab) if x['line'] != y['line']]
print('ablation differs from head in %d PASS lines' % len(changed))
K.write_json(out / 'ABLATION.json', dict(name=name, reverted=reverted, changed_passes=len(changed),
                                         plan=dict(path=str(plan), sha256=K.sha(plan)),
                                         head=dict(path=str(head), sha256=K.sha(head))))
print('plan', K.sha(plan))
