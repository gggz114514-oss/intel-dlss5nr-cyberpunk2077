"""N02 F20/F21 tail card build: parent byte reproduction, candidate compile,
SPIR-V validation, chain-head plan rewrite (one PASS), CPU plan check.

No GPU.  Writes only inside this card directory.
"""
import json, pathlib, subprocess, sys, hashlib, shutil

CARD = pathlib.Path(r'@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/fullnet-continuous-20261027/cards/n02-f20-f21-tail')
CAMP = pathlib.Path(r'@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/fullnet-continuous-20261027')
W = pathlib.Path(r'@NR_WORKSPACE@')
GLSLANG = W / 'src/project/tools/glslang-16.5.0/bin/glslang.exe'
VALIDATE = W / 'candidates/amd-map-20261015/s4-final-output-fusion-fix/production-repair/spv_validate_cpu.exe'
RUNNER = W / 'work/codex-astra-fp16-int8-audit-20261024/phase-diagnosis-20261025/direct-clock-build/Release/clock_runner.exe'
PIN = CAMP / 'cards/idt-identity-recipe/evidence/headers/b49ee2a8ef02'
SITE = W / 'work/codex-astra-fp16-int8-audit-20261024/c32-boundary-candidates-20261026/f20-four-output'

sys.path.insert(0, str(CAMP))
import COMMON as K


def sha(p):
    return hashlib.sha256(pathlib.Path(p).read_bytes()).hexdigest()


def run(argv):
    r = subprocess.run([str(x) for x in argv], capture_output=True, text=True)
    return r.returncode, r.stdout, r.stderr


def main():
    (CARD / 'build').mkdir(exist_ok=True)
    out = {'card': 'n02-f20-f21-tail', 'family': 'F21', 'gpu_used': False}

    # ---- 1. parent byte reproduction from the pristine site source+headers ----
    rc, so, se = run([GLSLANG, '-V', '--target-env', 'vulkan1.3', '-I' + str(PIN),
                      SITE / 'tail.comp', '-o', CARD / 'parent-rebuild.spv'])
    out['parent_rebuild'] = {
        'argv': [str(GLSLANG), '-V', '--target-env', 'vulkan1.3', '-I' + str(PIN),
                 str(SITE / 'tail.comp'), '-o', str(CARD / 'parent-rebuild.spv')],
        'rc': rc, 'stderr_tail': se[-200:],
        'sha256': sha(CARD / 'parent-rebuild.spv'),
        'baseline_spv_sha256': 'b49ee2a8ef028fefb07d2b139227d0f7fba7b99a23b2ef213615445fd288c223',
    }
    out['parent_rebuild']['byte_match'] = (out['parent_rebuild']['sha256'] ==
                                           out['parent_rebuild']['baseline_spv_sha256'])
    assert out['parent_rebuild']['byte_match'], 'parent did not reproduce'

    # ---- 2. candidate compile with the card-local header copy ----
    rc, so, se = run([GLSLANG, '-V', '--target-env', 'vulkan1.3', '-I' + str(CARD / 'headers'),
                      CARD / 'candidate.comp', '-o', CARD / 'candidate.spv'])
    out['candidate'] = {
        'argv': [str(GLSLANG), '-V', '--target-env', 'vulkan1.3', '-I' + str(CARD / 'headers'),
                 str(CARD / 'candidate.comp'), '-o', str(CARD / 'candidate.spv')],
        'rc': rc, 'stderr_tail': se[-200:],
        'sha256': sha(CARD / 'candidate.spv'),
        'bytes': (CARD / 'candidate.spv').stat().st_size,
    }

    # ---- 3. SPIR-V validation ----
    rc, so, se = run([VALIDATE, CARD / 'candidate.spv'])
    out['spv_validate_cpu'] = {'candidate': {'rc': rc, 'stdout': so.strip(), 'stderr': se.strip()}}
    rc, so, se = run([VALIDATE, CARD / 'parent-rebuild.spv'])
    out['spv_validate_cpu']['parent'] = {'rc': rc, 'stdout': so.strip(), 'stderr': se.strip()}

    # ---- 4. plan on the current chain head, replacing the F21 PASS only ----
    base_plan, base_sha, chain = K.current_best()
    n = K.make_plan(CARD / 'candidate.spv', CARD / 'plan.txt', parent_plan=base_plan,
                    spv_shas={'b49ee2a8ef028fefb07d2b139227d0f7fba7b99a23b2ef213615445fd288c223'})
    out['plan'] = {
        'path': str(CARD / 'plan.txt'), 'sha256': sha(CARD / 'plan.txt'),
        'parent_plan': {'path': str(base_plan), 'sha256': base_sha},
        'chain_head_label': chain[-1]['label'] if chain else None,
        'replaced_passes': n, 'expected_passes': 1,
    }
    assert n == 1, n

    # ---- 5. CPU plan check (candidate and chain-head parent) ----
    rc, so, se = run([RUNNER, '--plan', CARD / 'plan.txt', '--cpu-check'])
    out['cpu_check'] = {'argv': [str(RUNNER), '--plan', str(CARD / 'plan.txt'), '--cpu-check'],
                        'rc': rc, 'stdout': so.strip(), 'stderr': se.strip()}
    rc, so2, se2 = run([RUNNER, '--plan', base_plan, '--cpu-check'])
    out['cpu_check']['parent_stdout'] = so2.strip()
    out['cpu_check']['identical_to_parent'] = (so.strip() == so2.strip())

    (CARD / 'BUILD.json').write_text(json.dumps(out, indent=2), encoding='utf-8')
    print(json.dumps({k: out[k] for k in ('parent_rebuild', 'candidate', 'spv_validate_cpu',
                                          'plan', 'cpu_check')}, indent=1)[:4000])


if __name__ == '__main__':
    main()
