# -*- coding: utf-8 -*-
"""Extra observation: is the bright-blue deficit multiplicative or additive?

Bins the FP16 reference blue channel inside the fixed 'bright' mask and reports the
signed INT8-FP16 blue error per bin (no fitting, no correction).
"""
import json
from pathlib import Path

import numpy as np

OUT = Path(__file__).resolve().parent.parent
B = Path(r"@NR_WORKSPACE@/work/dsh-fullnet-static-20261024".replace("fullnet", "codex-astra-fp16-int8-audit-20261024"))
Q = Path(r"@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/int8-quality-first-20261029")
W = Path(r"@NR_WORKSPACE@/work/dsh-fullnet-static-20261029")
LUMA = np.array([0.2126, 0.7152, 0.0722])
FRAMES = [(34189, "day-street-static-formal-01"),
          (41444, "day-street-motion-formal-01"),
          (43654, "day-street-motion-formal-02")]
ARMS = {"xe": lambda f: W / f"S02-f01/runs24/run-{f}/replayed-network.bin",
        "c64": lambda f: Q / f"real-int8-24/c64-family-static/run-{f}/replayed-network.bin"}
EDGES = [0.70, 0.80, 0.90, 1.00, 1.01]

out = {"question": "does the bright-bucket blue deficit scale with the reference blue value?",
       "method": "fixed mask bright = FP16 luma >= 0.70; bins on FP16 plane0 B; "
                 "signed error = INT8 - FP16; no fit, no correction",
       "frames": []}
for frame, seq in FRAMES:
    ref = np.fromfile(Q / f"dataset/{seq}/0-{frame}/network-output.bin",
                      dtype="<f4").reshape(2, 720, 1280, 4)[0, :, :, :3].astype(np.float64)
    y = ref @ LUMA
    bright = y >= 0.70
    fr = {"frame": frame, "arms": {}}
    for arm, fn in ARMS.items():
        cand = np.fromfile(fn(frame), dtype="<f4").reshape(2, 720, 1280, 4)[0, :, :, :3].astype(np.float64)
        d = (cand - ref)[bright]
        bref = ref[bright][:, 2]
        rows = []
        for lo, hi in zip(EDGES[:-1], EDGES[1:]):
            m = (bref >= lo) & (bref < hi)
            if m.sum() < 20:
                rows.append({"bin": f"[{lo:.2f},{hi:.2f})", "n": int(m.sum())})
                continue
            rows.append({"bin": f"[{lo:.2f},{hi:.2f})", "n": int(m.sum()),
                         "ref_B_mean": float(bref[m].mean()),
                         "dB_mean": float(d[m, 2].mean()),
                         "dB_relative": float(d[m, 2].mean() / max(bref[m].mean(), 1e-9)),
                         "dR_mean": float(d[m, 0].mean()),
                         "dG_mean": float(d[m, 1].mean()),
                         "frac_ref_at_clamp": float((bref[m] >= 1.0).mean())})
        fr["arms"][arm] = rows
        print(frame, arm)
        for r in rows:
            if "dB_mean" in r:
                print(f"   B{r['bin']:12s} n={r['n']:6d} refB={r['ref_B_mean']:.3f} "
                      f"dB={r['dB_mean']:+.4f} ({r['dB_relative']*100:+.1f}%) "
                      f"dR={r['dR_mean']:+.4f} dG={r['dG_mean']:+.4f} atClamp={r['frac_ref_at_clamp']*100:.1f}%")
            else:
                print(f"   B{r['bin']:12s} n={r['n']:6d} (too few samples)")
    out["frames"].append(fr)

(OUT / "evidence/bright-B-conditioning.json").write_text(json.dumps(out, indent=1), encoding="utf-8")
doc = json.loads((OUT / "COLOR-BASELINE.json").read_text(encoding="utf-8"))
doc["bright_B_conditioning"] = out
(OUT / "COLOR-BASELINE.json").write_text(json.dumps(doc, indent=1), encoding="utf-8")
print("added bright_B_conditioning to COLOR-BASELINE.json")
