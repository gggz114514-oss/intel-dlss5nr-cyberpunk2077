"""n23-f05-shape build: parent byte reproduction, B1-V candidate, plans, cpu-check, NC identity.

No GPU. Everything here is CPU-only and read-only outside this card directory.

Arms produced:
  plan.txt                       B1-V candidate on all 12 F05 C128 PASS rows (SPV token only;
                                 BUFFER rows untouched, by the main-agent ruling - the asset rows
                                 live in BUFFER-EDITS.json and are applied by the main agent)
  probe/lanemap-probe.plan.txt   the lane-map probe: PASS 19 -> the probe SPV, plus the one POLICY
                                 token that makes the runner dump swin_9_debug (see RESUME)
"""
import json
import shutil
import sys
from pathlib import Path

CARD = Path(__file__).parent
sys.path.insert(0, str(CARD.parent.parent))          # campaign root (COMMON.py)
import COMMON as K                                    # noqa: E402

C = K.C
PARENT_SHA = '20f4074fc70c32c0e74f0ad000053492e5f5c420259da6044e7a00cfc3ee3f4f'
PROBE_PASS = 19
DBG_RESOURCE = 50            # swin_9_debug, binding 4 of the replaced PASS


def sha(p):
    return K.sha(p)


def log(*a):
    print(*a, flush=True)


def main():
    out = {}
    rec = K.read_json(C / 'evidence' / 'RECIPES.json')[PARENT_SHA]
    row = K.row_by_ordinal(PROBE_PASS)
    parent_src = Path(row['source'])
    parent_spv = Path(row['spv'])
    log('parent source', parent_src)
    log('parent spv   ', parent_spv, sha(parent_spv))
    assert sha(parent_spv) == PARENT_SHA, 'baseline row spv sha drifted'
    ords = [r['ordinal'] for r in K.rows() if r['spv_sha256'] == PARENT_SHA]
    log('F05 C128 group ordinals', ords)
    assert len(ords) == 12, ords

    # ---- 1. parent byte reproduction (RECIPES.json: raw compile, pinned headers)
    incs = [Path(rec['compile']['pinned_include_dir'])] + \
           [Path(d) for d in rec['compile']['include_dirs']]
    b = CARD / 'build'
    b.mkdir(exist_ok=True)
    raw = b / 'parent-raw.spv'
    K.compile_glsl(parent_src, raw, incs)
    parent_ok = sha(raw) == PARENT_SHA
    if parent_ok:
        shutil.copyfile(raw, CARD / 'parent-rebuild.spv')
    log('parent rebuild', sha(raw), 'match', parent_ok)
    assert parent_ok, 'parent byte reproduction failed; identity trio cannot be claimed'

    # ---- 2. candidate compile + validate
    cand = CARD / 'candidate.spv'
    K.compile_glsl(CARD / 'candidate.comp', cand, incs)
    K.validate_spv(cand)
    log('candidate', sha(cand), 'validated')

    # ---- 3. candidate plan: all 12 PASS rows of the group, SPV token only
    base_plan, base_sha, _ = K.current_best()
    n = K.make_plan(cand, CARD / 'plan.txt', parent_plan=base_plan, spv_shas={PARENT_SHA})
    plan = CARD / 'plan.txt'
    log('plan', sha(plan), 'rows replaced', n)
    assert n == 12, n
    head_txt = Path(base_plan).read_text(encoding='utf-8-sig').splitlines()
    new_txt = plan.read_text(encoding='utf-8').splitlines()
    assert len(head_txt) == len(new_txt)
    diffs = [i for i, (a, c) in enumerate(zip(head_txt, new_txt)) if a != c]
    log('plan lines differing from head:', len(diffs), diffs)
    assert len(diffs) == 12, len(diffs)
    for i in diffs:                     # only the SPV token may differ
        a, c = head_txt[i].split(), new_txt[i].split()
        assert a[:3] == c[:3] and a[4:] == c[4:] and a[3] != c[3], (i, a[:4], c[:4])
    cc = K.cpu_check(plan)
    (b / 'cpu-check.txt').write_text(cc, encoding='utf-8')
    log('cpu-check:', cc.strip())

    # ---- 4. the probe arm: PASS 19 -> probe SPV plus the POLICY token that enables the dump
    probe_spv = CARD / 'probe' / 'lanemap_probe.spv'
    K.validate_spv(probe_spv)
    lines = Path(base_plan).read_text(encoding='utf-8-sig').splitlines(True)
    recs = K.pass_lines_of(base_plan)
    target = recs[PROBE_PASS]['line']
    hit = 0
    for i, l in enumerate(lines):
        if l.rstrip('\n') == target:
            lines[i] = l.replace(recs[PROBE_PASS]['spv'], probe_spv.as_posix(), 1)
            hit += 1
    assert hit == 1, hit
    pol = 0
    for i, l in enumerate(lines):
        t = l.split()
        if t and t[0] == 'POLICY' and t[1] == str(DBG_RESOURCE) and t[2] == '1':
            t[2] = '0'
            lines[i] = ' '.join(t) + '\n'
            pol += 1
    assert pol == 1, pol
    probe_plan = CARD / 'probe' / 'lanemap-probe.plan.txt'
    probe_plan.write_text(''.join(lines), encoding='utf-8')
    pcc = K.cpu_check(probe_plan)
    (b / 'cpu-check-probe.txt').write_text(pcc, encoding='utf-8')
    log('probe plan', sha(probe_plan), '| cpu-check:', pcc.strip())

    # ---- 5. NoContraction accounting
    st = K.ST
    nc_parent = st.no_contraction_ids(st.Module(parent_spv))
    nc_cand = st.no_contraction_ids(st.Module(cand))
    log('NC parent', len(nc_parent), 'candidate', len(nc_cand))

    out.update(card='n23-f05-shape',
               parent=dict(plan=str(base_plan), plan_sha256=base_sha,
                           spv=str(parent_spv), spv_sha256=PARENT_SHA,
                           source=str(parent_src), source_sha256=sha(parent_src),
                           rebuild=str(CARD / 'parent-rebuild.spv'),
                           rebuild_sha256=sha(CARD / 'parent-rebuild.spv'),
                           rebuild_matches=True,
                           recipe=rec['nc_recipe']),
               candidate=dict(spv=str(cand), spv_sha256=sha(cand),
                              source=str(CARD / 'candidate.comp'),
                              source_sha256=sha(CARD / 'candidate.comp'),
                              identity='B1-V stage 1: V0+V1 merged into one 8x32 accumulator fed '
                                       'by one 16x32 B fragment; Q/K keep 8x16 but read the '
                                       '32-channel-blocked pack; no other source change',
                              validated=True),
               probe=dict(spv=str(probe_spv), spv_sha256=sha(probe_spv),
                          source=str(CARD / 'probe' / 'lanemap_probe.comp'),
                          plan=str(probe_plan), plan_sha256=sha(probe_plan),
                          cpu_check=pcc.strip().splitlines(),
                          status='AWAITING-GPU (1 dispatch, dump only)'),
               plan=dict(path=str(plan), sha256=sha(plan), rows_replaced=n, pass_ordinals=ords,
                         baseline_plan=str(base_plan), baseline_plan_sha256=base_sha),
               float_NC_checked=len(nc_cand),
               float_NC_parent=len(nc_parent),
               float_NC_identical=len(nc_cand) == len(nc_parent),
               cpu_check=dict(status='PASS', output=cc.strip().splitlines()),
               toolchain=dict(glslang=str(K.GLSLANG), validate=str(K.VALIDATE),
                              runner=str(K.RUNNER), args='-V --target-env vulkan1.3'))
    out['plan_token_changes'] = {
        'grid_changed': False,
        'grid_note': 'no PASS token other than the SPV path changed; grid stays 21 12 2 and the '
                     'header fields stay [32, 24576, 8, 32, 21, 12, 2]',
        'pass_rows_repointed': [dict(ordinal=o, spv='cards/n23-f05-shape/candidate.spv') for o in ords],
        'buffer_rows_changed': 0,
        'buffer_note': 'the 12 BUFFER rows that point at the permuted packs are delivered in '
                       'BUFFER-EDITS.json and are NOT applied to plan.txt (main-agent ruling)',
        'probe_plan_extras': ['PASS 19 SPV -> probe/lanemap_probe.spv',
                              'POLICY 50 1 0 233 0 -> POLICY 50 0 0 233 0 (the runner dumps every '
                              'non-resident buffer, so this is what makes swin_9_debug dumpable)'],
    }
    out['coverage_argument'] = {
        'grid_unchanged': True,
        'total_wg': 21 * 12 * 2,
        'tokens_per_wg': 64,
        'note': 'grid, ownership and the published address law are untouched, so the covered set '
                'equals the parent set exactly',
    }
    K.write_json(CARD / 'BUILD.json', out)
    log('BUILD.json written')


if __name__ == '__main__':
    main()
