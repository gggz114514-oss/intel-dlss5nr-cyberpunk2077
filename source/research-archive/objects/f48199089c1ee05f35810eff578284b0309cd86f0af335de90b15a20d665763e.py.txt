"""Campaign health check: chain identity, plan integrity, evidence completeness.

CPU only. Writes evidence/HEALTH.json and prints a summary.
"""
import json, sys
from pathlib import Path
sys.path.insert(0, str(Path(__file__).parent))
import COMMON as K
from audit_card import pass_lines

out = dict(checks={}, problems=[], notes=[])

# 1. chain
cur = K.read_json(K.C / 'CURRENT.json')
chain = cur['chain']
steps = []
for i, e in enumerate(chain):
    p = Path(e['path'])
    ok = p.exists() and K.sha(p) == e['sha256']
    n = len(pass_lines(p)) if p.exists() else 0
    steps.append(dict(index=i, label=e.get('label'), exists=p.exists(), sha_ok=ok, passes=n,
                      saving_ms=e.get('saving_ms'), kind=e.get('kind', 'adopt')))
    if not ok:
        out['problems'].append('chain step %d (%s): plan missing or sha mismatch' % (i, e.get('label')))
    if p.exists() and n != 232:
        out['problems'].append('chain step %d (%s): %d PASS lines, expected 232' % (i, e.get('label'), n))
out['checks']['chain'] = dict(len=len(chain), steps=steps, head=cur['best_plan'])

# 2. head rows vs handoff rows
head = Path(cur['best_plan']['path'])
hr = pass_lines(head)
rows = K.rows()
changed = []
for r, h in zip(rows, hr):
    if r['name'] != h['name']:
        out['problems'].append('head row %d name mismatch %s vs %s' % (r['ordinal'], r['name'], h['name']))
    spv = Path(h['spv'])
    if not spv.exists():
        out['problems'].append('head row %d: missing spv %s' % (r['ordinal'], spv))
        continue
    if str(r['spv']).replace('/', chr(92)) != str(h['spv']).replace('/', chr(92)):
        changed.append(r['ordinal'])
out['checks']['head'] = dict(path=str(head), sha256=K.sha(head), changed_rows=len(changed),
                             spvs_present=True)

# 3. measured cards: RESULTS.json + TIMING.json
cards = [d for d in (K.C / 'cards').iterdir() if d.is_dir()]
measured, accepted, blocked = [], [], []
for d in cards:
    if (d / 'TIMING.json').exists():
        measured.append(d.name)
    if (d / 'RESULTS.json').exists():
        r = K.read_json(d / 'RESULTS.json')
        st = str(r.get('status') or '')
        if st.startswith('ADOPTED'):
            accepted.append(d.name)
        if st.startswith('BLOCKED'):
            blocked.append(d.name)
out['checks']['cards'] = dict(total=len(cards), with_timing=len(measured),
                              accepted=len(accepted), blocked=len(blocked),
                              accepted_names=sorted(accepted), blocked_names=sorted(blocked))

# 4. evidence files
ev = K.C / 'evidence'
need = ['RECIPES.json', 'COVERAGE-232.json', 'NATIVE-INDEX.json', 'LESSONS.json',
        'PASS-TIME-RANKING.json', 'CHAIN-CONFIRMATION.json', 'COVERAGE-NOW.json', 'CHAIN-AUDIT.json']
missing = [n for n in need if not (ev / n).exists()]
if missing:
    out['problems'].append('missing evidence: %s' % missing)
out['checks']['evidence'] = dict(dir=str(ev), files=len(list(ev.glob('*'))), missing=missing)

# 5. environment
lock = K.LOCK
lock_state = K.read_json(lock).get('state') if lock.exists() else None
out['checks']['environment'] = dict(lock=str(lock), lock_state=lock_state,
                                   baseline_sha=K.sha(K.BASELINE), baseline_ok=K.sha(K.BASELINE) == K.BASELINE_SHA)
if not out['checks']['environment']['baseline_ok']:
    out['problems'].append('handoff baseline plan changed')

out['ok'] = not out['problems']
K.write_json(ev / 'HEALTH.json', out)
print(json.dumps({k: (v if k != 'chain' else {kk: vv for kk, vv in v.items() if kk != 'steps'})
                  for k, v in out['checks'].items()}, indent=2, ensure_ascii=False)[:2200])
print('problems:', out['problems'])
print('ok:', out['ok'])
