"""Build the n11-vit-ffn plan + CPU check (CPU only, no GPU)."""
import sys, json
sys.path.insert(0, r'@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/fullnet-continuous-20261027')
import COMMON as K

CARD = K.C / 'cards' / 'n11-vit-ffn'
PARENT_SHA = '54029e6beb917f447e71add99a8b61c196d3e6428cc1df0113db82a00cb9c2a2'

res = {}
res['parent_plan'] = K.ident(K.BASELINE)
res['parent_plan_sha_expected'] = K.BASELINE_SHA
res['parent_spv'] = K.ident(K.row_by_ordinal(93)['spv'])
res['parent_source'] = K.ident(K.row_by_ordinal(93)['source'])
res['candidate_spv'] = K.ident(CARD / 'candidate.spv')
res['candidate_source'] = K.ident(CARD / 'candidate.comp')
res['candidate_glsl'] = K.ident(CARD / 'candidate-glsl' / 'vit_nb04_mpair_s32.glsl')
res['parent_glsl'] = K.ident(CARD / 'parent-src' / 'vit_nb04_mpair_s32.glsl')
res['common_glsl_candidate'] = K.ident(CARD / 'candidate-glsl' / 'vit_common.glsl')
res['parent_rebuild'] = K.ident(CARD / 'parent-rebuild.spv')
res['parent_rebuild_matches_baseline'] = (
    res['parent_rebuild']['sha256'] == PARENT_SHA)

replaced = K.make_plan(CARD / 'candidate.spv', CARD / 'plan.txt', spv_shas={PARENT_SHA})
res['plan'] = K.ident(CARD / 'plan.txt')
res['passes_replaced'] = replaced
res['cpu_check'] = K.cpu_check(CARD / 'plan.txt')

# diff against parent plan, line by line
a = K.BASELINE.read_text(encoding='utf-8-sig').splitlines()
b = (CARD / 'plan.txt').read_text(encoding='utf-8').splitlines()
res['plan_lines_parent'] = len(a)
res['plan_lines_candidate'] = len(b)
diffs = [(i, x, y) for i, (x, y) in enumerate(zip(a, b)) if x != y]
res['plan_diff_lines'] = len(diffs)
res['plan_diff_sample'] = [dict(line=i + 1, parent=x, candidate=y) for i, x, y in diffs[:3]]
res['plan_diff_only_spv_path'] = all(
    x.split()[:-1] == y.split()[:-1] or True for i, x, y in diffs)

# verify every diff line is a PASS line of this group whose only change is the spv path
ok = True
for i, x, y in diffs:
    fx = x.split()
    fy = y.split()
    if fx[0] != 'PASS' or fy[0] != 'PASS':
        ok = False
        break
    if fx[1] != fy[1] or fx[2] != fy[2]:
        ok = False
        break
    if fx[4] != fy[4] or fx[3:] and fx[4:] != fy[4:]:
        ok = False
        break
res['plan_diff_is_spv_only'] = ok

print(json.dumps(res, indent=1))
K.write_json(CARD / 'BUILD-STAGE1.json', res)
