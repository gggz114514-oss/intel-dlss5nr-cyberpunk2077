import shlex, hashlib
def toks(line):
    try: return shlex.split(line)
    except ValueError: return line.split()

class Plan:
    def __init__(self, path):
        self.path = path; self.buffers = []; self.passes = {}; self.pass_order = []
        self.outputs = []; self.binds = collections.defaultdict(list)
        for ln in open(path, encoding="utf-8", errors="replace"):
            s = ln.strip()
            if not s: continue
            t = toks(s); k = t[0]
            if k == "PORT_NETWORK_1":
                self.header = [int(x) for x in t[1:5]]
            elif k == "BUFFER":
                self.buffers.append({"name": t[1], "bytes": int(t[2]), "file": t[3] if len(t) > 3 else "-"})
            elif k == "PASS":
                d = {"group": int(t[1]), "name": t[2], "spv": t[3], "args": tuple(t[4:]), "ordinal": len(self.pass_order)}
                self.passes.setdefault(t[2], []).append(d); self.pass_order.append(d)
            elif k == "BIND":
                self.binds[int(t[1])].append((int(t[2]), int(t[3]), int(t[4])))
            elif k == "OUTPUT":
                self.outputs.append({"buffer_index": int(t[1]), "format": t[2], "file": t[3]})
    def bname(self, i):
        return self.buffers[i]["name"] if 0 <= i < len(self.buffers) else "?%d" % i
    def bsize(self, i):
        return self.buffers[i]["bytes"] if 0 <= i < len(self.buffers) else -1
    def sha(self):
        return hashlib.sha256(open(self.path, "rb").read()).hexdigest()
