def grids_from_plan(plan_path):
    out = {}
    for line in Path(plan_path).read_text(encoding='utf-8-sig').split('\n'):
        if not line.startswith('PASS '):
            continue
        p = line.split(' ')
        out.setdefault(p[3].strip('"').replace('\\', '/'), set()).add((p[-3], p[-2], p[-1]))
    return {k: sorted(v) for k, v in out.items()}


def main():
    plan_path, plan_sha, _ = K.current_best()
    grids = grids_from_plan(plan_path)
    rows_by_spv = {}
    for r in K.rows():
        rows_by_spv.setdefault(r['spv'].strip('"').replace('\\', '/'), set()).add(
            tuple(r['grid']) if r.get('grid') else None)
    uniq = K.unique_shaders()
    rows = []
    for u in uniq:
        src = Path(u['source'])
        txt = src.read_text(encoding='utf-8-sig', errors='replace') if src.exists() else ''
        hits = [ln.strip() for ln in txt.splitlines() if 'gl_WorkGroupID' in ln]
        key = (u['family'], u['name'])
        cls, reason = CLASS.get(key, ('unclassified', ''))
        g = grids.get(u['spv'].strip('"').replace('\\', '/'), [])
        if not g:
            g = sorted(x for x in rows_by_spv.get(u['spv'].strip('"').replace('\\', '/'), set()) if x)
        rows.append({
            'family': u['family'], 'name': u['name'],
            'spv_sha256': u['spv_sha256'], 'ordinals': u['ordinals'],
            'passes': len(u['ordinals']),
            'segments_covered': len(u['ordinals']),
            'grids': g,
            'grid_y_values': sorted({int(x[1]) for x in g}) if g else [],
            'wgrid_lines': hits[:8],
            'class': cls, 'transposable_without_content_change': cls in ('token_x_group_y', 'mn_2d'),
            'reason': reason,
        })
    by_class = {}
    for r in rows:
        by_class.setdefault(r['class'], []).append(r['family'] + '/' + r['name'])

    summary = {
        'n03d': 'short census: which of the 39 unique shaders could host the same *kind* of '
                'workgroup reorder (transpose of two independent dispatch axes) without changing '
                'workgroup content',
        'plan': K.ident(plan_path),
        'unique_shaders': len(rows), 'passes': 232,
        'classes': {k: {'shaders': v, 'count': len(v)} for k, v in sorted(by_class.items())},
        'transposable_shaders': [r['family'] + '/' + r['name'] for r in rows
                                 if r['transposable_without_content_change']],
        'f08_l0': {
            'grid': [r['grids'] for r in rows if r['name'] == 'split16_b23_l0'],
            'class': [r['class'] for r in rows if r['name'] == 'split16_b23_l0'],
            'note': 'F08 has a single width (C512, 16 PASS); its sibling layers l1/l2/l3 are '
                    'separate families (F09/F10/F11) with their own grids (66,4,1) / (6,3..4,16) / (33,8,1)',
        },
        'rows': rows,
    }
    K.write_json(CARD / 'census.json', summary)
    print(json.dumps({k: summary[k] for k in ('unique_shaders', 'classes', 'transposable_shaders', 'f08_l0')}, indent=1))
    print('grids per family:')
    for r in rows:
        print('  %-4s %-28s %-40s %s' % (r['family'], r['name'], r['grids'], r['class']))


if __name__ == '__main__':
    main()
