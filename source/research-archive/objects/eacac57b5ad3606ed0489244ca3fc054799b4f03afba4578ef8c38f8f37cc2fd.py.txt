"""n37-f10-arena-shrink: negative control for _addr_audit.py.

n36 learned the hard way that "the gate passes" means nothing until it has been shown to
fail on the bug it was written for.  This script doctors the candidate source in five ways
that keep every *structural* check green (Workgroup bytes are still legal, spv_validate
would still pass, the parent rebuild is untouched) but break the address space, and asserts
that _addr_audit.py reports FAIL for each of them:

  N1  QKV_OFF left at 4096 (the handoff keeps the old base while the arena shrinks)
  N2  the STORE_QKV V literal left at the old plane base (8192) while VBASE moved to 4096
      -- this is exactly n36's bug shape: a plane literal that is not the section stride
  N3  one staging window base left at 4096 but the init/toggle disagreeing (STAGE_OFF=8192)
  N4  the arena declaration left at 10240 (slack: the tightness rule must fail)
  N5  QKV_OFF = 2048, which keeps every plane inside the arena but makes V overlap Q

Writes build/audit-negcontrol.json.
"""
import json
import pathlib
import sys

CARD = pathlib.Path(__file__).resolve().parent
sys.path.insert(0, str(CARD))
import _addr_audit as A               # noqa: E402

SRC = CARD / 'candidate.comp'
SPV = CARD / 'candidate.spv'

CASES = [
    ('N1 QKV_OFF left at 4096',
     [('#define QKV_OFF 0u', '#define QKV_OFF 4096u')]),
    ('N2 STORE_QKV V literal left at the old plane',
     [('coopMatStore(value,smem,QKV_OFF+4096u', 'coopMatStore(value,smem,QKV_OFF+8192u')]),
    ('N3 staging window base left at 8192',
     [('#define STAGE_OFF 4096u', '#define STAGE_OFF 8192u')]),
    ('N4 arena left at 10240 (slack)',
     [('shared float16_t smem[8192];', 'shared float16_t smem[10240];')]),
    ('N5 QKV_OFF 2048 (planes overlap)',
     [('#define QKV_OFF 0u', '#define QKV_OFF 2048u')]),
]


def main():
    base = SRC.read_text(encoding='utf-8')
    out = {'baseline': {'verdict': A.audit('candidate')['verdict']}, 'cases': []}
    for label, edits in CASES:
        text = base
        for old, new in edits:
            assert text.count(old) == 1, ('anchor', label, old, text.count(old))
            text = text.replace(old, new, 1)
        tmp = CARD / 'build' / ('negcontrol-%s.comp' % label.split()[0])
        tmp.write_text(text, encoding='utf-8')
        rec = A.audit(label.split()[0], source=tmp, spv=SPV, model_key='candidate')
        out['cases'].append(dict(case=label, edits=[list(e) for e in edits],
                                 verdict=rec['verdict'],
                                 resolved_planes=rec['planes'],
                                 windows=rec['windows'],
                                 max_index=rec['max_index'],
                                 arena_halves=rec['arena_halves'],
                                 violation_rules=sorted({v['rule'] for v in rec['violations']}),
                                 violation_whys=[v['why'] for v in rec['violations']][:6]))
        print('%-42s %s  rules=%s' % (label, rec['verdict'],
                                      sorted({v['rule'] for v in rec['violations']})))
        tmp.unlink()
    ok = (out['baseline']['verdict'] == 'PASS'
          and all(c['verdict'] == 'FAIL' for c in out['cases']))
    out['verdict'] = 'PASS' if ok else 'FAIL'
    out['reading'] = ('the gate passes on the candidate and fails on every doctored variant, '
                      'so it is sensitive to the exact bug class it exists for')
    (CARD / 'build/audit-negcontrol.json').write_text(json.dumps(out, indent=2),
                                                      encoding='utf-8')
    print('NEGATIVE CONTROL', out['verdict'])
    return 0 if ok else 1


if __name__ == '__main__':
    sys.exit(main())
