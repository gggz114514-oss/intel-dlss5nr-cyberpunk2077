import json, os, sys

ROOT = r"@NR_WORKSPACE@\work\codex-astra-fp16-int8-audit-20261024\fullnet-continuous-20261027\evidence"

def show(f, keys_only=False):
    p = os.path.join(ROOT, f)
    d = json.load(open(p, encoding="utf-8"))
    print("===", f, type(d).__name__, "len", len(d))
    if isinstance(d, dict):
        for k in list(d)[:20]:
            v = d[k]
            if isinstance(v, (list, dict)):
                s = json.dumps(v, ensure_ascii=False)
                print("  ", k, type(v).__name__, len(v), "sample:", s[:700])
            else:
                print("  ", k, "=", repr(v)[:300])

if __name__ == "__main__":
    for f in sys.argv[1:]:
        show(f)
