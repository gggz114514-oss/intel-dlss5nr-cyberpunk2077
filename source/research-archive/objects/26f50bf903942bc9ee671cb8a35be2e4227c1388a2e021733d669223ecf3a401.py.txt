#!/usr/bin/env python3
"""Ad-hoc: opcode histograms and instruction contexts for the R03 delta rows."""
import sys
from collections import Counter
from pathlib import Path

sys.path.insert(0, str(Path(__file__).resolve().parent))
import spv_read as S  # noqa: E402

CARD = Path("@NR_WORKSPACE@/work/dsh-xmx-handoff-validation-20261024/candidates/r03-ioport4")
parent = S.Module(CARD / "spv/parent_repro.spv")
half = S.Module(CARD / "spv/r03_ioport4_i8.spv")

hp = Counter(op for op, a, wc in parent.ins)
hc = Counter(op for op, a, wc in half.ins)
for op in sorted(set(hp) | set(hc)):
    if hp.get(op, 0) != hc.get(op, 0):
        print("%-32s p=%3d c=%3d d=%+d" % (parent.name(op), hp.get(op, 0), hc.get(op, 0),
                                           hc.get(op, 0) - hp.get(op, 0)))
print()
for op in (142, 133, 4460, 245, 246):
    print("opcode", op, parent.name(op), "parent", hp.get(op, 0), "cand", hc.get(op, 0))
    for label, m in (("p", parent), ("c", half)):
        for i, (o, a, wc) in enumerate(m.ins):
            if o == op:
                t = m.types.get(a[0]) if a else None
                print("   %s idx=%d wc=%d type=%s ops=%s" % (label, i, wc, t, a[:4]))
                break
