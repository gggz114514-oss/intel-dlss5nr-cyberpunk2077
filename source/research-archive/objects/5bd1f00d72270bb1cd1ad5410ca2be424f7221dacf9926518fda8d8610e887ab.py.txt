from pathlib import Path
import json,hashlib,statistics,math,sys
D=Path(__file__).resolve().parent
def rd(p):return json.loads(p.read_text(encoding='utf-8-sig'))
def check(j):
 d=Path(j['runner_output']);r=rd(d/'port_network_receipt.json')
 assert hashlib.sha256(Path(j['plan']['path']).read_bytes()).hexdigest()==j['plan']['sha256']
 assert hashlib.sha256((d/'frame_out.bin').read_bytes()).hexdigest()==j['expected_frame_out']
 assert r['normal_close'] and r['warm']==1 and r['reps']==30 and r['feedback'] and r['gpu_passes']==232 and r['nan']==r['inf']==0
 return d,r
def ci(xs):
 m=statistics.mean(xs);w=4.302652729911275*statistics.stdev(xs)/math.sqrt(3) if len(xs)==3 else None
 return dict(values=xs,mean=m,ci95=None if w is None else [m-w,m+w])
mode=sys.argv[1];vals={};stats={};definitions={}
if mode=='timestamps':
 for j in rd(D/'LUNA-TIMESTAMPS.json')['jobs']:
  d,r=check(j);assert r['per_pass_timing'] and len(r['per_pass_gpu'])==232
  for i,x in enumerate(r['per_pass_gpu']):assert x['index']==i and len(x['samples_ms'])==30
  f20=statistics.mean(r['per_pass_gpu'][230]['samples_ms']);total=statistics.mean(r['gpu_samples_ms'])
  vals[j['pair'],j['candidate']]=dict(F20=f20,network=total,other_intervals=sum(statistics.mean(x['samples_ms']) for x in r['per_pass_gpu'])-f20)
 comparisons={}
 for ca,pa in [('dualweight','parent'),('fouracc','parent'),('fouracc','dualweight')]:
  comparisons[ca+'-minus-'+pa]={k:ci([vals[i,ca][k]-vals[i,pa][k] for i in range(3)]) for k in ['F20','network','other_intervals']}
 out=dict(process_means={str(k):v for k,v in vals.items()},comparisons=comparisons,all270_samples_retained=True)
 (D/'TIMESTAMP-REVIEW.json').write_text(json.dumps(out,indent=2));print(json.dumps(comparisons,indent=2))
else:
 jobs=rd(D/'LUNA-PILOT.json')['jobs']+(rd(D/'LUNA-REST.json')['jobs'] if mode=='counters' else [])
 for j in jobs:
  d,r=check(j);c=rd(d/'hardware-counters.json');assert c['hardware_passes']==1 and len(c['frames'])==30 and c['feedback']
  names=[x['name'] for x in c['metrics']];assert set(names)==set(j['expected_metrics']);definitions[j['metric_group']]=c['metrics']
  for f in c['frames']:assert len(f['passes'])==232 and f['passes'][230]['index']==230
  mat=[f['passes'][230]['values'] for f in c['frames']];assert all(len(x)==len(names) and all(math.isfinite(v) for v in x) for x in mat)
  vals[j['metric_group'],j['pair'],j['candidate']]={name:statistics.mean(x[k] for x in mat) for k,name in enumerate(names)}
  stats[j['id']]=next(x for x in rd(d/'pipeline-statistics.json')['pipelines'] if x['index']==38)
 out=dict(process_means={str(k):v for k,v in vals.items()},pipeline38=stats,definitions=definitions,all_frames_gold=True,hardware_passes_one=True)
 if mode=='counters':
  comparisons={}
  for group in ['traffic','occupancy','stalls']:
   for name in vals[group,0,'parent']:
    entry={a:[vals[group,i,a][name] for i in range(3)] for a in ['parent','dualweight','fouracc']}
    for ca,pa in [('dualweight','parent'),('fouracc','parent'),('fouracc','dualweight')]:entry[ca+'-minus-'+pa]=ci([entry[ca][i]-entry[pa][i] for i in range(3)])
    comparisons[group+':'+name]=entry
  out['comparisons']=comparisons
 (D/('COUNTER-REVIEW.json' if mode=='counters' else 'PILOT-REVIEW.json')).write_text(json.dumps(out,indent=2))
 print('Verified jobs',len(jobs),'all gold, single hardware pass; pipeline stats saved')
 for (g,i,a),v in vals.items():
  if i==0:print(g,a,{k:x for k,x in v.items() if k in ['GpuTime','XVE_STALL','XVE_ACTIVE','XVE_THREAD_OCCUPANCY','XVE_STALL_SBID','XVE_STALL_ALUWR','XVE_STALL_BARRIER']})
