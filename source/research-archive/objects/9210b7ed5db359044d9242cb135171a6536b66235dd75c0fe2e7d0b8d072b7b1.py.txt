"""Consistent local-block error census from the existing controlled arms.

Reads the arm outputs as fp16 (the dump/plan convention); rejects size
mismatches instead of guessing.  GPU vs CPU origin is recorded per row.
"""
from __future__ import annotations

import json
from pathlib import Path

import numpy as np

D = Path(r"@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/int8-quality-first-20261029/real-int8-24")
OUT = Path(r"@NR_WORKSPACE@/work/dsh-fullnet-static-20261029/reports")


def metrics(arm: Path, ref: Path) -> dict:
    if not arm.is_file() or not ref.is_file():
        return {"status": "MISSING", "arm": str(arm), "ref": str(ref)}
    a = np.fromfile(arm, dtype="<f2").astype(np.float64)
    b = np.fromfile(ref, dtype="<f2").astype(np.float64)
    if a.size != b.size:
        return {"status": "SIZE_MISMATCH", "arm_elements": int(a.size), "ref_elements": int(b.size)}
    e = a - b
    return {
        "status": "OK",
        "n": int(a.size),
        "rel_l2": float(np.linalg.norm(e) / np.linalg.norm(b)),
        "rmse": float(np.sqrt(np.mean(e * e))),
        "max_abs": float(np.abs(e).max()),
        "ref_rms": float(np.sqrt(np.mean(b * b))),
    }


def main() -> None:
    rows = []
    # Baseline C32 body1-3, then the equalised candidate.
    for block in (1, 2, 3):
        ref = D / f"dump-fp16/dump_block{block}_out.bin"
        for arm_name, arm in [
            ("baseline-int8", D / f"local-resident-v2/local-b{block}/replayed-network.bin"),
            ("c32-equalized", D / f"c32-equalized/local-b{block}/replayed-network.bin"),
        ]:
            m = metrics(arm, ref)
            rows.append({"family": "C32", "block": block, "arm": arm_name, "origin": "GPU local arm", **m})
    # C32 body4 reported by the previous card (the dump pair is not size-compatible).
    q4 = json.loads((D / "c32-equalized-b4/QUALITY.json").read_text())
    for r in q4:
        if r["arm"] == "local-b4":
            rows.append({"family": "C32", "block": 4, "arm": "c32-minmax", "origin": "GPU local arm (QUALITY.json)",
                         "status": "OK", "rel_l2": r["rel_l2"], "psnr": r["psnr"], "note": "body4 X equalisation reported 12.699%->11.397% in the card; this arm is the minmax baseline"})
    # C64 baseline vs static family.
    for block in (5, 6, 7):
        ref = D / f"dump-fp16/dump_block{block}_out.bin"
        for arm_name, arm in [
            ("baseline-int8", D / f"family-isolation/run-block{block}/replayed-network.bin"),
            ("c64-static", D / f"c64-family-static/local-b{block}/replayed-network.bin"),
        ]:
            m = metrics(arm, ref)
            rows.append({"family": "C64", "block": block, "arm": arm_name, "origin": "GPU local arm", **m})
    # C256 and ViT baseline local arms.
    for block, family, arm in [
        (15, "C256", D / "family-isolation/run-block15/replayed-network.bin"),
        (31, "ViT", D / "family-isolation/run-block31/replayed-network.bin"),
        (38, "ViT", D / "family-isolation/run-block38/replayed-network.bin"),
    ]:
        ref = D / f"dump-fp16/dump_block{block}_out.bin"
        m = metrics(arm, ref)
        rows.append({"family": family, "block": block, "arm": "baseline-int8", "origin": "GPU local arm", **m})
    # C128 b9 CPU baseline from S05 (not a GPU local arm).
    s05 = Path(r"@NR_WORKSPACE@/work/dsh-fullnet-static-20261029/S05-c128/evidence/BASELINE-CPU.json")
    if s05.is_file():
        b = json.loads(s05.read_text())
        rows.append({"family": "C128", "block": 9, "arm": "baseline-int8", "origin": "CPU reconstruction (S05)",
                     "status": "CPU_ONLY", "rel_l2": b["mean_old_rel_l2"],
                     "note": "common FP16 teacher input, qX residual, D=1; GPU oracle not yet aligned"})
    (OUT / "LOCAL-ERROR-CENSUS.json").write_text(json.dumps({
        "scope": "controlled local arms, FP16 teacher input, block output compared; not whole-image quality",
        "rows": rows,
    }, indent=2), encoding="utf-8")
    for r in rows:
        if r["status"] == "OK":
            print(f"{r['family']:4s} b{r['block']:<2d} {r['arm']:16s} {r['origin']:32s} rel_l2={r.get('rel_l2', float('nan')):.4f}")
        else:
            print(f"{r['family']:4s} b{r['block']:<2d} {r['arm']:16s} {r['status']} {r.get('rel_l2')}")


if __name__ == "__main__":
    main()
