"""Write the final timing report for the merged backend."""
import json, sys
from pathlib import Path
sys.path.insert(0, str(Path(__file__).parent))
import COMMON as K

cur = K.read_json(K.C / 'CURRENT.json')
tim = K.read_json(K.C / 'queue' / 'final-confirm' / 'TIMING.json')
c = tim['comparisons'][0]
pairs = c['group_means']
head_ms = sum(p['head'] for p in pairs) / len(pairs)
base_ms = sum(p['handoff'] for p in pairs) / len(pairs)
adopt = [e for e in cur['chain'] if not e.get('kind') and e.get('saving_ms') is not None]
rows = K.read_json(K.C / 'backend-merged' / 'MANIFEST.json')

lines = ['# 融合后端的耗时报告', '',
         '## 结论', '',
         '| 项 | 值 |', '|---|---|',
         '| handoff 基线（绝对） | **%.3f ms/帧** |' % base_ms,
         '| 融合后端（绝对） | **%.3f ms/帧** |' % head_ms,
         '| 同批配对差值 | **+%.3f ms** CI95[%.3f, %.3f]，6/6 组同向 |' % (
             c['saving_ms'], c['CI95_ms'][0], c['CI95_ms'][1]),
         '| 相对提升 | **%.2f%%** |' % (100 * c['saving_ms'] / base_ms),
         '| 链上采纳步数 | %d |' % len(adopt),
         '| 相对 handoff 改变的 PASS 行 | %d（族：%s） |' % (
             rows['changed_rows'], '、'.join('%s %d' % (f, n) for f, n in rows['changed_per_family'].items())),
         '', '计时口径：同批 6 组交替（每组先跑完两臂各 30 样本），全部样本保留，不剔尖峰、不补跑；'
         '差值为组均值的配对 t 检验（df=5）。', '',
         '## 逐步增量（各自同批、各自基线）', '',
         '| # | 步骤 | 同批增量 | 机制 |', '|---|---|---|---|']
mech = {
    'n05a-f05-c256-mreuse': '一块权重喂两个 M8 片段',
    'n06-f06-output': '权重对跨两 M 块复用',
    'n09b-f09-sx-stage': '删 sx 的 global→shared→读回往返 + 1 道屏障',
    'pkg-vit-16pass': 'live 判定外提 + 块对齐 clamp（16 PASS 包）',
    'n06b-f06-c64': 'A tile 2×2→1×4 块化',
    'n09c-f09-b-share': '一份 B tile 服务两个 8-token 组',
    'n18-f01-entry': '删 clamp 后多余的镜像取模',
    'n10-f10-next': 'kc 地址递推 + arena 合并删 8 道屏障/WG',
    'n27-f07': 'staging 行连续化，8×32B 合成 128 条 256B 读',
    'n29-f01-alu-chain': '删 live 路径上 2 条整数除法（math.inv 21→19）',
}
for i, e in enumerate(adopt, 1):
    lines.append('| %d | %s | +%.6f ms | %s |' % (i, e['label'], e['saving_ms'],
                                                   mech.get(e['label'], '')))
lines += ['',
          '**注意**：逐步增量之和（%.6f ms）**不能当作端到端收益**。实测同一配置在两批里分别测得 '
          '%.4f 与 %.4f ms（相差约 8%%），这是批间漂移，不是配置变化；端到端以最新一批的配对差值为准，'
          '跨批只报区间。' % (cur['cumulative_ms_in_chain'], 1.005080, c['saving_ms']),
          '', '## 每批的端到端测量（同一配置的重复测量）', '',
          '| 批次 | handoff | 链头 | 差值 | 95%% CI |', '|---|---|---|---|---|',
          '| confirm3b（链头 n29） | — | — | +1.005080 | [0.9811, 1.0291] |',
          '| final-confirm（同一链头） | %.3f | %.3f | +%.4f | [%.4f, %.4f] |' % (
              base_ms, head_ms, c['saving_ms'], c['CI95_ms'][0], c['CI95_ms'][1]),
          '', '两批的中位数约 **0.97 ms**，所以对外报「**约 0.93–1.01 ms，约 5.2%**」。', '',
          '## 未采纳的尝试（同批实测，供后续参考）', '',
          '| 项 | 结果 | 原因 |', '|---|---|---|',
          '| F20 的 4 张删消息卡 | 全部 ns | 该核 SENDWR 只有 0.029 ms，消息不是预算 |',
          '| F20 tail owner 对齐（删一道真屏障） | −0.1809 ms | lane 内 ILP 4→1（L25） |',
          '| F20 arena 窗轮转（删 WAR 屏障） | −0.0150 ms | 屏障代价低于计数器分辨率 |',
          '| F05 36 行延迟隐藏重排 | A 类未落地 | 驱动调度器own基本块内顺序（L26） |',
          '| 冗余倒数额删除（F02/F19） | 前提证伪 | 24 条 = 24 个不同定值，非冗余（L27） |',
          '| F01 `sel` 恒等删除（144 条） | −0.0396 ms | `sel` 是廉价操作（L24） |']
out = K.C / 'FINAL-REPORT.md'
out.write_text('\n'.join(lines) + '\n', encoding='utf-8')
print('wrote', out)
print('base %.3f  head %.3f  saving %.4f  CI [%.4f, %.4f]' % (base_ms, head_ms, c['saving_ms'],
                                                             c['CI95_ms'][0], c['CI95_ms'][1]))
