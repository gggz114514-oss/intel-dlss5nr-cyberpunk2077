"""Generate SOURCE-DIFF.patch (parent header -> candidate header) and prove it
round-trips: applying the hunks to the parent lines must reproduce the candidate
bytes exactly.  CPU only.
"""
import difflib, hashlib
from pathlib import Path

CARD = Path(__file__).resolve().parent.parent
PARENT = CARD / 'parent-src/vit_nb04_mpair.glsl'
CAND = CARD / 'candidate-glsl/vit_nb04_mpair.glsl'
OUT = CARD / 'SOURCE-DIFF.patch'

a = PARENT.read_text(encoding='utf-8').splitlines(keepends=True)
b = CAND.read_text(encoding='utf-8').splitlines(keepends=True)
fa = ('a/candidates/codex-fp16-continuation-20261021/A/u03-f18-v2/source/vit_nb04_mpair.glsl')
fb = ('b/cards/n11b-vit-projection-hoist/candidate-glsl/vit_nb04_mpair.glsl')
diff = list(difflib.unified_diff(a, b, fromfile=fa, tofile=fb, n=4))
text = ''.join(diff)
OUT.write_text(text, encoding='utf-8')

# ---------------------------------------------------------------- apply + verify
def apply(hunks, src):
    out, i = [], 0
    for h in hunks:
        m = h[0]
        assert m.startswith('@@'), m
        span = m.split('@@')[1].strip().split()
        a0, alen = span[0][1:].split(',')
        a0 = int(a0) - 1
        out.extend(src[i:a0])
        i = a0
        for line in h[1:]:
            if line.startswith(' '):
                out.append(line[1:]); i += 1
            elif line.startswith('-'):
                i += 1
            elif line.startswith('+'):
                out.append(line[1:])

    out.extend(src[i:])
    return out


hunks, cur = [], None
for line in diff[2:]:
    if line.startswith('@@'):
        cur = [line]
        hunks.append(cur)
    else:
        cur.append(line)
got = ''.join(apply(hunks, a))
ok = got == ''.join(b)
print('patch hunks:', len(hunks), 'roundtrip identical:', ok)
print('patch bytes:', OUT.stat().st_size, 'sha256:', hashlib.sha256(OUT.read_bytes()).hexdigest())
print('parent sha:', hashlib.sha256(PARENT.read_bytes()).hexdigest())
print('cand   sha:', hashlib.sha256(CAND.read_bytes()).hexdigest())
print('common identical:', hashlib.sha256((CARD / 'candidate-glsl/vit_common.glsl').read_bytes()).hexdigest() ==
      hashlib.sha256((CARD / 'parent-src/vit_common.glsl').read_bytes()).hexdigest())
assert ok
