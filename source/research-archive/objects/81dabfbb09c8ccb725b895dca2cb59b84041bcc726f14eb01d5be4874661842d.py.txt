def tiled(token, c, C):
    return ((token // 16) * (C // 16) + c // 16) * 256 + (token % 16) * 16 + c % 16


# ---------------------------------------------------------------- SPIR-V diff
def normalized(m):
    """Opcode + operand stream with result ids replaced by 'id' and constant ids
    replaced by their literal value, so renumbering cannot masquerade as a change."""
    const = {}
    for op, ops, _, _ in m.instructions:
        if op == 43:                       # OpConstant type result value
            const[ops[1]] = ops[2]
        elif op == 44:                     # OpConstantComposite type result ...
            const[ops[1]] = tuple(const.get(x, ('raw', x)) for x in ops[2:])
    results = set()
    skip_result_first = {2, 3, 4, 5, 6, 8, 14, 15, 16, 17, 71, 72, 73, 74, 75, 77}
    for op, ops, _, _ in m.instructions:
        if not ops or op in skip_result_first:
            continue
        if K.ST.result_id_of(op, ops) is not None:
            results.add(ops[0] if op in K.ST.RESULT_AT_ZERO_OPS else ops[1])
    out = []
    for op, ops, _, _ in m.instructions:
        if op == 43:
            out.append((op, ('const', ops[1] if False else None, ops[2])))
            continue
        toks = []
        for o in ops:
            if o in const and not isinstance(const[o], tuple):
                toks.append(('c', const[o]))
            elif o in results:
                toks.append('id')
            else:
                toks.append(('lit', o))
        out.append((op, tuple(toks)))
    return out, const


def instr_diff(pa, pb):
    a = K.ST.Module(Path(pa))
    b = K.ST.Module(Path(pb))
    assert len(a.instructions) == len(b.instructions), (len(a.instructions), len(b.instructions))
    na, ca = normalized(a)
    nb, cb = normalized(b)
    assert [x[0] for x in na] == [x[0] for x in nb], 'opcode sequence differs'
    diffs = []
    for idx, ((opa, ta), (opb, tb)) in enumerate(zip(na, nb)):
        if ta != tb:
            diffs.append({'index': idx, 'opcode': opa, 'name': NAMES.get(opa, 'op%d' % opa),
                          'parent_normalized': list(ta), 'candidate_normalized': list(tb)})
    # which literal constants were introduced / removed
    changed_consts = []
    for idx, d in enumerate(diffs):
        pass
    return diffs, len(a.instructions), len(a.words)


# ---------------------------------------------------------------- mappings
def parent_sg(wgx, wgy, sg):
    token = wgx * 16 + (sg % 2) * 8
    return token, wgy * 4 + sg // 2


def cand_sg(wgx, wgy, sg):
    token = wgx * 32 + (sg % 4) * 8
    return token, wgy * 2 + sg // 4


def enumerate_map(fn, grid):
    gx, gy, gz = grid
    out = {}
    for y in range(gy):
        for x in range(gx):
            for sg in range(SG):
                token, g = fn(x, y, sg)
                out[(x, y, sg)] = (token, g)
    return out


def check_bijection(mapping, grid, label):
    pairs = list(mapping.values())
    uniq = set(pairs)
    expect = {(t, g) for t in range(0, T, 8) for g in range(G)}
    missing = sorted(expect - uniq)
    extra = sorted(uniq - expect)
    dup = len(pairs) - len(uniq)
    return {'label': label, 'grid': list(grid),
            'workgroups': grid[0] * grid[1] * grid[2],
            'sg_slots': len(pairs),
            'distinct_token8_group_pairs': len(uniq),
            'expected_pairs': len(expect),
            'duplicate_assignments': dup,
            'missing_pairs': [list(m) for m in missing],
            'extra_pairs': [list(m) for m in extra],
            'max_token': max(t for t, g in pairs),
            'min_token': min(t for t, g in pairs),
            'groups_seen': sorted({g for t, g in pairs}),
            'bijection_ok': not missing and not extra and dup == 0}


def wg_row(fn, x, y):
    return [(sg,) + fn(x, y, sg) for sg in range(SG)]


def wg_weight_slabs(fn, x, y):
    groups = sorted({fn(x, y, sg)[1] for sg in range(SG)})
    return groups


# half-element offsets of the three weight stages for one group
def slab_offsets(g, push):
    p2x, p2y, p2z = push
    stage1 = [(p2x + g * 32768 + k * 16 + off) for k in range(32) for off in (0, 8192, 16384, 24576)]
    # covered element ranges, not every load base
    return {'stage1_halfs': [p2x + g * 32768, p2x + (g + 1) * 32768],
            'expand_halfs': [p2y + g * 16384, p2y + (g + 1) * 16384],
            'contract_halfs': [p2z + g * 16384, p2z + (g + 1) * 16384]}
