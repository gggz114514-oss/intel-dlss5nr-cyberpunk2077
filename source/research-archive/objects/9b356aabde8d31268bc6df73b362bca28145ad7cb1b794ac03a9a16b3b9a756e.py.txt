POOL_ELEMS = 8 * K                # 256 pooled code points per workgroup


def sha(p):
    return hashlib.sha256(Path(p).read_bytes()).hexdigest()


def ident(p):
    p = Path(p)
    return {"path": p.as_posix(), "bytes": p.stat().st_size, "sha256": sha(p)}

        "coop_stores": WG * 2,
        "mma_instructions": WG * 4,      # 2 N16 outputs x 2 K16 steps
        "weight_bytes_read": WG * 4 * 512,   # 4 K16/N16 tiles of 512 B
        "scale_bytes_read": 0,
        "barriers": WG * 1,
        "weight_tile_bytes": 2048,
    }
    cand = {
        "shared_bytes_per_wg": 256,
        "pool_int8_stores": WG * POOL_ELEMS,
        "pool_int8_store_bytes": WG * POOL_ELEMS,
        "coop_loads": WG * 5,            # A x1 + B x2 + scale fragment x2
        "coop_stores": WG * 2,
        "mma_instructions": WG * 2,      # 2 N16 outputs, one K32 step
        "weight_bytes_read": WG * 2 * 512,   # 2 K32/N16 tiles of 512 B
        "scale_bytes_read": WG * 2 * 128 * 4,
        "barriers": WG * 1,
        "weight_tile_bytes": 1024,
        "quantization_code_points": WG * POOL_ELEMS,
        "int32_to_f32_components": WG * 2 * 4,
        "dequant_multiplies": WG * 2 * 4,
        "f32_to_f16_narrows": WG * 2,
        "extra_pack_files": 2,
    }
    return {
        "workgroups_per_dispatch": WG,
        "pooled_code_points_per_dispatch": WG * POOL_ELEMS,
        "note": ("source-level counts from the frozen plan grid and the two modules; the pooled "
                 "value is recomputed once per output-channel half (gy=2), exactly as the parent "
                 "does, so the pool count is 2x the 2,064,384-element tensor in both arms"),
        "parent": parent, "candidate": cand,
        "delta": {
            "shared_bytes_per_wg": cand["shared_bytes_per_wg"] - parent["shared_bytes_per_wg"],
            "coop_loads": cand["coop_loads"] - parent["coop_loads"],
            "mma_instructions": cand["mma_instructions"] - parent["mma_instructions"],
            "weight_bytes_read": cand["weight_bytes_read"] - parent["weight_bytes_read"],
            "scale_bytes_read": cand["scale_bytes_read"],
            "shared_write_bytes": cand["pool_int8_store_bytes"] - parent["pool_half_store_bytes"],
            "barriers": cand["barriers"] - parent["barriers"],
            "coop_stores": cand["coop_stores"] - parent["coop_stores"],
        },
        "owner_and_live_range": {
            "a_i8[256]": ("workgroup shared; written by all 32 lanes in the pool loop, read by the "
                          "single A fragment load after the one barrier; live range = pool loop -> "
                          "A load, 1 barrier, identical to the parent's a_half"),
            "aa / bb / bb1": "int8 fragments, live from their loads to the two MMAs",
            "ci0 / ci1": "int32 accumulators, live until their own dequant block (two disjoint blocks)",
            "st0 / st1": "one f32 scale fragment per block, dies in the dequant multiply",
            "output": "one direct fragment store per N16 block, unchanged addressing",
        },
        "not_claimed": ("not measured DRAM bytes, not native SLM/GRF, not a millisecond estimate; "
                        "the 5 KB scale table and the 2 KB weight pack are constant and cache-resident"),
