"""Apply a card's BUFFER-EDITS.json to a plan, verifying identities.

Usage: python apply_buffer_edits.py <card-dir> <plan-in> <plan-out>
"""
import sys
from pathlib import Path
sys.path.insert(0, str(Path(__file__).parent))
import COMMON as K

card = Path(sys.argv[1]).resolve()
plan_in = Path(sys.argv[2]).resolve()
plan_out = Path(sys.argv[3]).resolve()
edits = K.read_json(card / 'BUFFER-EDITS.json')
recs = edits.get('edits') or edits.get('buffer_edits') or []
assert recs, 'no edits found in BUFFER-EDITS.json (keys: %s)' % list(edits)[:8]
raw = plan_in.read_text(encoding='utf-8-sig')
lines = raw.splitlines()
applied = []
for e in recs:
    idx = e['buffer_line']
    assert lines[idx] == e['original_line'], ('original line mismatch at %d' % idx, lines[idx][:90], e['original_line'][:90])
    nf = Path(e['new_file'])
    assert nf.is_absolute(), ('new_file must be absolute', nf)
    assert nf.exists(), ('new_file missing', nf)
    got = K.sha(nf)
    assert got == e['new_sha256'], ('sha mismatch', nf, got, e['new_sha256'])
    assert nf.stat().st_size == e['new_bytes'], ('size mismatch', nf)
    lines[idx] = e['new_line']
    applied.append(dict(line=idx, file=nf.name, bytes=e['new_bytes']))
plan_out.parent.mkdir(parents=True, exist_ok=True)
plan_out.write_text('\n'.join(lines) + ('\n' if raw.endswith('\n') else ''), encoding='utf-8')
print('applied %d buffer edits' % len(applied))
for a in applied:
    print('  line %d %s (%d B)' % (a['line'], a['file'], a['bytes']))
print('plan sha', K.sha(plan_out))
out = K.cpu_check(plan_out)
print('cpu_check:', out.strip()[-120:])
