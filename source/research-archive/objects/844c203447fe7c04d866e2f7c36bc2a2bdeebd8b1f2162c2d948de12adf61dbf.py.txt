"""Make the F14-only plan from the current baseline plan and run cpu-check.

K.make_plan replaces only the SPV paths of the eight F14 PASS rows
(ordinals 95/100/105/110/115/120/125/130, PASS labels 58/63/68/73/78/83/88/93)
and asserts the replacement count equals the group size.
"""
import json
import sys
from pathlib import Path

HERE = Path(__file__).resolve().parent
CARD = HERE                              # this script lives in the card directory
EVID = CARD / 'evidence'
EVID.mkdir(exist_ok=True)
sys.path.insert(0, str(CARD.parent.parent))
import COMMON as K       # noqa: E402

SPV_SHA = '72e2df7c7b1e542f375f5c7c2bdc572433a08c5f1a1a0d706026efa54b9a1e87'
ORDINALS = [95, 100, 105, 110, 115, 120, 125, 130]


def main():
    out = CARD / 'plan.txt'
    replaced = K.make_plan(CARD / 'candidate.spv', out, spv_shas={SPV_SHA})
    plan_sha = K.sha(out)
    base_check = K.cpu_check(K.BASELINE)
    cand_check = K.cpu_check(out)
    (EVID / 'baseline-cpu-check.txt').write_text(base_check, encoding='utf-8')
    (EVID / 'candidate-cpu-check.txt').write_text(cand_check, encoding='utf-8')
    # structural diff of the two plans
    a = Path(K.BASELINE).read_text(encoding='utf-8-sig').splitlines()
    b = out.read_text(encoding='utf-8').splitlines()
    diff = [(i + 1, x, y) for i, (x, y) in enumerate(zip(a, b)) if x != y]
    res = dict(
        schema='n12-vit-qkv/plan/v1',
        parent_plan=dict(path=str(K.BASELINE), sha256=K.sha(K.BASELINE),
                         sha256_expected=K.BASELINE_SHA),
        candidate_plan=dict(path=str(out), sha256=plan_sha, lines=len(b),
                            parent_lines=len(a)),
        replaced_passes=replaced, expected_passes=len(ORDINALS), ordinals=ORDINALS,
        changed_lines=[dict(line=i, before=x, after=y) for i, x, y in diff],
        changed_line_count=len(diff),
        unchanged_elsewhere=(len(diff) == len([1 for i, x, y in diff])),
        cpu_check=dict(parent_stdout=base_check.strip(),
                       candidate_stdout=cand_check.strip(),
                       identical=base_check.strip() == cand_check.strip()),
    )
    assert replaced == len(ORDINALS), replaced
    assert len(diff) == len(ORDINALS), len(diff)
    (EVID / 'PLAN.json').write_text(json.dumps(res, indent=2), encoding='utf-8')
    print(json.dumps(res, indent=2))


if __name__ == '__main__':
    main()
