"""Which swin tile does the debug dump actually hold?  (validates the X gather too)"""
import sys
from pathlib import Path
import numpy as np

sys.path.insert(0, str(Path(__file__).resolve().parent))
import c32lib as L

HERE = Path(__file__).resolve().parent.parent
dbg = np.fromfile(HERE / "runs/S01-B4-ORACLE/swin_4_debug.bin", dtype="<f2")
info = L.block_bindings(L.D / "c32-minmax/local-b4.txt", ordinal=4)
TILES_X, TILES_Y = info["tiles_x"], info["tiles_y"]
x_flat = np.fromfile(info["input"], dtype="<f2")

for sg in (0, 1):
    X_dbg = np.stack([dbg[22528 + sg * 512 + col * 32: 22528 + sg * 512 + col * 32 + 32]
                      for col in range(16)], axis=1).astype(np.float32)
    best = []
    for tile in range(TILES_X * TILES_Y):
        TT = np.full(16, tile); CC = np.arange(16)
        X = L.gather(x_flat, TT, CC).astype(np.float32)
        best.append((L.rel_l2(X, X_dbg), tile))
    best.sort()
    print(f"sg={sg} dbg X absmax {np.abs(X_dbg).max():.4f}")
    for r, t in best[:3]:
        print(f"   tile {t:5d} (tx={t % TILES_X}, ty={t // TILES_X}) relL2={r:.6e}")
    # also try the transpose hypothesis on the best tile
    r, t = best[0]
    TT = np.full(16, t); CC = np.arange(16)
    X = L.gather(x_flat, TT, CC).astype(np.float32)
    print("   best tile: bitexact frac", float(np.mean(X == X_dbg)),
          "| absmax cpu/dbg", float(np.abs(X).max()), float(np.abs(X_dbg).max()))
