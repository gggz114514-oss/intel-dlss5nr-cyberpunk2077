import struct
p = r"@NR_WORKSPACE@/candidates/int8-coverage-v2-20261020/B/spv/c64_d1.spv"
data=open(p,'rb').read(); n=len(data)//4
w=struct.unpack("<%dI"%n, data[:n*4]); i=5
want={319,43,6,48,30,18,23,46,50,452,361,164,311,458}
while i<n:
    wc=w[i]>>16; op=w[i]&0xFFFF
    if wc==0: break
    ops=list(w[i+1:i+wc])
    if op in (19,20,21,22,23,24,28,29,30,32,33):
        if ops and ops[0] in want:
            print("type decl op",op,"id",ops[0],"operands",ops[1:])
    i+=wc
