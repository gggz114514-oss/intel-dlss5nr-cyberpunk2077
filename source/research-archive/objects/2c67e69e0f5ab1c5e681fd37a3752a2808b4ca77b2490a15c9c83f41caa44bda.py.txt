import sys, json, subprocess
from pathlib import Path
C = Path("@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/fullnet-continuous-20261027")
sys.path.insert(0, str(C))
import COMMON as K

CARD = C / "cards" / "n05a-f05-c256-mreuse"
ok = True

# 1. candidate.comp -> spv reproducibility against the shipped candidate.spv
re = CARD / "_build" / "cand-recheck.spv"
K.compile_glsl(CARD / "candidate.comp", re, K.include_dirs_for(37, CARD / "candidate.comp"))
same = K.sha(re) == K.sha(CARD / "candidate.spv")
print("candidate.spv == recompile(candidate.comp):", same, K.sha(re))
ok &= same

# 2. parent rebuild identity
print("parent-rebuild.spv == group spv:", K.sha(CARD / "parent-rebuild.spv") ==
      "9d61c4e3825265c42e7815372f6411a5e4777ab6267170a79a40dbfe862f66d8")
ok &= K.sha(CARD / "parent-rebuild.spv") == "9d61c4e3825265c42e7815372f6411a5e4777ab6267170a79a40dbfe862f66d8"

# 3. validate + cpu check again
K.validate_spv(CARD / "candidate.spv")
print("spv_validate_cpu: exit 0")
r = subprocess.run([str(K.RUNNER), "--plan", str(CARD / "plan.txt"), "--cpu-check"],
                   capture_output=True, text=True, encoding="utf-8", errors="replace")
print("cpu_check exit:", r.returncode, "|", r.stdout.strip())
ok &= r.returncode == 0

# 4. plan: 16 replacements only
txt = (CARD / "plan.txt").read_text(encoding="utf-8")
base = K.BASELINE.read_text(encoding="utf-8-sig")
print("candidate path occurrences:", txt.count((CARD / "candidate.spv").as_posix()))
print("baseline group path occurrences:", txt.count(Path(K.row_by_ordinal(37)["spv"]).as_posix()))
diff = sum(1 for a, b in zip(base.splitlines(), txt.splitlines()) if a != b)
print("plan lines differing from baseline:", diff, "| line counts equal:",
      len(base.splitlines()) == len(txt.splitlines()))
ok &= diff == 16 and len(base.splitlines()) == len(txt.splitlines())

print()
print("== deliverables ==")
want = ["CARD.txt", "ANALYSIS.md", "SOURCE-DIFF.patch", "PROOF.json", "COST-LEDGER.json",
        "NATIVE-PREDICTION.md", "BUILD.json", "candidate.comp", "candidate.spv",
        "parent-rebuild.spv", "plan.txt", "GPU-REQUEST.json", "RESUME.txt"]
for n in want:
    p = CARD / n
    print("%-22s %-8s %9d  %s" % (n, "OK" if p.exists() else "MISSING",
                                  p.stat().st_size if p.exists() else 0,
                                  K.sha(p)[:16] if p.exists() else ""))
    ok &= p.exists()
print()
print("ALL CHECKS:", "PASS" if ok else "FAIL")
