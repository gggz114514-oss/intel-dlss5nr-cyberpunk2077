"""Minimal SPIR-V scanner: cooperative-matrix types, loads/stores, muladds.

CPU only.  Used by n01b-f02-load-side to name which coopMatLoad sites exist and
what shape each cooperative matrix type is, without any GPU work.
"""
import struct, sys, json
from collections import Counter

OP = {
    59: 'OpVariable', 61: 'OpLoad', 62: 'OpStore', 65: 'OpAccessChain',
    4456: 'OpTypeCooperativeMatrixKHR', 4457: 'OpCooperativeMatrixLoadKHR',
    4458: 'OpCooperativeMatrixStoreKHR', 4459: 'OpCooperativeMatrixMulAddKHR',
    4460: 'OpCooperativeMatrixLengthKHR', 4461: 'OpConstantCompositeKHR',
    4455: 'OpCooperativeMatrixConstruct',
}


def parse(path):
    b = open(path, 'rb').read()
    assert b[:4] == b'\x03\x02\x23\x07', 'not spirv'
    n = len(b) // 4
    words = struct.unpack_from('<%dI' % (n - 5), b, 20)
    i = 0
    out = []
    while i < len(words):
        w = words[i]
        wc = w >> 16
        op = w & 0xFFFF
        out.append((op, words[i + 1:i + wc]))
        i += wc
    return out


def scan(path):
    inst = parse(path)
    types = {}
    consts = {}
    names = {}
    matrix_types = {}
    loads = []
    stores = []
    muladds = []
    for op, a in inst:
        if op == 4456:      # OpTypeCooperativeMatrixKHR result, component, scope, rows, cols, use
            matrix_types[a[0]] = dict(component=a[1], scope=a[2], rows=a[3],
                                      cols=a[4], use=a[5])
        elif op == 16:      # OpString-ish? no: OpSource; ignored
            pass
        elif op == 43:      # OpConstant
            consts[a[1]] = ('const', a[0], a[2:])
        elif op == 31:      # OpConstantNull
            consts[a[1]] = ('null', a[0])
        elif op == 4457:    # OpCooperativeMatrixLoadKHR  resulttype result ptr memlayout stride colmajor
            loads.append(dict(type=a[0], id=a[1], ptr=a[2], layout=a[3]))
        elif op == 4458:
            stores.append(dict(ptr=a[0], obj=a[1], layout=a[2]))
        elif op == 4459:
            muladds.append(dict(type=a[0], id=a[1], A=a[2], B=a[3], C=a[4]))
        elif op == 5:       # OpName
            if a:
                pass
    # resolve strides: operand j may be a constant id or a literal
    def cval(tok):
        if tok < len(inst) and tok in consts:
            c = consts[tok]
            if c[0] == 'const':
                return c[2][0] if c[2] else None
            return 0 if c[0] == 'null' else None
        return None

    for l in loads:
        l['stride_val'] = cval(l['layout'])
    return dict(matrix_types=matrix_types, loads=loads, stores=stores,
                muladds=muladds, n_consts=len(consts))


if __name__ == '__main__':
    d = scan(sys.argv[1])
    tc = Counter(l['type'] for l in d['loads'])
    print(json.dumps(d['matrix_types'], indent=2))
    print('loads by result type:')
    for t, c in tc.items():
        mt = d['matrix_types'][t]
        print('  type', t, mt, 'count', c)
    print('stores', len(d['stores']), 'muladds', len(d['muladds']))
    print('load strides:')
    print(Counter((l['type'], l['stride_val']) for l in d['loads']))
