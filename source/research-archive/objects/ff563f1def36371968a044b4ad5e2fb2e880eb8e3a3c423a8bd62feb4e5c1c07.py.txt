Enough of the binary format to answer:
  * the exact (opcode, word count) multiset of a module,
  * every cooperative-matrix MMA site with its A/B/C component kinds and widths,
  * cooperative matrix load/store/convert sites with their operand types,
  * shared (Workgroup) variables and their byte sizes,
  * descriptor bindings and the push-constant block size,
  * NoContraction decorations.
"""

class Module:
    def __init__(self, path):
        data = open(path, "rb").read()
        self.path = str(path)
        self.words = list(struct.unpack("<%dI" % (len(data) // 4), data))
        assert self.words[0] == 0x07230203, "not a SPIR-V module"
        self.ins = []
        i = 5
        while i < len(self.words):
            head = self.words[i]
            wc = head >> 16
            op = head & 0xFFFF
            self.ins.append((op, self.words[i + 1:i + wc], wc))
            i += wc
        self.types = {}
        self.consts = {}
        self.names = {}
        self.decor = {}
        self.vars = {}
        for idx, (op, a, wc) in enumerate(self.ins):
            if op == 21:
                self.types[a[0]] = ("int", a[1], a[2])
            elif op == 22:
                self.types[a[0]] = ("float", a[1])
            elif op == 23:
                self.types[a[0]] = ("vec", a[1], a[2])
            elif op == 4456:
                self.types[a[0]] = ("coopmat", a[1], a[2], a[3], a[4], a[5])
            elif op == 32:
                self.types[a[0]] = ("ptr", a[1], a[2])
            elif op == 28:
                self.types[a[0]] = ("array", a[1], a[2])
            elif op == 30:
                self.types[a[0]] = ("struct", list(a[1:]))
            elif op == 43:
                self.consts[a[1]] = ("const", a[0], a[2])
            elif op == 46:
                self.consts[a[1]] = ("null", a[0])
            elif op == 5:
                self.names[a[0]] = a[1].to_bytes(4, "little").split(b"\0")[0].decode("latin1")
            elif op == 71:
                self.decor.setdefault(a[0], []).append((a[1], a[2:]))

    # ---------- helpers ----------
    def type_of(self, tid):
        return self.types.get(tid)

    def comp_kind_width(self, tid):
        t = self.types.get(tid)
        if not t:
            return None
        if t[0] in ("int",):
            return (t[0], t[1])
        if t[0] == "float":
            return (t[0], t[1])
        if t[0] == "coopmat":
            return self.comp_kind_width(t[1])
        return None

    def defining(self, rid):
        for op, a, wc in self.ins:
            if op in (21, 22, 23, 28, 30, 32, 4456):
                if a[0] == rid:
                    return (op, a, wc)
            elif op in (43, 44, 46):
                if a[1] == rid:
                    return (op, a, wc)
            elif op in (59, 61, 62, 4457, 4458, 4459, 4460, 5358, 12, 82, 84, 117,
                        110, 111, 115, 113, 114, 133, 126, 130, 124, 141, 143, 152, 81,
                        65, 66, 67):
                if len(a) >= 2 and a[1] == rid:
                    return (op, a, wc)
        return None

    def resolve_coop_type(self, rid, depth=0):
        if depth > 8:
            return None
        t = self.types.get(rid)
        if t and t[0] == "coopmat":
            return t
        d = self.defining(rid)
        if not d:
            return None
        op, a, _ = d
        if op in (4457, 4458, 5358, 4459, 4460):     # result type of the instruction
            return self.resolve_coop_type(a[0], depth + 1)
        if op in (59, 61, 82, 84, 81, 117, 152, 44):
            return self.resolve_coop_type(a[0], depth + 1)
        return None

    def cval(self, cid):
        """Numeric value of a scalar integer/float constant id."""
        c = self.consts.get(cid)
        if not c:
            return None
        if c[0] == "null":
            return 0
        t = self.types.get(c[1])
        w = t[1] if t else 32
        v = c[2]
        if t and t[0] == "float":
            import struct as _s
            return _s.unpack("<f", _s.pack("<I", v))[0] if w == 32 else v
        if t and t[0] == "int":
            return v - (1 << w) if (w < 32 and v >= (1 << (w - 1))) else v
        return v

    def mma_sites(self):
        out = []
        for idx, (op, a, wc) in enumerate(self.ins):
            if op != 4459:
                continue
            at = self.resolve_coop_type(a[2])
            bt = self.resolve_coop_type(a[3])
            ct = self.resolve_coop_type(a[0])
            out.append({
                "index": idx, "word_offset": self.word_offset(idx),
                "result_id": a[1], "function": "main",
                "A_id": a[2], "B_id": a[3], "C_id": a[4],
                "A_comp": self.comp_kind_width(at[1]) if at else None,
                "B_comp": self.comp_kind_width(bt[1]) if bt else None,
                "C_comp": self.comp_kind_width(ct[1]) if ct else None,
                "A_shape": (self.cval(at[3]), self.cval(at[4])) if at else None,
                "B_shape": (self.cval(bt[3]), self.cval(bt[4])) if bt else None,
                "C_shape": (self.cval(ct[3]), self.cval(ct[4])) if ct else None,
                "words": wc,
            })
        return out

    def word_offset(self, idx):
        off = 5
        for i in range(idx):
            off += self.ins[i][2]
        return off

    def coop_io(self):
        """Cooperative matrix loads/stores/converts with component type."""
        out = {"load_sites": [], "store_sites": [], "convert_sites": []}
        for idx, (op, a, wc) in enumerate(self.ins):
            if op in (4457, 4458):
                ct = self.resolve_coop_type(a[0] if op == 4457 else a[0])
                rec = {"index": idx, "comp": self.comp_kind_width(a[0]),
                       "type": ct, "words": wc}
                (out["load_sites"] if op == 4457 else out["store_sites"]).append(rec)
            elif op == 5358:
                out["convert_sites"].append({"index": idx, "from": self.comp_kind_width(a[2]),
                                             "to": self.comp_kind_width(a[0]), "words": wc})
        return out

    def shared_vars(self):
        res = []
        for idx, (op, a, wc) in enumerate(self.ins):
            if op != 59 or len(a) < 3:
                continue
            if a[2] != 4:            # Workgroup storage class
                continue
            t = self.types.get(a[0])
            size = None
            if t and t[0] == "ptr":
                pt = self.types.get(t[2])
                if pt and pt[0] == "array":
                    el = self.comp_kind_width(pt[1])
                    n = self.consts.get(pt[2])
                    if el and n:
                        size = el[1] // 8 * n[2]
                elif pt and pt[0] in ("int", "float"):
                    size = pt[1] // 8
            res.append({"index": idx, "id": a[1], "name": self.names.get(a[1], ""),
                        "bytes": size})
        return res

    def bindings(self):
        res = []
        for idx, (op, a, wc) in enumerate(self.ins):
            if op != 59 or len(a) < 3 or a[2] != 12:
                continue
            d = dict(self.decor.get(a[1], []))
            res.append({"id": a[1], "name": self.names.get(a[1], ""),
                        "set": d.get(34, (None,))[0], "binding": d.get(33, (None,))[0]})
        return sorted(res, key=lambda r: (r["set"] if r["set"] is not None else -1,
                                          r["binding"] if r["binding"] is not None else -1))

    def push_constant_bytes(self):
        for idx, (op, a, wc) in enumerate(self.ins):
            if op == 59 and len(a) >= 3 and a[2] == 9:
                t = self.types.get(a[0])
                if t and t[0] == "ptr":
                    st = self.types.get(t[2])
                    if st and st[0] == "coopmat":
                        continue
                    if st and st[0] == "ptr":
                        pass
                # struct type
                for op2, a2, wc2 in self.ins:
                    if op2 == 30 and a2[0] == (t[2] if t and t[0] == "ptr" else None):
                        return sum(self._member_bytes(m) for m in a2[1:])
        return None

    def _member_bytes(self, tid):
        t = self.types.get(tid)
        if not t:
            return None
        if t[0] == "vec":
            base = self.comp_kind_width(t[1])
            return base[1] // 8 * t[2]
        if t[0] in ("int", "float"):
            return t[1] // 8
        if t[0] == "array":
            el = self._member_bytes(t[1])
            n = self.consts.get(t[2])
            return (el or 0) * (n[2] if n else 0)
        return None

    def no_contraction(self):
        ids = set()
        for rid, ds in self.decor.items():
            for d, args in ds:
                if d == 42:
                    ids.add(rid)
        return ids

    def histogram(self, code_only=True):
        hist = {}
        for op, a, wc in self.code() if code_only else self.ins:
            key = (op, wc)
            hist[key] = hist.get(key, 0) + 1
        return hist

    def code(self):
        start = next(i for i, (op, _, _) in enumerate(self.ins) if op == 54)
        return self.ins[start:]

    def extinst_hist(self):
        h = {}
        for op, a, wc in self.ins:
            if op == 12 and len(a) >= 4:
                h[a[3]] = h.get(a[3], 0) + 1
        return h

    def name(self, op):
        return OP.get(op, "Op%d" % op)
