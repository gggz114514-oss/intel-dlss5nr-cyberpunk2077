"""planlib.py - PORT_NETWORK_1 plan parser (read only).

Grammar (from work/dsh-int8-same-family-20261023/tools/plan_lint.py and the
plans themselves):

    PORT_NETWORK_1 <rows=152> <nbuffers> <npasses> <noutputs>
    BUFFER "<name>" <size> "<path|->"          x nbuffers
    PASS <row> "<name>" "<spv>" <sg=32> <shared> <nbind> <npush> <gx> <gy> <gz>
    BIND <res> <offset> <length>               x nbind
    PUSH <w0> ... <w{n-1}>                     (npush words)
    OUTPUT <res> "..." ...
    POLICY <res> <resident> <first> <last>
    HISTORY ...
"""
import re
from pathlib import Path

TOK = re.compile(r'"[^"]*"|\S+')


class Plan:
    def __init__(self, path):
        self.path = Path(path)
        self.raw = self.path.read_text(encoding="utf-8").splitlines()
        h = self.raw[0].split()
        assert h[0] == "PORT_NETWORK_1", self.raw[0]
        self.rows = int(h[1])
        self.nb = int(h[2])
        self.npass = int(h[3])
        self.nout = int(h[4])
        self.buffers = []          # (name, size, path)
        for ln in self.raw[1:1 + self.nb]:
            m = re.match(r'BUFFER "([^"]+)" (\d+) "([^"]*)"', ln)
            self.buffers.append((m.group(1), int(m.group(2)), m.group(3)))
        rest = self.raw[1 + self.nb:]
        self.passes = []           # dicts
        i = 0
        for pi in range(self.npass):
            t = TOK.findall(rest[i])
            p = {"ordinal": pi, "row": int(t[1]), "name": t[2].strip('"'),
                 "spv": t[3].strip('"'), "sg": int(t[4]), "shared": int(t[5]),
                 "nbind": int(t[6]), "npush": int(t[7]),
                 "grid": (int(t[8]), int(t[9]), int(t[10])),
                 "line": i + 1 + self.nb + 1, "binds": [], "push": []}
            for b in range(p["nbind"]):
                # BIND <binding> <resource> <offset_bytes> <bytes>   (runner format)
                w = rest[i + 1 + b].split()
                assert w[0] == "BIND", rest[i + 1 + b]
                p["binds"].append({"binding": int(w[1]), "res": int(w[2]),
                                   "offset": int(w[3]), "bytes": int(w[4])})
                assert p["binds"][-1]["binding"] == b or True
            pw = rest[i + 1 + p["nbind"]].split()
            assert pw[0] == "PUSH", rest[i + 1 + p["nbind"]]
            p["push"] = [int(x) for x in pw[1:]]
            assert len(p["push"]) == p["npush"], (p["name"], len(p["push"]))
            self.passes.append(p)
            i += 1 + p["nbind"] + 1
        self.outputs = []
        for ln in rest[i:i + self.nout]:
            w = TOK.findall(ln)
            self.outputs.append({"res": int(w[1]), "rest": ln})
        tail = rest[i + self.nout:]
        self.policy = []
        self.history = []
        for ln in tail:
            w = ln.split()
            if not w:
                continue
            if w[0] == "POLICY":
                self.policy.append({"res": int(w[1]), "resident": int(w[2]),
                                    "first": int(w[3]), "last": int(w[4]),
                                    "line": ln})
            elif w[0] == "HISTORY":
                self.history.append(ln)
            else:
                raise SystemExit("unexpected tail line: " + ln[:80])

    def bname(self, res):
        return self.buffers[res][0]

    def bsize(self, res):
        return self.buffers[res][1]

    def by_name(self, name):
        out = [p for p in self.passes if p["name"] == name]
        return out

    def names(self):
        return [p["name"] for p in self.passes]

    def index_of(self, name, nth=0):
        hits = [i for i, p in enumerate(self.passes) if p["name"] == name]
        return hits[nth]


def summarize(p):
    print("%-4d row %-3d %-22s %-72s sg=%d sh=%-6d nb=%d np=%d grid=%s" % (
        p["ordinal"], p["row"], p["name"], Path(p["spv"]).name,
        p["sg"], p["shared"], p["nbind"], p["npush"], p["grid"]))


def dump(plan, p, with_push=True):
    summarize(p)
    print("     spv " + p["spv"])
    for b in p["binds"]:
        nm = plan.bname(b["res"]) if b["res"] < plan.nb else "OUT-OF-RANGE"
        print("     BIND %d res %-4d %-22s off=%-9d bytes=%-9d buf=%d" % (
            b["binding"], b["res"], nm, b["offset"], b["bytes"],
            plan.bsize(b["res"]) if b["res"] < plan.nb else -1))
    if with_push:
        print("     PUSH " + " ".join(str(x) for x in p["push"]))
