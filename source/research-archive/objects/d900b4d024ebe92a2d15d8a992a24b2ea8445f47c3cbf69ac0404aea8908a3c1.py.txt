"""Which SPV does each of the three F3 down-sample passes actually dispatch?

Three sources are compared for every pass:

  * the audit's ledger pin (FULLNET-INT8-IMPLEMENTATION-LEDGER.json) -- the
    module whose word offsets FULLNET-RESIDUAL-MMA-SITES.json enumerates;
  * the INT8 control plan G3 (plan-g3-production.txt), the plan the ledger's
    residual sites were taken from;
  * the FP16 232 plan (combo-d01-d034-s32k11-v1), the current FP16 research
    parent.

For each referenced SPV the card records bytes + sha256 and re-enumerates its
static MulAdd sites, so a divergence is quantified rather than asserted.

usage: python tools/module_identity.py
"""
import json
import sys
from pathlib import Path

sys.dont_write_bytecode = True
HERE = Path(__file__).resolve().parent
CARD = HERE.parent
sys.path.insert(0, str(HERE))
import spvlib  # noqa: E402

for _op, _name in ((3, "OpSource"), (4, "OpSourceExtension"),
                   (185, "OpFOrdLessThan"), (186, "OpFOrdGreaterThan"),
                   (42, "OpConstantFalse")):
    spvlib.OP.setdefault(_op, _name)

R = Path("@NR_WORKSPACE@")
AUDIT = R / "work/codex-astra-fp16-int8-audit-20261024"
G3 = R / "candidates/int8-coverage-v2-20261020/plan-g3-production.txt"
P232 = R / "candidates/dsh-fp16-same-family-20261023/combo-d01-d034-s32k11-v1/plan-combo-d01-d034-s32k11-v1.txt"
PASSES = ["f3c64_body_ds", "f3c128_body_ds", "f3c256_body_ds"]
REPRO = {"f3c64_body_ds": CARD / "spv/parent_repro-f3c64_body_ds.spv",
         "f3c128_body_ds": CARD / "spv/parent_repro-f3c128_body_ds.spv",
         "f3c256_body_ds": CARD / "spv/parent_repro-f3c256_body_ds.spv"}


def sha(p):
    return __import__("hashlib").sha256(Path(p).read_bytes()).hexdigest()


def plan_pass(path, name):
    lines = Path(path).read_text(encoding="utf-8").splitlines()
    h = lines[0].split()
    nb, npass = int(h[2]), int(h[3])
    i = 1 + nb
    for _ in range(npass):
        t = lines[i].split()
        if t[0] != "PASS":
            return None
        nbind = int(t[6])
        if t[2].strip('"') == name:
            spv = t[3].strip('"')
            return {"plan": Path(path).as_posix(), "order": int(t[1]),
                    "spv": spv, "shared": int(t[5]), "nbind": nbind,
                    "grid": [int(t[8]), int(t[9]), int(t[10])],
                    "binds": [lines[i + 1 + b].split() for b in range(nbind)],
                    "push": lines[i + 1 + nbind].split()[1:]}
        i += nbind + 2
    return None


def mma(path):
    m = spvlib.Mod(path)
    out = {"fp16": 0, "int8": 0, "total": 0, "sites": []}
    for it in m.insts:
        if it.name != "OpCooperativeMatrixMulAddKHR":
            continue
        rc = m.cm_type(it.a[0]) or {}
        comp = rc.get("component")
        out["total"] += 1
        out["fp16" if comp == "f32" else "int8"] += 1
        out["sites"].append({"word_offset": it.woff, "result_id": it.a[1],
                             "function": m.funcs.get(it.func, "?"),
                             "accumulator": comp})
    return out


def main():
    led = json.load(open(AUDIT / "FULLNET-INT8-IMPLEMENTATION-LEDGER.json", encoding="utf-8"))
    bypass = {p["pass_name"]: p for p in led["positions"]}
    out = {"scope": "G3 dispatch identity of the three F3 down-sample passes",
           "ledger": (AUDIT / "FULLNET-INT8-IMPLEMENTATION-LEDGER.json").as_posix(),
           "g3_plan": {"path": G3.as_posix(), "sha256": sha(G3)},
           "fp16_232_plan": {"path": P232.as_posix(), "sha256": sha(P232)},
           "passes": {}}
    for name in PASSES:
        lp = bypass[name]
        g3 = plan_pass(G3, name)
        p232 = plan_pass(P232, name)
        rec = {"card": lp["card"], "ledger": {
                   "spv": lp["spv"], "sha256": lp["sha256"],
                   "bytes": Path(lp["spv"]).stat().st_size,
                   "grid": lp["grid"], "functions": lp["functions"],
                   "fp16_static_mma": lp["fp16_static_mma_count"]},
               "g3": g3, "fp16_232": p232,
               "replay": {"file": REPRO[name].as_posix(),
                          "bytes": REPRO[name].stat().st_size,
                          "sha256": sha(REPRO[name]),
                          "byte_identical_to_ledger_pin": sha(REPRO[name]) == lp["sha256"]}}
        for k, v in (("ledger", lp["spv"]), ("g3", g3 and g3["spv"]),
                     ("fp16_232", p232 and p232["spv"])):
            if not v:
                continue
            rec["%s_module" % k] = {"spv": v, "bytes": Path(v).stat().st_size,
                                    "sha256": sha(v), "mma": mma(v)}
        same = (g3 and g3["spv"] == lp["spv"]) and (p232 and p232["spv"] == lp["spv"])
        g3_same = bool(g3 and g3["spv"] == lp["spv"])
        p232_same = bool(p232 and p232["spv"] == lp["spv"])
        rec["verdict"] = {
            "g3_dispatch_is_ledger_pin": g3_same,
            "fp16_232_dispatch_is_ledger_pin": p232_same,
            "identity_scope": ("BOTH ARMS" if same else
                               "G3 ARM ONLY" if g3_same else "LEDGER PIN NOT DISPATCHED"),
            "fp16_232_module_fp16_mma": rec.get("fp16_232_module", {}).get("mma", {}).get("fp16"),
            "ledger_module_fp16_mma": rec.get("ledger_module", {}).get("mma", {}).get("fp16"),
        }
        out["passes"][name] = rec
    out["summary"] = {n: {"g3_same": v["verdict"]["g3_dispatch_is_ledger_pin"],
                          "p232_same": v["verdict"]["fp16_232_dispatch_is_ledger_pin"],
                          "scope": v["verdict"]["identity_scope"],
                          "ledger_fp16": v["verdict"]["ledger_module_fp16_mma"],
                          "p232_fp16": v["verdict"]["fp16_232_module_fp16_mma"]}
                      for n, v in out["passes"].items()}
    (CARD / "evidence").mkdir(parents=True, exist_ok=True)
    (CARD / "evidence/MODULE-IDENTITY.json").write_text(
        json.dumps(out, indent=1, ensure_ascii=False) + "\n", encoding="utf-8")
    print(json.dumps({"summary": out["summary"],
                      "details": {n: {"ledger": v["ledger"]["spv"].split("/")[-1],
                                      "g3": (v["g3"] or {}).get("spv", "-").split("/")[-1],
                                      "p232": (v["fp16_232"] or {}).get("spv", "-").split("/")[-1],
                                      "g3_shared": (v["g3"] or {}).get("shared"),
                                      "p232_shared": (v["fp16_232"] or {}).get("shared"),
                                      "replay_matches_pin": v["replay"]["byte_identical_to_ledger_pin"]}
                                  for n, v in out["passes"].items()}}, indent=1))
    return 0


if __name__ == "__main__":
    sys.exit(main())
