"""Sweep the two activation quantiser scales; objective = free-fit residual."""
import sys, itertools
from pathlib import Path
import numpy as np

sys.path.insert(0, str(Path(__file__).resolve().parent))
import c32lib as L

BLOCK = 4
info = L.block_bindings(L.D / "c32-minmax/local-b4.txt", ordinal=BLOCK)
x_flat = np.fromfile(info["input"], dtype="<f2")
gpu = np.fromfile(L.D / "c32-minmax/local-b4/replayed-network.bin", dtype="<f2")
wb = np.fromfile(info["wb"], dtype="<f2")
we, wc = L.load_w8(info["w8"])
fsc = L.load_fsc(info["fsc"])
rs = wb[info["rs_base"] + 8:info["rs_base"] + 40].astype(np.float16)

rng = np.random.default_rng(5)
tiles = np.sort(rng.choice(168 * 96, size=800, replace=False))
TT, CC = np.meshgrid(tiles, np.arange(16), indexing="ij")
TT = TT.ravel(); CC = CC.ravel()
base = TT * 512 + CC * 16
ref = np.empty((32, TT.size), np.float16)
for c in range(32):
    off = (c // 16) * 256 + ((c // 8) % 2) * 8 + (c % 8)
    ref[c] = gpu[base + off]
ref = ref.astype(np.float64)
x = L.gather(x_flat, TT, CC)
seed = L.widen(L.to_half(x * rs[:, None]))

tap_e = fsc[4:132]
tap_c = fsc[132:164]
sX0, sE0 = np.float32(fsc[0]), np.float32(fsc[2])
best = None
print("gx  ge   resid  (target with seed)   resid_noseed")
for gx, ge in itertools.product([0.5, 0.7, 1.0, 1.4, 2.0, 2.8, 4.0, 5.6], repeat=2):
    invx = np.float32(1.0 / (sX0 * gx))
    inve = np.float32(1.0 / (sE0 * ge))
    tapx = np.float32(sX0 * gx) * (tap_e / sX0)     # keep sWe fixed, change sX
    tape = np.float32(sE0 * ge) * (tap_c / sE0)
    xq = L.code8_ch(x, np.full(32, invx))
    acc = np.rint(we.astype(np.float64) @ xq.astype(np.float64)).astype(np.int32)
    a_h = L.act_h(L.to_half(acc.astype(np.float32) * tapx[:, None]))
    eq = L.code8_ch(a_h, np.full(128, inve))
    for with_seed in (True, False):
        target = ref - seed if with_seed else ref
        A, *_ = np.linalg.lstsq(eq.astype(np.float64).T, target.T, rcond=None)
        r = float(np.linalg.norm(target.T - eq.astype(np.float64).T @ A) / np.linalg.norm(target))
        if best is None or r < best[0]:
            best = (r, gx, ge, with_seed)
    print(f"{gx:4.1f} {ge:4.1f}  {r:.5f}")
print("best", best)
