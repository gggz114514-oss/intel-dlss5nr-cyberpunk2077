import sys, os, json, collections
sys.path.insert(0, r"@NR_WORKSPACE@/work/dsh-fullnet-static-20261029/tools")
import planlib
P = r"@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/int8-quality-first-20261029/real-int8-24/c64-family-static/combo-34189.txt"
p = planlib.Plan(P)
spvs = collections.OrderedDict()
for q in p.passes:
    spvs.setdefault(q.spv, []).append(q.ordinal)
print("unique spv", len(spvs))
for s, ords in spvs.items():
    sha = planlib.sha256_file(s)[:16] if os.path.isfile(s) else "MISSING"
    d = os.path.dirname(s)
    base = os.path.basename(s)
    stem = os.path.splitext(base)[0]
    cands = []
    for root in [d, os.path.join(d,'source'), os.path.dirname(d), os.path.join(os.path.dirname(d),'source')]:
        for ext in ['.comp','.glsl']:
            f = os.path.join(root, stem+ext)
            if os.path.isfile(f): cands.append(f)
    print(f"{sha} n={len(ords):3d} {base:42s} {d}  SRC={cands}")
