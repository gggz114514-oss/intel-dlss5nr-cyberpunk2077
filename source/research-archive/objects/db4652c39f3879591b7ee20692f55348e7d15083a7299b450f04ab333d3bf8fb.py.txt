import json, os
import numpy as np

W = r"@NR_WORKSPACE@/work/dsh-fullnet-static-20261029"
Q = r"@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/int8-quality-first-20261029"
D = os.path.join(Q, "real-int8-24")
REF = os.path.join(Q, "dataset/day-street-static-formal-01/0-34189/network-output.bin")

def rd(p):
    return np.fromfile(p, dtype="<f4").reshape(2, 720, 1280, 4)

ref = rd(REF)
def bucket_report(name, path):
    x = rd(path)
    a = x[0, :, :, :3].astype(np.float64)
    b = ref[0, :, :, :3].astype(np.float64)
    d = a - b
    lum = b.mean(axis=2)          # reference luma proxy (RGB mean), fixed mask from reference
    masks = {"dark(<0.25)": lum < 0.25, "mid(0.25-0.7)": (lum >= 0.25) & (lum < 0.7), "bright(>=0.7)": lum >= 0.7}
    rec = {"arm": name, "file": path, "overall": {}, "buckets": {}}
    for c, ch in enumerate("RGB"):
        rec["overall"][ch] = {"signed_mean": float(d[:, :, c].mean()), "mae": float(np.abs(d[:, :, c]).mean()),
                              "rmse": float(np.sqrt((d[:, :, c] ** 2).mean()))}
    rec["overall"]["psnr_db"] = float(-10 * np.log10(np.mean(d ** 2)))
    rec["overall"]["clip_frac_ref_at_1"] = float((b >= 1.0 - 1e-6).mean())
    rec["overall"]["clip_frac_arm_at_1"] = float((a >= 1.0 - 1e-6).mean())
    rec["overall"]["ref_absmax"] = float(np.abs(b).max())
    rec["overall"]["arm_absmax"] = float(np.abs(a).max())
    for k, m in masks.items():
        e = d[m]
        rec["buckets"][k] = {"pixel_fraction": float(m.mean()), "n": int(m.sum()),
                             "signed_mean_R": float(e[:, 0].mean()), "signed_mean_G": float(e[:, 1].mean()),
                             "signed_mean_B": float(e[:, 2].mean()),
                             "mean_abs": float(np.abs(e).mean()),
                             "ref_R_mean": float(b[m][:, 0].mean()), "ref_G_mean": float(b[m][:, 1].mean()),
                             "ref_B_mean": float(b[m][:, 2].mean()),
                             "clip_frac_arm_at_1": float((a[m] >= 1 - 1e-6).mean()),
                             "clip_frac_ref_at_1": float((b[m] >= 1 - 1e-6).mean())}
    print("\n%s  psnr=%.3f dB  clip ref/arm=%.5f/%.5f  absmax ref/arm=%.3f/%.3f" % (
        name, rec["overall"]["psnr_db"], rec["overall"]["clip_frac_ref_at_1"], rec["overall"]["clip_frac_arm_at_1"],
        rec["overall"]["ref_absmax"], rec["overall"]["arm_absmax"]))
    print("  overall signed R %+.5f G %+.5f B %+.5f" % (rec["overall"]["R"]["signed_mean"], rec["overall"]["G"]["signed_mean"], rec["overall"]["B"]["signed_mean"]))
    for k, v in rec["buckets"].items():
        print("  %-16s frac=%.4f  dR %+.5f dG %+.5f dB %+.5f  mean|e| %.5f  clip arm/ref %.5f/%.5f" % (
            k, v["pixel_fraction"], v["signed_mean_R"], v["signed_mean_G"], v["signed_mean_B"], v["mean_abs"],
            v["clip_frac_arm_at_1"], v["clip_frac_ref_at_1"]))
    return rec

out = []
for name, p in [("c64-family-static combo-34189 (XE candidate)", os.path.join(D, "c64-family-static/run-34189/replayed-network.bin")),
                ("corrected-INT8 baseline (run-34189)", os.path.join(D, "run-34189/replayed-network.bin")),
                ("c32-equalized (run-34189)", os.path.join(D, "c32-equalized/run-34189/replayed-network.bin"))]:
    if os.path.exists(p):
        out.append(bucket_report(name, p))
    else:
        print("MISSING", p)
json.dump(out, open(os.path.join(W, "C00-first-divergence/_work/rgb-buckets-34189.json"), "w", encoding="utf-8"), ensure_ascii=False, indent=1)
print("\nwrote rgb-buckets-34189.json")
