"""Register-index watermark helper for the n02d card (CPU only, reads a parked ISA).

The kernel is a SIMD32 compute shader whose allocator already reaches r127
(128 GRF/thread = the ceiling).  This script answers one question for BUDGET.md:
how much register-index watermark does the *fb-bias region* actually use today,
so the analytical peak of a wide-fragment variant can be compared against the
same region instead of against the whole kernel.

It never talks to a GPU: it parses a text ISA file that a previous GPU batch
parked under queue/*/native/captures.
"""
import re
import sys
from pathlib import Path

LINE = re.compile(r'/\* \[([0-9A-Fa-f]+)\]\s*\*/\s*(.*)$')
REG = re.compile(r'\b(?:acc\d*|[rf]?(\d+))(?:\.[0-9A-Fa-f]+)?(?:<[^>]*>)?(?::[a-z0-9]+)?')


def scan(path, lo=None, hi=None):
    """Return (max_reg, {reg: count}) over instructions whose pc is in [lo,hi]."""
    hist = {}
    total = 0
    for raw in Path(path).read_text(encoding='utf-8', errors='replace').splitlines():
        m = LINE.search(raw)
        if not m:
            continue
        pc = int(m.group(1), 16)
        if lo is not None and pc < lo:
            continue
        if hi is not None and pc > hi:
            continue
        total += 1
        # drop the byte-offset/comment tail so immediates are not read as registers
        body = m.group(2).split('//')[0]
        for g in re.finditer(r'\br(\d+)\b', body):
            n = int(g.group(1))
            hist[n] = hist.get(n, 0) + 1
    return (max(hist) if hist else None), hist, total


def main(argv):
    path = Path(argv[1])
    windows = [('all', None, None)]
    if len(argv) > 2:
        for spec in argv[2:]:
            name, lo, hi = spec.split(':')
            windows.append((name, int(lo, 16), int(hi, 16)))
    out = {}
    for name, lo, hi in windows:
        mx, hist, n = scan(path, lo, hi)
        top = sorted(hist.items(), key=lambda kv: -kv[0])[:8]
        out[name] = dict(instructions=n, max_reg=mx, top_regs=top,
                         regs_over_96=sorted(k for k in hist if k > 96))
        print('%-24s instrs=%-5d max_r=%-4s >96: %s'
              % (name, n, mx, out[name]['regs_over_96'][:12]))
    return out


if __name__ == '__main__':
    sys.dont_write_bytecode = True
    main(sys.argv)
