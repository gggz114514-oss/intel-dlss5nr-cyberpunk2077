"""Correct L27: text-identical operands are NOT redundancy; record the three-criteria test."""
import json, sys
from pathlib import Path
sys.path.insert(0, str(Path(__file__).parent))
import COMMON as K

# 1. annotate the census with the caveat that invalidates its headline number
p = K.C / 'evidence' / 'REDUNDANT-LATENCY-OPS.json'
d = K.read_json(p)
d['REFUTED'] = True
d['refuted_by'] = ('n33 applied a dword-granularity reaching-definition analysis to the F02 kernel and found that '
                   'the 24 expensive ops are 24 DISTINCT definitions: between any two sites of the same text group '
                   'there is at least one write to the source dword, and zero pairs share a definition. The '
                   'text-identical operand is register-name reuse, not redundancy.')
d['refuted_scope'] = ('cross-checked on two more kernels with the same method: a0888c5a (F01, rsqt 10) -> 0 genuine, '
                      'f3942341 (F19, rsqt 9) -> 0 genuine. Only 575e6253 (F21) showed genuine redundancy: 12 copies of '
                      'the same 1/x, exec-1 scalar, each segment re-deriving it from the same mov r2.x<-r1.0.')
d['required_test'] = ['same reaching definition (dword granularity)',
                      'no write to the source dword between the copies',
                      'same exec width, and the value is a push constant or loop invariant',
                      'each copy has consumers reading the same value']
d['headline_number_invalid'] = d.pop('total_redundant', None)
K.write_json(p, d)

# 2. rewrite L27
lp = K.C / 'evidence' / 'LESSONS.json'
L = K.read_json(lp)
for x in L['append_after_L7']:
    if x['id'] == 'L27':
        x['claim'] = ('Text-identical operands are not redundancy: a register name is reused across sites, and only a '
                      'reaching-definition analysis can tell a duplicated computation from a reused register')
        x['evidence'] = [
            'a census that grouped math.inv/math.rsqt by their source operand text reported 173 redundant copies in '
            '40 kernels (63% of the 275 expensive ops); n33 then ran a dword-granularity reaching-definition analysis '
            'on the F02 kernel and found the 24 ops there are 24 DISTINCT definitions, with at least one intervening '
            'write to the source dword between any two same-text sites and zero shared definitions',
            'the same check on a0888c5a (F01, rsqt 10) and f3942341 (F19, rsqt 9) also returned zero genuine duplicates',
            'only 575e6253 (F21) showed real redundancy: 12 copies of the same 1/x, exec-1 scalar, each block '
            're-deriving it from the same mov',
            'deleting a genuine-looking but non-redundant copy is E-class (the gold frames change), not A-class'
        ]
        x['consequence'] = ('Before opening a redundancy card, prove all four: same reaching definition at dword '
                            'granularity, no intervening write, same exec width, and consumers reading the same value. '
                            'This is the fourth time static text accounting overstated the machine (see L11 lane masks, '
                            'L14 fragment shapes, L18 source redundancy).')
        print('L27 corrected')
K.write_json(lp, L)
print('census annotated; L27 rewritten')
