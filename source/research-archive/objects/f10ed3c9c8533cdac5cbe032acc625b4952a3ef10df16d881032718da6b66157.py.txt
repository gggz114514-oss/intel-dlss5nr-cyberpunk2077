"""Rank PASSes by the measured stall time they actually spend (ms), not by isolated time.

For each PASS: stall_ms = GpuTime_ns * stall_percent / 100. This turns the per-PASS counters
into a millisecond budget and shows where the remaining serialisation money is.
"""
import json, sys
from pathlib import Path
sys.path.insert(0, str(Path(__file__).parent))
import COMMON as K

p = list((K.C / 'queue' / 'counters-xmx2').rglob('hardware-counters.json'))[0]
d = K.read_json(p)
names = [m['name'] for m in d['metrics']]
idx = {n: i for i, n in enumerate(names)}
passes = d['frames'][0]['passes']

def col(n):
    return [pp['values'][idx[n]] for pp in passes]

gpu = col('GpuTime')
stalls = {n: col(n) for n in names if n.startswith('XVE_STALL_')}
active = col('XVE_ACTIVE')
total_ms = sum(gpu) / 1e6

print('frame: %.3f ms (sum of per-PASS GpuTime)' % total_ms)
print('\nstall budget (time-weighted, ms) — how much wall time each stall reason accounts for:')
rows = []
for n, v in stalls.items():
    w = sum(a * b for a, b in zip(v, gpu)) / sum(gpu)
    rows.append((w * total_ms / 100.0, w, n))
for ms, w, n in sorted(rows, reverse=True):
    if ms > 0.01:
        print('  %-28s %5.1f%% of time  = %6.3f ms' % (n, w, ms))

def weighted(v):
    return sum(a * b for a, b in zip(v, gpu)) / sum(gpu)

print('\nidle inside dispatches: 100 - XVE_ACTIVE = %.1f%%  = %.3f ms'
      % (100 - weighted(active), (100 - weighted(active)) * total_ms / 100.0))

xve = active
cand = []
for i, pp in enumerate(passes):
    ms = gpu[i] / 1e6
    idl = (100 - xve[i]) * ms / 100.0
    aluwr = stalls['XVE_STALL_ALUWR'][i] * ms / 100.0
    sendwr = stalls['XVE_STALL_SENDWR'][i] * ms / 100.0
    bar = stalls['XVE_STALL_BARRIER'][i] * ms / 100.0
    cand.append((idl, ms, pp['name'], pp['index'], aluwr, sendwr, bar))
cand.sort(reverse=True)
print('\ntop 10 PASSes by measured idle time inside the dispatch (ms):')
print('  %-28s %7s %7s %7s %7s %7s' % ('name', 'gpu_ms', 'idle_ms', 'aluwr', 'sendwr', 'barrier'))
for idl, ms, name, i, aluwr, sendwr, bar in cand[:10]:
    print('  %-28s %7.3f %7.3f %7.3f %7.3f %7.3f' % (name, ms, idl, aluwr, sendwr, bar))
print('\nsum of idle across all 232: %.3f ms' % sum(c[0] for c in cand))
print('sum of ALUWR stall: %.3f ms, SENDWR: %.3f ms, BARRIER: %.3f ms'
      % (sum(c[4] for c in cand), sum(c[5] for c in cand), sum(c[6] for c in cand)))

K.write_json(K.C / 'evidence' / 'STALL-BUDGET.json', dict(
    frame_ms=total_ms, idle_ms=sum(c[0] for c in cand),
    stall_ms={n: sum(a * b for a, b in zip(v, gpu)) / 1e6 / 100.0 for n, v in stalls.items()},
    top_idle=[dict(name=c[2], index=c[3], gpu_ms=c[1], idle_ms=c[0], aluwr=c[4], sendwr=c[5], barrier=c[6])
              for c in cand[:20]]))
print('\nwrote evidence/STALL-BUDGET.json')
