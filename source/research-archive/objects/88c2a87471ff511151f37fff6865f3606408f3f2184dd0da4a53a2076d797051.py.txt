"""Tolerant PORT_NETWORK_1 reader for S03-tail auditing.

planlib (the shared tool) requires quoted PASS name/SPV, which the incumbent FP16
plan does not use.  This reader accepts both forms and never writes to the plan.
It is read-only and produces plain dicts; it is *not* a patcher.
"""
from __future__ import annotations

import hashlib
import shlex
from dataclasses import dataclass, field
from pathlib import Path
from typing import Dict, List, Optional


def sha256_file(path) -> str:
    h = hashlib.sha256()
    with open(path, "rb") as f:
        for chunk in iter(lambda: f.read(1 << 20), b""):
            h.update(chunk)
    return h.hexdigest()


@dataclass
class Buf:
    index: int
    name: str
    bytes: int
    file: str
    line: int


@dataclass
class Disp:
    ordinal: int
    group: int
    name: str
    spv: str
    grid: List[int]
    line: int
    binds: List[dict] = field(default_factory=list)
    push: Optional[List[str]] = None
    push_line: Optional[int] = None


@dataclass
class Plan:
    path: Path
    header: List[str] = field(default_factory=list)
    buffers: List[Buf] = field(default_factory=list)
    passes: List[Disp] = field(default_factory=list)
    outputs: List[dict] = field(default_factory=list)
    policies: List[dict] = field(default_factory=list)
    by_name: Dict[str, Buf] = field(default_factory=dict)
    unparsed: List[str] = field(default_factory=list)

    @property
    def sha256(self) -> str:
        return sha256_file(self.path)

    def buffer(self, name: str) -> Buf:
        return self.by_name[name]

    def named(self, *needles: str) -> List[Disp]:
        return [p for p in self.passes if all(n in p.name for n in needles)]


def load(path) -> Plan:
    path = Path(path)
    plan = Plan(path=path)
    text = path.read_text(encoding="utf-8", errors="replace")
    ordinal = 0
    cur: Optional[Disp] = None
    for i, raw in enumerate(text.splitlines(), 1):
        line = raw.strip()
        if not line:
            continue
        toks = shlex.split(line, posix=False)
        head = toks[0].upper()
        if head == "PORT_NETWORK_1":
            plan.header = toks[1:]
        elif head == "BUFFER":
            name = toks[1].strip('"')
            buf = Buf(len(plan.buffers), name, int(toks[2]), toks[3].strip('"'), i)
            plan.buffers.append(buf)
            plan.by_name[name] = buf
        elif head == "PASS":
            cur = Disp(ordinal, int(toks[1]), toks[2].strip('"'), toks[3].strip('"'),
                       [int(t) for t in toks[4:]], i)
            plan.passes.append(cur)
            ordinal += 1
        elif head == "BIND":
            if cur is None:
                plan.unparsed.append(line)
                continue
            cur.binds.append({"binding": int(toks[1]), "buffer": int(toks[2]),
                              "offset": int(toks[3]), "bytes": int(toks[4]), "line": i})
        elif head == "PUSH":
            if cur is None:
                plan.unparsed.append(line)
                continue
            cur.push = toks[1:]
            cur.push_line = i
        elif head == "OUTPUT":
            plan.outputs.append({"buffer": int(toks[1]), "format": toks[2],
                                 "file": toks[3].strip('"'), "line": i})
        elif head == "POLICY":
            plan.policies.append({"buffer": int(toks[1]), "resident": int(toks[2]),
                                  "first": int(toks[3]), "last": int(toks[4]),
                                  "clear_first": int(toks[5]), "line": i})
        else:
            plan.unparsed.append(line)
    return plan
