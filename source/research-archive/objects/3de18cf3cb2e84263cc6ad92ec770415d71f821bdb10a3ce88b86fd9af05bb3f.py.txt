"""Bind verified real snapshots to corrected INT8; preserve its internal ABI."""
from pathlib import Path
import json, shlex, hashlib
Q=Path(__file__).resolve().parent
A=Q.parent
base=A/'int8-v15-offline-20261028/corrected-20261029/plan.txt'
out=Q/'real-int8-24';out.mkdir(exist_ok=False)
lines=base.read_text().splitlines()
buffers=[shlex.split(l) for l in lines if l.startswith('BUFFER ')]
oldpush=list(map(int,next(l for l in lines if l.startswith('PUSH ')).split()[1:]))
jobs=[]
for j in json.loads((Q/'capture-buffered/formal-replay-24/JOBS.json').read_text()):
    cap=next((Q/'dataset'/j['sequence']).glob(f'*-{j["frame"]}/capture.json'))
    m=json.loads(cap.read_text()); push=m['first_push']
    assert all(a==b for i,(a,b) in enumerate(zip(oldpush,push)) if i not in (22,23))
    assert len(oldpush)==len(push)==32
    assert int(buffers[483][2])==m['output']['bytes']
    snapshots={v['resource']:v for v in m['buffers']}
    for n,v in snapshots.items():
        assert int(buffers[n][2])==v['bytes']
        assert hashlib.sha256((cap.parent/v['file']).read_bytes()).hexdigest()==v['sha256']
    result=[];bi=-1;ordinal=-1
    for l in lines:
        t=shlex.split(l)
        if not t:result.append(l);continue
        if t[0]=='HISTORY':continue
        if t[0]=='BUFFER':
            bi+=1
            if bi in snapshots:l=f'BUFFER "{t[1]}" {t[2]} "{(cap.parent/snapshots[bi]["file"]).as_posix()}"'
        if t[0]=='PASS':ordinal+=1
        if t[0]=='PUSH' and ordinal==0:l='PUSH '+' '.join(map(str,push))
        if t[0]=='OUTPUT':
            assert int(t[1])==483
            l='OUTPUT 483 float32 "replayed-network.bin"'
        result.append(l)
    p=out/f'int8-{j["frame"]}.txt';p.write_text('\n'.join(result)+'\n')
    jobs.append(dict(sequence=j['sequence'],frame=j['frame'],plan=str(p),out=str(out/f'run-{j["frame"]}'),expected_output_sha256=None,reference=str(cap.parent/m['output']['file']),dump=False))
# Dump just one real frame initially. No timing inference from alias-disabled dumps.
first=json.loads((Q/'capture-buffered/formal-replay-24/JOBS.json').read_text())[0]
for arm,p in [('fp16',first['plan']),('int8',jobs[0]['plan'])]:
    jobs.append(dict(sequence='diagnostic-'+arm,frame=first['frame'],plan=p,out=str(out/f'dump-{arm}'),expected_output_sha256=first['expected_output_sha256'] if arm=='fp16' else None,dump=True))
(out/'JOBS.json').write_text(json.dumps(jobs,indent=2))
(out/'CONTRACT.json').write_text(json.dumps({'int8_source':str(base),'sha256':hashlib.sha256(base.read_bytes()).hexdigest(),'input_resources':[452,478],'output_resource':483,'push_change':[22,23],'history':'captured FP16 history, independent frames','jobs':len(jobs)},indent=2))
print('Prepared',len(jobs),'jobs')
