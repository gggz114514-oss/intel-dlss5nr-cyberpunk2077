def nm(op):
    return NAMES.get(op, 'op%d' % op)


def census(path):
    m = ST.Module(path)
    ops = {}
    for op, o, _, _ in m.instructions:
        ops[op] = ops.get(op, 0) + 1
    ext = {}
    for op, o, _, _ in m.instructions:
        if op == 12:
            if o[3] > 0x10000:
                ext['set%d/%d' % (o[2] & 0xff, o[3])] = ext.get('set%d/%d' % (o[2] & 0xff, o[3]), 0) + 1
            else:
                key = 'ExtInst_%d%s' % (o[3], '' if o[3] not in GLSL_STD else '(' + GLSL_STD[o[3]] + ')')
                ext[key] = ext.get(key, 0) + 1
    # variable / memory census
    vars_ = []
    for op, o, _, _ in m.instructions:
        if op == 59:
            vars_.append(dict(id=o[1], storage=ST_NAMES.get(o[2], str(o[2]))))
    mem = {}
    for op, o, _, _ in m.instructions:
        mem[nm(op)] = mem.get(nm(op), 0) + 1
    return dict(path=str(path), sha256=hashlib.sha256(Path(path).read_bytes()).hexdigest(),
                words=len(m.words), bytes=Path(path).stat().st_size,
                nc=sorted(ST.no_contraction_ids(m)), nc_count=len(ST.no_contraction_ids(m)),
                opcodes={nm(k): v for k, v in sorted(ops.items())},
                opcodes_by_id={str(k): v for k, v in sorted(ops.items())},
                ext=ext, variable_storage={s: sum(1 for v in vars_ if v['storage'] == s)
                                           for s in {v['storage'] for v in vars_}},
                variables=vars_, memory=mem)


# ---------------------------------------------------------------- 1. F08 capture identity of the quoted parent
def f08_identity():
    f08 = C / 'cards/n11-vit-ffn'  # placeholder, real path below
    p = W / 'work/codex-astra-fp16-int8-audit-20261024/fullnet-mechanism-transfer-20261026/f08-register-chain-20261027/plan.txt'
    out = {'plan': str(p), 'exists': p.exists()}
    if not p.exists():
        return out
    lines = p.read_text(encoding='utf-8-sig').splitlines()
    hits = [l for l in lines if 'u03-f18-v2/shader.spv' in l]
    out['f16_pass_lines'] = hits
    out['f16_lines'] = len(hits)
    out['spv_sha256'] = hashlib.sha256(Path(BASE_SPV).read_bytes()).hexdigest()
    out['equals_current_baseline_spv'] = out['spv_sha256'] == hashlib.sha256((CARD / 'parent-rebuild.spv').read_bytes()).hexdigest()
    return out


res = {'card': 'n11b-vit-projection-hoist', 'stage': 'census+plan'}
res['f08_capture_identity'] = f08_identity()
res['parent'] = census(PARENT)
res['candidate'] = census(CAND)

pa, ca = res['parent']['opcodes_by_id'], res['candidate']['opcodes_by_id']
delta = {}
for k in sorted(set(pa) | set(ca), key=lambda x: int(x)):
    a, b = pa.get(k, 0), ca.get(k, 0)
    if a != b:
        delta[nm(int(k))] = {'parent': a, 'candidate': b, 'delta': b - a}
res['opcode_delta'] = delta
ea, ec = res['parent']['ext'], res['candidate']['ext']
res['ext_delta'] = {k: [ea.get(k, 0), ec.get(k, 0)] for k in sorted(set(ea) | set(ec)) if ea.get(k, 0) != ec.get(k, 0)}
res['variable_storage_delta'] = {k: [res['parent']['variable_storage'].get(k, 0), res['candidate']['variable_storage'].get(k, 0)]
                                for k in sorted(set(res['parent']['variable_storage']) | set(res['candidate']['variable_storage']))}
res['nc_identical'] = res['parent']['nc'] == res['candidate']['nc']

Path(CARD / '_tools/spv-diff.json').write_text(json.dumps(res, indent=2), encoding='utf-8')

# ---------------------------------------------------------------- 2. plan on the accepted chain head
text = CHAIN_PLAN.read_text(encoding='utf-8-sig')

n = text.count(BASE_SPV)
assert n == 8, 'chain head has %d F16 pass lines' % n
new = text.replace(BASE_SPV, (CARD / 'candidate.spv').as_posix())
plan = CARD / 'plan.txt'
plan.write_text(new, encoding='utf-8')
# replacement diff
a, b = text.splitlines(), new.splitlines()
assert len(a) == len(b)
changed = [(i + 1, x, y) for i, (x, y) in enumerate(zip(a, b)) if x != y]
CAND_PATH = (CARD / 'candidate.spv').as_posix()


def masked(s):
    return s.replace(BASE_SPV, '<SPV>').replace(CAND_PATH, '<SPV>')


def fields(s):
    return [t for t in s.split('"')]


plan_res = dict(parent_plan=str(CHAIN_PLAN), parent_plan_sha256=chain_sha,
                parent_plan_sha256_expected=CHAIN_SHA, parent_matches_current_json=True,
                f16_occurrences=n, replaced=n, expected=8, diff_lines=len(changed),
                only_spv_path_changed=all(masked(x) == masked(y) for _, x, y in changed),
                changed=[dict(line=i, old=x, new=y,
                              name=fields(x)[1],
                              spv_before=fields(x)[3],
                              numeric_tail=fields(x)[4].split(),
                              grid=' '.join(fields(x)[4].split()[-3:]))
                         for i, x, y in changed],
                plan=dict(path=str(plan), sha256=hashlib.sha256(plan.read_bytes()).hexdigest()),
                bytes=plan.stat().st_size)
res['plan'] = plan_res

# ---------------------------------------------------------------- 3. cpu check
r = subprocess.run([str(RUNNER), '--plan', str(plan), '--cpu-check'],
                   capture_output=True, text=True, encoding='utf-8', errors='replace')
res['cpu_check'] = {'argv': [str(RUNNER), '--plan', str(plan), '--cpu-check'],
                    'rc': r.returncode, 'stdout': r.stdout, 'stderr': r.stderr}
Path(CARD / '_tools/verify-and-plan.json').write_text(json.dumps(res, indent=2), encoding='utf-8')

print('cpu_check out:', r.stdout)
print('cpu_check err:', r.stderr)
