"""Address-layer census: are ugm message addresses already base+imm (a shared base
register with an immediate offset), or is a fresh address GRF built per message?

Read-only over the CURRENT baseline ISA captures. Writes _addr/<ordinal>.json.
"""
import json, os, re, glob, sys
from collections import Counter

CARD = os.path.dirname(os.path.abspath(__file__))
EVID = os.path.join(os.path.dirname(os.path.dirname(CARD)), "evidence")
IDX = json.load(open(os.path.join(EVID, "NATIVE-INDEX.json"), encoding="utf-8"))
COV = {r["ordinal"]: r for r in IDX["coverage"]}
H2F = json.load(open(os.path.join(CARD, "_hash2asm.json"), encoding="utf-8"))
LINE_RE = re.compile(r"^/\* \[([0-9A-Fa-f]+)\]\s+\*/\s*(.*)$")


def resolve(o):
    for c in (H2F.get(str(o)) or []):
        if os.path.exists(c):
            return c


def sends(path):
    prev_addr = None
    rows = []
    for raw in open(path, encoding="utf-8", errors="replace"):
        m = LINE_RE.match(raw)
        if not m:
            continue
        body = m.group(2)
        if "send.ugm" not in body:
            continue
        left = body.split("//")[0]
        t = [x for x in left.split() if not re.match(r"^\(\d+\|", x)]
        # t[0]='send.ugm', t[1]=dst, t[2]=addr, t[3]=data, t[4]=aN, t[5]=desc
        addr = t[2] if len(t) > 2 else None
        imm = None
        for x in t[2:]:
            mm = re.match(r"^(0x[0-9A-Fa-f]+):a\d", x)
            if mm:
                imm = mm.group(1)
        rows.append({"addr": m.group(1), "addr_reg": addr, "imm": imm,
                     "msg": re.search(r"(load|store)\.ugm\.([A-Za-z0-9]+)", body).group(0)
                     if re.search(r"(load|store)\.ugm\.", body) else None,
                     "same_reg_as_prev": (addr is not None and addr == prev_addr)})
        prev_addr = addr
    return rows


def main():
    out = os.path.join(CARD, "_addr")
    os.makedirs(out, exist_ok=True)
    args = sys.argv[1:] or [str(o) for o in sorted(COV)]
    for a in args:
        o = int(a)
        p = resolve(o)
        if not p:
            continue
        rows = sends(p)
        c = Counter()
        for r in rows:
            c["sends"] += 1
            if r["imm"]:
                c["with_immediate"] += 1
            if r["same_reg_as_prev"]:
                c["same_addr_reg_as_prev"] += 1
            if r["msg"] and re.match(r"^d(8|16|32)(u(8|16|32))?$", r["msg"].split(".")[-1]):
                c["narrow"] += 1
                if r["imm"]:
                    c["narrow_with_immediate"] += 1
                if r["same_reg_as_prev"]:
                    c["narrow_same_addr_reg"] += 1
        c["family"] = COV[o]["family"]
        c["name"] = COV[o]["name"]
        c["kernel_hash"] = COV[o]["baseline_kernel_hash"]
        json.dump(dict(c), open(os.path.join(out, "%03d.json" % o), "w"), indent=1)


if __name__ == "__main__":
    main()
