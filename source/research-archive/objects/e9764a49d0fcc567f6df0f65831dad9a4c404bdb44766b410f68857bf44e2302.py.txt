"""R01 F1 - exact hypothesis test, no accumulator inversion.

Test whether dumped E row of an offending token is reproduced bit-exactly by
qx + d*e_k for some channel k and integer d (single code change), or by
qx + d1*e_k1 + d2*e_k2 (pair), by direct recomputation of the half E.
"""
import json
import sys
from pathlib import Path

ROOT = Path("@NR_WORKSPACE@")
DUMP = ROOT / "run/int8-coverage-v2-20261020/C/c32-i8-diag-f1-scalex2-001/diag_f1_debug.bin"
WEIGHTS = ROOT / "candidates/int8-coverage-v2-20261020/C/pack/c32_f1_weights.i8.bin"
SCALES = ROOT / "candidates/int8-coverage-v2-20261020/C/c32-scale-from-x-v2/pack/c32_f1_scales.f32.bin"
FP16 = ROOT / "candidates/amd-map-20261015/s1-integrated/candidate/packs/ioport_0_ioport_pack.fp16.bin"

sys.path.insert(0, str(ROOT / "work/int8-coverage-v2-20261020/C/tools"))
from ref_c32 import Ref, half_array, f32, h16, code  # noqa: E402

dbg = half_array(DUMP)
ref = Ref(WEIGHTS, SCALES, FP16, 17280, 8)
BAD = [23, 24, 30, 46, 52]


def act_h(x):
    t = max(-4.0, min(4.0, x))
    u = h16((-0.055908203125) * abs(t) + 0.447265625)
    v = h16(t * u + 0.89453125)
    return h16(x * v)


W = [[ref.w[(f // 8) * 1024 + k * 32 + (f % 8)] for k in range(32)] for f in range(128)]
MULT = ref.mult_e


def e_row(q):
    acc = [sum(W[f][k] * q[k] for k in range(32)) for f in range(128)]
    return [act_h(h16(f32(a * MULT[f]))) for f, a in enumerate(acc)], acc


out = {"tokens": []}
for t in BAD:
    x = [dbg[22528 + t * 32 + c] for c in range(32)]
    qx = [code(x[c], ref.invx) for c in range(32)]
    ej = [dbg[t * 128 + f] for f in range(128)]
    e0, acc0 = e_row(qx)
    rec = {"token": t, "base_ok": sum(1 for f in range(128) if e0[f] == ej[f])}
    single = []
    for k in range(32):
        for d in range(-127, 128):
            if d == 0:
                continue
            q = list(qx)
            q[k] += d
            if not (-127 <= q[k] <= 127):
                continue
            e, _ = e_row(q)
            hits = sum(1 for f in range(128) if e[f] == ej[f])
            if hits >= 128:
                single.append({"k": k, "d": d})
    rec["exact_single"] = single
    if not single:
        # best partial single
        best = (0, None, None)
        for k in range(32):
            for d in range(-6, 7):
                if d == 0:
                    continue
                q = list(qx)
                q[k] += d
                if not (-127 <= q[k] <= 127):
                    continue
                e, _ = e_row(q)
                hits = sum(1 for f in range(128) if e[f] == ej[f])
                if hits > best[0]:
                    best = (hits, k, d)
        rec["best_single_partial"] = {"hits": best[0], "k": best[1], "d": best[2]}
    # column / block permutation hypotheses on X
    perms = {
        "rot8": [(c + 8) % 32 for c in range(32)],
        "rot16": [(c + 16) % 32 for c in range(32)],
        "rot4": [(c + 4) % 32 for c in range(32)],
        "fragswap_01_23": [c ^ 8 for c in range(32)],
        "frag_swap_02": [c ^ 16 for c in range(32)],
        "rev": [31 - c for c in range(32)],
        "token_col_swap_adj": None,
    }
    permhits = {}
    for name, p in perms.items():
        if p is None:
            continue
        q = [code(x[p[c]], ref.invx) for c in range(32)]
        e, _ = e_row(q)
        permhits[name] = sum(1 for f in range(128) if e[f] == ej[f])
    rec["permutation_hits"] = permhits
    out["tokens"].append(rec)

print(json.dumps(out["tokens"], indent=1))
Path(__file__).with_name("f1_probe7.out.json").write_text(json.dumps(out, indent=1), encoding="utf-8")
