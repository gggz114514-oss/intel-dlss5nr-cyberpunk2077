"""Run the 8-frame autonomous-feedback chain with the current best candidate.

This is a GPU orchestration helper, not a candidate builder.  It calls the
existing feedback_plan.py to build one frame at a time (each frame needs the
previous frame's output history) and the existing locked queue runner for the
actual replay.  Writes only under S13-quality/ and gpu/.
"""
from __future__ import annotations

import hashlib
import json
import subprocess
import sys
from pathlib import Path

B = Path("@NR_WORKSPACE@")
W = B / "work/dsh-fullnet-static-20261029"
S13 = W / "S13-quality"
PY = B / "work/codex-astra-fp16-int8-audit-20261024/int8-quality-first-20261029/tool-env/Scripts/python.exe"
FEEDBACK = S13 / "feedback_plan.py"
RUNNER = W / "gpu/run-queue.ps1"
ROOT = S13 / "runs-static01"
STAGE = S13 / "staged-live"
HEAD = B / "work/codex-astra-fp16-int8-audit-20261024/int8-quality-first-20261029/real-int8-24/c64-family-static/run-34189"
SEQ = "day-street-static-formal-01"


def sha(p: Path) -> str:
    h = hashlib.sha256()
    with open(p, "rb") as f:
        for b in iter(lambda: f.read(1 << 20), b""):
            h.update(b)
    return h.hexdigest()


def build(frame: int) -> bool:
    cmd = [str(PY), "-X", "utf8", str(FEEDBACK), "chain", "--sequence", SEQ,
           "--frames", "34189-34196", "--run-root", str(ROOT), "--stage-dir", str(STAGE),
           "--head-run", str(HEAD), "--head-output-name", "replayed-network.bin"]
    r = subprocess.run(cmd, capture_output=True, text=True)
    print("chain", frame, r.returncode, r.stdout.strip()[:400])
    if r.returncode != 0:
        print(r.stderr)
        return False
    return (ROOT / f"plan-{frame}.txt").is_file()


def run(frame: int) -> bool:
    plan = ROOT / f"plan-{frame}.txt"
    out = ROOT / f"run-{frame}"
    if out.exists():
        print("out exists, skip", out)
        return True
    queue = {
        "schema": "dsh-gpu-queue/1", "status": "NOT_RUN",
        "runner": str(B / "work/codex-astra-fp16-int8-audit-20261024/phase-diagnosis-20261025/direct-clock-build/Release/clock_runner.exe"),
        "lock": "@NR_GPU_WORKSPACE@/artifacts/parallel-port-20261013/GPU.lock",
        "authorization": "User authorized in DSH session on 2026-10-29 19:5x Asia/Tokyo; exclusive lock protocol, no game concurrent",
        "jobs": [{
            "card": "S13-feedback", "priority": 1, "sequence": "feedback", "frame": frame,
            "plan": str(plan).replace("\\", "/"), "out": str(out).replace("\\", "/"),
            "plan_sha256": sha(plan), "expected_output_sha256": None,
            "output_name": "feedback-network.bin", "dump": False,
            "note": "autonomous feedback chain", "preflight_problems": [], "ready": True,
        }],
        "ready": 1, "blocked": 0,
    }
    qp = W / f"gpu/QUEUE-FEEDBACK-{frame}.json"
    qp.write_text(json.dumps(queue, indent=2), encoding="utf-8")
    r = subprocess.run(["powershell.exe", "-NoProfile", "-ExecutionPolicy", "Bypass",
                        "-File", str(RUNNER), "-Queue", str(qp),
                        "-Owner", f"DSH-FEEDBACK-{frame}-20261029"],
                       capture_output=True, text=True)
    print("run", frame, r.returncode, (r.stdout or "").strip()[-300:])
    if r.returncode != 0:
        print((r.stderr or "").strip()[-1000:])
        return False
    return (out / "feedback-network.bin").is_file()


def main() -> None:
    for frame in range(34191, 34197):
        if not (ROOT / f"plan-{frame}.txt").is_file() and not build(frame):
            print("build failed at", frame)
            break
        if not run(frame):
            print("run failed at", frame)
            break
    else:
        print("chain complete 34191..34196")


if __name__ == "__main__":
    main()
