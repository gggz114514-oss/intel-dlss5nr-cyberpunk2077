# S09: dump one row (bindings/push/pass tokens) by ordinal from CURRENT-ROWS.json
import json, sys, os
ROWS = r"@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/int8-quality-first-20261029/dsh-static-handoff/CURRENT-ROWS.json"
j = json.load(open(ROWS, encoding="utf-8"))
rows = j["rows"]
bufs = j["buffers"]
want = [int(x) for x in sys.argv[1:]]
for r in rows:
    if r["ordinal"] not in want:
        continue
    print("=" * 100)
    print("ordinal", r["ordinal"], "pass_no", r["pass_number"], "name", r["name"])
    print("spv", r["spv"])
    print("sha256", r["sha256"])
    print("pass_tokens", r.get("pass_tokens"))
    print("push(%d):" % len(r.get("push") or []), r.get("push"))
    for b, v in sorted(r["bindings"].items(), key=lambda kv: int(kv[0])):
        bi, off, nbytes = v
        nm = "?"
        try:
            nm = bufs[bi].get("name") if isinstance(bufs[bi], dict) else str(bufs[bi])
        except Exception:
            nm = "?"
        print(f"  bind {b:>2} -> buffer {bi:>3} {nm:<34} off={off:<12} bytes={nbytes}")
