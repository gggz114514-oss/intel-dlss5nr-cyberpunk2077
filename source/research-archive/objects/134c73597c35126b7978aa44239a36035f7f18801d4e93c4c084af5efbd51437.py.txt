STORAGE = {0: 'UniformConstant', 1: 'Input', 2: 'Uniform', 3: 'Output',
           4: 'Workgroup', 5: 'CrossWorkgroup', 6: 'Private', 7: 'Function',
           8: 'Generic', 9: 'PushConstant', 10: 'AtomicCounter', 11: 'Image',
           12: 'StorageBuffer'}


def scan(path):
    b = Path(path).read_bytes()
    words = list(struct.unpack('<%dI' % (len(b) // 4), b))
    assert words[0] == 0x07230203, hex(words[0])
    i, counts, wg, vars_, decs = 5, collections.Counter(), [], [], collections.Counter()
    nc = 0
    while i < len(words):
        wc = words[i] >> 16
        op = words[i] & 0xFFFF
        counts[op] += 1
        if op == 59:  # OpVariable
            sc = words[i + 3]
            vars_.append(STORAGE.get(sc, sc))
            if sc == 4:
                wg.append(words[i + 2])
        if op == 71:  # OpDecorate
            decs[words[i + 2]] += 1
            if words[i + 2] == 44:
                nc += 1
        i += wc if wc else 1
    return words, counts, wg, vars_, decs, nc


def main():
    p = Path(sys.argv[1])
    words, counts, wg, vars_, decs, nc = scan(p)
    print('file     ', p)
    print('bytes    ', p.stat().st_size, ' words', len(words))
    print('variables by storage class:', dict(collections.Counter(vars_)))
    print('workgroup variables:', len(wg), wg)
    print('barriers: ControlBarrier %d  MemoryBarrier %d' % (counts[224], counts[225]))
    print('coopmat: Load %d  Store %d  MulAdd %d  Length %d'
          % (counts[4457], counts[4459], counts[4458], counts[4460]))
    print('NoContraction decorations:', nc)
    print('decorations total:', sum(decs.values()))
    print('float ops: FAdd %d FMul %d FConvert %d FNegate %d'
          % (counts[128], counts[130], counts[180] + counts[168], counts[140]))
    print('int ops: IAdd %d IMul %d Shl %d Shr %d AccessChain %d Select %d'
          % (counts[87], counts[94], counts[109], counts[110], counts[65] + counts[67],
             counts[141]))
    print('mem: Load %d Store %d' % (counts[61], counts[62]))
    print('blocks: Label %d  BranchConditional %d  LoopMerge %d  SelectionMerge %d'
          % (counts[248], counts[250], counts[246], counts[247]))
    print('top opcodes:', [(OP.get(k, k), v) for k, v in counts.most_common(20)])


if __name__ == '__main__':
    main()
