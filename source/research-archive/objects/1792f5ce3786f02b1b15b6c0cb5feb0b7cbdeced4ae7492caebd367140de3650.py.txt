"""n06d-f06-c256-atile build: parent byte reproduction, candidate compile, SPIR-V
validation, plan replacement on the CURRENT CHAIN HEAD (15 PASS), CPU plan check,
SPIR-V accounting, GPU-REQUEST.

CPU only. No GPU, no deployment, no baseline-plan / pointer / COMMON.py edits.
"""
import json, sys, subprocess
from pathlib import Path

CARD = Path(__file__).resolve().parent.parent
CAMP = CARD.parent.parent
sys.path.insert(0, str(CAMP))
sys.path.insert(0, str(CARD / 'scratch'))
import COMMON as K
import spvstat

TARGET_SPV_SHA = '881dff8c51454f79181a567d6b976ea21ec7c81174bf075ca7be723b09c4a3f0'
TARGET_SRC_SHA = 'c3efee68a1935d28fbcbbadcfb5d44381277c6936401b464cf37b132f6221c16'
PARENT_SRC = K.D / 'f06-scale-fragments/family/c256/candidate.comp'
PARENT_SPV = K.D / 'f06-scale-fragments/family/c256/candidate.spv'
PINNED = CAMP / 'cards/idt-identity-recipe/evidence/headers/881dff8c5145'
BUILD = CARD / 'build'
PLAN = CARD / 'plan.txt'


def run(argv):
    return subprocess.run([str(x) for x in argv], capture_output=True, text=True,
                          encoding='utf-8', errors='replace')


def main():
    BUILD.mkdir(parents=True, exist_ok=True)
    out = {'card': CARD.name, 'kind': 'build+plan+cpu-check (CPU only)',
           'mechanism': 'publish shT in consumer tile order (tile contiguous 256 B, '
                        'read stride 16); shT kept, one 32 KiB global read, not segmented'}

    # ---- group identity
    assert K.sha(PARENT_SRC) == TARGET_SRC_SHA, 'parent source sha drifted'
    assert K.sha(PARENT_SPV) == TARGET_SPV_SHA, 'parent spv sha drifted'
    ordinals = sorted(r['ordinal'] for r in K.rows() if r['spv_sha256'] == TARGET_SPV_SHA)
    assert len(ordinals) == 15, ordinals
    assert ordinals == [38, 41, 44, 47, 50, 53, 56, 170, 173, 176, 179, 182, 185, 188, 191]
    rec = K.read_json(CAMP / 'evidence' / 'RECIPES.json')[TARGET_SPV_SHA]
    out['group'] = dict(family='F06 C256', spv_sha256=TARGET_SPV_SHA,
                        source_sha256=TARGET_SRC_SHA, ordinals=ordinals,
                        n_passes=len(ordinals), recipe_mode=rec['nc_recipe']['mode'])
    out['parent'] = dict(path=str(PARENT_SPV), sha256=TARGET_SPV_SHA,
                         source=str(PARENT_SRC), source_sha256=TARGET_SRC_SHA)

    # ---- 1. parent byte reproduction (raw recipe: no -D, pinned header dir, no NC transfer)
    parent_rebuild = CARD / 'parent-rebuild.spv'
    r = run([K.GLSLANG, '-V', '--target-env', 'vulkan1.3', '-I' + str(PINNED),
             PARENT_SRC, '-o', parent_rebuild])
    (BUILD / 'parent-rebuild-stdout.txt').write_text(r.stdout + r.stderr, encoding='utf-8')
    assert r.returncode == 0, r.stdout + r.stderr
    out['parent_rebuild'] = dict(path=str(parent_rebuild), sha256=K.sha(parent_rebuild),
                                 target=str(PARENT_SPV), target_sha256=TARGET_SPV_SHA,
                                 matches_parent_bytes=K.sha(parent_rebuild) == TARGET_SPV_SHA,
                                 recipe='glslang -V --target-env vulkan1.3 -I pinned_headers '
                                        '(no -D, no NoContraction transfer; RECIPES.json '
                                        'nc_recipe.mode = %s)' % rec['nc_recipe']['mode'])
    assert out['parent_rebuild']['matches_parent_bytes'], 'parent bytes not reproduced'

    # ---- 2. candidate compile + validate
    cand_spv = CARD / 'candidate.spv'
    r = run([K.GLSLANG, '-V', '--target-env', 'vulkan1.3', '-I' + str(PINNED),
             CARD / 'candidate.comp', '-o', cand_spv])
    (BUILD / 'candidate-stdout.txt').write_text(r.stdout + r.stderr, encoding='utf-8')
    assert r.returncode == 0, r.stdout + r.stderr
    vr = run([K.VALIDATE, cand_spv])
    (BUILD / 'validate-stdout.txt').write_text(vr.stdout + vr.stderr, encoding='utf-8')
    out['candidate'] = dict(path=str(cand_spv), sha256=K.sha(cand_spv),
                            source=str(CARD / 'candidate.comp'),
                            source_sha256=K.sha(CARD / 'candidate.comp'),
                            bytes=cand_spv.stat().st_size,
                            validate_returncode=vr.returncode,
                            validate_stdout=(vr.stdout + vr.stderr).strip()[:400])
    assert vr.returncode == 0, 'candidate.spv failed spv_validate_cpu'

    # ---- 3. plan: current chain head, replace only this group's 15 PASS lines
    head, head_sha, _chain = K.current_best()
    out['chain_head'] = dict(path=str(head), sha256=head_sha)
    out['handoff_baseline_plan'] = dict(path=str(K.BASELINE), sha256=K.sha(K.BASELINE),
                                        expected_sha256=K.BASELINE_SHA)
    assert K.sha(K.BASELINE) == K.BASELINE_SHA, 'handoff baseline plan drifted'
    out['handoff_baseline_spv_of_group'] = dict(path=str(PARENT_SPV), sha256=TARGET_SPV_SHA,
                                                status='OFFLINE_VALIDATED_NOT_DEPLOYED')
    replaced = K.make_plan(cand_spv, PLAN, parent_plan=head, spv_shas={TARGET_SPV_SHA})
    out['plan'] = dict(path=str(PLAN), sha256=K.sha(PLAN), replaced_passes=replaced,
                       expected_passes=15, parent_plan=str(head), parent_plan_sha256=head_sha)
    assert replaced == 15, replaced
    assert K.sha(head) == head_sha, 'chain head changed under us'
    # SPV lines in the plan: exactly 15 point at the candidate, none at the parent path
    txt = PLAN.read_text(encoding='utf-8-sig')
    out['plan']['lines_to_candidate'] = txt.count(cand_spv.as_posix())
    out['plan']['lines_to_parent_spv'] = txt.count(PARENT_SPV.as_posix())
    assert out['plan']['lines_to_candidate'] == 15
    assert out['plan']['lines_to_parent_spv'] == 0
    # diff vs chain head must be exactly 15 changed lines
    a = Path(head).read_text(encoding='utf-8-sig').splitlines()
    b = txt.splitlines()
    assert len(a) == len(b)
    changed = [i for i, (x, y) in enumerate(zip(a, b)) if x != y]
    out['plan']['diff_lines'] = len(changed)
    out['plan']['diff_line_index_first'] = changed[0]
    out['plan']['diff_line_index_last'] = changed[-1]
    assert len(changed) == 15, len(changed)

    # ---- 4. CPU plan check (candidate vs chain head)
    cc = K.cpu_check(PLAN)
    (BUILD / 'cpu-check.txt').write_text(cc, encoding='utf-8')
    out['cpu_check'] = cc.strip()
    head_cc = K.cpu_check(head)
    (BUILD / 'cpu-check-chainhead.txt').write_text(head_cc, encoding='utf-8')
    out['cpu_check_chain_head'] = head_cc.strip()
    out['cpu_check_identical_to_chain_head'] = cc.strip() == head_cc.strip()
    assert cc.strip().startswith('CPU_PLAN_OK'), cc

    # ---- 5. GPU-REQUEST (contract structure) then cross-check rebase_by/rebuild
    gpu_req = {
        'card': CARD.name,
        'shader_group': {'spv_sha256_baseline': [TARGET_SPV_SHA], 'pass_ordinals': ordinals},
        'plan': {'path': PLAN.as_posix(), 'sha256': out['plan']['sha256']},
        'arms': [{'id': 'base', 'plan': 'baseline', 'role': 'baseline'},
                 {'id': 'cand', 'plan': 'candidate', 'role': 'candidate'}],
        'gates': ['fixed', 'fb2'],
        'native': True,
        'resources': True,
        'ordinary': {'design': 'two-arm-6-alternating', 'reps': 30, 'warm': 1, 'feedback': True},
        'hypothesis': 'If native shows the A-read SLM messages per subgroup falling from 1024 '
                      'to 128..256 with identical bytes/sectors and dpas=256, then the '
                      'consumer-tile publish landed',
        'native_observables': [
            'send.slm load.slm.d32x8t.a32 : 1024 -> 0 (per SG), replaced by 128..256 wider '
            'non-transposed SLM read messages (d32x2/d32x4/d32x8)',
            'total load.slm messages on the A read sites: 1024 -> 128..256 per SG',
            'send.ugm load.ugm.d32x8.a32.bss (weight tiles): 256 -> 256 per SG',
            'dpas.8x8: 256 -> 256 per SG',
            'workgroup memory 32768 B, scratch 0 B',
            'send.slm store for the publish: 16 d16u32 -> 4 wide stores/warp (32 B/lane)',
            'sync.bar dynamic: 1 -> 1 (no barrier added; the card adds no segmentation)',
        ],
    }
    K.write_json(CARD / 'GPU-REQUEST.json', gpu_req)
    alt = CARD / 'build' / 'plan-rebased-check.txt'
    p2, sha2, n2, base2 = K.rebase_candidate_plan(CARD.name, alt)
    out['rebase_cross_check'] = dict(plan=str(p2), sha256=sha2, replaced=n2,
                                     baseline=str(base2['baseline']),
                                     baseline_sha256=base2['baseline_sha'],
                                     identical_to_plan=sha2 == out['plan']['sha256'])
    assert n2 == 15 and sha2 == out['plan']['sha256'], 'rebase disagrees with make_plan'

    # ---- 6. SPIR-V accounting parent vs candidate
    pm, cm = spvstat.Mod(parent_rebuild), spvstat.Mod(cand_spv)
    ps, cs = pm.summary(), cm.summary()
    K.write_json(BUILD / 'spv-stats.json', dict(parent=ps, candidate=cs))
    out['spirv'] = dict(
        parent=dict(sha256=TARGET_SPV_SHA, words=ps['words'], bytes=ps['bytes'],
                    instructions=ps['instructions'], no_contraction=ps['no_contraction'],
                    float_ops=ps['float_ops'], cooperative_matrix=ps['cooperative_matrix'],
                    int_ops=ps['int_ops'], control_flow=ps['control_flow'],
                    non_function_storage=ps['non_function_storage']),
        candidate=dict(sha256=K.sha(cand_spv), words=cs['words'], bytes=cs['bytes'],
                       instructions=cs['instructions'], no_contraction=cs['no_contraction'],
                       float_ops=cs['float_ops'], cooperative_matrix=cs['cooperative_matrix'],
                       int_ops=cs['int_ops'], control_flow=cs['control_flow'],
                       non_function_storage=cs['non_function_storage']),
        interface_equal=dict(
            capabilities=ps['capabilities'] == cs['capabilities'],
            extensions=ps['extensions'] == cs['extensions'],
            entry_points=ps['entry_points'] == cs['entry_points'],
            execution_modes=ps['execution_modes'] == cs['execution_modes'],
            memory_model=ps['memory_model'] == cs['memory_model'],
            non_function_storage=ps['non_function_storage'] == cs['non_function_storage'],
            decorations=ps['decorations'] == cs['decorations'],
            float_ops=ps['float_ops'] == cs['float_ops'],
            cooperative_matrix=ps['cooperative_matrix'] == cs['cooperative_matrix'],
        ),
        float_NC_checked=cs['no_contraction'],
        float_NC_note='0 is the true count, not a missing measurement: neither arm has a '
                      'NoContraction decoration (decoration histogram has no key 43), so '
                      'there is no FP site to align; the FP expression set is identical '
                      '(OpFMul 2, OpFConvert 4, OpFAdd/OpFSub 0, OpQuantizeToF16 0)',
    )
    out['spirv']['interface_all_equal'] = all(out['spirv']['interface_equal'].values())
    assert out['spirv']['interface_all_equal'], out['spirv']['interface_equal']
    assert cs['no_contraction'] == ps['no_contraction'] == 0
    assert cs['float_ops'] == ps['float_ops'], (cs['float_ops'], ps['float_ops'])

    K.write_json(CARD / 'BUILD.json', out)
    print('parent rebuild matches bytes:', out['parent_rebuild']['matches_parent_bytes'])
    print('cand spv sha', out['candidate']['sha256'], out['candidate']['bytes'], 'B')
    print('plan sha', out['plan']['sha256'], 'replaced', out['plan']['replaced_passes'],
          'diff lines', out['plan']['diff_lines'])
    print('cpu-check:', out['cpu_check'], '| identical to chain head:',
          out['cpu_check_identical_to_chain_head'])
    print('instr %d -> %d, words %d -> %d' % (ps['instructions'], cs['instructions'],
                                              ps['words'], cs['words']))
    print('coop', ps['cooperative_matrix'], '->', cs['cooperative_matrix'])
    print('int ops', ps['int_ops'], '->', cs['int_ops'])
    print('control flow', ps['control_flow'], '->', cs['control_flow'])
    print('non-function storage', ps['non_function_storage'], '->', cs['non_function_storage'])


if __name__ == '__main__':
    main()
