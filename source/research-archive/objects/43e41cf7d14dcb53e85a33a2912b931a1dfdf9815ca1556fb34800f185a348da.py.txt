"""n21 step-0: reproduce the parent SPV of ord 194 byte-exactly, and dump SPIR-V facts."""
import sys, json, hashlib, subprocess
from pathlib import Path

CARD = Path(__file__).resolve().parent

sys.path.insert(0, str(C))
import COMMON as K

BUILD = CARD / 'build'
BUILD.mkdir(exist_ok=True)

PARENT_COMP = Path(K.row_by_ordinal(194)['source'])
PARENT_SPV = Path(K.row_by_ordinal(194)['spv'])
PINNED = C / 'cards/idt-identity-recipe/evidence/headers/ffde3c3d2379'

print('parent comp :', PARENT_COMP, K.sha(PARENT_COMP))
print('parent spv  :', PARENT_SPV, K.sha(PARENT_SPV))
print('pinned dir  :', PINNED, PINNED.exists())
print('glslang     :', K.GLSLANG, K.GLSLANG.exists())

raw = BUILD / 'parent-rebuild-raw.spv'
K.compile_glsl(PARENT_COMP, raw, [PINNED])
s = K.sha(raw)
print('rebuild raw :', s, raw.stat().st_size, 'match=', s == K.row_by_ordinal(194)['spv_sha256'])

# cross-check with the shared helper (must also hit raw)
res = K.reproduce_current(194, BUILD / 'repro-current')
print('reproduce_current:', json.dumps({k: v for k, v in res.items() if k in
      ('raw_match', 'recipe', 'nc_count', 'decorated_match')}))
