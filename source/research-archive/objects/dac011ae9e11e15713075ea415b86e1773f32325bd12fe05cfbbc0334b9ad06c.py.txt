import pathlib,json,time,datetime,configparser
R=pathlib.Path(__file__).resolve().parent
P=pathlib.Path('@NR_WORKSPACE@/work/github-experimental-game-install-20261024/package/config')
result=[]
for label,key,build in [('C1','small','optimized-full-gpu-handoff-v1'),('B','102','optimized-full-20261028'),('C2','small','optimized-full-gpu-handoff-v1')]:
 c=configparser.ConfigParser();c.read(P/'nr_panel.ini',encoding='utf8');c['NR']['backend']=key
 tmp=P/'nr_panel.aba.tmp'
 with tmp.open('w',encoding='utf8') as f:c.write(f)
 tmp.replace(P/'nr_panel.ini')
 deadline=time.monotonic()+20
 while True:
  try:status=json.loads((P/'component_status.json').read_text());panel=json.loads((P/'panel_state.json').read_text())
  except (ValueError,OSError):status={};panel={}
  if status.get('build_id')==build and status.get('init_state')==2 and panel.get('backend')==key:break
  if time.monotonic()>deadline:raise RuntimeError('Identity switch timeout '+label)
  time.sleep(.25)
 start=time.monotonic();print('START '+label,flush=True)
 while time.monotonic()-start<20:
  elapsed=time.monotonic()-start
  try:
   status=json.loads((P/'component_status.json').read_text());panel=json.loads((P/'panel_state.json').read_text())
   row=dict(segment=label,elapsed=elapsed,settling=elapsed<5,utc=datetime.datetime.now(datetime.timezone.utc).isoformat(),component=status,panel=panel)
   result.append(row)
   (R/'stationary-cbc.json').write_text(json.dumps(result,indent=2),encoding='utf8')
   if status.get('build_id')!=build or panel.get('backend')!=key:raise RuntimeError('Identity changed '+label)
   if status.get('gpu_timeout',0) or status.get('last_status',0):raise RuntimeError('GPU status failure '+label)
  except (json.JSONDecodeError,PermissionError) as e:
   result.append(dict(segment=label,elapsed=elapsed,read_error=str(e)))
  time.sleep(.5)
 print('END '+label,flush=True)
print('CBC_DONE; game remains running; no acceptance or deployment')
