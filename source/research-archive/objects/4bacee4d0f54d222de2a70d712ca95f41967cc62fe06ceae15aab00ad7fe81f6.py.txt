#!/usr/bin/env python3
"""MANIFEST of this work directory: every file's size and sha256, grouped.

Skips nothing silently: binary evidence is listed too, so a reviewer can pin the exact
artefacts a claim rests on.

usage: make_manifest.py <root> [--out MANIFEST.json]
"""
import hashlib
import json
import sys
from pathlib import Path

SKIP_DIRS = {"__pycache__", "tmp"}


def sha(p, buf=1 << 20):
    h = hashlib.sha256()
    with open(p, "rb") as f:
        for b in iter(lambda: f.read(buf), b""):
            h.update(b)
    return h.hexdigest()


def main(argv):
    root = Path(argv[0])
    out = None
    if "--out" in argv:
        out = Path(argv[argv.index("--out") + 1])
    files, groups = [], {}
    for p in sorted(root.rglob("*")):
        if not p.is_file():
            continue
        if any(part in SKIP_DIRS for part in p.relative_to(root).parts):
            continue
        rel = p.relative_to(root).as_posix()
        top = rel.split("/")[0]
        rec = {"path": rel, "bytes": p.stat().st_size, "sha256": sha(p)}
        files.append(rec)
        groups.setdefault(top, {"files": 0, "bytes": 0})
        groups[top]["files"] += 1
        groups[top]["bytes"] += rec["bytes"]
    doc = {"root": str(root), "file_count": len(files), "total_bytes": sum(f["bytes"] for f in files),
           "groups": groups, "files": files}
    dst = out or (root / "MANIFEST.json")
    dst.write_text(json.dumps(doc, indent=1, ensure_ascii=False) + "\n",
                   encoding="utf-8", newline="\n")
    print(json.dumps({"files": doc["file_count"], "bytes": doc["total_bytes"],
                      "groups": {k: v["files"] for k, v in groups.items()}}, indent=1))
    return 0


if __name__ == "__main__":
    sys.exit(main(sys.argv[1:]))
