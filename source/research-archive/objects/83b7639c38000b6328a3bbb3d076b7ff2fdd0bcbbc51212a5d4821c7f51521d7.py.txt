"""Append the n29-era three-arm confirmation to evidence/CHAIN-CONFIRMATION.json."""
import json, sys
from pathlib import Path
sys.path.insert(0, str(Path(__file__).parent))
import COMMON as K

tim = K.read_json(K.C / 'queue' / 'confirm4' / 'TIMING.json')
cur = K.read_json(K.C / 'CURRENT.json')
comp = {c['name']: c for c in tim['comparisons']}
entry = dict(
    batch='confirm4 (three-arm: handoff / previous head n27 / current head n29)',
    chain_len=len(cur['chain']), head=cur['best_plan'],
    end_to_end=dict(saving_ms=comp['A_minus_C']['saving_ms'], CI95_ms=comp['A_minus_C']['CI95_ms'],
                    status=comp['A_minus_C']['status'], groups=comp['A_minus_C']['groups']),
    last_step_contribution=dict(saving_ms=comp['B_minus_C']['saving_ms'], CI95_ms=comp['B_minus_C']['CI95_ms'],
                                status=comp['B_minus_C']['status'], label='n29-f01-alu-chain'),
    previous_head_end_to_end=dict(saving_ms=comp['A_minus_B']['saving_ms'], CI95_ms=comp['A_minus_B']['CI95_ms']))
p = K.C / 'evidence' / 'CHAIN-CONFIRMATION.json'
d = K.read_json(p)
d.setdefault('measurements', []).append(entry)
d['authoritative'] = dict(
    kind='three-arm balanced ABC batch, 6 permutations x 30 samples/arm',
    end_to_end_ms=entry['end_to_end']['saving_ms'], CI95_ms=entry['end_to_end']['CI95_ms'],
    head=cur['best_plan'], chain_len=len(cur['chain']),
    conclusion='the accepted chain now measures %.4f ms end to end (handoff baseline -> current head); '
               'the last step (n29) contributes %.4f ms in the same batch'
               % (entry['end_to_end']['saving_ms'], entry['last_step_contribution']['saving_ms']))
K.write_json(p, d)
print('end-to-end %.6f CI [%.6f, %.6f]' % (entry['end_to_end']['saving_ms'],
                                           entry['end_to_end']['CI95_ms'][0], entry['end_to_end']['CI95_ms'][1]))
print('n29 contribution %.6f CI [%.6f, %.6f]' % (entry['last_step_contribution']['saving_ms'],
                                                 entry['last_step_contribution']['CI95_ms'][0],
                                                 entry['last_step_contribution']['CI95_ms'][1]))
