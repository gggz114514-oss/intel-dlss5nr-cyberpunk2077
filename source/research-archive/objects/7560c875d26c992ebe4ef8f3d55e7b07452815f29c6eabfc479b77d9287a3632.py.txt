from pathlib import Path

p = Path(__file__).with_name('COMMON.py')
lines = p.read_text(encoding='utf-8').splitlines()
new = r"""        return str(s).strip('"').replace('\\', '/').rstrip('/')"""
hit = 0
for i, l in enumerate(lines):
    if 'return str(s).replace(' in l:
        lines[i] = new
        hit += 1
if hit != 1:
    raise SystemExit('expected exactly one _norm return line, found %d' % hit)
p.write_text('\n'.join(lines) + '\n', encoding='utf-8')
print('patched', hit)
for l in lines:
    if 'return str(s)' in l:
        print('now:', l)
