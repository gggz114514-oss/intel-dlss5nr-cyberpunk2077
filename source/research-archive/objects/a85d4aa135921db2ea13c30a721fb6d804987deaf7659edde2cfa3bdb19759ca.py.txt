"""Compare small's shader set against vk102's (its ancestor), not against the 310.8 full net.

small was derived from vk102 with few changes, so the meaningful overlap question is
small-vs-vk102, not small-vs-our-baseline.
"""
import hashlib, json, re, sys
from collections import Counter
from pathlib import Path
sys.path.insert(0, str(Path(__file__).parent))
import COMMON as K

SMALL = Path('@NR_WORKSPACE@/work/vk102-small-310-9-16-20261019/'
             'package/manifest/port_network_runtime.txt')
VK = Path('@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/'
          'vk102-current-paired-20261026/vk102-path-normalized.txt')


def load(plan):
    raw = plan.read_text(encoding='utf-8-sig', errors='replace')
    out = []
    for line in raw.splitlines():
        if not line.startswith('PASS '):
            continue
        parts = line.split('"')
        if len(parts) < 4:
            continue
        p = Path(re.sub(r'\\{2,}', r'\\', parts[3].strip()))
        out.append((parts[1].strip(), p))
    return out


def shas(rows):
    m = {}
    for name, p in rows:
        if p.exists():
            m.setdefault(hashlib.sha256(p.read_bytes()).hexdigest(), []).append((name, p))
    return m


small_rows, vk_rows = load(SMALL), load(VK)
small, vk = shas(small_rows), shas(vk_rows)
print('small PASS %d (%d distinct shaders); vk102 PASS %d (%d distinct shaders)'
      % (len(small_rows), len(small), len(vk_rows), len(vk)))

shared = sorted(set(small) & set(vk))
shared_small_pass = sum(len(small[s]) for s in shared)
shared_vk_pass = sum(len(vk[s]) for s in shared)
print('\nshaders present in BOTH: %d' % len(shared))
print('  small PASS using them : %d / %d (%.1f%%)' % (shared_small_pass, len(small_rows),
                                                      100 * shared_small_pass / len(small_rows)))
print('  vk102 PASS using them : %d / %d' % (shared_vk_pass, len(vk_rows)))
only_small = sorted(set(small) - set(vk))
only_vk = sorted(set(vk) - set(small))
print('\nonly in small: %d shaders / %d PASS' % (len(only_small), sum(len(small[s]) for s in only_small)))
for s in only_small:
    print('   %s %2d PASS  %s' % (s[:10], len(small[s]), small[s][0][1].stem.split('__')[-1]))
print('only in vk102: %d shaders / %d PASS' % (len(only_vk), sum(len(vk[s]) for s in only_vk)))
K.write_json(K.C / 'evidence' / 'SMALL-VS-VK102.json', dict(
    small_pass=len(small_rows), vk_pass=len(vk_rows), shared_shaders=len(shared),
    shared_small_pass=shared_small_pass, shared_vk_pass=shared_vk_pass,
    only_small=[dict(sha=s, pass_=len(small[s]), name=small[s][0][1].stem.split('__')[-1]) for s in only_small],
    only_vk=[dict(sha=s, pass_=len(vk[s]), name=vk[s][0][1].stem.split('__')[-1]) for s in only_vk]))
