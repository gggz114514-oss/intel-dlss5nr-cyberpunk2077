def opcode_delta():
    b = ST.Module(CARD / 'build/parent-raw.spv')
    c = ST.Module(CARD / 'build/candidate-raw.spv')
    hb = Counter(op for op, _, _, _ in b.instructions)
    hc = Counter(op for op, _, _, _ in c.instructions)
    out = []
    for op in sorted(set(hb) | set(hc)):
        d = hc.get(op, 0) - hb.get(op, 0)
        if d:
            out.append(dict(opcode=op, name=NB.get(op, ''), delta=d,
                            baseline=hb.get(op, 0), candidate=hc.get(op, 0)))
    return out, sum(hb.values()), sum(hc.values())


def per_function_fp():
    b = ST.Module(CARD / 'build/parent-raw.spv')
    c = ST.Module(CARD / 'build/candidate-raw.spv')
    br = ST.function_regions(b)
    cr = ST.function_regions(c)
    p2c, _, _ = N.match_functions(b, c), None, None
    p2c = N.match_functions(b, c)[2]
    rows = []
    for pi, ci in sorted(p2c.items()):
        hb = Counter(b.instructions[x][0] for x in br[pi] if b.instructions[x][0] in FP_OPS)
        hc = Counter(c.instructions[x][0] for x in cr[ci] if c.instructions[x][0] in FP_OPS)
        rows.append(dict(parent_fn=pi, candidate_fn=ci,
                         parent_fp_ops=sum(hb.values()), candidate_fp_ops=sum(hc.values()),
                         equal=hb == hc,
                         detail={str(k): [hb.get(k, 0), hc.get(k, 0)] for k in sorted(set(hb) | set(hc))}))
    return rows
