import json, os

W = r"@NR_WORKSPACE@/work/dsh-fullnet-static-20261029"

def probe(path, label, maxdepth=3):
    print("\n" + "=" * 100)
    print("FILE", label, os.path.getsize(path), "bytes")
    o = json.load(open(path, encoding="utf-8"))
    def walk(x, d=0, key=""):
        pad = "  " * d
        if isinstance(x, dict):
            print("%s%s{%d keys}: %s" % (pad, key, len(x), list(x)[:14]))
            if d < maxdepth:
                for k in list(x)[:6]:
                    walk(x[k], d + 1, k + ": ")
        elif isinstance(x, list):
            print("%s%s[%d]" % (pad, key, len(x)))
            if x and d < maxdepth:
                walk(x[0], d + 1, "[0] ")
        else:
            s = str(x)
            print("%s%s= %s" % (pad, key, s[:200]))
    walk(o)
    return o
