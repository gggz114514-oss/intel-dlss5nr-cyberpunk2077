"""Task 1 (decisive): test the C64 canonical-layout hypothesis.

FP16 C64 canonical image layout (from c64_qx_producer.comp):
    idx = tile*1024 + (c/16)*256 + slot*16 + (c%16)      # [tile][g=4][slot=16][k=16]
INT8 token-major (qx / block*_out) layout:
    idx = tile*1024 + slot*64 + c                        # [tile][slot][c=64]

So a naive reshape(-1,64) of the canonical tensor puts
    col = (slot%4)*16 + k
i.e. the column index mixes slot bits with channel bits -> a period-16 pattern
in per-column statistics that has nothing to do with the channel axis.
"""
import json
import numpy as np
from pathlib import Path

B = Path("@NR_WORKSPACE@")
W = B / "work/dsh-fullnet-static-20261029"
CHAIN = W / "C00-color-baseline/runs/diag-chain62-66"
TEACH = (B / "work/codex-astra-fp16-int8-audit-20261024/int8-quality-first-20261029"
         / "real-int8-24/dump-fp16")


def load(p):
    return np.fromfile(p, dtype="<f2").astype(np.float32)


def canon_to_token(a):
    """[tile][g][slot][k] -> [tile][slot][g*16+k]"""
    t = a.reshape(-1, 4, 16, 16)
    return np.ascontiguousarray(t.transpose(0, 2, 1, 3)).reshape(-1)


def token_to_canon(a):
    t = a.reshape(-1, 16, 4, 16)
    return np.ascontiguousarray(t.transpose(0, 2, 1, 3)).reshape(-1)


def stats(g, t, C=64):
    n = min(g.size, t.size)
    g = g[:n].reshape(-1, C).astype(np.float64)
    t = t[:n].reshape(-1, C).astype(np.float64)
    gr = np.sqrt((g * g).mean(0)); tr = np.sqrt((t * t).mean(0))
    ratio = gr / np.maximum(tr, 1e-30)
    G = g; T = t
    R = (G.T @ T) / np.maximum(np.outer(np.sqrt((G * G).sum(0)), np.sqrt((T * T).sum(0))), 1e-30)
    diag = np.diag(R)
    off = np.abs(R - np.diag(diag))
    return dict(
        rms_gpu=float(np.sqrt((g * g).mean())), rms_teach=float(np.sqrt((t * t).mean())),
        ratio_global=float(np.sqrt((g * g).mean() / (t * t).mean())),
        relL2=float(np.linalg.norm(g - t) / np.linalg.norm(t)),
        corr=float(np.corrcoef(g.ravel(), t.ravel())[0, 1]),
        slope=float((g * t).sum() / (t * t).sum()),
        ch_ratio_min=float(ratio.min()), ch_ratio_max=float(ratio.max()),
        ch_ratio_std=float(ratio.std()),
        ch_ratio_16=[round(float(x), 4) for x in ratio[:16]],
        xcorr_diag_mean=float(diag.mean()), xcorr_offdiag_max=float(off.max()),
    )


def main():
    out = {}
    for blk in (62, 63, 64, 65):
        g = load(CHAIN / f"chain-block{blk}_out.bin")
        t = load(TEACH / f"dump_block{blk}_out.bin")
        variants = {
            "naive_reshape_both": stats(g, t),
            "teacher_canon_to_token": stats(g, canon_to_token(t)),
            "gpu_canon_to_token": stats(token_to_canon(g), t),
            "teacher_canon_gpu_canon": stats(token_to_canon(g), canon_to_token(t)),
        }
        out[blk] = variants
    print(json.dumps(out, indent=1))


if __name__ == "__main__":
    main()
