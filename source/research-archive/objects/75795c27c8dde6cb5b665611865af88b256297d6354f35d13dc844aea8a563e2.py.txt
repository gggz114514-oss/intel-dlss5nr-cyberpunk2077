from pathlib import Path
import json
N=Path(__file__).resolve().parent
old={};new={}
def emit(dst,stage,m,n,ks,seed):
 key=(stage,m,n);assert key not in dst
 dst[key]=[seed]+[("MMA",k,16) for k in ks]+["half"]+(["nr_act_h"] if stage=="expand" else [])
for sg in range(8):
 for rd in range(2):
  for m in [(sg//4+4*rd)*8,(sg//4+4*rd)*8+16]:
   for n in [sg%4*32,sg%4*32+16]:emit(old,"expand",m,n,[0,16],"zero")
 for n in [0,16]:emit(old,"contract",sg*8,n,range(0,128,16),"half(X*rs)")
 for j in range(3):
  for n in [j*32,j*32+16]:emit(old,"qkv",sg*8,n,[0,16],"zero")
for sg in range(4):
 for m in [sg*16,sg*16+8]:
  for ki in range(8):emit(new,"expand",m,ki*16,[0,16],"zero")
  for n in [0,16]:emit(new,"contract",m,n,range(0,128,16),"half(X*rs)")
  for j in range(3):
   for n in [j*32,j*32+16]:emit(new,"qkv",m,n,[0,16],"zero")
assert old==new
