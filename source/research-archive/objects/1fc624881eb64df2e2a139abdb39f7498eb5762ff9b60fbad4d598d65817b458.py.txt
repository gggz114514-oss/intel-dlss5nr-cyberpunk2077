from pathlib import Path
import hashlib,json,shlex
A=Path('@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024')
D=A/'c32-boundary-audit-20261026'
p=A/'c32-no-gain-diagnosis-20261026/combined-no-root-body/plan.txt'
sha=lambda p:hashlib.sha256(p.read_bytes()).hexdigest()
assert sha(p)=='015da1d924457a436de18db2a0947001508e0d97c64f5cd20281a6c8ff32b019'
buffers=[];passes=[];history=[]
for line in p.read_text().splitlines():
 t=shlex.split(line)
 if not t:continue
 if t[0]=='BUFFER':buffers.append({'name':t[1],'bytes':int(t[2])})
 elif t[0]=='PASS':passes.append({'ordinal':len(passes),'name':t[2],'spv':t[3],'binds':[]})
 elif t[0]=='BIND':passes[-1]['binds'].append(list(map(int,t[1:])))
 elif t[0]=='HISTORY':history.append(list(map(int,t[1:])))
assert len(passes)==232
edges={}
for b in [2,15,289,453,480]:
 edges[b]={**buffers[b],'bindings':[{'ordinal':x['ordinal'],'name':x['name'],'slot':v[0]} for x in passes for v in x['binds'] if v[1]==b]}
assert [x['ordinal'] for x in edges[480]['bindings']]==[230,231]
assert not any(480 in h[:2] for h in history)
# Symbolically execute the existing B -> A permutation, retaining (token,feature).
v=[[(l//2,2*j+(l&1)) for j in range(8)] for l in range(32)]
for bit,pairs in [(4,[(0,1),(2,3),(4,5),(6,7)]),(8,[(0,2),(1,3),(4,6),(5,7)]),(16,[(0,4),(1,5),(2,6),(3,7)])]:
 old=[r[:] for r in v]
 for l in range(32):
  for a,b in pairs:
   src=l^bit;r=old[src][a if src&bit else b]
   v[l][a]=r if l&bit else old[l][a]
   v[l][b]=old[l][b] if l&bit else r
out=[v[(l&1)|((l>>3)&2)|((l&14)<<1)] for l in range(32)]
assert all(out[l][j]==(2*j+l//16,l%16) for l in range(32) for j in range(8))
# Exhaust all F20 tile owners, pixels and 16-channel block starts.
seen=bytearray(1344*768);cropped=0;addresses=0
for wy in range(97):
 for wx in range(169):
  for t in range(64):
   q,s=divmod(t,16);tx=2*wx-1+(q&1);ty=2*wy-1+(q>>1)
   if not (0<=tx<336 and 0<=ty<192):continue
   px=tx*4+s%4;py=ty*4+s//4;idx=py*1344+px
   assert seen[idx]==0;seen[idx]=1
   cropped+=int(px<1280 and py<720)
   for c in [0,16]:
    producer=((ty*336+tx)*2+c//16)*256+s*16+c%16
    token=((py//4)*336+px//4)*16+4*(py%4)+px%4
    consumer=(token//16*2+c//16)*256+token%16*16+c%16
    assert producer==consumer;addresses+=1
assert sum(seen)==1344*768 and cropped==1280*720
cost={'existing_full_write':1344*768*32*2,'existing_valid_read':1280*720*32*2,'proposed_valid_residual_write_and_read':2*1280*720*4*4}
cost['logical_bytes_saved']=cost['existing_full_write']+cost['existing_valid_read']-cost['proposed_valid_residual_write_and_read']
result={'status':'CPU_AUDIT_ONLY_NO_CANDIDATE_NO_GPU','baseline_sha256':sha(p),'edges':edges,'history':history,'entry_mapping':'A0[l][j]=(token=2*j+l//16, feature=l%16); A1 token +=8, j=0..3','entry_mapping_checks':256,'entry_source_shuffle_calls':{'prepare_B':8,'B_to_A':10,'total':18},'f20_unique_pixels':sum(seen),'f20_valid_crop_pixels':cropped,'address_block_start_checks':addresses,'cost_bytes_source_level_not_DRAM':cost}
(D/'EVIDENCE.json').write_text(json.dumps(result,indent=2),encoding='utf-8')
print(json.dumps({'checks':'PASS','cost':cost,'edges':edges},ensure_ascii=False))
