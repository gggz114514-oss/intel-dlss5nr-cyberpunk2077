import json, struct, statistics, hashlib
from pathlib import Path
D=Path(__file__).resolve().parent
def read(p):return json.loads(p.read_text(encoding='utf-8-sig'))
prep=read(D/'direct-clock/PREPARATION.json');gx,gy,_=prep['grid']
out={'status':'PILOT_ONLY','unit':'raw subgroup-clock ticks; NOT Vulkan timestamp units','jobs':[]}
for mode in ['control','null','whole']:
 root=D/'run'/('direct-f20-'+mode);rp=root/'port_network_receipt.json';cp=root/'clock.bin'
 if not rp.exists():continue
 receipt=read(rp);row=next(x for x in receipt['per_pass_gpu'] if x['index']==230)
 entry={'id':mode,'target_samples_ms':row['samples_ms'],'target_median_ms':statistics.median(row['samples_ms']),'gold':hashlib.sha256((root/'frame_out.bin').read_bytes()).hexdigest()=='1755e9c2fadba97f08cb5b02b7a4294cde3675379bdec75352287505f7e129d6'}
 if cp.exists():
  data=cp.read_bytes();assert len(data)==prep['clock_buffer_bytes'];values=struct.unpack('<%dI'%(len(data)//4),data)
  records=[];unexpected=[];missing=[]
  for y in range(gy):
   for x in range(gx):
    selected=x%16==8 and y%16==8
    for sg in range(8):
     k=((y*gx+x)*8+sg)*4;a,b,c,d=values[k:k+4];nonzero=bool(a or b or c or d)
     if mode=='control':
      if nonzero:unexpected.append([x,y,sg])
     elif selected:
      if not nonzero:missing.append([x,y,sg])
      else:records.append({'x':x,'y':y,'sg':sg,'ticks':(((c|(d<<32))-(a|(b<<32)))&((1<<64)-1))})
     elif nonzero:unexpected.append([x,y,sg])
  entry.update({'expected_records':sum(x%16==8 for x in range(gx))*sum(y%16==8 for y in range(gy))*8 if mode!='control' else 0,'records':records,'missing':missing,'unexpected':unexpected})
  if records:
   ticks=[r['ticks'] for r in records];entry.update({'tick_median':statistics.median(ticks),'tick_min':min(ticks),'tick_max':max(ticks),'per_sg_median':{sg:statistics.median(r['ticks'] for r in records if r['sg']==sg) for sg in range(8)}})
 out['jobs'].append(entry)
(D/'MAIN-DIRECT-CLOCK-REVIEW.json').write_text(json.dumps(out,indent=2),encoding='utf-8')
for e in out['jobs']:print({k:v for k,v in e.items() if k not in ['records','target_samples_ms']})
