"""Build step for n32-f05-aluwr: parent byte reproduction + candidate compile.

CPU only (glslang + spv_validate_cpu).  No GPU, no plan dispatch.

identity: chain head = accepted/n29-f01-alu-chain-20261027/plan.txt (sha d8875665...)
  c64  source/target = fullnet-mechanism-transfer-20261026/f05-kv-consumer-layout/family/c64/candidate.{comp,spv}
  c128 source/target = .../family/c128/candidate.{comp,spv}
  c256 source/target = cards/n05a-f05-c256-mreuse/candidate.{comp,spv}   (n05a adopted in the chain head)
include dirs are the ones recorded in evidence/RECIPES.json / n05a BUILD.json (header pins).
"""
import json, shutil, sys, hashlib
from pathlib import Path

HERE = Path(__file__).parent
CAMP = HERE.parent.parent
sys.path.insert(0, str(CAMP))
import COMMON as K

D = K.D                                  # fullnet-mechanism-transfer-20261026
GROUPS = {
    "c64": dict(
        src=D / "f05-kv-consumer-layout/family/c64/candidate.comp",
        spv=D / "f05-kv-consumer-layout/family/c64/candidate.spv",
        inc=D.parent.parent / "zcode-fp16-fullnet-handoff-20261024/candidates/f01-f02-f05-f21/f05-c64-join-t3-v1/source",
        sha="0d83192b5368cc6074e00bbefed67ec71f8f540a9088d4d96752887390459641",
        ords=[7, 10, 13, 16, 215, 218, 221, 224]),
    "c128": dict(
        src=D / "f05-kv-consumer-layout/family/c128/candidate.comp",
        spv=D / "f05-kv-consumer-layout/family/c128/candidate.spv",
        inc=D.parent.parent / "zcode-fp16-fullnet-handoff-20261024/candidates/f01-f02-f05-f21/f05-c128-join-t3-v1/source",
        sha="20f4074fc70c32c0e74f0ad000053492e5f5c420259da6044e7a00cfc3ee3f4f",
        ords=[19, 22, 25, 28, 31, 34, 195, 198, 201, 204, 207, 210]),
    "c256": dict(
        src=CAMP / "cards/n05a-f05-c256-mreuse/candidate.comp",
        spv=CAMP / "cards/n05a-f05-c256-mreuse/candidate.spv",
        inc=D.parent.parent / "zcode-fp16-fullnet-handoff-20261024/candidates/f01-f02-f05-f21/f05-c256-join-t3-v1/source",
        sha="7a9576d15a18b4698b2e930c82b9658f2345e40479351e6ad0a65b034abc0af1",
        ords=[37, 40, 43, 46, 49, 52, 55, 58, 169, 172, 175, 178, 181, 184, 187, 190]),
}

BUILD = HERE / "build"
BUILD.mkdir(exist_ok=True)
res = {}


def sha(p):
    return hashlib.sha256(Path(p).read_bytes()).hexdigest()


for g, spec in GROUPS.items():
    d = BUILD / g
    d.mkdir(exist_ok=True)
    work = d / "parent.comp"
    shutil.copyfile(spec["src"], work)
    incs = [work.parent, spec["inc"]]
    raw = d / "parent-raw.spv"
    K.compile_glsl(work, raw, incs)
    r = dict(group=g, parent_source=str(spec["src"]), parent_source_sha=sha(spec["src"]),
             parent_source_copy=str(work), parent_source_copy_sha=sha(work),
             include_dir=str(spec["inc"]),
             parent_spv_target=str(spec["spv"]), parent_spv_target_sha=sha(spec["spv"]),
             rebuild_sha=sha(raw), rebuilt=sha(raw) == spec["sha"],
             expected_sha=spec["sha"], ordinals=spec["ords"])
    res[g] = r
    print("%-5s parent %s  rebuild %s  %s" % (g, spec["sha"][:16], sha(raw)[:16],
                                              "BYTE-MATCH" if r["rebuilt"] else "MISMATCH"))
    if not r["rebuilt"]:
        # keep the artefact for inspection; the contract allows recording a toolchain delta
        shutil.copyfile(raw, d / "parent-rebuild.spv")

(HERE / "build" / "parent-repro.json").write_text(json.dumps(res, indent=1), encoding="utf-8")
print("\nall parents reproduced:", all(v["rebuilt"] for v in res.values()))
