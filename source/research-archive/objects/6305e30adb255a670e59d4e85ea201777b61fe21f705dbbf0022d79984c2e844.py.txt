"""n21 step-7: build plan.txt on the chain head, replacing only ord 194 (replaced == 1)."""
import sys, json, difflib
from pathlib import Path

CARD = Path(__file__).resolve().parent
C = CARD.parent.parent
sys.path.insert(0, str(C))
import COMMON as K

HEAD, HEAD_SHA, _ = K.current_best()
assert HEAD_SHA == '719f100b5210249f315094b24160ecfd5b2fb707afb9a3543e1beec61d3f3f49', HEAD_SHA
parent_spv_sha = K.row_by_ordinal(194)['spv_sha256']

out = CARD / 'plan.txt'
replaced = K.make_plan(CARD / 'candidate.spv', out, parent_plan=HEAD, spv_shas=[parent_spv_sha])
assert replaced == 1, replaced

a = Path(HEAD).read_text(encoding='utf-8-sig').splitlines(keepends=True)
b = out.read_text(encoding='utf-8').splitlines(keepends=True)
assert len(a) == len(b), (len(a), len(b))
changed = [(i, x, y) for i, (x, y) in enumerate(zip(a, b)) if x != y]
assert len(changed) == 1, changed
i, x, y = changed[0]

recs = K.pass_lines_of(out)
assert len(recs) == 232
assert recs[194]['spv'].endswith('cards/n21-debug-regrelief/candidate.spv'), recs[194]['spv']
assert recs[193]['spv'] == K.pass_lines_of(HEAD)[193]['spv'], 'ord 193 must be untouched'

print('head plan     :', HEAD, HEAD_SHA)
print('candidate plan:', out, K.sha(out))
print('replaced      :', replaced, 'changed line index', i)
print('  parent:', x.strip())
print('  child :', y.strip())
print('ord193 unchanged:', recs[193]['spv'])

cp = K.cpu_check(out)
print('--- cpu-check ---')
print(cp.strip()[:3000])
(CARD / 'build' / 'cpu-check.txt').write_text(cp, encoding='utf-8')
