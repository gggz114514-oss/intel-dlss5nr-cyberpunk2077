"""Run NNCF on the real, equalized lift calibration subgraph; export contracts."""
from pathlib import Path
import json,numpy as np,onnx,nncf
from onnx import numpy_helper
from nncf.quantization.advanced_parameters import AdvancedQuantizationParameters,QuantizationParameters,AdvancedSmoothQuantParameters
from nncf.quantization.range_estimator import RangeEstimatorParametersSet
from nncf.common.quantization.structs import QuantizationScheme as QuantizationMode
Q=Path(__file__).resolve().parent;D=Q/'real-int8-24';O=D/'lift-equalized';cal=np.load(O/'CALIBRATION.npz')
model=onnx.load(Q/'reference-lift/lift.onnx')
model.graph.initializer[0].CopyFrom(numpy_helper.from_array((cal['weight']*cal['d'][:,None]).astype(np.float32),'weight'))
data=[]
for frame in [34189,41444,43654]:
 x=np.fromfile(D/f'feature-diagnostics/run-{frame}/features.bin',dtype='<f2').reshape(-1,16).astype(np.float32)/cal['d']
 data.append({'features':x})
params=AdvancedQuantizationParameters(
 activations_quantization_params=QuantizationParameters(num_bits=8,per_channel=False),
 weights_quantization_params=QuantizationParameters(num_bits=8,per_channel=True),
 activations_range_estimator_params=RangeEstimatorParametersSet.MINMAX,
 smooth_quant_alphas=AdvancedSmoothQuantParameters(matmul=-1))
quant=nncf.quantize(model,nncf.Dataset(data),subset_size=3,preset=nncf.QuantizationPreset.PERFORMANCE,target_device=nncf.TargetDevice.ANY,advanced_parameters=params)
onnx.save(quant,O/'lift-real-nncf.onnx');initializers={v.name:numpy_helper.to_array(v) for v in quant.graph.initializer};rows=[]
for n in quant.graph.node:
 if n.op_type in ['QuantizeLinear','DequantizeLinear','Clip']:
  rows.append({'name':n.name,'type':n.op_type,'inputs':list(n.input),'constants':{k:initializers[k].tolist() for k in n.input if k in initializers}})
(O/'NNCF-REAL-CONTRACT.json').write_text(json.dumps({'status':'REAL_THREE_FRAME_CALIBRATION_SUBGRAPH','vulkan_activation_range':[-127,127],'tool_constraint':'ANY rejected forced symmetric/signed/narrow settings. Tool defaults retained and actual constants exported for inspection; not a drop-in Vulkan pack.','equalization':'offline power-of-two diagonal, tool SmoothQuant disabled to avoid applying twice','quantizers':rows,'vulkan_candidate_activation_scale':float(cal['activation_scale']),'note':'Subgraph calibration only; not full ONNX model or holdout acceptance'},indent=2));print('NNCF real three-frame subgraph exported')
