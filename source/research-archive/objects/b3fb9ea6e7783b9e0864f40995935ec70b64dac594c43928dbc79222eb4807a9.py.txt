"""Exhaustive check of the identity that licenses the F01 candidate change.

parent  : a = base + mirror(y,H)*W*4 + mirror(x,W)*4
candidate: a = base +        y *W*4 +        x *4
with (x,y) = clamp(ij, 0, (W-1,H-1)).

Checks
  1. lemma  : mirror(i,size) == i for every i in [0,size-1]
  2. clamped: mirror(clamp(i,0,size-1),size) == clamp(i,0,size-1) for a wide i range
  3. address: the two address formulas are equal for every (W,H,ij) in the sweep
  4. corners: for random bilinear positions the four corner addresses are equal
Also emulates the parent's integer `%` (truncated, like SPIR-V OpSRem on signed) so the
lemma is checked against the same semantics the shader uses.
"""
import json, random
from pathlib import Path

CARD = Path(__file__).resolve().parents[1]
OUT = CARD / 'tools' / 'mirror-identity.json'


def mirror_parent(i, size):
    period = 2 * size
    j = i - period * int(i / period) if period else None   # truncated division (C/SPIR-V)
    if period == 0:
        return None
    if j < 0:
        j += period
    return j if j < size else period - 1 - j


def clamp(i, lo, hi):
    return lo if i < lo else (hi if i > hi else i)


def addr_parent(base, W, H, x, y):
    return base + mirror_parent(y, H) * W * 4 + mirror_parent(x, W) * 4


def addr_cand(base, W, H, x, y):
    return base + y * W * 4 + x * 4


def main():
    lemma = dict(cases=0, bad=[])
    clamped = dict(cases=0, bad=[])
    for size in range(1, 130):
        for i in range(0, size):
            lemma['cases'] += 1
            if mirror_parent(i, size) != i:
                lemma['bad'].append([i, size])
        for i in range(-400, 4200):
            clamped['cases'] += 1
            c = clamp(i, 0, size - 1)
            if mirror_parent(c, size) != c:
                clamped['bad'].append([i, size])

    addr = dict(cases=0, bad=[])
    for W in range(1, 65):
        for H in range(1, 65):
            for ix in (-3, -1, 0, 1, 2, W - 2, W - 1, W, 2 * W - 2):
                for iy in (-3, -1, 0, 1, 2, H - 2, H - 1, H, 2 * H - 2):
                    x = clamp(ix, 0, W - 1)
                    y = clamp(iy, 0, H - 1)
                    addr['cases'] += 1
                    if addr_parent(1000, W, H, x, y) != addr_cand(1000, W, H, x, y):
                        addr['bad'].append([W, H, ix, iy])

    rng = random.Random(20261027)
    corners = dict(cases=0, bad=[], sample=None)
    for _ in range(20000):
        W = rng.randint(1, 4096)
        H = rng.randint(1, 4096)
        fx = rng.uniform(-10.0, W + 10.0)
        fy = rng.uniform(-10.0, H + 10.0)
        ij = (int(fx // 1), int(fy // 1))
        hi = (W - 1, H - 1)
        seen = []
        for ox, oy in ((0, 0), (1, 0), (0, 1), (1, 1)):
            x = clamp(ij[0] + ox, 0, hi[0])
            y = clamp(ij[1] + oy, 0, hi[1])
            corners['cases'] += 1
            a = addr_parent(777, W, H, x, y)
            b = addr_cand(777, W, H, x, y)
            seen.append(a)
            if a != b:
                corners['bad'].append([W, H, ij, ox, oy])
        if corners['sample'] is None:
            corners['sample'] = dict(W=W, H=H, ij=list(ij), corner_addrs=seen)

    out = dict(
        note='mirror(i,size) == i on [0,size-1]; the history corner reads are clamped to that '
             'range before they reach the compile-unit mirror, so the candidate address is the '
             'same integer for every input in the sweep.',
        lemma_in_range=dict(cases=lemma['cases'], mismatches=len(lemma['bad']), bad=lemma['bad'][:5]),
        lemma_after_clamp=dict(cases=clamped['cases'], mismatches=len(clamped['bad']),
                               bad=clamped['bad'][:5]),
        address_equality=dict(cases=addr['cases'], mismatches=len(addr['bad']), bad=addr['bad'][:5]),
        corner_address_equality=dict(cases=corners['cases'], mismatches=len(corners['bad']),
                                     bad=corners['bad'][:5], sample=corners['sample']),
    )
    out['PASS'] = all(v['mismatches'] == 0 for k, v in out.items() if isinstance(v, dict) and 'mismatches' in v)
    OUT.write_text(json.dumps(out, indent=1), encoding='utf-8')
    print(json.dumps(out, indent=1)[:1200])
    print('PASS =', out['PASS'])


if __name__ == '__main__':
    main()
