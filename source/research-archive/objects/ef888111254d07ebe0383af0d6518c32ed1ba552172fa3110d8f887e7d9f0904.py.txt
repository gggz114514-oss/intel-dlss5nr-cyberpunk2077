# -*- coding: utf-8 -*-
"""Emit zoom crops of the four ROI families + stats for candidate boxes."""
import sys
from pathlib import Path

import numpy as np

sys.path.insert(0, str(Path(__file__).resolve().parent))
import pngio  # noqa: E402

Q = Path(r"@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/int8-quality-first-20261029")
OUT = Path(__file__).resolve().parent.parent
CROPS = OUT / "PNG" / "roi-refine"
ref = np.fromfile(Q / "dataset/day-street-static-formal-01/0-34189/network-output.bin",
                  dtype="<f4").reshape(2, 720, 1280, 4)[0, :, :, :3].astype(np.float64)
img8 = np.rint(np.clip(ref, 0, 1) * 255).astype(np.uint8)
W_ = np.array([0.2126, 0.7152, 0.0722])

CROPS_ = {
    "A-sky-signleft": (540, 110, 780, 290),
    "B-sign-right": (760, 110, 1000, 290),
    "C-ncpd": (400, 220, 680, 330),
    "D-blue-underpass": (600, 330, 820, 460),
    "E-neutral-right": (840, 170, 1140, 300),
}
for name, (x0, y0, x1, y1) in CROPS_.items():
    crop = img8[y0:y1, x0:x1]
    # 2x nearest-neighbour zoom for legibility, documented, no interpolation
    zoom = np.repeat(np.repeat(crop, 2, axis=0), 2, axis=1)
    pngio.write_rgb(CROPS / f"{name}.png", zoom)
    print(f"{name}: ({x0},{y0})-({x1},{y1}) zoom2x shape={zoom.shape}")

BOXES = {
    "P1-sky": (570, 155, 715, 212),
    "P1b-sky-strict": (600, 160, 700, 215),
    "P2-sign-right": (790, 165, 850, 240),
    "P2b-building-left": (690, 150, 760, 240),
    "P2c-building-mid": (700, 175, 760, 235),
    "P3-ncpd-letters": (455, 248, 620, 296),
    "P3b-ncpd-core": (470, 252, 610, 292),
    "P4-blue-underpass": (640, 360, 800, 440),
    "P5-blue-sidewalk": (100, 600, 300, 672),
    "P6-neutral-864-224": (864, 224, 896, 256),
    "P6b-neutral-1088-192": (1088, 192, 1120, 224),
    "P6c-neutral-1088-160": (1088, 160, 1120, 192),
    "P6d-concrete-beam": (540, 30, 700, 90),
    "P7-road": (850, 630, 1000, 700),
}
print()
for name, (x0, y0, x1, y1) in BOXES.items():
    c = ref[y0:y1, x0:x1].reshape(-1, 3)
    m = c.mean(0)
    L = c @ W_
    print(f"{name:24s} ({x0:4d},{y0:3d},{x1:4d},{y1:3d}) mean=({m[0]:.3f},{m[1]:.3f},{m[2]:.3f}) "
          f"L={L.mean():.3f} R-G={m[0]-m[1]:+.3f} B-G={m[2]-m[1]:+.3f} "
          f"fracL>0.7={float((L>0.7).mean()):.3f} fracL<0.25={float((L<0.25).mean()):.3f} "
          f"clip_any={float((c>=1.0).any(1).mean()):.4f}")
