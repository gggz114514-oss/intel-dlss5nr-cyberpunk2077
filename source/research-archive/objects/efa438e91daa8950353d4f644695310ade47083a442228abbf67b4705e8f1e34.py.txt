"""Generate FAMILY-COVERAGE.md: per-family status derived from the chain, the ranking and the cards."""
import json, sys
from collections import defaultdict
from pathlib import Path
sys.path.insert(0, str(Path(__file__).parent))
import COMMON as K

cur = K.read_json(K.C / 'CURRENT.json')
cov = K.read_json(K.C / 'evidence' / 'COVERAGE-NOW.json')
rank = K.read_json(K.C / 'evidence' / 'PASS-TIME-RANKING.json')
led = K.read_json(K.C / 'CAMPAIGN.json')

fam_rows = defaultdict(int)
fam_changed = defaultdict(int)
fam_time = {}
for t in cov['table']:
    fam_rows[t['family']] += 1
    if t['introduced_by_step'] != 0:
        fam_changed[t['family']] += 1
for f, ms in rank['families'].items():
    fam_time[f] = ms

# which cards touched each family
touched = defaultdict(list)
for c in led['cards']:
    r = c.get('card')
    fam = None
    p = K.C / 'cards' / r
    if (p / 'GPU-REQUEST.json').exists():
        try:
            fam = (K.read_json(p / 'GPU-REQUEST.json').get('card') or '')
        except Exception:
            fam = ''
    t = c.get('timing') or {}
    if not t:
        continue
    # map card id prefix to family heuristically from the id
    for f in fam_time:
        if f.lower() in r.lower():
            touched[f].append((r, t['saving_ms'], t['status']))

lines = ['# 族级覆盖与状态（按当前接受链头 %s）' % cur['best_plan']['sha256'][:12], '',
         '| 族 | PASS | 隔离时长 ms | 占比 | 链上已改 | 已测卡（同批 saving，正值=更快） | 状态 |',
         '|---|---|---|---|---|---|---|']
total_time = rank['sum_fixed_ms']
for f in sorted(fam_time, key=lambda x: -fam_time[x]):
    n = fam_rows.get(f, 0)
    ch = fam_changed.get(f, 0)
    cards = touched.get(f, [])
    cards_s = ', '.join('%s %+.4f(%s)' % (c[0].split('-')[0], c[1], 'acc' if c[2] == 'FASTER_95' else ('neg' if c[2] == 'SLOWER_95' else 'ns')) for c in cards) or '—'
    if ch and any(c[2] == 'FASTER_95' for c in cards):
        state = '已采纳 %d/%d 行' % (ch, n)
    elif ch:
        state = '已改 %d/%d 行（未采纳）' % (ch, n)
    elif cards:
        state = '已测未采纳'
    else:
        state = '未开卡'
    lines.append('| %s | %d | %.3f | %.1f%% | %d | %s | %s |' % (
        f, n, fam_time[f], 100 * fam_time[f] / total_time, ch, cards_s, state))

lines += ['', '说明：隔离时长来自交接的逐 PASS 计时（不可相加、非 ordinary）；"链上已改"指该族有多少 PASS 行的 SPV 在链上被换过。']
(K.C / 'FAMILY-COVERAGE.md').write_text('\n'.join(lines) + '\n', encoding='utf-8')
print('\n'.join(lines))
