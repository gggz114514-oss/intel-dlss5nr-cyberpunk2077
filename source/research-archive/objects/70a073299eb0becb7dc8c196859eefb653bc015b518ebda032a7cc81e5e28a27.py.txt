def kind(abc):
    ks = [(t.get("kind"), t.get("width")) for t in abc]
    if len(ks) >= 3 and all(k[0] == "float" and k[1] == 16 for k in ks[:2]) and ks[2][0] == "float":
        return "fp16"
    if len(ks) >= 2 and all(k[0] in ("int", "signed int", "unsigned int") for k in ks[:2]):
        return "int8"
    return "mixed"

            counts["all_fp16"] += 1
        elif s == {"int8"}:
            counts["all_int8"] += 1
        elif not s:
            counts["no_mma"] += 1

            counts["mixed"] += 1
