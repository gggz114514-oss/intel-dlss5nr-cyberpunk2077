#!/usr/bin/env python3
"""Per-variant residual counts for the C128 WO quantiser."""
import sys
from pathlib import Path

import numpy as np

sys.path.insert(0, str(Path(__file__).resolve().parent))
import gen_c128_wo_packs as G  # noqa: E402
import diag_wo_quantiser as V  # noqa: E402


def main():
    b = int(sys.argv[1]) if len(sys.argv) > 1 else 12
    info = G.parse_plan()[b]
    pack = np.fromfile(info["pack_path"], dtype="<f2")
    w = G.decode_w(pack, info["wo"])
    amax = np.abs(w).max(axis=1)
    sW = np.where(amax > 0, np.float32(amax) / np.float32(127.0),
                  np.float32(1.0)).astype(np.float32)
    shipped = (G.SHIPPED / f"c128_b{b}_wo.codes.i8.bin").read_bytes()
    old = np.frombuffer(shipped, np.int8).astype(np.int32)
    for name, fn in V.VARIANTS.items():
        q = np.clip(fn(w, sW), -127, 127).astype(np.int32).ravel()
        new = np.zeros(G.N * G.K, np.int32)
        new[V.OFF] = q
        d = old != new
        print("%-16s differing %5d  max|d| %d" % (name, int(d.sum()),
                                                 int(np.abs(old - new).max())))
    # where do the remaining differences sit relative to a tie?
    q = np.clip(V.VARIANTS["f64_div_f64s"](w, sW), -127, 127).astype(np.int32).ravel()
    new = np.zeros(G.N * G.K, np.int32)
    new[V.OFF] = q
    d = np.nonzero(old != new)[0]
    print("f64_div_f64s differing", d.size)
    if d.size:
        print(" sample", [(int(p), int(old[p]), int(new[p])) for p in d[:8]])


if __name__ == "__main__":
    main()
