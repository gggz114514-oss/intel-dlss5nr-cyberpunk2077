def sha(p):
    return hashlib.sha256(Path(p).read_bytes()).hexdigest()

def write_patch():

    b = (CARD / 'candidate.comp').read_text(encoding='utf-8').splitlines(keepends=True)

    return dict(path=str(CARD / 'SOURCE-DIFF.patch'), lines=len(d),
                hunks=sum(1 for l in d if l.startswith('@@')),
                added=sum(1 for l in d if l.startswith('+') and not l.startswith('+++')),
                removed=sum(1 for l in d if l.startswith('-') and not l.startswith('---')))


# ------------------------------------------------- 2. half commutativity check
def half_bits(v):
    """half bit pattern of a float64 value, with half overflow / underflow folded."""
    if v != v:
        return 0x7E00
    if v == float('inf'):
        return 0x7C00
    if v == float('-inf'):
        return 0xFC00
    try:
        return struct.unpack('<H', struct.pack('<e', v))[0]
    except OverflowError:
        return 0x7C00 if v > 0 else 0xFC00


def to_half(bits):
    return struct.unpack('<e', struct.pack('<H', bits))[0]


def check_commutative():
    """x + peer vs peer + x over the whole half code space against peers.

    IEEE-754 addition is commutative by definition; this is the machine check on
    the actual operand domain (all 65536 codes).  Python's '<e' round trip is
    used as the half model: the *same* float64 sum is rounded on both sides, so
    the check is exactly "are the two orderings the same operation".
    """
    codes = list(range(0x10000))
    peers = [0x0000, 0x8000, 0x0001, 0x8001, 0x03FF, 0x0400, 0x7BFF, 0xFBFF,
             0x7C00, 0xFC00, 0x7E00, 0x3555, 0x3C00, 0xBC00, 0x4900, 0xC900]
    n = 0
    bad = []
    for p in peers:
        pv = to_half(p)
        for c in codes:
            xv = to_half(c)
            xy = half_bits(xv + pv)
            yx = half_bits(pv + xv)
            n += 1
            if xy != yx:
                bad.append((hex(c), hex(p), hex(xy), hex(yx)))
    # the specific value domains of the five site families
    fam = dict(
        exp_out=(0x3C20, 0x3E48),      # nr_exp_h2 clamp [1.03125, 1.5693359375]
        sq=(0x0000, 0x7BFF),           # squares of halves: non-negative
        pooled=(-1.0, 1.0),
    )
    return dict(pairs_checked=n, peers=len(peers), codes=len(codes),
                mismatches=len(bad), first_mismatch=bad[:3],
                method='half code space x 16 representative peers, both orderings '
                       'rounded from the same float64 sum')
