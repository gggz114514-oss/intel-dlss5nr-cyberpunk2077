"""Compact structural dump of the F14 native ISA around the scale-load sites.

Read-only.  Prints instruction indices (address, opcode, operand tail) for the
interesting opcodes in a byte window, plus labels/branches.
"""
import re
import sys
from pathlib import Path

Q = Path('@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/'
         'fullnet-continuous-20261027/queue/n12-vit-qkv/native/captures')
BASE = Q / 'native-native-base' / 'ISA' / 'CS_asm2ef42bf87fecdff2_simd32_0000000000000019_24_orig.asm'
CAND = Q / 'native-native-cand' / 'ISA' / 'CS_asm220ce9995b78f549_simd32_0000000000000019_24_orig.asm'
INSN = re.compile(r'^\s*/\*\s*\[([0-9A-Fa-f]+)\]\s*\*/\s*(.*)$')
OPC = re.compile(r'^(?:\([^)]*\)\s+)*([a-z][a-z0-9_.]*)\s*(.*)$')

KEEP = ('send.ugm', 'send.gtwy', 'cmp', 'goto', 'join', 'jmpi', 'sync.nop', 'sync.allrd',
        'dpas.8x8', 'math.rsqt', 'add', 'shl', 'mov', 'and', 'sel', 'or', 'add3', 'mad')


def main(path, lo, hi):
    lines = path.read_text(encoding='utf-8', errors='replace').splitlines()
    print('== %s  window %X..%X' % (path.name, lo, hi))
    for i, ln in enumerate(lines):
        m = INSN.match(ln)
        if not m:
            if re.match(r'^\s*(L\d+|//\s*BBO|BBO)', ln):
                pass
            continue
        a = int(m.group(1), 16)
        if not (lo <= a <= hi):
            continue
        body = m.group(2)
        o = OPC.match(body)
        op = o.group(1) if o else '?'
        tail = o.group(2) if o else body
        if op in KEEP or 'bss' in body or 'goto' in body or 'join' in body:
            print('%7X  %-10s %s' % (a, op, tail[:150]))


if __name__ == '__main__':
    which = sys.argv[1] if len(sys.argv) > 1 else 'base'
    p = BASE if which == 'base' else CAND
    lo = int(sys.argv[2], 16) if len(sys.argv) > 2 else 0x16000
    hi = int(sys.argv[3], 16) if len(sys.argv) > 3 else 0x19000
    main(p, lo, hi)
