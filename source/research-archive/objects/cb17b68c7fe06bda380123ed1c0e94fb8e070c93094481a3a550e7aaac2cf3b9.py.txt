"""Unified diff parent source -> candidate source (this card), and a read-only
ISA census of the parent identity capture that the prediction is priced against.

The ISA capture belongs to cards/n09b-f09-sx-stage (read only here); nothing is
written outside this card directory.
"""
from pathlib import Path
import json, re, sys
import difflib

CARD = Path(__file__).resolve().parent
sys.path.insert(0, str(CARD.parent.parent))
import COMMON as K

PARENT = CARD.parent / 'n09b-f09-sx-stage' / 'candidate.comp'
CAND = CARD / 'candidate.comp'

# parent identity ISA (n09b native capture, read only): this is the shader the
# chain head plan actually binds for these 16 PASS.
ASM = (CARD.parent.parent / 'queue' / 'n09b-f09-sx-stage' / 'native' / 'captures' /
       'native-native-cand' / 'ISA' / 'CS_asm873c258babc364b2_simd32_0000000000000012_17_orig.asm')


def make_patch():
    a = PARENT.read_text(encoding='utf-8-sig').splitlines(keepends=True)
    b = CAND.read_text(encoding='utf-8-sig').splitlines(keepends=True)
    diff = list(difflib.unified_diff(
        a, b, fromfile='a/cards/n09b-f09-sx-stage/candidate.comp',
        tofile='b/cards/n09c-f09-b-share/candidate.comp', n=4))
    (CARD / 'SOURCE-DIFF.patch').write_text(''.join(diff), encoding='utf-8')
    return dict(path=str(CARD / 'SOURCE-DIFF.patch'), sha256=K.sha(CARD / 'SOURCE-DIFF.patch'),
                lines=len(diff),
                added=sum(1 for l in diff if l.startswith('+') and not l.startswith('+++')),
                removed=sum(1 for l in diff if l.startswith('-') and not l.startswith('---')),
                hunks=sum(1 for l in diff if l.startswith('@@')))


def isa_census(path):
    text = Path(path).read_text(encoding='utf-8', errors='replace')
    ins = re.findall(r'^/\* \[([0-9A-F]+)\]  \*/ (.*)$', text, re.M)
    mn, msgs, dpas, syncs = {}, {}, 0, 0
    maxr, first_dpas = -1, None
    for off, body in ins:
        body = body.strip()
        b2 = re.sub(r'^\((?:W|[0-9]+\|M0)\)\s+', '', body)
        op = b2.split()[0] if b2.split() else '?'
        mn[op] = mn.get(op, 0) + 1
        if op.startswith('dpas'):
            dpas += 1
            if first_dpas is None:
                first_dpas = int(off, 16)
        if op.startswith('sync'):
            syncs += 1
        m = re.search(r'//.*;\s*(\S+)$', body)
        if op.startswith('send') and m:
            k = '%s %s' % (op, m.group(1))
            msgs[k] = msgs.get(k, 0) + 1
        for r in re.findall(r'\br(\d+)\.', body):
            maxr = max(maxr, int(r))
    total_send = sum(v for k, v in msgs.items() if k.startswith('send'))
    return dict(path=str(path), instructions=len(ins), sends=total_send, dpas=dpas,
                syncs=syncs, max_r=maxr, first_dpas_hex=first_dpas,
                mnemonic_histogram=dict(sorted(mn.items(), key=lambda x: -x[1])),
                messages=dict(sorted(msgs.items(), key=lambda x: -x[1])),
                mov=mn.get('mov', 0), add=mn.get('add', 0), mul=mn.get('mul', 0),
                shl=mn.get('shl', 0), sync_nop=mn.get('sync.nop', 0),
                sync_allwr=mn.get('sync.allw', 0) + mn.get('sync.allwr', 0))


def main():
    out = dict(patch=make_patch())
    if ASM.exists():
        out['parent_isa'] = isa_census(ASM)
        out['parent_isa']['source'] = ('cards/n09b-f09-sx-stage native capture (read only); this is '
                                       'the SPV the chain-head plan binds for these 16 PASS')
    K.write_json(CARD / 'PARENT-EVIDENCE.json', out)
    print(json.dumps(out['patch'], indent=1))
    if 'parent_isa' in out:
        p = out['parent_isa']
        print('parent ISA: instr %d sends %d dpas %d syncs %d max_r %d first_dpas %s mov %d add %d mul %d shl %d sync.nop %d'
              % (p['instructions'], p['sends'], p['dpas'], p['syncs'], p['max_r'],
                 p['first_dpas_hex'], p['mov'], p['add'], p['mul'], p['shl'], p['sync_nop']))
        print('messages:', json.dumps(p['messages'], indent=1))


if __name__ == '__main__':
    main()
