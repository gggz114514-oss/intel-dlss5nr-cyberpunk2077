#!/usr/bin/env python3
"""Diagnose the residual difference between recomputed and shipped C128 WO codes.

Answers two different questions separately:
  * permutation?  -> same code multiset, different positions (layout wrong)
  * rounding?     -> same positions, |diff| == 1 and the scaled value sits at a tie
"""
import sys
from pathlib import Path

import numpy as np

sys.path.insert(0, str(Path(__file__).resolve().parent))
import gen_c128_wo_packs as G  # noqa: E402

b = int(sys.argv[1]) if len(sys.argv) > 1 else 12
passes = G.parse_plan()
info = passes[b]
i8b, scb, rec = G.build_block(info)
new = np.frombuffer(i8b, np.int8).astype(np.int32)
old = np.frombuffer((G.SHIPPED / f"c128_b{b}_wo.codes.i8.bin").read_bytes(),
                    np.int8).astype(np.int32)
d = np.nonzero(old != new)[0]
print("block", b, "differing positions", d.size, "of", old.size)
print("same multiset:", np.array_equal(np.sort(old), np.sort(new)))
print("max |diff|", int(np.abs(old - new).max()), "diff histogram",
      {int(k): int(v) for k, v in zip(*np.unique(old - new, return_counts=True))})
# recompute the scaled value at each differing position
pack = np.fromfile(info["pack_path"], dtype="<f2")
w = G.decode_w(pack, info["wo"])
amax = np.abs(w).max(axis=1)
sW = np.where(amax > 0, np.float32(amax) / np.float32(127.0), np.float32(1.0)).astype(np.float32)
invs = (np.float32(1.0) / sW).astype(np.float32)
ni, ki = np.arange(G.N)[:, None], np.arange(G.K)[None, :]
off = ((ni // 16) * 2048 + (ki // 32) * 512 + (ni % 16) * 32 + (ki % 32)).ravel()
pos2nk = {int(o): (int(n), int(k)) for o, n, k in zip(off, np.repeat(ni, G.K, 1).ravel(),
                                                      np.tile(ki, (G.N, 1)).ravel())}
rows = []
for p in d[:12]:
    n, k = pos2nk[int(p)]
    sc = float(w[n, k] * invs[n])
    rows.append((int(p), n, k, float(w[n, k]), sc, int(old[p]), int(new[p])))
for r in rows:
    print("pos=%6d n=%3d k=%3d w=%+.6f scaled=%+.6f frac=%+.4f old=%+4d new=%+4d"
          % (r[0], r[1], r[2], r[3], r[4], r[4] - np.floor(r[4]) - 0.5, r[5], r[6]))
# how many differing positions are exactly at a tie (|frac-0.5| < 1e-6)?
frac = np.abs(w * invs[:, None] - np.round(w * invs[:, None]))
tie = frac[np.array([pos2nk[int(p)] for p in d])[:, 0],
           np.array([pos2nk[int(p)] for p in d])[:, 1]] if d.size else np.array([])
print("differing positions with |scaled-round| < 1e-6:", int(np.count_nonzero(tie < 1e-6)))
print("median distance to integer at differing positions:", float(np.median(tie)) if d.size else 0)
