"""n36-f05-arena-shrink: generate the three candidate sources from the chain-head parents.

Every edit is an explicitly asserted literal replacement, so the produced candidate is
provably the parent plus exactly these edits (and SOURCE-DIFF.patch is generated from
the parent/candidate pair, not hand-written).

The single mechanism: the V plane stops being a third 4096 B plane per head.  The V
projection is a pure function of gY and the V weight tiles, so it is relocated to after
the last Q/K read of the workgroup and stores into the retired Q plane.  Token
ownership of the V projection moves from {8t, 8t+32} to {16t, 16t+8} so this
subgroup's V store footprint is exactly the 512 half of Q it read itself.
"""

GROUPS = {
    'c64': dict(src=D / 'f05-kv-consumer-layout/family/c64/candidate.comp',
                edits=C64_EDITS, old_decl=DECL_OLD, pv_loads=16),
    'c128': dict(src=D / 'f05-kv-consumer-layout/family/c128/candidate.comp',
                 edits=C128_EDITS, old_decl=DECL_OLD, pv_loads=16),

OLD_BASE = '(gl_SubgroupID / 4u) * 6144u'
NEW_BASE = '(gl_SubgroupID / 4u) * 4096u'
# The PV A-operand reads address the V plane with the intra-section literal VBASE
# (`... + 4096u + key*32u + dim`).  VBASE is NOT the section stride: when V moves into
# the Q plane this literal must become 0u, or PV reads 4096 half past the head section
# (and out of the 8192-half declaration for the second head).  This was the E-class bug
# of the first build and is now an asserted edit with an exact count.
PV_OLD = '(gl_SubgroupID / 4u) * 6144u) + 4096u + '
PV_NEW = '(gl_SubgroupID / 4u) * 6144u) + 0u + '


def main():
    report = {}
    patches = []
    for g, spec in GROUPS.items():
        text = pathlib.Path(spec['src']).read_text(encoding='utf-8')
        edits = []
        for old, new in spec['edits']:
            n = text.count(old)
            assert n == 1, (g, 'anchor count %d (expected 1)' % n, old[:90])
            text = text.replace(old, new, 1)
            edits.append(dict(anchor=old.splitlines()[0][:96], occurrences=n))
        # declaration
        n = text.count(spec['old_decl'])
        assert n == 1, (g, 'declaration count', n)
        text = text.replace(spec['old_decl'], CXX_DECL_NEW, 1)
        edits.append(dict(anchor='shared float16_t qkv[12288]', occurrences=n))
        # PV reads the V plane with an intra-section literal: VBASE 4096 -> 0
        n_pv = text.count(PV_OLD)
        assert n_pv == spec['pv_loads'], (g, 'PV base occurrences', n_pv, spec['pv_loads'])
        text = text.replace(PV_OLD, PV_NEW)
        edits.append(dict(anchor='PV A-load base  + 4096u -> + 0u',
                          occurrences=n_pv,
                          note='the V plane moved into the Q plane; this literal is the '
                               'intra-section VBASE, not the section stride'))
        # arena base: every per-head section stride 6144 -> 4096 half
        n = text.count(OLD_BASE)
        assert n >= 20, (g, 'base stride occurrences', n)
        text = text.replace(OLD_BASE, NEW_BASE)
        edits.append(dict(anchor=OLD_BASE, occurrences=n, replacement=NEW_BASE))
        # residual V-plane offsets in the QKV pass must be gone
        left = text.count('4096u + m0')
        assert left == 0, (g, 'V store left in the QKV pass', left)
        out = CARD / ('candidate-%s.comp' % g)
        out.write_text(text, encoding='utf-8')
        parent = pathlib.Path(spec['src']).read_text(encoding='utf-8').splitlines(True)
        cand = text.splitlines(True)
        diff = list(difflib.unified_diff(
            parent, cand,
            fromfile='A-parent/%s' % pathlib.Path(spec['src']).name,
            tofile='B-n36/candidate-%s.comp' % g, n=4))
        patches.append(''.join(diff))
        report[g] = dict(parent=str(spec['src']), candidate=str(out),
                         edits=edits, lines=len(cand), diff_lines=len(diff),
                         barrier_calls=text.count('barrier();'))
        print('%-5s lines=%4d diff=%4d barrier()=%d base_stride_edits=%d'
              % (g, len(cand), len(diff), text.count('barrier();'), n))
    (CARD / 'SOURCE-DIFF.patch').write_text('\n'.join(patches), encoding='utf-8')
    (CARD / 'build/candidate-gen.json').write_text(json.dumps(report, indent=2),
                                                   encoding='utf-8')
    return 0


if __name__ == '__main__':
    sys.exit(main())
