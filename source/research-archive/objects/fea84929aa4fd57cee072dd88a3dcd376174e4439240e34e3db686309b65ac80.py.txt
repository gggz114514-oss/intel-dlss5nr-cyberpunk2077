from pathlib import Path
import re,json
D=Path('@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/fullnet-mechanism-transfer-20261026');S=D/'f05-four-weight-window-20261027';T=D/'f10-weight-return-separation-20261027'
out=[]
for w,i,n in [('c64',5,24),('c128',9,48)]:
 for arm,root in [('parent',T),('candidate',S)]:
  p=next((root/'captures').glob(f'*/ISA/*_{i}_orig.asm'));lines=[]
  for l in p.read_text().splitlines():
   m=re.search(r'\[([0-9A-F]+)\]',l)
   if m:lines.append((int(m[1],16),l))
  loads={};edges=[];dp=0
  for pc,l in lines:
   if 'load.ugm.d32x8.a32' in l:
    reg=int(re.search(r'send.ugm\s+\(16\|M0\)\s+r(\d+)',l)[1]);loads[reg]=(pc,l)
   if 'dpas.8x8' in l:
    regs=re.search(r'\s+r(\d+):hf\s+r(\d+)\.0:hf',l);assert regs,l
    r=int(regs[1]);lp,ll=loads[r];edges.append({'load_pc':lp,'consume_pc':pc,'weight_bank':r,'A_bank':int(regs[2]),'send':ll,'dpas':l});dp+=1
    if dp==n:break
  assert len(edges)==n and len({e['load_pc'] for e in edges})==n
  # Static issue-to-source-use payload intervals, not hardware latency or allocated GRF.
  events=sorted([(e['load_pc'],1) for e in edges]+[(e['consume_pc'],-1) for e in edges]);live=peak=0
  for pc,delta in events:live+=delta;peak=max(peak,live)
  phases=[edges[k:k+6] for k in range(0,n,6)]
  arows=[{'pc':pc,'bank':int(re.search(r'send.ugm\s+\(1\|M0\)\s+r(\d+)',l)[1]),'line':l} for pc,l in lines if pc<=edges[-1]['consume_pc'] and 'load.ugm.d32x8t.a32' in l]
  out.append({'width':w,'arm':arm,'QKV_dpas':n,'unique_weight_sends':n,'issue_to_use_peak_weight_blocks':peak,'A_row_reads':arows,'phases':phases})
(S/'QKV-WEIGHT-INTERVALS.json').write_text(json.dumps(out,indent=2),encoding='utf-8')
for r in out:print(r['width'],r['arm'],r['QKV_dpas'],r['issue_to_use_peak_weight_blocks'])
