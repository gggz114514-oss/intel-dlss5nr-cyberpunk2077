"""why do 1383 records show SPV-match but header mismatch?"""
import json, re, sys
from pathlib import Path
from collections import Counter

CARD = Path(__file__).parent
W = Path('@NR_WORKSPACE@')
BASELINE = (W / 'work/codex-astra-fp16-int8-audit-20261024/fullnet-mechanism-transfer-20261026/'
            'f10-weight-return-separation-20261027/plan.txt')
TOKEN_RE = re.compile(r'"([^"]*)"|(\S+)')


def parse_plan(p):
    out = []
    for line in Path(p).read_text(encoding='utf-8-sig', errors='replace').splitlines():
        if not line.startswith('PASS '):
            continue
        t = [a if a else b for a, b in TOKEN_RE.findall(line)]
        out.append(dict(ordinal=len(out), pass_no=int(t[1]), name=t[2], spv=t[3], rest=t[4:]))
    return out


base = parse_plan(BASELINE)
j = json.loads((CARD / 'evidence' / 'NATIVE-INDEX.json').read_text(encoding='utf-8'))
caps = [c for c in j['captures'] if c.get('header_diff_ordinals')]
print('captures with header diffs:', len(caps), 'of', len(j['captures']))
print('diff count histogram:', Counter(len(c['header_diff_ordinals']) for c in caps).most_common(8))

field = Counter()
examples = []
for c in caps[:400]:
    p = (c['plan'] or {}).get('path')
    if not p or not Path(p).exists():
        continue
    other = parse_plan(p)
    if len(other) != 232:
        continue
    for o in c['header_diff_ordinals'][:60]:
        b, x = base[o]['rest'], other[o]['rest']
        for i in range(min(len(b), len(x))):
            if b[i] != x[i]:
                field[i] += 1
    if len(examples) < 4:
        o = c['header_diff_ordinals'][0]
        examples.append((c['isa_dir'], o, base[o]['rest'], other[o]['rest']))
print('differing field index counts:', sorted(field.items()))
for e in examples:
    print('\n', e[0])
    print('   ordinal', e[1])
    print('   baseline:', e[2])
    print('   capture :', e[3])
