`c32_f2_i8.spv` (f52e5264...) byte for byte; every splice below is anchored on a
unique line and asserted, and the script writes a machine-checkable account.

Six splices:
  R1  binding + shared declarations   (int8 K/V/Q/P staging, new b9/b10)

  R4  Y -> int8 publishing            (replaces the two NRH_B joins)
  R5  QKV                             (24 FP16 MMA -> 12 int8 MMA)
  R6  post-norm .. stores             (16 score + 16 PV + 8 output -> 8 + 8 + 4)

R2 = """// ---- C32-I8 body helpers (c32-f1-f2) --------------------------------------
// Weight tile bases inside C32I8BodyWeights (bytes).  An 8x32 I8A tile occupies
// 1024 bytes of address space with 256 live codes (stride 32, LCOL), exactly the
// convention the deployed FFN pack already uses.
#define C32I8B_WQ 32768u
#define C32I8B_WK 36864u
#define C32I8B_WV 40960u
#define C32I8B_WO 45056u
#define C32I8B_WP 49152u
// Static per-output-channel weight scales inside C32I8BodyScales.
#define C32I8B_SQ 0u
#define C32I8B_SK 32u
#define C32I8B_SV 64u
#define C32I8B_SO 96u
#define C32I8B_SP 128u
// Shared int8 staging carved out of sE8, which is dead once the contract has
// consumed E: Q8[sg] is [tok][dim] stride 32 (512 B), P8[sg] is two [tok][key]
// tiles of 512 B.  Total 2048 + 4096 = 6144 of sE8's 8192 bytes.
#define C32I8B_Q8 0u
#define C32I8B_P8 2048u

// Largest |component| a lane holds in one 8x16 half fragment.
float nr_i8_amax(NRH_CH a) {
    float m = 0.0;
    for (int i = 0; i < 4; ++i) m = max(m, abs(float(a[i])));
    return m;
}
// Subgroup max by shuffle-xor (the module only enables the shuffle extension,
// so subgroupMax/subgroupAdd are deliberately not used).
float nr_i8_sgmax(float v) {
    for (uint m = 16u; m > 0u; m >>= 1u) v = max(v, subgroupShuffleXor(v, uint(m)));
    return v;
}
// sX8 staging with a caller-supplied inverse scale: element (row, col) of the
// fragment lands at base + col*32 + rowoff + row, i.e. the [token][channel]
// stride-32 I8B tile that coopMatLoad(...,32,LCOL) reads.
void nr_i8_put_ys(NRH_CH a, uint rowoff, uint base, float inv) {
    for (int i = 0; i < 4; ++i) {
        uvec2 rc = nrh_coord_CH(uint(i));
        sX8[base + rc.y * 32u + rowoff + rc.x] = int8_t(nr_i8_code(a[i], inv));
    }
}
// sE8 Q8 tile: [sg][token][dim] stride 32.
void nr_i8_put_q(NRH_CH a, uint rowoff, uint base, float inv) {
    for (int i = 0; i < 4; ++i) {
        uvec2 rc = nrh_coord_CH(uint(i));
        sE8[C32I8B_Q8 + base + rc.y * 32u + rowoff + rc.x] = int8_t(nr_i8_code(a[i], inv));
    }
}
// Published K/V tiles: [key 0..63][lane 0..31] stride 32, key = keybase + token.
void nr_i8_put_k(NRH_CH a, uint rowoff, uint keybase, float inv) {
    for (int i = 0; i < 4; ++i) {
        uvec2 rc = nrh_coord_CH(uint(i));
        shK8[(keybase + rc.y) * 32u + rowoff + rc.x] = int8_t(nr_i8_code(a[i], inv));
    }
}
void nr_i8_put_v(NRH_CH a, uint rowoff, uint keybase, float inv) {
    for (int i = 0; i < 4; ++i) {
        uvec2 rc = nrh_coord_CH(uint(i));
        shV8[(keybase + rc.y) * 32u + rowoff + rc.x] = int8_t(nr_i8_code(a[i], inv));
    }
}
// sE8 P tile: [sg][key tile 32][token][key%32] stride 32.
void nr_i8_put_p(NRH_CH a, uint keyrow, uint tileoff, uint base, float inv) {
    for (int i = 0; i < 4; ++i) {
        uvec2 rc = nrh_coord_CH(uint(i));
        sE8[C32I8B_P8 + base + tileoff + rc.y * 32u + keyrow + rc.x] =
            int8_t(nr_i8_code(a[i], inv));
    }
}
// Dequantisation, folded exactly once: static per-output-channel weight scale
// times the activation scale that was used to build the int8 operand.
NRH_C nr_i8_deq2(I8C a, uint sbase, float sAct) {
    I8F f = I8F(a);
    NRH_C o = NRH_C(0.0);
    for (int i = 0; i < 4; ++i) {
        uvec2 rc = nrh_coord_C(uint(i));
        o[i] = f[i] * (fsc2[sbase + rc.x] * sAct);
    }
    return o;
}
NRH_C nr_i8_deq2_add(I8C a, uint sbase, float sAct, NRH_C seed) {
    I8F f = I8F(a);
    NRH_C o = NRH_C(0.0);
    for (int i = 0; i < 4; ++i) {
        uvec2 rc = nrh_coord_C(uint(i));
        o[i] = f[i] * (fsc2[sbase + rc.x] * sAct) + seed[i];
    }
    return o;
}
// Pure dynamic-operand dequantisation (no static weight column): score and PV
// both sit between two self-calibrated activations.
NRH_C nr_i8_deq_s(I8C a, float s) {
    I8F f = I8F(a);
    NRH_C o = NRH_C(0.0);
    for (int i = 0; i < 4; ++i) o[i] = f[i] * s;
    return o;
}

    o.append("        NRH_C zq0, zq1, zq2, zq3, zk0, zk1, zk2, zk3, zv0, zv1, zv2, zv3;")
    o.append("        {")
    o.append("            I8A wa; I8C a0;")
    for grp, (wbase, sbase, dst) in enumerate([
            (QW, "C32I8B_SQ", "zq"), (KW, "C32I8B_SK", "zk"), (VW, "C32I8B_SV", "zv")]):
        o.append("            // %s" % ("QKV"[grp]))
        for f in range(4):
            o.append("            a0 = I8C(0);")
            o.append("            coopMatLoad(wa, wb8, %s + %du, 32u, LCOL);" % (wbase, 1024 * f))
            o.append("            a0 = coopMatMulAdd(wa, yq8, a0);")
            o.append("            %s%d = nr_i8_deq2(a0, %s + %du, sY);" % (dst, f, sbase, 8 * f))
    o.append("        }")
    for f in range(4):

    o.append("        f16vec2 norm_pair=nr_norm32_pair(q0,q1,q2,q3,k0,k1,k2,k3);")
    o.append("        float16_t scalar0 = float16_t(fb[uint(pc.p2.z)]);")
    for f in range(4):
        for j in range(4):
            o.append("        {")
            o.append("            float16_t qn=norm_pair.x, kn=norm_pair.y;")
            o.append("            float16_t qr = q%d[%d] * qn;" % (f, j))
            o.append("            qr = qr * scalar0;")
            o.append("            q%d[%d] = qr;" % (f, j))
            o.append("            k%d[%d] = k%d[%d] * kn;" % (f, j, f, j))

    o.append("        }")
    return "\n".join(o)


def pv_and_out():
    o = []
    o.append("        // Ctx^T = V^T * P^T; reciprocal and per-entry H already applied.")
    o.append("        // INT8: 4 channel fragments x 2 key chunks of 32 (was 16 FP16 MMA).")
    o.append("        I8C tc0 = I8C(0);")
    o.append("        I8C tc1 = I8C(0);")
    o.append("        I8C tc2 = I8C(0);")
    o.append("        I8C tc3 = I8C(0);")
    o.append("        {")
    o.append("            I8A a; I8B pb;")
    for kc in range(2):
        o.append("            coopMatLoad(pb, sE8, C32I8B_P8 + sg * 1024u + %du, 32u, LCOL);" % (512 * kc))
        for f in range(4):
            o.append("            coopMatLoad(a, shV8, %du, 32u, LCOL);" % (1024 * kc + 8 * f))
            o.append("            tc%d = coopMatMulAdd(a, pb, tc%d);" % (f, f))
    o.append("        }")
    for f in range(4):
        o.append("        NRH_C tcd%d = nr_i8_deq_s(tc%d, sV * sP);" % (f, f))
    for f in range(4):
        o.append("        NRH_CH ctx%d = nrh_narrow(tcd%d);" % (f, f))
        o.append("        if (debug) coopMatStore(ctx%d, dbg, 18432u+t0*32u+%du, 32u, LCOL);" % (f, 8 * f))
    o.append("        // ---- C32-I8 Ctx tile for the output projection ----------")
    o.append("        float sC;")
    o.append("        {")
    o.append("            float ca = max(max(nr_i8_amax(ctx0), nr_i8_amax(ctx1)), max(nr_i8_amax(ctx2), nr_i8_amax(ctx3)));")
    o.append("            float cs = nr_i8_sgmax(ca);")
    o.append("            sC = max(cs, 1e-12) / 127.0;")
    o.append("            float invC = 127.0 / max(cs, 1e-12);")
    for f in range(4):
        o.append("            nr_i8_put_ys(ctx%d, %du, sg * 512u, invC);" % (f, 8 * f))
    o.append("            subgroupMemoryBarrierShared();")
    o.append("            subgroupBarrier();")
    o.append("        }")
    o.append("        I8B cb8;")
    o.append("        coopMatLoad(cb8, sX8, sg * 512u, 32u, LCOL);")
    o.append("        // Projection residual is H(Y*ars) BEFORE original K0,16.")
    for f in range(4):
        o.append("        NRH_C oc%d;" % f)
        o.append("        {")
        o.append("            NRH_CH scale;")
        o.append("            coopMatLoad(scale,wb,uint(pc.p2.y)+%du,0u,LCOL);" % (8 * f))
        o.append("            NRH_CH seed = y%d * scale;" % f)
        o.append("            oc%d = nrh_widen_exact(seed);" % f)
        o.append("        }")
    o.append("        {")
    o.append("            I8A wa; I8C a0;")
    for f in range(4):
        o.append("            a0 = I8C(0);")
        o.append("            coopMatLoad(wa, wb8, C32I8B_WO + %du, 32u, LCOL);" % (1024 * f))
        o.append("            a0 = coopMatMulAdd(wa, cb8, a0);")
        o.append("            oc%d = nr_i8_deq2_add(a0, C32I8B_SO + %du, sC, oc%d);" % (f, 8 * f, f))
    o.append("        }")
    for f in range(4):
        o.append("        NRH_CH o%d = nrh_narrow(oc%d);" % (f, f))
        o.append("        if (debug) coopMatStore(o%d,dbg,20480u+t0*32u+%du,32u,LCOL);" % (f, 8 * f))
        o.append("        bool o_oob%d;" % f)
        o.append("        uint o_base%d=nr_tok_addr(t0,%du,wx,wy,shift,tiles_x,o_oob%d);" % (f, 8 * f, f))
        o.append("        if (!o_oob%d) coopMatStore(o%d,imgout,o_base%d,16u,LCOL);" % (f, f, f))
    return "\n".join(o)


def main():
    src = SRC.read_text(encoding="utf-8")
    lines = src.split("\n")
    acc = {"parent": str(SRC), "splices": []}

    def expect(old, new, tag):
        nonlocal src
        n = src.count(old)
        if n != 1:
            die("%s: anchor count %d (expected 1)" % (tag, n))
        src = src.replace(old, new, 1)
        acc["splices"].append({"id": tag, "old_bytes": len(old.encode()),
                               "new_bytes": len(new.encode())})

    expect(R1_OLD, R1_NEW, "R1-declarations")
    expect("void nrf2_project_and_blend() {", R2 + "\nvoid nrf2_project_and_blend() {",
           "R2-helpers")
    expect(R3_OLD, R3_NEW, "R3-projection")
    expect(R4_OLD, R4_NEW, "R4-y-quantise")

    # R5: the whole Q/K/V section, from its banner comment up to (excluding) the
    # norm comment.  Anchor on both ends.
    i0 = src.index("        // Q/K/V share the two resident Y operands; original K0,16.")
    i1 = src.index("        // Norm and reciprocal are per token; no subgroupAdd or alternate tree.")
    if i1 <= i0:
        die("R5 span")
    old = src[i0:i1]
    src = src[:i0] + qkv_block() + "\n" + src[i1:]
    acc["splices"].append({"id": "R5-qkv", "old_bytes": len(old.encode()),
                           "new_bytes": len(qkv_block().encode()),
                           "old_fp16_mma": old.count("coopMatMulAdd"),
                           "new_int8_mma": qkv_block().count("coopMatMulAdd")})

    # R6: from the norm block through the last imgout store.
    j0 = src.index("        // Norm and reciprocal are per token; no subgroupAdd or alternate tree.")
    j1 = src.index("        if (!o_oob3) coopMatStore(o3,imgout,o_base3,16u,LCOL);")
    j1 += len("        if (!o_oob3) coopMatStore(o3,imgout,o_base3,16u,LCOL);")
    old6 = src[j0:j1]
    new6 = "\n".join([norm_and_publish(), score_block(), denom_and_p(),
                      pv_and_out()])
    src = src[:j0] + new6 + src[j1:]
    acc["splices"].append({"id": "R6-attention", "old_bytes": len(old6.encode()),
                           "new_bytes": len(new6.encode()),
                           "old_fp16_mma": old6.count("coopMatMulAdd"),
                           "new_int8_mma": new6.count("coopMatMulAdd"),
                           "old_coopmat_load": old6.count("coopMatLoad"),
                           "new_coopmat_load": new6.count("coopMatLoad")})

    OUT.parent.mkdir(parents=True, exist_ok=True)
    OUT.write_text(src, encoding="utf-8")
    acc["out"] = str(OUT)
    acc["out_bytes"] = len(src.encode())
    acc["out_lines"] = src.count("\n") + 1
    acc["total_fp16_mma_removed"] = 24 + (old6.count("coopMatMulAdd"))
    ACC.parent.mkdir(parents=True, exist_ok=True)
    ACC.write_text(json.dumps(acc, indent=1), encoding="utf-8")
    print(json.dumps(acc, indent=1))


if __name__ == "__main__":
