def cmd_parent():
    out = CARD / 'build'
    out.mkdir(parents=True, exist_ok=True)
    raw = out / 'parent-raw.spv'
    K.compile_glsl(CONTRACT['source'], raw, [CONTRACT['pin']])
    res = dict(source=str(CONTRACT['source']), source_sha256=K.sha(CONTRACT['source']),
               source_sha_matches=(K.sha(CONTRACT['source']) == CONTRACT['source_sha']),
               include_dir=str(CONTRACT['pin']),
               raw_sha256=K.sha(raw), target_sha256=K.sha(CONTRACT['spv']),
               raw_match=(K.sha(raw) == CONTRACT['spv_sha']))
    if res['raw_match']:
        shutil.copyfile(raw, CARD / 'parent-rebuild.spv')
    print(json.dumps(res, indent=2))
    return res


def cmd_cand():
    out = CARD / 'build'
    out.mkdir(parents=True, exist_ok=True)
    spv = CARD / 'candidate.spv'
    K.compile_glsl(CARD / 'candidate.comp', spv, [CARD, CONTRACT['pin']])
    sha = K.sha(spv)
    print('candidate sha256', sha, 'bytes', spv.stat().st_size)
    try:
        K.validate_spv(spv)
        print('validate: OK')
    except Exception as e:
        print('validate: FAIL', e)
    # swap back to the pinned dir only -> must NOT reproduce (header shadow proof)
    alt = out / 'noshaodow.spv'
    K.compile_glsl(CARD / 'candidate.comp', alt, [CONTRACT['pin'], CARD])
    print('sha with pinned headers first:', K.sha(alt),
          '(must differ from candidate unless the edit is inert)')
    return dict(sha256=sha, bytes=spv.stat().st_size)

                out_plan=str(out), out_plan_sha256=K.sha(out), replaced=n)

def cmd_check():

    (CARD / 'build' / 'cpu-check.txt').write_text(txt, encoding='utf-8')
    return txt


def spv_census(path):
    m = ST.Module(Path(path))
    ops = Counter()
    nc = set()
    body = m.words if hasattr(m, 'words') else None
    # use the library's own instruction walk if available
    try:
        nc = set(ST.no_contraction_ids(m))
    except Exception:
        nc = set()
    try:
        for ins in m.instructions():
            ops[ins.opcode] += 1
    except Exception:
        # fallback: scan words naively (opcode = word & 0xffff)
        for w in (body or []):
            ops[w & 0xFFFF] += 1
    return dict(instructions=sum(ops.values()), opcodes={str(k): v for k, v in sorted(ops.items())},
                no_contraction=sorted(nc), nc_count=len(nc))
