from pathlib import Path

p = Path(__file__).with_name('counters_two_arm.py')
t = p.read_text(encoding='utf-8')
old = "    res[aid] = list((out / ('spec-' + aid)).rglob('hardware-counters.json'))[0]"
new = ("    hits = list(out.rglob('hardware-counters.json'))\n"
       "    res[aid] = [h for h in hits if ('run-' + aid) in str(h)][0]")
assert t.count(old) == 1, t.count(old)
p.write_text(t.replace(old, new, 1), encoding='utf-8')
print('patched')
