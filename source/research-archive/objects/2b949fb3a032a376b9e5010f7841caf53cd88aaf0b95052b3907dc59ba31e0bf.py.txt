"""Three-arm same-batch confirmation: handoff baseline (A), previous head (B), current head (C).

Six balanced ABC permutations, 30 samples per arm per permutation, all samples kept,
two-sided t(df=5) on the six permutation means. This is the only way to compare the
end-to-end gain of two chain states without cross-batch drift.

Usage: python confirm3.py <name>
"""
import json, sys
from pathlib import Path
sys.path.insert(0, str(Path(__file__).parent))
import COMMON as K
import STAGE
import ANALYZE

PERM3 = [('A', 'B', 'C'), ('A', 'C', 'B'), ('B', 'A', 'C'), ('B', 'C', 'A'), ('C', 'A', 'B'), ('C', 'B', 'A')]
name = sys.argv[1] if len(sys.argv) > 1 else 'confirm3'
cur = K.read_json(K.C / 'CURRENT.json')
chain = cur['chain']
C_plan = Path(cur['best_plan']['path'])
B_plan = Path(chain[-2]['path'])
A_plan = K.BASELINE
arms = dict(A=(A_plan, K.sha(A_plan)), B=(B_plan, K.sha(B_plan)), C=(C_plan, K.sha(C_plan)))
out = K.C / 'queue' / name
out.mkdir(parents=True, exist_ok=True)

jobs = []
for pi, order in enumerate(PERM3):
    for aid in order:
        p, s = arms[aid]
        jobs.append(K.ordinary_job('perm-%d-%s' % (pi, aid), p, s,
                                   out / 'run' / ('perm-%d-%s' % (pi, aid)), aid, pi))
spec = dict(stage='ordinary', owner='LUNA-CONTINUOUS-CONFIRM3',
            authorization='Main (this session) authorized the three-arm chain confirmation only: handoff baseline, previous head, current head, gold required.',
            output_root=str(out), identities=[str(a[0]) for a in arms.values()] + [str(K.RUNNER)],
            jobs=jobs, rules='three-arm same-batch confirmation; all samples retained')
sp = out / 'spec-ordinary.json'
K.write_json(sp, spec)
manifest, ps1 = STAGE.prepare(spec)
r = STAGE.run(manifest, ps1)
s = STAGE.summarize(spec['output_root'])
K.write_json(out / 'STAGE-SUMMARY.json', s)
print('ordinary', s['status'])
if s['status'] != 'COMPLETED':
    print([(j['id'], j['status'], j['frame']) for j in s['jobs']])
    sys.exit(1)
rep = ANALYZE.report(out / 'LUNA-ORDINARY.json',
                     [dict(arms=('A', 'C')), dict(arms=('B', 'C')), dict(arms=('A', 'B'))],
                     out / 'TIMING.json')
for c in rep['comparisons']:
    print('%s: %s saving=%.6f CI95=[%.6f, %.6f] groups=%d' % (
        c['name'], c['status'], c['saving_ms'], c['CI95_ms'][0], c['CI95_ms'][1], c['groups']))
