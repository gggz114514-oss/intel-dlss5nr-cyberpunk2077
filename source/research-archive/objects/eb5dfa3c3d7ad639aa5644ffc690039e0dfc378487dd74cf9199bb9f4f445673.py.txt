"""SPIR-V structural comparison for the n11-vit-ffn card (CPU only)."""
import json, struct, sys, collections
from pathlib import Path

sys.path.insert(0, r'@NR_WORKSPACE@/build/zcode-fp16-compute-backend-20261021/scripts')
import spv_tools_local as ST

OPS = {
    248: 'OpLabel', 249: 'OpBranch', 250: 'OpBranchConditional', 251: 'OpSwitch',
    245: 'OpPhi', 246: 'OpLoopMerge', 247: 'OpSelectionMerge', 252: 'OpKill',
    54: 'OpFunction', 56: 'OpFunctionEnd', 57: 'OpFunctionCall',
    205: 'OpCooperativeMatrixLoadKHR', 208: 'OpCooperativeMatrixStoreKHR',
    207: 'OpCooperativeMatrixMulAddKHR', 4458: 'OpCooperativeMatrixMulAddKHR(4458)',
    61: 'OpLoad', 62: 'OpStore', 126: 'OpFma',
    81: 'OpConvertFToH', 90: 'OpConvertFToS', 112: 'OpConvertSToF',
    15: 'OpFNegate', 129: 'OpFAdd', 132: 'OpFMul', 128: 'OpFDiv',
    230: 'OpSelect', 177: 'OpUMin', 173: 'OpUMax',
}
NO_CONTRACTION = 44


def words(path):
    data = Path(path).read_bytes()
    w = list(struct.unpack('<%dI' % (len(data) // 4), data))
    assert w[0] == 0x07230203, path
    return w


def instrs(w):
    i = 5
    out = []
    while i < len(w):
        op, count = w[i] & 0xFFFF, w[i] >> 16
        out.append((op, w[i + 1:i + count], i))
        i += count
    return out


def census(path):
    w = words(path)
    c = collections.Counter()
    for op, ops, off in instrs(w):
        c[OPS.get(op, 'op%d' % op)] += 1
    # loops: OpLoopMerge count, and for each, whether the merge block contains a back edge
    return dict(path=str(path), bytes=len(Path(path).read_bytes()), words=len(w),
                counts=dict(c))


if __name__ == '__main__':
    out = {}
    for p in sys.argv[1:]:
        m = ST.Module(Path(p))
        nc = ST.no_contraction_ids(m)
        d = census(p)
        d['no_contraction_count'] = len(nc)
        d['no_contraction_ids_sample'] = sorted(nc)[:12]
        out[Path(p).stem] = d
    print(json.dumps(out, indent=1))
