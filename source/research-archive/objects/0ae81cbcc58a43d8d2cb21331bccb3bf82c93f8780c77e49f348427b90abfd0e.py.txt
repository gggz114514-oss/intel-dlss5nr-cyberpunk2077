"""Assemble SOURCE-DIFF.patch and extend BUILD.json with the verification block.

Everything is derived from files already on disk (parents, candidates, plan,
spv-diff.json, reader-set.json, mapping-proof.json, fold-shape.json).
"""
import json
import difflib
import sys
from pathlib import Path

CARD = Path(__file__).resolve().parent
C = CARD.parents[1]
sys.path.insert(0, str(C))
import COMMON as K

PAIR = [
    ('producer ord 193',
     Path('@NR_WORKSPACE@/candidates/amd-netorg-t3b-20261018/'
          'c128-canonical-only-v1/source/c3x_body_c128_r09.comp'),
     CARD / 'candidate-193.comp'),
    ('consumer ord 194',
     Path('@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/'
          'fullnet-mechanism-transfer-20261026/contract-scale-fragments/family/c128/candidate.comp'),
     CARD / 'candidate-194.comp'),
]


def rel(p):
    try:
        return Path(p).resolve().relative_to(Path('@NR_WORKSPACE@')).as_posix()
    except Exception:
        return Path(p).as_posix()


def main():
    out = []
    for label, parent, cand in PAIR:
        a = parent.read_text(encoding='utf-8-sig').splitlines(keepends=True)
        b = cand.read_text(encoding='utf-8').splitlines(keepends=True)
        d = difflib.unified_diff(a, b, fromfile='a/' + rel(parent), tofile='b/' + rel(cand),
                                 n=3, lineterm='\n')
        out.append('# ---- %s: parent -> candidate ----\n# parent sha256 %s\n# cand   sha256 %s\n%s\n'
                   % (label, K.sha(parent), K.sha(cand), ''.join(d)))

    # plan appendix (2 lines, SPV paths only)
    base = K.current_best()[0]
    plan = CARD / 'plan.txt'
    a = base.read_text(encoding='utf-8-sig').splitlines(keepends=True)
    b = plan.read_text(encoding='utf-8').splitlines(keepends=True)
    d = difflib.unified_diff(a, b, fromfile='a/' + rel(base), tofile='b/' + rel(plan),
                             n=0, lineterm='\n')
    out.append('# ---- plan (appendix, not source): chain head -> card plan ----\n%s\n'
               % ''.join(d))
    (CARD / 'SOURCE-DIFF.patch').write_text(''.join(out), encoding='utf-8')

    # ---- BUILD.json extension
    bj = json.loads((CARD / 'BUILD.json').read_text(encoding='utf-8'))
    sd = json.loads((CARD / 'spv-diff.json').read_text(encoding='utf-8'))
    rs = json.loads((CARD / 'reader-set.json').read_text(encoding='utf-8'))
    mp = json.loads((CARD / 'mapping-proof.json').read_text(encoding='utf-8'))
    fs = json.loads((CARD / 'fold-shape.json').read_text(encoding='utf-8'))
    bj['float_NC_checked'] = {
        'ord_193': {'parent': sd['passes']['ord_193']['parent_nc'],
                    'candidate': sd['passes']['ord_193']['candidate_nc']},
        'ord_194': {'parent': sd['passes']['ord_194']['parent_nc'],
                    'candidate': sd['passes']['ord_194']['candidate_nc']},
        'decoration_scanned': 44,
        'note': 'both PASSes are on the raw recipe (evidence/RECIPES.json nc_recipe.mode=raw, '
                'nc_count 0); the candidate introduces no new NoContraction decoration',
    }
    bj['spv_structural_diff'] = {
        'ord_193': {'parent_instructions': sd['passes']['ord_193']['parent_instructions'],
                    'candidate_instructions': sd['passes']['ord_193']['candidate_instructions'],
                    'opcode_count_delta': sd['passes']['ord_193']['opcode_count_delta'],
                    'removed_operand_words_only': True,
                    'changed': sd['passes']['ord_193']['removed_instructions']},
        'ord_194': {'parent_instructions': sd['passes']['ord_194']['parent_instructions'],
                    'candidate_instructions': sd['passes']['ord_194']['candidate_instructions'],
                    'opcode_count_delta': sd['passes']['ord_194']['opcode_count_delta'],
                    'note': 'glslang renumbers ids after the extra IMul, so the instruction '
                            'multiset diff is unreadable; the opcode multiset delta is the '
                            'authoritative statement: only OpIMul changes (45 -> 49, the four '
                            'k0*8 terms). No float, conversion, dpas or cm-load/store count '
                            'changes.'},
    }
    bj['fold_shape_evidence'] = {
        'ord_193_parent': fs['ord_193_parent'],
        'ord_193_candidate': fs['ord_193_candidate'],
        'ord_194_parent': fs['ord_194_parent'],
        'ord_194_candidate': fs['ord_194_candidate'],
        'spv_diff_json': str(CARD / 'spv-diff.json'),
        'fold_shape_json': str(CARD / 'fold-shape.json'),
    }
    bj['reader_set_evidence'] = {
        'file': str(CARD / 'reader-set.json'),
        'resource_208_swin_56_M': {
            'writers': rs['resources'][0]['writers'],
            'readers': rs['resources'][0]['readers'],
            'bound_not_used': rs['resources'][0]['bound_not_used'],
            'bound_by': rs['resources'][0]['bound_by'],
        },
    }
    bj['mapping_proof'] = {'file': str(CARD / 'mapping-proof.json'), 'verdict': mp['verdict'],
                           'checks': mp['checks']}
    bj['plan_pass_lines_changed'] = 2
    bj['artifacts'] = {
        'candidate_193_comp': K.ident(CARD / 'candidate-193.comp'),
        'candidate_193_spv': K.ident(CARD / 'candidate-193.spv'),
        'candidate_194_comp': K.ident(CARD / 'candidate-194.comp'),
        'candidate_194_spv': K.ident(CARD / 'candidate-194.spv'),
        'plan': K.ident(CARD / 'plan.txt'),
        'source_diff_patch': K.ident(CARD / 'SOURCE-DIFF.patch'),
        'parent_rebuild_193': K.ident(CARD / 'parent-rebuild-193.spv'),
        'parent_rebuild_194': K.ident(CARD / 'parent-rebuild-194.spv'),
    }
    bj['status'] = 'BUILT / AWAITING-GPU (native first, then ordinary)'
    K.write_json(CARD / 'BUILD.json', bj)
    print(json.dumps(bj['artifacts'], indent=2))
    print('patch bytes', (CARD / 'SOURCE-DIFF.patch').stat().st_size)
    return 0


if __name__ == '__main__':
    sys.exit(main())
