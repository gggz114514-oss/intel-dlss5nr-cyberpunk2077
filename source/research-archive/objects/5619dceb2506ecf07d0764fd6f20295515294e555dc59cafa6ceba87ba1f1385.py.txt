"""Does the candidate really emit the intended order?  (SPIR-V = IGC's input)

For each group and arm: position of every InverseSqrt (GLSL.std.450 #32) and of every
subgroup shuffle in the expanded post-inline stream, the number of reduction shuffles
that come before the first InverseSqrt, and the distance from each InverseSqrt to its
first consumer.
"""
import json, sys
from pathlib import Path

HERE = Path(__file__).parent
CAMP = HERE.parent.parent
sys.path.insert(0, str(CAMP))
import COMMON as K
sys.path.insert(0, str(K.SPV_TOOLS))
import spv_tools_local as ST  # noqa: E402

sys.path.insert(0, str(HERE))
from _b4_spv import expand, parse_functions  # noqa: E402

PARENT = {
    "c64": K.D / "f05-kv-consumer-layout/family/c64/candidate.spv",
    "c128": K.D / "f05-kv-consumer-layout/family/c128/candidate.spv",
    "c256": CAMP / "cards/n05a-f05-c256-mreuse/candidate.spv",
}
CAND = {g: HERE / ("candidate-%s.spv" % g) for g in ("c64", "c128", "c256")}
INVSQRT = 32
SHUF_XOR = 345        # OpGroupNonUniformShuffle  (broadcast form)
SHUF = 346            # OpGroupNonUniformShuffleXor (the 4 tree steps)
SHUF_BOTH = (345, 346)


def flat(path):
    m = ST.Module(Path(path))
    funcs, entry = parse_functions(m)
    out = []
    expand(funcs, entry, ("root",), {}, out)
    return out


def report(path):
    f = flat(path)
    rs = [n["idx"] for n in f if n["op"] == 12 and len(n["ops"]) > 3 and n["ops"][3] == ("lit", INVSQRT)]
    xor = [n["idx"] for n in f if n["op"] == SHUF]
    both = [n["idx"] for n in f if n["op"] in SHUF_BOTH]
    return dict(invs=len(rs), rs_idx=rs,
                rsqt_span=(rs[-1] - rs[0]) if rs else 0,
                shuffles_total=len(both), shuffle_xor=len(xor),
                shuffles_before_first_rsqrt=len([i for i in both if rs and i < rs[0]]),
                xors_before_first_rsqrt=len([i for i in xor if rs and i < rs[0]]),
                shuffles_after_last_rsqrt=len([i for i in both if rs and i > rs[-1]]))


out = {}
for g in ("c64", "c128", "c256"):
    p, c = report(PARENT[g]), report(CAND[g])
    out[g] = dict(parent=p, candidate=c)
    print("=" * 76)
    print("%s  InverseSqrt: parent %d / candidate %d" % (g, p["invs"], c["invs"]))
    print("   reduction shuffles BEFORE the first rsqrt : %d -> %d"
          % (p["xors_before_first_rsqrt"], c["xors_before_first_rsqrt"]))
    print("   span first..last rsqrt (instructions)     : %d -> %d"
          % (p["rsqt_span"], c["rsqt_span"]))
    print("   shuffles after the last rsqrt             : %d -> %d"
          % (p["shuffles_after_last_rsqrt"], c["shuffles_after_last_rsqrt"]))
    print("   total shuffles (broadcast+xor) %d -> %d" % (p["shuffles_total"], c["shuffles_total"]))
(HERE / "build" / "spv-order.json").write_text(json.dumps(out, indent=1), encoding="utf-8")
print("\nwrote", HERE / "build" / "spv-order.json")
