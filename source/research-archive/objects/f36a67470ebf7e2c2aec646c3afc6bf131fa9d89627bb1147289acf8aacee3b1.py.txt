"""Audit every card's GPU-REQUEST pass_ordinals against the ledger.

Checks that each declared ordinal points at a row whose family/shader plausibly matches the
card, and prints the SPV sha per ordinal so a wrong group is visible at a glance.
"""
import json, sys
from collections import Counter, defaultdict
from pathlib import Path
sys.path.insert(0, str(Path(__file__).parent))
import COMMON as K

rows = K.rows()
by_ord = {r['ordinal']: r for r in rows}
bad, ok = [], []
for d in sorted((K.C / 'cards').iterdir()):
    req = d / 'GPU-REQUEST.json'
    if not req.exists():
        continue
    j = K.read_json(req)
    ords = ((j.get('shader_group') or {}).get('pass_ordinals')) or []
    if not ords:
        continue
    fams = Counter(by_ord[o]['family'] for o in ords if o in by_ord)
    shas = Counter(by_ord[o]['spv_sha256'][:8] for o in ords if o in by_ord)
    declared = str((j.get('shader_group') or {}).get('family') or '')
    mismatch = [f for f in fams if declared and f.split()[0].lower() not in declared.lower()
                and declared.lower() not in f.lower()]
    rec = dict(card=d.name, n=len(ords), families=dict(fams), shas=dict(shas), declared=declared)
    if mismatch or len(shas) > 3:
        bad.append(rec)
    else:
        ok.append(rec)
print('checked %d cards with pass_ordinals' % (len(ok) + len(bad)))
print('consistent: %d' % len(ok))
print('suspicious: %d' % len(bad))
for r in bad:
    print('  %-24s declared=%-8s families=%s shas=%s' % (r['card'], r['declared'], r['families'], r['shas']))
K.write_json(K.C / 'evidence' / 'ORDINAL-AUDIT.json', dict(ok=ok, suspicious=bad))
