"""Read-only ISA scanner for the n11-vit-ffn card (no GPU).

Parses a GTPin medisa .asm dump into (label, offset, opcode, operands, comment)
and produces feed/consume maps for the F13 expand kernel.
"""
import re, sys, json, collections

LINE = re.compile(r'^/\* \[([0-9A-Fa-f]+)\]\s+\*/\s*(.*)$')
LABEL = re.compile(r'^(L\d+):')


def parse(path):
    out = []
    label = None
    for raw in open(path, encoding='utf-8', errors='replace'):
        s = raw.rstrip('\n')
        m = LABEL.match(s.strip())
        if m:
            label = m.group(1)
            continue
        m = LINE.match(s)
        if not m:
            continue
        off = int(m.group(1), 16)
        body = m.group(2).strip()
        # split off // comment
        body2, _, com = body.partition('//')
        body2 = body2.strip()
        com = com.strip()
        if not body2:
            continue
        pred = None
        pm = re.match(r'^\(([^)]*)\)\s*', body2)
        if pm:
            pred = pm.group(1)
            body2 = body2[pm.end():].strip()
        toks = body2.split()
        if not toks:
            continue
        op = toks[0]
        rest = body2[len(op):].strip()
        # operands: dst/src fields separated by 2+ spaces were collapsed; keep tokens
        out.append(dict(label=label, off=off, pred=pred, op=op, text=body2,
                        operands=toks[1:], comment=com))
    return out


def regs_of(text):
    return sorted(set(re.findall(r'\b(?:r|acc|a0|cr0)\d+', text)))


def summarize(ins):
    sends = [i for i in ins if i['op'].startswith('send')]
    dpas = [i for i in ins if i['op'].startswith('dpas')]
    return dict(total=len(ins), sends=len(sends), dpas=len(dpas),
                other=len(ins) - len(sends) - len(dpas),
                send_ops=collections.Counter(i['op'] for i in sends),
                dpas_ops=collections.Counter(i['op'] for i in dpas))


if __name__ == '__main__':
    p = sys.argv[1]
    ins = parse(p)
    mode = sys.argv[2] if len(sys.argv) > 2 else 'summary'
    if mode == 'summary':
        print(json.dumps(summarize(ins), default=str, indent=1))
    elif mode == 'dpas':
        for i in ins:
            if i['op'].startswith('dpas'):
                print('%05X %s' % (i['off'], i['text']))
    elif mode == 'window':
        lo = int(sys.argv[3], 16)
        hi = int(sys.argv[4], 16)
        for i in ins:
            if lo <= i['off'] <= hi:
                print('%05X %-4s %s' % (i['off'], i['label'] or '', i['text']))
    elif mode == 'sends':
        for i in ins:
            if i['op'].startswith('send'):
                print('%05X %s' % (i['off'], i['text']))
