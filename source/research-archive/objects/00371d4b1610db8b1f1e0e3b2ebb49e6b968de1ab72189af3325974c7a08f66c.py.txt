"""n22-f15-den-pv: P->PV transport census in the parent SPIR-V (read-only, CPU only).

Answers, for the parent F15 module:
  * how many OpGroupNonUniformShuffle / ShuffleXor / OpCooperativeMatrix* survive,
  * for every surviving shuffle: its value-operand signature and index-operand
    signature, so duplicates (same value, same index) and identity shuffles
    (index == the caller's own lane) are enumerable instead of assumed.
"""
import sys, json
from collections import Counter, defaultdict
from pathlib import Path

BASE = Path('@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/fullnet-continuous-20261027')
sys.path.insert(0, str(BASE))
sys.path.insert(0, str(BASE.parents[0] / ''))
TOOLS = Path('@NR_WORKSPACE@/build/zcode-fp16-compute-backend-20261021/scripts')
sys.path.insert(0, str(TOOLS))
import spv_tools_local as ST  # noqa: E402

PARENT = Path('@NR_WORKSPACE@/work/zcode-fp16-fullnet-handoff-20261024/candidates/f15l-vit-pv/spv/vit_attn_f15l.spv')
OUT = BASE / 'cards/n22-f15-den-pv/build/SHUF-CENSUS-parent.json'

NAMES = {345: 'OpGroupNonUniformShuffle', 346: 'OpGroupNonUniformShuffleXor',
         4457: 'OpCooperativeMatrixLoadKHR', 4458: 'OpCooperativeMatrixStoreKHR',
         4459: 'OpCooperativeMatrixMulAddKHR', 224: 'OpControlBarrier'}


def dump(module_path, out_path):
    m = ST.Module(Path(module_path))
    inst = list(m.instructions)
    hist = Counter(i[0] for i in inst)
    shuf = []
    for i in inst:
        if i[0] in (345, 346):
            ops = [int(x) for x in i[1]]
            shuf.append(dict(opcode=NAMES[i[0]], ops=ops,
                             result=int(ops[0]) if ops else None))
    # signature of (value, index) per shuffle, following the operand order of SPIR-V:
    #   OpGroupNonUniformShuffle  %type %result %exec %value %index
    #   OpGroupNonUniformShuffleXor %type %result %exec %value %index
    rows = []
    for s in shuf:
        ops = [int(x) for x in s['ops']]
        value, index = ops[3], ops[4]
        try:
            vsig = m.operand_signature(value, depth=3)
        except Exception as e:
            vsig = 'ERR:%s' % e
        try:
            isig = m.operand_signature(index, depth=3)
        except Exception as e:
            isig = 'ERR:%s' % e
        rows.append(dict(kind=s['opcode'], result=ops[1], value=value, index=index,
                         value_sig=str(vsig), index_sig=str(isig)))
    dup = Counter((r['kind'], r['value_sig'], r['index_sig']) for r in rows)
    res = dict(module=str(module_path), total_instructions=sum(hist.values()),
               hist={NAMES.get(k, 'op%d' % k): v for k, v in sorted(hist.items()) if k in NAMES},
               shuffles=rows,
               shuffle_count=len(rows),
               distinct_key_counts={str(k): v for k, v in dup.items()},
               duplicated_groups=sum(1 for v in dup.values() if v > 1),
               redundant_copies=sum(v - 1 for v in dup.values()))
    Path(out_path).parent.mkdir(parents=True, exist_ok=True)
    Path(out_path).write_text(json.dumps(res, indent=2), encoding='utf-8')
    return res


if __name__ == '__main__':
    src = sys.argv[1] if len(sys.argv) > 1 else str(PARENT)
    out = sys.argv[2] if len(sys.argv) > 2 else str(OUT)
    r = dump(src, out)
    print(json.dumps({k: v for k, v in r.items() if k != 'shuffles'}, indent=2)[:4000])
    for row in r['shuffles']:
        print(row['kind'], 'result', row['result'], 'val', row['value'], 'idx', row['index'])
        print('   value_sig', row['value_sig'][:150])
        print('   index_sig', row['index_sig'][:150])
