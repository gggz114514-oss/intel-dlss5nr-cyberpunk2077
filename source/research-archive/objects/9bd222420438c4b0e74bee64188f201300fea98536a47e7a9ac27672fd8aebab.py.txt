import json,struct,statistics,sys
from pathlib import Path
D=Path(__file__).resolve().parent;family=sys.argv[1];sgn=8 if family=='f20' else 4;ordinal={'f20':230,'c32':1,'f01':0}[family]
def read(p):return json.loads(p.read_text(encoding='utf-8-sig'))
prep=read(D/({'f20':'direct-clock','c32':'direct-c32','f01':'direct-f01'}[family])/'PREPARATION.json');gx,gy,_=prep['grid']
names=['fold','second-barrier','expand','contract','qkv','norm','scale','score-exp','den','p-normalize','pv','output'] if family=='f20' else ['stage-'+x for x in ['ffn','qkv','norm-publish','score-exp','den','p-normalize','pv','output']]
if family=='f01':names.insert(0,'stage-input-lift')
whole='clock-whole' if family=='f20' else 'whole'
rows=[]
for rep in range(3):
 arms={}
 for name in ['control',*names,whole]:
  p=D/'run'/f'{family}-stage-r{rep}-{name}';rp=p/'port_network_receipt.json'
  if not rp.exists():continue
  r=read(rp);timing=next(x for x in r['per_pass_gpu'] if x['index']==ordinal);data=(p/'clock.bin').read_bytes();values=struct.unpack('<%dI'%(len(data)//4),data);ticks=[];by_sg={i:[] for i in range(sgn)};missing=[];unexpected=[]
  for y in range(gy):
   for x in range(gx):
    for sg in range(sgn):
     k=((y*gx+x)*sgn+sg)*4;a,b,c,d=values[k:k+4];nz=bool(a or b or c or d);selected=x%16==8 and y%16==8 and name!='control'
     if selected and nz:
      dt=((c|(d<<32))-(a|(b<<32)))&((1<<64)-1);ticks.append(dt);by_sg[sg].append(dt)
     elif selected:missing.append([x,y,sg])
     elif nz:unexpected.append([x,y,sg])
  arms[name]={'median_ms':statistics.median(timing['samples_ms']),'mean_ticks':statistics.mean(ticks) if ticks else None,'median_ticks':statistics.median(ticks) if ticks else None,'per_sg_mean_ticks':{sg:statistics.mean(t) for sg,t in by_sg.items() if t},'missing':missing,'unexpected':unexpected}
 entry={'rep':rep,'arms':arms}
 if len(arms)==len(names)+2:
  entry['sum_to_whole_ratio']=sum(arms[n]['mean_ticks'] for n in names)/arms[whole]['mean_ticks']
  entry['perturbation']={n:arms[n]['median_ms']/arms['control']['median_ms']-1 for n in names+[whole]}
 rows.append(entry)
out={'family':family,'status':'SOURCE_WINDOW_MEASUREMENTS_PENDING_NATIVE_AUDIT','unit':'subgroup clock ticks; not additive exclusive network milliseconds','repeats':rows}
if all('sum_to_whole_ratio' in r for r in rows):
 out['stage_summary']={n:{'mean_ticks':statistics.mean(r['arms'][n]['mean_ticks'] for r in rows),'whole_fraction':statistics.mean(r['arms'][n]['mean_ticks']/r['arms'][whole]['mean_ticks'] for r in rows),'spread':max(r['arms'][n]['mean_ticks'] for r in rows)/min(r['arms'][n]['mean_ticks'] for r in rows)-1,'max_abs_perturbation':max(abs(r['perturbation'][n]) for r in rows)} for n in names}
(D/f'MAIN-{family.upper()}-STAGES-REVIEW.json').write_text(json.dumps(out,indent=2),encoding='utf-8')
for r in rows:print({'rep':r['rep'],'completed_arms':len(r['arms']),'closure':r.get('sum_to_whole_ratio'),'max_abs_perturbation':max(map(abs,r['perturbation'].values())) if 'perturbation'in r else None})
for k,v in out.get('stage_summary',{}).items():print(k,v)
