# -*- coding: utf-8 -*-
"""Verify the DISPLAY-CONTRACT mapping and emit ROI candidate crops."""
import sys
from pathlib import Path

import numpy as np

sys.path.insert(0, str(Path(__file__).resolve().parent))
import pngio  # noqa: E402

Q = Path(r"@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/int8-quality-first-20261029")
OUT = Path(__file__).resolve().parent.parent
PNG = OUT / "PNG"
CROPS = PNG / "roi-candidates"

ref = np.fromfile(Q / "dataset/day-street-static-formal-01/0-34189/network-output.bin",
                  dtype="<f4").reshape(2, 720, 1280, 4)
plane0 = ref[0, :, :, :3]
img = np.rint(np.clip(plane0.astype(np.float64), 0.0, 1.0) * 255.0).astype(np.uint8)
pngio.write_rgb(PNG / "fp16-34189-from-bin.png", img)

src = pngio.read_rgb(Q / "visual-comparison/fp16.png")
print("fp16.png shape", src.shape)
if src.shape == img.shape:
    d = np.abs(src.astype(np.int16) - img.astype(np.int16))
    print("fp16.png vs round(clip(plane0)*255): maxdiff", int(d.max()),
          "npx_diff", int((d > 0).sum()), "of", d.size)
    print("proof: visual-comparison/fp16.png IS frame 34189 plane0 under the documented mapping"
          if int(d.max()) == 0 else "MISMATCH -> mapping or frame differs")

CANDS = {
    "01-sky-bright":            (600, 120, 700, 190),
    "02-facade-bright":         (745, 165, 860, 215),
    "03-ncpd-sign":             (445, 240, 640, 300),
    "04-blue-ground-sidewalk":  (90, 590, 300, 675),
    "05-blue-ground-underpass": (630, 505, 780, 555),
    "06-neutral-wall-left":     (60, 330, 200, 395),
    "07-neutral-pillar-right":  (1180, 250, 1270, 400),
    "08-neutral-road":          (850, 630, 1000, 700),
}
over = img.copy()
colors = [(255, 0, 0), (0, 255, 0), (0, 128, 255), (255, 255, 0),
          (255, 0, 255), (0, 255, 255), (255, 128, 0), (128, 255, 128)]
for i, (name, (x0, y0, x1, y1)) in enumerate(CANDS.items()):
    c = colors[i % len(colors)]
    over[y0:y1, x0:x0 + 1] = c
    over[y0:y1, x1 - 1:x1] = c
    over[y0:y0 + 1, x0:x1] = c
    over[y1 - 1:y1, x0:x1] = c
    crop = img[y0:y1, x0:x1]
    pngio.write_rgb(CROPS / f"{name}.png", crop)
    print(f"{name}: box=({x0},{y0},{x1},{y1}) size={x1-x0}x{y1-y0} "
          f"meanRGB={crop.reshape(-1,3).mean(0).round(2).tolist()} "
          f"luma={float((crop[:,:,0]*0.2126+crop[:,:,1]*0.7152+crop[:,:,2]*0.0722).mean())/255:.3f}")
pngio.write_rgb(CROPS.parent / "roi-candidates-overlay.png", over)
print("wrote overlay:", CROPS.parent / "roi-candidates-overlay.png")
