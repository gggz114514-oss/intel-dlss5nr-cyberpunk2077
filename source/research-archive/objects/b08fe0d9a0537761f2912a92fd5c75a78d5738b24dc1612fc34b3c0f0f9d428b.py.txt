"""Print an ISA address range of one F05 kernel: _t04_range.py <64|128|256> <from> <to>"""
import sys
from pathlib import Path
HERE = Path(__file__).parent
sys.path.insert(0, str(HERE))
import _regions as R

w = int(sys.argv[1]); lo = int(sys.argv[2], 16); hi = int(sys.argv[3], 16)
ins, labels = R.load(w)
ins, guard = R.classify(w, ins, labels)
for it in ins:
    if lo <= it["addr"] <= hi:
        print("[%06X] %-4s %s" % (it["addr"], "L" if it["live"] else "D", it["text"]))
