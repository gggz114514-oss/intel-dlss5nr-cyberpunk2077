"""Regenerate SOURCE-DIFF.patch (parent source -> candidate.comp)."""
from pathlib import Path
import difflib

CARD = Path(__file__).resolve().parent
PARENT = Path('@NR_WORKSPACE@/candidates/zcode-fp16-b5-global-expansion-20261020/'
              'split16-s21-seed-direct-v1/source/split16_projection_fp16.comp')
a = PARENT.read_text(encoding='utf-8').splitlines()
b = (CARD / 'candidate.comp').read_text(encoding='utf-8').splitlines()
d = list(difflib.unified_diff(a, b,
                              'a/candidates/zcode-fp16-b5-global-expansion-20261020/'
                              'split16-s21-seed-direct-v1/source/split16_projection_fp16.comp',
                              'b/cards/n09b-f09-sx-stage/candidate.comp',
                              n=3, lineterm=''))
(CARD / 'SOURCE-DIFF.patch').write_text('\n'.join(d) + '\n', encoding='utf-8')
added = sum(1 for l in d if l.startswith('+') and not l.startswith('+++'))
removed = sum(1 for l in d if l.startswith('-') and not l.startswith('---'))
print('SOURCE-DIFF.patch: +%d -%d lines' % (added, removed))
