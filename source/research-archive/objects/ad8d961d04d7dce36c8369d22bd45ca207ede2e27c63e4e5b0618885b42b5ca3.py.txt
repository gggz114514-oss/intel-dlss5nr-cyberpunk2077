"""Build identity for card n28: reproduce the *parent* SPIR-V of ord 230 byte-for-byte from
its source with the recorded recipe, and record the toolchain identities.

This card produced no candidate (see RESUME.txt / BARRIER-AUDIT.md: both budgeted directions
are empty), so BUILD.json records the parent rebuild only.

Recipe: evidence/RECIPES.json -> shader 215aca679aaa... (verified: true)
  compile: no defines, include dir = cards/idt-identity-recipe/evidence/headers/215aca679aaa
  then: transfer the decoration pattern from the recorded decoration parent and re-apply the
        parent's NoContraction set (nc_count 45) -- plain glslang output does not match.
"""
import sys, json, shutil, subprocess
from pathlib import Path

sys.stdout.reconfigure(encoding='utf-8')
sys.path.insert(0, str(Path(__file__).resolve().parents[3]))
import COMMON as K
import spv_tools_local as ST

CARD = Path('cards/n28-f20-barrier-alu')
OUT = CARD / 'build'
OUT.mkdir(parents=True, exist_ok=True)

row = K.row_by_ordinal(230)
src = Path(row['source'])
target = Path(row['spv'])
rec = K.read_json(Path('evidence/RECIPES.json'))[row['spv_sha256']]
pinned = Path(rec['compile']['pinned_include_dir'])
dec_parent = Path(rec['decoration_parent'])

res = dict(ordinal=230, family='F20', name=row['name'],
           source=K.ident(src), target=K.ident(target),
           recipe_verified=rec['verified'], recipe_sha=row['spv_sha256'],
           include_dirs=[str(pinned), str(dec_parent.parent)],
           decoration_parent=K.ident(dec_parent))

raw = OUT / 'raw.spv'
K.compile_glsl(src, raw, [pinned])
res['raw'] = K.ident(raw)
res['raw_match'] = K.sha(raw) == row['spv_sha256']
print('raw sha      ', res['raw']['sha256'])
print('target sha   ', row['spv_sha256'])
print('raw_match    ', res['raw_match'])

pm = ST.Module(dec_parent)
rm = ST.Module(raw)
keep, info = ST.transfer_decoration_pattern_aligned(pm, rm, set())
fixed = OUT / 'parent-rebuild.spv'
ST.set_no_contraction(rm, keep, fixed)
res['rebuild'] = K.ident(fixed)
res['decorated_match'] = res['rebuild']['sha256'] == row['spv_sha256']
res['decoration_info'] = {k: info[k] for k in (
    'functions_mapped', 'parent_decorations_total',
    'child_own_decorations_outside_pattern')}
res['nc_count'] = len(keep)
res['nc_parent'] = len(ST.no_contraction_ids(pm))
res['float_NC_checked'] = len(keep)
print('rebuild sha  ', res['rebuild']['sha256'])
print('decorated_match', res['decorated_match'], 'nc', len(keep), 'parent nc', res['nc_parent'])

# SPIR-V validation of both the raw and the reproduced module (CPU tool only)
val = {}
for tag, p in (('raw', raw), ('parent-rebuild', fixed)):
    r = subprocess.run([str(K.VALIDATE), str(p)], capture_output=True, text=True,
                       encoding='utf-8', errors='replace')
    val[tag] = dict(returncode=r.returncode, stdout=r.stdout.strip()[:400],
                    stderr=r.stderr.strip()[:400])
    print('validate %-14s rc=%d' % (tag, r.returncode))
res['spv_validate'] = val

# toolchain identities
res['toolchain'] = dict(
    glslang=K.ident(K.GLSLANG),
    validator=K.ident(K.VALIDATE),
    runner=K.ident(K.RUNNER),
    python=sys.version.split()[0],
)

# what the card decided NOT to build, with the reason (kept next to the identity)
res['candidate'] = None
res['candidate_reason'] = (
    'BLOCKED: no budget in either direction. BARRIER: the 12 static sync.bar are 5 runtime '
    '(7 are inside if(debug) blocks and the plan carries debug=0); all 5 are protected by a '
    'cross-subgroup RAW publish or by the arena WAR between stage 3 reads and stage 4 writes '
    '(BARRIER-AUDIT.md sections 3-4). ALUWR: every address cone feeding the 13 SPIR-V / 38 '
    'unrolled shared-source coopMatLoad sites is 2-8 instructions long and only the last one '
    'is site-local (base+imm); the three deepest chains in the kernel (33/31/29) are the '
    'frozen reduction trees and the tail FP32 accumulation order (CONTRACT section 3).')

K.write_json(CARD / 'BUILD.json', res)
print('\nwrote', CARD / 'BUILD.json')
