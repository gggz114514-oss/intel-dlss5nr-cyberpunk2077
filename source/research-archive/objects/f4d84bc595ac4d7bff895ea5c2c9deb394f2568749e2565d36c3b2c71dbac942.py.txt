"""n36-f05-arena-shrink: the two contract-required alignment checks that need the modules.

1. NoContraction site multiset: an id-free (function, opcode, operand-signature) multiset
   must be identical between parent and candidate, site for site.
2. Integer control-flow opcode set {128,130,132,134,135,137,138,139} (OpIAdd/ISub/IMul/
   SDiv/UDiv/SRem/UMod per SPIR-V numbering used by the campaign) is reported separately
   from the floating-point multiset, as CONTRACT.md 2 requires.

Writes build/verify.json.  CPU only.
"""
import collections
import json
import pathlib
import sys

sys.path.insert(0, r'@NR_WORKSPACE@/build/zcode-fp16-compute-backend-20261021/scripts')
import spv_tools_local as ST          # noqa: E402

CARD = pathlib.Path(__file__).resolve().parent
INT_CF = [128, 130, 132, 134, 135, 137, 138, 139]
OPNAME = {128: 'OpIAdd', 130: 'OpISub', 132: 'OpIMul', 134: 'OpSDiv', 135: 'OpUDiv',
          137: 'OpSRem', 138: 'OpSMod', 139: 'OpUMod'}


def nc_multiset(m):
    ids = ST.no_contraction_ids(m)
    by = m.id_to_instruction()
    func_of, cur = {}, None
    for op, ops, _, _ in m.instructions:
        if op == 54:                                  # OpFunction
            cur = ops[1]
        rid = ST.result_id_of(op, ops)
        if rid is not None:
            func_of[rid] = cur
    bag = collections.Counter()
    for i in ids:
        op, ops = by[i]
        bag[(func_of.get(i), op, ST.Module.operand_signature(m, i))] += 1
    return bag, len(ids)


def main():
    out = {}
    for g in ('c64', 'c128', 'c256'):
        par = ST.Module(CARD / ('parent-rebuild-%s.spv' % g))
        cand = ST.Module(CARD / ('candidate-%s.spv' % g))
        bp, np_ = nc_multiset(par)
        bc, nc_ = nc_multiset(cand)
        hp = collections.Counter(op for op, _, _, _ in par.instructions)
        hc = collections.Counter(op for op, _, _, _ in cand.instructions)
        rec = dict(
            no_contraction=dict(parent=np_, candidate=nc_, identical=bp == bc,
                                parent_only=[str(k) for k in (bp - bc)][:10],
                                candidate_only=[str(k) for k in (bc - bp)][:10]),
            int_control_flow={OPNAME[k]: [hp.get(k, 0), hc.get(k, 0)] for k in INT_CF},
            int_control_flow_total=[sum(hp.get(k, 0) for k in INT_CF),
                                    sum(hc.get(k, 0) for k in INT_CF)],
            private_storage_variables=dict(
                parent=sum(1 for op, ops, _, _ in par.instructions
                           if op == 59 and len(ops) > 2 and ops[2] == 6),
                candidate=sum(1 for op, ops, _, _ in cand.instructions
                              if op == 59 and len(ops) > 2 and ops[2] == 6),
                note='no Private (spill) object in either module; only the ISA can show '
                     'the spill area size'),
            module_words=[len(par.words), len(cand.words)],
            instructions=[len(par.instructions), len(cand.instructions)],
        )
        out[g] = rec
        print('%-5s NC %d -> %d identical=%s | int-cf %s -> %s | Private %d/%d'
              % (g, np_, nc_, bp == bc,
                 rec['int_control_flow_total'][0], rec['int_control_flow_total'][1],
                 rec['private_storage_variables']['parent'],
                 rec['private_storage_variables']['candidate']))
    (CARD / 'build/verify.json').write_text(json.dumps(out, indent=2), encoding='utf-8')
    ok = all(v['no_contraction']['identical'] and
             v['private_storage_variables']['parent'] == 0 and
             v['private_storage_variables']['candidate'] == 0 for v in out.values())
    print('alignment checks pass:', ok)
    return 0 if ok else 1


if __name__ == '__main__':
    sys.exit(main())
