"""Assemble the card's build identity artefacts: parent-rebuild.spv, SOURCE-DIFF.patch,
BUILD.json.  Everything is measured from files on disk in this run.
"""
import difflib
import json
import shutil
import subprocess
import sys
from pathlib import Path

sys.path.insert(0, r'@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/fullnet-continuous-20261027')
import COMMON as K

CARD = K.C / 'cards' / 'n04a-f05-c64-feed'
BUILD = CARD / 'build'
GROUP = [7, 10, 13, 16, 215, 218, 221, 224]
ORD = 7


def main():
    row = K.row_by_ordinal(ORD)
    # parent rebuild artefact next to the candidate
    shutil.copyfile(BUILD / 'parent-raw.spv', CARD / 'parent-rebuild.spv')
    assert K.sha(CARD / 'parent-rebuild.spv') == row['spv_sha256']

    a = (CARD / 'parent-rebuild.comp').read_text(encoding='utf-8').splitlines(keepends=True)
    b = (CARD / 'candidate.comp').read_text(encoding='utf-8').splitlines(keepends=True)
    diff = list(difflib.unified_diff(a, b, fromfile='a/parent-rebuild.comp',
                                     tofile='b/candidate.comp', n=4))
    (CARD / 'SOURCE-DIFF.patch').write_text(''.join(diff), encoding='utf-8')
    added = sum(1 for l in diff if l.startswith('+') and not l.startswith('+++'))
    removed = sum(1 for l in diff if l.startswith('-') and not l.startswith('---'))
    print('patch hunks:', sum(1 for l in diff if l.startswith('@@')), 'added', added, 'removed', removed)

    pairs = K.read_json(BUILD / 'nc-pairs.json')
    float_nc = [p for p in pairs if p['cls'] == 'float_NC']
    int_nc = [p for p in pairs if p['cls'] == 'integer_NC']
    cpu = K.read_json(BUILD / 'cpu-check.json')
    val = subprocess.run([str(K.VALIDATE), str(CARD / 'candidate.spv')],
                         capture_output=True, text=True, encoding='utf-8', errors='replace')

    base_plan = K.BASELINE
    build = dict(
        card='n04a-f05-c64-feed',
        family='F05', width=64, pass_ordinals=GROUP,
        baseline_plan=dict(path=str(base_plan), sha256=K.sha(base_plan),
                           expected_sha256=K.BASELINE_SHA,
                           match=K.sha(base_plan) == K.BASELINE_SHA),
        parent=dict(path=str(row['spv']), sha256=K.sha(row['spv'])),
        parent_source=dict(path=str(row['source']), sha256=K.sha(row['source'])),
        parent_source_copy=dict(path=str(CARD / 'parent-rebuild.comp'),
                                sha256=K.sha(CARD / 'parent-rebuild.comp')),
        parent_rebuild=dict(path=str(CARD / 'parent-rebuild.spv'),
                            sha256=K.sha(CARD / 'parent-rebuild.spv'),
                            byte_identical=K.sha(CARD / 'parent-rebuild.spv') == row['spv_sha256'],
                            recipe='glslang 16.5.0 -V --target-env vulkan1.3, include dirs '
                                   '[card dir, census site dir]; no NoContraction transfer'),
        candidate_source=dict(path=str(CARD / 'candidate.comp'),
                              sha256=K.sha(CARD / 'candidate.comp')),
        candidate_spv=dict(path=str(CARD / 'candidate.spv'),
                           sha256=K.sha(CARD / 'candidate.spv'),
                           differs_from_parent=K.sha(CARD / 'candidate.spv') != row['spv_sha256']),
        plan=dict(path=str(CARD / 'plan.txt'), sha256=K.sha(CARD / 'plan.txt'),
                  passes_replaced=8, parent_plan=str(base_plan),
                  parent_plan_sha256=K.sha(base_plan)),
        float_NC_checked=len(float_nc),
        integer_NC_checked=len(int_nc),
        float_NC_sites=[dict(parent_id=p['parent_id'], candidate_id=p['candidate_id'],
                             op=p['parent_op_name'], signature=p['signature'],
                             candidate_signature=p['candidate_signature'],
                             signature_equal=p['signature_equal']) for p in float_nc],
        integer_NC_sites=[dict(parent_id=p['parent_id'], candidate_id=p['candidate_id'],
                               op=p['parent_op_name'], signature=p['signature'],
                               candidate_signature=p['candidate_signature'],
                               signature_equal=p['signature_equal']) for p in int_nc],
        no_contraction_method=(
            'Both modules are decorated only by glslang from the same source lineage '
            '(parent source compiles byte-identically to the parent SPV with no '
            'decoration transfer). The candidate keeps 80/80 decorations: each parent '
            'NoContraction site is aligned to exactly one candidate site by '
            'longest-matching-blocks over the ID-free (opcode, wordcount) skeleton and '
            'verified by identical ID-free operand signatures; the aligned set equals '
            'the candidate\'s own decorated set exactly, so no site was added or dropped.'),
        spv_validation=dict(argv=[str(K.VALIDATE), str(CARD / 'candidate.spv')],
                            returncode=val.returncode,
                            stdout=(val.stdout or '').strip(),
                            stderr=(val.stderr or '').strip()),
        cpu_check=dict(
            baseline=dict(argv=[str(K.RUNNER), '--plan', str(base_plan), '--cpu-check'],
                          stdout=cpu['baseline']['stdout'].strip()),
            candidate=dict(argv=[str(K.RUNNER), '--plan', str(CARD / 'plan.txt'), '--cpu-check'],
                           stdout=cpu['candidate']['stdout'].strip()),
            identical=cpu['baseline']['stdout'].strip() == cpu['candidate']['stdout'].strip()),
        spv_opcode_delta=K.read_json(BUILD / 'spv-opcode-delta.json'),
        tools=dict(python='@NR_USER_HOME@/AppData/Local/Programs/Python/Python312/python.exe',
                   glslang=str(K.GLSLANG), validate=str(K.VALIDATE), runner=str(K.RUNNER)),
    )
    K.write_json(CARD / 'BUILD.json', build)
    print(json.dumps({k: build[k] for k in
                      ('float_NC_checked', 'integer_NC_checked')}, indent=1))
    print('candidate sha', build['candidate_spv']['sha256'])
    print('plan sha     ', build['plan']['sha256'])
    print('cpu identical', build['cpu_check']['identical'])


if __name__ == '__main__':
    main()
