def cmp(a, b, an, bn):
    n = min(len(a.passes), len(b.passes))
    same_name = sum(1 for i in range(n) if a.passes[i].name == b.passes[i].name)
    same_spv  = sum(1 for i in range(n) if a.passes[i].spv == b.passes[i].spv)
    same_sha  = 0
    for i in range(n):
        sa = planlib.sha256_file(a.passes[i].spv) if Path(a.passes[i].spv).is_file() else None
        sb = planlib.sha256_file(b.passes[i].spv) if Path(b.passes[i].spv).is_file() else None
        if sa and sa == sb: same_sha += 1
    print(f"{an} vs {bn}: common={n} same_name={same_name} same_spv_path={same_spv} same_spv_sha={same_sha} added={len(a.passes)-n} removed={len(b.passes)-n}")
cmp(cur, c32, "current", "c32-equalized")
cmp(cur, base, "current", "corrected-base")
