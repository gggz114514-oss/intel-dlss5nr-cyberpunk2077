import json, os, collections

W = r"@NR_WORKSPACE@/work/dsh-fullnet-static-20261029"
o = json.load(open(os.path.join(W, "ledger/FULLNET-ROWS-V2.json"), encoding="utf-8"))
rows = o["rows"]

prod = collections.defaultdict(list)
cons = collections.defaultdict(list)
for r in rows:
    for b in r.get("bindings") or []:
        nm, ac = b["buffer"], b.get("spv_access")
        if ac == "write":
            prod[nm].append(r["name"])
        elif ac == "read":
            cons[nm].append(r["name"])

targets = ["block0_out", "block1_out", "block2_out", "block3_out", "skip4", "block4_out",
           "block5_out", "block6_out", "block7_out", "skip8", "block14_out", "block15_out",
           "block22_out", "block30_out", "block31_out", "block38_out",
           "block62_out", "block63_out", "block64_out", "block65_out", "block66_out",
           "block67_out", "block68_out", "block69_out", "frame_out", "skip0"]
print("=== producer / consumer map ===")
for t in targets:
    p = prod.get(t, [])
    c = cons.get(t, [])
    print("%-14s P=%-46s C=%s" % (t, ",".join(p)[:46], ",".join(c)[:120]))

print("\n=== F2 / tail / F20 rows ===")
for r in rows:
    if r["family"] in ("F2", "tail", "F20"):
        ins = [b["buffer"] for b in r["bindings"] if b.get("spv_access") == "read"]
        outs = [b["buffer"] for b in r["bindings"] if b.get("spv_access") == "write"]
        print("\nord=%d %-22s family=%s stage=%s" % (r["ordinal"], r["name"], r["family"], r["stage"]))
        print("   spv:", r["spv"]["basename"], "sha", r["spv"]["sha256"][:16], "int8_mma", r["spv"].get("int8_mma"), "fp16_mma", r["spv"].get("fp16_mma"))
        print("   reads(%d): %s" % (len(ins), ins))
        print("   writes(%d): %s" % (len(outs), outs))
        print("   quant_edges:", json.dumps(r.get("quant_edges"), ensure_ascii=False)[:500])
