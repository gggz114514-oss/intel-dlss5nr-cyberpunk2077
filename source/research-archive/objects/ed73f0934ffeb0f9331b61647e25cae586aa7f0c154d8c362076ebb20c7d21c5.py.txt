"""Read-only peek at the F14 native ISA (baseline and n12 candidate).

Prints the scalar bss loads (the scale coefficient) with context and the
send.ugm message histogram.  Nothing is captured or written outside stdout.
"""
import hashlib
import re
import sys
from collections import Counter
from pathlib import Path

Q = Path('@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/'
         'fullnet-continuous-20261027/queue/n12-vit-qkv/native/captures')
BASE = Q / 'native-native-base' / 'ISA' / 'CS_asm2ef42bf87fecdff2_simd32_0000000000000019_24_orig.asm'
CAND = Q / 'native-native-cand' / 'ISA' / 'CS_asm220ce9995b78f549_simd32_0000000000000019_24_orig.asm'
INSN = re.compile(r'^\s*/\*\s*\[([0-9A-Fa-f]+)\]\s*\*/\s*(.*)$')


def hist(lines):
    c = Counter()
    for ln in lines:
        m = INSN.match(ln)
        if not m:
            continue
        body = m.group(2)
        # send.ugm operand message is after the opcode
        toks = body.split()
        if toks and toks[0] in ('send.ugm', 'send.gtwy', 'dpas.8x8', 'sync.nop', 'sync.allrd',
                                'math.rsqt', 'mov', 'add', 'mul', 'shl', 'goto', 'join', 'jmpi'):
            if toks[0] == 'send.ugm':
                c['send.ugm ' + (toks[1] if len(toks) > 1 else '?')] += 1
            else:
                c[toks[0]] += 1
        elif toks and toks[0].startswith('/*'):
            pass
    return c


def main():
    for name, p in (('BASE', BASE), ('CAND', CAND)):
        data = p.read_bytes()
        print('=' * 20, name, p.name, len(data), 'sha256', hashlib.sha256(data).hexdigest())
        lines = data.decode('utf-8', 'replace').splitlines()
        n_insn = sum(1 for ln in lines if INSN.match(ln))
        print('instruction_lines', n_insn)
        c = hist(lines)
        for k, v in c.most_common():
            print('   %-60s %d' % (k, v))
        # scalar bss loads with context
        hits = [i for i, ln in enumerate(lines) if 'd32x1t' in ln and 'bss' in ln]
        print('scalar bss d32x1t load lines:', len(hits))
        for i in hits:
            print('--- context around line', i + 1)
            for j in range(max(0, i - 8), min(len(lines), i + 9)):
                print(('>> ' if j == i else '   ') + lines[j].rstrip()[:200])
        print()


if __name__ == '__main__':
    main()
