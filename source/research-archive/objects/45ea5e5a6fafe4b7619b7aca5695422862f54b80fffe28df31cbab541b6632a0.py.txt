"""C00 probe: verify array layout/identity for the three comparison frames.

Read-only. No GPU. Writes nothing outside stdout.
"""
import hashlib
import json
import os
import sys

import numpy as np

B = r"@NR_WORKSPACE@"
Q = B + "/work/codex-astra-fp16-int8-audit-20261024/int8-quality-first-20261029"
W = B + "/work/dsh-fullnet-static-20261029"
XE = W + "/S02-f01/runs24"
C64 = Q + "/real-int8-24/c64-family-static"

FRAMES = [
    (34189, "day-street-static-formal-01"),
    (41444, "day-street-motion-formal-01"),
    (43654, "day-street-motion-formal-02"),
]
SHAPE = (2, 720, 1280, 4)
H, Wd = 720, 1280


def load(path):
    a = np.fromfile(path, dtype="<f4")
    assert a.size == 2 * H * Wd * 4, (path, a.size)
    return a.reshape(SHAPE)


def sha(path, bs=1 << 20):
    h = hashlib.sha256()
    with open(path, "rb") as f:
        while True:
            b = f.read(bs)
            if not b:
                break
            h.update(b)
    return h.hexdigest()


def stats(x):
    return dict(min=float(x.min()), max=float(x.max()), mean=float(x.mean()),
                nan=int(np.isnan(x).sum()), inf=int(np.isinf(x).sum()))


for frame, seq in FRAMES:
    ref_p = f"{Q}/dataset/{seq}/0-{frame}/network-output.bin"
    xe_p = f"{XE}/run-{frame}/replayed-network.bin"
    c64_p = f"{C64}/run-{frame}/replayed-network.bin"
    print("=" * 70)
    print("frame", frame, seq)
    for tag, p in (("ref", ref_p), ("xe", xe_p), ("c64", c64_p)):
        print(f"  {tag}: {sha(p)}  {os.path.getsize(p)}")
    ref, xe, c64 = load(ref_p), load(xe_p), load(c64_p)
    for tag, a in (("ref", ref), ("xe", xe), ("c64", c64)):
        print(f"  {tag} plane0 rgb {stats(a[0, :, :, :3])}")
        print(f"  {tag} plane0 a   {stats(a[0, :, :, 3])}")
        print(f"  {tag} plane1 rgb {stats(a[1, :, :, :3])}")
        print(f"  {tag} plane1 a   {stats(a[1, :, :, 3])}")
    # history-plane identity across arms
    for tag, a in (("xe", xe), ("c64", c64)):
        d1 = np.abs(a[1] - ref[1])
        print(f"  plane1 |{tag}-ref| max={d1.max():.3e} mean={d1.mean():.3e}")
    d1 = np.abs(xe[1] - c64[1])
    print(f"  plane1 |xe-c64| max={d1.max():.3e} mean={d1.mean():.3e}")
    # is plane1 of ref equal to plane0 shifted? (sanity: history may be previous frame)
    print("  ref plane0==plane1 ?", bool(np.array_equal(ref[0], ref[1])))
    # where do values exit [0,1]?
    for tag, a in (("ref", ref), ("xe", xe), ("c64", c64)):
        r = a[0, :, :, :3]
        lo = int((r < 0.0).sum())
        hi = int((r > 1.0).sum())
        print(f"  {tag} plane0 rgb out-of-[0,1]: below={lo} above={hi} of {r.size}")
