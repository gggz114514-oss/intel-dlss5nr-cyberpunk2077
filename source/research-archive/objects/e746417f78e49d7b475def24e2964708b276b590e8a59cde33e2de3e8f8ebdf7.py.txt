"""validate the index: per-pipeline native-hash agreement among CURRENT records."""
import json
from pathlib import Path
from collections import Counter, defaultdict

CARD = Path(__file__).parent
j = json.loads((CARD / 'evidence' / 'NATIVE-INDEX.json').read_text(encoding='utf-8'))

cur = defaultdict(Counter)
allh = defaultdict(Counter)
for c in j['captures']:
    for e in c['records']:
        if e['identity'] == 'CURRENT':
            for p in (e.get('pass_ordinals') or []):
                cur[p][e['native_hash']] += 1
        for p in (e.get('pass_ordinals') or []):
            allh[p][e['native_hash']] += 1

print('== CURRENT native-hash agreement per pass ==')
bad = []
for o in range(232):
    hs = cur[o]
    if len(hs) != 1:
        bad.append((o, hs))
print('passes with !=1 CURRENT native hash:', len(bad))
for o, hs in bad[:15]:
    print('  ordinal', o, j['coverage'][o]['name'], dict(hs))

print('\n== the single CURRENT hash per pipeline (sample) ==')
seen = set()
for o in range(232):
    p = j['coverage'][o]['pipeline']
    if p in seen:
        continue
    seen.add(p)
    hs = list(cur[o])
    print('  p%-2d pass %-3d %-32s current_hashes=%s  all_hashes=%d'
          % (p, o, j['coverage'][o]['name'], hs, len(allh[o])))

print('\n== UNKNOWN records by capture ==')
unk = Counter()
for c in j['captures']:
    n = sum(1 for e in c['records'] if e['identity'] == 'UNKNOWN')
    if n:
        unk[(c['capture_root'], c['mapping_mode'], (c['plan'] or {}).get('n_passes'))] += n
print('captures with UNKNOWN:', len(unk), ' records:', sum(unk.values()))
for k, v in list(unk.items())[:12]:
    print('  ', v, k[1], k[2], k[0])

print('\n== mapping modes ==')
print(Counter(c['mapping_mode'] for c in j['captures']))
print('\n== plan n_passes ==')
print(Counter((c['plan'] or {}).get('n_passes') for c in j['captures']))
print('\n== plan stale / baseline-flag ==')
print('stale:', sum(1 for c in j['captures'] if c.get('plan_stale')))
print('baseline:', sum(1 for c in j['captures'] if (c['plan'] or {}).get('equals_baseline')))
print('codegen-header diffs:', sum(len(c.get('header_codegen_diff_ordinals') or []) for c in j['captures']))
print('host-side diffs:', sum(len(c.get('host_side_diff_ordinals') or []) for c in j['captures']))
