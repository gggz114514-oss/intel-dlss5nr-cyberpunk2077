import json,struct,statistics as s,sys
from pathlib import Path
D=Path(__file__).resolve().parent;folder=sys.argv[1]
def read(p):return json.loads(p.read_text(encoding='utf-8-sig'))
out=[]
for spec in read(D/folder/'PREPARATION.json'):
 rows=[];gx,gy,gz=spec['grid'];sgn=spec['sg'];phases=[a for a in spec['plans'] if a not in ['control','whole','null','loop']]
 for rep in range(3):
  arms={}
  for arm in spec['plans']:
   p=D/'run'/f"remaining-{spec['label']}-r{rep}-{arm}"
   if not (p/'port_network_receipt.json').exists() or not (p/'clock.bin').exists():continue
   r=read(p/'port_network_receipt.json');t=next(x for x in r['per_pass_gpu'] if x['index']==spec['ordinal']);raw=(p/'clock.bin').read_bytes();assert len(raw)==gx*gy*gz*sgn*16
   ticks=[];by_sg=[[] for _ in range(sgn)];missing=unexpected=0
   for i,(a,b,c,d) in enumerate(struct.iter_unpack('<IIII',raw)):
    nz=bool(a or b or c or d);selected=(i//sgn)%16==0 and arm!='control'
    if selected and nz:
     dt=((c+(d<<32))-(a+(b<<32)))%2**64;ticks.append(dt);by_sg[i%sgn].append(dt)
    elif selected:missing+=1
    elif nz:unexpected+=1
   assert missing==unexpected==0,(spec['label'],rep,arm,missing,unexpected)
   arms[arm]={'ticks':s.mean(ticks) if ticks else 0,'median':s.median(ticks) if ticks else 0,'max':max(ticks) if ticks else 0,'records':len(ticks),'by_sg':[s.mean(v) if v else 0 for v in by_sg],'ms':s.median(t['samples_ms'])}
  row={'rep':rep,'arms':arms}
  if len(arms)==len(spec['plans']):
   row['closure']=sum(arms[a]['ticks'] for a in phases)/arms['whole']['ticks'];row['perturbation']={a:arms[a]['ms']/arms['control']['ms']-1 for a in arms if a!='control'}
   row['empty_fraction']={a:arms['null']['ticks']/arms[a]['ticks'] for a in phases}
   row['active_sg_closure']=sum(sum(arms[a]['by_sg'][:4]) for a in phases)/sum(arms['whole']['by_sg'][:4])
   if 'loop' in arms:row['parts_to_loop']=sum(arms[a]['ticks'] for a in phases)/arms['loop']['ticks'];row['loop_to_projection']=arms['loop']['ticks']/arms['whole']['ticks']
  rows.append(row)
 entry={'label':spec['label'],'ordinal':spec['ordinal'],'phases':phases,'repeats':rows};out.append(entry)
 print(spec['label'],[(r['rep'],len(r['arms']),round(r.get('closure',0),3),round(max(map(abs,r.get('perturbation',{'x':0}).values())),3)) for r in rows])
 if all('closure' in r for r in rows):
  summary={a:{'fraction':s.mean(r['arms'][a]['ticks']/r['arms']['whole']['ticks'] for r in rows),'ticks':s.mean(r['arms'][a]['ticks'] for r in rows),'active_sg_fraction':s.mean(sum(r['arms'][a]['by_sg'][:4])/sum(r['arms']['whole']['by_sg'][:4]) for r in rows),'spread':max(r['arms'][a]['ticks'] for r in rows)/min(r['arms'][a]['ticks'] for r in rows)-1} for a in phases}
  entry['summary']=summary;print(summary)
(D/('MAIN-'+folder.upper()+'-REVIEW.json')).write_text(json.dumps({'units':'raw subgroup elapsed ticks, never exclusive kernel ms','rows':out},indent=2),encoding='utf-8')
