# ---------------------------------------------------------------- shT footprint check
def workgroup_arrays(mod):
    """[(var_id, array_len, elem_bits)] for every Workgroup OpVariable of an array type."""
    out = []
    types, ptr, arr, cons = {}, {}, {}, {}
    for op, o in mod.instr:
        if op == 32:                     # OpTypePointer (result, storage, type)
            ptr[o[0]] = (o[1], o[2])
        elif op == 28:                   # OpTypeArray (result, elem, length_id)
            arr[o[0]] = (o[1], o[2])
        elif op == 43:                   # OpConstant
            cons[o[1]] = o[2:]
        elif op == 22:                   # OpTypeFloat (result, bits)
            types[o[0]] = o[1]
        elif op == 59 and len(o) >= 3 and o[2] == 4:   # Workgroup variable
            pt = ptr.get(o[0])
            if pt and pt[1] in arr:
                elem, ln = arr[pt[1]]
                bits = types.get(elem)
                out.append((o[1], cons.get(ln, [None])[0], bits))
    return out

               and bool(shT_cand) and shT_cand[0][1] == 16384 and shT_cand[0][2] == 16),
    note='shT is not deleted, resized or re-typed: one half array of 16384 elements in both arms',
)
