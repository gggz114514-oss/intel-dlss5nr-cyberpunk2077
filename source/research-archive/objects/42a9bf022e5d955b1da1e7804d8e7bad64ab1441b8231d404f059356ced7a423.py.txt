"""Pack-B plans: identical to the pack-A plans except the vit_block31_weights BUFFER path.

Pack B = per-channel basis fitted on the union of the three FP16-teacher frames AND the actual
quantised-upstream input (4 samples). It is strictly better on the deployed input
(1.608% -> 1.235% Y relL2) and unchanged on the teacher frames, but it must not be promoted
before the pack-A arm proves the CPU<->GPU math (diag2 gate).
"""
from pathlib import Path
import hashlib
import json
import sys

R = Path("@NR_WORKSPACE@")
S = R / "work/dsh-fullnet-static-20261029/S08-vit"
sys.path.insert(0, str(R / "work/dsh-fullnet-static-20261029/tools"))
import planlib  # noqa: E402

PACK_B = S / "PACK/vit_block31_weights.int8.s08b.bin"


def sha(p):
    return hashlib.sha256(Path(p).read_bytes()).hexdigest()


def clone(src_name, dst_name):
    src, dst = S / src_name, S / dst_name
    p = planlib.Plan(src)
    p.assert_valid()
    log = p.set_buffer("vit_block31_weights", file=PACK_B.as_posix())
    lines = [l.rstrip("\r") for l in p.to_text().split("\n")]
    lines = [l for l in lines if l != ""]
    dst.write_bytes(("\r\n".join(lines) + "\r\n").encode("utf-8"))
    q = planlib.Plan(dst)
    q.assert_valid()
    a = [l.rstrip("\r") for l in src.read_text(encoding="utf-8").split("\n") if l.strip()]
    b = [l.rstrip("\r") for l in dst.read_text(encoding="utf-8").split("\n") if l.strip()]
    changed = [(i + 1, x, y) for i, (x, y) in enumerate(zip(a, b)) if x != y]
    assert len(changed) == 1 and len(a) == len(b), changed
    return dict(source=dict(path=str(src), sha256=sha(src)), out=dict(path=str(dst), sha256=sha(dst),
                 header=q.header, dispatch_count=len(q.passes), outputs=len(q.outputs)),
                buffer_edit=log, changed_lines=[dict(line=i, old=x, new=y) for i, x, y in changed])


def main():
    rep = dict(pack_B=dict(path=str(PACK_B), sha256=sha(PACK_B), bytes=PACK_B.stat().st_size),
               prod=clone("candidate-34189.txt", "candidate-34189b.txt"),
               diag=clone("candidate-34189-diag2.txt", "candidate-34189-diag2b.txt"),
               rule="gate pack A first (candidate-34189-diag2.txt); pack B is a one-line BUFFER swap "
                    "on the SAME SPVs and must be gated the same way before any 24-frame run")
    (S / "PLANS-B.json").write_text(json.dumps(rep, indent=1) + "\n", encoding="utf-8")
    print(json.dumps({k: rep[k] for k in ("pack_B",)}, indent=1))
    print("prod", rep["prod"]["out"]["sha256"], "diag", rep["diag"]["out"]["sha256"])
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
