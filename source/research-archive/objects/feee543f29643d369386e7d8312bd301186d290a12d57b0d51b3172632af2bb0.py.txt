"""Print a compact interface/MMA summary of evidence/spv-tail-scan.json."""
from __future__ import annotations

import json
from pathlib import Path

HERE = Path(__file__).resolve().parent
d = json.loads((HERE.parent / "evidence" / "spv-tail-scan.json").read_text(encoding="utf-8"))

for key, v in d.items():
    if "error" in v:
        print(f"=== {key}: ERROR {v['error']}")
        continue
    print(f"=== {key}")
    print(f"    sha256 {v['sha256']}  bytes {v['bytes']}  spv {v['version']}")
    print(f"    mma int8={v['mma_int8']} fp16={v['mma_fp16']} other={v['mma_other']} "
          f"shapes={v['mma_shapes']} int_dots={v['int_dots']}")
    print(f"    float widths {v['float_widths']} int widths {v['int_widths']} "
          f"127={v['has_float_127']} inv127={v['has_float_inv127']}")
    print(f"    ext {v['ext_inst']}")
    print(f"    iface ({v['interface_count']}):")
    for it in v["interface"]:
        bind = "-" if it["binding"] is None else it["binding"]
        print(f"      set={it['set']} bind={str(bind):<3} {str(it['pointee']):<18} "
              f"storage={it['storage']:<14} written={it['written_by_kernel']!s:<5} "
              f"nonwritable={it['non_writable']!s:<5} stride={it['array_stride']}")
