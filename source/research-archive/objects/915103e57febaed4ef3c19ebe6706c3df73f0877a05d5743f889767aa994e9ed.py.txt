# -*- coding: utf-8 -*-
"""Compact human-readable audit of evidence/metrics-raw.json (no new maths)."""
import json
from pathlib import Path

import numpy as np

OUT = Path(__file__).resolve().parent.parent
raw = json.loads((OUT / "evidence/metrics-raw.json").read_text(encoding="utf-8"))


def fmt(v, w=9, p=5):
    return f"{v:+.{p}f}".rjust(w)


print("=== GLOBAL (plane0 RGB, all 2,764,800 samples/channel) ===")
for f in raw["frames"]:
    for arm in ("xe", "c64"):
        g = f["arms"][arm]["scopes"]["global"]
        print(f"{f['frame']} {arm:4s} " + " ".join(
            f"{c}: mean{fmt(g[c]['signed_mean'])} mae{g[c]['mae']:.5f} rmse{g[c]['rmse']:.5f} "
            f"p95{g[c]['abs_p95']:.4f} p99{g[c]['abs_p99']:.4f}" for c in "RGB"))
        print(f"          Y mean{fmt(g['Y']['signed_mean'])} rmse{g['Y']['rmse']:.5f} "
              f"RG ref{g['RG']['ref_mean']:+.5f}->cand{g['RG']['cand_mean']:+.5f} "
              f"(d{fmt(g['RG']['signed_mean'])}) "
              f"BG ref{g['BG']['ref_mean']:+.5f}->cand{g['BG']['cand_mean']:+.5f} "
              f"(d{fmt(g['BG']['signed_mean'])})")
        print(f"          clip any ref{g['ANY']['ref_clip_frac']*100:.3f}% "
              f"cand{g['ANY']['cand_clip_frac']*100:.3f}% | per-ch ref/cand " + " ".join(
              f"{c}:{g[c]['ref_clip_frac']*100:.3f}/{g[c]['cand_clip_frac']*100:.3f}%" for c in "RGB"))

print("\n=== BUCKETS (FP16 luma masks) ===")
for f in raw["frames"]:
    for b in ("dark", "mid", "bright"):
        for arm in ("xe", "c64"):
            s = f["arms"][arm]["scopes"]["buckets"][b]
            print(f"{f['frame']} {b:6s} {arm:4s} n={s['R']['npx']:7d} " + " ".join(
                f"{c}{fmt(s[c]['signed_mean'])}/m{s[c]['mae']:.4f}/p99{s[c]['abs_p99']:.4f}"
                for c in "RGB") + f" | RG{fmt(s['RG']['signed_mean'])} BG{fmt(s['BG']['signed_mean'])}")

print("\n=== CLIP / PLANE1 / ALPHA ===")
for f in raw["frames"]:
    for arm in ("xe", "c64"):
        c = f["arms"][arm]["scopes"]["clip"]
        print(f"{f['frame']} {arm:4s} ref-clipped px={c['clipped_any']['R']['npx']} "
              f"({c['clipped_any']['R']['npx']/921600*100:.3f}%) "
              f"dRGB=({c['clipped_any']['R']['signed_mean']:+.5f},"
              f"{c['clipped_any']['G']['signed_mean']:+.5f},"
              f"{c['clipped_any']['B']['signed_mean']:+.5f}) "
              f"maeB={c['clipped_any']['B']['mae']:.5f}")
        print(f"          plane1 exceed ref={c['plane1_exceed_ref']} cand={c['plane1_exceed_cand']} | "
              f"plane1 dRGB=({c['plane1_global']['R']['signed_mean']:+.5f},"
              f"{c['plane1_global']['G']['signed_mean']:+.5f},"
              f"{c['plane1_global']['B']['signed_mean']:+.5f}) "
              f"| alpha ref_mean={c['plane1_alpha']['ref_mean']:.5f} "
              f"cand={c['plane1_alpha']['cand_mean']:.5f} mae={c['plane1_alpha']['mae']:.5f}")

print("\n=== TILES: worst per frame/arm ===")
for f in raw["frames"]:
    for arm in ("xe", "c64"):
        t = f["arms"][arm]["tiles"]
        def worst(key, use_abs=True):
            k = max(t, key=lambda r: abs(r[key]) if use_abs else r[key])
            return k
        rows = []
        for ch in "RGB":
            k = worst(f"signed_mean_{ch}")
            rows.append(f"{ch}@({k['x']},{k['y']}){k[f'signed_mean_{ch}']:+.5f}")
        mx = worst("mae_Y", use_abs=False)
        mxm = worst("max_abs", use_abs=False)
        print(f"{f['frame']} {arm:4s} signedR/G/B " + " ".join(rows)
              + f" | mae_Y@({mx['x']},{mx['y']})={mx['mae_Y']:.5f}"
              + f" | max|d|@{mxm['x']},{mxm['y']}={mxm['max_abs']:.4f}")
        # top-5 by mae_Y
        top = sorted(t, key=lambda r: -r["mae_Y"])[:5]
        print("          top5 mae_Y: " + " ".join(
            f"({r['x']},{r['y']}){r['mae_Y']:.4f}/B{r['signed_mean_B']:+.3f}" for r in top))

print("\n=== ROI ===")
for f in raw["frames"]:
    for arm in ("xe", "c64"):
        m = f["arms"][arm]
        for name, s in m["scopes"]["roi"].items():
            meta = m["roi_meta"][name]
            print(f"{f['frame']} {arm:4s} {name:16s} n={meta['npx']:6d} L={meta['ref_Y_mean']:.3f} "
                  + " ".join(f"{c}{fmt(s[c]['signed_mean'])}/m{s[c]['mae']:.4f}" for c in "RGB")
                  + f" RG{fmt(s['RG']['signed_mean'])} BG{fmt(s['BG']['signed_mean'])} "
                  f"p99B={s['B']['abs_p99']:.4f}")
