"""Plan generation for n05d-f05-c64-mbatch (CPU only).

Rebases on the *current accepted chain head* (CURRENT.json best_plan =
pkg-vit-16pass, sha 58449908...), replaces only the 8 F05 C64 PASS lines (the
shader group identified by the baseline SPV sha 0d83192b...), asserts the
replacement count is exactly 8, checks that no other line changed and that the
only changed token is the SPV path, then runs the CPU plan check.
"""
import json
import sys
from pathlib import Path

CARD = Path(__file__).resolve().parent.parent
BUILD = CARD / '_build'
C = CARD.parent.parent
sys.path.insert(0, str(C))
import COMMON as K                                   # noqa: E402

GROUP_SHA = '0d83192b5368cc6074e00bbefed67ec71f8f540a9088d4d96752887390459641'
EXPECT_PASSES = 8


def main():
    base_plan, base_sha, chain = K.current_best()
    print('chain head:', base_plan.name, base_sha)
    assert K.sha(base_plan) == base_sha, 'chain-head plan sha mismatch'

    rows = [r for r in K.rows() if r['spv_sha256'] == GROUP_SHA]
    print('group rows:', len(rows), [r['ordinal'] for r in rows])
    assert len(rows) == EXPECT_PASSES, len(rows)
    paths = sorted({Path(r['spv']).as_posix() for r in rows})
    print('distinct baseline spv paths in group:', paths)
    assert len(paths) == 1, paths
    text = Path(base_plan).read_text(encoding='utf-8-sig')
    occ = {p: text.count(p) for p in paths}
    print('occurrences in chain-head plan:', occ)

    out = CARD / 'plan.txt'
    n = K.make_plan(CARD / 'candidate.spv', out, parent_plan=base_plan,
                    spv_shas={GROUP_SHA})
    print('replaced passes:', n)
    assert n == EXPECT_PASSES, 'replaced %d passes, expected %d' % (n, EXPECT_PASSES)

    a = Path(base_plan).read_text(encoding='utf-8-sig').splitlines()
    b = out.read_text(encoding='utf-8-sig').splitlines()
    assert len(a) == len(b), 'line count changed'
    diff = [(i, x, y) for i, (x, y) in enumerate(zip(a, b)) if x != y]
    cand = (CARD / 'candidate.spv').as_posix()

    def only_spv(x, y):
        tx, ty = x.split(), y.split()
        if len(tx) != len(ty):
            return False
        changed = [(p.strip('"'), q.strip('"')) for p, q in zip(tx, ty) if p != q]
        return len(changed) == 1 and changed[0][1] == cand and changed[0][0] in paths

    only_spv_ok = all(only_spv(x, y) for _, x, y in diff)
    print('differing lines:', len(diff), 'only spv path changed:', only_spv_ok)
    assert len(diff) == EXPECT_PASSES, len(diff)
    assert only_spv_ok

    ck = K.cpu_check(out)
    print('cpu_check:', ck.strip())

    res = dict(
        chain_head=dict(path=str(base_plan), sha256=base_sha,
                        chain=[c['label'] for c in chain]),
        plan=dict(path=str(out), sha256=K.sha(out),
                  bytes=len(out.read_bytes()),
                  replaced_passes=n, expected_passes=EXPECT_PASSES,
                  differing_lines_vs_chain_head=len(diff),
                  only_spv_path_changed=only_spv_ok,
                  baseline_path_occurrences=occ,
                  group_ordinals=[r['ordinal'] for r in rows],
                  group_names=[r['name'] for r in rows]),
        cpu_check=dict(argv=[str(K.RUNNER), '--plan', str(out), '--cpu-check'],
                       exit=0, stdout=ck.strip()),
    )
    (BUILD / 'plan.json').write_text(json.dumps(res, indent=2), encoding='utf-8')


if __name__ == '__main__':
    main()
