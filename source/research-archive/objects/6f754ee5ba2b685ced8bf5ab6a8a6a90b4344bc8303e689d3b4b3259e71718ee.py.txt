"""Per-k0-step census of the F16 projection parent ISA (read only).

Answers, for the parent capture:
  * how many sends of each kind/exec-width the kernel has, split by role
  * the per-step instruction budget (steps found from the conditional jmpi sites)
  * where the mov(32) instructions sit relative to the two DPAS
"""
import sys, json, collections
sys.path.insert(0, r'@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/fullnet-continuous-20261027/cards/n11b-vit-projection-hoist/_tools')
import isa_scan as S

ISA = r'@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/fullnet-mechanism-transfer-20261026/f08-register-chain-20261027/captures/native-f20-pixel-owner/ISA/CS_asm61c5144f4358fd6f_simd32_000000000000001b_26_orig.asm'
EXPAND = r'@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/fullnet-mechanism-transfer-20261026/f08-register-chain-20261027/captures/native-f20-pixel-owner/ISA/CS_asm48ac0230660a2c52_simd32_0000000000000017_22_orig.asm'


def role(i):
    com = i['comment']
    tail = com.split(';')[-1].strip() if ';' in com else com
    return tail[:60]


def census(path):
    ins = S.parse(path)
    sends = [i for i in ins if i['op'].startswith('send')]
    cat = collections.Counter()
    for i in sends:
        cat[(i['op'], i['width'], role(i))] += 1
    jmpi = [i for i in ins if i['op'].startswith('jmpi')]
    cond = [i for i in jmpi if i['pred'] and ('&' in i['pred'])]
    mov32 = [i['off'] for i in ins if i['op'] == 'mov' and i['width'] == 32]
    dpas = [i['off'] for i in ins if i['op'].startswith('dpas')]
    sync = [i for i in ins if i['op'].startswith('sync')]
    return dict(total=len(ins), sends=len(sends),
                send_roles={'%s|exec%s|%s' % k: v for k, v in cat.items()},
                jmpi=len(jmpi), jmpi_conditional=len(cond), jmpi_unconditional=len(jmpi) - len(cond),
                mov32=len(mov32), dpas=len(dpas), sync=len(sync),
                first_dpas=dpas[:4], last_dpas=dpas[-4:],
                first_mov32=mov32[:6],
                first_cond_jmpi=[(i['off'], i['text']) for i in cond[:6]])


if __name__ == '__main__':
    out = dict(projection=census(ISA), expand=census(EXPAND))
    print(json.dumps(out, indent=1))
