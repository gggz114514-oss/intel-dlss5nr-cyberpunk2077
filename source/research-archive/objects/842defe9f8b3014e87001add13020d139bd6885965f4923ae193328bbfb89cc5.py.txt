"""Per-function SPIR-V view of the F14 module (parent vs candidate).

Prints, for every OpFunction, its OpName, its instruction count and its
(opcode -> count) histogram, plus the full instruction stream of the norm
helpers so the deleted broadcast can be read directly.
"""
import struct
import sys
from collections import Counter


def words_of(path):
    data = open(path, 'rb').read()
    return list(struct.unpack('<%dI' % (len(data) // 4), data))


def parse(words):
    insts = []
    i = 5
    while i < len(words):
        w = words[i]
        op, count = w & 0xffff, w >> 16
        insts.append((op, words[i + 1:i + count]))
        i += count
    return insts


NAMES = {59: 'OpVariable', 61: 'OpLoad', 62: 'OpStore', 65: 'OpAccessChain',
         43: 'OpConstant', 44: 'OpConstantComposite', 81: 'OpCompositeConstruct',
         82: 'OpCompositeExtract', 83: 'OpCompositeInsert', 124: 'OpBitcast',
         126: 'OpSNegate', 128: 'OpIAdd', 129: 'OpFAdd', 132: 'OpIMul',
         133: 'OpFMul', 134: 'OpUDiv', 197: 'OpNot', 198: 'OpBitFieldInsert',
         199: 'OpBitFieldSExtract', 200: 'OpBitFieldUExtract',
         194: 'OpBitwiseOr', 195: 'OpBitwiseXor', 196: 'OpBitwiseAnd',
         191: 'OpShiftRightLogical', 193: 'OpShiftLeftLogical',
         345: 'GROUP_SHUFFLE(op345)', 346: 'GROUP_SHUFFLE_XOR(op346)',
         327: 'OpSelect', 249: 'GROUP_BROADCAST_ELECT(op249)',
         252: 'OpKill', 253: 'OpReturn', 254: 'OpReturnValue',
         60: 'OpCopyMemory', 84: 'OpCopyObject', 115: 'op115(extinst-fp16?)',
         174: 'op174', 176: 'op176', 170: 'op170', 171: 'op171', 169: 'op169',
         137: 'op137', 246: 'op246', 247: 'op247', 6: 'OpMemberName',
         3: 'OpSource', 4: 'OpSourceExtension', 10: 'OpExtension',
         11: 'OpExtInstImport', 12: 'OpExtInst', 14: 'OpMemoryModel',
         15: 'OpEntryPoint', 16: 'OpExecutionMode', 17: 'OpCapability',
         19: 'OpTypeVoid', 20: 'OpTypeBool', 21: 'OpTypeInt', 22: 'OpTypeFloat',
         23: 'OpTypeVector', 29: 'OpTypeRuntimeArray', 30: 'OpTypeStruct',
         32: 'OpTypePointer', 33: 'OpTypeFunction', 71: 'OpDecorate',
         72: 'OpMemberDecorate', 5: 'OpName', 54: 'OpFunction',
         55: 'OpFunctionParameter', 56: 'OpFunctionEnd', 57: 'OpFunctionCall',
         248: 'OpLabel', 249: 'OpBranch', 250: 'OpBranchConditional'}


def name(op):
    return NAMES.get(op, 'op%d' % op)


def functions(insts):
    """[(name, [ (op, ops) ])] using OpName for function ids."""
    names = {}
    for op, ops in insts:
        if op == 5 and len(ops) >= 2:
            names[ops[0]] = ''.join(chr(c & 0xff) for c in ops[1:])
    funcs = []
    cur = None
    for op, ops in insts:
        if op == 54:
            cur = [names.get(ops[1], 'func%s' % ops[1]), []]
            funcs.append(cur)
            continue
        if cur is not None:
            cur[1].append((op, ops))
        if op == 56:
            cur = None
    return funcs


def main(parent, cand):
    for tag, path in (('PARENT', parent), ('CANDIDATE', cand)):
        insts = parse(words_of(path))
        funcs = functions(insts)
        print('======== %s   %s' % (tag, path))
        print('total module instructions %d, functions %d' % (len(insts), len(funcs)))
        for nm, body in funcs:
            c = Counter(op for op, _ in body)
            print('  fn %-34s instrs=%4d  %s' % (nm, len(body),
                  ', '.join('%s:%d' % (name(k), v) for k, v in sorted(c.items())
                            if v and name(k) not in ('OpLabel', 'OpReturn', 'OpReturnValue'))))
        for nm, body in funcs:
            if 'norm_rcp' in nm or 'rcp_own' in nm or 'tree' in nm:
                print('  --- body of %s ---' % nm)
                for op, ops in body:
                    print('     %-28s %s' % (name(op), list(ops)))


if __name__ == '__main__':
    main(sys.argv[1], sys.argv[2])
