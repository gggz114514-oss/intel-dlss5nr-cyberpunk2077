#!/usr/bin/env python3
"""cumulative-v4 = cumulative-v3 + the nine C128 WO passes (R07 C128 11→2 left).

Same two-namespace discipline as v2: pass identity is the pass NAME (ordinals moved
when R13/R14 inserted a pass), and only BIND resource ids in the card's tail namespace
(>= 739) are remapped onto fresh ids after the base plan's buffer count.
"""
import hashlib
import json
import sys
from pathlib import Path

W = Path(r"@NR_WORKSPACE@/work/dsh-xmx-handoff-validation-20261024")
BASE = W / "candidates/cumulative-v3/plan-cumulative-v3.txt"
CARD = W / "candidates/r07-c128-wo-i8/plan-r07-c128-9x-i8.txt"
G3_NBUF = 739


def parse(p):
    L = Path(p).read_text(encoding="utf-8").splitlines()
    h = L[0].split()
    bufs, passes, tail = {}, {}, []
    i, bid = 1, 0
    while i < len(L):
        l = L[i]
        if l.startswith("BUFFER "):
            bufs[bid] = l
            bid += 1
            i += 1
            continue
        if l.startswith("PASS "):
            nb = int(l.split()[6])
            passes[l.split()[2].strip('"')] = L[i:i + nb + 2]
            i += nb + 2
            continue
        tail.append(l)
        i += 1
    return h, bufs, passes, tail


def main():
    hv, bv, pv, tv = parse(BASE)
    _g3 = Path(r"@NR_WORKSPACE@/candidates/int8-coverage-v2-20261020/plan-g3-production.txt")
    _hg, _bg, g3p, _tg = parse(_g3)
    hc, bc, pc, tc = parse(CARD)
    base_n = len(bv)
    remap = {}
    for bid in sorted(bc):
        if bid >= G3_NBUF:
            remap[bid] = base_n + len(remap)
    mb, mp = dict(bv), dict(pv)
    changed = []
    for name, blk in pc.items():
        if blk == pv.get(name):
            continue
        # precedence: a pass the base already changed (v3's own cards, incl. R13/R14)
        # must not be reverted to the G3 form by this card.
        if pv.get(name) != g3p.get(name):
            continue
        new = []
        for l in blk:
            if l.startswith("BIND "):
                f = l.split()
                r = int(f[2])
                if r in remap:
                    f[2] = str(remap[r])
                new.append(" ".join(f))
            else:
                new.append(l)
        mp[name] = new
        changed.append(name)
    for bid in sorted(remap):
        mb[remap[bid]] = bc[bid]
    out = [" ".join([hv[0], hv[1], str(len(mb)), hv[3], hv[4]])]
    out += [mb[b] for b in sorted(mb)]
    for name in pv:                     # keep the base plan's order
        out += mp[name]
    out += tv
    dst = W / "candidates/cumulative-v4/plan-cumulative-v4.txt"
    dst.parent.mkdir(parents=True, exist_ok=True)
    dst.write_text("\n".join(out) + "\n", encoding="utf-8", newline="\n")
    rep = {"base": str(BASE), "base_sha256": hashlib.sha256(BASE.read_bytes()).hexdigest(),
           "card": str(CARD), "card_sha256": hashlib.sha256(CARD.read_bytes()).hexdigest(),
           "changed_pass_names": changed, "added_buffers": len(remap),
           "cumulative_v4": {"path": str(dst),
                             "sha256": hashlib.sha256(dst.read_bytes()).hexdigest(),
                             "buffers": len(mb), "passes": len(mp)}}
    (W / "candidates/cumulative-v4/MERGE.json").write_text(
        json.dumps(rep, indent=1, ensure_ascii=False) + "\n", encoding="utf-8", newline="\n")
    print(json.dumps({"buffers": len(mb), "passes": len(mp), "changed": len(changed),
                      "sha": rep["cumulative_v4"]["sha256"][:16]}, ensure_ascii=False))


if __name__ == "__main__":
    sys.exit(main())
