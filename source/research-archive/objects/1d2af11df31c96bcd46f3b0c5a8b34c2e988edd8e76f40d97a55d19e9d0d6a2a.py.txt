def tiled(token, c, C):
    return ((token // 16) * (C // 16) + c // 16) * 256 + (token % 16) * 16 + c % 16


def enumerate_map(fn, grid):
    gx, gy, gz = grid
    return {(x, y, sg): fn(x, y, sg) for y in range(gy) for x in range(gx) for sg in range(SG)}


def bijection(mapping, grid, label):
    pairs = list(mapping.values())
    uniq = set(pairs)
    expect = {(t, g) for t in range(0, T, 8) for g in range(G)}
    return {'label': label, 'grid': list(grid),
            'workgroups': grid[0] * grid[1] * grid[2],
            'sg_slots': len(pairs),
            'distinct_token8_group_pairs': len(uniq),
            'expected_pairs': len(expect),
            'duplicate_assignments': len(pairs) - len(uniq),
            'missing_pairs': sorted(map(list, expect - uniq)),
            'extra_pairs': sorted(map(list, uniq - expect)),
            'min_token': min(t for t, g in pairs), 'max_token': max(t for t, g in pairs),
            'groups_seen': sorted({g for t, g in pairs}),
            'bijection_ok': expect == uniq}


def write_sets(mapping):
    offs, dup, total = set(), 0, 0
    for (x, y, sg), (token, g) in mapping.items():
        for t in range(4):
            base = tiled(token, g * 64 + t * 16, 512)
            for r in range(8):
                for c in range(16):
                    off = base + r * 16 + c
                    total += 1
                    if off in offs:
                        dup += 1
                    offs.add(off)
    return offs, dup, total


def wg_membership(mapping, grid):
    gx, gy, _ = grid
    out = {}
    for y in range(gy):
        for x in range(gx):
            toks = sorted({mapping[(x, y, sg)][0] for sg in range(SG)})
            out[(x, y)] = {'tiles': [t // 8 for t in toks],
                           'token_lo': toks[0], 'token_hi': toks[-1] + 7,
                           'groups': sorted({mapping[(x, y, sg)][1] for sg in range(SG)})}
    return out


def launch_order(mapping, grid, wg):
    """Linear workgroup id order x + gx*(y + gy*z): what a wavefront sees."""
    gx, gy, gz = grid
    seq = []
    for y in range(gy):
        for x in range(gx):
            seq.append((x, y, wg[(x, y)]['groups'], tuple(wg[(x, y)]['tiles'])))
    return seq


def window_stats(seq, sizes):
    out = {}
    for w in sizes:
        if w > len(seq):
            continue
        gsets, tsets = [], []
        for i in range(len(seq) - w + 1):
            gs, ts = set(), set()
            for k in range(w):
                gs.update(seq[i + k][2])
                ts.update(seq[i + k][3])
            gsets.append(len(gs))
            tsets.append(len(ts))
        out[str(w)] = {'mean_distinct_group_slabs': sum(gsets) / len(gsets),
                       'max_distinct_group_slabs': max(gsets),
                       'mean_distinct_token8_tiles': sum(tsets) / len(tsets),
                       'max_distinct_token8_tiles': max(tsets),
                       'windows': len(gsets)}
    return out


def runs(seq, key):
    """Length of consecutive runs sharing the same value of seq[i][key]."""
    vals = [s[key] for s in seq]
    lens, cur = [], 1
    for i in range(1, len(vals)):
        if vals[i] == vals[i - 1]:
            cur += 1
        else:
            lens.append(cur)
            cur = 1
    lens.append(cur)
    counts = {}
    for l in lens:
        counts[l] = counts.get(l, 0) + 1
    return {'max': max(lens), 'mean': sum(lens) / len(lens),
            'run_length_counts': {str(k): v for k, v in sorted(counts.items())}}


def row_stats(seq, gx):
    """Per dispatch row (fixed y): distinct group slabs and token tiles in that row."""
    out = []
    for i in range(0, len(seq), gx):
        row = seq[i:i + gx]
        gs, ts = set(), set()
        for r in row:
            gs.update(r[2])
            ts.update(r[3])
        out.append({'y': row[0][1], 'workgroups': len(row),
                    'distinct_group_slabs': len(gs), 'distinct_token8_tiles': len(ts)})
    return {'rows': len(out), 'groups_per_row': out[0]['distinct_group_slabs'],
            'tiles_per_row': out[0]['distinct_token8_tiles'],
            'uniform': all(o['distinct_group_slabs'] == out[0]['distinct_group_slabs'] and
                           o['distinct_token8_tiles'] == out[0]['distinct_token8_tiles'] for o in out),
            'first_two_rows': out[:2], 'last_row': out[-1]}
