"""Splice the N05-c QKV block into candidate.comp (card-local, CPU only).

Replaces the parent's QKV projection block (from the "// AN6:" comment through
the "// row_phase values retire before next phase" line, inclusive) with the
block in _build/qkv-block.glsl.  Everything outside those lines is byte-identical
to the parent source, so SOURCE-DIFF.patch stays minimal.
"""
from pathlib import Path

CARD = Path(__file__).resolve().parent.parent
BUILD = CARD / '_build'

src = (CARD / 'parent-rebuild.comp').read_text(encoding='utf-8')
lines = src.splitlines(keepends=True)

lo = next(i for i, l in enumerate(lines) if l.startswith('    // AN6:'))
hi = next(i for i, l in enumerate(lines)
          if 'row_phase values retire before next phase' in l)
assert lo < hi, (lo, hi)
print('replacing source lines %d..%d (1-based %d..%d)' % (lo, hi, lo + 1, hi + 1))
print('first:', lines[lo].rstrip())
print('last :', lines[hi].rstrip())

block = (BUILD / 'qkv-block.glsl').read_text(encoding='utf-8')
assert block.endswith('\n')

out = lines[:lo] + [block] + lines[hi + 1:]
(CARD / 'candidate.comp').write_text(''.join(out), encoding='utf-8')
print('candidate.comp lines: %d -> %d' % (len(lines), len(''.join(out).splitlines())))
