"""S06 block15 probe: identities, decoder proof, dump semantics, residual formula.

Writes evidence/01-probe.json.  CPU only.
"""
import json
import sys
from pathlib import Path

import numpy as np

sys.path.insert(0, str(Path(__file__).parent))
import c256b15 as L

OUT = L.W / "evidence/01-probe.json"
ev = {"card": "S06-c256", "stage": "probe", "block": 15}

# ---------------------------------------------------------------- identities
ids = {}
for name, p, want in [
    ("plan_combo_34189", L.Q / "real-int8-24/c64-family-static/combo-34189.txt",
     "2b5481d5ce6b8ec8c1f62e68d2057b39d5267825d655e844a2ee2c5500f1c690"),
    ("fp16_pack_swin_15_weights", L.FP16_PACK, L.PF16_SHA),
    ("pack_c256_b15_int8", L.PACK_OLD, L.PACK_OLD_SHA),
    ("spv_c256_expand_mid_int8", L.SPV_EXPAND, "7059c61d2467d8a2b65a7d91616b187b3cce9fec7e24f3ea707931287d7b80ae"),
    ("spv_c256_contract_int8", L.SPV_CONTRACT, "56519c8fc2b33fccd4a74e6f8302f5912c3e2f53815305a11ff05476bb73a126"),
    ("spv_c256_early_qx_producer", L.SPV_PRODUCER, "36707a220903a6bb7fff3669fa5ae158da6ff76003c5f133e70f326c81ab007d"),
    ("spv_r07m_c256_wo_qx_i8", L.SPV_WO, "ab1c011d2e2dfd15dfb0a4ae419308c74f37ebcd27b84424f94425911244c548"),
]:
    got = L.sha256(p) if p.exists() else None
    ids[name] = dict(path=str(p), bytes=p.stat().st_size if p.exists() else None,
                     sha256=got, expected=want, ok=got == want)
ev["identities"] = ids
assert all(v["ok"] for v in ids.values()), "identity mismatch"

XDUMP = L.D / "dump-fp16/dump_block14_out.bin"
MDUMP = L.D / "dump-fp16/dump_swin_15_M.bin"
YDUMP = L.D / "dump-fp16/dump_swin_15_Y.bin"
B15OUT = L.D / "dump-fp16/dump_block15_out.bin"
ev["dumps"] = {p.name: dict(bytes=p.stat().st_size, sha256=L.sha256(p))
               for p in (XDUMP, MDUMP, YDUMP, B15OUT)}

# ------------------------------------------------------------- decode W + q
We, Wm, Wc, rs = L.decode_weights()
old = np.fromfile(L.PACK_OLD, dtype=np.uint8)

qWe, sWe = L.pack_codes(We, 256)
qWm, sWm = L.pack_codes(Wm, 128)
qWc, sWc = L.pack_codes(Wc, 256)
my_we = L.to_pack_bytes(qWe, 256)
my_wm = L.to_pack_bytes(qWm, 128)
my_wc = L.to_pack_bytes(qWc, 256)
old_wc = old[L.WC_OFF:L.WE_OFF].view(np.int8)
old_we = old[L.WE_OFF:L.WM_OFF].view(np.int8)
old_wm = old[L.WM_OFF:L.SC_OFF].view(np.int8)
ev["decoder_proof"] = {
    "note": "decode fp16 pack -> per-output-channel requantise -> byte-compare with shipped int8 pack",
    "wc_equal": int((my_wc == old_wc).sum()), "wc_n": int(my_wc.size),
    "we_equal": int((my_we == old_we).sum()), "we_n": int(my_we.size),
    "wm_equal": int((my_wm == old_wm).sum()), "wm_n": int(my_wm.size),
    "wc_bitexact": bool((my_wc == old_wc).all()),
    "we_bitexact": bool((my_we == old_we).all()),
    "wm_bitexact": bool((my_wm == old_wm).all()),
}

# ------------------------------------------------------ shipped pack scalars
sc = np.frombuffer(old[L.SCAL_OFF:L.SCAL_OFF + 128].tobytes(), dtype="<f4")
ev["shipped_scal"] = dict(invX=float(sc[0]), invA=float(sc[1]), invM=float(sc[2]),
                          sX=float(sc[3]), sA=float(sc[4]), sM=float(sc[5]),
                          absX=float(sc[6]), absA=float(sc[7]), absM=float(sc[8]),
                          K=dict(expand=float(sc[9]), mid=float(sc[10]), contract=float(sc[11])))
ew = np.frombuffer(old[L.EW_OFF:L.EW_OFF + 4096].tobytes(), dtype="<f4")
mw = np.frombuffer(old[L.MW_OFF:L.MW_OFF + 1024].tobytes(), dtype="<f4")
alpha = np.frombuffer(old[L.ALPHA_OFF:L.ALPHA_OFF + 1024].tobytes(), dtype="<f4")
ev["shipped_scale_pack"] = dict(
    ew_equals_sX_times_sWe=bool(np.array_equal(ew, (np.float64(sc[3]) * sWe.astype(np.float64)).astype("<f4"))),
    mw_equals_sA_times_sWm=bool(np.array_equal(mw, (np.float64(sc[4]) * sWm.astype(np.float64)).astype("<f4"))),
    alpha_equals_sM_times_sWc=bool(np.array_equal(alpha, (np.float64(sc[5]) * sWc.astype(np.float64)).astype("<f4"))),
    qx_region_all_zero=bool((old[L.QX_OFF:L.QX_OFF + L.QX_BYTES] == 0).all()),
    qa_region_all_zero=bool((old[L.QA_OFF:L.QA_OFF + L.QA_BYTES] == 0).all()),
    pad_all_zero=bool((old[L.PAD_OFF:L.PAD_OFF + L.PAD_BYTES] == 0).all()),
    rs_equals_fp16_rs=bool(np.array_equal(
        np.frombuffer(old[L.RS_OFF:L.RS_OFF + 1024].tobytes(), dtype="<f4"), rs)),
)

# ------------------------------------------------------------------- token map
idx, oob = L.window_to_canonical()
ev["token_map"] = dict(n_tok=L.N_TOK, n_valid=L.N_VALID, oob=oob,
                       slices_from_c3x_pre="4032 valid tokens; 192 padding tokens")

Xc = L.load_x_canonical(XDUMP)
M = L.load_window_major(MDUMP, 256)
Y = L.load_window_major(YDUMP, 256)
ev["tensor_absmax"] = dict(
    X=float(np.abs(Xc).max()), M=float(np.abs(M).max()), Y=float(np.abs(Y).max()),
    X_rows=int(Xc.shape[0]), M_rows=int(M.shape[0]))

# OOB slots must be zero in the parent dumps
oob_mask = idx < 0
ev["oob_zero_check"] = dict(
    M_oob_absmax=float(np.abs(M[oob_mask]).max()),
    Y_oob_absmax=float(np.abs(Y[oob_mask]).max()),
    M_oob_all_zero=bool((M[oob_mask] == 0).all()),
    Y_oob_all_zero=bool((Y[oob_mask] == 0).all()),
    n_oob=int(oob_mask.sum()),
)

# ------------------------------------- residual formula identification (FP16)
# Y = narrow( f32(M_fp16 @ Wc) * 1.0 + seed ) ; identify seed
Xw = np.zeros((L.N_TOK, L.C_IN), dtype=np.float32)
Xw[~oob_mask] = Xc[idx[~oob_mask]]
base = (M.astype(np.float64) @ Wc.astype(np.float64)).astype(np.float32)
Yd = Y.astype(np.float64)
cand = {
    "seed=0": np.zeros_like(base),
    "seed=X": Xw,
    "seed=X*rs": L.narrow_half(Xw * rs[None, :]),
    "seed=half(X)*rs": L.narrow_half(L.narrow_half(Xw) * rs[None, :]),
}
target = Yd - base.astype(np.float64)
res = {}
for k, s in cand.items():
    d = target - s.astype(np.float64)
    res[k] = dict(rel_l2_of_residual=float(np.sqrt(np.mean(d * d)) / np.sqrt(np.mean(target * target))),
                  full_rel_l2=float(np.sqrt(np.mean((base.astype(np.float64) + s - Yd) ** 2))
                                    / np.sqrt(np.mean(Yd ** 2))))
ev["residual_identification"] = dict(
    note="Y_dump - fp32(M_dump @ Wc); compare against candidate seed branches",
    valid_only={k: {kk: vv for kk, vv in v.items()} for k, v in res.items()},
)
# same, restricted to valid tokens
tgt_v = target[~oob_mask]
ev["residual_identification"]["valid_only"] = {
    k: dict(rel_l2_of_residual=float(np.sqrt(np.mean((tgt_v - s[~oob_mask]) ** 2))
                                     / np.sqrt(np.mean(tgt_v ** 2))))
    for k, s in cand.items()}

# ------------------------------------------------- FP16 parent full reference
Aref = (Xw.astype(np.float64) @ We.astype(np.float64).T).astype(np.float32)
Aact = L.act_half32(L.narrow_half(Aref))
Mref = np.zeros((L.N_TOK, 256), dtype=np.float32)
for g in range(L.MID_GROUPS):
    sl = slice(32 * g, 32 * g + 32)
    Mref[:, sl] = (Aact[:, 128 * g:128 * g + 128].astype(np.float64)
                   @ Wm[sl].astype(np.float64).T).astype(np.float32)
_Mref_h = Mref.astype(np.float16).view(np.uint16)
_M_h = np.ascontiguousarray(M).view(np.uint16)
_Mstats = L.stats(Mref, M.astype(np.float32), ~oob_mask, "fp32 cpu M vs dump M")
_Mstats.update(bit_equal=int((_Mref_h == _M_h).sum()), n_bits=int(_Mref_h.size),
               bit_equal_pct=float((_Mref_h == _M_h).mean() * 100.0))
ev["fp16_reference_vs_dump"] = dict(
    A_absmax=float(np.abs(Aact).max()),
    M=_Mstats,
)
for k, s in cand.items():
    Yref = L.narrow_half(base + s)
    ev["fp16_reference_vs_dump"].setdefault("Y", {})[k] = L.stats(Yref, Y, ~oob_mask, "cpu fp16 ref Y vs dump Y, " + k)

OUT.write_text(json.dumps(ev, indent=1), encoding="utf-8")
print(json.dumps({k: ev[k] for k in ("decoder_proof", "tensor_absmax", "oob_zero_check",
                                     "residual_identification", "fp16_reference_vs_dump")},
                 indent=1)[:6000])
print("written", OUT)
