import re, collections
from pathlib import Path
F = Path(r"@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/fullnet-mechanism-transfer-20261026/f05-k-consumer-tile/c256/captures/native-f20-pixel-owner/ISA/CS_asmfd78c91bb8744f0f_simd32_000000000000000e_13_orig.asm")
lines = F.read_text(encoding="utf-8", errors="replace").splitlines()
FENCE = next(i for i, l in enumerate(lines) if "fence.slm" in l)

def dsc(l):
    m = re.search(r"rd:\d+; ([a-z0-9_.]+)", l.split("//", 1)[1]) if "//" in l else None
    return m.group(1) if m else None

# register writer tracking: key = (regname, suboffset) as it appears in the dst
SEND = re.compile(r"send\.(\w+)\s+\((\d+)\|M0\)\s+(r\d+)\s+(r\d+).*?(0x[0-9A-Fa-f]+)?\s*// (wr:[^;]+; (\S+))")
DPAS = re.compile(r"dpas\.[\d x]+\((\d+)\|M0\)\s+(\S+)\s+(\S+)\s+(\S+)\s+(\S+)")

writes = {}          # regbase -> (line, desc, exec)
dpas_recs = []
for i, l in enumerate(lines):
    m = SEND.search(l)
    if m:
        exec_ = int(m.group(2)); dst = m.group(3); desc = m.group(6)
        writes[dst] = (i, desc, exec_)
    m2 = DPAS.search(l)
    if m2:
        ex = int(m2.group(1))
        dst = m2.group(2).split(":")[0]
        a = m2.group(4).split(":")[0]
        b = m2.group(5).split(":")[0]
        dpas_recs.append((i, ex, dst, a, b))

print("dpas:", len(dpas_recs))
qkv = [d for d in dpas_recs if d[0] < FENCE]
att = [d for d in dpas_recs if d[0] >= FENCE]
print("qkv dpas:", len(qkv), "att dpas:", len(att))
print("exec sizes:", collections.Counter(d[1] for d in dpas_recs))

def prov(reg):
    # exact base match first, then prefix
    if reg in writes:
        return writes[reg]
    for k in writes:
        if k == reg:
            return writes[k]
    return None

for lab, recs in (("QKV", qkv), ("ATT", att)):
    a_prov = collections.Counter()
    b_prov = collections.Counter()
    for i, ex, dst, a, b in recs:
        pa, pb = prov(a), prov(b)
        a_prov[pa[1] if pa else "none"] += 1
        b_prov[pb[1] if pb else "none"] += 1
    print()
    print("== %s : A-operand provenance ==" % lab)
    for k, v in a_prov.most_common():
        print("   %4d %s" % (v, k))
    print("== %s : B-operand provenance ==" % lab)
    for k, v in b_prov.most_common():
        print("   %4d %s" % (v, k))

# how many distinct B registers are consumed, and how many dpas per B register
for lab, recs in (("QKV", qkv), ("ATT", att)):
    cb = collections.Counter(b for _, _, _, _, b in recs)
    ca = collections.Counter(a for _, _, _, a, _ in recs)
    print()
    print("%s distinct B regs %d (uses per reg: %s)" % (lab, len(cb), sorted(collections.Counter(cb.values()).items())))
    print("%s distinct A regs %d (uses per reg: %s)" % (lab, len(ca), sorted(collections.Counter(ca.values()).items())))
