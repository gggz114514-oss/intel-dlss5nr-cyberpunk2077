"""Markdown table of the 73 gathers of ordinal 0 (CPU only, read-only ISA).

Classification is by address range, i.e. by which source site emitted the block:
  0x0B90..0x0BC0  source-pixel read (nrf_rgba / nrf_rgba_clamped), 3 gathers
  0x0CE0..0x1CD0  depth 3x3 search (centre + 4 neighbours), 5 gathers
  0x1CD0..0x3740  history block (60 bilinear corners + 2 motion reads), 62
  0x3810..0x3830  cold-noise read (f[ni..ni+2]), 3 gathers
The two motion gathers sit inside the history block and are not separated here:
they are in the same (skipped) arm and this card does not touch them.

Depths are reported twice: over the whole backward slice (which walks out of the
region into the prologue / push-constant block, hence 49/201) and over the
sub-DAG induced by the entry region only, which is the number the card asks for.
"""
import sys
from collections import Counter
from pathlib import Path

CARD = Path(__file__).resolve().parents[1]
sys.path.insert(0, str(CARD))
from tools.isa_chain import (load, gathers, slice_of, int_tail, depth_of,  # noqa: E402
                             region_of, LAT)

DEAD = [(0x0CE0, 0x3750), (0x3850, 0x47A0)]
DEPTH_BLOCK = 0x1CD0
SOURCE = {'0xb90', '0xba0', '0xbc0'}
NOISE = {'0x3810', '0x3820', '0x3830'}


def is_dead(a):
    return any(lo <= a < hi for lo, hi in DEAD)


def classify(addr):
    a = hex(addr)
    if a in SOURCE:
        return 'source-pixel'
    if a in NOISE:
        return 'cold-noise'
    if addr < DEPTH_BLOCK:
        return 'depth'
    return 'history-block'


def depth_region(isa, nodes):
    """Longest path over edges whose *both* ends are inside the entry region."""
    memo = {}

    def d(k):
        if k in memo:
            return memo[k]
        best = 0
        for p in nodes.get(k, []):
            if p in nodes and region_of(isa[p]['addr']) and region_of(isa[k]['addr']):
                best = max(best, d(p) + 1)
        memo[k] = best
        return best

    return max((d(k) for k in nodes), default=0)


def main():
    isa, by_dw = load()
    gs = gathers(isa)
    rows = []
    for g in gs:
        nodes = slice_of(isa, by_dw, g['idx'])
        tail, _ = int_tail(isa, nodes, g['idx'])
        d, _ = depth_of(isa, nodes)
        w, _ = depth_of(isa, nodes, weight=True)
        td, _ = depth_of(isa, tail)
        reg_nodes = [k for k in nodes if region_of(isa[k]['addr'])]
        rows.append(dict(addr=hex(g['addr']), cls=classify(g['addr']),
                         live='live' if not is_dead(g['addr']) else 'skipped',
                         off=g['ops'][1] if len(g['ops']) > 1 else '?',
                         nreg=len(reg_nodes), dreg=depth_region(isa, nodes),
                         n=len(nodes), d=d, w=w, tail=len(tail), td=td,
                         msg=g['comment'].split(';')[-1].strip()))
    print('| # | gather | class | mask | off | region nodes | **chain depth** | '
          'int-tail | **tail depth** | slice nodes | slice depth | wdepth | message |')
    print('|---|---|---|---|---|---|---|---|---|---|---|---|---|')
    for i, r in enumerate(rows, 1):
        print('| %d | `%s` | %s | %s | %s | %d | %d | %d | %d | %d | %d | %d | `%s` |' % (
            i, r['addr'], r['cls'], r['live'], r['off'], r['nreg'], r['dreg'],
            r['tail'], r['td'], r['n'], r['d'], r['w'], r['msg']))
    by_cls = Counter(r['cls'] for r in rows)
    live = [r for r in rows if r['live'] == 'live']
    print()
    print('classes:', dict(by_cls), 'total', len(rows))
    print('live gathers: %d %s' % (len(live), [r['addr'] for r in live]))
    print('per-gather int-tail sum, all 73: %d ; live 6: %d' % (
        sum(r['tail'] for r in rows), sum(r['tail'] for r in live)))
    print('max region chain depth: %d ; max int-tail depth: %d ; max tail: %d' % (
        max(r['dreg'] for r in rows), max(r['td'] for r in rows),
        max(r['tail'] for r in rows)))


if __name__ == '__main__':
    main()
