"""Resolve the small plan's shader identities by unescaping the doubled backslashes correctly."""
import hashlib, json, sys
from collections import Counter
from pathlib import Path
sys.path.insert(0, str(Path(__file__).parent))
import COMMON as K

SMALL = Path('@NR_WORKSPACE@/work/vk102-small-310-9-16-20261019')
plan = SMALL / 'package' / 'manifest' / 'port_network_runtime.txt'
raw = plan.read_text(encoding='utf-8-sig', errors='replace')
print('sample raw bytes:', raw.splitlines()[3][:120].encode('unicode_escape')[:120])

rows = []
for line in raw.splitlines():
    if not line.startswith('PASS '):
        continue
    # the path is the single double-quoted field that ends in .spv
    start = line.find('"')
    end = line.find('"', start + 1)
    path = line[start + 1:end]
    rows.append((line, path))
print('PASS lines parsed:', len(rows))
p0 = rows[0][1]
print('first path repr:', p0[:60].replace('\\', '/'))
alt = p0.replace('\\\\', '\\')
print('exists raw  :', Path(p0).exists())
print('exists fixed:', Path(alt).exists())

small = {}
for line, path in rows:
    for cand in (path, path.replace('\\\\', '\\')):
        f = Path(cand)
        if f.exists():
            small.setdefault(cand, hashlib.sha256(f.read_bytes()).hexdigest())
            break
print('resolved files:', len(small), 'of', len(rows))
shas = Counter(small.values())
base = {}
for r in K.rows():
    base.setdefault(r['spv_sha256'], set()).add(r['family'])
opt = K.read_json(K.C / 'backend-merged' / 'MANIFEST.json')
opt_shas = {r['spv_sha256'] for r in opt['rows']}
hit = sum(n for s, n in shas.items() if s in base)
print('\nsmall PASS using a shader that exists in the 232-row baseline: %d / %d' % (hit, len(rows)))
for s, n in shas.most_common(15):
    fam = ','.join(sorted(base.get(s, []))) or 'NOT-IN-BASELINE'
    print('  %s %3d PASS  %s' % (s[:12], n, fam))
K.write_json(K.C / 'evidence' / 'SMALL-TRANSFER.json',
             dict(small_passes=len(rows), resolved=len(small), baseline_hit=hit,
                  shas=dict(shas), baseline_families={k: sorted(v) for k, v in base.items()},
                  optimised=len(opt_shas)))
