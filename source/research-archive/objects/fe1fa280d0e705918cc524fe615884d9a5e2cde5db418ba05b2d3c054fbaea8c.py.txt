"""SPIR-V level delta between the reproduced parent and the candidate."""
import sys, json, collections
from pathlib import Path

CARD = Path(__file__).resolve().parents[1]
sys.path.insert(0, str(CARD.parents[1]))
import COMMON as K          # sets up sys.path for spv_tools_local
import spv_tools_local as ST
NAMES = {
    10: 'OpExtension', 11: 'OpExtInstImport', 14: 'OpMemoryModel', 15: 'OpEntryPoint',
    16: 'OpExecutionMode', 19: 'OpTypeVoid', 22: 'OpTypeInt', 23: 'OpTypeFloat',
    24: 'OpTypeVector', 54: 'OpFunction', 55: 'OpFunctionEnd', 56: 'OpFunctionParameter',
    57: 'OpFunctionCall', 59: 'OpVariable', 61: 'OpLoad', 62: 'OpStore',
    65: 'OpAccessChain', 66: 'OpDecorate', 71: 'OpCompositeExtract',
    77: 'OpVectorShuffle', 79: 'OpCompositeConstruct', 80: 'OpCompositeExtract2' ,
    81: 'OpCompositeInsert', 86: 'OpFNegate', 127: 'OpBitcast', 128: 'OpIAdd', 129: 'OpFAdd',
    130: 'OpISub', 131: 'OpFSub', 132: 'OpIMul', 133: 'OpFMul', 134: 'OpUDiv', 135: 'OpSDiv',
    136: 'OpFDiv', 137: 'OpUMod', 138: 'OpSRem', 139: 'OpSMod', 140: 'OpSlessThan',
    164: 'OpLogicalAnd', 169: 'OpSelect', 171: 'OpIEqual', 172: 'OpINotEqual',
    173: 'OpUGreaterThan', 175: 'OpSGreaterThan', 177: 'OpULessThan', 179: 'OpSLessThan',
    199: 'OpGroupNonUniformShuffle', 200: 'OpGroupNonUniformShuffleXor',
    205: 'OpGroupNonUniformBroadcastFirst', 207: 'OpGroupNonUniformBallot',
    245: 'OpBitFieldInsert', 246: 'OpBitFieldSExtract', 247: 'OpBitFieldUExtract',
    250: 'OpBitReverse', 251: 'OpBitCount', 253: 'OpBitwiseOr', 254: 'OpBitwiseXor',
    255: 'OpBitwiseAnd', 194: 'OpShiftRightLogical', 196: 'OpShiftLeftLogical',
    126: 'OpSRem2',
}
INT_SET = {128, 130, 132, 134, 135, 137, 138, 139}


def hist(p):
    m = ST.Module(Path(p))
    c = collections.Counter(ins[0] for ins in m.instructions)
    return m, c


def main():
    pm, pc = hist(CARD / 'parent-rebuild.spv')
    cm, cc = hist(CARD / 'candidate.spv')
    keys = sorted(set(pc) | set(cc))
    rows = []
    for k in keys:
        if pc.get(k, 0) == cc.get(k, 0):
            continue
        rows.append(dict(opcode=k, name=NAMES.get(k, 'op%d' % k),
                         parent=pc.get(k, 0), candidate=cc.get(k, 0),
                         delta=cc.get(k, 0) - pc.get(k, 0)))
    out = dict(
        parent=dict(path=str(CARD / 'parent-rebuild.spv'), sha256=K.sha(CARD / 'parent-rebuild.spv'),
                    instructions=len(pm.instructions)),
        candidate=dict(path=str(CARD / 'candidate.spv'), sha256=K.sha(CARD / 'candidate.spv'),
                       instructions=len(cm.instructions)),
        changed_opcodes=rows,
        int_cf_set=dict(parent={str(k): pc.get(k, 0) for k in sorted(INT_SET)},
                        candidate={str(k): cc.get(k, 0) for k in sorted(INT_SET)}),
        nc_parent=len(ST.no_contraction_ids(pm)),
        nc_candidate=len(ST.no_contraction_ids(cm)),
    )
    K.write_json(CARD / 'tools' / 'spv-delta.json', out)
    print(json.dumps(out, indent=1))


if __name__ == '__main__':
    main()
