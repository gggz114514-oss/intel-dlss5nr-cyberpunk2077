"""Run the zprof profiler on the current chain head with a metric set and dump the counters.

Usage: python run_counters.py <name> [metrics-csv]

The run is measurement-only: it does not change the chain, uses the unique GPU lock, and
enforces the fixed-frame gold. The tool also always writes counter-catalog.json, which lists
every counter the driver exposes (name/description/unit/scope) - that catalogue is how we
find out whether an XMX/matrix-specific counter exists at all.
"""
import json, sys
from pathlib import Path
sys.path.insert(0, str(Path(__file__).parent))
import COMMON as K
import STAGE

DEFAULT = ('GpuTime,AvgGpuCoreFrequencyMHz,XVE_ACTIVE,XVE_STALL,XVE_THREADS_OCCUPANCY_ALL,'
           'XVE_STALL_ALUWR,XVE_STALL_BARRIER,XVE_STALL_CONTROL,XVE_STALL_INSTFETCH,XVE_STALL_OTHER,'
           'XVE_STALL_PIPESTALL,XVE_STALL_PS_DEPENDENCY,XVE_STALL_SBID,XVE_STALL_SENDWR,'
           'ICACHE_HIT,ICACHE_MISS,LOAD_STORE_CACHE_BYTE_READ,LOAD_STORE_CACHE_BYTE_WRITE,'
           'XVE_INST_EXECUTED_BARRIER')

name = sys.argv[1]
metrics = sys.argv[2] if len(sys.argv) > 2 else DEFAULT
base, base_sha, chain = K.current_best()
out = K.C / 'queue' / name
out.mkdir(parents=True, exist_ok=True)
run = out / 'run'
job = dict(id='counters', candidate='head', pair=0,
           plan=dict(path=str(base), sha256=base_sha),
           argv=[str(K.ZPROF), '--plan', str(base), '--out', str(run / 'counters'), '--pool',
                 '--warm', '0', '--reps', '1', '--metrics', metrics],
           runner_output=str(run / 'counters'),
           expected_frame_out=K.FIXED_GOLD, expected_fixed_frame_out=K.FIXED_GOLD,
           expected_frame0=K.FIXED_GOLD,
           metric_group='resource_only')
spec = dict(stage='resources', owner='LUNA-CONTINUOUS-COUNTERS',
            authorization='Main (this session) authorized a counter-profiling run of the chain head; measurement only, no chain change.',
            output_root=str(out), identities=[str(base), str(K.ZPROF)], jobs=[job],
            rules='measurement only; gold required; counters reported raw')
K.write_json(out / 'spec.json', spec)
manifest, ps1 = STAGE.prepare(spec)
STAGE.run(manifest, ps1)
s = STAGE.summarize(spec['output_root'], slim=False)
K.write_json(out / 'STAGE-SUMMARY.json', s)
print('status', s['status'], [(j['id'], j['status'], j['frame']) for j in s['jobs']])
for p in sorted(run.rglob('counter-catalog.json')):
    cat = K.read_json(p)
    names = [(c['name'], c.get('unit'), c.get('scope')) for c in cat['counters']]
    print('catalog:', p)
    print('  counters:', cat['counter_count'])
    key = [n for n in names if any(t in n[0].upper() for t in ('XMX', 'MATRIX', 'DPAS', 'XVE', 'PIPE', 'ALU'))]
    for n in key[:60]:
        print('   ', n)
for p in sorted(run.rglob('hardware-counters.json')):
    d = K.read_json(p)
    print('counters file:', p, 'hardware_passes', d.get('hardware_passes'))
    print('  requested metric entries:', [m['name'] for m in d.get('metrics', [])])
