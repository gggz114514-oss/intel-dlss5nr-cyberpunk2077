"""List all dispatches compactly: ordinal, group, name, spv basename."""
import sys
from pathlib import Path
B = Path("@NR_WORKSPACE@")
W = B / "work/dsh-fullnet-static-20261029"
sys.path.insert(0, str(W / "tools"))
import planlib

plan = planlib.Plan(W / "C00-color-baseline/DIAG-chain62-66-34189.txt")
for p in plan.passes:
    print(f"{p.ordinal:4d} g{p.group:4d} {p.name:34s} {Path(p.spv).name:28s} "
          f"binds={len(p.binds)} push={len(p.push or [])}")
print("--- outputs ---")
for idx, fmt, name, ln in plan.outputs:
    print(idx, plan.buffers[idx].name, fmt, name)
