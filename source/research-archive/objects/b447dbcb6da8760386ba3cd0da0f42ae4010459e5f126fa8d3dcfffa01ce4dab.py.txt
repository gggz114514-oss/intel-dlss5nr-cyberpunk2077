"""S12: inventory of WO / DS / cross-family projection / tail dispatches in combo-34189."""
import json, sys, os
sys.path.insert(0, r"@NR_WORKSPACE@/work/dsh-fullnet-static-20261029/tools")
import planlib

PLAN = r"@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/int8-quality-first-20261029/real-int8-24/c64-family-static/combo-34189.txt"
p = planlib.Plan(PLAN)
print("header", p.header, "passes", len(p.passes), "buffers", len(p.buffers),
      "sha", planlib.sha256_file(PLAN))

targets = ("_body_4", "body_ds", "pre_c", "project_blend", "ioport_block70",
           "ioport_block4_", "block66_fused", "b70", "f2c32")
sel = []
for ps in p.passes:
    nm = ps.name
    if any(t in nm for t in targets):
        sel.append(ps)
print("selected dispatches:", len(sel))

out = []
for ps in sel:
    spvs = ps.spv
    spv_path = spvs if os.path.isabs(spvs) else spvs
    binds = []
    for b in ps.binds:
        buf = p.buffers[b.buffer]
        binds.append({
            "binding": b.binding, "buffer_index": b.buffer, "buffer": buf.name,
            "file": buf.file, "offset": b.offset, "bytes": b.bytes,
            "buffer_bytes": buf.bytes,
        })
    out.append({
        "ordinal": ps.ordinal, "group": ps.group, "name": ps.name,
        "spv": spvs, "spv_sha256": (planlib.sha256_file(spvs) if os.path.exists(spvs) else None),
        "push": ps.push, "binds": binds,
    })

json.dump(out, open(r"@NR_WORKSPACE@/work/dsh-fullnet-static-20261029/S12-proj/_work/plan-targets.json", "w", encoding="utf-8"), indent=1)
for r in out:
    print(r["ordinal"], r["name"], os.path.basename(r["spv"]), (r["spv_sha256"] or "")[:12],
          "push=", r["push"])
    for b in r["binds"]:
        print(f'      b{b["binding"]:<2d} buf#{b["buffer_index"]:<3d} {b["buffer"]:<34s} off={b["offset"]:<7d} len={b["bytes"]:<7d} cap={b["buffer_bytes"]:<7d} {os.path.basename(b["file"])}')
