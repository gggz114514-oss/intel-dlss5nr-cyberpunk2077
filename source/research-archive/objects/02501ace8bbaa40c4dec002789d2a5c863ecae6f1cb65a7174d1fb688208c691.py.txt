class M:
    def __init__(self, path):
        data = Path(path).read_bytes()
        self.words = list(struct.unpack('<%dI' % (len(data) // 4), data))
        self.bound = self.words[3]
        self.ins = []
        i = 5
        while i < len(self.words):
            w = self.words[i]
            op, cnt = w & 0xffff, w >> 16
            self.ins.append((op, self.words[i + 1:i + cnt], i, cnt))
            i += cnt


def main():
    m = M(SPV)
    print('bound ids:', m.bound, 'instructions:', len(m.ins))
    c = collections.Counter(op for op, _, _, _ in m.ins)
    print('\ncensus (top 40):')
    for op, n in c.most_common(40):
        print('  %5d  %s' % (n, NAMES.get(op, 'Op%d' % op)))
    # type table
    types = {}
    for op, ops, _, _ in m.ins:
        if op in (21, 22, 23, 30, 32, 28, 26, 4456):
            types[ops[0]] = (op, ops)
    # 64-bit int types
    i64 = {tid for tid, (op, ops) in types.items() if op == 21 and len(ops) > 1 and ops[1] == 64}
    print('\n64-bit int types:', sorted(i64))
    c64 = collections.Counter()
    for op, ops, _, _ in m.ins:
        if op in (128, 130, 132, 134, 135, 191, 194, 184, 186) and ops[0] in i64:
            c64[NAMES[op]] += 1
    print('64-bit integer arithmetic:', dict(c64))
    conv = collections.Counter()
    for op, ops, _, _ in m.ins:
        if op in (113, 114) and ops[0] in i64:
            conv[NAMES[op]] += 1
    print('to-64-bit conversions:', dict(conv))
    loads = [i for i in m.ins if i[0] == 4457]
    print('\ncooperative matrix loads:', len(loads))
    stores = [i for i in m.ins if i[0] == 4458]
    print('cooperative matrix stores:', len(stores))
    mma = [i for i in m.ins if i[0] == 4459]
    print('cooperative matrix muladd:', len(mma))


if __name__ == '__main__':
    main()
