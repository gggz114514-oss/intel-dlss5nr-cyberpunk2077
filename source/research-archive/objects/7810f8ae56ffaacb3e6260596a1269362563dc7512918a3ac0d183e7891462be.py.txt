"""Run the n02d lane-map probe once with --dump-intermediates, under the unique lock.

The probe writes the 8x64 fragment layout into the debug buffer; the dump is what
probe_read.py needs. The job is expected to be numerically irrelevant to the frame, so a
gold failure is acceptable and simply stops the manifest after that one job.
"""
import sys
from pathlib import Path
sys.path.insert(0, str(Path(__file__).parent))
import COMMON as K
import STAGE

name = sys.argv[1]
plan = Path(sys.argv[2])
out = K.C / 'queue' / name
out.mkdir(parents=True, exist_ok=True)
run = out / 'run'
j = K.gate_job('lanemap-probe', plan, K.sha(plan), run / 'lanemap-probe', 'fixed', 'probe')
j['argv'] = j['argv'] + ['--dump-intermediates']
spec = dict(stage='gates', owner='LUNA-CONTINUOUS-LANEMAP',
            authorization='Main (this session) authorized the n02d lane-map probe only (debug dump); the probe is not a candidate.',
            output_root=str(out / 'gates'), identities=[str(plan), str(K.RUNNER)],
            jobs=[j], rules='lane-map probe; dump only; not adopted')
sp = out / 'spec-gates.json'
K.write_json(sp, spec)
manifest, ps1 = STAGE.prepare(spec)
r = STAGE.run(manifest, ps1)
s = STAGE.summarize(spec['output_root'], slim=False)
K.write_json(Path(spec['output_root']) / 'STAGE-SUMMARY.json', s)
print('probe', s['status'], [(x['id'], x['status'], x['frame'], x['frame_sha']) for x in s['jobs']])
dumps = sorted((run / 'lanemap-probe').glob('dump_*'))
print('dumps', [(d.name, d.stat().st_size) for d in dumps])
