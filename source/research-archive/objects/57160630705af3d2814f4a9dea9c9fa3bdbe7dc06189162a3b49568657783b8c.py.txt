"""N02: apply SOURCE-DIFF.patch to the pristine site files and compare with the card copies."""
import pathlib, re, difflib

CARD = pathlib.Path(r'@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/fullnet-continuous-20261027/cards/n02-f20-f21-tail')
SITE = pathlib.Path(r'@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/c32-boundary-candidates-20261026/f20-four-output')


def parse(block):
    lines = block.splitlines()
    src = lines[0][len('--- '):].split('\t')[0]
    dst = lines[1][len('+++ '):].split('\t')[0]
    hunks, cur = [], None
    for l in lines[2:]:
        m = re.match(r'@@ -(\d+)(?:,(\d+))? \+\d+', l)
        if m:
            cur = {'start': int(m.group(1)) - 1, 'body': []}
            hunks.append(cur)
            continue
        if cur is not None:
            cur['body'].append(l)
    return src, dst, hunks


def apply(text, hunks):
    out = text.split('\n')
    for h in reversed(hunks):
        idx = h['start']
        for l in h['body']:
            if l.startswith(' '):
                assert out[idx] == l[1:], (idx, out[idx], l)
                idx += 1
            elif l.startswith('-'):
                assert out[idx] == l[1:], (idx, out[idx], l)
                del out[idx]
            elif l.startswith('+'):
                out.insert(idx, l[1:])
                idx += 1
    return '\n'.join(out)


patch = (CARD / 'SOURCE-DIFF.patch').read_text(encoding='utf-8')
ok = True
for block in re.split(r'(?m)^(?=--- )', patch):
    if not block.strip():
        continue
    src, dst, hunks = parse(block)
    p = SITE / src.split('/')[-1] if src.startswith('a/') else CARD / src.split('/')[-1]
    new = apply(p.read_text(encoding='utf-8'), hunks)
    tgt = CARD / 'candidate.comp' if 'tail.comp' in src else CARD / 'headers/ioport_common.glsl'
    want = tgt.read_text(encoding='utf-8')
    same = new == want
    ok = ok and same
    print(src, '->', dst, '| hunks', len(hunks), '| target equal:', same)
    if not same:
        for l in difflib.unified_diff(want.split('\n'), new.split('\n'), 'want', 'applied', lineterm=''):
            print('   ', l)
print('PATCH OK', ok)
