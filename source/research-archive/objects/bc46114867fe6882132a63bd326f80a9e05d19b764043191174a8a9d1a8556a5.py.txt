import importlib.util
import json
import os
from pathlib import Path
import tempfile
import time
import unittest

P = Path('@NR_WORKSPACE@/work/github-experimental-game-install-20261024/package/panel/nr_control_panel.py')
spec = importlib.util.spec_from_file_location('panel_under_test', P)
m = importlib.util.module_from_spec(spec)
spec.loader.exec_module(m)

class StatusTests(unittest.TestCase):
    def component(self, key):
        return dict(m.IDENTITIES[key], init_state=2, last_status=0, gpu_timeout=0, fallback=0)
    def test_full_on_small_route(self):
        s=m.describe_backend({'backend':'small','mounted':1,'processor':1},self.component('102'),'102',True)
        self.assertEqual(s['label'],'Full（已挂载）')
        self.assertEqual(s['detail'],'已应用')
    def test_small_and_pending_full(self):
        s=m.describe_backend({'backend':'102','mounted':1,'processor':1},self.component('small'),'102',True)
        self.assertEqual(s['label'],'Small（已挂载）')
        self.assertIn('尚未生效',s['detail'])
    def test_unknown_runtime_not_guessed(self):
        c=self.component('102');c['runtime_sha256']='unknown'
        s=m.describe_backend({'mounted':1,'processor':1},c,'102',True)
        self.assertIsNone(s['actual_key']);self.assertFalse(s['show_metrics'])
    def test_unmounted(self):
        s=m.describe_backend({'mounted':0,'processor':0},self.component('102'),'102',True)
        self.assertEqual(s['label'],'未挂载');self.assertFalse(s['show_metrics'])
    def test_live_and_stale_files(self):
        old=(m.STATE,m.COMPONENT,m.INI)
        try:
            with tempfile.TemporaryDirectory() as d:
                m.STATE=Path(d)/'panel.json';m.COMPONENT=Path(d)/'component.json';m.INI=Path(d)/'panel.ini'
                m.INI.write_text('[NR]\nbackend=102\n',encoding='utf8')
                m.STATE.write_text(json.dumps({'backend':'small','mounted':1,'processor':1}),encoding='utf8')
                m.COMPONENT.write_text(json.dumps(self.component('102')),encoding='utf8')
                self.assertEqual(m.live_status()['backend_status']['actual_key'],'102')
                os.utime(m.COMPONENT,(time.time()-10,time.time()-10))
                s=m.live_status()['backend_status']
                self.assertFalse(s['show_metrics']);self.assertIn('过期',s['label'])
        finally:
            m.STATE,m.COMPONENT,m.INI=old

if __name__=='__main__':unittest.main()
