import json,hashlib
N=Path(__file__).resolve().parent;R=N.parent;A=R.parent;C=N/'combo'
def read(p):return json.loads(p.read_text(encoding='utf-8-sig'))
def save(p,x):p.write_text(json.dumps(x,indent=2),encoding='utf-8')
def ident(p):return dict(path=str(p),sha256=hashlib.sha256(p.read_bytes()).hexdigest())
perf=read(C/'ORDINARY-REVIEW.json');res=read(C/'RESOURCE-REVIEW.json');cov=read(N/'FAMILY-COVERAGE.json')
assert res['status']=='GATES_AND_RESOURCES_PASS'
assert perf['candidate']==ident(C/'plan.txt') and perf['parent']==cov['baseline']
latest=A/'LATEST-FP16-COMBO.json';old=read(latest)
assert old['plan']==cov['baseline']
out=dict(status='COMPLETE',verdict=perf['verdict'],coverage='F01 + 7 body + F2 + F20; 10/232 passes',baseline=cov['baseline'],candidate=perf['candidate'],fixed_fb2='BYTE_IDENTICAL',all_target_scratch_zero=True,all40_scratch_not_increased=True,ordinary=perf,adopted=perf['verdict']=='FASTER',limits=['One combined comparison cannot attribute savings to a single family.','Static rsqt reduction does not identify exposed GPU cycles.','No maximum-register/occupancy inference.','Existing fixed/feedback fixtures, not new game-media validation.'])
save(N/'RESULTS.json',out)
if out['adopted']:
 save(C/'PREVIOUS-LATEST-FP16-COMBO.json',old)
 new=dict(status='VERIFIED_CANDIDATE_NOT_DEPLOYED',plan=perf['candidate'],components=[f['spv'] for f in cov['families']]+[cov['F20_existing']],previous_baseline=cov['baseline'],previous_record=str(C/'PREVIOUS-LATEST-FP16-COMBO.json'),incremental_ordinary=perf,resource_review=ident(C/'RESOURCE-REVIEW.json'),scope='10 C32-family pass replacements; rest inherited unchanged; no deployment')
 save(latest,new)
cov['status']='COMPLETE_'+perf['verdict'];save(N/'FAMILY-COVERAGE.json',cov)
mean=perf['mean_savings_ms'];lo,hi=perf['ci95']
text=f'''# C32 family norm owner results

Outcome: {perf['verdict']}. Parent minus candidate: {mean:+.6f} ms/frame; 95% CI [{lo:+.6f}, {hi:+.6f}]. Three alternating process pairs, 30 samples each arm, all 180 samples retained. No identical rerun to seek significance.

Coverage: F01, seven body passes, F2 and F20. Combined fixed and fb2 frames are byte-identical to the expected outputs. Driver scratch remains zero for all five target pipeline identities; no other pipeline scratch allocation increased. F01/body/F2 shared sizes unchanged; F20 25088 -> 24832 bytes.

Body/F01/F2 math.rsqt 16 -> 8 per native, MOV +14, SEL -16, static instructions -10; DPAS/SLM/UGM/barrier counts unchanged. This verifies arithmetic ownership redistribution, not elimination of the entire norm bridge. F20 uses the previously gated compact coefficient-pair implementation, not the failed distributed norm implementation.

{'The combined plan becomes the verified candidate baseline, with the previous record preserved. It is not deployed.' if out['adopted'] else 'The accepted f20-score-owner baseline stays unchanged. This candidate is retained as a numerical asset; no proven net speed gain is claimed.'}

The combined measurement cannot separate family contributions. Remaining bottleneck attribution is unresolved: this experiment does not establish whether rsqrt was hidden, whether exchange costs offset it, or whether another stage dominates. These require a separately discriminating diagnosis; no speculative new register allocation is added.

See METHOD.md for mapping, rounding/lifetime proof and F20 migration boundaries; NATIVE-REVIEW.json, combo/RESOURCE-REVIEW.json and combo/ORDINARY-REVIEW.json for evidence.
'''
(N/'RESULTS.md').write_text(text,encoding='utf-8')
save(N/'STATE.json',dict(status='COMPLETE',verdict=perf['verdict'],accepted_baseline=read(latest)['plan'],no_gpu_jobs_pending=True))
(N/'RESUME.txt').write_text('This bounded family experiment is complete. Read RESULTS.md and METHOD.md. Do not rerun the identical timing batch. Preserve all candidates and accepted baseline. F20 fully register-owned norm still requires a separate exact arithmetic/owner proof.\n',encoding='utf-8')
files=[p for p in N.rglob('*') if p.is_file() and p.suffix in ['.json','.md','.txt','.py','.comp','.spv'] and p.name!='MANIFEST.json']
save(N/'MANIFEST.json',dict(files=[ident(p) for p in sorted(files)],note='Evidence/source/plan inventory; large native and frame files retained at paths recorded in execution receipts and native review.'))
