"""Build identity for n05e-f05-atile (CPU only: glslang-16.5.0 + spv_validate_cpu).

1. compile the parent source with the RECIPES.json recipe headers (pinned
   idt-identity-recipe copy of the two .glsl headers) -> must be byte-identical
   to the baseline SPV 20f4074f...;
2. compile candidate.comp the same way -> candidate.spv, validate it;
3. structural comparison parent vs candidate at SPIR-V level (cooperative
   matrix opcode counts, loop merges, workgroup storage) -> _build/build.json.

No GPU, no Vulkan device task, no plan, no baseline file is written.
"""
import json
import shutil
import sys
from pathlib import Path

CARD = Path(__file__).resolve().parent.parent
BUILD = CARD / '_build'
C = CARD.parent.parent
sys.path.insert(0, str(C))
import COMMON as K                                   # noqa: E402
import spv_tools_local as ST                         # noqa: E402

PARENT_SHA = '20f4074fc70c32c0e74f0ad000053492e5f5c420259da6044e7a00cfc3ee3f4f'
PIN = C / 'cards/idt-identity-recipe/evidence/headers/20f4074fc70c'

OP = {'TypeCooperativeMatrixKHR': 4456, 'CoopMatLoad': 4457,
      'CoopMatStore': 4458, 'CoopMatMulAdd': 4459, 'CoopMatLength': 4460,
      'LoopMerge': 246, 'Decorate': 71, 'ExtInst': 12, 'Barrier': 224}


def stats(path):
    m = ST.Module(Path(path))
    counts = {}
    for _, op, _ in ST.code_instructions(m):
        counts[op] = counts.get(op, 0) + 1
    return m, counts


def main():
    incs = [PIN]
    for p in sorted(PIN.glob('*.glsl')):
        print('header', p.name, K.sha(p))

    K.compile_glsl(CARD / 'parent-rebuild.comp', BUILD / 'parent-raw.spv', incs)
    psha = K.sha(BUILD / 'parent-raw.spv')
    print('parent raw sha', psha, 'match', psha == PARENT_SHA)
    if psha == PARENT_SHA:
        shutil.copyfile(BUILD / 'parent-raw.spv', CARD / 'parent-rebuild.spv')
    K.validate_spv(BUILD / 'parent-raw.spv')

    K.compile_glsl(CARD / 'candidate.comp', BUILD / 'cand-raw.spv', incs)
    csha = K.sha(BUILD / 'cand-raw.spv')
    print('candidate sha', csha)
    shutil.copyfile(BUILD / 'cand-raw.spv', CARD / 'candidate.spv')
    K.validate_spv(CARD / 'candidate.spv')
    print('validate: exit 0 for both')

    pm, pc = stats(BUILD / 'parent-raw.spv')
    cm, cc = stats(BUILD / 'cand-raw.spv')
    out = dict(
        parent=dict(path=str(K.row_by_ordinal(19)['spv']), sha256=PARENT_SHA,
                    source=str(K.row_by_ordinal(19)['source']),
                    source_sha256=K.sha(CARD / 'parent-rebuild.comp'),
                    rebuilt_sha256=psha, raw_match=psha == PARENT_SHA,
                    bytes=len((BUILD / 'parent-raw.spv').read_bytes())),
        candidate=dict(path=str(CARD / 'candidate.spv'), sha256=csha,
                       source=str(CARD / 'candidate.comp'),
                       source_sha256=K.sha(CARD / 'candidate.comp'),
                       bytes=len((CARD / 'candidate.spv').read_bytes())),
        include_dirs=[str(p) for p in incs],
        recipe='raw (RECIPES.json 20f4074f: pinned headers, no build defines, '
               'no decoration transfer)',
        static=dict(parent={k: pc.get(v, 0) for k, v in OP.items()},
                    candidate={k: cc.get(v, 0) for k, v in OP.items()},
                    delta={k: cc.get(v, 0) - pc.get(v, 0) for k, v in OP.items()}),
        nc=dict(parent=sorted(ST.no_contraction_ids(pm)),
                candidate=sorted(ST.no_contraction_ids(cm))),
    )
    out['nc']['parent_n'] = len(out['nc']['parent'])
    out['nc']['candidate_n'] = len(out['nc']['candidate'])
    out['nc']['same_sites'] = out['nc']['parent'] == out['nc']['candidate']
    (BUILD / 'build.json').write_text(json.dumps(out, indent=2), encoding='utf-8')

    print(json.dumps(out['static'], indent=1))
    print('NC parent=%d candidate=%d same=%s' % (out['nc']['parent_n'],
                                                 out['nc']['candidate_n'],
                                                 out['nc']['same_sites']))


if __name__ == '__main__':
    main()
