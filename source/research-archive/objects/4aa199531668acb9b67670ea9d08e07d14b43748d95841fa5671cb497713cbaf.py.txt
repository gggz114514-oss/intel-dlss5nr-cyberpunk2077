import sys, os, re, collections
sys.path.insert(0, r"@NR_WORKSPACE@/work/dsh-fullnet-static-20261029/ledger/_work")
import spvscan
from pathlib import Path
P = Path(r"@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/fullnet-continuous-20261027/backend-merged/plan.txt")
lines = P.read_text(encoding="utf-8", errors="replace").splitlines()
spvs=[]
for l in lines:
    if l.startswith("PASS "):
        t = l.split()
        # PASS group name spv ... or PASS group "name" "spv" ...
        cand = [x.strip('"') for x in t[2:] if x.strip('"').lower().endswith(".spv")]
        if cand: spvs.append(cand[0])
print("PASS lines", len(spvs), "unique", len(set(spvs)))
tot=collections.Counter()
for s in dict.fromkeys(spvs):
    v = spvscan.scan(Path(s))
    tot[(v["mma_int8"], v["mma_fp16"], v["mma_other"])]+=1
    print(f"{os.path.basename(s):44s} i8={v['mma_int8']:3d} f16={v['mma_fp16']:3d} oth={v['mma_other']:3d} 127={str(v['has_float_127'])[0]} i127={str(v['has_float_inv127'])[0]}")
print("histogram (int8,fp16,other)->count", dict(tot))
