"""R01 F1 - derive the parent-arm (FP16) gate plan from the frozen diag plan.

The parent arm needs the same pinned window, the same buffers and the same PUSH
as the frozen `plan-c32-i8-diag-f1-scalex2.txt`; only PASS 0's module, its
declared LDS field and its binding count change, because the FP16 parent has no
INT8 weight/scale bindings.  The result is verified against the parent's own
frozen plan block (T3/c32a-f1-transport-v1/port_network_runtime.txt:489-497).
"""
import hashlib
import json
from pathlib import Path

ROOT = Path("@NR_WORKSPACE@")
CARD = ROOT / "work/dsh-xmx-handoff-validation-20261024/candidates/r01-f1"
SRC = ROOT / "candidates/int8-coverage-v2-20261020/C/c32-scale-from-x-v2/plans/plan-c32-i8-diag-f1-scalex2.txt"
PARENT_PLAN = ROOT / "candidates/dhs-netorg-long-20261018/T3/c32a-f1-transport-v1/port_network_runtime.txt"
PARENT_SPV = ROOT / "candidates/dhs-netorg-long-20261018/T3/c32a-f1-transport-v1/c32a_f1_transport.spv"
OUT = CARD / "plan-r01-f1-gate-parent.txt"


def sha(p):
    return hashlib.sha256(Path(p).read_bytes()).hexdigest()


lines = SRC.read_text(encoding="utf-8").splitlines(True)
idx = next(i for i, l in enumerate(lines) if l.startswith("PASS 0 "))
old = lines[idx]
new = ('PASS 0 "f1c32_image_pre" "%s" 32 10240 8 32 168 96 1\n' % PARENT_SPV.as_posix())
assert old.split()[6] == "10"
lines[idx] = new
# drop BIND 8/9 (the INT8 weight and scale buffers, unused by the parent)
drop = [i for i in range(idx + 1, idx + 11)
        if lines[i].startswith("BIND 8 ") or lines[i].startswith("BIND 9 ")]
assert len(drop) == 2 and drop[0] == idx + 9 and drop[1] == idx + 10, drop
removed = [lines[i] for i in reversed(drop)]
for i in sorted(drop, reverse=True):
    del lines[i]
OUT.write_text("".join(lines), encoding="utf-8")

# verification against the parent's own frozen block
t3 = PARENT_PLAN.read_text(encoding="utf-8").splitlines()
t3i = next(i for i, l in enumerate(t3) if l.startswith("PASS 0 "))
mine = OUT.read_text(encoding="utf-8").splitlines()
mi = next(i for i, l in enumerate(mine) if l.startswith("PASS 0 "))
block_mine = mine[mi:mi + 10]
block_t3 = t3[t3i:t3i + 10]
same = [a == b for a, b in zip(block_mine, block_t3)]
push_mine = block_mine[9].split()
push_t3 = block_t3[9].split()
push_word_diff = [{"word_after_size": i, "parent_plan": a, "diag_plan": b}
                  for i, (a, b) in enumerate(zip(push_t3[2:], push_mine[2:])) if a != b]
delta = {
    "schema": "r01-f1/plan-delta/1",
    "purpose": "parent (FP16) arm of the F1 stage gate on the same pinned window as the frozen INT8 capture",
    "base_plan": {"path": SRC.as_posix(), "bytes": SRC.stat().st_size, "sha256": sha(SRC)},
    "new_plan": {"path": OUT.as_posix(), "bytes": OUT.stat().st_size, "sha256": sha(OUT)},
    "changes": [
        {"line": idx + 1, "kind": "PASS 0 replaced", "before": old.rstrip("\n"), "after": new.rstrip("\n"),
         "fields": "module = FP16 parent; declared LDS 20480 -> 10240 (sX8 2048 + sE8 8192 removed); binding count 10 -> 8"},
        {"line": idx + 10, "kind": "BIND 8 removed", "before": removed[1].rstrip("\n"), "after": None,
         "reason": "c32_f1_i8_weights, resource 604, unused by the FP16 parent"},
        {"line": idx + 11, "kind": "BIND 9 removed", "before": removed[0].rstrip("\n"), "after": None,
         "reason": "c32_f1_i8_scales, resource 605, unused by the FP16 parent"},
    ],
    "unchanged": "every BUFFER line, every other PASS/BIND/PUSH/HISTORY/POLICY line, and the F1 PUSH line (pinned window + debug flag) are byte-identical to the base plan",
    "verification": {
        "against": {"path": PARENT_PLAN.as_posix(), "line": t3i + 1, "sha256": sha(PARENT_PLAN)},
        "parent_block_lines_equal": same,
        "push_word_differences": push_word_diff,
        "note": "PASS 0 and BIND 0..7 reproduce the parent's own frozen block line for line. The PUSH line is the base plan's diagnostic PUSH and differs from the parent's production PUSH in exactly the words listed above; both are read by the same kernel and only the diagnostic pin/debug word differ, so the resource wiring is unchanged.",
    },
    "line_count": {"base": len(SRC.read_text(encoding='utf-8').splitlines()), "new": len(mine)},
}
(CARD / "PLAN-DELTA.json").write_text(json.dumps(delta, indent=1, ensure_ascii=False) + "\n", encoding="utf-8")
print(json.dumps({"new_sha256": delta["new_plan"]["sha256"], "bytes": delta["new_plan"]["bytes"],
                  "parent_block_equal": same, "n_changes": len(delta["changes"])}, indent=1))
