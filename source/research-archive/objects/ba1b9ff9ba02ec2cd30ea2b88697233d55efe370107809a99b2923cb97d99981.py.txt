import json, sys
p = r'@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/fullnet-mechanism-transfer-20261026/fullnet-connection-atlas-20261027/MODULE-TIMINGS.json'
d = json.load(open(p, encoding='utf-8-sig'))
tot = 0.0
for e in d:
    t = e['timing']
    fx = t.get('fixed-frame0', {}).get('median')
    fb = t.get('fb2-frame1', {}).get('median')
    n = len(e['ordinals'])
    if fx:
        tot += fx * n
    if e['family'] in ('F13', 'F14', 'F15', 'F16'):
        print('%s n=%d ord0=%d fixed=%s fb2=%s ordinals=%s' % (e['family'], n, e['ordinals'][0], fx, fb, e['ordinals']))
print('sum of fixed medians over all families (ns->ms):', round(tot, 4))
