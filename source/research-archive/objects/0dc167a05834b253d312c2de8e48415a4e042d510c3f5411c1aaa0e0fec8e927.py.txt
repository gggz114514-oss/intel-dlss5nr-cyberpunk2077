"""make_candidate.py - apply the R02 QKV INT8 edit as an exact line-range
replacement of the verbatim parent source.

The parent file is the byte-identical copy in ../source/parent; this script
replaces parent lines 507..667 (candidate numbering) with the INT8 QKV block and
records every replaced/inserted line in ../evidence/CHANGE-SET.json so the edit
set is enumerable.  Run once; the output must not be hand-edited afterwards.
"""
import hashlib
import json
import sys
from pathlib import Path

CARD = Path(__file__).resolve().parents[1]
PARENT = CARD / "source/parent/c32_body1_i8.comp"
CAND = CARD / "source/candidate/c32_body1_i8.comp"
EV = CARD / "evidence/CHANGE-SET.json"

FIRST = 506   # "NRH_B yb0 = nrh_join_rows_to_B(y0, y1);"
LAST = 666    # closing brace of the V projection block

NEW = '''        // ---- R02 QKV INT8 (replaces 24 static FP16 MMA) ---------------------
        // Stage-1 explicit shared INT8 handoff: the contract row Y is quantized
        // once per subgroup into sY8 in the measured I8B K32 tile (byte n*32+k),
        // then each projection is one K=32 integer step per 8 output rows,
        // dequantized once per output channel and narrowed.  The FP16 register B
        // handoff (nrh_join_rows_to_B) is gone: the B operand is loaded from the
        // shared INT8 tile, never cast from a FP16 K16 fragment.
        nr_i8_put_y(y0, 0u, sg * 512u);
        nr_i8_put_y(y1, 8u, sg * 512u);
        nr_i8_put_y(y2, 16u, sg * 512u);
        nr_i8_put_y(y3, 24u, sg * 512u);
        subgroupMemoryBarrierShared();
        subgroupBarrier();
        NRH_CH k0;
        NRH_CH k1;
        NRH_CH k2;
        NRH_CH k3;
        NRH_CH v0;
        NRH_CH v1;
        NRH_CH v2;
        NRH_CH v3;
        {
            I8B yb8;
            coopMatLoad(yb8, sY8, sg * 512u, 32u, LCOL);
            I8A wa;
            I8C za;
            // Q: output channels 0..31 over the 32 contract channels, K=32 once.
            za = I8C(0);
            coopMatLoad(wa, w8, C32_I8_WQ + 0u*1024u, 32u, LCOL);
            za = coopMatMulAdd(wa, yb8, za);
            q0 = nrh_narrow(nr_i8_deq(za, 0u, C32_I8_DQ_Q));
            za = I8C(0);
            coopMatLoad(wa, w8, C32_I8_WQ + 1u*1024u, 32u, LCOL);
            za = coopMatMulAdd(wa, yb8, za);
            q1 = nrh_narrow(nr_i8_deq(za, 8u, C32_I8_DQ_Q));
            za = I8C(0);
            coopMatLoad(wa, w8, C32_I8_WQ + 2u*1024u, 32u, LCOL);
            za = coopMatMulAdd(wa, yb8, za);
            q2 = nrh_narrow(nr_i8_deq(za, 16u, C32_I8_DQ_Q));
            za = I8C(0);
            coopMatLoad(wa, w8, C32_I8_WQ + 3u*1024u, 32u, LCOL);
            za = coopMatMulAdd(wa, yb8, za);
            q3 = nrh_narrow(nr_i8_deq(za, 24u, C32_I8_DQ_Q));
            // K: identical shape, its own weight tiles and channel factors.
            za = I8C(0);
            coopMatLoad(wa, w8, C32_I8_WK + 0u*1024u, 32u, LCOL);
            za = coopMatMulAdd(wa, yb8, za);
            k0 = nrh_narrow(nr_i8_deq(za, 0u, C32_I8_DQ_K));
            za = I8C(0);
            coopMatLoad(wa, w8, C32_I8_WK + 1u*1024u, 32u, LCOL);
            za = coopMatMulAdd(wa, yb8, za);
            k1 = nrh_narrow(nr_i8_deq(za, 8u, C32_I8_DQ_K));
            za = I8C(0);
            coopMatLoad(wa, w8, C32_I8_WK + 2u*1024u, 32u, LCOL);
            za = coopMatMulAdd(wa, yb8, za);
            k2 = nrh_narrow(nr_i8_deq(za, 16u, C32_I8_DQ_K));
            za = I8C(0);
            coopMatLoad(wa, w8, C32_I8_WK + 3u*1024u, 32u, LCOL);
            za = coopMatMulAdd(wa, yb8, za);
            k3 = nrh_narrow(nr_i8_deq(za, 24u, C32_I8_DQ_K));
            // V: identical shape; V never passes through the norm below.
            za = I8C(0);
            coopMatLoad(wa, w8, C32_I8_WV + 0u*1024u, 32u, LCOL);
            za = coopMatMulAdd(wa, yb8, za);
            v0 = nrh_narrow(nr_i8_deq(za, 0u, C32_I8_DQ_V));
            za = I8C(0);
            coopMatLoad(wa, w8, C32_I8_WV + 1u*1024u, 32u, LCOL);
            za = coopMatMulAdd(wa, yb8, za);
            v1 = nrh_narrow(nr_i8_deq(za, 8u, C32_I8_DQ_V));
            za = I8C(0);
            coopMatLoad(wa, w8, C32_I8_WV + 2u*1024u, 32u, LCOL);
            za = coopMatMulAdd(wa, yb8, za);
            v2 = nrh_narrow(nr_i8_deq(za, 16u, C32_I8_DQ_V));
            za = I8C(0);
            coopMatLoad(wa, w8, C32_I8_WV + 3u*1024u, 32u, LCOL);
            za = coopMatMulAdd(wa, yb8, za);
            v3 = nrh_narrow(nr_i8_deq(za, 24u, C32_I8_DQ_V));
        }
'''


def sha(p):
    return hashlib.sha256(p.read_bytes()).hexdigest()


def main():
    src = PARENT.read_text(encoding="utf-8").splitlines(keepends=True)
    old = src[FIRST - 1:LAST]
    if old[0].strip() != "NRH_B yb0 = nrh_join_rows_to_B(y0, y1);":
        sys.exit("anchor 1 mismatch: %r" % old[0])
    if old[-1].strip() != "}":
        sys.exit("anchor 2 mismatch: %r" % old[-1])
    if "v3 = nrh_narrow(z3);" not in old[-2]:
        sys.exit("anchor 3 mismatch: %r" % old[-2])
    new = src[:FIRST - 1] + [NEW] + src[LAST:]
    # edit B (recorded separately): declare the QKV activation tile beside the
    # existing int8 arrays, i.e. before the include that defines nr_i8_put_y.
    SE = "shared int8_t sE8[8192];\n"
    i = new.index(SE)
    SY = ("shared int8_t sY8[2048];   // R02 QKV input tile: 4 subgroups x "
          "(16 tokens x 32 ch)\n")
    new = new[:i + 1] + [SY] + new[i + 1:]
    CAND.write_text("".join(new), encoding="utf-8", newline="")
    removed = [l.rstrip("\n") for l in old]
    added = NEW.splitlines()
    rec = {
        "parent": str(PARENT), "parent_sha256": sha(PARENT),
        "candidate": str(CAND), "candidate_sha256": sha(CAND),
        "parent_lines": len(src), "candidate_lines": len(new),
        "edits": [
            {"name": "A: replace the FP16 QKV block",
             "replaced_range_parent_lines": [FIRST, LAST],
             "replaced_line_count": len(removed),
             "inserted_line_count": len(added),
             "removed": removed, "added": added},
            {"name": "B: declare the QKV int8 activation tile",
             "anchor": "shared int8_t sE8[8192];",
             "inserted_line": SY.rstrip("\n"),
             "inserted_line_count": 1},
        ],
        "summary": {
            "removed_fp16_mma": 24,
            "removed_fp16_weight_loads": 24,
            "removed_fp16_register_b": ["nrh_join_rows_to_B(y0,y1)",
                                        "nrh_join_rows_to_B(y2,y3)"],
            "added_int8_mma": 12,
            "added_int8_weight_tiles": 12,
            "added_int8_activation_tile_bytes_per_sg": 512,
            "added_declared_shared_bytes": 2048,
            "added_barriers": ["subgroupMemoryBarrierShared()",
                               "subgroupBarrier()"],
            "added_quantization_writes_per_sg": "4 calls x 32 lanes x 4 components = 512 int8 codes per subgroup instance (16 tokens x 32 channels); 2048 per workgroup with 4 active subgroups",
        },
    }
    EV.write_text(json.dumps(rec, ensure_ascii=False, indent=1) + "\n", encoding="utf-8")
    print(json.dumps({"parent_sha256": rec["parent_sha256"],
                      "candidate_sha256": rec["candidate_sha256"],
                      "parent_lines": rec["parent_lines"],
                      "candidate_lines": rec["candidate_lines"],
                      "edit_A_replaced": rec["edits"][0]["replaced_line_count"],
                      "edit_A_inserted": rec["edits"][0]["inserted_line_count"]},
                     indent=1))


if __name__ == "__main__":
    main()
