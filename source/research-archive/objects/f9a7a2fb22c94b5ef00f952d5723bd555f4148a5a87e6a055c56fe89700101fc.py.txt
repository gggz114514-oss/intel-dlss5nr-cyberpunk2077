#!/usr/bin/env python3
"""Apply the R12 PV-INT8 change set (C1..C8) to a split16 l2 source file.

The same eight edits apply to both module identities of R12:
  * the 15 ordinary positions (`split16_attention_qkv_int8_inline.comp`, prod build with
    -DL2_INLINE_QT=1 -DL2_NO_T_WRITE=1),
  * block30 (`split16_attention_qkv_int8.comp`, no macros: it still writes the half T
    that its own l3 consumes).
They differ only in the PV-exit tail, which this change set does not touch.

Every anchor must occur exactly once; the script fails loudly otherwise, so an
unexpected source drift cannot silently produce a partial patch.
"""
import hashlib
import json
import sys
from pathlib import Path

EDITS = [
    ("C1", """shared int8_t sy8[4096];
#define sy phase
#define sp phase
#if !ROW_OWNER

#endif
// PV uses the retired Q prefix as its standard half bridge; P stays live.
// Q[0..1024) is dead after the score WG barrier, V remains at [4096..6144).
uint v_raw(uint e) { return e<4096u ? e : 4096u+((e-4096u)%32u)*64u+(e-4096u)/32u; }
uint wq_tile(uint r,uint k) { return uint(pc.p3.y)+((r/16u)*32u+k/16u)*256u; }
""",

def apply(src_text):
    text = src_text
    applied = []
    for cid, old, new in EDITS:
        n = text.count(old)
        if n != 1:
            raise SystemExit("anchor %s occurs %d times (expected 1)" % (cid, n))
        text = text.replace(old, new)
        applied.append({"id": cid, "anchor_lines": old.count("\n"),
                        "added_lines": new.count("\n")})
    return text, applied


def main(argv):
    src, dst, rep = argv[0], argv[1], argv[2]
    raw = Path(src).read_bytes()
    text = raw.decode("utf-8")
    crlf = "\r\n" in text
    lf = text.replace("\r\n", "\n")
    out, applied = apply(lf)
    if crlf:
        out = out.replace("\n", "\r\n")
    Path(dst).write_bytes(out.encode("utf-8"))
    rec = {"source": src, "source_sha256": hashlib.sha256(raw).hexdigest(),
           "source_line_endings": "CRLF" if crlf else "LF",
           "candidate": dst, "candidate_sha256": hashlib.sha256(out.encode("utf-8")).hexdigest(),
           "source_lines": lf.count("\n") + (0 if lf.endswith("\n") else 1),
           "candidate_lines": out.count("\r\n" if crlf else "\n") + (0 if out.endswith("\n") else 1),
           "edits": applied, "edit_count": len(applied),
           "removed_lines": sum(e["anchor_lines"] for e in applied),
           "added_lines": sum(e["added_lines"] for e in applied)}
    Path(rep).write_text(json.dumps(rec, indent=1) + "\n", encoding="utf-8", newline="\n")
    print("applied", len(applied), "edits ->", dst, rec["candidate_sha256"][:16])
    return 0


if __name__ == "__main__":
    sys.exit(main(sys.argv[1:]))
