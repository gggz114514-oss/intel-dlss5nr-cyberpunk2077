"""Regression test: re-rebasing each adopted card onto its predecessor must reproduce the accepted plan.

This guards the tool change that made rebase carry per-token plan field edits (grid etc.).
It also proves each accepted step is reproducible from (predecessor plan, card candidate).
"""
import sys
from pathlib import Path
sys.path.insert(0, str(Path(__file__).parent))
import COMMON as K
from audit_card import pass_lines

cur = K.read_json(K.C / 'CURRENT.json')
chain = cur['chain']
cards = {i: chain[i].get('label') for i in range(1, len(chain))}
ok, bad = [], []
for i in range(1, len(chain)):
    step = chain[i]
    label = step.get('label') or ''
    if step.get('kind', '').startswith('ablation') or step.get('kind', '').startswith('restore'):
        continue
    d = K.C / 'cards' / label
    if not (d / 'candidate.spv').exists():
        continue
    prev = Path(chain[i - 1]['path'])
    try:
        plan, sha, n, meta = K.rebase_by_ordinals(label, out_plan=K.C / 'queue' / '_rebase_test' / ('%02d.txt' % i))
    except Exception as e:
        bad.append((i, label, 'rebase raised: %r' % (e,)))
        continue
    acc = Path(step['path'])
    a = [x['line'] for x in pass_lines(plan)]
    b = [x['line'] for x in pass_lines(acc)]
    diff = [j for j in range(232) if a[j] != b[j]]
    if diff:
        bad.append((i, label, 'differs from accepted plan on %d rows: %s' % (len(diff), diff[:6])))
    else:
        ok.append((i, label, n, meta.get('field_carried')))
print('reproduced exactly: %d' % len(ok))
for r in ok:
    print('  step %d %-24s replaced=%d fields_carried=%s' % r)
print('mismatches: %d' % len(bad))
for r in bad:
    print('  step %d %-24s %s' % r)
