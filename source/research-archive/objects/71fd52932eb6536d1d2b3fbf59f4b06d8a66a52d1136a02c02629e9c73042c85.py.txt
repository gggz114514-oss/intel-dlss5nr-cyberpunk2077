import sys
from pathlib import Path
CAMP = Path('@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/fullnet-continuous-20261027')
sys.path.insert(0, str(CAMP))
sys.path.insert(0, r'@NR_WORKSPACE@/build/zcode-fp16-compute-backend-20261021/scripts')
import spv_tools_local as ST

SC = {0: 'UniformConstant', 1: 'Input', 2: 'Uniform', 3: 'Output', 4: 'Workgroup',
      5: 'CrossWorkgroup', 6: 'Private', 7: 'Function', 8: 'Generic', 9: 'PushConstant',
      10: 'AtomicCounter', 11: 'Image', 12: 'StorageBuffer'}

def trace(path, tag):
    m = ST.Module(Path(path))
    print('==', tag)
    for op, ops, wi, wc in m.instructions:
        if op == 4458:
            chain = []
            cid = ops[0]
            for _ in range(8):
                dop, dops = m.defining_instruction(cid)
                if dop is None:
                    chain.append('?')
                    break
                if dop == 59:
                    chain.append('var(sc=%s)' % SC.get(dops[2], dops[2]))
                    break
                if dop in (65, 66):
                    chain.append('chain')
                    cid = dops[2]
                    continue
                if dop == 61:
                    chain.append('load')
                    cid = dops[2]
                    continue
                chain.append('op%d' % dop)
                break
            print('   coopMatStore ptr id', ops[0], '->', ' > '.join(chain))
    # variables with Workgroup storage
    for op, ops, wi, wc in m.instructions:
        if op == 59 and ops[2] == 4:
            print('   Workgroup var id', ops[1], 'type', ops[0])

trace('@NR_WORKSPACE@/candidates/amd-netorg-full-scope-workbuddy-20261017/c2-layout/layout0-unify-l2-v4/spv/c2_S19_fused_c256_body_ds.spv', 'parent')
