def opcode_hist(m):
    h = {}
    for op, ops, i, n in m.instructions:
        h[op] = h.get(op, 0) + 1
    return h


def normalize(m):
    """(opchain, tokenstream, const_values): result ids -> 'id', constant ids ->
    ('c', value), other literals kept.  Renumbering cannot masquerade as a change."""
    const = {}
    for op, ops, _, _ in m.instructions:
        if op == 43:
            const[ops[1]] = ops[2]
        elif op == 44:
            const[ops[1]] = tuple(const.get(x, ('raw', x)) for x in ops[2:])
    results = set()
    for op, ops, _, _ in m.instructions:
        rid = K.ST.result_id_of(op, ops)
        if rid is not None:
            results.add(rid)
    chain, toks = [], []
    for op, ops, _, _ in m.instructions:
        chain.append(op)
        if op == 43:                                   # OpConstant type result value
            toks.append((op, ('const', None, ops[2])))
            continue
        seq = []
        for o in ops:
            if o in const and not isinstance(const[o], tuple):
                seq.append(('c', const[o]))
            elif o in results:
                seq.append('id')
            else:
                seq.append(('lit', o))
        toks.append((op, tuple(seq)))
    return chain, toks, const


def spirv_diff(a_path, b_path, label):
    a, b = K.ST.Module(Path(a_path)), K.ST.Module(Path(b_path))
    ca, ta, consta = normalize(a)
    cb, tb, constb = normalize(b)
    assert ca == cb, 'opcode sequence differs (%s)' % label
    assert len(ca) == len(cb)
    diffs = []
    for idx, ((op1, t1), (op2, t2)) in enumerate(zip(ta, tb)):
        if t1 != t2:
            diffs.append({'index': idx, 'opcode': op1, 'parent': list(map(str, t1)),
                          'candidate': list(map(str, t2))})
    return {
        'label': label,
        'instructions_parent': len(a.instructions),
        'instructions_candidate': len(b.instructions),
        'words_parent': len(a.words),
        'words_candidate': len(b.words),
        'opcode_sequence_identical': True,
        'differing_instructions': len(diffs),
        'differences': diffs,
        'parent_constant_values': sorted(repr(v) for v in consta.values()),
        'candidate_constant_values': sorted(repr(v) for v in constb.values()),
        'constant_multiset_equal': sorted(repr(v) for v in consta.values()) ==
                                   sorted(repr(v) for v in constb.values()),
    }


def analyze_vs_parent(pm, cm):
    out = {}
    nc_p, nc_c = K.ST.no_contraction_ids(pm), K.ST.no_contraction_ids(cm)
    out['no_contraction_parent'] = len(nc_p)
    out['no_contraction_candidate'] = len(nc_c)
    out['no_contraction_ids_equal'] = sorted(nc_p) == sorted(nc_c)
    hp, hc = opcode_hist(pm), opcode_hist(cm)
    out['opcode_histogram_delta'] = {str(k): {'parent': hp.get(k, 0), 'candidate': hc.get(k, 0)}
                                     for k in sorted(set(hp) | set(hc)) if hp.get(k, 0) != hc.get(k, 0)}
    fp = [op for op, *_ in pm.instructions if op in FLOAT_ARITH]
    fc = [op for op, *_ in cm.instructions if op in FLOAT_ARITH]
    out['float_arith_sites_parent'] = len(fp)
    out['float_arith_sites_candidate'] = len(fc)
    out['float_arith_opcode_sequence_equal'] = fp == fc
    ip = [op for op, *_ in pm.instructions if op in INT_CFG]
    ic = [op for op, *_ in cm.instructions if op in INT_CFG]
    out['int_cfg_sites_parent'] = len(ip)
    out['int_cfg_sites_candidate'] = len(ic)
    fa_p = [pm.fma_signature(i) for i in pm.ext_inst_fma_ordinals()]
    fa_c = [cm.fma_signature(i) for i in cm.ext_inst_fma_ordinals()]
    out['ext_inst_fma_sites_parent'] = len(fa_p)
    out['ext_inst_fma_sites_candidate'] = len(fa_c)
    out['ext_inst_fma_signatures_equal'] = fa_p == fa_c
    out['float_NC_checked'] = len(fa_p) + min(len(fp), len(fc))
    sites = []
    ipos = [i for i, (op, *_ ) in enumerate(pm.instructions) if op in FLOAT_ARITH or
            (op == 12 and len(pm.instructions[i][1]) >= 5 and pm.instructions[i][1][3] == 50)]
    ipos_c = [i for i, (op, *_ ) in enumerate(cm.instructions) if op in FLOAT_ARITH or
              (op == 12 and len(cm.instructions[i][1]) >= 5 and cm.instructions[i][1][3] == 50)]
    assert len(ipos) == len(ipos_c)
    for a_i, b_i in zip(ipos, ipos_c):
        opa, oa, _, _ = pm.instructions[a_i]
        opb, ob, _, _ = cm.instructions[b_i]
        e = {'parent_index': a_i, 'candidate_index': b_i, 'opcode': opa,
             'same_opcode': opa == opb,
             'no_contraction_parent': oa[1] in nc_p if opa != 12 else oa[1] in nc_p,
             'no_contraction_candidate': ob[1] in nc_c}
        if opa == 12 and len(oa) >= 5 and oa[3] == 50:
            e['ext_inst'] = 'GLSL.std.450 Fma(50)'
            e['parent_signature'] = repr(pm.fma_signature(oa[1]))
            e['candidate_signature'] = repr(cm.fma_signature(ob[1]))
            e['signature_equal'] = e['parent_signature'] == e['candidate_signature']
        sites.append(e)
    out['float_sites'] = sites
    sp = [op for op, *_ in pm.instructions]
    sc = [op for op, *_ in cm.instructions]
    sm = difflib.SequenceMatcher(a=sp, b=sc, autojunk=False)
    changes, n_eq = [], 0
    for tag, i1, i2, j1, j2 in sm.get_opcodes():
        if tag == 'equal':
            n_eq += i2 - i1
            continue
        changes.append({'tag': tag, 'parent_index': i1, 'parent_opcodes': sp[i1:i2],
                        'candidate_index': j1, 'candidate_opcodes': sc[j1:j2]})
    out['opcode_stream'] = {'parent_len': len(sp), 'candidate_len': len(sc),
                            'equal_opcodes': n_eq, 'change_regions': len(changes)}
    out['opcode_change_regions'] = changes[:40]
    out['opcode_change_regions_total'] = len(changes)
    out['dpas_sites_parent'] = hp.get(OPCMULADD, 0)
    out['dpas_sites_candidate'] = hc.get(OPCMULADD, 0)
    out['spirv_words_parent'] = len(pm.words)
    out['spirv_words_candidate'] = len(cm.words)
    return out


# ---------------------------------------------------------------- plan
def build_plan(parent_plan):
    """Replace the 16 F08 PASS lines of `parent_plan`: SPV path + grid 66 2 1 -> 4 33 1."""
    raw = Path(parent_plan).read_bytes()
    bom = raw.startswith(b'\xef\xbb\xbf')
    text = raw.decode('utf-8-sig')
    lines = text.split('\n')
    parent_posix = PARENT_SPV.as_posix()
    cand_posix = (CARD / 'candidate.spv').as_posix()
    changed = []
    for i, line in enumerate(lines):
        cr = '\r' if line.endswith('\r') else ''
        body = line[:-1] if cr else line
        if not body.startswith('PASS ') or parent_posix not in body:
            continue
        parts = body.split(' ')
        assert tuple(parts[-3:]) == PARENT_GRID, parts[-6:]
        assert parts[3] == parent_posix, parts[3]
        parts[-3:] = list(CAND_GRID)
        parts[3] = cand_posix
        lines[i] = ' '.join(parts) + cr
        changed.append({'line': i + 1, 'pass': parts[1], 'name': parts[2],
                        'old_grid': ' '.join(PARENT_GRID), 'new_grid': ' '.join(CAND_GRID)})
    assert len(changed) == 16, len(changed)
    out = CARD / 'plan.txt'
    out.write_bytes((('\ufeff' if bom else '') + '\n'.join(lines)).encode('utf-8'))
    return out, changed, bom


def plan_diff(plan_path, parent_plan):
    a = Path(parent_plan).read_text(encoding='utf-8-sig').split('\n')
    b = Path(plan_path).read_text(encoding='utf-8-sig').split('\n')
    assert len(a) == len(b), (len(a), len(b))
    return [{'line': i + 1, 'parent': a[i], 'candidate': b[i]}
            for i, (x, y) in enumerate(zip(a, b)) if x != y]
