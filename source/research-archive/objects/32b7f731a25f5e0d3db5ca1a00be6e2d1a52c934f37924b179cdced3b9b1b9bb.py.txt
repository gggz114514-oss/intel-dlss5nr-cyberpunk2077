"""Internal-consistency test of the tile->channel mapping.

The pack declares  sWe[m] = max_k |We[m][k]|/127  inside fsc[4+m] = sX*sWe[m], and
sWc[o] = max_f |Wc[o][f]|/127 inside fsc[132+o] = sE*sWc[o].  If the tile->channel
mapping is right, max|code| * sX / 127 must reproduce fsc[4+m] up to the int8
rounding of one code (relative error ~1/127 of the max).
"""
import sys
from pathlib import Path
import numpy as np

sys.path.insert(0, str(Path(__file__).resolve().parent))
import c32lib as L

for tag, plan in [("A", L.D / "c32-minmax/local-b4.txt"),
                  ("B", L.D / "c32-equalized-b4/local-b4.txt")]:
    info = L.block_bindings(plan, ordinal=4)
    we, wc = L.load_w8(info["w8"])
    fsc = L.load_fsc(info["fsc"])
    sX, sE = np.float32(fsc[0]), np.float32(fsc[2])
    amax_e = np.abs(we.astype(np.float32)).max(axis=1)
    amax_c = np.abs(wc.astype(np.float32)).max(axis=1)
    pred_e = amax_e / 127.0 * sX
    pred_c = amax_c / 127.0 * sE
    got_e = fsc[4:132]
    got_c = fsc[132:164]
    def stats(p, g, label):
        ok = g > 0
        r = p[ok] / g[ok]
        print(f"  {label}: ratio pred/declared  min={r.min():.4f} med={np.median(r):.4f} max={r.max():.4f}"
              f"  frac|r-1|<0.02 = {np.mean(np.abs(r-1)<0.02):.3f}")
        print(f"     corr={np.corrcoef(p, g)[0,1]:.5f}")
    print("arm", tag, "spv", info["spv_sha256"][:12], "fsc", fsc.size, "sX", sX, "sE", sE)
    stats(pred_e, got_e, "expand 128 taps")
    stats(pred_c, got_c, "contract 32 taps")
    # what if the expand channel order were 4 interleaved groups of 32?
    for perm_name, perm in [("identity", np.arange(128)),
                            ("g4s_swap", np.arange(128).reshape(4, 32).T.reshape(-1)),
                            ("stride32", np.argsort(np.arange(128).reshape(4, 32).T.reshape(-1)))]:
        p2 = got_e[perm]
        ok = p2 > 0
        r = pred_e[ok] / p2[ok]
        print(f"     perm {perm_name:10s} corr={np.corrcoef(pred_e, p2)[0,1]:.5f} med_ratio={np.median(r):.4f}")
