"""S05 step 4: build the self-contained candidate plan(s) with the shared planlib.

candidate-<frame>.txt      : combo-<frame> with exactly two BUFFER file paths replaced
                             (block9's int8 weight pack and its scale table).  Byte
                             counts are unchanged, so no BIND/POLICY/PUSH line moves.
candidate-<frame>-gate.txt : the same, plus three OUTPUT dumps (block8_out qX, swin_9_M,
                             swin_9_Y) for the GPU CPU/GPU audit.  The c64-stage dump
                             plan uses the identical pattern (declared OUTPUT "raw").

python -X utf8 tools/build_plans.py [--frame 34189]
"""
import json
import sys
from pathlib import Path

sys.path.insert(0, str(Path(__file__).resolve().parent))
import s05_common as C

sys.path.insert(0, str(C.B / 'work/dsh-fullnet-static-20261029/tools'))
import planlib

FRAME = '34189'
if '--frame' in sys.argv:
    FRAME = sys.argv[sys.argv.index('--frame') + 1]
BASE = C.D / ('c64-family-static/combo-%s.txt' % FRAME)
NEW_W = C.CARD / 'PACK/c128_b9_weights.i8.bin'
NEW_S = C.CARD / 'PACK/c128_b9_scales.f32.bin'
OUT_PLAN = C.CARD / ('candidate-%s.txt' % FRAME)
OUT_GATE = C.CARD / ('candidate-%s-gate.txt' % FRAME)

assert BASE.is_file() and NEW_W.is_file() and NEW_S.is_file()
base_sha = planlib.sha256_file(BASE)
if FRAME == '34189':
    assert base_sha == '2b5481d5ce6b8ec8c1f62e68d2057b39d5267825d655e844a2ee2c5500f1c690', base_sha

plan = planlib.Plan(BASE)
base_problems = plan.validate()
assert base_problems == [], base_problems
old_buf = dict(w=plan.buffer('chain_qweights_C'), s=plan.buffer('chain_scales'))
assert old_buf['w'].bytes == NEW_W.stat().st_size == 106496
assert old_buf['s'].bytes == NEW_S.stat().st_size == 4160

edits = []
edits.append(plan.set_buffer('chain_qweights_C', file=NEW_W.as_posix()))
edits.append(plan.set_buffer('chain_scales', file=NEW_S.as_posix()))
problems = plan.validate()
assert problems == [], problems
plan.write(OUT_PLAN)

delta = planlib.diff_plans(planlib.Plan(BASE), planlib.Plan(OUT_PLAN))
report = dict(
    card='S05-c128',
    base=dict(path=str(BASE), sha256=base_sha),
    new=dict(path=str(OUT_PLAN), sha256=planlib.sha256_file(OUT_PLAN)),
    edits=edits,
    problems=problems,
    delta=delta,
)
(C.CARD / 'evidence' / 'PLAN-DELTA.json').write_text(json.dumps(report, indent=2))

# ---- gate plan: add three decl-ared OUTPUT dumps (same pattern as c64-stage/old.txt)
text = OUT_PLAN.read_text(encoding='utf-8').splitlines()
out_lines = [i for i, l in enumerate(text) if l.startswith('OUTPUT')]
first_policy = next(i for i, l in enumerate(text) if l.startswith('POLICY'))
assert out_lines and first_policy == max(out_lines) + 1, (out_lines, first_policy)
ins = ['OUTPUT 48 raw "block9_qx.bin"',
       'OUTPUT 51 raw "block9_M.bin"',
       'OUTPUT 52 raw "block9_Y.bin"']
for k, line in enumerate(ins):
    text.insert(first_policy + k, line)
nb_out = len(out_lines) + len(ins)
head = text[0].split()
assert head[0] == 'PORT_NETWORK_1' and int(head[4]) == 1, head
head[4] = str(nb_out)
text[0] = ' '.join(head)
OUT_GATE.write_text('\n'.join(text) + '\n', encoding='utf-8')

gp = planlib.Plan(OUT_GATE)
gp_problems = gp.validate()
assert gp_problems == [], gp_problems
report['gate'] = dict(path=str(OUT_GATE), sha256=planlib.sha256_file(OUT_GATE), problems=gp_problems,
                     outputs=gp.outputs, header=gp.lines[0])
(C.CARD / 'evidence' / 'PLAN-DELTA.json').write_text(json.dumps(report, indent=2))

print(json.dumps(dict(candidate_sha=report['new']['sha256'], gate_sha=report['gate']['sha256'],
                      delta=delta, edits=edits, gate_outputs=gp.outputs, header=gp.lines[0]), indent=2))
