"""Where else in the frame is residency capped by shared memory?

For every PASS we know (a) its declared shared memory, (b) its measured occupancy, (c) its GPU
time. Grouping by shared memory shows which groups sit below the best occupancy achieved in the
same frame (86.1% at 8 KB), i.e. where the lever measured on the 32 KB group could also apply.
"""
import json, sys
from collections import defaultdict
from pathlib import Path
sys.path.insert(0, str(Path(__file__).parent))
import COMMON as K

rows = K.read_json(K.C / 'evidence' / 'OCCUPANCY-PROBE.json')['rows']
BEST = max(x['occ'] for x in rows)
by_slm = defaultdict(list)
for r in rows:
    by_slm[r['slm']].append(r)
print('best occupancy in the frame: %.1f%%' % BEST)
print('\n%-8s %3s %9s %9s %9s %9s %s' % ('slm', 'n', 'occ%', 'gap%', 'gpu_ms', 'aluwr%', 'families'))
cand = []
for slm in sorted(by_slm):
    sel = by_slm[slm]
    n = len(sel)
    occ = sum(x['occ'] for x in sel) / n
    gpu = sum(x['gpu'] for x in sel)
    fam = defaultdict(int)
    for x in sel:
        fam[x['family']] += 1
    gap = BEST - occ
    print('%-8d %3d %9.1f %9.1f %9.3f %9.1f %s' % (
        slm, n, occ, gap, gpu, sum(x['aluwr'] for x in sel) / n,
        ', '.join('%s×%d' % (f, c) for f, c in sorted(fam.items()))))
    if gap > 5 and gpu > 0.05:
        cand.append(dict(slm=slm, n=n, occ=occ, gap=gap, gpu_ms=gpu,
                         families={f: c for f, c in fam.items()},
                         score=gap * gpu / 100.0))
print('\nranked by (occupancy gap x gpu time) — the heuristic for "how much residency is on the table":')
print('  %-8s %3s %8s %8s %10s  %s' % ('slm', 'n', 'gap%', 'gpu_ms', 'score', 'families'))
for c in sorted(cand, key=lambda z: -z['score']):
    print('  %-8d %3d %8.1f %8.3f %10.4f  %s' % (c['slm'], c['n'], c['gap'], c['gpu_ms'], c['score'],
                                                 ', '.join('%s×%d' % (f, n) for f, n in c['families'].items())))
K.write_json(K.C / 'evidence' / 'OCCUPANCY-CANDIDATES.json', dict(best_occ=BEST, candidates=cand))
