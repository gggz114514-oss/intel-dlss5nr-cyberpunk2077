param([string]$ManifestPath)
$ErrorActionPreference='Stop';$task=Get-Content $ManifestPath -Raw|ConvertFrom-Json;$root=$task.output_root;$base=Split-Path $ManifestPath -Parent;$template=Join-Path (Split-Path $base -Parent) 'execute-nativeonly-pilot.ps1';$master=Join-Path (Split-Path $base -Parent) 'LUNA-NATIVE-WINDOWS-JOBS.json';$masterHash=(Get-FileHash $master -Algorithm SHA256).Hash.ToLowerInvariant();$results=[Collections.ArrayList]::new();$statusPath=Join-Path $root 'LUNA-NATIVE-WINDOWS-STATUS.json';$resultsPath=Join-Path $root 'LUNA-NATIVE-WINDOWS-RESULTS.json';$stop=$null
function SaveJson($p,$v){[IO.File]::WriteAllText($p,($v|ConvertTo-Json -Depth 40),[Text.UTF8Encoding]::new($false))}
function SaveProgress($state,$next){$done=@($results|Where-Object {$_.status -eq 'COMPLETED'}).Count;$skip=@($results|Where-Object {$_.status -eq 'SKIPPED_GATE'}).Count;SaveJson $resultsPath ([ordered]@{status=$state;jobs=@($results);executed_jobs=$done;skipped_jobs=$skip;stop_reason=$stop;source_manifest=$master;source_manifest_sha256=$masterHash;authorization='Parent authorized nativeonly captures for gate-passed arms only; not performance evidence.'});SaveJson $statusPath ([ordered]@{status=$state;updated_at=[DateTime]::UtcNow.ToString('o');planned_jobs=@($task.jobs).Count;executed_jobs=$done;skipped_jobs=$skip;next_job=$next;stop_reason=$stop;lock_state='released between jobs';results_path=$resultsPath})}
try{
 $gates=@{
 'direct-c32-stage-ffn-native'=@{file=(Join-Path (Split-Path $base -Parent) 'LUNA-C32-STAGE-RESULTS.json');pattern='-stage-ffn$'}
 'direct-clock-fold-work-native'=@{file=(Join-Path (Split-Path $base -Parent) 'LUNA-F20-HANDOFF-EXECUTION\LUNA-F20-HANDOFF-RESULTS.json');pattern='-fold-work$'}
 'direct-clock-scale-work-native'=@{file=(Join-Path (Split-Path $base -Parent) 'LUNA-F20-HANDOFF-EXECUTION\LUNA-F20-HANDOFF-RESULTS.json');pattern='-scale-work$'}
 'direct-clock-scale-handoff-native'=@{file=(Join-Path (Split-Path $base -Parent) 'LUNA-F20-HANDOFF-EXECUTION\LUNA-F20-HANDOFF-RESULTS.json');pattern='-scale-handoff$'}
 'direct-f01-whole-native'=@{file=(Join-Path (Split-Path $base -Parent) 'LUNA-F01-CLOSURE-EXECUTION\LUNA-F01-CLOSURE-RESULTS.json');pattern='-whole$'}
 }
 foreach($i in $task.identities){if((Get-FileHash $i.path -Algorithm SHA256).Hash.ToLowerInvariant() -ne $i.sha256){throw ('Identity mismatch '+$i.path)}}
 $nativeScriptHash=(Get-FileHash $template -Algorithm SHA256).Hash.ToLowerInvariant();$planBad=@($task.jobs|Where-Object {(Get-FileHash $_.plan.path -Algorithm SHA256).Hash.ToLowerInvariant() -ne $_.plan.sha256});if($planBad.Count){throw 'Native plan hash mismatch'}
 SaveProgress 'RUNNING' $task.jobs[0].id
 foreach($job in $task.jobs){if($stop){[void]$results.Add([ordered]@{id=$job.id;status='SKIPPED_AFTER_STOP';reason=$stop;actual_argv=$job.argv});SaveProgress 'STOPPED' $null;continue}
  $g=$gates[$job.id];$gr=Get-Content $g.file -Raw|ConvertFrom-Json;$matched=@($gr.jobs|Where-Object {$_.id -match $g.pattern});$gateOk=($matched.Count -ge 3 -and @($matched|Where-Object {$_.status -ne 'COMPLETED' -or $_.exit_code -ne 0 -or -not $_.frame_matches_gold}).Count -eq 0)
  if(-not $gateOk){[void]$results.Add([ordered]@{id=$job.id;status='SKIPPED_GATE';gate_result_path=$g.file;gate_pattern=$g.pattern;gate_jobs=$matched;actual_argv=$job.argv});SaveProgress 'RUNNING' 'next eligible job';continue}
  $sub=Join-Path $root $job.id;if(Test-Path $sub){throw ('Sub-output exists '+$sub)};New-Item -ItemType Directory -Path $sub|Out-Null;$one=$task|ConvertTo-Json -Depth 50|ConvertFrom-Json;$one.output_root=$sub;$one.jobs=@($job);$mf=Join-Path $sub ($job.id+'.json');SaveJson $mf $one
  if(@(Get-Process|Where-Object {$_.ProcessName -match 'Cyberpunk|REDlauncher|port_network|profile_zprof|gtpin|clock_runner|clock_query'}).Count){throw ('Active process before '+$job.id)};$l=Get-Content '@NR_GPU_WORKSPACE@\artifacts\parallel-port-20261013\GPU.lock' -Raw|ConvertFrom-Json;if($l.state -ne 'FINISHED'){throw ('GPU lock unavailable before '+$job.id)}
  $start=[DateTime]::UtcNow.ToString('o');$child=Start-Process -FilePath 'powershell.exe' -ArgumentList @('-NoProfile','-ExecutionPolicy','Bypass','-File',$template,'-ManifestPath',$mf) -Wait -PassThru -NoNewWindow;$end=[DateTime]::UtcNow.ToString('o');$oneResult=Join-Path $sub 'LUNA-NATIVEONLY-PILOT-RESULTS.json';$oneStatus=Join-Path $sub 'LUNA-NATIVEONLY-PILOT-STATUS.json';if(-not (Test-Path $oneResult)){throw ('No nativeonly result '+$job.id)};$rr=Get-Content $oneResult -Raw|ConvertFrom-Json;$entry=$rr.jobs[0];$gateOk=($child.ExitCode -eq 0 -and $rr.status -eq 'COMPLETED' -and $entry.exit_code -eq 0 -and $entry.normal_close -eq $true -and $entry.frame_matches_gold -eq $true);$entry|Add-Member -NotePropertyName orchestration_started_utc -NotePropertyValue $start -Force;$entry|Add-Member -NotePropertyName orchestration_ended_utc -NotePropertyValue $end -Force;$entry|Add-Member -NotePropertyName native_runner_script_sha256 -NotePropertyValue $nativeScriptHash -Force;$entry|Add-Member -NotePropertyName gate_source_results -NotePropertyValue $g.file -Force;$entry|Add-Member -NotePropertyName gate_source_pattern -NotePropertyValue $g.pattern -Force;$entry|Add-Member -NotePropertyName executor_result_path -NotePropertyValue $oneResult -Force;$entry|Add-Member -NotePropertyName executor_status_path -NotePropertyValue $oneStatus -Force;$entry|Add-Member -NotePropertyName status -NotePropertyValue $(if($gateOk){'COMPLETED'}else{'FAILED'}) -Force;$entry|Add-Member -NotePropertyName orchestration_status -NotePropertyValue $(if($gateOk){'COMPLETED'}else{'FAILED'}) -Force;[void]$results.Add($entry)
  if(-not $gateOk){$stop='nativeonly exit/receipt/frame gate failed for '+$job.id;SaveProgress 'STOPPED' 'none';break};SaveProgress 'RUNNING' 'next listed job'
 }
 SaveProgress $(if($stop){'STOPPED'}else{'COMPLETED'}) $null
}catch{$stop=$_.Exception.Message;SaveProgress 'FAILED_OR_STOPPED' $null;throw}
