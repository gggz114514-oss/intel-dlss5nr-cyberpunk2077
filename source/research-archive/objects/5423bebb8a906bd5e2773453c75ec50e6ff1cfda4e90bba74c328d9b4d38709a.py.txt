"""n36-f05-arena-shrink: close the card — RESULTS.json + the residency threshold table.

import hashlib
import json
import pathlib

CARD = pathlib.Path(__file__).resolve().parent


def sha(p):
    return hashlib.sha256(pathlib.Path(p).read_bytes()).hexdigest()


def read(name):
    return json.loads((CARD / name).read_text(encoding='utf-8'))


def threshold_table():
    """Residency thresholds implied by the two measured reverse probes.

    F05 probe: 24576 B -> 2 residents, 49152 B -> 1 resident
    F06 probe: 32768 B -> 2 residents, 49152 B -> 1 resident
    => 65536 <= P < 73728 (SLM bytes per SM), and residents(W) = floor(P / W).

    The boundary W for exactly k residents is floor(P/W) == k, i.e. the largest W that
    still fits k: W_k = floor(P_lo / k).  A table entry is *exact* when floor(P_lo/W)
    equals floor(P_hi/W), i.e. the count does not depend on where P sits in the interval.
    """
    lo, hi = 65536, 73727
    rows = []
    for k in (1, 2, 3, 4, 5, 6, 8):
        w = lo // k
        rows.append(dict(residents=k, boundary_workgroup_bytes=w,
                         at_lo=lo // w, at_hi=hi // w,
                         exact=(lo // w) == (hi // w),
                         bytes_per_head_pair=w,
                         note='largest workgroup shared bytes that still fits %d per SM' % k))
    # the named thresholds the campaign will quote
    named = []
    for w in (32768, 24576, 21845, 20480, 16384, 13107, 12288, 8192):
        named.append(dict(workgroup_bytes=w, residents=(lo // w, hi // w),
                          exact=(lo // w) == (hi // w)))
    return dict(pool_bytes_interval=[lo, hi + 1], basis=[
        'F05 reverse probe (evidence/OCCUPANCY-ELASTICITY-F05.json): 24576 B -> 2, 49152 B -> 1',
        'F06 reverse probe (evidence/OCCUPANCY-ELASTICITY.json): 32768 B -> 2, 49152 B -> 1'],
        residents_formula='residents(W) = floor(P / W), P in [65536, 73727]',
        boundaries=rows, named=named)
