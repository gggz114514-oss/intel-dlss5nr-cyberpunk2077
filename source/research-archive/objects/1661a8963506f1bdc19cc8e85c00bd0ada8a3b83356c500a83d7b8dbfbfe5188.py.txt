"""Classify the live `add` corpus of the three F05 kernels (address vs data).

- add with an a0 destination operand  -> address formation for the (W) register-indirect
  moves that implement the cross-lane exchange
- add with :hf -> f16 data add (norm tree / denominator tree)
- add with :f  -> f32 data add
Also reports the per-kernel size of the "norm section" (first rsqt .. last rsqt) and
the instruction count between consecutive rsqt.
"""
import sys, re
from collections import Counter
from pathlib import Path
HERE = Path(__file__).parent
sys.path.insert(0, str(HERE))
import _regions as R

for w in (64, 128, 256):
    ins, labels = R.load(w)
    ins, guard = R.classify(w, ins, labels)
    live = [i for i in ins if i["live"]]
    c = Counter()
    for it in live:
        if it["op"] != "add":
            continue
        if "a0" in it["args"]:
            c["add -> a0 (address)"] += 1
        elif ":hf" in it["args"]:
            c["add :hf (f16 data)"] += 1
        elif ":f" in it["args"]:
            c["add :f (f32 data)"] += 1
        else:
            c["add other"] += 1
    rs = [i for i in live if "rect" not in i["op"] and i["op"] == "math.rsqt"]
    addrs = [i["addr"] for i in rs]
    gaps = [(addrs[k + 1] - addrs[k]) // 16 for k in range(len(addrs) - 1)]
    print("=" * 70)
    print("c%d live %d  add %d" % (w, len(live), sum(v for k, v in c.items() if k.startswith('add'))))
    for k, v in c.most_common():
        print("   %-22s %5d" % (k, v))
    print("   rsqt %d  first %X last %X" % (len(addrs), addrs[0], addrs[-1]))
    print("   rsqt-to-rsqt distance in instructions:", gaps)
    print("   norm section span: %d instructions (first rsqt -> last rsqt)" %
          (len([i for i in live if addrs[0] <= i["addr"] <= addrs[-1]])))
