"""Assemble BUILD.json for n29-f01-alu-chain (CPU only)."""
import json
import subprocess
import sys
from pathlib import Path

CARD = Path(__file__).resolve().parents[1]
CAMP = CARD.parents[1]
sys.path.insert(0, str(CAMP))
import COMMON as K  # noqa: E402

BASE_PLAN = CAMP / 'accepted' / 'n27-f07-20261027' / 'plan.txt'
PARENT = CARD.parent / 'n18-f01-entry' / 'candidate.spv'
ISA = CAMP / 'queue/n27-f07/native/captures/native-native-base/ISA/' \
            'CS_asm54249010bf7e593b_simd32_0000000000000001_0_orig.asm'
PUSH_P4 = [1344, 768, 0, 10137620]
PUSH_P5 = [1280, 720, 0, 4]
PUSH_P6 = [3686400, 3686420, 7372820, 9216020]


def sha(p):
    return K.sha(Path(p))


def cpu_check():
    out = subprocess.run([str(K.RUNNER), '--plan', str(CARD / 'plan.txt'), '--cpu-check'],
                         capture_output=True, text=True)
    return dict(argv=[str(K.RUNNER), '--plan', str(CARD / 'plan.txt'), '--cpu-check'],
                returncode=out.returncode,
                stdout=out.stdout.strip(), stderr=out.stderr.strip())


def main():
    parent_build = K.read_json(CARD / 'parent-rebuild.json')
    cand_build = K.read_json(CARD / 'candidate-build.json')
    delta = K.read_json(CARD / 'tools' / 'spv-delta.json')
    base_plan_text = BASE_PLAN.read_text(encoding='utf-8-sig').splitlines()
    card_plan_text = (CARD / 'plan.txt').read_text(encoding='utf-8-sig').splitlines()
    diff = [(i, a, b) for i, (a, b) in enumerate(zip(base_plan_text, card_plan_text)) if a != b]
    res = dict(
        card='n29-f01-alu-chain',
        ordinal=0, pass_name='f1c32_image_pre',
        baseline_plan=dict(path=str(BASE_PLAN), sha256=sha(BASE_PLAN),
                           note='accepted chain head the parent SPV lives in; byte-identical '
                                'row for PASS 0 to queue/n27-f07/plan-rebased.txt'),
        card_plan=dict(path=str(CARD / 'plan.txt'), sha256=sha(CARD / 'plan.txt'),
                       lines=len(card_plan_text), passes=232, buffers=502,
                       replaced_rows=1, diff=dict(line_numbers=[i + 1 for i, _, _ in diff],
                                                   base=[a for _, a, _ in diff],
                                                   card=[b for _, _, b in diff])),
        parent=dict(spv=dict(path=str(PARENT), sha256=sha(PARENT)),
                    source=parent_build['source'],
                    rebuild_raw=parent_build['raw'], rebuild=parent_build['decorated'],
                    raw_match=parent_build['raw_match'],
                    decorated_match=parent_build['decorated_match'],
                    decor=parent_build['decor'],
                    note='raw compile differs from the shipped parent (NoContraction '
                         'placement); after the recorded F01 nc_recipe the rebuild is '
                         'byte-identical to the shipped parent, which is what the card '
                         'requires (decorated_match=true)'),
        candidate=dict(source=dict(path=str(CARD / 'candidate.comp'), sha256=sha(CARD / 'candidate.comp')),
                       raw=dict(path=str(CARD / 'candidate-raw.spv'), sha256=sha(CARD / 'candidate-raw.spv')),
                       spv=dict(path=str(CARD / 'candidate.spv'), sha256=sha(CARD / 'candidate.spv')),
                       build_json=str(CARD / 'candidate-build.json'),
                       decor=cand_build['decor']),
        float_NC_checked=dict(parent=delta['nc_parent'], candidate=delta['nc_candidate'],
                              rule='F01 nc_recipe: transfer the parent decoration pattern, '
                                   'then set the 189-id NoContraction set',
                              equal=delta['nc_parent'] == delta['nc_candidate']),
        spv_delta=dict(parent_instructions=delta['parent']['instructions'],
                       candidate_instructions=delta['candidate']['instructions'],
                       changed_opcodes=delta['changed_opcodes'],
                       int_cf_set=delta['int_cf_set'],
                       note='the delta is function-level: nrf_hist_linear is a real SPIR-V '
                            'function with 5 calls, so its -1 body instruction costs 5 after '
                            'inlining; no float opcode changed'),
        corpus_identity=dict(
            kernel_hash='54249010bf7e593b',
            basis='queue/n27-f07/native/captures/native-native-base (base arm of the n27 '
                  'native run; its plan PASS 0 row is the same parent SPV, same grid)',
            isa=dict(path=str(ISA), sha256=sha(ISA), bytes=ISA.stat().st_size),
            push_constants=dict(p4=PUSH_P4, p5=PUSH_P5, p6=PUSH_P6,
                                source='plan.txt PUSH line of PASS 0'),
            grid=[168, 96], local_size_x=128, lds_bytes=9216),
        cpu_check=cpu_check(),
        interpreter=dict(build=sys.executable,
                         proofs='@NR_USER_HOME@/AppData/Local/Programs/Python/Python310/python.exe '
                                '(only interpreter on this host with numpy; CONTRACT python312 '
                                'has no numpy, so the exhaustive proofs run under 310)'),
    )
    K.write_json(CARD / 'BUILD.json', res)
    print(json.dumps({k: res[k] for k in ('baseline_plan', 'card_plan', 'cpu_check')}, indent=1))
    print('parent', res['parent']['spv']['sha256'], 'decorated_match',
          res['parent']['decorated_match'])
    print('candidate', res['candidate']['spv']['sha256'])
    print('isa', res['corpus_identity']['isa']['sha256'], res['corpus_identity']['isa']['bytes'])


if __name__ == '__main__':
    main()
