"""Identify the F02 body kernels inside a 40-pipeline native capture, then census
their send-message composition.

CPU only, read-only on the capture.
"""
import sys, json, collections
from pathlib import Path
sys.path.insert(0, str(Path(__file__).parent))
import isaparse as P


def census(path):
    isa = P.parse(path)
    c = collections.Counter(i['op'] for i in isa)
    sends = [i for i in isa if i['op'].startswith('send')]
    forms = collections.Counter((i['op'], P.rd_of(i), P.msg_of(i)) for i in sends)
    return dict(path=str(path), instr=len(isa), dpas=c.get('dpas', 0),
                sends=len(sends), send_slm=sum(1 for i in sends if i['op'] == 'send.slm'),
                send_ugm=sum(1 for i in sends if i['op'] == 'send.ugm'),
                bar=c.get('sync.bar', 0), mov=c.get('mov', 0), forms=forms)


if __name__ == '__main__':
    d = Path(sys.argv[1])
    rows = []
    for f in sorted(d.glob('*.asm')):
        r = census(f)
        rows.append(r)
    # F02 body signature: dpas 128
    for r in rows:
        if r['dpas'] >= 100:
            print('%-58s instr=%5d dpas=%3d sends=%4d ugm=%4d slm=%3d bar=%2d mov=%4d'
                  % (Path(r['path']).name, r['instr'], r['dpas'], r['sends'],
                     r['send_ugm'], r['send_slm'], r['bar'], r['mov']))
    print()
    print('--- full table (all 40) instr sorted')
    for r in sorted(rows, key=lambda x: -x['instr'])[:12]:
        print('%-58s instr=%5d dpas=%3d' % (Path(r['path']).name, r['instr'], r['dpas']))
