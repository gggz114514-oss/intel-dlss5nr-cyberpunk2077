"""Task 1b: cross-channel correlation matrix GPU x teacher, to separate
   (a) layout/permutation error  from  (b) genuine amplitude compression.
"""
import json
import numpy as np
from pathlib import Path

B = Path("@NR_WORKSPACE@")
W = B / "work/dsh-fullnet-static-20261029"
CHAIN = W / "C00-color-baseline/runs/diag-chain62-66"
TEACH = (B / "work/codex-astra-fp16-int8-audit-20261024/int8-quality-first-20261029"
         / "real-int8-24/dump-fp16")


def load(p):
    return np.fromfile(p, dtype="<f2").astype(np.float32)


def xcorr(g, t, C):
    G = g.reshape(-1, C).astype(np.float64)
    T = t.reshape(-1, C).astype(np.float64)
    M = G.T @ T                      # C x C
    gn = np.sqrt((G * G).sum(0))
    tn = np.sqrt((T * T).sum(0))
    R = M / np.maximum(np.outer(gn, tn), 1e-30)
    return R


def best_perm(R):
    """greedy assignment gpu-channel -> teacher-channel by |corr|"""
    C = R.shape[0]
    A = np.abs(R).copy()
    perm = []
    used = set()
    for _ in range(C):
        i, j = np.unravel_index(np.argmax(A), A.shape)
        perm.append((int(i), int(j), float(R[i, j])))
        A[i, :] = -1
        A[:, j] = -1
        used.add(j)
    return perm


def main():
    out = {}
    for blk, C in ((62, 64), (63, 64), (64, 64), (65, 64), (66, 32)):
        g = load(CHAIN / f"chain-block{blk}_out.bin")
        t = load(TEACH / f"dump_block{blk}_out.bin")
        R = xcorr(g, t, C)
        perm = best_perm(R)
        diag = np.diag(R)
        offmax = np.max(np.abs(R - np.diag(diag)), axis=1)
        out[blk] = {
            "C": C,
            "diag_corr": [round(float(x), 4) for x in diag],
            "diag_corr_mean": float(diag.mean()),
            "offdiag_absmax_per_ch": [round(float(x), 4) for x in offmax],
            "offdiag_absmax_global": float(offmax.max()),
            "best_perm": perm,
            "perm_is_identity": all(i == j for i, j, _ in perm),
            "perm_corr_mean": float(np.mean([c for _, _, c in perm])),
            "R16x16": np.round(R[:16, :16], 3).tolist() if C >= 16 else None,
        }
    print(json.dumps(out, indent=1))


if __name__ == "__main__":
    main()
