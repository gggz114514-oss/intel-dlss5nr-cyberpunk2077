"""final spot checks on the delivered index."""
import json
import os
from pathlib import Path

CARD = Path(__file__).parent
j = json.loads((CARD / 'evidence' / 'NATIVE-INDEX.json').read_text(encoding='utf-8'))

cov = j['coverage']
print('passes:', len(cov), 'CURRENT:', sum(1 for c in cov if c['status'] == 'CURRENT'),
      'HISTORICAL-ONLY:', sum(1 for c in cov if c['status'] == 'HISTORICAL-ONLY'),
      'MISSING:', sum(1 for c in cov if c['status'] == 'MISSING'))

sampled = [r for c in cov for r in c['current']]
missing_files = [r['isa'] for r in sampled if not os.path.exists(r['isa'])]
print('sampled current refs:', len(sampled), 'missing on disk:', len(missing_files))
print('PASS 0 sample:', cov[0]['current'][0]['isa'])
print('PASS 0 current_count:', cov[0]['current_count'],
      'baseline hash:', cov[0]['baseline_kernel_hash'])

# every record's isa must exist
allrec = [e for c in j['captures'] for e in c['records']]
gone = [e['isa'] for e in allrec if not os.path.exists(e['isa'])]
print('records:', len(allrec), 'records whose file vanished since the scan:', len(gone))

# sha of a sample re-verified
import hashlib
bad = 0
for e in allrec[:200]:
    h = hashlib.sha256()
    with open(e['isa'], 'rb') as f:
        for b in iter(lambda: f.read(1 << 20), b''):
            h.update(b)
    if h.hexdigest() != e['sha256']:
        bad += 1
print('re-hashed 200 records, sha mismatches:', bad)

print('captures:', len(j['captures']),
      'full-current:', j['summary']['captures_full_current'],
      'summary passes_current:', j['summary']['passes_current'])
print('summary keys:', sorted(j['summary']))
