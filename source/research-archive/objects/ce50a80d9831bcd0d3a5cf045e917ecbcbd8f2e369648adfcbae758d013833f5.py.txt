"""n33 task B7: plan.txt (7 PASS of F02) + CPU plan check + BUILD.json.

The F02 group is 7 PASS rows that all dispatch the same SPV (4c034a532ce6...).
`K.make_plan` replaces exactly those rows on the chain head and asserts the
replacement count equals the group size; `K.cpu_check` runs clock_runner
--cpu-check (plan / bindings / pool upload / grid / buffer lifetime, CPU only).
"""
import sys, json
from pathlib import Path

HERE = Path(__file__).parent
CAMP = HERE.parent.parent
sys.path.insert(0, str(CAMP))
import COMMON as K  # noqa: E402

PARENT_SHA = "4c034a532ce60f5fa608d1ce67f6368819c052c3d5268eb51f7f680ea83aaf10"
PLAN = HERE / "plan.txt"
OUT = HERE / "BUILD.json"


def main():
    base, base_sha, _chain = K.current_best()
    n = K.make_plan(HERE / "candidate.spv", PLAN, parent_plan=base, spv_shas={PARENT_SHA})
    print("chain head   : %s (sha %s)" % (base, base_sha[:16]))
    print("PASS replaced: %d (F02 group)" % n)
    assert n == 7, n
    print("plan sha     : %s" % K.sha(PLAN)[:16])
    cpu = K.cpu_check(PLAN)
    print("cpu_check:\n%s" % cpu.rstrip())

    build = json.loads((HERE / "build/build.json").read_text()) if (HERE / "build/build.json").exists() else {}
    nc = json.loads((HERE / "build/nc-final.json").read_text())
    parent_src = Path("@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/"
                      "c32-rest-20261026/body-pv-den-overlap/candidate.comp")

    rec = dict(
        card="n33-f02-latency",
        status="BLOCKED",
        family="F02",
        parent=dict(plan=str(base), plan_sha256=base_sha),
        baseline_plan=dict(path=str(K.BASELINE), sha256=K.BASELINE_SHA),
        shader_group=dict(kernel_hash="7a38a17947f21ae6", ordinals=[1, 2, 3, 4, 227, 228, 229],
                          spv=str(Path(nc["parent_spv"])), spv_sha256=nc["parent_spv_sha"]),
        parent_spv=dict(path=str(nc["parent_spv"]), sha256=nc["parent_spv_sha"],
                        bytes=nc["parent_bytes"], source=str(parent_src),
                        source_sha256=K.sha(parent_src)),
        parent_rebuild=dict(path=str(HERE / "build/parent-rebuild.spv"),
                            sha256=K.sha(HERE / "build/parent-rebuild.spv"),
                            byte_match=K.sha(HERE / "build/parent-rebuild.spv") == PARENT_SHA,
                            nc_keep=nc["parent_nc"], nc_parent=nc["parent_nc"],
                            recipe="evidence/RECIPES.json 4c034a532ce6 (glslang 16.5.0 + kv-tile NoContraction pattern)"),
        candidate=dict(source=str(HERE / "candidate.comp"), source_sha256=K.sha(HERE / "candidate.comp"),
                       spv=str(HERE / "candidate.spv"), spv_sha256=K.sha(HERE / "candidate.spv"),
                       spv_bytes=(HERE / "candidate.spv").stat().st_size,
                       spv_validated=True,
                       nc=nc["candidate_nc"], nc_own=nc["candidate_raw_nc"],
                       nc_transferred_exp_helper=2,
                       identical_skeleton_functions=nc["identical_skeleton_functions"],
                       note=("NC 185 = 183 own `precise` sites (identical opcode histogram to the parent's "
                             "raw compile) + 2 exp-helper FrexpStruct sites; only `main` changed, all 9 "
                             "helper functions have an identical instruction skeleton")),
        plan=dict(path=str(PLAN), sha256=K.sha(PLAN), passes_replaced=n),
        cpu_check=cpu,
        float_NC_checked=nc["candidate_nc"],
        float_NC_parent=nc["parent_nc"],
        dispatch_authorization="NONE - card is BLOCKED; do not dispatch plan.txt",
    )
    OUT.write_text(json.dumps(rec, indent=1), encoding="utf-8")
    print("wrote", OUT)

if __name__ == "__main__":
