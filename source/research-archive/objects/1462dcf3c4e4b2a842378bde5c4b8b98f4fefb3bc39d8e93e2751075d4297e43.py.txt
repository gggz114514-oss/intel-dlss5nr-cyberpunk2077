"""For each static sync.bar in the F20 ISA capture, print the surrounding window so the
barrier can be tied to a source stage and to an execution path (normal vs debug).

Read-only.
"""
import re, sys
from pathlib import Path

sys.stdout.reconfigure(encoding='utf-8')

ISA = Path(r'@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024'
           r'/fullnet-mechanism-transfer-20261026/f10-weight-return-separation-20261027'
           r'/captures/native-f20-pixel-owner/ISA'
           r'/CS_asm76902ef344609c8a_simd32_0000000000000027_38_orig.asm')

lines = ISA.read_text(encoding='utf-8', errors='replace').splitlines()
bar_lines = [i + 1 for i, l in enumerate(lines) if re.search(r'\bsync\.bar\b', l)]
print('bar lines:', bar_lines)
W = 14
for k, bl in enumerate(bar_lines):
    print('\n########## bar %d at line %d' % (k + 1, bl))
    for i in range(max(1, bl - W), min(len(lines), bl + W) + 1):
        mark = '>>' if i == bl else '  '
        print('%s %4d %s' % (mark, i, lines[i - 1].rstrip()))
