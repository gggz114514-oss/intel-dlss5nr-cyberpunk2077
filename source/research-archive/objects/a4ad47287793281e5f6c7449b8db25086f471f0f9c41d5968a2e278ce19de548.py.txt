import sys, json, re
from pathlib import Path
CAMP = Path('@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/fullnet-continuous-20261027')
sys.path.insert(0, str(CAMP))
sys.path.insert(0, r'@NR_WORKSPACE@/build/zcode-fp16-compute-backend-20261021/scripts')
import COMMON as K
import spv_tools_local as ST

plan = K.BASELINE.read_text(encoding='utf-8-sig').splitlines()
by_name = {}
for i, l in enumerate(plan):
    if not l.startswith('PASS '):
        continue
    m = re.match(r'PASS\s+(\d+)\s+(?:"([^"]+)"|(\S+))\s+(?:"([^"]+)"|(\S+))\s+(.*)$', l)
    if not m:
        print('NOPARSE', l[:80]); continue
    name = m.group(2) or m.group(3)
    spv = m.group(4) or m.group(5)
    nums = m.group(6).split()
    by_name[name] = dict(line=i + 1, spv=spv, nums=nums, shared=int(nums[1]), grid=nums[-3:])

print('%-22s %-10s %-10s %-8s %s' % ('name', 'plan_sh', 'spv_wg', 'grid', 'arrays'))
for n in (59, 35, 17, 166, 167, 168, 192, 193, 194, 212, 213, 214, 133, 5):
    r = K.row_by_ordinal(n)
    mm = ST.Module(Path(r['spv']))
    sb = mm.storage_bytes()
    wg = sb.get('Workgroup')
    arrs = []
    for op, ops, _, _ in mm.instructions:
        if op == 59 and len(ops) >= 3 and ops[2] == 3:
            arrs.append(ops[1])
    p = by_name.get(r['name'])
    print('%-22s %-10s %-10s %-8s %s' % (r['name'], p and p['shared'], wg and wg['bytes'], p and 'x'.join(p['grid']), arrs))

json.dump({k: dict(line=v['line'], shared=v['shared'], grid=v['grid'], nums=v['nums']) for k, v in by_name.items()},
          open(CAMP / 'cards/n07n14-arena/tools/plan-passes.json', 'w'), indent=1)
