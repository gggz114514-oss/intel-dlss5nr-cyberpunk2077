from pathlib import Path
import sys,json,hashlib
import numpy as np
Q=Path(__file__).resolve().parent.parent
sys.path.insert(0,str(Q))
from make_capture_replay import build
root=Path(sys.argv[1]);dest=Path(sys.argv[2]);dest.mkdir(exist_ok=False)
rows=[]
for p in sorted((root/'snapshots').glob('*/capture.json'),key=lambda p:int(p.parent.name.split('-')[-1])):
    m=json.loads(p.read_text(encoding='utf-8'))
    for item in m['buffers']+[m['output']]:
        f=p.parent/item['file']
        assert f.stat().st_size==item['bytes'] and hashlib.sha256(f.read_bytes()).hexdigest()==item['sha256']
    y=np.fromfile(p.parent/m['output']['file'],dtype='<f4');assert np.isfinite(y).all()
    out=dest/f'frame-{m["frame_id"]}.txt'
    receipt=build(Q/'capture/capture-source-plan.txt',p,out)
    rows.append({'frame_id':m['frame_id'],'plan':str(out),'expected':str(p.parent/m['output']['file']),**receipt})
assert len(rows)==8 and all(rows[i]['frame_id']==rows[0]['frame_id']+i for i in range(8))
(dest/'JOBS.json').write_text(json.dumps(rows,indent=2),encoding='utf-8')
print('8 contiguous captured frames verified; replay plans ready',dest)
