#!/usr/bin/env python3
"""Verify the claimed vit_block31 pack defect: does the G3-bound pack carry the
denormal / non-scale ws[] header that makes the deployed qT codes collapse to 0?

Reports ws[0:16] for the pack the plan binds and for the X5-fixed candidate, plus the
pairing identity 1/ws[8] == ws[3] that X5 used as its correctness criterion.
"""
import glob
import hashlib
import sys
from pathlib import Path

import numpy as np

B = Path("@NR_WORKSPACE@")
sys.path.insert(0, str(B / "work/dsh-int8-fullnet-impl-20261024/tools"))
import planlib as L  # noqa: E402

PLAN = B / "work/dsh-int8-fullnet-impl-20261024/candidates/cumulative-v11/plan-cumulative-v11.txt"
DENORM = 1.1754944e-38


def show(label, path, n=16):
    p = Path(path)
    if not p.exists():
        print(f"{label}: MISSING {p}")
        return None
    a = np.fromfile(p, dtype="<f2", count=n).astype(np.float32)
    print(f"{label}\n  path {p}\n  sha256 {hashlib.sha256(p.read_bytes()).hexdigest()[:24]}"
          f"  bytes {p.stat().st_size}")
    print("  ws[0:16] =", np.array2string(a, precision=6))
    print(f"  denormals {int(np.sum((np.abs(a) < DENORM) & (a != 0)))}   "
          f"ws[3]={a[3]:.6g} ws[8]={a[8]:.6g} ws[10]={a[10]:.6g} "
          f"ws[8]*ws[3]={float(a[8]) * float(a[3]):.6g}")
    return a


def main():
    plan = L.Plan(PLAN)
    p = plan.by_name("vit_block31_attention")[0]
    b0 = p["binds"][0]
    name, size, path = plan.buffers[b0["res"]]
    print(f"plan {PLAN.name}  PASS {p['name']}  BIND0 -> {name} {size} B")
    show("G3-BOUND (what the cumulative plan actually reads)", path)
    print()
    pats = ["**/vit21-blocks31-38-v1/**/*.bin", "**/vit_21_weights.fp16.bin"]
    seen = set()
    for pat in pats:
        for g in glob.glob(str(B / "candidates" / pat), recursive=True)[:6]:
            q = Path(g)
            if q.stat().st_size < 64 or q in seen:
                continue
            seen.add(q)
            show("CANDIDATE FIX (X5 v1 pack)", q)
            print()
    return 0


if __name__ == "__main__":
    sys.exit(main())
