"""Print the context of every send.ugm with a given rd width. CPU only."""
import sys
sys.path.insert(0, '@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/fullnet-continuous-20261027/cards/n01b-f02-load-side')
import isaparse as P

path = sys.argv[1]
want = int(sys.argv[2])
before = int(sys.argv[3]) if len(sys.argv) > 3 else 8
after = int(sys.argv[4]) if len(sys.argv) > 4 else 10
only = sys.argv[5] if len(sys.argv) > 5 else None

isa = P.parse(path)
for k, i in enumerate(isa):
    if i['op'] != 'send.ugm' or P.rd_of(i) != want:
        continue
    if only and only not in P.msg_of(i):
        continue
    print('=== 0x%05X  %s' % (i['addr'], i['raw'][:150]))
    for j in range(max(0, k - before), min(len(isa), k + after)):
        mark = '>>' if j == k else '  '
        print('%s %5d 0x%05X %s' % (mark, j, isa[j]['addr'], isa[j]['raw'][:135]))
