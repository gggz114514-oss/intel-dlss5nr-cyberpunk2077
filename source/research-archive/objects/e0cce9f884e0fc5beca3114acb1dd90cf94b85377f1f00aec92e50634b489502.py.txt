"""n33 B3 diagnostic: which NC sites are in the parent identity but not in the
parent's own raw compile, and where does the block-aware match lose them?"""
import sys, json
from pathlib import Path
from collections import Counter

HERE = Path(__file__).parent
CAMP = HERE.parent.parent
sys.path.insert(0, str(CAMP))
import COMMON as K  # noqa: E402

ST = K.ST
BUILD = HERE / "build"
PARENT_SPV = Path("@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/"
                  "c32-rest-20261026/body-pv-den-overlap/candidate.spv")

pm = ST.Module(PARENT_SPV)
pm_raw = ST.Module(BUILD / "parent-raw.spv")
cm_raw = ST.Module(BUILD / "candidate-raw.spv")


def where(m, rid):
    for idx, (op, ops, *_) in enumerate(m.instructions):
        if ST.result_id_of(op, ops) == rid:
            reg = [fi for fi, r in enumerate(ST.function_regions(m)) if idx in r]
            return idx, op, ops, reg[0] if reg else None
    return None


def decorated_sites(m):
    dec = set(ST.no_contraction_ids(m))
    out = []
    for idx, (op, ops, *_) in enumerate(m.instructions):
        rid = ST.result_id_of(op, ops)
        if rid is not None and rid in dec:
            out.append((idx, op, tuple(ops)))
    return out


p_dec = decorated_sites(pm)
pr_dec = decorated_sites(pm_raw)
print("parent identity NC %d   parent raw NC %d" % (len(p_dec), len(pr_dec)))
print("opcode histogram parent identity:", Counter(o for _, o, _ in p_dec).most_common())
print("opcode histogram parent raw    :", Counter(o for _, o, _ in pr_dec).most_common())

# the parent identity was built from the raw parent by an in-order transfer, so the
# raw decs are a subset of the identity decs by position: 
pr_pos = [x[0] for x in pr_dec]
p_pos = [x[0] for x in p_dec]
extra = [i for i in p_pos if i not in pr_pos]
print("\npositions decorated in the identity but not in the raw parent: %s" % extra)
for i in extra:
    op, ops = pm.instructions[i][0], pm.instructions[i][1]
    reg = [fi for fi, r in enumerate(ST.function_regions(pm)) if i in r]
    print("   idx %d  op %d %s  function %s" % (i, op, ops, reg))
    for k in range(i - 2, i + 3):
        print("        %d %s" % (k, pm.instructions[k][:2]))

# where the candidate raw gets decorated
print("\ncandidate raw NC %d" % len(ST.no_contraction_ids(cm_raw)))
