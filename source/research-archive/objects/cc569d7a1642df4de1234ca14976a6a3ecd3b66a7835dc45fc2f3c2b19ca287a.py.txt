This script runs NO GPU work. It only reads queue/n06-f06-output/{gates,native,resources} and
records the observed ISA counts and per-pass GpuTime, so the card can state what landed.
"""
import hashlib, json, re, collections, sys

PARENT_KERNEL = 'a558554d30105096'
CAND_KERNEL = 'e662a7728e588f29'
ORD = [20, 23, 26, 29, 32, 196, 199, 202, 205, 208, 211]


def sha(p):
    return hashlib.sha256(Path(p).read_bytes()).hexdigest()


def census(path):
    isa = ISA.parse(path)
    ops = collections.Counter(i['op'] for i in isa)
    forms = collections.Counter()
    for i in isa:
        if i['op'].startswith('send'):
            m = re.search(r'(load|store)\.ugm\.([A-Za-z0-9]+)\.', i['comment'] or '')
            forms[(m.group(1) + '.' + m.group(2)) if m else 'other'] += 1
    regs = set()
    for i in isa:
        for o in i['ops']:
            for m in re.finditer(r'\br(\d+)', o):
                regs.add(int(m.group(1)))
    first = next(k for k, i in enumerate(isa) if i['op'].startswith('dpas'))
    return dict(path=str(path), instructions_isa_lines=len(isa),
                dpas=ops['dpas.8x8'], sends=sum(v for k, v in ops.items() if k.startswith('send')),
                send_forms=dict(forms), mov=ops['mov'], add=ops['add'], mul=ops['mul'],
                sync_nop=ops['sync.nop'], allwr=ops['allwr'], allrd=ops['sync.allrd'],
                syncs=(ops['sync.nop'] + ops['sync.allwr'] + ops['sync.allrd'] + ops['sync.bar']),
                shl=ops['shl'], max_r=max(regs),
                first_dpas_addr=hex(isa[first]['addr']), first_dpas_instruction_index=first)


def pipeline_stats(arm):
    d = json.loads((QROOT / 'resources/run' / arm / 'pipeline-statistics.json')
                   .read_text(encoding='utf-8-sig'))
    st = {s['name']: s['value'] for s in d['pipelines'][10]['executables'][0]['statistics']}
    return {k: st[k] for k in ('Instruction Count', 'Basic Block Count', 'Scratch Memory Size',
                               'Loop Count', 'Workgroup Memory Size')}


def passes(arm):
    d = json.loads((QROOT / 'resources/run' / arm / 'hardware-counters.json')
                   .read_text(encoding='utf-8-sig'))
    return {p['index']: (p['name'], p['values'][0]) for p in d['frames'][0]['passes']}
