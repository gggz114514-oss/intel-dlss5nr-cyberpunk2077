def decoration_attribution(mod):
    """NoContraction / RelaxedPrecision decorations attributed to the defining function."""
    defs = {}
    for idx, (off, opc, w) in enumerate(mod.insts):
        if opc in S.RESULT_OPS or opc in (4456, 22, 21, 32, 28, 30, 43, 44, 46, 54, 55, 59, 80, 81, 82):
            if len(w) > 2:
                defs[w[2]] = idx
    out = {}
    for (target, name, rest) in mod.decor:
        idx = defs.get(target)
        fn = "module"
        if idx is not None:
            for f in mod.functions:
                if f["first"] <= idx <= f["last"]:
                    fn = f["name"].split("(")[0]
        out.setdefault(name, {}).setdefault(fn, 0)
        out[name][fn] += mod.decor[(target, name, rest)]
    return out

def oracle():
    """CPU INT8 reference for one owner SG: exact int32 dot vs the FP16 path."""
    rng = np.random.default_rng(20261024)
    keys, dims, queries = 64, 32, 16
    # V: half values, mixed sign, the range a swin QKV projection produces
    V = (rng.normal(0.0, 0.6, size=(keys, dims))).astype(np.float16)
    # P: half probabilities already normalised by the parent's tree per query column
    logits = rng.normal(0.0, 1.2, size=(keys, queries))
    logits -= logits.max(axis=0, keepdims=True)
    P = np.exp(logits)
    P = P / P.sum(axis=0, keepdims=True)
    P = P.astype(np.float16)
    # parent: FP32 accumulation of the half operands
    ref64 = (V.astype(np.float64).T @ P.astype(np.float64))       # exact reference
    ref32 = (V.astype(np.float32).T @ P.astype(np.float32))       # FP32 MMA-like
    # candidate: one per-workgroup V scale, one per-query P scale, RNE, int32 dot
    sV = np.abs(V.astype(np.float32)).max() / 127.0
    qV = np.clip(np.rint(V.astype(np.float32) / sV), -127, 127).astype(np.int32)
    pmx = P.astype(np.float32).max(axis=0)
    sP = pmx * (1.0 / 127.0)
    qP = np.clip(np.rint(P.astype(np.float32) / sP[None, :]), -127, 127).astype(np.int32)
    acc = (qV.T.astype(np.int64) @ qP.astype(np.int64))            # exact int32 range
    cand = acc.astype(np.float64) * (sV * sP)[None, :]
    # the fragment maps are bijections, so the int32 dot above is the MMA's arithmetic;
    # here we only check the numeric contract, not the transport.
    err = np.abs(cand - ref32)
    rel = err / np.maximum(np.abs(ref32), 1e-12)
    bound = (sV * 0.5) * np.abs(P.astype(np.float64)).sum(axis=0)[None, :] + \
            (sP[None, :] * 0.5) * np.abs(V.astype(np.float64)).sum(axis=0)[:, None]
    return {
        "label": "R04-PV-INT8-LOCAL-ORACLE",
        "what": "the shader's PV arithmetic on one owner SG: ctx^T = sum over 64 keys of "
                "V^T.SP^-1 * P^T.SP, exact int32 accumulation, dequantised once in FP32",
        "who_generates_it": "this card, on CPU, from the same formulas the shader encodes; "
                            "no GPU and no capture involved",
        "status": "LOCAL_REFERENCE_ONLY; the numeric gate against this reference has NOT been "
                  "run on a device (no GPU slot)",
        "inputs": {"generator": "numpy default_rng(20261024)",
                   "V": "64x32 half, N(0,0.6)", "P": "64x16 half, softmax per column",
                   "non_zero": bool(np.count_nonzero(V) and np.count_nonzero(P)),
                   "V_absmax": float(np.abs(V.astype(np.float32)).max()),
                   "P_max": float(pmx.max())},
        "int8_codes": {"qV_non_zero": int(np.count_nonzero(qV)),
                       "qV_zero_share": float(1.0 - np.count_nonzero(qV) / qV.size),
                       "qP_non_zero": int(np.count_nonzero(qP)),
                       "sV": float(sV), "sP_range": [float(sP.min()), float(sP.max())],
                       "qV_range": [int(qV.min()), int(qV.max())],
                       "qP_range": [int(qP.min()), int(qP.max())],
                       "int32_accumulator_maxabs": int(np.abs(acc).max()),
                       "int32_overflow_free": bool(np.abs(acc).max() < 2 ** 31)},
        "error_vs_fp32_parent_path": {
            "err_rms": float(np.sqrt((err ** 2).mean())),
            "ref_rms": float(np.sqrt((ref32.astype(np.float64) ** 2).mean())),
            "err_rms_over_ref_rms": float(np.sqrt((err ** 2).mean()) /
                                          np.sqrt((ref32.astype(np.float64) ** 2).mean())),
            "share_err_above_1pct_ref_rms": float((err > 0.01 * np.sqrt(
                (ref32.astype(np.float64) ** 2).mean())).mean()),
            "max_abs": float(err.max()), "mean_abs": float(err.mean()),
            "max_rel": float(rel.max()), "p99_rel": float(np.percentile(rel, 99)),
            "analytic_bound_max": float(bound.max()),
            "within_analytic_bound": bool(err.max() <= bound.max() * 1.05),
            "reference_magnitude": float(np.abs(ref32).max()),
        },
        "why_this_is_attributable": [
            "the int32 dot over 64 keys is exact: no FP accumulation order difference",
            "the only two roundings are the per-element quantisation of V and P with RNE",
            "the parent's half P, its tree, its reciprocal and its single rounding point are "
            "untouched, so the quantiser input is byte-identical to the FP16 consumer's input",
            "the per-query P scale is provably <= 1 scale: max_k P <= 1 (P = exp*inv with "
            "inv = 1/max(sum,2^-14)), so pmx*(127/pmx) = 127 never saturates",
        ],
        "not_covered": "native execution, the pipeline's actual P values, and the full-net "
                       "quality contract; those are the GPU gate jobs in run/r04-c64-attn",
    }
