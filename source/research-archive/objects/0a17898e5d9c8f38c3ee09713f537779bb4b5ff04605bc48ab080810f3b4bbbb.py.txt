"""Record L22 (measured stall budget as the targeting metric) and L23 (AMD comparison)."""
import json, sys
from pathlib import Path
sys.path.insert(0, str(Path(__file__).parent))
import COMMON as K

p = K.C / 'evidence' / 'LESSONS.json'
d = K.read_json(p)
ids = [x['id'] for x in d['append_after_L7']]

new = []
if 'L22' not in ids:
    new.append({
        'id': 'L22',
        'claim': 'Hardware stall counters, converted to milliseconds per PASS, are the first targeting metric that predicted our outcomes; instruction and message counts do not',
        'evidence': [
            'measured on the chain head: frame 17.142 ms of per-PASS GpuTime, of which 9.399 ms (54.8%) the XVE is not active; time-weighted stalls are ALUWR 3.670 ms (21.4%), SENDWR 2.161 ms (12.6%), PIPESTALL 1.680 ms (9.8%), BARRIER 1.433 ms (8.4%), PS_DEPENDENCY 0.292 ms (1.7%)',
            'inter-dispatch gaps are only 17.524 - 17.142 = 0.382 ms (2.2%), so the 232 full pipeline barriers between dispatches are NOT where the serialisation money is - a persistent/tile-chain redesign is bounded by ~0.38 ms',
            'the metric explains the campaign: n18 deleted a modulo chain that fed addresses (on the dependency path) for +0.079 ms, n19 deleted duplicate gathers off the path for +0.005 ms, and F20s four message-deleting cards were all null while its SENDWR is only 0.029 ms',
            'message cost is position dependent: n05a paid 126 ps per deleted message where the message was serialised, while the frame average is 31 ps, so a linear message-price model over-predicts by 4x (a naive global application predicted 405% of the frame)'
        ],
        'consequence': 'Target passes by measured stall milliseconds (ALUWR + BARRIER), and ask whether a change shortens a dependency chain or removes a barrier together with the data flow that forces it. Do not open persistent/tile-chain work expect to recover more than a few tenths of a millisecond.'
    })
if 'L23' not in ids:
    new.append({
        'id': 'L23',
        'claim': 'The AMD RDNA4 port of the same network has the same LDS-staging-plus-barrier structure; it hides the cost with double buffering and FP8 payloads rather than avoiding barriers',
        'evidence': [
            'the AMD reference stages fragments in shared memory exactly as we do (ffwd3_t.comp: shared fe4m3vec4 lds_fr[...], lds_fh[2*...]; attn.comp: lds_q/lds_k/lds_v/lds_p) and its barrier counts are comparable (fswin_t.comp 27, fswin_body.glsl 16, attn.comp 13)',
            'it double buffers with a [buf] dimension and hh / 1-hh index flipping, so the write of stage i+1 overlaps the read of stage i instead of waiting',
            'its LDS payloads are NR_E4M3, so the same 16 KB holds twice the elements and the handoff needs half the transactions',
            'its workgroups are smaller: 8x8 = 64 threads typical (2 waves) versus 205 of our 232 passes using 256 threads (8 subgroups), so its barriers fan in over 4x fewer waves'
        ],
        'consequence': 'Of the three AMD treatments only window rotation / barrier coalescing transfers without re-quantisation, and we have already proven it once (n10 merged two arenas and alternated the A-slice windows for +0.094 ms). Memory-doubling ping-pong is not free for us: F20 already uses 29184 B of workgroup memory, so doubling would cut resident workgroups (L4).'
    })
if new:
    d['append_after_L7'].extend(new)
    K.write_json(p, d)
    print('added', [x['id'] for x in new])
else:
    print('already present')
