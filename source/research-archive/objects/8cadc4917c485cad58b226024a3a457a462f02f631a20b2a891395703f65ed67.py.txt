#!/usr/bin/env python3
"""Probe 2: fix the cooperative-matrix type encoding, then trace the seed value.

SPV_KHR_cooperative_matrix as emitted by glslang 16.5.0:
  OpTypeCooperativeMatrixKHR  %id %comp %scope %rows %cols %use   (all ids)
  OpCooperativeMatrixLoadKHR  %type %id %ptr %stride %layout %memop...
  OpCooperativeMatrixStoreKHR %ptr %obj %stride %layout %memop...
  OpCooperativeMatrixMulAddKHR %type %id %A %B %C %ops
"""
import sys

sys.path.insert(0, r"@NR_WORKSPACE@/work/dsh-xmx-handoff-validation-20261024/candidates/b3-c32-seed/tools")
from b3_spv import Module  # noqa: E402

P = r"@NR_WORKSPACE@/work/dsh-xmx-handoff-validation-20261024/build/b3-c32-seed/parent_repro.spv"
M = Module(P)


def konst(cid):
    ins = M.by_id.get(cid)
    if ins is None:
        return None
    if ins.op == 43:
        return ins.ops[2]
    return None


# 1. coopmat types, rid at operand 0
print("== cooperative matrix types ==")
cm = {}
for ins in M.insts:
    if ins.op == 4456:
        rid = ins.ops[0]
        comp, scope, rows, cols, use = ins.ops[1:]
        cm[rid] = dict(comp=comp, rows=konst(rows), cols=konst(cols), use=konst(use))
        print("  type %-6d comp=%-4d rows=%-3s cols=%-3s use=%-3s" %
              (rid, comp, konst(rows), konst(cols), konst(use)))
COMPNAME = {6: "f16", 28: "f32", 58: "i32", 510: "i8"}
for t, d in cm.items():
    d["name"] = "cm[%s %sx%s use%s]" % (COMPNAME.get(d["comp"], "?"), d["rows"], d["cols"], d["use"])
print("names:", {k: v["name"] for k, v in sorted(cm.items()) })

# constants behind scope/use/etc
print("\n== constants used as type params ==")
for cid in (29, 30, 31, 32, 41, 233, 518):
    ins = M.by_id.get(cid)
    print("  const %d -> %s" % (cid, (ins.op, list(ins.ops)) if ins else None))

# 2. all instructions producing a cooperative matrix type
print("\n== instructions whose result type is a cooperative mat ==")
census = {}
for ins in M.insts:
    if ins.rtype in cm and ins.op not in (4456,):
        census[(ins.rtype, ins.op)] = census.get((ins.rtype, ins.op), 0) + 1
for (t, op), c in sorted(census.items()):
    print("  %-30s %-34s x%d" % (cm[t]["name"], M.by_id and __import__("b3_spv").OPNAME.get(op, op), c))

# 3. OpFConvert: what types
print("\n== OpFConvert (115) sites ==")
for ins in M.insts:
    print("  word %-6d type=%-4s (%s) -> %-4s from %-4s (%s)" % (
        ins.pos, ins.rtype, cm.get(ins.rtype, {}).get("name", "scalar"),
        ins.rid, ins.ops[2], cm.get(ins.ops[2], {}).get("name", "scalar")))
