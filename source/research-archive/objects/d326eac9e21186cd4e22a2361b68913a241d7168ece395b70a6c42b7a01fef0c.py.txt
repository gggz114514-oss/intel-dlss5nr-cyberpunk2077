"""n21 step-10: read the two native captures of this card (main agent's same-batch run)
and answer the card's question directly: max_r_data / max_r_any / distinct regs.

Read only: the captures live under queue/n21-debug-regrelief/ (main agent's output).
"""
import sys, re, json, collections
from pathlib import Path

CARD = Path(__file__).resolve().parent
C = CARD.parent.parent
sys.path.insert(0, str(C))
import COMMON as K

Q = C / 'queue/n21-debug-regrelief/native/captures'
BASE = Q / 'native-native-base/ISA/CS_asm4e6ba3ddc1931535_simd32_0000000000000022_33_orig.asm'
CAND = Q / 'native-native-cand/ISA/CS_asm966fc69aa087705b_simd32_0000000000000022_33_orig.asm'

LINE = re.compile(r'^/\* \[([0-9A-F]+)\]\s+\*/\s+(?:\(([^)]*)\)\s+)?([a-z0-9_.]+)\s*(.*)$')
REG = re.compile(r'\b(r|acc)(\d+)(?:\.(\d+))?')


def parse(path):
    ins = []
    for raw in Path(path).read_text(encoding='utf-8', errors='replace').splitlines():
        m = LINE.match(raw.rstrip())
        if not m:
            continue
        addr, mask, op, rest = m.groups()
        ins.append(dict(addr=int(addr, 16), mask=mask or '', op=op, rest=rest.strip(), text=raw.strip()))
    return ins


def regs(rest):
    return {int(n) for kind, n, _ in REG.findall(rest) if kind == 'r'}


def facts(path):
    ins = parse(path)
    allr = collections.Counter()
    for i in ins:
        for r in regs(i['rest']):
            allr[r] += 1
    # r127-only-and rule: the highest register whose only use is the `and r0` flag temp
    def is_flag_temp(i):
        return i['op'] == 'and' and 'r127' in i['rest'] and 'r0.0' in i['rest']
    flag = [i for i in ins if is_flag_temp(i)]
    data_regs = set(allr)
    for i in flag:
        data_regs.discard(127)
    # stores / sends / dpas
    store = sum(1 for i in ins if 'store.ugm' in i['rest'])
    dpas = sum(1 for i in ins if i['op'].startswith('dpas'))
    sends = sum(1 for i in ins if i['op'].startswith('send'))
    labels = sum(1 for i in ins if i['op'] == 'jmpi') + 1
    return dict(
        path=str(path), sha256=K.sha(path), instructions=len(ins),
        max_r_any=max(allr) if allr else None,
        max_r_data=max(data_regs) if data_regs else None,
        distinct_r=len(allr),
        flag_temp_127=len(flag),
        store_ugm=store, dpas=dpas, sends=sends,
        top_regs={str(k): allr[k] for k in sorted(allr) if k >= 118},
    )


b, c = facts(BASE), facts(CAND)
res = dict(base=b, cand=c,
           delta=dict(instructions=c['instructions'] - b['instructions'],
                      max_r_any=c['max_r_any'] - b['max_r_any'],
                      max_r_data=c['max_r_data'] - b['max_r_data'],
                      distinct_r=c['distinct_r'] - b['distinct_r'],
                      store_ugm=c['store_ugm'] - b['store_ugm'],
                      sends=c['sends'] - b['sends'],
                      dpas=c['dpas'] - b['dpas']))
K.write_json(CARD / 'build' / 'measured-native.json', res)
print(json.dumps(res, indent=1))
