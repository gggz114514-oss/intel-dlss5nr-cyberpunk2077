"""Compare base vs candidate ISA for the F05 package, pairing kernels by pipeline id.

Reports, per changed kernel: instruction count, dpas/rsqt/inv counts, and the rsqt ->
first-consumer distance histogram (the hard-fail criterion of the latency-hiding card).
"""
import re, sys
from collections import Counter
from pathlib import Path
sys.path.insert(0, str(Path(__file__).parent))
import COMMON as K
import ISA

REG = re.compile(r'\br(\d+)(?:\.(\d+))?')


def reg(o):
    m = REG.search(o or '')
    return (int(m.group(1)), m.group(2)) if m else None


def distances(path):
    isa = ISA.parse(path)
    ops = [i['ops'] for i in isa]
    kinds = [i['op'] for i in isa]
    out = []
    for i, k in enumerate(kinds):
        if k != 'math.rsqt':
            continue
        srcs = [reg(o) for o in ops[i]]
        srcs = [s for s in srcs if s]
        if not srcs:
            continue
        d = None
        for j in range(i + 1, min(i + 120, len(kinds))):
            if any(reg(o) == srcs[0] for o in ops[j]):
                d = j - i
                break
        out.append(d)
    return out, Counter(kinds), isa


def pid(name):
    m = re.search(r'_(\d+)_orig\.asm$', name)
    return int(m.group(1)) if m else -1


base = Path('queue/n32-run/run/captures-base')
cand = Path('queue/n32-run/run/captures-cand')
fb = {pid(p.name): p for p in base.rglob('*.asm')}
fc = {pid(p.name): p for p in cand.rglob('*.asm')}
common = sorted(set(fb) & set(fc))
changed = []
for i in common:
    a = fb[i].read_bytes()
    b = fc[i].read_bytes()
    if a != b:
        changed.append(i)
print('pipelines:', len(common), 'changed:', changed)
for i in changed:
    for tag, p in (('base', fb[i]), ('cand', fc[i])):
        d, kinds, isa = distances(p)
        cnt = Counter(x['op'] for x in isa)
        s = ISA.summary(p)
        hist = Counter(d) if d else {}
        print('  pipe %-3d %-4s instr %5d dpas %3d rsqt %2d inv %2d maxr %3d  dist %s'
              % (i, tag, s['instructions'], s['dpas'], cnt.get('math.rsqt', 0), cnt.get('math.inv', 0),
                 s['max_r'], dict(sorted(hist.items(), key=lambda kv: (kv[0] is None, kv[0])))))
