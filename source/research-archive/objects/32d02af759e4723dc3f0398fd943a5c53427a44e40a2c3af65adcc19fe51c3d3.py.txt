"""Show one narrow message class with its 3 preceding instructions (read-only)."""
import json, os, re, sys

CARD = os.path.dirname(os.path.abspath(__file__))
EVID = os.path.join(os.path.dirname(os.path.dirname(CARD)), "evidence")
IDX = json.load(open(os.path.join(EVID, "NATIVE-INDEX.json"), encoding="utf-8"))
COV = {r["ordinal"]: r for r in IDX["coverage"]}
H2F = json.load(open(os.path.join(CARD, "_hash2asm.json"), encoding="utf-8"))
LINE_RE = re.compile(r"^/\* \[([0-9A-Fa-f]+)\]\s+\*/\s*(.*)$")


def resolve(o):
    for c in (H2F.get(str(o)) or []):
        if os.path.exists(c):
            return c


def main():
    o = int(sys.argv[1]); pat = sys.argv[2]
    before = int(sys.argv[3]) if len(sys.argv) > 3 else 3
    limit = int(sys.argv[4]) if len(sys.argv) > 4 else 6
    p = resolve(o)
    print("### ord %d %s %s" % (o, COV[o]["family"], COV[o]["name"]))
    print("### ISA", p)
    rows = [(m.group(1), m.group(2).strip()) for raw in open(p, encoding="utf-8", errors="replace")
            for m in [LINE_RE.match(raw)] if m]
    hits = [i for i, (a, b) in enumerate(rows) if pat in b]
    print("### matches", len(hits))
    for i in hits[:limit]:
        print("---- addr %s" % rows[i][0])
        for j in range(max(0, i - before), i + 1):
            print("   [%s] %s" % (rows[j][0], rows[j][1][:160]))


if __name__ == "__main__":
    main()
