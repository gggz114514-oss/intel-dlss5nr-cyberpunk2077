"""make_candidate_json.py - assemble CANDIDATE.json from the evidence files.

No new measurement: every field is read back from the receipts/evidence this card
already produced, so the identity record cannot drift from the artifacts.
"""
import hashlib
import json
from pathlib import Path

CARD = Path(__file__).resolve().parents[1]
EV = CARD / "evidence"


def sha(p):
    return hashlib.sha256(Path(p).read_bytes()).hexdigest()


def ident(rel):
    p = CARD / rel
    return {"path": rel, "bytes": p.stat().st_size, "sha256": sha(p)}
