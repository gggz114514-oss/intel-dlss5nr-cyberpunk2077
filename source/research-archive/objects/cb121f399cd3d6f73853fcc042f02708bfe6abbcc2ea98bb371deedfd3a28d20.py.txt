import sys, json, hashlib
from pathlib import Path

HERE = Path(__file__).parent
CAMP = HERE.parent.parent
sys.path.insert(0, str(CAMP))
import COMMON as K  # noqa: E402

ST = K.ST
BUILD = HERE / "build"
BUILD.mkdir(exist_ok=True)

PARENT_SRC = Path("@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/"
                  "c32-rest-20261026/body-pv-den-overlap/candidate.comp")
PARENT_SPV = Path("@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/"
                  "c32-rest-20261026/body-pv-den-overlap/candidate.spv")
PARENT_SPV_SHA = "4c034a532ce60f5fa608d1ce67f6368819c052c3d5268eb51f7f680ea83aaf10"
DECO_PARENT = Path("@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/"
                   "c32-feed-owner-20261025/kv-tile/candidate.spv")
DECO_PARENT_SHA = "c5600f9a4595620b637fc8c84fd17fc197118d3c3a0a3717d8901659d363fba3"
INC = Path("@NR_WORKSPACE@/candidates/zcode-fp16-compute-backend-20261021/"
           "u09-join-c32-body1-v1/source")


def sha(p):
    return hashlib.sha256(Path(p).read_bytes()).hexdigest()


def main():
    assert sha(PARENT_SPV) == PARENT_SPV_SHA, sha(PARENT_SPV)
    assert sha(DECO_PARENT) == DECO_PARENT_SHA, sha(DECO_PARENT)
    raw = BUILD / "parent-raw.spv"
    K.compile_glsl(PARENT_SRC, raw, [INC])
    cm = ST.Module(raw)
    pm = ST.Module(DECO_PARENT)
    keep, info = ST.transfer_decoration_pattern_aligned(pm, cm, [])
    out = BUILD / "parent-rebuild.spv"
    ST.set_no_contraction(cm, keep, out)
    rec = dict(parent_source=str(PARENT_SRC), parent_source_sha=sha(PARENT_SRC),
               parent_spv=str(PARENT_SPV), parent_spv_sha=PARENT_SPV_SHA,
               parent_bytes=PARENT_SPV.stat().st_size,
               rebuild=str(out), rebuild_sha=sha(out), byte_match=sha(out) == PARENT_SPV_SHA,
               raw_sha=sha(raw), raw_match=sha(raw) == PARENT_SPV_SHA,
               nc_keep=len(keep), nc_parent=len(ST.no_contraction_ids(pm)),
               functions_mapped=info["functions_mapped"],
               child_functions_without_parent_counterpart=len(
                   info["child_functions_without_parent_counterpart"]),
               recipe="evidence/RECIPES.json 4c034a532ce6",
               decoration_parent=str(DECO_PARENT), decoration_parent_sha=DECO_PARENT_SHA)
    print("parent rebuild: NC keep=%d/%d  byte_match=%s  sha=%s"
          % (rec["nc_keep"], rec["nc_parent"], rec["byte_match"], rec["rebuild_sha"][:16]))
    (BUILD / "build.json").write_text(json.dumps(rec, indent=1), encoding="utf-8")
    print("wrote", BUILD / "build.json")
    assert rec["byte_match"], "parent bytes not reproduced - do not continue"


if __name__ == "__main__":
    main()
