"""Correct the record: cross-run counter noise exceeds the effects we were reading.

Two profiling runs of the SAME chain head (before and after the n27+n29 adoptions, but the
two kernels of interest are unchanged in both) differ by ~4.5 percentage points on every
stall counter and by +0.36 ms on the frame sum. That is measurement drift, not a regression:
the XMX slot count is bit-identical and the per-kernel GpuTime is flat.
"""
import json, sys
from pathlib import Path
sys.path.insert(0, str(Path(__file__).parent))
import COMMON as K

# 1. record the noise measurement
K.write_json(K.C / 'evidence' / 'COUNTER-NOISE.json', dict(
    method='two separate profiler runs of the same plan, warm 0, reps 1, same metric set',
    frame_sum_ms=[17.142, 17.498],
    stall_point_deltas={'ALUWR': 3.79, 'SENDWR': 4.58, 'BARRIER': 4.56, 'PIPESTALL': 4.42,
                        'PS_DEPENDENCY': 4.88},
    stable_quantities={'XMX_FP16_slots': 9.8037e8,
                       'f1c32_image_pre_gpu_ms': [1.247, 1.199],
                       's37_fold_body_gpu_ms': [1.346, 1.357]},
    conclusion='aggregate stall percentages drift by up to ~4.9 points between runs while '
               'deterministic counts and per-kernel times do not, so only differences much '
               'larger than ~5 points can be read across runs'))

# 2. correct n30's attribution
p = K.C / 'cards' / 'n30-f20-arena-rotate' / 'NOTES-main.json'
d = K.read_json(p)
d['attribution']['counter_result'] = (
    '两臂计数器实测该 PASS：XVE_STALL_BARRIER 16.109%→15.959%（−0.149 点）、折算 0.2173→0.2167 ms。'
    '**但这个差不能被解读为“没效果”**：后续用同一链头连做两次 profiling（evidence/COUNTER-NOISE.json）显示，'
    '聚合 stall 百分比在两次同条件运行之间会漂移 **±3.8…4.9 点**，而确定性计数（XMX slots）与逐核 GpuTime 稳定。'
    '所以 −0.149 点**落在噪声内，既不能证实也不能否证**；本卡的判定只依据 ordinary（−0.0150，CI 跨 0）。')
d['attribution']['why'] = (
    '不是“屏障数量与代价不成比例”这种强结论，而是：**这道屏障的代价可能真实存在但小于计数器分辨率**。'
    '要判它必须用更大的样本或更好的指标（例如把该 PASS 单独跑到 ns 级重复、或看 n28 那样真删一道后 ordinary 的变化）。')
K.write_json(p, d)

# 3. refine L22
lp = K.C / 'evidence' / 'LESSONS.json'
L = K.read_json(lp)
for x in L['append_after_L7']:
    if x['id'] == 'L22':
        x['claim'] = ('Hardware stall counters converted to milliseconds rank passes usefully within one run, '
                      'but their aggregate percentages drift by ~5 points between runs and cannot resolve small effects')
        x['evidence'].append('two runs of the same plan differed by ALUWR +3.79, SENDWR +4.58, BARRIER +4.56, '
                             'PIPESTALL +4.42 and PS_DEPENDENCY +4.88 points while the deterministic XMX slot count '
                             'was identical (9.8037e8) and the two largest kernels times were flat')
        x['consequence'] = ('Use the counter budget to rank passes inside one run and to find big asymmetries. '
                            'Do not read a difference of a few points across runs as an effect, and never let a '
                            'counter delta override ordinary timing: the timing is the arbiter.')
        print('L22 refined')
K.write_json(lp, L)
print('n30 attribution corrected; noise recorded')
