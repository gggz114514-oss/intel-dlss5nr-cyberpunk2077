"""Structural parent-vs-candidate diff for ord 59 (F07 C256 DS-end)."""
import sys, json, subprocess
from collections import Counter
from pathlib import Path

CAMP = Path('@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/fullnet-continuous-20261027')
sys.path.insert(0, str(CAMP))
sys.path.insert(0, r'@NR_WORKSPACE@/build/zcode-fp16-compute-backend-20261021/scripts')
import COMMON as K
import spv_tools_local as ST
sys.path.insert(0, str(CAMP / 'cards/n07n14-arena/tools'))
import spv_report as R

CARD = CAMP / 'cards/n07n14-arena'
row = K.row_by_ordinal(59)
P = Path(row['spv'])
C = CARD / 'candidate.spv'

mp, mc = ST.Module(P), ST.Module(C)
print('parent    instr %d words %d' % (len(mp.instructions), len(mp.words)))
print('candidate instr %d words %d' % (len(mc.instructions), len(mc.words)))

np_, nc_ = R.names(mp), R.names(mc)
print('named objects parent   ', sorted((k, v) for k, v in np_.items() if k < 60))
print('named objects candidate', sorted((k, v) for k, v in nc_.items() if k < 60))

fp = {name: reg for fid, name, reg in R.funcs(mp)}
fc = {name: reg for fid, name, reg in R.funcs(mc)}
print('parent functions   ', [k for k in fp])
print('candidate functions', [k for k in fc])

KEYS = (57, 61, 62, 65, 71, 72, 224, 225, 4457, 4458, 4459, 128, 130, 132, 134, 135, 137, 138, 139, 112, 113, 79, 80, 81, 82)
def h(m, reg):
    c = Counter(m.instructions[i][0] for i in reg)
    return {R.OPNAME.get(k, k): c.get(k, 0) for k in KEYS}

for f in fp:
    print('--', f)
    print('   parent   ', h(mp, fp[f]))
    print('   candidate', h(mc, fc.get(f, [])))

tp = Counter(op for op, _, _, _ in mp.instructions)
tc = Counter(op for op, _, _, _ in mc.instructions)
print('TOTAL delta', {R.OPNAME.get(k, k): (tp.get(k, 0), tc.get(k, 0)) for k in set(tp) | set(tc) if tp.get(k, 0) != tc.get(k, 0)})

ncp, ncc = ST.no_contraction_ids(mp), ST.no_contraction_ids(mc)
print('NoContraction parent', len(ncp), sorted(ncp))
print('NoContraction cand  ', len(ncc), sorted(ncc))
for tag, m, ids in (('parent', mp, ncp), ('cand', mc, ncc)):
    for cid in sorted(ids):
        op, ops = m.defining_instruction(cid)
        print('  ', tag, cid, R.OPNAME.get(op, op), m.operand_signature(cid, 4))

print('decorations total', len([1 for op, _, _, _ in mp.instructions if op in (71, 72)]),
      len([1 for op, _, _, _ in mc.instructions if op in (71, 72)]))
print('storage parent   ', mp.storage_bytes())
print('storage candidate', mc.storage_bytes())
print('bindings parent   ', mp.binding_ids())
print('bindings candidate', mc.binding_ids())

# code-region instruction sequence diff
def seq(m):
    regs = ST.function_regions(m)
    idxs = sorted(set(i for reg in regs for i in reg))
    return [m.instructions[i][0] for i in idxs]
sp, sc = seq(mp), seq(mc)
print('code region length', len(sp), '->', len(sc))
