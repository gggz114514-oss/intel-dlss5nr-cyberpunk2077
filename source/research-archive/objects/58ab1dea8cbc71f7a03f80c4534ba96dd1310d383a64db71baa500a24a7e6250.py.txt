"""Final self-check for the n11-vit-ffn card: rebuild everything and re-verify identities."""
import json, sys, hashlib, subprocess
from pathlib import Path
sys.path.insert(0, r'@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/fullnet-continuous-20261027')
import COMMON as K

CARD = K.C / 'cards' / 'n11-vit-ffn'
GLSLANG = str(K.GLSLANG)
VALIDATE = str(K.VALIDATE)
PARENT = '54029e6beb917f447e71add99a8b61c196d3e6428cc1df0113db82a00cb9c2a2'
CAND = 'b68712e15bf6bf76eeb1db151d546195987aeec285fc0cc9bf79f805e6616fad'
PLAN = '7c16fbf835f4e03bd0ae8d63dbffc56c126fc6c6ab339fe80ea134bff564e920'

res = {}
tmp = CARD / '_recheck'
tmp.mkdir(exist_ok=True)

def comp(src, inc, out):
    r = subprocess.run([GLSLANG, '-V', '--target-env', 'vulkan1.3', '-I' + str(inc), str(src), '-o', str(out)],
                       capture_output=True, text=True, encoding='utf-8', errors='replace')
    return r.returncode

res['rebuild_parent_rc'] = comp(CARD / 'parent-src' / 'vit_nb04_expand_s32.comp', CARD / 'parent-src', tmp / 'p.spv')
res['rebuild_parent_sha'] = K.sha(tmp / 'p.spv')
res['rebuild_parent_ok'] = res['rebuild_parent_sha'] == PARENT
res['rebuild_cand_rc'] = comp(CARD / 'candidate.comp', CARD / 'candidate-glsl', tmp / 'c.spv')
res['rebuild_cand_sha'] = K.sha(tmp / 'c.spv')
res['rebuild_cand_ok'] = res['rebuild_cand_sha'] == CAND

r = subprocess.run([VALIDATE, str(CARD / 'candidate.spv')], capture_output=True, text=True)
res['validate_rc'] = r.returncode
res['validate_out'] = (r.stdout + r.stderr).strip()

res['plan_sha'] = K.sha(CARD / 'plan.txt')
res['plan_ok'] = res['plan_sha'] == PLAN
res['cpu_check'] = K.cpu_check(CARD / 'plan.txt').strip()

# pass/buffer counts in plan header vs parent
head = (CARD / 'plan.txt').read_text(encoding='utf-8').splitlines()[0].split()
phead = K.BASELINE.read_text(encoding='utf-8-sig').splitlines()[0].split()
res['plan_header'] = head
res['parent_header'] = phead
res['header_equal'] = head == phead
res['pass_lines'] = sum(1 for l in (CARD / 'plan.txt').read_text(encoding='utf-8').splitlines() if l.startswith('PASS '))
res['passes_targeting_candidate'] = (CARD / 'plan.txt').read_text(encoding='utf-8').count(
    (CARD / 'candidate.spv').as_posix())
res['passes_targeting_parent_spv'] = (CARD / 'plan.txt').read_text(encoding='utf-8').count('K11_s32.spv')

deliverables = ['CARD.txt', 'ANALYSIS.md', 'PROOF.json', 'SOURCE-DIFF.patch', 'candidate.comp',
                'candidate.spv', 'BUILD.json', 'COST-LEDGER.json', 'NATIVE-PREDICTION.md',
                'GPU-REQUEST.json', 'RESUME.txt', 'plan.txt', 'parent-rebuild.spv',
                'NATIVE-ANALYSIS.json', 'VIT-FFN-NATIVE-COMPARISON.json']
res['deliverables'] = {f: dict(exists=(CARD / f).exists(),
                               bytes=(CARD / f).stat().st_size if (CARD / f).exists() else 0)
                       for f in deliverables}
bad_json = []
for f in ['PROOF.json', 'BUILD.json', 'COST-LEDGER.json', 'GPU-REQUEST.json',
          'NATIVE-ANALYSIS.json', 'VIT-FFN-NATIVE-COMPARISON.json']:
    try:
        json.loads((CARD / f).read_text(encoding='utf-8'))
    except Exception as e:
        bad_json.append((f, str(e)))
res['json_ok'] = not bad_json
res['bad_json'] = bad_json
proof = json.loads((CARD / 'PROOF.json').read_text(encoding='utf-8'))
res['proof_all_ok'] = proof['coverage_ok']
res['proof_read_domain_288'] = proof['read_domain']['288']
print(json.dumps(res, indent=1))
