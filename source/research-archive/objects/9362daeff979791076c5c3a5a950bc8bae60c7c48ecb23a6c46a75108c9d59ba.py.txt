"""Geometry-only probe (CPU): parent attention source with ONLY the candidate's
workgroup geometry applied (local_size 256->128, owner span 16->32 query columns,
fill stride 256->128) and no second query block.

Purpose: show mechanically that the geometry change alone leaves the parent's
per-query arithmetic and control structure untouched (identical SPIR-V opcode
histogram), so the candidate's extra arithmetic is exactly one duplicated
per-block chain (block B) and nothing else.

Run:  python probe-geom.py
Writes build/probe-geom.comp, build/probe-geom.spv, build/PROBE-GEOM.json
"""
import json, sys
from pathlib import Path

CAMP = Path('@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/fullnet-continuous-20261027')
sys.path.insert(0, str(CAMP))
import COMMON as K
import build as B          # reuse spv_stats()

CARD = CAMP / 'cards/n13-vit-attn'
PARENT_SRC = Path(K.row_by_ordinal(96)['source'])


def main():
    text = PARENT_SRC.read_text(encoding='utf-8-sig')
    subs = [
        ('layout(local_size_x=256) in;', 'layout(local_size_x=128) in;'),
        ('uint qg=(sg/2u)*16u,dg=(sg%2u)*16u;', 'uint qg=(sg/2u)*32u;'),
        ('e+=256u', 'e+=128u'),
    ]
    for a, b in subs:
        n = text.count(a)
        assert n, 'missing ' + a
        text = text.replace(a, b)
        print('applied %-46s x%d' % (a[:46], n))
    out_src = CARD / 'build/probe-geom.comp'
    out_src.parent.mkdir(parents=True, exist_ok=True)
    # the probe keeps the parent's includes; point them at the card's pinned headers
    out_src.write_text(text, encoding='utf-8')
    out_spv = CARD / 'build/probe-geom.spv'
    K.compile_glsl(out_src, out_spv, [CARD])
    K.validate_spv(out_spv)

    par = B.spv_stats(CARD / 'build/repro96/rebuild.spv')
    prb = B.spv_stats(out_spv)
    cand = B.spv_stats(CARD / 'candidate.spv')
    rep = dict(parent=str(PARENT_SRC), probe=str(out_src),
               parent_sha=B.sh(Path(CARD / 'build/repro96/rebuild.spv')),
               probe_sha=B.sh(out_spv), candidate_sha=B.sh(CARD / 'candidate.spv'),
               parent_stats=par, probe_stats=prb, candidate_stats=cand,
               probe_equals_parent_opcodes=par['opcodes'] == prb['opcodes'],
               probe_equals_parent_nc=par['no_contraction'] == prb['no_contraction'],
               probe_tracked=prb['tracked'], parent_tracked=par['tracked'])
    opcode_diff = {k: (par['opcodes'].get(k), prb['opcodes'].get(k))
                   for k in set(par['opcodes']) | set(prb['opcodes'])
                   if par['opcodes'].get(k) != prb['opcodes'].get(k)}
    rep['opcode_diff_parent_vs_probe'] = opcode_diff
    K.write_json(CARD / 'build/PROBE-GEOM.json', rep)
    print('probe total instr', prb['total_instructions'], 'parent', par['total_instructions'])
    print('opcode histogram identical:', rep['probe_equals_parent_opcodes'], 'diff:', opcode_diff)
    print('NC probe', prb['no_contraction'], 'parent', par['no_contraction'])
    print('candidate total', cand['total_instructions'])


if __name__ == '__main__':
    main()
