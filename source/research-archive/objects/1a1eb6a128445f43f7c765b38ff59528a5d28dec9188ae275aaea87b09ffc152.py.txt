import json,hashlib,statistics,math,sys
from pathlib import Path
N=Path(__file__).resolve().parent;R=N.parent
def sha(p):return hashlib.sha256(p.read_bytes()).hexdigest()
def ident(p):return dict(path=str(p),sha256=sha(p))
if sys.argv[1]=='prepare':
 task=json.loads((R/'f20-peer8-dualweight/LUNA-ORDINARY.json').read_text());task['output_root']=str(N)
 task['identities']=task['identities'][:1]+[ident(N/'candidate.spv'),ident(R/'f20-peer8-dualweight/candidate.spv')]
 for j in task['jobs']:
  arm='dualweight' if j['candidate']=='single' else j['candidate'];j['candidate']=arm
  label=f'ordinary-fouracc-{j["pair"]}-{arm}';j['id']=label;dest=N/'run'/label;j['runner_output']=str(dest)
  plan=Path(j['plan']['path']) if arm=='parent' else (N if arm=='candidate' else R/'f20-peer8-dualweight')/'plan.txt'
  j['plan']=ident(plan);j['argv'][j['argv'].index('--plan')+1]=str(plan);j['argv'][j['argv'].index('--out')+1]=str(dest)
 task['rules']=['Only after explicit main native gate approval. Nine exact serial ordinary processes, same three-arm balanced order; P03/dualweight/fouracc, warm1/reps30/feedback. Unique lock. No profiler/per-pass timing. Any error/nongold stop. No retries or sample removal.']
 (N/'LUNA-ORDINARY.json').write_text(json.dumps(task,indent=2));print('Prepared; awaiting native approval')
else:
 task=json.loads((N/'LUNA-ORDINARY.json').read_text());means={};frames={}
 for j in task['jobs']:
  d=Path(j['runner_output']);r=json.loads((d/'port_network_receipt.json').read_text());h=sha(d/'frame_out.bin');assert h==j['expected_frame_out']
  assert r['normal_close'] and r['warm']==1 and r['reps']==30 and r['feedback'] and r['gpu_passes']==232 and r['nan']==r['inf']==0
  a=r['gpu_samples_ms'];assert len(a)==30 and all(math.isfinite(x) and x>0 for x in a)
  means[j['id']]=statistics.mean(a);frames[j['id']]=h
 results={}
 for arm in ['parent','dualweight']:
  ds=[means[f'ordinary-fouracc-{i}-{arm}']-means[f'ordinary-fouracc-{i}-candidate'] for i in range(3)]
  m=statistics.mean(ds);hw=4.302652729911275*statistics.stdev(ds)/math.sqrt(3);ci=[m-hw,m+hw]
  results[arm]=dict(savings_ms=ds,mean_savings_ms=m,ci95=ci,verdict='FASTER' if ci[0]>0 else 'SLOWER' if ci[1]<0 else 'INDISTINCT')
 (N/'ORDINARY-REVIEW.json').write_text(json.dumps(dict(comparisons=results,means=means,all_270_samples_retained=True,frame_hashes=frames),indent=2));print(json.dumps(results,indent=2))
