"""Rank passes by measured ALUWR / BARRIER / SENDWR milliseconds to pick the next targets.

Within a single profiling run the ranking is usable (cross-run drift is ~5 points, see
evidence/COUNTER-NOISE.json), so this answers "where is the remaining dependency money".
"""
import json, sys
from pathlib import Path
sys.path.insert(0, str(Path(__file__).parent))
import COMMON as K

p = list((K.C / 'queue' / 'counters-head2').rglob('hardware-counters.json'))[0]
d = K.read_json(p)
names = [m['name'] for m in d['metrics']]
idx = {n: i for i, n in enumerate(names)}
ps = d['frames'][0]['passes']
rows = []
for x in ps:
    g = x['values'][idx['GpuTime']] / 1e6
    rows.append(dict(name=x['name'], index=x['index'], gpu=g,
                     aluwr=g * x['values'][idx['XVE_STALL_ALUWR']] / 100,
                     sendwr=g * x['values'][idx['XVE_STALL_SENDWR']] / 100,
                     barrier=g * x['values'][idx['XVE_STALL_BARRIER']] / 100,
                     pipestall=g * x['values'][idx['XVE_STALL_PIPESTALL']] / 100,
                     active=x['values'][idx['XVE_ACTIVE']]))
tot = sum(r['gpu'] for r in rows)
print('frame %.3f ms over %d passes' % (tot, len(rows)))
for key in ('aluwr', 'sendwr', 'barrier'):
    s = sum(r[key] for r in rows)
    print('  total %-8s %.3f ms (%.1f%%)' % (key, s, 100 * s / tot))
print('\ntop 12 by ALUWR ms:')
print('  %-30s %7s %7s %7s %7s %7s' % ('name', 'gpu_ms', 'aluwr', 'sendwr', 'barrier', 'active%'))
for r in sorted(rows, key=lambda z: -z['aluwr'])[:12]:
    print('  %-30s %7.3f %7.3f %7.3f %7.3f %7.1f' % (r['name'], r['gpu'], r['aluwr'], r['sendwr'],
                                                      r['barrier'], r['active']))
print('\ntop 8 by BARRIER ms:')
for r in sorted(rows, key=lambda z: -z['barrier'])[:8]:
    print('  %-30s %7.3f %7.3f %7.3f %7.3f %7.1f' % (r['name'], r['gpu'], r['aluwr'], r['sendwr'],
                                                      r['barrier'], r['active']))
K.write_json(K.C / 'evidence' / 'STALL-TARGETS.json', dict(frame_ms=tot, rows=sorted(rows, key=lambda z: -z['aluwr'])))
