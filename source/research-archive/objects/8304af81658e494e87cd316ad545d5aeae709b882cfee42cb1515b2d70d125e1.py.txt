#!/usr/bin/env python3
"""Runner-contract check for a PORT_NETWORK_1 plan.

For every BIND: bytes > 0 and offset + bytes <= the declared BUFFER size.  This is the
check the runner itself enforces, and it is the one a mis-parsed lint can miss (the old
lint read the BIND fields in the wrong order, so "0 problems" did not mean "safe").

usage: strict_bind_check.py <plan> [--json out.json]
"""
import json
import sys
from pathlib import Path

sys.path.insert(0, r"@NR_WORKSPACE@/work/dsh-int8-fullnet-impl-20261024/tools")
import planlib as L  # noqa: E402


def main(argv):
    out = None
    if "--json" in argv:
        i = argv.index("--json")
        out = Path(argv[i + 1])
        argv = argv[:i] + argv[i + 2:]
    plan = L.Plan(argv[0])
    bad, nbind, unbound = [], 0, 0
    bound = set()
    for p in plan.passes:
        for b in p["binds"]:
            nbind += 1
            if b["res"] >= plan.nb:
                bad.append({"pass": p["name"], "why": "resource out of range", **b})
                continue
            bound.add(b["res"])
            size = plan.bsize(b["res"])
            if b["bytes"] <= 0:
                bad.append({"pass": p["name"], "why": "bytes<=0", **b})
            elif b["offset"] + b["bytes"] > size:
                bad.append({"pass": p["name"], "why": "offset+bytes>declared", **b,
                            "declared": size})
    file_backed = [r for r in range(plan.nb) if plan.buffers[r][2] != "-"]
    missing = [r for r in file_backed
               if not Path(plan.buffers[r][2]).exists()]
    size_mismatch = [r for r in file_backed if Path(plan.buffers[r][2]).exists()
                     and Path(plan.buffers[r][2]).stat().st_size != plan.bsize(r)]
    unbound = len([r for r in file_backed if r not in bound])
    doc = {"plan": str(argv[0]), "buffers": plan.nb, "passes": plan.npass,
           "binds_checked": nbind, "violations": bad,
           "file_backed": len(file_backed), "missing_files": missing,
           "size_mismatch": size_mismatch,
           "file_backed_never_bound": unbound,
           "ok": not bad and not missing and not size_mismatch}
    print(json.dumps({k: (len(v) if isinstance(v, list) and k != "violations" else v)
                      for k, v in doc.items() if k != "violations"}, indent=1))
    if bad:
        print("first violations:", json.dumps(bad[:5], indent=1))
    if out:
        out.write_text(json.dumps(doc, indent=1) + "\n", encoding="utf-8", newline="\n")
    return 0 if doc["ok"] else 1


if __name__ == "__main__":
    sys.exit(main(sys.argv[1:]))
