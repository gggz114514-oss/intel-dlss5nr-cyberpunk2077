"""Final consistency scan for the n29 card deliverables (read-only)."""
from pathlib import Path

CARD = Path(__file__).resolve().parents[1]
DOCS = ('CHAIN-AUDIT.md', 'ANALYSIS.md', 'NATIVE-PREDICTION.md', 'RESUME.txt')
FLAGS = ('<!--GATHER-TABLE-->', '错位', 'TODO', 'FIXME')
STALE = ('408', '943')


def main():
    for n in DOCS:
        t = (CARD / n).read_text(encoding='utf-8')
        lines = t.splitlines()
        bad = [s for s in FLAGS if s in t]
        stale = [s for s in STALE if s in t]
        print('%-24s %4d lines  flags=%s  stale_tokens=%s' % (n, len(lines), bad, stale))
    audit = (CARD / 'CHAIN-AUDIT.md').read_text(encoding='utf-8')
    print('gather rows embedded:', sum(1 for line in audit.splitlines()
                                       if line.startswith('| ') and '`0x' in line))


if __name__ == '__main__':
    main()
