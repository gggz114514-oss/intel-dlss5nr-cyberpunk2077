"""S09 diagnostic: identify the ViT QKV dump layout and the T dump semantics."""
import os
import numpy as np

DUMPS = r"@NR_WORKSPACE@/work/codex-astra-fp16-int8-audit-20261024/int8-quality-first-20261029/real-int8-24/dump-fp16"
TOK, HEADS, HD = 288, 32, 32


def load(block, tag):
    a = np.fromfile(os.path.join(DUMPS, "dump_vit_block%d_%s.bin" % (block, tag)), dtype=np.float16)
    return a


qkv = load(31, "QKV")
print("QKV elems", qkv.size, "expect", TOK * 3072)
a = qkv.reshape(TOK, 3072)
for name, sl in (("blk0", slice(0, 1024)), ("blk1", slice(1024, 2048)), ("blk2", slice(2048, 3072))):
    x = a[:, sl].reshape(TOK, HEADS, HD).astype(np.float32)
    nrm = np.linalg.norm(x, axis=2)
    print(name, "absmax=%.5f" % np.abs(x).max(),
          "row-norm mean=%.5f cv=%.3f min=%.5f max=%.5f" % (
              nrm.mean(), nrm.std() / max(nrm.mean(), 1e-9), nrm.min(), nrm.max()))
    # per head row norm constancy
    print("      per-head row-norm mean:", np.round(nrm.mean(axis=0)[:8], 4))
    print("      per-head row-norm cv  :", np.round((nrm.std(axis=0) / np.maximum(nrm.mean(axis=0), 1e-9))[:8], 5))

for tag in ("T", "Y"):
    y = load(31, tag)
    print(tag, "elems", y.size, "absmax=%.5f" % np.abs(y.astype(np.float32)).max(),
          "mean=%.6f" % y.astype(np.float32).mean())
